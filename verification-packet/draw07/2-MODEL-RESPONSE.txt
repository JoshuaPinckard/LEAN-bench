from AlgorithmImportBaseClasses import *
from AlgorithmImportIndicators import *

class QuantConnectAlgorithm(QCAlgorithm):
    def Initialize(self):
        self.SetStartDate(2006, 1, 3)
        self.SetEndDate(2015, 12, 31)
        self.SetCash(1000000)
        
        tickers = ["SPY", "AAPL", "IBM", "BAC", "AIG"]
        self.symbols = {}
        for ticker in tickers:
            symbol = self.AddEquity(ticker, Resolution.Daily).Symbol
            self.symbols[ticker] = symbol
        
        self.spy_sma = SimpleMovingAverage(20)
        self.RegisterIndicator(self.symbols["SPY"], self.spy_sma, Resolution.Daily)
        
        self.aapl_rsi = RelativeStrengthIndex(14)
        self.RegisterIndicator(self.symbols["AAPL"], self.aapl_rsi, Resolution.Daily)
        
        self.aapl_red_days = 0
        self.prev_spy_close = None
        self.prev_spy_sma = None
        self.prev_aapl_close = None
        self.trading_started = False
        
    def OnData(self, data):
        required_tickers = ["SPY", "AAPL", "IBM", "BAC", "AIG"]
        for ticker in required_tickers:
            if self.symbols[ticker] not in data:
                return
        
        spy_close = self.Securities[self.symbols["SPY"]].Close
        aapl_close = self.Securities[self.symbols["AAPL"]].Close
        
        if self.prev_aapl_close is not None and aapl_close < self.prev_aapl_close:
            self.aapl_red_days += 1
        else:
            self.aapl_red_days = 0
        
        if not self.spy_sma.IsReady or not self.aapl_rsi.IsReady:
            self.prev_spy_close = spy_close
            self.prev_aapl_close = aapl_close
            self.prev_spy_sma = self.spy_sma.Current.Value
            return
        
        if not self.trading_started:
            self.trading_started = True
            self.prev_spy_close = spy_close
            self.prev_aapl_close = aapl_close
            self.prev_spy_sma = self.spy_sma.Current.Value
            return
        
        spy_sma_value = self.spy_sma.Current.Value
        aapl_rsi_value = self.aapl_rsi.Current.Value
        
        spy_quantity = self.Portfolio[self.symbols["SPY"]].Quantity
        aapl_quantity = self.Portfolio[self.symbols["AAPL"]].Quantity
        
        if spy_quantity > 0:
            if self.prev_spy_close > self.prev_spy_sma and spy_close <= spy_sma_value:
                self.MarketOrder(self.symbols["SPY"], -spy_quantity)
        
        if aapl_quantity > 0:
            if aapl_rsi_value > 60:
                self.MarketOrder(self.symbols["AAPL"], -aapl_quantity)
        
        if spy_quantity == 0:
            if self.prev_spy_close <= self.prev_spy_sma and spy_close > spy_sma_value:
                portfolio_value = self.Portfolio.TotalPortfolioValue
                target_value = portfolio_value * 0.4
                shares_to_buy = int(target_value / spy_close)
                if shares_to_buy > 0:
                    self.MarketOrder(self.symbols["SPY"], shares_to_buy)
        
        if aapl_quantity == 0:
            if self.aapl_red_days >= 3:
                available_cash = self.Portfolio.Cash
                if available_cash >= 20000:
                    shares_to_buy = int(20000 / aapl_close)
                    if shares_to_buy > 0:
                        self.MarketOrder(self.symbols["AAPL"], shares_to_buy)
        
        self.prev_spy_close = spy_close
        self.prev_aapl_close = aapl_close
        self.prev_spy_sma = spy_sma_value
