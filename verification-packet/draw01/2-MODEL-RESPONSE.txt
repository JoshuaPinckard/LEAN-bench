from AlgorithmImports import *


class TwoSlotStrategyAlgorithm(QCAlgorithm):
    def Initialize(self):
        self.SetStartDate(2006, 1, 3)
        self.SetEndDate(2015, 12, 31)
        self.SetCash(1000000)

        self.spy = self.AddEquity("SPY", Resolution.Daily).Symbol
        self.aapl = self.AddEquity("AAPL", Resolution.Daily).Symbol
        self.ibm = self.AddEquity("IBM", Resolution.Daily).Symbol
        self.bac = self.AddEquity("BAC", Resolution.Daily).Symbol
        self.aig = self.AddEquity("AIG", Resolution.Daily).Symbol

        self.subscribed_symbols = [
            self.spy,
            self.aapl,
            self.ibm,
            self.bac,
            self.aig,
        ]

        self.spy_sma = self.SMA(self.spy, 100, Resolution.Daily)
        self.aapl_rsi = self.RSI(
            self.aapl,
            14,
            MovingAverageType.Wilders,
            Resolution.Daily
        )

        self.previous_spy_close = None
        self.previous_spy_sma = None

        self.previous_aapl_close = None
        self.aapl_red_streak = 0

        self.strategy_a_lot = 0
        self.strategy_b_lot = 0

    def OnData(self, data):
        spy_bar = data.Bars[self.spy] if data.Bars.ContainsKey(self.spy) else None
        aapl_bar = data.Bars[self.aapl] if data.Bars.ContainsKey(self.aapl) else None

        spy_buy_signal = False
        spy_sell_signal = False

        if spy_bar is not None:
            spy_close = float(spy_bar.Close)

            if self.spy_sma.IsReady:
                current_spy_sma = float(self.spy_sma.Current.Value)

                if (
                    self.previous_spy_close is not None
                    and self.previous_spy_sma is not None
                ):
                    spy_buy_signal = (
                        self.previous_spy_close <= self.previous_spy_sma
                        and spy_close > current_spy_sma
                    )
                    spy_sell_signal = (
                        self.previous_spy_close >= self.previous_spy_sma
                        and spy_close < current_spy_sma
                    )

                self.previous_spy_sma = current_spy_sma

            self.previous_spy_close = spy_close

        aapl_buy_signal = False
        aapl_sell_signal = False

        if aapl_bar is not None:
            aapl_close = float(aapl_bar.Close)

            if self.previous_aapl_close is None:
                self.aapl_red_streak = 0
            elif aapl_close < self.previous_aapl_close:
                self.aapl_red_streak += 1
            else:
                self.aapl_red_streak = 0

            self.previous_aapl_close = aapl_close

            if self.aapl_rsi.IsReady:
                aapl_sell_signal = float(self.aapl_rsi.Current.Value) > 60

            aapl_buy_signal = self.aapl_red_streak >= 3

        if not all(
            data.Bars.ContainsKey(symbol)
            for symbol in self.subscribed_symbols
        ):
            return

        spy_close = float(data.Bars[self.spy].Close)
        aapl_close = float(data.Bars[self.aapl].Close)

        # Evaluate sells before buys.
        if self.strategy_a_lot > 0 and spy_sell_signal:
            self.MarketOrder(self.spy, -self.strategy_a_lot)
            self.strategy_a_lot = 0

        if self.strategy_b_lot > 0 and aapl_sell_signal:
            self.MarketOrder(self.aapl, -self.strategy_b_lot)
            self.strategy_b_lot = 0

        if self.strategy_a_lot == 0 and spy_buy_signal and self.spy_sma.IsReady:
            target_value = 0.40 * float(self.Portfolio.TotalPortfolioValue)
            quantity = int(target_value / spy_close)

            if quantity > 0:
                self.MarketOrder(self.spy, quantity)
                self.strategy_a_lot = quantity

        if (
            self.strategy_b_lot == 0
            and aapl_buy_signal
            and float(self.Portfolio.Cash) >= 20000
        ):
            quantity = int(20000 / aapl_close)

            if quantity > 0:
                self.MarketOrder(self.aapl, quantity)
                self.strategy_b_lot = quantity
