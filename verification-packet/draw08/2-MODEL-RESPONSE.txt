from AlgorithmImports import *


class ExactStrategySystemAlgorithm(QCAlgorithm):

    def Initialize(self):
        self.SetStartDate(2006, 1, 3)
        self.SetEndDate(2015, 12, 31)
        self.SetCash(1_000_000)

        self.spy = self.AddEquity("SPY", Resolution.Daily).Symbol
        self.aapl = self.AddEquity("AAPL", Resolution.Daily).Symbol
        self.ibm = self.AddEquity("IBM", Resolution.Daily).Symbol
        self.bac = self.AddEquity("BAC", Resolution.Daily).Symbol
        self.aig = self.AddEquity("AIG", Resolution.Daily).Symbol

        self.symbols = [
            self.spy,
            self.aapl,
            self.ibm,
            self.bac,
            self.aig
        ]

        self.spy_sma = SimpleMovingAverage(100)
        self.aapl_rsi = RelativeStrengthIndex(
            14,
            MovingAverageType.Wilders
        )

        self.previous_spy_close = None
        self.previous_spy_sma = None

        self.previous_aapl_close = None
        self.aapl_red_streak = 0
        self.aapl_observation_count = 0

        self.spy_lot_shares = 0
        self.aapl_lot_shares = 0

        # One prior fully formed SMA value is also required for crossing logic.
        self.SetWarmUp(101, Resolution.Daily)

    def OnData(self, data):
        previous_spy_close = self.previous_spy_close
        previous_spy_sma = self.previous_spy_sma

        spy_bar = None
        current_spy_sma = None

        if data.Bars.ContainsKey(self.spy):
            spy_bar = data.Bars[self.spy]
            self.spy_sma.Update(spy_bar.EndTime, spy_bar.Close)

            if self.spy_sma.IsReady:
                current_spy_sma = self.spy_sma.Current.Value
                self.previous_spy_close = spy_bar.Close
                self.previous_spy_sma = current_spy_sma

        aapl_bar = None

        if data.Bars.ContainsKey(self.aapl):
            aapl_bar = data.Bars[self.aapl]
            self.aapl_rsi.Update(aapl_bar.EndTime, aapl_bar.Close)

            if self.previous_aapl_close is not None:
                if aapl_bar.Close < self.previous_aapl_close:
                    self.aapl_red_streak += 1
                else:
                    self.aapl_red_streak = 0

            self.previous_aapl_close = aapl_bar.Close
            self.aapl_observation_count += 1

        if self.IsWarmingUp:
            return

        if any(not data.Bars.ContainsKey(symbol) for symbol in self.symbols):
            return

        if (
            previous_spy_close is None
            or previous_spy_sma is None
            or current_spy_sma is None
            or not self.aapl_rsi.IsReady
            or self.aapl_observation_count < 4
        ):
            return

        spy_crossed_below = (
            previous_spy_close >= previous_spy_sma
            and spy_bar.Close < current_spy_sma
        )
        spy_crossed_above = (
            previous_spy_close <= previous_spy_sma
            and spy_bar.Close > current_spy_sma
        )

        # Evaluate all sells before any buys.
        if self.spy_lot_shares > 0 and spy_crossed_below:
            self.MarketOrder(
                self.spy,
                -self.spy_lot_shares,
                False,
                "Strategy A: close crossed below 100-day SMA"
            )
            self.spy_lot_shares = 0

        if (
            self.aapl_lot_shares > 0
            and self.aapl_rsi.Current.Value > 60
        ):
            self.MarketOrder(
                self.aapl,
                -self.aapl_lot_shares,
                False,
                "Strategy B: 14-day RSI above 60"
            )
            self.aapl_lot_shares = 0

        if self.spy_lot_shares == 0 and spy_crossed_above:
            quantity = int(self.CalculateOrderQuantity(self.spy, 0.40))
            if quantity > 0:
                self.MarketOrder(
                    self.spy,
                    quantity,
                    False,
                    "Strategy A: close crossed above 100-day SMA"
                )
                self.spy_lot_shares = quantity

        if self.aapl_lot_shares == 0 and self.aapl_red_streak >= 3:
            if self.Portfolio.Cash >= 20_000:
                quantity = int(20_000 // aapl_bar.Close)
                if quantity > 0:
                    self.MarketOrder(
                        self.aapl,
                        quantity,
                        False,
                        "Strategy B: red for at least 3 consecutive days"
                    )
                    self.aapl_lot_shares = quantity
