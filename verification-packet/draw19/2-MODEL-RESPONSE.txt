from AlgorithmImports import *


class ExactStrategySystem(QCAlgorithm):

    def Initialize(self):
        self.SetStartDate(2006, 1, 3)
        self.SetEndDate(2015, 12, 31)
        self.SetCash(1_000_000)

        self.symbols = [
            self.AddEquity("SPY", Resolution.Daily).Symbol,
            self.AddEquity("AAPL", Resolution.Daily).Symbol,
            self.AddEquity("IBM", Resolution.Daily).Symbol,
            self.AddEquity("BAC", Resolution.Daily).Symbol,
            self.AddEquity("AIG", Resolution.Daily).Symbol
        ]

        self.spy = self.Symbol("SPY")
        self.aapl = self.Symbol("AAPL")

        self.spy_sma = self.SMA(self.spy, 100, Resolution.Daily)
        self.aapl_rsi = self.RSI(self.aapl, 14, MovingAverageType.Wilders, Resolution.Daily)

        self.previous_closes = {}
        self.previous_spy_sma = None
        self.previous_spy_close = None
        self.aapl_red_streak = 0

        self.strategy_a_shares = 0
        self.strategy_b_shares = 0

    def OnData(self, data: Slice):
        # Keep lot state synchronized with filled holdings.
        self.strategy_a_shares = int(self.Portfolio[self.spy].Quantity)
        self.strategy_b_shares = int(self.Portfolio[self.aapl].Quantity)

        # If any subscribed ticker is missing, skip all rule evaluation.
        if any(symbol not in data.Bars for symbol in self.symbols):
            return

        spy_close = data.Bars[self.spy].Close
        aapl_close = data.Bars[self.aapl].Close

        prior_aapl_close = self.previous_closes.get(self.aapl)
        if prior_aapl_close is None:
            self.aapl_red_streak = 0
        elif aapl_close < prior_aapl_close:
            self.aapl_red_streak += 1
        else:
            self.aapl_red_streak = 0

        all_rules_ready = (
            self.spy_sma.IsReady
            and self.aapl_rsi.IsReady
            and self.previous_spy_sma is not None
            and self.previous_spy_close is not None
        )

        if all_rules_ready:
            spy_sma_value = self.spy_sma.Current.Value

            spy_crosses_above = (
                self.previous_spy_close <= self.previous_spy_sma
                and spy_close > spy_sma_value
            )
            spy_crosses_below = (
                self.previous_spy_close >= self.previous_spy_sma
                and spy_close < spy_sma_value
            )

            # Slot 1: Strategy A
            if self.strategy_a_shares > 0:
                if spy_crosses_below:
                    self.MarketOrder(self.spy, -self.strategy_a_shares)
            else:
                if spy_crosses_above:
                    target_value = self.Portfolio.TotalPortfolioValue * 0.40
                    quantity = int(target_value / spy_close)
                    if quantity > 0:
                        self.MarketOrder(self.spy, quantity)

            # Slot 2: Strategy B
            if self.strategy_b_shares > 0:
                if self.aapl_rsi.Current.Value > 60:
                    self.MarketOrder(self.aapl, -self.strategy_b_shares)
            else:
                if self.aapl_red_streak >= 3 and self.Portfolio.Cash >= 20_000:
                    quantity = int(20_000 / aapl_close)
                    if quantity > 0:
                        self.MarketOrder(self.aapl, quantity)

        self.previous_spy_close = spy_close
        self.previous_spy_sma = self.spy_sma.Current.Value if self.spy_sma.IsReady else None

        for symbol in self.symbols:
            self.previous_closes[symbol] = data.Bars[symbol].Close
