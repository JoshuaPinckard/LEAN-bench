from AlgorithmImports import *


class ExactStrategyAlgorithm(QCAlgorithm):
    def Initialize(self):
        self.SetStartDate(2006, 1, 3)
        self.SetEndDate(2015, 12, 31)
        self.SetCash(1000000)

        self.spy = self.AddEquity("SPY", Resolution.Daily).Symbol
        self.aapl = self.AddEquity("AAPL", Resolution.Daily).Symbol
        self.ibm = self.AddEquity("IBM", Resolution.Daily).Symbol
        self.bac = self.AddEquity("BAC", Resolution.Daily).Symbol
        self.aig = self.AddEquity("AIG", Resolution.Daily).Symbol

        self.subscribed_symbols = [
            self.spy,
            self.aapl,
            self.ibm,
            self.bac,
            self.aig
        ]

        self.spy_sma = self.SMA(self.spy, 100, Resolution.Daily)
        self.aapl_rsi = self.RSI(
            self.aapl,
            14,
            MovingAverageType.Wilders,
            Resolution.Daily
        )

        self.previous_spy_sma = None
        self.previous_aapl_close = None
        self.aapl_red_streak = 0
        self.aapl_close_count = 0

        self.strategy_a_lot = 0
        self.strategy_b_lot = 0

        self.SetWarmUp(100, Resolution.Daily)

    def OnData(self, data):
        spy_bar = data.Bars[self.spy] if self.spy in data.Bars else None
        aapl_bar = data.Bars[self.aapl] if self.aapl in data.Bars else None

        previous_spy_sma = self.previous_spy_sma
        current_spy_sma = None

        if spy_bar is not None and self.spy_sma.IsReady:
            current_spy_sma = float(self.spy_sma.Current.Value)
            self.previous_spy_sma = current_spy_sma

        if aapl_bar is not None:
            current_aapl_close = float(aapl_bar.Close)

            if self.previous_aapl_close is None:
                self.aapl_red_streak = 0
            elif current_aapl_close < self.previous_aapl_close:
                self.aapl_red_streak += 1
            else:
                self.aapl_red_streak = 0

            self.previous_aapl_close = current_aapl_close
            self.aapl_close_count += 1

        if self.IsWarmingUp:
            return

        if not all(symbol in data.Bars for symbol in self.subscribed_symbols):
            return

        spy_ready = (
            self.spy_sma.IsReady
            and previous_spy_sma is not None
            and current_spy_sma is not None
        )

        aapl_ready = (
            self.aapl_rsi.IsReady
            and self.aapl_close_count >= 4
        )

        if spy_ready and self.strategy_a_lot > 0:
            crosses_below = (
                previous_spy_sma >= previous_spy_sma
                and float(spy_bar.Close) < current_spy_sma
            )

            if crosses_below:
                self.MarketOrder(self.spy, -self.strategy_a_lot)
                self.strategy_a_lot = 0

        if aapl_ready and self.strategy_b_lot > 0:
            if float(self.aapl_rsi.Current.Value) > 60:
                self.MarketOrder(self.aapl, -self.strategy_b_lot)
                self.strategy_b_lot = 0

        if spy_ready and self.strategy_a_lot == 0:
            spy_close = float(spy_bar.Close)
            crosses_above = (
                previous_spy_sma <= previous_spy_sma
                and spy_close > current_spy_sma
            )

            if crosses_above:
                quantity = int(
                    (0.40 * float(self.Portfolio.TotalPortfolioValue))
                    / spy_close
                )

                if quantity > 0:
                    self.MarketOrder(self.spy, quantity)
                    self.strategy_a_lot = quantity

        if aapl_ready and self.strategy_b_lot == 0:
            aapl_close = float(aapl_bar.Close)

            if self.aapl_red_streak >= 3:
                if float(self.Portfolio.Cash) >= 20000:
                    quantity = int(20000 / aapl_close)

                    if quantity > 0:
                        self.MarketOrder(self.aapl, quantity)
                        self.strategy_b_lot = quantity
