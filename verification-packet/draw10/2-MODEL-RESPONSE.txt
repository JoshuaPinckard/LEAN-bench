from AlgorithmImports import *


class SpecifiedStrategySystem(QCAlgorithm):

    def Initialize(self):
        self.SetStartDate(2006, 1, 3)
        self.SetEndDate(2015, 12, 31)
        self.SetCash(1_000_000)

        self.symbols = {}
        for ticker in ["SPY", "AAPL", "IBM", "BAC", "AIG"]:
            self.symbols[ticker] = self.AddEquity(ticker, Resolution.Daily).Symbol

        self.spy = self.symbols["SPY"]
        self.aapl = self.symbols["AAPL"]

        self.spy_sma = self.SMA(self.spy, 100, Resolution.Daily)
        self.aapl_rsi = self.RSI(self.aapl, 14, MovingAverageType.Wilders, Resolution.Daily)

        self.previous_spy_close = None
        self.previous_spy_sma = None
        self.previous_aapl_close = None
        self.aapl_red_streak = 0

        self.strategy_a_shares = 0
        self.strategy_b_shares = 0

    def OnData(self, slice):
        spy_bar = slice.Bars.get(self.spy)
        aapl_bar = slice.Bars.get(self.aapl)

        # Market-state quantities continue updating whenever their own data exists.
        if aapl_bar is not None:
            if self.previous_aapl_close is not None and aapl_bar.Close < self.previous_aapl_close:
                self.aapl_red_streak += 1
            else:
                self.aapl_red_streak = 0
            self.previous_aapl_close = aapl_bar.Close

        current_spy_sma = self.spy_sma.Current.Value if self.spy_sma.IsReady else None

        # All subscribed tickers must have a bar before any rule evaluation.
        if not all(symbol in slice.Bars for symbol in self.symbols.values()):
            if spy_bar is not None and current_spy_sma is not None:
                self.previous_spy_close = spy_bar.Close
                self.previous_spy_sma = current_spy_sma
            return

        # No trading until every indicator or rolling window used by a rule is formed.
        if not self.spy_sma.IsReady or not self.aapl_rsi.IsReady:
            if current_spy_sma is not None:
                self.previous_spy_close = spy_bar.Close
                self.previous_spy_sma = current_spy_sma
            return

        spy_crosses_above = (
            self.previous_spy_close is not None
            and self.previous_spy_sma is not None
            and self.previous_spy_close <= self.previous_spy_sma
            and spy_bar.Close > current_spy_sma
        )

        spy_crosses_below = (
            self.previous_spy_close is not None
            and self.previous_spy_sma is not None
            and self.previous_spy_close >= self.previous_spy_sma
            and spy_bar.Close < current_spy_sma
        )

        # Slot 1: Strategy A
        if self.strategy_a_shares > 0 and spy_crosses_below:
            self.Liquidate(self.spy)
            self.strategy_a_shares = 0
        elif self.strategy_a_shares == 0 and spy_crosses_above:
            self.SetHoldings(self.spy, 0.40)
            self.strategy_a_shares = self.Portfolio[self.spy].Quantity

        # Slot 2: Strategy B
        if self.strategy_b_shares > 0 and self.aapl_rsi.Current.Value > 60:
            self.Liquidate(self.aapl)
            self.strategy_b_shares = 0
        elif self.strategy_b_shares == 0 and self.aapl_red_streak >= 3:
            if self.Portfolio.Cash >= 20_000:
                quantity = int(20_000 // aapl_bar.Close)
                if quantity > 0:
                    self.MarketOrder(self.aapl, quantity)
                    self.strategy_b_shares = quantity

        self.previous_spy_close = spy_bar.Close
        self.previous_spy_sma = current_spy_sma
