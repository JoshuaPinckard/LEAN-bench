import backtrader as bt

class TradingStrategy(bt.Strategy):
    params = (
        ('atr_period', 14),
        ('ema_period', 50),
        ('pullback_atr', 1.5),
        ('tp_atr', 2.5),
        ('max_entries', 4),
        ('risk_pct', 0.01),
    )

    def __init__(self):
        self.atr = bt.indicators.ATR(self.data, period=self.params.atr_period)
        self.ema = bt.indicators.EMA(self.data.close, period=self.params.ema_period)
        self.order = None
        self.entry_count = 0
        self.peak = 0.0
        self.entry_prices = []
        self.atr_at_entries = []
        self.first_entry_price = 0.0
        self.atr_at_first_entry = 0.0
        self.max_entries_reached = False

    def next(self):
        if self.order:
            return

        current_close = self.data.close[0]
        current_high = self.data.high[0]

        if current_high > self.peak:
            self.peak = current_high

        if self.position:
            if current_close < self.ema[0]:
                self._close_all()
                return

            if self.first_entry_price > 0:
                tp = self.first_entry_price + self.params.tp_atr * self.atr_at_first_entry
                if current_close >= tp:
                    self._close_all()
                    return

            if self.entry_prices:
                sl = self.entry_prices[-1] - self.params.pullback_atr * self.atr_at_entries[-1]
                if current_close < sl:
                    self._close_all()
                    return

            if self.entry_count < self.params.max_entries and self.entry_prices:
                pullback_level = self.peak - self.params.pullback_atr * self.atr[0]
                if (current_close <= pullback_level
                        and current_close > self.ema[0]
                        and self.peak > self.entry_prices[-1]):
                    self._enter_position(current_close)

        elif not self.max_entries_reached:
            if current_close > self.ema[0] and self.peak > 0:
                pullback_level = self.peak - self.params.pullback_atr * self.atr[0]
                if current_close <= pullback_level:
                    self._enter_position(current_close)

    def _enter_position(self, price):
        portfolio_value = self.broker.getvalue()
        size = int((portfolio_value * self.params.risk_pct) / price)
        if size < 1:
            return
        if self.entry_count == 0:
            self.first_entry_price = price
            self.atr_at_first_entry = self.atr[0]
        self.entry_prices.append(price)
        self.atr_at_entries.append(self.atr[0])
        self.entry_count += 1
        self.peak = price
        if self.entry_count >= self.params.max_entries:
            self.max_entries_reached = True
        self.order = self.buy(size=size)

    def _close_all(self):
        self.order = self.close()

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            if order.status == order.Completed and order.issell():
                self.entry_count = 0
                self.entry_prices = []
                self.atr_at_entries = []
                self.first_entry_price = 0.0
                self.atr_at_first_entry = 0.0
                self.max_entries_reached = False
                self.peak = 0.0
            elif order.status in [order.Canceled, order.Margin, order.Rejected] and order.isbuy():
                if self.entry_prices:
                    self.entry_prices.pop()
                    self.atr_at_entries.pop()
                    self.entry_count = max(0, self.entry_count - 1)
                    if self.entry_count == 0:
                        self.first_entry_price = 0.0
                        self.atr_at_first_entry = 0.0
                    self.max_entries_reached = self.entry_count >= self.params.max_entries
            self.order = None
