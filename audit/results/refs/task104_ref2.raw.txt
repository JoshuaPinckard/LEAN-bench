import math
import backtrader as bt


class ConsolidationBreakout(bt.Strategy):
    params = (
        ('sma_period', 120),
        ('vol_sma_period', 20),
        ('consolidation_min', 180.0),
        ('consolidation_max', 193.0),
        ('tp_level', 200.0),
        ('vol_mult', 1.5),
        ('trend_lookback', 5),
        ('risk_pct', 0.01),
    )

    def __init__(self):
        self.sma120 = bt.indicators.SimpleMovingAverage(
            self.data.close, period=self.p.sma_period
        )
        self.vol_sma = bt.indicators.SimpleMovingAverage(
            self.data.volume, period=self.p.vol_sma_period
        )
        self.order = None

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        if self.order:
            return

        if len(self) < self.p.sma_period + self.p.trend_lookback:
            return

        if not self.position:
            # Condition 1: Close above long-term SMA(120)
            cond1 = self.data.close[0] > self.sma120[0]

            # Condition 2: Prior candle was in consolidation range [$180.00, $193.00]
            cond2 = self.p.consolidation_min <= self.data.close[-1] <= self.p.consolidation_max

            # Condition 3: Selling volume on prior consolidation bar does not exceed 1.5x 20-day SMA volume
            if self.data.close[-1] < self.data.open[-1]:
                cond3 = self.data.volume[-1] <= self.p.vol_mult * self.vol_sma[-1]
            else:
                cond3 = True

            # Condition 4: Current candle close breaks out above upper consolidation boundary ($193.00)
            cond4 = self.data.close[0] > self.p.consolidation_max

            # Additional Rule: SMA(120) trending upward over the last 5 days
            cond5 = self.sma120[0] > self.sma120[-self.p.trend_lookback]

            if cond1 and cond2 and cond3 and cond4 and cond5:
                capital = self.broker.getvalue()
                risk_amount = capital * self.p.risk_pct
                risk_per_share = self.data.close[0] - self.p.consolidation_min

                if risk_per_share > 0:
                    size = int(risk_amount / risk_per_share)
                    max_shares = int(self.broker.getcash() / self.data.close[0])
                    size = min(size, max_shares)
                    if size > 0:
                        self.order = self.buy(size=size)
                else:
                    self.order = self.buy()
        else:
            # Exit Conditions
            # 1. Take-profit at target level $200.00
            # 2. Stop-loss if price drops below SMA(120)
            # 3. Alternative exit on close below consolidation lower boundary $180.00
            if self.data.high[0] >= self.p.tp_level or self.data.close[0] >= self.p.tp_level:
                self.order = self.close()
            elif self.data.close[0] < self.sma120[0]:
                self.order = self.close()
            elif self.data.close[0] < self.p.consolidation_min:
                self.order = self.close()
