import backtrader as bt
import math


class BreakoutStrategy(bt.Strategy):
    def __init__(self):
        self.bb = bt.indicators.BollingerBands(
            self.data.close, period=20, devfactor=2
        )
        self.sma = bt.indicators.SimpleMovingAverage(self.data.close, period=20)

        self.order = None
        self.entry_price = None

        self.addminperiod(23)

    def notify_order(self, order):
        if order.status == order.Completed:
            if order.isbuy():
                self.entry_price = order.executed.price
            elif order.issell():
                self.entry_price = None
            self.order = None
        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        if self.order:
            return

        upper_band = self.bb.lines.top[0]

        if self.position:
            if self.data.close[0] < upper_band:
                self.order = self.close()
            elif self.entry_price is not None and self.data.close[0] >= self.entry_price * 1.02:
                self.order = self.close()
            return

        breakout = (
            self.data.close[0] > upper_band
            and self.data.close[-1] <= self.bb.lines.top[-1]
            and self.data.close[-2] <= self.bb.lines.top[-2]
            and self.data.close[-3] <= self.bb.lines.top[-3]
            and self.sma[0] > self.sma[-1]
        )

        if breakout:
            risk_per_unit = self.data.close[0] - upper_band
            risk_capital = self.broker.getvalue() * 0.01
            size = math.floor(risk_capital / risk_per_unit)

            if size > 0:
                self.order = self.buy(size=size)
