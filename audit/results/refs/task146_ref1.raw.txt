import backtrader as bt


class CupAndHandleStrategy(bt.Strategy):
    params = (
        ("cup_bars", 250),
        ("handle_bars", 20),
        ("volume_period", 50),
        ("risk_fraction", 0.01),
    )

    def __init__(self):
        self.order = None
        self.handle_stop = None
        self.volume_average = bt.indicators.SimpleMovingAverage(
            self.data.volume, period=self.p.volume_period
        )

    def notify_order(self, order):
        if order.status in (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            if order.status in (order.Canceled, order.Margin, order.Rejected):
                self.handle_stop = None
            self.order = None

    def next(self):
        if self.order:
            return

        if self.position:
            if self.handle_stop is not None and self.data.low[0] < self.handle_stop:
                self.order = self.close()
            return

        required_history = self.p.cup_bars + self.p.handle_bars
        if len(self.data) <= required_history:
            return

        highs = [
            self.data.high[-offset]
            for offset in range(required_history, 0, -1)
        ]
        lows = [
            self.data.low[-offset]
            for offset in range(required_history, 0, -1)
        ]

        cup_highs = highs[:self.p.cup_bars]
        cup_lows = lows[:self.p.cup_bars]
        handle_highs = highs[self.p.cup_bars:]
        handle_lows = lows[self.p.cup_bars:]

        neckline = max(cup_highs)
        cup_low = min(cup_lows)
        cup_bottom_index = cup_lows.index(cup_low)
        handle_low = min(handle_lows)

        left_rim = max(cup_highs[:20])
        right_rim = max(cup_highs[-20:])

        cup_is_u_shaped = (
            20 <= cup_bottom_index < self.p.cup_bars - 20
            and left_rim >= neckline * 0.95
            and right_rim >= neckline * 0.95
        )
        handle_is_valid = (
            handle_low > cup_low
            and handle_low >= neckline * 0.90
            and max(handle_highs) <= neckline
        )

        breakout = (
            self.data.close[0] > neckline
            and self.data.volume[0] > self.volume_average[0]
        )

        if cup_is_u_shaped and handle_is_valid and breakout:
            risk_per_share = self.data.close[0] - handle_low
            if risk_per_share <= 0:
                return

            risk_capital = self.broker.getvalue() * self.p.risk_fraction
            size_by_risk = int(risk_capital / risk_per_share)
            size_by_cash = int(
                self.broker.getcash() / (self.data.close[0] * (1.0 + 0.001))
            )
            size = min(size_by_risk, size_by_cash)

            if size > 0:
                self.handle_stop = handle_low
                self.order = self.buy(size=size)
