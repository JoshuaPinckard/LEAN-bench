# complete strategy class
import backtrader as bt


class CupAndHandleStrategy(bt.Strategy):
    params = (
        ('cup_period', 220),           # bars forming the cup
        ('handle_period', 30),         # bars forming the handle (cup+handle = 250 ~ 1 year)
        ('vol_period', 50),            # average-volume lookback for breakout confirmation
        ('rim_frac', 0.20),            # fraction of cup edges used to locate each rim
        ('rim_tol', 0.10),             # max relative difference between left/right cup rims
        ('min_cup_depth', 0.10),       # minimum cup depth as a fraction of resistance
        ('center_lo', 0.25),           # cup bottom must lie within the central band ...
        ('center_hi', 0.75),           # ... of the cup (U-shape)
        ('handle_max_retrace', 0.50),  # handle depth <= this fraction of the cup depth
        ('risk_pct', 0.01),            # risk per trade as fraction of total capital (1-2% -> 1%)
    )

    def __init__(self):
        # Volume indicator: average volume over the last `vol_period` candles.
        self.vol_avg = bt.indicators.SimpleMovingAverage(
            self.data.volume, period=self.p.vol_period)
        self.order = None
        self.stop_price = None
        self.min_bars = self.p.cup_period + self.p.handle_period + 1

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return
        if order.status in (order.Completed, order.Canceled,
                            order.Margin, order.Rejected):
            self.order = None

    def next(self):
        if self.order:
            return

        # Manage an open position: exit on a drop below the handle low (stop-loss).
        if self.position:
            if self.data.close[0] < self.stop_price:
                self.order = self.close()
            return

        # Need a full cup+handle (>= ~1 year) of history plus the prior bar.
        if len(self.data) < self.min_bars:
            return

        H = self.p.handle_period
        C = self.p.cup_period

        # Handle window: the H bars immediately preceding the current (breakout) bar.
        handle_highs = [self.data.high[-i] for i in range(1, H + 1)]
        handle_lows = [self.data.low[-i] for i in range(1, H + 1)]

        # Cup window: the C bars preceding the handle, in chronological order.
        cup_highs = [self.data.high[-i] for i in range(H + C, H, -1)]
        cup_lows = [self.data.low[-i] for i in range(H + C, H, -1)]

        # --- Cup geometry (U-shaped formation) ---------------------------
        cup_bottom = min(cup_lows)
        edge = max(1, int(self.p.rim_frac * C))
        left_rim = max(cup_highs[:edge])
        right_rim = max(cup_highs[-edge:])
        resistance = max(left_rim, right_rim)     # upper boundary / neckline
        cup_depth = resistance - cup_bottom
        if cup_depth <= 0:
            return

        # Cup must be a genuine U (meaningful depth).
        if cup_depth < self.p.min_cup_depth * resistance:
            return

        # Bottom of the U must sit in the central band of the cup.
        bottom_idx = cup_lows.index(cup_bottom)
        if not (self.p.center_lo * C <= bottom_idx <= self.p.center_hi * C):
            return

        # Left and right rims at similar levels -> symmetric cup.
        if abs(left_rim - right_rim) > self.p.rim_tol * resistance:
            return

        # --- Handle geometry (consolidation / small decline) -------------
        handle_high = max(handle_highs)
        handle_low = min(handle_lows)

        # Handle stays below the resistance (it has not broken out yet).
        if handle_high >= resistance:
            return
        # Handle is only a small decline relative to the cup, and above the cup bottom.
        if handle_low <= cup_bottom:
            return
        handle_depth = resistance - handle_low
        if handle_depth > self.p.handle_max_retrace * cup_depth:
            return

        # --- Breakout with volume confirmation ---------------------------
        # Break upward above the resistance; the breakout candle closes above it.
        broke_out = (self.data.close[0] > resistance and
                    self.data.close[-1] <= resistance)
        volume_ok = self.data.volume[0] > self.vol_avg[0]
        if not (broke_out and volume_ok):
            return

        # --- Risk-based position sizing (stop = handle low) --------------
        entry_price = self.data.close[0]
        risk_per_share = entry_price - handle_low
        if risk_per_share <= 0:
            return
        risk_amount = self.p.risk_pct * self.broker.getvalue()
        size = int(risk_amount / risk_per_share)

        # Do not attempt to buy more than the available cash allows.
        affordable = int(self.broker.getcash() / (entry_price * 1.001))
        size = min(size, affordable)
        if size <= 0:
            return

        self.stop_price = handle_low
        self.order = self.buy(size=size)
