import backtrader as bt


class FibTrendlinePullbackStrategy(bt.Strategy):
    params = (
        ('fractal_n', 2),
        ('sma_period', 50),
        ('significant_move_pct', 0.03),
        ('fib_ratio', 0.618),
        ('risk_pct', 0.01),
        ('reward_risk_ratio', 2.0),
        ('stop_buffer_pct', 0.001),
    )

    def __init__(self):
        self.sma = bt.indicators.SMA(self.data.close, period=self.p.sma_period)
        self.order = None
        self.stop_order = None
        self.limit_order = None
        self.swing_lows = []
        self.swing_highs = []
        self.awaiting_fib = False
        self.fib_level = None
        self._stop_basis = None
        self._pending_stop_basis = None

    def _check_fractals(self):
        n = self.p.fractal_n
        if len(self) < 2 * n + 1:
            return
        window_low = [self.data.low[i] for i in range(-2 * n, 1)]
        window_high = [self.data.high[i] for i in range(-2 * n, 1)]
        candidate_low = self.data.low[-n]
        candidate_high = self.data.high[-n]
        bar_num = len(self) - 1 - n
        if candidate_low == min(window_low):
            self.swing_lows.append((bar_num, candidate_low))
        if candidate_high == max(window_high):
            self.swing_highs.append((bar_num, candidate_high))

    def _trendline_value(self, bar_num):
        if len(self.swing_lows) < 2:
            return None
        i1, p1 = self.swing_lows[-2]
        i2, p2 = self.swing_lows[-1]
        if p2 <= p1:
            return None
        slope = (p2 - p1) / (i2 - i1)
        return p1 + slope * (bar_num - i1)

    def _last_significant_move(self):
        for low_idx, low_price in reversed(self.swing_lows):
            for high_idx, high_price in reversed(self.swing_highs):
                if high_idx > low_idx and low_price > 0:
                    if (high_price - low_price) / low_price >= self.p.significant_move_pct:
                        return low_price, high_price
        return None

    def _cancel_exit_orders(self):
        if self.stop_order is not None and self.stop_order.alive():
            self.cancel(self.stop_order)
        if self.limit_order is not None and self.limit_order.alive():
            self.cancel(self.limit_order)
        self.stop_order = None
        self.limit_order = None

    def next(self):
        self._check_fractals()

        cur_bar = len(self) - 1
        trend_now = self._trendline_value(cur_bar)
        trend_prev = self._trendline_value(cur_bar - 1)

        if self.position:
            if trend_now is not None and trend_prev is not None:
                if self.data.close[-1] > trend_prev and self.data.close[0] < trend_now:
                    self._cancel_exit_orders()
                    self.close()
                    self.awaiting_fib = False
                    self.fib_level = None
            return

        if self.order is not None:
            return

        if trend_now is not None and trend_prev is not None:
            if self.data.close[-1] < trend_prev and self.data.close[0] > trend_now:
                move = self._last_significant_move()
                if move is not None:
                    low_price, high_price = move
                    self.fib_level = high_price - self.p.fib_ratio * (high_price - low_price)
                    self.awaiting_fib = True
                    self._stop_basis = min(trend_now, self.swing_lows[-1][1])

        if self.awaiting_fib and self.fib_level is not None:
            if self.data.low[0] <= self.fib_level:
                sma_rising = len(self) > self.p.sma_period and self.sma[0] > self.sma[-1]
                if sma_rising:
                    entry_price_est = self.data.close[0]
                    stop_price = self._stop_basis * (1 - self.p.stop_buffer_pct)
                    risk_per_share = entry_price_est - stop_price
                    if risk_per_share > 0:
                        size = int((self.broker.getvalue() * self.p.risk_pct) / risk_per_share)
                        if size > 0:
                            self._pending_stop_basis = stop_price
                            self.order = self.buy(size=size)
                self.awaiting_fib = False
                self.fib_level = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order == self.order:
            if order.status == order.Completed:
                entry_price = order.executed.price
                size = order.executed.size
                stop_price = self._pending_stop_basis
                risk = entry_price - stop_price
                take_profit = entry_price + self.p.reward_risk_ratio * risk
                self.stop_order = self.sell(exectype=bt.Order.Stop, price=stop_price, size=size)
                self.limit_order = self.sell(exectype=bt.Order.Limit, price=take_profit, size=size)
            self.order = None
            return

        if order == self.stop_order or order == self.limit_order:
            if order.status == order.Completed:
                self._cancel_exit_orders()
            elif order.status in [order.Canceled, order.Margin, order.Rejected]:
                if order == self.stop_order:
                    self.stop_order = None
                if order == self.limit_order:
                    self.limit_order = None
