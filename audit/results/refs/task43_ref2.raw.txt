import backtrader as bt


class BollingerBandsStrategy(bt.Strategy):
    params = (
        ("period", 20),
        ("devfactor", 2.0),
    )

    def __init__(self):
        self.boll = bt.indicators.BollingerBands(
            self.data.close,
            period=self.params.period,
            devfactor=self.params.devfactor,
        )
        self.order = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return
        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            self.order = None

    def next(self):
        if self.order:
            return

        close_prev = self.data.close[-1]
        close_curr = self.data.close[0]

        bot_prev = self.boll.lines.bot[-1]
        bot_curr = self.boll.lines.bot[0]

        top_prev = self.boll.lines.top[-1]
        top_curr = self.boll.lines.top[0]

        mid_prev = self.boll.lines.mid[-1]
        mid_curr = self.boll.lines.mid[0]

        if not self.position:
            # Long entry: close crosses lower_bb from below to above
            if close_prev < bot_prev and close_curr > bot_curr:
                cash = self.broker.getcash()
                size = int(cash / (close_curr * 1.001))
                if size > 0:
                    self.order = self.buy(size=size)
            # Short entry: close crosses upper_bb from above to below
            elif close_prev > top_prev and close_curr < top_curr:
                cash = self.broker.getcash()
                size = int(cash / (close_curr * 1.001))
                if size > 0:
                    self.order = self.sell(size=size)
        else:
            if self.position.size > 0:
                # Long exit: close crosses middle_bb from above to below
                if close_prev > mid_prev and close_curr < mid_curr:
                    self.order = self.close()
            elif self.position.size < 0:
                # Short exit: close crosses middle_bb from below to above
                if close_prev < mid_prev and close_curr > mid_curr:
                    self.order = self.close()
