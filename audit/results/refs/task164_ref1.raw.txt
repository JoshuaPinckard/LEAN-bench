import backtrader as bt


class CSILiteMomentum(bt.Strategy):
    params = (
        ("atr_period", 14),
        ("macd_fast", 12),
        ("macd_slow", 26),
        ("macd_signal", 9),
        ("sma_period", 50),
        ("rsi_period", 14),
    )

    def __init__(self):
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.macd = bt.indicators.MACD(
            self.data,
            period_me1=self.p.macd_fast,
            period_me2=self.p.macd_slow,
            period_signal=self.p.macd_signal,
        )
        self.sma_price = bt.indicators.SMA(self.data.close, period=self.p.sma_period)
        self.sma_atr = bt.indicators.SMA(self.atr, period=self.p.sma_period)
        self.sma_volume = bt.indicators.SMA(self.data.volume, period=self.p.sma_period)
        self.rsi = bt.indicators.RSI(self.data, period=self.p.rsi_period)
        self.macd_cross_down = bt.indicators.CrossDown(
            self.macd.macd, self.macd.signal
        )

        self.order = None
        self.tranches = 0
        self.peak_s = None

    def notify_order(self, order):
        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            if order.status == order.Completed:
                if order.isbuy():
                    self.tranches += 1
                    self.peak_s = None
                elif order.issell():
                    self.tranches = 0
                    self.peak_s = None
            self.order = None

    def next(self):
        if self.order:
            return

        macd_hist = max(0.0, self.macd.macd[0] - self.macd.signal[0])
        score = (
            macd_hist
            * (self.atr[0] / self.sma_atr[0])
            * (self.data.volume[0] / self.sma_volume[0])
            * 100.0
        )

        if self.position.size:
            if self.peak_s is None or score > self.peak_s:
                self.peak_s = score

            if (
                score < 0.70 * self.peak_s
                or self.macd_cross_down[0] > 0
                or self.rsi[0] < 45
            ):
                self.order = self.close()
                return

        if (
            score > 1.5
            and self.data.close[0] > self.sma_price[0]
            and self.macd.macd[0] > self.macd.signal[0]
            and self.tranches < 3
        ):
            capital = self.broker.getvalue()
            exposure = abs(self.position.size) * self.data.close[0]

            if exposure < 0.50 * capital:
                tranche_value = 0.05 * capital
                available_exposure = 0.50 * capital - exposure
                size = int(min(tranche_value, available_exposure) / self.data.close[0])

                if size > 0:
                    self.order = self.buy(size=size)
