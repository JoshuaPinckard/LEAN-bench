import backtrader as bt


class GridMartingaleStrategy(bt.Strategy):
    params = (
        ("grid_step", 0.01),  # Grid step distance (1.0%)
        ("tp_pct", 0.005),  # Take profit margin (0.5%)
        ("base_volume", 1),  # Base volume for initial order
        ("martingale_mult", 2.0),  # Volume multiplier for reinvestments
        ("max_reinvestments", 5),  # Max number of reinvestment levels
    )

    def __init__(self):
        self.order_buy_grid = None
        self.order_sell_grid = None
        self.order_entry = None
        self.order_tp = None

        self.position_side = 0  # 1 for Long, -1 for Short, 0 for Flat
        self.reinvest_level = 0
        self.averaged_price = 0.0
        self.total_size = 0
        self.last_entry_price = 0.0

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Canceled, order.Margin, order.Rejected]:
            if (
                self.order_buy_grid
                and order.ref == self.order_buy_grid.ref
            ):
                self.order_buy_grid = None
            if (
                self.order_sell_grid
                and order.ref == self.order_sell_grid.ref
            ):
                self.order_sell_grid = None
            if self.order_entry and order.ref == self.order_entry.ref:
                self.order_entry = None
            if self.order_tp and order.ref == self.order_tp.ref:
                self.order_tp = None
            return

        if order.status == order.Completed:
            if (
                self.order_buy_grid
                and order.ref == self.order_buy_grid.ref
            ):
                self.order_buy_grid = None
                self._handle_initial_fill(order, side=1)
            elif (
                self.order_sell_grid
                and order.ref == self.order_sell_grid.ref
            ):
                self.order_sell_grid = None
                self._handle_initial_fill(order, side=-1)
            elif self.order_entry and order.ref == self.order_entry.ref:
                self.order_entry = None
                self._handle_reinvestment_fill(order)
            elif self.order_tp and order.ref == self.order_tp.ref:
                self.order_tp = None
                self._handle_tp_fill(order)

    def _handle_initial_fill(self, order, side):
        if side == 1 and self.order_sell_grid:
            self.cancel(self.order_sell_grid)
            self.order_sell_grid = None
        elif side == -1 and self.order_buy_grid:
            self.cancel(self.order_buy_grid)
            self.order_buy_grid = None

        self.position_side = side
        fill_price = order.executed.price
        fill_size = abs(int(order.executed.size))

        self.averaged_price = fill_price
        self.total_size = fill_size
        self.last_entry_price = fill_price
        self.reinvest_level = 0

        if side == 1:
            tp_price = self.averaged_price * (1.0 + self.p.tp_pct)
            self.order_tp = self.sell(
                exectype=bt.Order.Limit, price=tp_price, size=self.total_size
            )
        else:
            tp_price = self.averaged_price * (1.0 - self.p.tp_pct)
            self.order_tp = self.buy(
                exectype=bt.Order.Limit, price=tp_price, size=self.total_size
            )

        if self.reinvest_level < self.p.max_reinvestments:
            next_level = self.reinvest_level + 1
            next_size = int(
                self.p.base_volume * (self.p.martingale_mult**next_level)
            )
            if side == 1:
                next_price = self.last_entry_price * (1.0 - self.p.grid_step)
                self.order_entry = self.buy(
                    exectype=bt.Order.Limit, price=next_price, size=next_size
                )
            else:
                next_price = self.last_entry_price * (1.0 + self.p.grid_step)
                self.order_entry = self.sell(
                    exectype=bt.Order.Limit, price=next_price, size=next_size
                )

    def _handle_reinvestment_fill(self, order):
        fill_price = order.executed.price
        fill_size = abs(int(order.executed.size))

        total_cost = (self.averaged_price * self.total_size) + (
            fill_price * fill_size
        )
        self.total_size += fill_size
        self.averaged_price = total_cost / self.total_size
        self.last_entry_price = fill_price
        self.reinvest_level += 1

        if self.order_tp:
            self.cancel(self.order_tp)
            self.order_tp = None

        if self.position_side == 1:
            tp_price = self.averaged_price * (1.0 + self.p.tp_pct)
            self.order_tp = self.sell(
                exectype=bt.Order.Limit, price=tp_price, size=self.total_size
            )
        else:
            tp_price = self.averaged_price * (1.0 - self.p.tp_pct)
            self.order_tp = self.buy(
                exectype=bt.Order.Limit, price=tp_price, size=self.total_size
            )

        if self.reinvest_level < self.p.max_reinvestments:
            next_level = self.reinvest_level + 1
            next_size = int(
                self.p.base_volume * (self.p.martingale_mult**next_level)
            )
            if self.position_side == 1:
                next_price = self.last_entry_price * (1.0 - self.p.grid_step)
                self.order_entry = self.buy(
                    exectype=bt.Order.Limit, price=next_price, size=next_size
                )
            else:
                next_price = self.last_entry_price * (1.0 + self.p.grid_step)
                self.order_entry = self.sell(
                    exectype=bt.Order.Limit, price=next_price, size=next_size
                )

    def _handle_tp_fill(self, order):
        self.position_side = 0
        self.averaged_price = 0.0
        self.total_size = 0
        self.reinvest_level = 0
        self.last_entry_price = 0.0

        if self.order_entry:
            self.cancel(self.order_entry)
            self.order_entry = None

    def next(self):
        if self.position_side == 0:
            current_close = self.data.close[0]
            buy_target_price = current_close * (1.0 - self.p.grid_step)
            sell_target_price = current_close * (1.0 + self.p.grid_step)

            if self.order_buy_grid:
                self.cancel(self.order_buy_grid)
                self.order_buy_grid = None

            if self.order_sell_grid:
                self.cancel(self.order_sell_grid)
                self.order_sell_grid = None

            self.order_buy_grid = self.buy(
                exectype=bt.Order.Limit,
                price=buy_target_price,
                size=self.p.base_volume,
            )
            self.order_sell_grid = self.sell(
                exectype=bt.Order.Limit,
                price=sell_target_price,
                size=self.p.base_volume,
            )
