class CCISlopeMeanReversionStrategy(bt.Strategy):
    params = (
        ("cci_period", 20),
        ("slope_period", 20),
        ("stats_period", 20),
    )

    def __init__(self):
        self.cci = bt.indicators.CommodityChannelIndex(
            self.data, period=self.p.cci_period
        )
        self.slope = bt.indicators.LinearRegressionSlope(
            self.cci, period=self.p.slope_period
        )
        self.mean_slope = bt.indicators.SMA(
            self.slope, period=self.p.stats_period
        )
        self.std_slope = bt.indicators.StdDev(
            self.slope, period=self.p.stats_period
        )

        self.order = None
        self.stop_order = None

    def notify_order(self, order):
        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            if order is self.order:
                if order.status == order.Completed and order.isbuy():
                    self.stop_order = self.sell(
                        exectype=bt.Order.Stop,
                        price=order.executed.price * 0.98,
                    )
                elif order.status == order.Completed and order.issell():
                    self.stop_order = self.buy(
                        exectype=bt.Order.Stop,
                        price=order.executed.price * 1.02,
                    )
                self.order = None

            if order is self.stop_order:
                self.stop_order = None

    def next(self):
        if self.order:
            return

        slope = self.slope[0]
        mean_slope = self.mean_slope[0]
        std_slope = self.std_slope[0]

        if not self.position:
            if std_slope == 0:
                return

            z = (slope - mean_slope) / std_slope

            if z > 2:
                self.order = self.buy()
            elif z < -2:
                self.order = self.sell()
        elif self.position.size > 0 and slope <= mean_slope:
            if self.stop_order:
                self.cancel(self.stop_order)
                self.stop_order = None
            self.order = self.close()
        elif self.position.size < 0 and slope >= mean_slope:
            if self.stop_order:
                self.cancel(self.stop_order)
                self.stop_order = None
            self.order = self.close()
