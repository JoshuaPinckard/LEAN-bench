import backtrader as bt


class MarketMakerGridStrategy(bt.Strategy):

    LEVEL_DEVIATIONS = (0.0005, 0.0010, 0.0015)  # 0.05%, 0.10%, 0.15%
    LEVEL_WEIGHTS = (0.2, 0.3, 0.5)
    TOTAL_BUDGET_PCT = 0.10
    ATR_THRESHOLD = 0.002
    SLIPPAGE_THRESHOLD = 0.001
    MIN_SIDE_VOLUME = 0.1

    def __init__(self):
        self.sma = bt.indicators.SMA(self.data.close, period=20)
        self.atr = bt.indicators.ATR(self.data, period=14)
        self.open_orders = []
        self.skip_cycle = False

    def notify_order(self, order):
        if order.status == order.Completed:
            limit_price = order.created.price
            fill_price = order.executed.price
            if abs(fill_price - limit_price) / limit_price > self.SLIPPAGE_THRESHOLD:
                for o in list(self.open_orders):
                    if o.alive():
                        self.cancel(o)
                self.skip_cycle = True

        if order.status in (order.Completed, order.Canceled, order.Margin,
                             order.Rejected, order.Expired):
            if order in self.open_orders:
                self.open_orders.remove(order)

    def next(self):
        # Slippage control: unconditionally suppress the immediately following cycle.
        if self.skip_cycle:
            self.skip_cycle = False
            return

        atr_pct = self.atr[0] / self.data.close[0]
        if atr_pct > self.ATR_THRESHOLD:
            for o in list(self.open_orders):
                if o.alive():
                    self.cancel(o)
            return

        capital = self.broker.getvalue()
        side_budget = capital * self.TOTAL_BUDGET_PCT / 2.0
        sma_val = self.sma[0]

        buy_prices = [sma_val * (1 - dev) for dev in self.LEVEL_DEVIATIONS]
        sell_prices = [sma_val * (1 + dev) for dev in self.LEVEL_DEVIATIONS]

        buy_sizes = [side_budget * w / p for w, p in zip(self.LEVEL_WEIGHTS, buy_prices)]
        sell_sizes = [side_budget * w / p for w, p in zip(self.LEVEL_WEIGHTS, sell_prices)]

        buy_total = sum(buy_sizes)
        if buy_total < self.MIN_SIDE_VOLUME:
            # Absolute floor takes precedence over the nominal budget percentage.
            scale = self.MIN_SIDE_VOLUME / buy_total
            buy_sizes = [s * scale for s in buy_sizes]

        sell_total = sum(sell_sizes)
        if sell_total < self.MIN_SIDE_VOLUME:
            scale = self.MIN_SIDE_VOLUME / sell_total
            sell_sizes = [s * scale for s in sell_sizes]

        # valid=bt.Order.DAY expires an unfilled order at the end of this bar (TTL = 1 bar).
        for price, size in zip(buy_prices, buy_sizes):
            o = self.buy(exectype=bt.Order.Limit, price=price, size=size,
                          valid=bt.Order.DAY)
            self.open_orders.append(o)

        for price, size in zip(sell_prices, sell_sizes):
            o = self.sell(exectype=bt.Order.Limit, price=price, size=size,
                           valid=bt.Order.DAY)
            self.open_orders.append(o)
