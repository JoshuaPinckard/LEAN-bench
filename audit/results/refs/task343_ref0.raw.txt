import backtrader as bt


class MeanReversionBBStrategy(bt.Strategy):
    params = dict(
        bb_period=20,
        bb_dev=2,
        rsi_period=14,
        rsi_upper=70,
        rsi_lower=30,
        atr_period=14,
        atr_mult=2,
    )

    def __init__(self):
        self.bb = bt.indicators.BollingerBands(
            self.data.close, period=self.p.bb_period, devfactor=self.p.bb_dev
        )
        self.rsi = bt.indicators.RSI(self.data.close, period=self.p.rsi_period)
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)

        self.order = None
        self.stop_price = None

    def notify_order(self, order):
        if order.status == order.Completed:
            if self.position:
                entry_price = order.executed.price
                offset = self.p.atr_mult * self.atr[0]
                if self.position.size > 0:
                    self.stop_price = entry_price - offset
                else:
                    self.stop_price = entry_price + offset
            else:
                self.stop_price = None
            self.order = None
        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        if self.order:
            return

        close = self.data.close[0]

        if self.position:
            in_channel = self.bb.bot[0] <= close <= self.bb.top[0]
            if self.position.size > 0:
                stop_hit = close <= self.stop_price
                opposing_signal = close > self.bb.top[0] and self.rsi[0] > self.p.rsi_upper
                if stop_hit or in_channel or opposing_signal:
                    self.order = self.close()
            else:
                stop_hit = close >= self.stop_price
                opposing_signal = close < self.bb.bot[0] and self.rsi[0] < self.p.rsi_lower
                if stop_hit or in_channel or opposing_signal:
                    self.order = self.close()
            return

        if close > self.bb.top[0] and self.rsi[0] > self.p.rsi_upper:
            self.order = self.sell()
        elif close < self.bb.bot[0] and self.rsi[0] < self.p.rsi_lower:
            self.order = self.buy()
