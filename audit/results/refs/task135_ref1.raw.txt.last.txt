import backtrader as bt
from datetime import date, datetime, timedelta


class ExchangeScheduleFilterStrategy(bt.Strategy):
    @staticmethod
    def _easter_sunday(year):
        a = year % 19
        b = year // 100
        c = year % 100
        d = b // 4
        e = b % 4
        f = (b + 8) // 25
        g = (b - f + 1) // 3
        h = (19 * a + b - d - g + 15) % 30
        i = c // 4
        k = c % 4
        l = (32 + 2 * e + 2 * i - h - k) % 7
        m = (a + 11 * h + 22 * l) // 451
        month = (h + l - 7 * m + 114) // 31
        day = (h + l - 7 * m + 114) % 31 + 1
        return date(year, month, day)

    @classmethod
    def is_business_day(cls, value):
        if isinstance(value, datetime):
            value = value.date()

        if value.weekday() >= 5:
            return False

        year = value.year
        holidays = set()

        new_year = date(year, 1, 1)
        holidays.add(new_year)
        if new_year.weekday() == 5:
            holidays.add(date(year - 1, 12, 31))
        elif new_year.weekday() == 6:
            holidays.add(date(year, 1, 2))

        holidays.add(date(year, 1, 1) + timedelta(days=(14 - date(year, 1, 1).weekday()) % 7))
        holidays.add(date(year, 2, 1) + timedelta(days=(14 - date(year, 2, 1).weekday()) % 7))
        holidays.add(cls._easter_sunday(year) - timedelta(days=2))

        memorial_day = date(year, 5, 31)
        holidays.add(memorial_day - timedelta(days=(memorial_day.weekday() - 0) % 7))

        if year >= 2022:
            juneteenth = date(year, 6, 19)
            holidays.add(juneteenth)
            if juneteenth.weekday() == 5:
                holidays.add(date(year, 6, 18))
            elif juneteenth.weekday() == 6:
                holidays.add(date(year, 6, 20))

        independence_day = date(year, 7, 4)
        holidays.add(independence_day)
        if independence_day.weekday() == 5:
            holidays.add(date(year, 7, 3))
        elif independence_day.weekday() == 6:
            holidays.add(date(year, 7, 5))

        holidays.add(date(year, 9, 1) + timedelta(days=(7 - date(year, 9, 1).weekday()) % 7))
        holidays.add(date(year, 11, 1) + timedelta(days=(3 - date(year, 11, 1).weekday()) % 7 + 21))

        christmas = date(year, 12, 25)
        holidays.add(christmas)
        if christmas.weekday() == 5:
            holidays.add(date(year, 12, 24))
        elif christmas.weekday() == 6:
            holidays.add(date(year, 12, 26))

        return value not in holidays

    @classmethod
    def is_last_business_day_of_week(cls, value):
        if isinstance(value, datetime):
            value = value.date()

        return cls.is_business_day(value) and not cls.is_business_day(
            value + timedelta(days=1)
        )

    def __init__(self):
        self.order = None

    def notify_order(self, order):
        if order.status in (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            self.order = None

    def next(self):
        current_date = self.data.datetime.date(0)

        if self.position and self.is_last_business_day_of_week(current_date):
            if self.order is None:
                self.order = self.close()
