import backtrader as bt


class RegimeVolatilityStrategy(bt.Strategy):
    params = dict(
        atr_period=14,
        bb_period=20,
        bb_dev=2,
        vol_ema_period=50,
        ema_fast_period=10,
        ema_slow_period=30,
        sma_trend_period=50,
        rsi_period=14,
        macd_fast=12,
        macd_slow=26,
        macd_signal=9,
        cvi_low=0.9,
        cvi_high=1.1,
        cvi_mid=1.0,
        rsi_mid=50,
        rsi_meanrev_long_entry=35,
        rsi_meanrev_long_exit=55,
        low_atr_mult=1.5,
        high_atr_mult=2.5,
        base_risk_pct=0.01,
        size_mult_low=1.3,
        size_mult_mid=1.0,
        size_mult_high=0.7,
    )

    def __init__(self):
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.bb = bt.indicators.BollingerBands(
            self.data, period=self.p.bb_period, devfactor=self.p.bb_dev
        )
        self.bb_width = self.bb.lines.top - self.bb.lines.bot
        self.atr_ema = bt.indicators.EMA(self.atr, period=self.p.vol_ema_period)
        self.bbw_ema = bt.indicators.EMA(self.bb_width, period=self.p.vol_ema_period)
        self.ema_fast = bt.indicators.EMA(self.data.close, period=self.p.ema_fast_period)
        self.ema_slow = bt.indicators.EMA(self.data.close, period=self.p.ema_slow_period)
        self.sma_trend = bt.indicators.SMA(self.data.close, period=self.p.sma_trend_period)
        self.rsi = bt.indicators.RSI(self.data.close, period=self.p.rsi_period)
        self.macd = bt.indicators.MACD(
            self.data.close,
            period_me1=self.p.macd_fast,
            period_me2=self.p.macd_slow,
            period_signal=self.p.macd_signal,
        )
        self.ema_cross = bt.indicators.CrossOver(self.ema_fast, self.ema_slow)

        self.order = None
        self.entry_type = None
        self.stop_price = None
        self.prev_cvi = None
        self.prev_macd_hist = None

    def notify_order(self, order):
        if order.status in (order.Completed, order.Canceled, order.Margin, order.Rejected):
            self.order = None

    def next(self):
        cvi = 0.5 * (self.atr[0] / self.atr_ema[0]) + 0.5 * (
            self.bb_width[0] / self.bbw_ema[0]
        )
        macd_hist = self.macd.macd[0] - self.macd.signal[0]

        prev_cvi = self.prev_cvi
        prev_macd_hist = self.prev_macd_hist
        self.prev_cvi = cvi
        self.prev_macd_hist = macd_hist

        is_last_bar = len(self.data) == self.data.buflen()

        if self.order:
            return

        close = self.data.close[0]

        if self.position:
            exit_signal = is_last_bar
            if not exit_signal:
                if self.entry_type == 'trend':
                    if prev_cvi is not None and prev_cvi <= self.p.cvi_mid < cvi:
                        exit_signal = True
                    elif close <= self.stop_price:
                        exit_signal = True
                else:
                    if prev_cvi is not None and prev_cvi >= self.p.cvi_mid > cvi:
                        exit_signal = True
                    elif close <= self.stop_price:
                        exit_signal = True
                    elif self.entry_type == 'meanrev' and (
                        close >= self.bb.lines.mid[0]
                        or self.rsi[0] > self.p.rsi_meanrev_long_exit
                    ):
                        exit_signal = True

            if exit_signal:
                self.order = self.close()
                self.entry_type = None
                self.stop_price = None
            return

        if is_last_bar:
            return

        cvi_mult = self.p.size_mult_mid
        if cvi < self.p.cvi_low:
            cvi_mult = self.p.size_mult_low
        elif cvi > self.p.cvi_high:
            cvi_mult = self.p.size_mult_high

        def submit_long(entry_type, atr_mult):
            stop_dist = atr_mult * self.atr[0]
            if stop_dist <= 0:
                return False
            equity = self.broker.getvalue()
            raw_size = (equity * self.p.base_risk_pct / stop_dist) * cvi_mult
            size = int(raw_size)
            if size < 1:
                return False
            self.stop_price = close - stop_dist
            self.entry_type = entry_type
            self.order = self.buy(size=size)
            return True

        if cvi < self.p.cvi_low:
            if (
                self.ema_cross[0] > 0
                and close > self.sma_trend[0]
                and self.rsi[0] > self.p.rsi_mid
            ):
                if submit_long('trend', self.p.low_atr_mult):
                    return
        elif cvi > self.p.cvi_high:
            if (
                close <= self.bb.lines.bot[0]
                and self.rsi[0] < self.p.rsi_meanrev_long_entry
            ):
                if submit_long('meanrev', self.p.high_atr_mult):
                    return

        if prev_cvi is not None and prev_macd_hist is not None:
            cvi_cross_up = prev_cvi < self.p.cvi_mid <= cvi
            macd_cross_up = prev_macd_hist < 0 <= macd_hist
            if (
                cvi_cross_up
                and macd_cross_up
                and self.rsi[0] > self.p.rsi_mid
                and close > self.sma_trend[0]
            ):
                submit_long('shift', self.p.high_atr_mult)
