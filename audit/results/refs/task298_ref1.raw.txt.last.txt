import backtrader as bt


class SniperPullback(bt.Strategy):
    params = (
        ("bb_period", 20),
        ("bb_devfactor", 2),
        ("atr_period", 14),
        ("volume_sma_period", 20),
    )

    def __init__(self):
        self.bbands = bt.indicators.BollingerBands(
            self.data.close,
            period=self.p.bb_period,
            devfactor=self.p.bb_devfactor,
        )
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.volume_sma = bt.indicators.SMA(
            self.data.volume,
            period=self.p.volume_sma_period,
        )

        self.order = None
        self.h_ref = None
        self.h_ref_bar = None

    def notify_order(self, order):
        if order.status in (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            self.order = None

    def next(self):
        current_bar = len(self)
        close = self.data.close[0]

        if close > self.bbands.top[0]:
            self.h_ref = close
            self.h_ref_bar = current_bar

        if self.order:
            return

        if self.position:
            entry_price = self.position.price
            if close >= entry_price * 1.05 or close <= entry_price * 0.97:
                self.order = self.close()
            return

        if self.h_ref is None or current_bar - self.h_ref_bar > 5:
            return

        pullback = (self.h_ref - close) / self.h_ref
        if (
            0.09 <= pullback <= 0.13
            and self.data.volume[0] > 1.5 * self.volume_sma[0]
        ):
            size = int((self.broker.getvalue() * 0.02) / close)
            if size > 0:
                self.order = self.buy(size=size)
