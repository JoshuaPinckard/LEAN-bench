```python
import collections
import statistics

import backtrader as bt


class MeanReversionBollingerStrategy(bt.Strategy):
    params = dict(
        bb_period=20,
        bb_devfactor=1.5,
        sma_period=200,
        atr_period=14,
        atr_lookback=252,
        atr_percentile_n=5,   # n=5 -> quantile cut points at 20/40/60/80
        exit_bars=5,
        position_value=1000.0,
        max_loss_pct=0.02,
    )

    def __init__(self):
        self.bb = bt.indicators.BollingerBands(
            self.data.close, period=self.p.bb_period, devfactor=self.p.bb_devfactor
        )
        self.sma200 = bt.indicators.SMA(self.data.close, period=self.p.sma_period)
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)

        self.cross_lower = bt.indicators.CrossOver(self.data.close, self.bb.bot)
        self.cross_upper = bt.indicators.CrossOver(self.data.close, self.bb.top)
        self.cross_mid = bt.indicators.CrossOver(self.data.close, self.bb.mid)

        self.atr_history = collections.deque(maxlen=self.p.atr_lookback)

        self.order = None
        self.bar_executed = None
        self.initial_capital = self.broker.getvalue()

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return

        if order.status == order.Completed:
            if self.position:
                self.bar_executed = len(self)
            else:
                self.bar_executed = None

        self.order = None

    def next(self):
        if self.order:
            return

        self.atr_history.append(self.atr[0])

        if self.position:
            floating_pnl = (self.data.close[0] - self.position.price) * self.position.size
            if floating_pnl < -self.p.max_loss_pct * self.initial_capital:
                self.order = self.close()
                return

            bars_in_trade = len(self) - self.bar_executed
            if self.cross_mid[0] != 0 or bars_in_trade >= self.p.exit_bars:
                self.order = self.close()
            return

        if len(self.atr_history) < self.p.atr_lookback:
            return

        atr_p20 = statistics.quantiles(list(self.atr_history), n=self.p.atr_percentile_n)[0]
        if self.atr[0] < atr_p20:
            return

        price = self.data.close[0]
        size = self.p.position_value / price

        if self.cross_lower[0] < 0 and price <= self.sma200[0]:
            self.order = self.buy(size=size)
        elif self.cross_upper[0] > 0 and price >= self.sma200[0]:
            self.order = self.sell(size=size)
```
