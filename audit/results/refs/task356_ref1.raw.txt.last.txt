import backtrader as bt


class BollingerEmaReversalStrategy(bt.Strategy):
    def __init__(self):
        self.bollinger = bt.indicators.BollingerBands(
            self.data.close, period=20, devfactor=2
        )
        self.ema50 = bt.indicators.EMA(self.data.close, period=50)

        self.cross_upper = bt.indicators.CrossOver(
            self.data.close, self.bollinger.top
        )
        self.cross_lower = bt.indicators.CrossOver(
            self.data.close, self.bollinger.bot
        )
        self.cross_ema = bt.indicators.CrossOver(self.data.close, self.ema50)

        self.pending_orders = set()

    def notify_order(self, order):
        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            self.pending_orders.discard(order.ref)

    def _track(self, order):
        self.pending_orders.add(order.ref)

    def next(self):
        if self.pending_orders:
            return

        long_entry = self.cross_upper[0] > 0 and self.data.close[0] > self.ema50[0]
        short_entry = self.cross_lower[0] < 0 and self.data.close[0] < self.ema50[0]
        long_exit = self.cross_ema[0] < 0
        short_exit = self.cross_ema[0] > 0

        if self.position.size > 0:
            if short_entry:
                self._track(self.close())
                self._track(self.sell())
            elif long_exit:
                self._track(self.close())

        elif self.position.size < 0:
            if long_entry:
                self._track(self.close())
                self._track(self.buy())
            elif short_exit:
                self._track(self.close())

        elif long_entry:
            self._track(self.buy())

        elif short_entry:
            self._track(self.sell())
