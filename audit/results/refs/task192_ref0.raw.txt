import math
import collections

import backtrader as bt


class MultiStrategyORBIncome(bt.Strategy):
    params = dict(
        bb_period=20, bb_dev=2, atr_period=14, ema_period=20, rsi_period=14,
        vol_sma_period=20, bw_lookback=252, or_bars=6,
        base_capital=150000.0, risk_pct=0.01,
        alloc_s1=0.40, alloc_s2=0.30, alloc_s3=0.30,
        s1_timeout_bars=4,
        session_end_hour=15, session_end_minute=55,
    )

    def __init__(self):
        self.bb = bt.indicators.BollingerBands(self.data, period=self.p.bb_period, devfactor=self.p.bb_dev)
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.ema = bt.indicators.EMA(self.data, period=self.p.ema_period)
        self.rsi = bt.indicators.RSI(self.data, period=self.p.rsi_period)
        self.vol_sma = bt.indicators.SMA(self.data.volume, period=self.p.vol_sma_period)

        self.bw_history = collections.deque(maxlen=self.p.bw_lookback)

        self.order = None
        self.active_strategy = None
        self.entry_price = None
        self.stop_price = None
        self.tp_price = None
        self.entry_bar = None
        self.risk_r = None

        self.session_date = None
        self.session_bar_idx = 0
        self.or_high = None
        self.or_low = None
        self.or_complete = False
        self.mid_or = None
        self.broke_above_or = False
        self.broke_below_or = False
        self.above_or_count = 0
        self.below_or_count = 0

        self.log_records = []

    def log(self, msg):
        dt = self.data.datetime.datetime(0)
        self.log_records.append("%s %s" % (dt.isoformat(), msg))

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return
        if order.status in (order.Margin, order.Rejected, order.Canceled) and not self.position:
            self.active_strategy = None
            self.entry_price = None
            self.stop_price = None
            self.tp_price = None
            self.risk_r = None
            self.entry_bar = None
        self.order = None

    def _new_session_reset(self):
        self.session_bar_idx = 0
        self.or_high = None
        self.or_low = None
        self.or_complete = False
        self.mid_or = None
        self.broke_above_or = False
        self.broke_below_or = False
        self.above_or_count = 0
        self.below_or_count = 0

    def _update_or(self):
        h = self.data.high[0]
        l = self.data.low[0]
        if self.or_high is None:
            self.or_high = h
            self.or_low = l
        else:
            self.or_high = max(self.or_high, h)
            self.or_low = min(self.or_low, l)
        if self.session_bar_idx == self.p.or_bars:
            self.or_complete = True
            self.mid_or = (self.or_high + self.or_low) / 2.0

    def _bandwidth_percentile(self, bw):
        self.bw_history.append(bw)
        hist = self.bw_history
        n = len(hist)
        rank = sum(1 for x in hist if x <= bw)
        return (rank / n) * 100.0

    def _position_size(self, risk_dist, alloc_pct, price):
        if risk_dist <= 0 or price <= 0:
            return 0
        risk_amount = self.p.base_capital * self.p.risk_pct
        risk_size = int(round(risk_amount / risk_dist))
        alloc_capital = self.p.base_capital * alloc_pct
        alloc_size = int(alloc_capital // price)
        return max(0, min(risk_size, alloc_size))

    def _register_trade(self, strategy, entry_price, stop, tp, risk_dist, direction, pct):
        self.active_strategy = strategy
        self.entry_price = entry_price
        self.stop_price = stop
        self.tp_price = tp
        self.risk_r = risk_dist
        self.entry_bar = len(self)
        self.log("ENTRY strat=%d dir=%d price=%.2f stop=%.2f tp=%.2f R=%.4f bw_pct=%.2f" %
                  (strategy, direction, entry_price, stop, tp, risk_dist, pct))

    def _reset_trade_state(self):
        self.log("EXIT strat=%s" % self.active_strategy)
        self.active_strategy = None
        self.entry_price = None
        self.stop_price = None
        self.tp_price = None
        self.risk_r = None
        self.entry_bar = None

    def _manage_position(self, close, is_eod):
        if self.order:
            return
        strat = self.active_strategy
        pos_size = self.position.size

        if strat == 1:
            bars_held = len(self) - self.entry_bar
            if pos_size > 0:
                tp_hit = close >= self.tp_price
                sl_hit = close <= self.stop_price
            else:
                tp_hit = close <= self.tp_price
                sl_hit = close >= self.stop_price
            timeout_hit = bars_held >= self.p.s1_timeout_bars
            if tp_hit or sl_hit or timeout_hit:
                self.order = self.close()
                self._reset_trade_state()
            return

        if strat == 2:
            if pos_size > 0:
                tp_hit = close >= self.tp_price
                sl_hit = close <= self.stop_price
            else:
                tp_hit = close <= self.tp_price
                sl_hit = close >= self.stop_price
            if tp_hit or sl_hit or is_eod:
                self.order = self.close()
                self._reset_trade_state()
            return

        if strat == 3:
            atr = self.atr[0]
            if pos_size > 0:
                tp_hit = close >= self.tp_price
                sl_hit = close <= self.stop_price
                trail_level = max(self.ema[0], close - atr)
                trail_hit = close <= trail_level
            else:
                tp_hit = close <= self.tp_price
                sl_hit = close >= self.stop_price
                trail_level = min(self.ema[0], close + atr)
                trail_hit = close >= trail_level
            if tp_hit or sl_hit or trail_hit or is_eod:
                self.order = self.close()
                self._reset_trade_state()
            return

    def _try_strategy1(self, close, top, bot, mid, pct):
        atr = self.atr[0]
        rsi = self.rsi[0]
        if close >= top and rsi > 60:
            stop = top + atr
            risk_dist = stop - close
            tp = max(mid, close - 0.5 * risk_dist)
            size = self._position_size(risk_dist, self.p.alloc_s1, close)
            if size > 0:
                self.order = self.sell(size=size)
                self._register_trade(1, close, stop, tp, risk_dist, -1, pct)
        elif close <= bot and rsi < 40:
            stop = bot - atr
            risk_dist = close - stop
            tp = min(mid, close + 0.5 * risk_dist)
            size = self._position_size(risk_dist, self.p.alloc_s1, close)
            if size > 0:
                self.order = self.buy(size=size)
                self._register_trade(1, close, stop, tp, risk_dist, 1, pct)

    def _try_strategy2(self, close, pct):
        atr = self.atr[0]
        high = self.data.high[0]
        low = self.data.low[0]
        if high > self.or_high:
            self.broke_above_or = True
        if low < self.or_low:
            self.broke_below_or = True

        if self.broke_above_or and close < self.or_high:
            stop = self.or_high + 0.5 * atr
            risk_dist = stop - close
            if risk_dist > 0:
                size = self._position_size(risk_dist, self.p.alloc_s2, close)
                if size > 0:
                    self.order = self.sell(size=size)
                    self._register_trade(2, close, stop, self.mid_or, risk_dist, -1, pct)
                    self.broke_above_or = False
            return
        if self.broke_below_or and close > self.or_low:
            stop = self.or_low - 0.5 * atr
            risk_dist = close - stop
            if risk_dist > 0:
                size = self._position_size(risk_dist, self.p.alloc_s2, close)
                if size > 0:
                    self.order = self.buy(size=size)
                    self._register_trade(2, close, stop, self.mid_or, risk_dist, 1, pct)
                    self.broke_below_or = False

    def _try_strategy3(self, close, pct):
        atr = self.atr[0]
        ema_rising = self.ema[0] > self.ema[-1]
        ema_falling = self.ema[0] < self.ema[-1]

        if self.above_or_count >= 2 and ema_rising:
            stop = max(close - atr, self.or_low)
            risk_dist = close - stop
            if risk_dist > 0:
                size = self._position_size(risk_dist, self.p.alloc_s3, close)
                if size > 0:
                    tp = close + 2 * risk_dist
                    self.order = self.buy(size=size)
                    self._register_trade(3, close, stop, tp, risk_dist, 1, pct)
            return
        if self.below_or_count >= 2 and ema_falling:
            stop = min(close + atr, self.or_high)
            risk_dist = stop - close
            if risk_dist > 0:
                size = self._position_size(risk_dist, self.p.alloc_s3, close)
                if size > 0:
                    tp = close - 2 * risk_dist
                    self.order = self.sell(size=size)
                    self._register_trade(3, close, stop, tp, risk_dist, -1, pct)

    def next(self):
        dt = self.data.datetime.datetime(0)
        date_key = (dt.year, dt.month, dt.day)
        if date_key != self.session_date:
            self.session_date = date_key
            self._new_session_reset()
        self.session_bar_idx += 1
        if self.session_bar_idx <= self.p.or_bars:
            self._update_or()

        close = self.data.close[0]
        mid = self.bb.mid[0]
        top = self.bb.top[0]
        bot = self.bb.bot[0]
        bw = (top - bot) / mid if mid else 0.0
        pct = self._bandwidth_percentile(bw)

        is_eod = (dt.hour == self.p.session_end_hour and dt.minute == self.p.session_end_minute)

        if self.position:
            self._manage_position(close, is_eod)
            return

        if self.order:
            return

        if pct < 10:
            return

        if self.data.volume[0] < self.vol_sma[0]:
            return

        if self.or_complete:
            if close > self.or_high:
                self.above_or_count += 1
            else:
                self.above_or_count = 0
            if close < self.or_low:
                self.below_or_count += 1
            else:
                self.below_or_count = 0

        if 0 <= pct <= 30:
            self._try_strategy1(close, top, bot, mid, pct)
            if self.order:
                return

        if self.or_complete and 70 <= pct <= 90:
            self._try_strategy2(close, pct)
            if self.order:
                return

        if self.or_complete:
            self._try_strategy3(close, pct)
