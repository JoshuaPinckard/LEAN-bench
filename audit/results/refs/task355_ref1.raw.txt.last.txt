import backtrader as bt


class IntradayOutsideBarReversal(bt.Strategy):
    params = (("trend_period", 6), ("risk_fraction", 0.01), ("tick_size", 0.01),)

    def __init__(self):
        self.pending_order = None
        self.stop_order = None
        self.exit_order = None
        self.outside_bar_low = None
        self.outside_bar_high = None

    def _in_market_hours(self):
        current = self.data.datetime.datetime(0)
        minutes = current.hour * 60 + current.minute
        return 9 * 60 + 30 <= minutes < 16 * 60

    def _submit_long_bracket(self):
        entry_price = self.data.close[0]
        self.outside_bar_low = self.data.low[0]
        self.outside_bar_high = self.data.high[0]
        stop_price = self.outside_bar_low - self.p.tick_size
        risk_per_share = entry_price - stop_price
        size = int((self.broker.getvalue() * self.p.risk_fraction) / risk_per_share)

        if size <= 0:
            return

        orders = self.buy_bracket(
            size=size,
            exectype=bt.Order.Market,
            stopprice=stop_price,
            stopexec=bt.Order.Stop,
        )
        self.pending_order = orders[0]
        self.stop_order = orders[2]

    def _submit_short_bracket(self):
        entry_price = self.data.close[0]
        self.outside_bar_low = self.data.low[0]
        self.outside_bar_high = self.data.high[0]
        stop_price = self.outside_bar_high + self.p.tick_size
        risk_per_share = stop_price - entry_price
        size = int((self.broker.getvalue() * self.p.risk_fraction) / risk_per_share)

        if size <= 0:
            return

        orders = self.sell_bracket(
            size=size,
            exectype=bt.Order.Market,
            stopprice=stop_price,
            stopexec=bt.Order.Stop,
        )
        self.pending_order = orders[0]
        self.stop_order = orders[2]

    def next(self):
        if not self._in_market_hours():
            return

        if self.position:
            if self.exit_order:
                return

            if ((self.position.size > 0 and self.data.close[0] < self.outside_bar_low) or
                    (self.position.size < 0 and self.data.close[0] > self.outside_bar_high)):
                if self.stop_order:
                    self.cancel(self.stop_order)
                self.exit_order = self.close()
            return

        if self.pending_order or self.exit_order:
            return

        outside_bar = (
            self.data.high[0] > self.data.high[-1] and
            self.data.low[0] < self.data.low[-1]
        )
        bullish_outside_bar = outside_bar and self.data.close[0] > self.data.open[0]
        bearish_outside_bar = outside_bar and self.data.close[0] < self.data.open[0]
        after_decline = self.data.close[0] < self.data.close[-self.p.trend_period]
        after_rally = self.data.close[0] > self.data.close[-self.p.trend_period]

        if bullish_outside_bar and after_decline:
            self._submit_long_bracket()
        elif bearish_outside_bar and after_rally:
            self._submit_short_bracket()

    def notify_order(self, order):
        terminal = (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        )

        if order is self.pending_order and order.status in terminal:
            self.pending_order = None

        if order is self.stop_order and order.status in terminal:
            self.stop_order = None

        if order is self.exit_order and order.status in terminal:
            self.exit_order = None
