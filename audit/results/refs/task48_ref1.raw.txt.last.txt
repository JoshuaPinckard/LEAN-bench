import backtrader as bt


class BullishDivergenceTSI(bt.Strategy):
    def __init__(self):
        momentum = self.data.close - self.data.close(-1)
        abs_momentum = abs(momentum)

        fast_momentum = bt.indicators.EMA(momentum, period=25)
        slow_momentum = bt.indicators.EMA(fast_momentum, period=13)
        fast_abs_momentum = bt.indicators.EMA(abs_momentum, period=25)
        slow_abs_momentum = bt.indicators.EMA(fast_abs_momentum, period=13)

        raw_tsi = bt.DivByZero(slow_momentum, slow_abs_momentum, zero=0.0) * 100.0
        self.tsi = bt.indicators.EMA(raw_tsi, period=2)

        self._pending_orders = []

    def next(self):
        if self.position or self._pending_orders:
            return

        if self.data.low[0] < self.data.low[-1] and self.tsi[0] > self.tsi[-1]:
            stop_level = self.data.low[0]

            entry = self.buy(transmit=False)
            size = entry.created.size

            stop_loss = self.sell(
                size=size,
                exectype=bt.Order.Stop,
                price=stop_level,
                parent=entry,
                transmit=False,
            )
            end_of_day_exit = self.sell(
                size=size,
                exectype=bt.Order.Close,
                parent=entry,
                transmit=True,
            )

            self._pending_orders = [entry, stop_loss, end_of_day_exit]

    def notify_order(self, order):
        if order.status in (
            bt.Order.Completed,
            bt.Order.Canceled,
            bt.Order.Margin,
            bt.Order.Rejected,
            bt.Order.Expired,
        ):
            if order in self._pending_orders:
                self._pending_orders.remove(order)
