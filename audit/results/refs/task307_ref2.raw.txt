import backtrader as bt


class AdxMomentumStrategy(bt.Strategy):
    params = (
        ("AdxPeriod", 14),
        ("AdxThreshold", 25.0),
        ("MomentumPeriod", 5),
        ("ATRMultiplier", 2.0),
        ("AtrPeriod", 14),
    )

    def __init__(self):
        self.adx = bt.ind.ADX(period=self.p.AdxPeriod)
        self.momentum = bt.ind.Momentum(period=self.p.MomentumPeriod)
        self.crossover = bt.ind.CrossOver(self.momentum, 0.0)
        self.atr = bt.ind.ATR(period=self.p.AtrPeriod)

        self.order = None
        self.entry_price = None
        self.stop_price = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            if self.position.size > 0:
                if self.stop_price is None:
                    self.entry_price = order.executed.price
                    self.stop_price = (
                        self.entry_price - self.p.ATRMultiplier * self.atr[0]
                    )
            elif self.position.size < 0:
                if self.stop_price is None:
                    self.entry_price = order.executed.price
                    self.stop_price = (
                        self.entry_price + self.p.ATRMultiplier * self.atr[0]
                    )
            else:
                self.entry_price = None
                self.stop_price = None
        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            pass

        self.order = None

    def next(self):
        if self.order:
            return

        if self.position.size > 0:
            if self.crossover[0] < 0 or (
                self.stop_price is not None
                and self.data.low[0] <= self.stop_price
            ):
                self.order = self.close()
        elif self.position.size < 0:
            if self.crossover[0] > 0 or (
                self.stop_price is not None
                and self.data.high[0] >= self.stop_price
            ):
                self.order = self.close()
        else:
            if self.adx[0] > self.p.AdxThreshold:
                if self.crossover[0] > 0:
                    self.order = self.buy()
                elif self.crossover[0] < 0:
                    self.order = self.sell()
