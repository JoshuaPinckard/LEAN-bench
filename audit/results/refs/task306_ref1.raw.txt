```python
import backtrader as bt
import math
import numpy as np
from hmmlearn.hmm import GaussianHMM


class MacdHmmAtrStrategy(bt.Strategy):
    # Intraday AAPL 5m strategy: MACD(12,26,9) cross confirmed by a 2-state
    # Gaussian HMM regime transition, with an ATR(14)-based stop of width
    # ATR * k, k = base_k * (slow/fast) = 1.0 * (26/12).
    params = dict(
        macd_fast=12,
        macd_slow=26,
        macd_signal=9,
        atr_period=14,
        base_k=1.0,
        hmm_states=2,
        train_window=390,      # rolling window of last 390 log returns
        hmm_n_iter=100,
        hmm_random_state=42,
        stake=1,               # fixed 1-share sizing
    )

    def __init__(self):
        self.macd = bt.indicators.MACD(
            self.data,
            period_me1=self.p.macd_fast,
            period_me2=self.p.macd_slow,
            period_signal=self.p.macd_signal,
        )
        # +1 when MACD crosses above signal, -1 when it crosses below
        self.mcross = bt.indicators.CrossOver(self.macd.macd, self.macd.signal)
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)

        # k = base_k * (slow_period / fast_period)
        self.k = self.p.base_k * (self.p.macd_slow / self.p.macd_fast)

        # HMM / regime state
        self.returns = []          # close-to-close log returns
        self.last_date = None      # for session-open (daily) detection
        self.model = None
        self.bull_state = None
        self.bear_state = None
        self.prev_label = None     # previous bar's semantic label ('bull'/'bear')

        # order tracking
        self.entry_order = None    # pending market entry / exit order
        self.stop_order = None     # currently live ATR stop order

    # ---- HMM helpers -----------------------------------------------------
    def _fit_hmm(self):
        # Train on the last `train_window` returns STRICTLY before the current
        # bar (past data only -> no lookahead).
        w = self.p.train_window
        train = np.asarray(self.returns[-(w + 1):-1], dtype=float).reshape(-1, 1)
        try:
            model = GaussianHMM(
                n_components=self.p.hmm_states,
                covariance_type='diag',
                n_iter=self.p.hmm_n_iter,
                random_state=self.p.hmm_random_state,
            )
            model.fit(train)
            means = model.means_.flatten()
            self.model = model
            # bull = state with the higher mean return, bear = the lower
            self.bull_state = int(np.argmax(means))
            self.bear_state = int(np.argmin(means))
        except Exception:
            # keep the previously fitted model on failure
            pass

    def _infer_label(self):
        if self.model is None:
            return None
        w = self.p.train_window
        if len(self.returns) < w:
            return None
        # Viterbi decode over the last `w` returns (incl. current); current
        # hidden state is the last decoded state.
        seq = np.asarray(self.returns[-w:], dtype=float).reshape(-1, 1)
        try:
            states = self.model.predict(seq)
        except Exception:
            return None
        return 'bull' if int(states[-1]) == self.bull_state else 'bear'

    # ---- stop management -------------------------------------------------
    def _refresh_stop(self):
        dist = self.atr[0] * self.k
        price = self.data.close[0]
        if self.stop_order is not None:
            self.cancel(self.stop_order)
            self.stop_order = None
        if self.position.size > 0:
            self.stop_order = self.sell(
                size=self.position.size,
                exectype=bt.Order.Stop,
                price=price - dist,
            )
        elif self.position.size < 0:
            self.stop_order = self.buy(
                size=abs(self.position.size),
                exectype=bt.Order.Stop,
                price=price + dist,
            )

    # ---- shared per-bar core --------------------------------------------
    def _core(self, trading):
        # 1) accumulate close-to-close log returns (also during warm-up)
        if len(self) > 1 and self.data.close[-1] > 0:
            self.returns.append(math.log(self.data.close[0] / self.data.close[-1]))

        # 2) refit HMM once per day, at the session open, on past data only
        cur_date = self.data.datetime.date(0)
        new_session = (self.last_date is None) or (cur_date != self.last_date)
        self.last_date = cur_date
        if new_session and len(self.returns) >= self.p.train_window + 1:
            self._fit_hmm()

        # 3) infer current regime label and detect a transition on this bar
        label = self._infer_label()
        trans_bull = (label == 'bull') and (self.prev_label is not None) and (self.prev_label != 'bull')
        trans_bear = (label == 'bear') and (self.prev_label is not None) and (self.prev_label != 'bear')
        if label is not None:
            self.prev_label = label

        if not trading:
            return

        # MACD crossover on this bar
        cross_up = self.mcross[0] > 0
        cross_down = self.mcross[0] < 0

        bull_confirm = cross_up and trans_bull      # long entry / short exit
        bear_confirm = cross_down and trans_bear    # short entry / long exit

        # guard: wait for any pending entry/exit market order to resolve
        if self.entry_order is not None:
            return

        # 4) risk management: refresh the ATR stop each bar while in position
        if self.position:
            self._refresh_stop()

        if self.position.size > 0:              # long open
            if bear_confirm:                    # opposite confirmation -> exit
                if self.stop_order is not None:
                    self.cancel(self.stop_order)
                    self.stop_order = None
                self.entry_order = self.close()
        elif self.position.size < 0:            # short open
            if bull_confirm:                    # opposite confirmation -> exit
                if self.stop_order is not None:
                    self.cancel(self.stop_order)
                    self.stop_order = None
                self.entry_order = self.close()
        else:                                   # flat
            if bull_confirm:
                self.entry_order = self.buy(size=self.p.stake)
            elif bear_confirm:
                self.entry_order = self.sell(size=self.p.stake)

    def prenext(self):
        # accumulate returns / regime state during indicator warm-up (no trading)
        self._core(trading=False)

    def next(self):
        self._core(trading=True)

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return
        if order.status == order.Completed:
            if order is self.entry_order:
                self.entry_order = None
            if order is self.stop_order:
                self.stop_order = None
            return
        if order.status in (order.Canceled, order.Margin, order.Rejected):
            if order is self.entry_order:
                self.entry_order = None
            if order is self.stop_order:
                self.stop_order = None
            return
```
