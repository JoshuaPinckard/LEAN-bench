import backtrader as bt


class MeanReversionStrategy(bt.Strategy):
    params = (
        ("rsi_period", 14),
        ("sma_fast_period", 20),
        ("sma_slow_period", 50),
        ("boll_period", 20),
        ("boll_dev", 2.0),
        ("vol_period", 20),
        ("vol_threshold", 0.0),
        ("min_candles", 200),
        ("pct_capital", 0.01),
        ("sl_sd_mult", 1.5),
    )

    def __init__(self):
        self.order = None
        self.entry_price = None
        self.sl_price = None
        self.tp_price = None

        self.rsi = bt.ind.RSI(self.data.close, period=self.p.rsi_period)
        self.sma_fast = bt.ind.SMA(
            self.data.close, period=self.p.sma_fast_period
        )
        self.sma_slow = bt.ind.SMA(
            self.data.close, period=self.p.sma_slow_period
        )
        self.boll = bt.ind.BollingerBands(
            self.data.close, period=self.p.boll_period, devfactor=self.p.boll_dev
        )
        self.sma_vol = bt.ind.SMA(self.data.volume, period=self.p.vol_period)

    def notify_order(self, order):
        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            self.order = None
            if order.status == order.Completed:
                if self.position:
                    self.entry_price = order.executed.price
                    sd = (self.boll.lines.top[0] - self.boll.lines.mid[0]) / 2.0
                    if self.position.size > 0:
                        self.sl_price = (
                            self.entry_price - self.p.sl_sd_mult * sd
                        )
                        self.tp_price = self.boll.lines.mid[0]
                    elif self.position.size < 0:
                        self.sl_price = (
                            self.entry_price + self.p.sl_sd_mult * sd
                        )
                        self.tp_price = self.boll.lines.mid[0]
                else:
                    self.entry_price = None
                    self.sl_price = None
                    self.tp_price = None

    def next(self):
        if self.order:
            return

        if len(self) < self.p.min_candles:
            return

        if not self.position:
            if self.sma_vol[0] <= self.p.vol_threshold:
                return

            long_cond = (
                self.data.close[0] < self.boll.lines.bot[0]
                and self.rsi[0] < 30
                and self.sma_fast[0] > self.sma_slow[0]
            )

            short_cond = (
                self.data.close[0] > self.boll.lines.top[0]
                and self.rsi[0] > 70
                and self.sma_fast[0] < self.sma_slow[0]
            )

            if long_cond:
                capital = self.broker.getvalue()
                raw_size = int((capital * self.p.pct_capital) / self.data.close[0])
                size = (
                    max(1, raw_size)
                    if self.broker.getcash() >= self.data.close[0]
                    else 0
                )
                if size > 0:
                    self.order = self.buy(size=size)
            elif short_cond:
                capital = self.broker.getvalue()
                raw_size = int((capital * self.p.pct_capital) / self.data.close[0])
                size = (
                    max(1, raw_size)
                    if self.broker.getcash() >= self.data.close[0]
                    else 0
                )
                if size > 0:
                    self.order = self.sell(size=size)
        else:
            if self.position.size > 0:
                cross_mid = self.data.close[0] >= self.boll.lines.mid[0]
                rsi_overbought = self.rsi[0] > 70
                sl_hit = (self.sl_price is not None) and (
                    self.data.low[0] <= self.sl_price
                )
                tp_hit = (self.tp_price is not None) and (
                    self.data.high[0] >= self.tp_price
                )

                if cross_mid or rsi_overbought or sl_hit or tp_hit:
                    self.order = self.close()

            elif self.position.size < 0:
                cross_mid = self.data.close[0] <= self.boll.lines.mid[0]
                rsi_oversold = self.rsi[0] < 30
                sl_hit = (self.sl_price is not None) and (
                    self.data.high[0] >= self.sl_price
                )
                tp_hit = (self.tp_price is not None) and (
                    self.data.low[0] <= self.tp_price
                )

                if cross_mid or rsi_oversold or sl_hit or tp_hit:
                    self.order = self.close()
