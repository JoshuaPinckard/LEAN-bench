import backtrader as bt
import math


class OpeningRangeBreakoutStrategy(bt.Strategy):
    params = (
        ('risk_fraction', 0.01),
    )

    def __init__(self):
        self.ema9 = bt.indicators.EMA(self.data.close, period=9)
        self.ema21 = bt.indicators.EMA(self.data.close, period=21)

        self.order = None
        self.current_day = None
        self.orh = None
        self.orl = None
        self.range_complete = False
        self.long_breakout = False
        self.short_breakout = False
        self.cum_pv = 0.0
        self.cum_volume = 0.0
        self.vwap = None

    def prenext(self):
        self._process(allow_entries=False)

    def next(self):
        self._process(allow_entries=True)

    def _process(self, allow_entries):
        dt = self.data.datetime.datetime(0)
        day = dt.date()
        clock = dt.time()

        if day != self.current_day:
            self.current_day = day
            self.orh = None
            self.orl = None
            self.range_complete = False
            self.long_breakout = False
            self.short_breakout = False
            self.cum_pv = 0.0
            self.cum_volume = 0.0
            self.vwap = None

        typical_price = (self.data.high[0] + self.data.low[0] + self.data.close[0]) / 3.0
        self.cum_pv += typical_price * self.data.volume[0]
        self.cum_volume += self.data.volume[0]
        if self.cum_volume:
            self.vwap = self.cum_pv / self.cum_volume

        if clock.hour == 9 and clock.minute == 30:
            self.orh = self.data.high[0]
            self.orl = self.data.low[0]
            self.range_complete = True
            return

        if not self.range_complete or self.orh is None or self.orl is None:
            return

        if self.order:
            return

        if clock.hour == 15 and clock.minute == 30 and self.position:
            self.order = self.close(exectype=bt.Order.Close)
            return

        if clock.hour == 15 and clock.minute >= 45:
            return

        orb = self.orh - self.orl
        fib382 = self.orl + (0.382 * orb)
        fib618 = self.orl + (0.618 * orb)

        if self.position:
            if self.position.size > 0 and self.data.low[0] <= self.orl:
                self.order = self.close()
            elif self.position.size < 0 and self.data.high[0] >= self.orh:
                self.order = self.close()
            return

        if not allow_entries or self.vwap is None:
            return

        if self.data.high[0] > self.orh:
            self.long_breakout = True
        if self.data.low[0] < self.orl:
            self.short_breakout = True

        ema_up = self.ema9[0] > self.ema9[-1] and self.ema21[0] > self.ema21[-1]
        ema_down = self.ema9[0] < self.ema9[-1] and self.ema21[0] < self.ema21[-1]

        long_pullback = self.data.low[0] <= fib382 or self.data.low[0] <= fib618
        short_pullback = self.data.high[0] >= fib382 or self.data.high[0] >= fib618

        if self.long_breakout and long_pullback and ema_up and self.data.close[0] > self.vwap:
            risk_per_share = abs(self.data.close[0] - self.orl)
            if risk_per_share > 0:
                size = math.floor((self.broker.getvalue() * self.p.risk_fraction) / risk_per_share)
                if size > 0:
                    self.order = self.buy(size=size)

        elif self.short_breakout and short_pullback and ema_down and self.data.close[0] < self.vwap:
            risk_per_share = abs(self.orh - self.data.close[0])
            if risk_per_share > 0:
                size = math.floor((self.broker.getvalue() * self.p.risk_fraction) / risk_per_share)
                if size > 0:
                    self.order = self.sell(size=size)

    def notify_order(self, order):
        if order.status in (order.Completed, order.Canceled, order.Margin, order.Rejected):
            self.order = None
