import backtrader as bt


class MeanReversionSwingStrategy(bt.Strategy):
    def __init__(self):
        self.ema20 = bt.indicators.EMA(self.data.close, period=20)
        self.ema_mean = bt.indicators.SMA(self.ema20, period=20)
        self.ema_std = bt.indicators.StdDev(self.ema20, period=20)
        self.upper_band = self.ema_mean + 2 * self.ema_std
        self.lower_band = self.ema_mean - 2 * self.ema_std
        self.ema_mean_cross = bt.indicators.CrossOver(self.ema20, self.ema_mean)

        self.entry_order = None
        self.stop_order = None
        self.exit_order = None
        self.exit_requested = False

    def next(self):
        if self.exit_order or self.entry_order:
            return

        if self.position:
            if self.ema_mean_cross[0] != 0:
                if self.stop_order:
                    self.exit_requested = True
                    self.cancel(self.stop_order)
                else:
                    self.exit_order = self.close()
            return

        if self.stop_order:
            return

        if self.ema20[-1] < self.lower_band[-1] and self.ema20[0] > self.lower_band[0]:
            self.entry_order = self.buy()
        elif self.ema20[-1] > self.upper_band[-1] and self.ema20[0] < self.upper_band[0]:
            self.entry_order = self.sell()

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            if order is self.entry_order:
                self.entry_order = None
                if order.status == order.Completed:
                    if order.isbuy():
                        self.stop_order = self.sell(
                            size=abs(self.position.size),
                            exectype=bt.Order.Stop,
                            price=order.executed.price * 0.98,
                        )
                    else:
                        self.stop_order = self.buy(
                            size=abs(self.position.size),
                            exectype=bt.Order.Stop,
                            price=order.executed.price * 1.02,
                        )

            elif order is self.stop_order:
                self.stop_order = None
                if self.exit_requested:
                    self.exit_requested = False
                    if self.position:
                        self.exit_order = self.close()

            elif order is self.exit_order:
                self.exit_order = None
