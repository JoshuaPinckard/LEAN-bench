import backtrader as bt


class SmaCrossoverStrategy(bt.Strategy):
    def __init__(self):
        self.order = None
        self.fast_sma = bt.indicators.SimpleMovingAverage(self.data.close, period=10)
        self.slow_sma = bt.indicators.SimpleMovingAverage(self.data.close, period=50)
        self.signal = bt.indicators.If(self.fast_sma > self.slow_sma, 1, 0)
        self.signal_difference = bt.indicators.Difference(self.signal, period=1)

    def next(self):
        if self.order:
            return

        if self.signal_difference[0] > 0 and not self.position:
            self.order = self.buy()
        elif self.signal_difference[0] < 0 and self.position:
            self.order = self.close()

    def notify_order(self, order):
        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            self.order = None
