import backtrader as bt


class SupertrendStrategy(bt.Strategy):
    params = (
        ("atr_period", 32),
        ("atr_multiplier", 4.35),
        ("sma_period", 200),
    )

    def __init__(self):
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.hl2 = (self.data.high + self.data.low) / 2.0
        self.sma200 = bt.indicators.SMA(self.data.close, period=self.p.sma_period)

        self.order = None
        self.final_ub = None
        self.final_lb = None
        self.prev_trend = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        basic_ub = self.hl2[0] + self.p.atr_multiplier * self.atr[0]
        basic_lb = self.hl2[0] - self.p.atr_multiplier * self.atr[0]

        if self.final_ub is None:
            final_ub = basic_ub
            final_lb = basic_lb
            trend = 1 if self.data.close[0] >= final_lb else -1
        else:
            if basic_ub < self.final_ub or self.data.close[-1] > self.final_ub:
                final_ub = basic_ub
            else:
                final_ub = self.final_ub

            if basic_lb > self.final_lb or self.data.close[-1] < self.final_lb:
                final_lb = basic_lb
            else:
                final_lb = self.final_lb

            if self.prev_trend == 1:
                trend = -1 if self.data.close[0] < final_lb else 1
            else:
                trend = 1 if self.data.close[0] > final_ub else -1

        st_flip_up = (trend == 1) and (self.prev_trend == -1)
        st_flip_down = (trend == -1) and (self.prev_trend == 1)

        self.final_ub = final_ub
        self.final_lb = final_lb
        self.prev_trend = trend

        if self.order:
            return

        if not self.position:
            if st_flip_up and (self.data.close[0] > self.sma200[0]):
                cash = self.broker.getcash()
                price = self.data.close[0]
                comm_rate = 0.001
                size = int(cash / (price * (1.0 + comm_rate)))
                if size > 0:
                    self.order = self.buy(size=size)
        else:
            if st_flip_down:
                self.order = self.close()
