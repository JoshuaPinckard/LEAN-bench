import backtrader as bt


class ConsolidationBreakoutStrategy(bt.Strategy):
    params = dict(
        ema_period=8,
        entry_level=8.47,
        first_exit_level=9.08,
        stop_loss_level=8.02,
    )

    def __init__(self):
        self.ema = bt.indicators.ExponentialMovingAverage(
            self.data.close, period=self.p.ema_period
        )
        self.order = None
        self.first_exit_done = False
        self.trailing_active = False

    def notify_order(self, order):
        if order.status in (order.Completed, order.Canceled, order.Margin, order.Rejected):
            self.order = None

    def next(self):
        if self.order:
            return

        close = self.data.close[0]

        if not self.position:
            if close > self.p.entry_level:
                self.order = self.buy()
            return

        if close < self.p.stop_loss_level:
            self.order = self.close()
            self.first_exit_done = False
            self.trailing_active = False
            return

        if not self.first_exit_done:
            if close >= self.p.first_exit_level:
                half_size = self.position.size // 2
                if half_size > 0:
                    self.order = self.sell(size=half_size)
                self.first_exit_done = True
                self.trailing_active = True
            return

        if self.trailing_active and close < self.ema[0]:
            self.order = self.close()
            self.first_exit_done = False
            self.trailing_active = False
