import backtrader as bt
import math


class HurstMeanReversionStrategy(bt.Strategy):
    class HurstExponent(bt.Indicator):
        lines = ("hurst",)
        params = (("period", 100),)

        def __init__(self):
            self.addminperiod(self.p.period)

        def next(self):
            values = [self.data[-i] for i in range(self.p.period - 1, -1, -1)]
            mean = sum(values) / self.p.period
            cumulative = []
            total = 0.0

            for value in values:
                total += value - mean
                cumulative.append(total)

            rescaled_range = max(cumulative) - min(cumulative)
            variance = sum((value - mean) ** 2 for value in values) / self.p.period
            standard_deviation = math.sqrt(variance)

            if standard_deviation == 0.0 or rescaled_range == 0.0:
                self.lines.hurst[0] = float("nan")
            else:
                self.lines.hurst[0] = math.log(
                    rescaled_range / standard_deviation
                ) / math.log(self.p.period)

    def __init__(self):
        self.sma = bt.indicators.SMA(self.data.close, period=20)
        self.hurst = self.HurstExponent(self.data.close, period=100)
        self.order = None
        self.stop_order = None

    def next(self):
        if self.order:
            return

        if self.position:
            if self.position.size > 0:
                if self.data.close[0] > self.sma[0] or self.hurst[0] >= 0.5:
                    if self.stop_order:
                        self.cancel(self.stop_order)
                    self.order = self.close()
            else:
                if self.data.close[0] < self.sma[0] or self.hurst[0] >= 0.5:
                    if self.stop_order:
                        self.cancel(self.stop_order)
                    self.order = self.close()
            return

        if self.hurst[0] < 0.5:
            if self.data.close[0] < self.sma[0]:
                self.order = self.buy()
            elif self.data.close[0] > self.sma[0]:
                self.order = self.sell()

    def notify_order(self, order):
        if order.status in [order.Completed]:
            if order == self.order:
                if order.isbuy():
                    self.stop_order = self.sell(
                        exectype=bt.Order.Stop,
                        price=order.executed.price * 0.98,
                    )
                elif order.issell() and self.position.size < 0:
                    self.stop_order = self.buy(
                        exectype=bt.Order.Stop,
                        price=order.executed.price * 1.02,
                    )
                self.order = None
            elif order == self.stop_order:
                self.stop_order = None

        elif order.status in [
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            if order == self.order:
                self.order = None
            elif order == self.stop_order:
                self.stop_order = None
