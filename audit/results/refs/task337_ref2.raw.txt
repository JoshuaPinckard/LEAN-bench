import backtrader as bt


class IntradayAccumDistSMA(bt.Strategy):
    params = (
        ("sma_period", 20),
        ("stop_pct", 0.0),
    )

    def __init__(self):
        self.sma = bt.indicators.SMA(self.data.close, period=self.p.sma_period)
        self.accum_dist = bt.indicators.AccumDist(self.data)
        self.order = None
        self.stop_order = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order == self.stop_order:
            if order.status in [
                order.Completed,
                order.Canceled,
                order.Margin,
                order.Rejected,
            ]:
                self.stop_order = None

        if order == self.order:
            if order.status == order.Completed:
                if (
                    self.position.size != 0
                    and self.p.stop_pct > 0
                    and self.stop_order is None
                ):
                    if self.position.size > 0:
                        stop_price = order.executed.price * (
                            1.0 - self.p.stop_pct
                        )
                        self.stop_order = self.sell(
                            exectype=bt.Order.Stop, price=stop_price
                        )
                    elif self.position.size < 0:
                        stop_price = order.executed.price * (
                            1.0 + self.p.stop_pct
                        )
                        self.stop_order = self.buy(
                            exectype=bt.Order.Stop, price=stop_price
                        )
                self.order = None
            elif order.status in [order.Canceled, order.Margin, order.Rejected]:
                self.order = None

    def next(self):
        if self.order:
            return

        if not self.position:
            if (
                self.data.close[0] > self.sma[0]
                and self.accum_dist[0] > self.accum_dist[-1]
            ):
                self.order = self.buy()
            elif (
                self.data.close[0] < self.sma[0]
                and self.accum_dist[0] < self.accum_dist[-1]
            ):
                self.order = self.sell()
        else:
            if (
                self.position.size > 0
                and self.accum_dist[0] <= self.accum_dist[-1]
            ):
                if self.stop_order:
                    self.cancel(self.stop_order)
                self.order = self.close()
            elif (
                self.position.size < 0
                and self.accum_dist[0] >= self.accum_dist[-1]
            ):
                if self.stop_order:
                    self.cancel(self.stop_order)
                self.order = self.close()
