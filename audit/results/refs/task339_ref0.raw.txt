import backtrader as bt


class ADXRocBreakoutStrategy(bt.Strategy):
    params = (
        ('adx_period', 14),
        ('roc_period', 20),
        ('band_period', 20),
        ('band_dev', 2.0),
        ('stop_loss_pct', 0.02),
    )

    def __init__(self):
        self.adx = bt.indicators.ADX(self.data, period=self.p.adx_period)
        self.roc = bt.indicators.ROC(self.adx.adx, period=self.p.roc_period)
        self.roc_sma = bt.indicators.SMA(self.roc, period=self.p.band_period)
        self.roc_std = bt.indicators.StandardDeviation(self.roc, period=self.p.band_period)
        self.crossover = bt.indicators.CrossOver(self.roc, self.roc_sma)
        self.order = None

    def notify_order(self, order):
        if order.status in (order.Completed, order.Canceled, order.Margin, order.Rejected):
            self.order = None

    def next(self):
        if self.order:
            return

        upper_band = self.roc_sma[0] + self.p.band_dev * self.roc_std[0]
        lower_band = self.roc_sma[0] - self.p.band_dev * self.roc_std[0]

        if not self.position:
            if self.roc[0] > upper_band:
                self.order = self.buy()
            elif self.roc[0] < lower_band:
                self.order = self.sell()
        else:
            if self.position.size > 0:
                stop_price = self.position.price * (1.0 - self.p.stop_loss_pct)
                if self.data.close[0] <= stop_price:
                    self.order = self.close()
                elif self.crossover[0] != 0:
                    self.order = self.close()
            else:
                stop_price = self.position.price * (1.0 + self.p.stop_loss_pct)
                if self.data.close[0] >= stop_price:
                    self.order = self.close()
                elif self.crossover[0] != 0:
                    self.order = self.close()
