import backtrader as bt


class MacdVolumeAtrStrategy(bt.Strategy):
    def __init__(self):
        self.macd = bt.indicators.MACD(
            self.data.close,
            period_me1=12,
            period_me2=26,
            period_signal=9,
        )
        self.volume_average = bt.indicators.SMA(self.data.volume, period=20)
        self.atr = bt.indicators.ATR(self.data, period=14)

        self.pending_order = None
        self.stop_order = None
        self.canceling_stop = False
        self.reverse_direction = 0

    def _open_position(self, direction):
        self.pending_order = self.buy() if direction > 0 else self.sell()

    def _submit_close(self):
        self.pending_order = self.close()

    def _exit_for_reversal(self):
        if self.stop_order is not None:
            self.canceling_stop = True
            self.cancel(self.stop_order)
        else:
            self._submit_close()

    def next(self):
        if self.pending_order is not None or self.canceling_stop:
            return

        bullish_cross = (
            self.macd.macd[0] > self.macd.signal[0]
            and self.macd.macd[-1] <= self.macd.signal[-1]
        )
        bearish_cross = (
            self.macd.macd[0] < self.macd.signal[0]
            and self.macd.macd[-1] >= self.macd.signal[-1]
        )
        volume_spike = self.data.volume[0] >= 2 * self.volume_average[0]

        bullish_signal = bullish_cross and volume_spike
        bearish_signal = bearish_cross and volume_spike

        if self.position.size > 0:
            if bearish_signal:
                self.reverse_direction = -1
                self._exit_for_reversal()
        elif self.position.size < 0:
            if bullish_signal:
                self.reverse_direction = 1
                self._exit_for_reversal()
        elif bullish_signal:
            self._open_position(1)
        elif bearish_signal:
            self._open_position(-1)

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order is self.stop_order:
            if order.status == order.Completed:
                self.stop_order = None
                self.reverse_direction = 0
            elif order.status in [order.Canceled, order.Margin, order.Rejected]:
                self.stop_order = None
                if self.canceling_stop:
                    self.canceling_stop = False
                    self._submit_close()
            return

        if order is self.pending_order:
            self.pending_order = None

            if order.status != order.Completed:
                self.reverse_direction = 0
                return

            if self.position.size == 0:
                if self.reverse_direction:
                    direction = self.reverse_direction
                    self.reverse_direction = 0
                    self._open_position(direction)
                return

            atr_distance = 2 * self.atr[0]
            if self.position.size > 0:
                self.stop_order = self.sell(
                    exectype=bt.Order.Stop,
                    price=order.executed.price - atr_distance,
                )
            else:
                self.stop_order = self.buy(
                    exectype=bt.Order.Stop,
                    price=order.executed.price + atr_distance,
                )
