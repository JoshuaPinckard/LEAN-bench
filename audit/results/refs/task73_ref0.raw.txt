import backtrader as bt


class MultiTimeframeCandlestickAggregationStrategy(bt.Strategy):
    params = (
        ('timeframes', (120, 60, 30, 15)),
        ('risk_perc', 0.01),
        ('tp_atr_mult', 1.5),
        ('sl_atr_mult', 0.5),
        ('atr_period', 14),
        ('printlog', False),
    )

    def __init__(self):
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.base_width = self.p.timeframes[0] // 15
        self.entry_width = self.p.timeframes[-1] // 15
        self.order = None
        self.tp_price = None
        self.sl_price = None
        self._pending_atr = None

    def log(self, txt):
        dt = self.data.datetime.datetime(0)
        print('%s %s' % (dt.isoformat(), txt))

    def _candle(self, width, k):
        # k=0 is the `width`-bar candle ending at the current bar;
        # k=1 is the immediately preceding candle of the same width, etc.
        ago_end = k * width
        ago_start = ago_end + width - 1
        o = self.data.open[-ago_start]
        c = self.data.close[-ago_end]
        h = max(self.data.high[-a] for a in range(ago_end, ago_start + 1))
        lo = min(self.data.low[-a] for a in range(ago_end, ago_start + 1))
        return o, h, lo, c

    def _bullish_engulfing(self, prev, curr):
        po, ph, pl, pc = prev
        co, ch, cl, cc = curr
        return pc < po and cc > co and co <= pc and cc >= po

    def _bearish_engulfing(self, prev, curr):
        po, ph, pl, pc = prev
        co, ch, cl, cc = curr
        return pc > po and cc < co and co >= pc and cc <= po

    def _drill_down(self):
        widths = [tf // 15 for tf in self.p.timeframes]
        prev_top = self._candle(widths[0], 1)
        curr_top = self._candle(widths[0], 0)
        if not self._bullish_engulfing(prev_top, curr_top):
            return False
        counts = {self.p.timeframes[0]: 1}
        prev_count = 1
        for tf, w in zip(self.p.timeframes[1:], widths[1:]):
            n = widths[0] // w
            cnt = 0
            for k in range(n):
                prevc = self._candle(w, k + 1)
                currc = self._candle(w, k)
                if self._bullish_engulfing(prevc, currc):
                    cnt += 1
            counts[tf] = cnt
            if self.p.printlog:
                self.log('pattern instance counts by timeframe: %s' % counts)
            if cnt <= prev_count:
                return False
            prev_count = cnt
        return True

    def notify_order(self, order):
        if order.status not in (order.Completed, order.Canceled, order.Margin, order.Rejected):
            return
        if order.status == order.Completed:
            if order.isbuy():
                self.tp_price = order.executed.price + self.p.tp_atr_mult * self._pending_atr
                self.sl_price = order.executed.price - self.p.sl_atr_mult * self._pending_atr
            else:
                self.tp_price = None
                self.sl_price = None
        self.order = None

    def next(self):
        if self.order:
            return

        if self.position:
            if len(self) >= 2 * self.entry_width:
                prevc = self._candle(self.entry_width, 1)
                currc = self._candle(self.entry_width, 0)
                if self._bearish_engulfing(prevc, currc):
                    self.order = self.close()
                    return
            if self.tp_price is not None and self.data.high[0] >= self.tp_price:
                self.order = self.close()
                return
            if self.sl_price is not None and self.data.low[0] <= self.sl_price:
                self.order = self.close()
                return
            return

        if len(self) < 2 * self.base_width:
            return
        bar_idx = len(self) - 1
        if (bar_idx + 1) % self.base_width != 0:
            return

        if not self._drill_down():
            return

        atr = self.atr[0]
        if atr is None or atr <= 0:
            return

        capital = self.broker.getvalue()
        sl_dist = self.p.sl_atr_mult * atr
        size = (capital * self.p.risk_perc) / sl_dist
        if size <= 0:
            return

        self._pending_atr = atr
        self.order = self.buy(size=size)
