import backtrader as bt


class MeanReversionStrategy(bt.Strategy):
    params = (
        ('target_position_value', 100000.0),
    )

    def __init__(self):
        self.bb_signal = bt.indicators.BollingerBands(
            self.data.close, period=200, devfactor=2
        )
        self.bb_stop = bt.indicators.BollingerBands(
            self.data.close, period=200, devfactor=3
        )
        self.rsi = bt.indicators.RSI(self.data.close, period=14)
        self.atr = bt.indicators.ATR(self.data, period=14)
        self.order = None

    def notify_order(self, order):
        if order.status in (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            self.order = None

    def next(self):
        if self.order:
            return

        close = self.data.close[0]

        if self.position:
            if self.position.size > 0:
                if self.data.low[0] <= self.bb_stop.bot[0]:
                    self.order = self.close()
                elif (
                    self.data.close[-1] < self.bb_signal.mid[-1]
                    and close >= self.bb_signal.mid[0]
                ):
                    self.order = self.close()
            else:
                if self.data.high[0] >= self.bb_stop.top[0]:
                    self.order = self.close()
                elif (
                    self.data.close[-1] > self.bb_signal.mid[-1]
                    and close <= self.bb_signal.mid[0]
                ):
                    self.order = self.close()
            return

        capital = self.broker.getvalue()

        if close < self.bb_signal.bot[0] and self.rsi[0] < 35:
            distance_to_stop = abs(close - self.bb_stop.bot[0])
            size = min(
                self.p.target_position_value / close,
                (0.02 * capital) / max(distance_to_stop, self.atr[0]),
            )
            self.order = self.buy(size=size)

        elif close > self.bb_signal.top[0] and self.rsi[0] > 65:
            distance_to_stop = abs(self.bb_stop.top[0] - close)
            size = min(
                self.p.target_position_value / close,
                (0.02 * capital) / max(distance_to_stop, self.atr[0]),
            )
            self.order = self.sell(size=size)
