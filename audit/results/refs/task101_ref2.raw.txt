import backtrader as bt


class TrailingStopRunner(bt.Strategy):
    def __init__(self):
        self.atr = bt.indicators.ATR(self.data, period=14)
        self.macd = bt.indicators.MACDHisto(
            self.data.close, period_me1=12, period_me2=9, period_signal=26
        )
        self.macd_cross = bt.indicators.CrossOver(self.macd.histo, 0.0)
        self.sma = bt.indicators.SMA(self.data.close, period=9)
        self.sma_vol = bt.indicators.SMA(self.data.volume, period=20)

        self.order = None
        self.stop_loss = None
        self.entry_price = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            if order.isbuy():
                self.entry_price = order.executed.price
                self.stop_loss = self.entry_price - 2.0 * self.atr[0]
            elif order.issell():
                self.entry_price = None
                self.stop_loss = None
            self.order = None

        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        if self.order:
            return

        if not self.position:
            if (
                self.data.close[0] > self.sma[0]
                and self.macd_cross[0] > 0
                and self.data.volume[0] > self.sma_vol[0]
            ):
                self.order = self.buy()
        else:
            candidate_stop = self.data.high[0] - 2.0 * self.atr[0]
            if self.stop_loss is not None:
                self.stop_loss = max(self.stop_loss, candidate_stop)
            else:
                self.stop_loss = candidate_stop

            if self.data.low[0] <= self.stop_loss:
                self.order = self.close()
                return

            if self.macd_cross[0] < 0:
                self.order = self.close()
                return
