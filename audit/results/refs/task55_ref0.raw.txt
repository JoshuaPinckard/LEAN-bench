import backtrader as bt


class RSIBearishDivergenceLiquidityExhaustionStrategy(bt.Strategy):
    params = dict(
        rsi_period=14,
        sl_pct=0.01,             # stop-loss distance above the triggering local high
        rr_ratio=2.0,            # take-profit = rr_ratio x the stop-loss distance
        rsi_exit_level=50,       # alternative exit: close if RSI rises above this
        risk_pct=0.01,           # fraction of portfolio value risked per trade
        trend_filter_period=35,  # proxy "weekly" SMA on the H1 feed (~5 trading days x 7 bars/day)
    )

    def __init__(self):
        self.rsi = bt.indicators.RSI(self.data.close, period=self.p.rsi_period)
        self.trend_sma = bt.indicators.SMA(self.data.close, period=self.p.trend_filter_period)

        self.order = None
        self.sl_order = None
        self.tp_order = None

        self.last_confirmed_high = None
        self.last_confirmed_high_rsi = None
        self.last_confirmed_high_vol = None
        self.prev_confirmed_high = None
        self.prev_confirmed_high_rsi = None
        self.prev_confirmed_high_vol = None

        self.entry_stop_price = None
        self.entry_take_price = None

    def notify_order(self, order):
        if order.status == order.Completed:
            if order == self.order:
                self.order = None
                if order.issell():
                    size = abs(order.executed.size)
                    self.sl_order = self.buy(exectype=bt.Order.Stop, price=self.entry_stop_price, size=size)
                    self.tp_order = self.buy(exectype=bt.Order.Limit, price=self.entry_take_price, size=size)
            elif order == self.sl_order:
                self.sl_order = None
                if self.tp_order is not None and self.tp_order.alive():
                    self.cancel(self.tp_order)
                    self.tp_order = None
            elif order == self.tp_order:
                self.tp_order = None
                if self.sl_order is not None and self.sl_order.alive():
                    self.cancel(self.sl_order)
                    self.sl_order = None
        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            if order == self.order:
                self.order = None
            elif order == self.sl_order:
                self.sl_order = None
            elif order == self.tp_order:
                self.tp_order = None

    def next(self):
        if len(self.data) >= 3:
            if self.data.high[-1] > self.data.high[-2] and self.data.high[-1] > self.data.high[0]:
                new_high_price = self.data.high[-1]
                new_high_rsi = self.rsi[-1]
                new_high_vol = self.data.volume[-1]

                self.prev_confirmed_high = self.last_confirmed_high
                self.prev_confirmed_high_rsi = self.last_confirmed_high_rsi
                self.prev_confirmed_high_vol = self.last_confirmed_high_vol

                self.last_confirmed_high = new_high_price
                self.last_confirmed_high_rsi = new_high_rsi
                self.last_confirmed_high_vol = new_high_vol

                if (self.prev_confirmed_high is not None
                        and not self.position
                        and self.order is None):
                    new_high_above_prev = self.last_confirmed_high > self.prev_confirmed_high
                    rsi_divergence = self.last_confirmed_high_rsi < self.prev_confirmed_high_rsi
                    volume_exhaustion = self.last_confirmed_high_vol < self.prev_confirmed_high_vol
                    trend_not_up = self.data.close[0] < self.trend_sma[0]

                    if new_high_above_prev and rsi_divergence and volume_exhaustion and trend_not_up:
                        sl_price = self.last_confirmed_high * (1 + self.p.sl_pct)
                        risk_per_share = sl_price - self.last_confirmed_high
                        if risk_per_share > 0:
                            risk_dollars = self.broker.getvalue() * self.p.risk_pct
                            size = int(risk_dollars / risk_per_share)
                            if size > 0:
                                self.entry_stop_price = sl_price
                                self.entry_take_price = self.last_confirmed_high - risk_per_share * self.p.rr_ratio
                                self.order = self.sell(size=size)

        if self.position and self.order is None and self.rsi[0] > self.p.rsi_exit_level:
            if self.sl_order is not None and self.sl_order.alive():
                self.cancel(self.sl_order)
            if self.tp_order is not None and self.tp_order.alive():
                self.cancel(self.tp_order)
            self.order = self.close()
