import backtrader as bt


class MACDBollingerATRStrategy(bt.Strategy):
    params = (
        ("macd_fast", 12),
        ("macd_slow", 26),
        ("macd_signal", 9),
        ("bb_period", 20),
        ("bb_dev", 2.0),
        ("atr_period", 14),
        ("atr_mult", 2.0),
    )

    def __init__(self):
        self.macd = bt.indicators.MACD(
            self.data.close,
            period_me1=self.p.macd_fast,
            period_me2=self.p.macd_slow,
            period_signal=self.p.macd_signal,
        )
        self.macd_cross = bt.indicators.CrossOver(
            self.macd.macd, self.macd.signal
        )
        self.boll = bt.indicators.BollingerBands(
            self.data.close, period=self.p.bb_period, devfactor=self.p.bb_dev
        )
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)

        self.order = None
        self.stop_price = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status == order.Completed:
            if order.isbuy():
                if self.position.size > 0:
                    self.stop_price = (
                        order.executed.price - self.p.atr_mult * self.atr[0]
                    )
            elif order.issell():
                if self.position.size < 0:
                    self.stop_price = (
                        order.executed.price + self.p.atr_mult * self.atr[0]
                    )

            if self.position.size == 0:
                self.stop_price = None

        self.order = None

    def next(self):
        if self.order:
            return

        if not self.position:
            if (
                self.macd_cross[0] > 0
                and self.data.close[0] <= self.boll.lines.bot[0]
            ):
                self.order = self.buy()
            elif (
                self.macd_cross[0] < 0
                and self.data.close[0] >= self.boll.lines.top[0]
            ):
                self.order = self.sell()
        else:
            if self.position.size > 0:
                reach_mid = self.data.close[0] >= self.boll.lines.mid[0]
                stop_hit = (
                    self.stop_price is not None
                    and self.data.close[0] <= self.stop_price
                )
                if reach_mid or stop_hit:
                    self.order = self.close()
            elif self.position.size < 0:
                reach_mid = self.data.close[0] <= self.boll.lines.mid[0]
                stop_hit = (
                    self.stop_price is not None
                    and self.data.close[0] >= self.stop_price
                )
                if reach_mid or stop_hit:
                    self.order = self.close()
