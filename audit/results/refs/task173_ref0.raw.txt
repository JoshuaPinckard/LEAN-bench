import backtrader as bt


class GapMomentumStrategy(bt.Strategy):
    params = (
        ('atr_period', 14),
        ('vol_sma_period', 20),
        ('liquidity_threshold', 1000000),
        ('volatility_threshold', 0.01),
        ('gap_atr_mult', 1.5),
        ('vol_mult', 1.2),
        ('take_profit_pct', 0.02),
        ('stop_loss_pct', 0.01),
        ('time_exit_bars', 12),
        ('risk_pct', 0.01),
    )

    def __init__(self):
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.vol_sma = bt.indicators.SMA(self.data.volume, period=self.p.vol_sma_period)

        self.order = None
        self.entry_price = None
        self.entry_bar = None
        self.stop_price = None
        self.tp_price = None
        self.direction = None

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            if order.status == order.Completed:
                if self.entry_price is None:
                    self.entry_price = order.executed.price
                    self.entry_bar = len(self)
                    self.direction = 1 if order.isbuy() else -1
                    if self.direction == 1:
                        self.stop_price = self.entry_price * (1 - self.p.stop_loss_pct)
                        self.tp_price = self.entry_price * (1 + self.p.take_profit_pct)
                    else:
                        self.stop_price = self.entry_price * (1 + self.p.stop_loss_pct)
                        self.tp_price = self.entry_price * (1 - self.p.take_profit_pct)
                else:
                    self.entry_price = None
                    self.entry_bar = None
                    self.stop_price = None
                    self.tp_price = None
                    self.direction = None
            self.order = None

    def _enter(self, direction):
        price_estimate = self.data.close[0]
        stop_points = self.p.stop_loss_pct * price_estimate
        risk_amount = self.p.risk_pct * self.broker.getvalue()
        size = risk_amount / stop_points
        if direction == 1:
            self.order = self.buy(size=size)
        else:
            self.order = self.sell(size=size)

    def next(self):
        if self.order:
            return

        if self.position:
            bars_held = len(self) - self.entry_bar
            if self.direction == 1:
                exit_signal = (self.data.high[0] >= self.tp_price or
                                self.data.low[0] <= self.stop_price or
                                bars_held >= self.p.time_exit_bars)
            else:
                exit_signal = (self.data.low[0] <= self.tp_price or
                                self.data.high[0] >= self.stop_price or
                                bars_held >= self.p.time_exit_bars)
            if exit_signal:
                self.order = self.close()
            return

        if self.vol_sma[0] < self.p.liquidity_threshold:
            return
        if self.atr[0] / self.data.close[-1] < self.p.volatility_threshold:
            return

        gap = self.data.open[0] - self.data.close[-1]
        atr_val = self.atr[0]

        long_gap = gap >= self.p.gap_atr_mult * atr_val and self.data.open[0] > self.data.high[-1]
        short_gap = (-gap) >= self.p.gap_atr_mult * atr_val and self.data.open[0] < self.data.low[-1]

        if long_gap:
            momentum_ok = (self.data.volume[0] > self.p.vol_mult * self.vol_sma[0] and
                           self.data.close[0] > self.data.open[0])
            if momentum_ok:
                self._enter(1)
        elif short_gap:
            momentum_ok = (self.data.volume[0] > self.p.vol_mult * self.vol_sma[0] and
                           self.data.close[0] < self.data.open[0])
            if momentum_ok:
                self._enter(-1)
