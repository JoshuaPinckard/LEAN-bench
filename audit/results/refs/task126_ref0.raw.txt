import backtrader as bt


class DemaSuperTrendStrategy(bt.Strategy):
    params = dict(
        dema_period=200,
        atr_period=12,
        multiplier=3.0,
    )

    def __init__(self):
        self.dema = bt.indicators.DEMA(self.data.close, period=self.p.dema_period)
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)

        self.order = None

        self.prev_final_upperband = None
        self.prev_final_lowerband = None
        self.prev_supertrend = None
        self.prev_direction = None

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        high = self.data.high[0]
        low = self.data.low[0]
        close = self.data.close[0]
        atr = self.atr[0]

        basic_upperband = (high + low) / 2.0 + self.p.multiplier * atr
        basic_lowerband = (high + low) / 2.0 - self.p.multiplier * atr

        if self.prev_final_upperband is None:
            final_upperband = basic_upperband
            final_lowerband = basic_lowerband
        else:
            prev_close = self.data.close[-1]

            if basic_upperband < self.prev_final_upperband or prev_close > self.prev_final_upperband:
                final_upperband = basic_upperband
            else:
                final_upperband = self.prev_final_upperband

            if basic_lowerband > self.prev_final_lowerband or prev_close < self.prev_final_lowerband:
                final_lowerband = basic_lowerband
            else:
                final_lowerband = self.prev_final_lowerband

        if self.prev_supertrend is None:
            supertrend = final_upperband if close <= final_upperband else final_lowerband
        elif self.prev_supertrend == self.prev_final_upperband:
            supertrend = final_upperband if close <= final_upperband else final_lowerband
        else:
            supertrend = final_lowerband if close >= final_lowerband else final_upperband

        direction = close > supertrend

        buy_signal = (self.prev_direction is not None) and (not self.prev_direction) and direction
        sell_signal = (self.prev_direction is not None) and self.prev_direction and (not direction)

        if self.order is None:
            if not self.position:
                if buy_signal and close > self.dema[0]:
                    cash = self.broker.getcash()
                    size = int(cash / close)
                    if size > 0:
                        self.order = self.buy(size=size)
            else:
                if sell_signal:
                    self.order = self.close()

        self.prev_final_upperband = final_upperband
        self.prev_final_lowerband = final_lowerband
        self.prev_supertrend = supertrend
        self.prev_direction = direction
