import backtrader as bt


class TechnicalBreakoutStrategy(bt.Strategy):
    params = (
        ('risk_fraction', 0.01),
    )

    def __init__(self):
        self.sma200 = bt.indicators.SMA(self.data.close, period=200)
        self.ema8 = bt.indicators.EMA(self.data.close, period=8)
        self.ema20 = bt.indicators.EMA(self.data.close, period=20)

        self.bb = bt.indicators.BollingerBands(
            self.data.close, period=20, devfactor=2.0
        )
        self.bb_width = self.bb.top - self.bb.bot
        self.bb_width_sma = bt.indicators.SMA(self.bb_width, period=20)

        self.atr = bt.indicators.ATR(self.data, period=14)
        self.atr_sma = bt.indicators.SMA(self.atr, period=14)
        self.volume_sma = bt.indicators.SMA(self.data.volume, period=20)

        self.macd = bt.indicators.MACD(
            self.data.close,
            period_me1=12,
            period_me2=26,
            period_signal=9,
        )

        self.order = None
        self.entry_atr = None
        self.take_profit = None
        self.stop_loss = None

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            if order.status == order.Completed and order.isbuy():
                self.take_profit = order.executed.price + (3.0 * self.entry_atr)
                self.stop_loss = order.executed.price - (1.5 * self.entry_atr)
            self.order = None

    def next(self):
        if self.order:
            return

        if self.position:
            bearish_macd_cross = (
                self.macd.macd[-1] > self.macd.signal[-1]
                and self.macd.macd[0] < self.macd.signal[0]
            )

            if (
                self.data.close[0] >= self.take_profit
                or self.data.close[0] <= self.stop_loss
                or self.data.close[0] < self.ema20[0]
                or bearish_macd_cross
            ):
                self.order = self.close()
            return

        squeeze_yesterday = self.bb_width[-1] < self.bb_width_sma[-1]

        entry_signal = (
            self.data.close[0] > self.bb.top[0]
            and squeeze_yesterday
            and self.data.close[0] > self.sma200[0]
            and self.ema8[0] > self.ema20[0]
            and self.data.close[0] > self.ema8[0]
            and self.atr[0] > self.atr_sma[0]
            and self.data.volume[0] > (1.2 * self.volume_sma[0])
        )

        if entry_signal:
            self.entry_atr = self.atr[0]
            risk_per_share = 1.5 * self.entry_atr
            size = int((self.broker.getcash() * self.p.risk_fraction) / risk_per_share)

            if size > 0:
                self.order = self.buy(size=size)
