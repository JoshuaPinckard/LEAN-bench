import backtrader as bt


class IntradayMiddayReversal(bt.Strategy):
    params = (
        ("session_start", (9, 30)),
        ("session_end", (16, 0)),
        ("morning_start", (9, 30)),
        ("morning_end", (11, 30)),
        ("midday_start", (11, 30)),
        ("midday_end", (13, 30)),
        ("eod_liquidation_time", (15, 45)),
    )

    def __init__(self):
        self.sma = bt.indicators.SMA(self.data.close, period=20)
        self.rsi = bt.indicators.RSI(self.data.close, period=14)
        self.rsi_cross_up = bt.indicators.CrossUp(self.rsi, 50.0)
        self.rsi_cross_down = bt.indicators.CrossDown(self.rsi, 50.0)

        self.order = None
        self.order_role = None
        self.stop_price = None
        self.morning_trend = None
        self.current_date = None

    def _time_tuple(self):
        current_time = self.data.datetime.time(0)
        return current_time.hour, current_time.minute

    def _in_window(self, current_time, start, end):
        return start <= current_time <= end

    def notify_order(self, order):
        if order.status != order.Completed:
            if order.status in (order.Canceled, order.Margin, order.Rejected):
                if order is self.order:
                    self.order = None
                    self.order_role = None
            return

        if order is self.order:
            completed_role = self.order_role
            self.order = None
            self.order_role = None

            if completed_role == "entry":
                if self.position.size > 0:
                    self.stop_price = order.executed.price * (1.0 - 0.02)
                elif self.position.size < 0:
                    self.stop_price = order.executed.price * (1.0 + 0.02)
            elif completed_role == "exit":
                self.stop_price = None

    def next(self):
        bar_date = self.data.datetime.date(0)
        current_time = self._time_tuple()

        if bar_date != self.current_date:
            self.current_date = bar_date
            self.morning_trend = None

        if not self._in_window(
            current_time, self.p.session_start, self.p.session_end
        ):
            return

        if current_time == self.p.morning_end:
            if self.data.close[0] > self.sma[0] and self.sma[0] > self.sma[-4]:
                self.morning_trend = "up"
            elif self.data.close[0] < self.sma[0] and self.sma[0] < self.sma[-4]:
                self.morning_trend = "down"
            else:
                self.morning_trend = None

        if self.position:
            if self.order is not None:
                return

            stop_hit = (
                (self.position.size > 0 and self.data.low[0] <= self.stop_price)
                or (self.position.size < 0 and self.data.high[0] >= self.stop_price)
            )
            opposite_cross = (
                (self.position.size > 0 and self.rsi_cross_down[0])
                or (self.position.size < 0 and self.rsi_cross_up[0])
            )

            if (
                stop_hit
                or opposite_cross
                or current_time >= self.p.eod_liquidation_time
            ):
                self.order = self.close()
                self.order_role = "exit"
            return

        if self.order is not None:
            return

        if not self._in_window(
            current_time, self.p.midday_start, self.p.midday_end
        ):
            return

        if self.morning_trend == "up" and self.rsi_cross_down[0]:
            self.order = self.sell()
            self.order_role = "entry"
        elif self.morning_trend == "down" and self.rsi_cross_up[0]:
            self.order = self.buy()
            self.order_role = "entry"
