import backtrader as bt


class PeriodicRebalanceStrategy(bt.Strategy):
    params = (("start_date", None),)

    def __init__(self):
        self.pending_order = None
        self.phase = None
        self.started = False
        self.current_week = None
        self.weeks_elapsed = 0
        self.finished = False

    def _week_key(self):
        iso = self.data.datetime.date(0).isocalendar()
        return (iso[0], iso[1])

    def _is_selected_start_date(self):
        if self.p.start_date is None:
            return True
        return self.data.datetime.date(0).isoformat() == str(self.p.start_date)[:10]

    def _rebalance(self, final_exit=False):
        self.phase = "final_exit" if final_exit else "rebalance_close"
        self.pending_order = self.close()

    def next(self):
        if self.finished or self.pending_order is not None:
            return

        if not self.started:
            if self._is_selected_start_date():
                self.started = True
                self.current_week = self._week_key()
                self.phase = "initial_buy"
                self.pending_order = self.order_target_percent(target=1.0)
            return

        week = self._week_key()
        if week == self.current_week:
            return

        self.current_week = week
        self.weeks_elapsed += 1

        if self.weeks_elapsed >= 52:
            if self.position:
                self._rebalance(final_exit=True)
            else:
                self.finished = True
        elif self.weeks_elapsed % 12 == 0 and self.position:
            self._rebalance()

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return

        if order.status == order.Completed and self.phase == "rebalance_close":
            self.phase = "rebalance_buy"
            self.pending_order = self.order_target_percent(target=1.0)
            return

        if order.status == order.Completed and self.phase == "final_exit":
            self.finished = True

        if order.status in (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            self.pending_order = None
            self.phase = None
