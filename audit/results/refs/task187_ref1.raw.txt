import backtrader as bt


class SingleAssetPeriodicRebalance(bt.Strategy):
    # Single-asset periodic rebalancing to a constant 100% target weight.
    # Enter on the start date, force-rebalance (full close + immediate re-open)
    # every 12 weeks, and close out at the 52-week forward horizon.
    params = dict(
        rebalance_weeks=12,   # forced rebalancing period
        horizon_weeks=52,     # forward horizon after which the position is closed
        target_pct=0.99,      # practical "100%" weight (headroom for commission / next-open fill)
    )

    def __init__(self):
        self.order_pending = False
        self.finished = False
        self.start_date = None
        self.next_rebal = self.p.rebalance_weeks

    def _commrate(self):
        try:
            return self.broker.getcommissioninfo(self.data).p.commission
        except Exception:
            return 0.0

    def _open_full(self, close_existing):
        # Optionally fully close the current position, then (re)open a position
        # sized to the target weight of the current portfolio value. Doing the
        # close together with the (re)buy incurs a sell and a buy commission,
        # modelling rebalancing with transaction costs.
        if close_existing and self.position:
            self.close()
        price = self.data.close[0]
        comm = self._commrate()
        value = self.broker.getvalue()
        size = (value * self.p.target_pct) / (price * (1.0 + comm))
        if size > 0:
            self.buy(size=size)
            self.order_pending = True

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return
        if order.status in (order.Completed, order.Canceled, order.Margin, order.Rejected):
            self.order_pending = False

    def next(self):
        if self.finished:
            return

        dt = self.data.datetime.date(0)
        if self.start_date is None:
            self.start_date = dt
        weeks = (dt - self.start_date).days // 7

        if self.order_pending:
            return

        # Exit: close at the end of the 52-week horizon and lock in the result.
        if weeks >= self.p.horizon_weeks:
            if self.position:
                self.close()
                self.order_pending = True
            self.finished = True
            return

        # Entry: on the start date buy the asset with (target) 100% of the portfolio.
        if not self.position:
            self._open_full(close_existing=False)
            return

        # Rebalancing: every 12 weeks fully close and immediately re-open at 100%.
        if weeks >= self.next_rebal:
            self._open_full(close_existing=True)
            self.next_rebal += self.p.rebalance_weeks
            return
