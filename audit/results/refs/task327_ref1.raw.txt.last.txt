import backtrader as bt


class Strategy(bt.Strategy):
    params = (("short_enabled", True),)

    def __init__(self):
        body = abs(self.data.close - self.data.open)
        self.body_sma = bt.indicators.SimpleMovingAverage(body, period=20)
        self.volume_sma = bt.indicators.SimpleMovingAverage(
            self.data.volume, period=20
        )
        self.order = None

    def notify_order(self, order):
        if order.status in (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            self.order = None

    def next(self):
        if self.order:
            return

        body = abs(self.data.close[0] - self.data.open[0])
        long_signal = (
            self.data.close[0] > self.data.open[0]
            and body >= 1.5 * self.body_sma[0]
            and self.data.volume[0] >= 1.5 * self.volume_sma[0]
        )
        short_signal = (
            self.data.close[0] < self.data.open[0]
            and body >= 1.5 * self.body_sma[0]
            and self.data.volume[0] >= 1.5 * self.volume_sma[0]
        )

        if self.position.size > 0 and short_signal:
            self.order = self.close()
            return

        if self.position.size < 0 and long_signal:
            self.order = self.close()
            return

        dt = self.data.datetime.datetime(0)
        seconds = dt.hour * 3600 + dt.minute * 60 + dt.second
        trading_time = 7 * 3600 <= seconds <= 17 * 3600

        if not self.position and trading_time:
            if long_signal:
                self.order = self.buy()
            elif self.p.short_enabled and short_signal:
                self.order = self.sell()
