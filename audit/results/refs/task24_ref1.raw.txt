import backtrader as bt


class InstantTradingWithDeltaTriggers(bt.Strategy):
    params = (
        ("buy_position_delta", 0.02),
        ("sell_position_delta", 0.02),
    )

    def __init__(self):
        self.order = None

    def notify_order(self, order):
        if order.status in (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            self.order = None

    def next(self):
        if self.order:
            return

        close = self.data.close[0]
        previous_close = self.data.close[-1]

        if not self.position:
            if close > previous_close * (1 + self.p.buy_position_delta):
                cash = self.broker.getcash()
                size = cash / (close * 1.001)
                self.order = self.buy(size=size)
            elif close < previous_close * (1 - self.p.sell_position_delta):
                cash = self.broker.getcash()
                size = cash / close
                self.order = self.sell(size=size)
            return

        entry_price = self.position.price

        if self.position.size > 0:
            if (
                close <= entry_price * (1 - self.p.sell_position_delta)
                and close > entry_price
            ):
                self.order = self.close()
        else:
            if (
                close >= entry_price * (1 + self.p.buy_position_delta)
                and close < entry_price
            ):
                self.order = self.close()
