import backtrader as bt


class MeanReversionStrategy(bt.Strategy):
    params = (
        ("sma_period", 20),
        ("atr_period", 14),
        ("atr_mult", 2.0),
    )

    def __init__(self):
        self.sma = bt.indicators.SimpleMovingAverage(
            self.data.close, period=self.p.sma_period
        )
        self.atr = bt.indicators.AverageTrueRange(
            self.data, period=self.p.atr_period
        )

        self.order = None
        self.stop_order = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            if order == self.order:
                self.order = None
                if self.position.size > 0:
                    stop_price = (
                        order.executed.price - self.p.atr_mult * self.atr[0]
                    )
                    self.stop_order = self.sell(
                        exectype=bt.Order.Stop, price=stop_price
                    )
                elif self.position.size < 0:
                    stop_price = (
                        order.executed.price + self.p.atr_mult * self.atr[0]
                    )
                    self.stop_order = self.buy(
                        exectype=bt.Order.Stop, price=stop_price
                    )
            elif order == self.stop_order:
                self.stop_order = None

        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            if order == self.order:
                self.order = None
            if order == self.stop_order:
                self.stop_order = None

    def next(self):
        if self.order:
            return

        if not self.position:
            if self.data.close[0] < self.sma[0] - self.p.atr_mult * self.atr[0]:
                self.order = self.buy()
            elif (
                self.data.close[0] > self.sma[0] + self.p.atr_mult * self.atr[0]
            ):
                self.order = self.sell()
        else:
            if self.position.size > 0:
                if self.data.close[0] >= self.sma[0]:
                    if self.stop_order:
                        self.cancel(self.stop_order)
                    self.order = self.close()
            elif self.position.size < 0:
                if self.data.close[0] <= self.sma[0]:
                    if self.stop_order:
                        self.cancel(self.stop_order)
                    self.order = self.close()
