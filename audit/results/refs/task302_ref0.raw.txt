import backtrader as bt


class Supertrend(bt.Indicator):
    lines = ('supertrend', 'final_upper', 'final_lower')
    params = (
        ('period', 10),
        ('multiplier', 3.0),
    )

    def __init__(self):
        self.atr = bt.indicators.ATR(self.data, period=self.p.period)
        self.hl2 = (self.data.high + self.data.low) / 2.0
        self._initialized = False

    def next(self):
        basic_upper = self.hl2[0] + self.p.multiplier * self.atr[0]
        basic_lower = self.hl2[0] - self.p.multiplier * self.atr[0]

        if not self._initialized:
            self.lines.final_upper[0] = basic_upper
            self.lines.final_lower[0] = basic_lower
            self.lines.supertrend[0] = basic_upper
            self._initialized = True
            return

        prev_final_upper = self.lines.final_upper[-1]
        prev_final_lower = self.lines.final_lower[-1]
        prev_close = self.data.close[-1]

        if basic_upper < prev_final_upper or prev_close > prev_final_upper:
            final_upper = basic_upper
        else:
            final_upper = prev_final_upper

        if basic_lower > prev_final_lower or prev_close < prev_final_lower:
            final_lower = basic_lower
        else:
            final_lower = prev_final_lower

        self.lines.final_upper[0] = final_upper
        self.lines.final_lower[0] = final_lower

        prev_supertrend = self.lines.supertrend[-1]
        close = self.data.close[0]

        if prev_supertrend == prev_final_upper:
            if close <= final_upper:
                st = final_upper
            else:
                st = final_lower
        else:
            if close >= final_lower:
                st = final_lower
            else:
                st = final_upper

        self.lines.supertrend[0] = st


class SupertrendReversalStrategy(bt.Strategy):
    def __init__(self):
        self.supertrend = Supertrend(self.data, period=10, multiplier=3.0)
        self.pending_orders = []

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            if order in self.pending_orders:
                self.pending_orders.remove(order)

    def next(self):
        if self.pending_orders:
            return

        if len(self.supertrend.supertrend) < 2:
            return

        prev_bearish = self.supertrend.supertrend[-1] > self.data.close[-1]
        prev_bullish = self.supertrend.supertrend[-1] < self.data.close[-1]
        curr_bearish = self.supertrend.supertrend[0] > self.data.close[0]
        curr_bullish = self.supertrend.supertrend[0] < self.data.close[0]

        if prev_bearish and curr_bullish:
            if self.position.size < 0:
                self.pending_orders.append(self.close())
            self.pending_orders.append(self.buy())
        elif prev_bullish and curr_bearish:
            if self.position.size > 0:
                self.pending_orders.append(self.close())
            self.pending_orders.append(self.sell())
