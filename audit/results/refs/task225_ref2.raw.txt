import math
import backtrader as bt


class BreakoutMomentumStrategy(bt.Strategy):
    def __init__(self):
        self.sma200 = bt.indicators.SMA(self.data.close, period=200)
        self.ema20 = bt.indicators.EMA(self.data.close, period=20)
        self.ema50 = bt.indicators.EMA(self.data.close, period=50)

        self.bb = bt.indicators.BollingerBands(
            self.data.close, period=20, devfactor=2.0
        )
        self.bb_width = (self.bb.top - self.bb.bot) / self.bb.mid
        self.sma_bb_width = bt.indicators.SMA(self.bb_width, period=100)

        self.sma_vol_20 = bt.indicators.SMA(self.data.volume, period=20)
        self.sma_vol_50 = bt.indicators.SMA(self.data.volume, period=50)

        self.rsi = bt.indicators.RSI(self.data.close, period=14)
        self.atr = bt.indicators.ATR(self.data, period=14)

        self.crossover = bt.indicators.CrossOver(self.data.close, self.bb.top)

        self.order = None
        self.entry_price = None
        self.entry_atr = None
        self.initial_stop = None
        self.max_close = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            if order.isbuy():
                self.entry_price = order.executed.price
                self.entry_atr = self.atr[0]
                self.initial_stop = self.entry_price - 1.5 * self.entry_atr
                self.max_close = self.data.close[0]
            elif order.issell():
                self.entry_price = None
                self.entry_atr = None
                self.initial_stop = None
                self.max_close = None

        self.order = None

    def next(self):
        if self.order:
            return

        if not self.position:
            uptrend = (self.data.close[0] > self.sma200[0]) and (
                self.ema20[0] > self.ema50[0]
            )
            squeeze = self.bb_width[0] < 0.8 * self.sma_bb_width[0]
            rvol_spike = (self.sma_vol_20[0] > 0) and (
                self.data.volume[0] >= 5.0 * self.sma_vol_20[0]
            )
            breakout = self.crossover[0] > 0
            momentum = self.rsi[0] > 55
            liquidity = self.sma_vol_20[0] >= self.sma_vol_50[0] * 0.7

            if (
                uptrend
                and squeeze
                and rvol_spike
                and breakout
                and momentum
                and liquidity
            ):
                risk_per_share = 1.5 * self.atr[0]
                if risk_per_share > 0:
                    equity = self.broker.getvalue()
                    size = math.floor((0.02 * equity) / risk_per_share)
                    if size > 0:
                        self.order = self.buy(size=size)
        else:
            self.max_close = max(self.max_close, self.data.close[0])

            take_profit = self.data.close[0] >= (
                self.entry_price + 2.0 * self.entry_atr
            )
            stop_loss = self.data.close[0] <= self.initial_stop
            trailing_stop = self.data.close[0] <= (
                self.max_close - 2.0 * self.atr[0]
            )
            weakening = self.data.close[0] < self.ema20[0]

            if take_profit or stop_loss or trailing_stop or weakening:
                self.order = self.close()
