import backtrader as bt


class MeanReversionZScoreStrategy(bt.Strategy):
    params = (
        ('lookback', 20),
        ('half_life', 5),
        ('entry_threshold', 1.0),
        ('stoploss_factor', 1.5),
        ('narrow_band_rel_eps', 1e-4),
    )

    def __init__(self):
        self.bb = bt.indicators.BollingerBands(
            self.data.close, period=self.p.lookback, devfactor=2.0)
        self.alpha = 2.0 / (self.p.half_life + 1.0)
        self.ema_z = None
        self.order = None

    def notify_order(self, order):
        if order.status in (order.Completed, order.Canceled, order.Margin, order.Rejected):
            self.order = None

    def next(self):
        if self.order:
            return

        mid = self.bb.mid[0]
        top = self.bb.top[0]
        band_width = top - mid

        if abs(band_width) < self.p.narrow_band_rel_eps * self.data.close[0]:
            return

        z = 2.0 * (self.data.close[0] - mid) / band_width

        prev_ema = self.ema_z
        if prev_ema is None:
            self.ema_z = z
        else:
            self.ema_z = self.alpha * z + (1.0 - self.alpha) * prev_ema

        cross_up = prev_ema is not None and prev_ema < 0.0 and self.ema_z >= 0.0
        cross_down = prev_ema is not None and prev_ema > 0.0 and self.ema_z <= 0.0

        stoploss_threshold = self.p.entry_threshold * self.p.stoploss_factor

        if self.position:
            if self.position.size > 0:
                if z >= stoploss_threshold:
                    self.order = self.close()
                elif cross_up:
                    self.order = self.close()
            elif self.position.size < 0:
                if z <= -stoploss_threshold:
                    self.order = self.close()
                elif cross_down:
                    self.order = self.close()
        else:
            if z < -self.p.entry_threshold:
                size = int(self.broker.getcash() / self.data.close[0])
                if size > 0:
                    self.order = self.buy(size=size)
            elif z > self.p.entry_threshold:
                size = int(self.broker.getcash() / self.data.close[0])
                if size > 0:
                    self.order = self.sell(size=size)
