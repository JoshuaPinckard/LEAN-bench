import backtrader as bt


class MeanReversionStrategy(bt.Strategy):
    params = (
        ('max_shares', 100),
    )

    def __init__(self):
        self.bb = bt.indicators.BollingerBands(
            self.data.close, period=20, devfactor=2
        )
        self.volume_sma_20 = bt.indicators.SimpleMovingAverage(
            self.data.volume, period=20
        )
        self.volume_sma_50 = bt.indicators.SimpleMovingAverage(
            self.data.volume, period=50
        )
        self.rsi = bt.indicators.RSI(self.data.close, period=14)
        self.cross_mid = bt.indicators.CrossOver(
            self.data.close, self.bb.mid
        )

        self.order = None
        self.entry_bar = None
        self.day_start_value = None
        self.current_day = None
        self.daily_loss_limit_reached = False

    def notify_order(self, order):
        if order.status == order.Completed:
            if order.isbuy():
                self.entry_bar = len(self.data)
            elif order.issell():
                self.entry_bar = None

        if order.status in (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            self.order = None

    def next(self):
        bar_day = self.data.datetime.date(0)

        if self.current_day != bar_day:
            self.current_day = bar_day
            self.day_start_value = self.broker.getvalue()
            self.daily_loss_limit_reached = False

        if (
            self.day_start_value is not None
            and self.broker.getvalue()
            <= self.day_start_value * (1.0 - 0.005)
        ):
            self.daily_loss_limit_reached = True

        if self.order:
            return

        if self.position:
            held_bars = len(self.data) - self.entry_bar

            if self.cross_mid[0] > 0 or held_bars >= 3:
                self.order = self.close()
            return

        if self.daily_loss_limit_reached:
            return

        if (
            self.data.close[0] < self.bb.bot[0]
            and self.data.volume[0] > 1.5 * self.volume_sma_20[0]
            and self.rsi[0] < 30
            and self.volume_sma_50[0] > 1000000
