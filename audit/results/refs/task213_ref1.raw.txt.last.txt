import backtrader as bt


class DynamicCVaRStrategy(bt.Strategy):
    params = (
        ("rebalance", "daily"),
        ("min_volume", 0.0),
        ("risk_threshold", 0.02),
        ("c", 2.0),
    )

    def __init__(self):
        self.order = None
        self.ema20 = bt.indicators.EMA(self.data.close, period=20)
        self.ema100 = bt.indicators.EMA(self.data.close, period=100)
        self.macd = bt.indicators.MACD(self.data.close)
        self.atr = bt.indicators.ATR(self.data, period=14)
        self.volume_sma = bt.indicators.SMA(self.data.volume, period=20)
        self.ema_crossover = bt.indicators.CrossOver(self.ema20, self.ema100)
        self.previous_bullish = None

    def notify_order(self, order):
        if order.status in (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            self.order = None

    def next(self):
        if self.order:
            return

        bullish = self.ema20[0] > self.ema100[0] and self.macd.macd[0] > 0
        regime_changed = (
            self.previous_bullish is not None
            and bullish != self.previous_bullish
        )
        self.previous_bullish = bullish

        liquid = self.volume_sma[0] > self.p.min_volume

        if self.position and (regime_changed or True):
            self.order = self.close()
            return

        if bullish and liquid and not self.position:
            weight = min(
                1.0,
                self.p.risk_threshold
                / (self.p.c * self.atr[0] / self.data.close[0]),
            )
            self.order = self.order_target_percent(target=weight)
