import backtrader as bt


class ReferenceStrategy(bt.Strategy):
    def __init__(self):
        self.ema20 = bt.indicators.EMA(self.data.close, period=20)
        self.ema50 = bt.indicators.EMA(self.data.close, period=50)
        self.rsi = bt.indicators.RSI(self.data.close, period=14)
        self.rsi_cross = bt.indicators.CrossOver(self.rsi, 50.0)

        self.order = None
        self.stop_order = None
        self.current_pattern_high = None
        self.current_stop_price = None

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            if self.order and order.ref == self.order.ref:
                self.order = None
            if self.stop_order and order.ref == self.stop_order.ref:
                self.stop_order = None

            if order.status == order.Completed:
                if self.position.size < 0 and self.stop_order is None and self.current_stop_price is not None:
                    self.stop_order = self.buy(
                        exectype=bt.Order.Stop,
                        price=self.current_stop_price
                    )

    def next(self):
        if self.order:
            return

        # Active Short Position Exits
        if self.position.size < 0:
            close_above_pattern_high = (
                self.current_pattern_high is not None and
                self.data.close[0] > self.current_pattern_high
            )

            rsi_cross_above = (self.rsi_cross[0] > 0)

            bullish_engulfing = (
                self.data.close[-1] < self.data.open[-1] and
                self.data.close[0] > self.data.open[0] and
                self.data.open[0] <= self.data.close[-1] and
                self.data.close[0] >= self.data.open[-1]
            )

            if close_above_pattern_high or rsi_cross_above or bullish_engulfing:
                if self.stop_order:
                    self.cancel(self.stop_order)
                    self.stop_order = None
                self.order = self.close()
            return

        # Entry logic
        # Trend confirmation (upswing filter): evaluated at bar -3 (prior to pattern start)
        upswing = (self.ema20[-3] > self.ema50[-3]) and (self.ema20[-3] > self.ema20[-4])

        # Three Black Crows logic across bars -2, -1, 0
        c1_bear = self.data.close[-2] < self.data.open[-2]
        c2_bear = self.data.close[-1] < self.data.open[-1]
        c3_bear = self.data.close[0] < self.data.open[0]
        three_bearish = c1_bear and c2_bear and c3_bear

        c2_open_in_c1 = (self.data.close[-2] <= self.data.open[-1] <= self.data.open[-2])
        c3_open_in_c2 = (self.data.close[-1] <= self.data.open[0] <= self.data.open[-1])
        opens_within = c2_open_in_c1 and c3_open_in_c2

        closes_lower = (self.data.close[-2] > self.data.close[-1] > self.data.close[0])

        three_black_crows = three_bearish and opens_within and closes_lower

        # Resistance filter: highest high of prior 20 bars (excluding current bar)
        resistance = max([self.data.high[-i] for i in range(1, 21)])
        near_resistance = self.data.close[0] >= 0.995 * resistance

        if upswing and three_black_crows and near_resistance:
            if self.position.size > 0:
                self.order = self.close()
            elif self.position.size == 0:
                pattern_high = max(self.data.high[0], self.data.high[-1], self.data.high[-2])
                self.current_pattern_high = pattern_high
                self.current_stop_price = pattern_high * 1.02
                self.order = self.sell()
