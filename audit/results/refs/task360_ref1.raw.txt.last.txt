import backtrader as bt


class BollingerATRBreakoutStrategy(bt.Strategy):
    params = (
        ('bb_period', 20),
        ('bb_devfactor', 2),
        ('atr_period', 14),
        ('atr_avg_lookback', 20),
        ('atr_stop_multiplier', 2.0),
    )

    def __init__(self):
        self.bbands = bt.indicators.BollingerBands(
            self.data.close,
            period=self.p.bb_period,
            devfactor=self.p.bb_devfactor,
        )
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.atr_sma = bt.indicators.SMA(
            self.atr,
            period=self.p.atr_avg_lookback,
        )

        self.order = None
        self.trailing_stop = None
        self.highest_close_since_entry = None
        self.lowest_close_since_entry = None

    def notify_order(self, order):
        if order.status in (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            if order.status == order.Completed and order.isbuy() and not self.position:
                self.highest_close_since_entry = self.data.close[0]
                self.lowest_close_since_entry = None
                self.trailing_stop = (
                    order.executed.price
                    - self.p.atr_stop_multiplier * self.atr[0]
                )
            elif order.status == order.Completed and order.issell() and not self.position:
                self.lowest_close_since_entry = self.data.close[0]
                self.highest_close_since_entry = None
                self.trailing_stop = (
                    order.executed.price
                    + self.p.atr_stop_multiplier * self.atr[0]
                )

            self.order = None

    def next(self):
        if self.order:
            return

        volatility_ok = self.atr[0] > self.atr_sma[0]
        upper_cross = (
            self.data.close[-1] <= self.bbands.top[-1]
            and self.data.close[0] > self.bbands.top[0]
        )
        lower_cross = (
            self.data.close[-1] >= self.bbands.bot[-1]
            and self.data.close[0] < self.bbands.bot[0]
        )

        if not self.position:
            if upper_cross and volatility_ok:
                self.order = self.buy()
            elif lower_cross and volatility_ok:
                self.order = self.sell()
            return

        if self.position.size > 0:
            self.highest_close_since_entry = max(
                self.highest_close_since_entry,
                self.data.close[0],
            )
            self.trailing_stop = max(
                self.trailing_stop,
                self.highest_close_since_entry
                - self.p.atr_stop_multiplier * self.atr[0],
            )

            if self.data.low[0] <= self.trailing_stop or (
                lower_cross and volatility_ok
            ):
                self.order = self.close()

        else:
            self.lowest_close_since_entry = min(
                self.lowest_close_since_entry,
                self.data.close[0],
            )
            self.trailing_stop = min(
                self.trailing_stop,
                self.lowest_close_since_entry
                + self.p.atr_stop_multiplier * self.atr[0],
            )

            if self.data.high[0] >= self.trailing_stop or (
                upper_cross and volatility_ok
            ):
                self.order = self.close()
