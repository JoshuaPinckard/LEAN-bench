import backtrader as bt


class GapMomentum(bt.Strategy):
    params = (
        ("time_exit_bars", 12),
    )

    def __init__(self):
        self.atr = bt.indicators.ATR(self.data, period=14)
        self.volume_sma = bt.indicators.SMA(self.data.volume, period=20)
        self.order = None
        self.entry_price = None
        self.entry_bar = None

    def notify_order(self, order):
        if order.status == order.Completed:
            if order.isbuy():
                self.entry_price = order.executed.price
                self.entry_bar = len(self)

        if order.status in (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            self.order = None

    def next(self):
        if self.order:
            return

        if self.position:
            if (
                self.data.close[0] >= self.entry_price * 1.02
                or self.data.close[0] <= self.entry_price * 0.99
                or len(self) - self.entry_bar >= self.p.time_exit_bars
            ):
                self.order = self.close()
            return

        liquid = self.volume_sma[0] >= 1000000
        volatile = self.atr[0] / self.data.close[-1] >= 0.01
        momentum = self.data.volume[0] > 1.2 * self.volume_sma[0]

        if not (liquid and volatile and momentum):
            return

        capital = self.broker.getvalue()
        stop_points = self.data.close[0] * 0.01
        size = (capital * 0.01) / stop_points

        long_gap = (
            self.data.open[0] - self.data.close[-1] >= 1.5 * self.atr[0]
            and self.data.open[0] > self.data.high[-1]
            and self.data.close[0] > self.data.open[0]
        )

        if long_gap:
            self.order = self.buy(size=size)
