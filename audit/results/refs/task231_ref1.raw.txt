import backtrader as bt
import math


class ImpliedVolatilityETFStrategy(bt.Strategy):
    params = (
        ("component_iv_period", 10),
        ("etf_iv_period", 5),
        ("calculated_iv_ma_period", 10),
        ("daily_bars", 78),
        ("annualization_days", 252),
    )

    def __init__(self):
        self.order = None
        self.stop_order = None
        self.target_order = None
        self.entry_bar = None

        self.log_return = bt.indicators.LogReturn(self.data.close, period=1)
        annualization = math.sqrt(self.p.annualization_days * self.p.daily_bars)

        self.component_iv = (
            bt.indicators.StandardDeviation(
                self.log_return, period=self.p.component_iv_period
            )
            * annualization
        )
        self.current_etf_iv = (
            bt.indicators.StandardDeviation(
                self.log_return, period=self.p.etf_iv_period
            )
            * annualization
        )

        self.calculated_iv = self.component_iv
        self.smoothed_calculated_iv = bt.indicators.SimpleMovingAverage(
            self.calculated_iv, period=self.p.calculated_iv_ma_period
        )

        self.average_bar_volume = bt.indicators.SimpleMovingAverage(
            self.data.volume, period=self.p.daily_bars
        )

    def notify_order(self, order):
        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            if order is self.order:
                if order.status == order.Completed and order.isbuy() != bool(self.position):
                    self.entry_bar = len(self)
                    entry_price = order.executed.price

                    if self.position.size > 0:
                        self.stop_order = self.sell(
                            size=self.position.size,
                            exectype=bt.Order.Stop,
                            price=entry_price * 0.98,
                        )
                        self.target_order = self.sell(
                            size=self.position.size,
                            exectype=bt.Order.Limit,
                            price=entry_price * 1.01,
                            oco=self.stop_order,
                        )
                    elif self.position.size < 0:
                        size = abs(self.position.size)
                        self.stop_order = self.buy(
                            size=size,
                            exectype=bt.Order.Stop,
                            price=entry_price * 1.02,
                        )
                        self.target_order = self.buy(
                            size=size,
                            exectype=bt.Order.Limit,
                            price=entry_price * 0.99,
                            oco=self.stop_order,
                        )

                self.order = None

            if order is self.stop_order:
                self.stop_order = None

            if order is self.target_order:
                self.target_order = None

            if not self.position:
                self.entry_bar = None

    def next(self):
        if self.order is not None:
            return

        calculated_iv = self.smoothed_calculated_iv[0]
        previous_calculated_iv = self.smoothed_calculated_iv[-1]
        current_iv = self.current_etf_iv[0]

        if calculated_iv <= 0 or previous_calculated_iv <= 0:
            return

        deviation = (current_iv - calculated_iv) / calculated_iv
        calculated_iv_change = abs(
            (calculated_iv - previous_calculated_iv) / previous_calculated_iv
        )

        liquid = self.data.volume[0] > self.average_bar_volume[0]
        vix_below_30 = current_iv <= 0.30
        calculated_iv_stable = calculated_iv_change <= 0.05

        if self.position:
            corrected = abs(deviation) < 0.05
            timed_out = (
                self.entry_bar is not None and len(self) - self.entry_bar >= 6
            )

            if corrected or timed_out:
                if self.stop_order is not None:
                    self.cancel(self.stop_order)
                if self.target_order is not None:
                    self.cancel(self.target_order)
                self.order = self.close()
            return

        if self.stop_order is not None or self.target_order is not None:
            return

        if not (liquid and vix_below_30 and calculated_iv_stable):
            return

        size = (self.broker.getvalue() * 0.10) / self.data.close[0]

        if deviation <= -0.15:
            self.order = self.buy(size=size)
        elif deviation >= 0.15:
            self.order = self.sell(size=size)
