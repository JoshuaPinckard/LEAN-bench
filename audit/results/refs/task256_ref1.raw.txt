import backtrader as bt
import math


class ManagedIncomeCapitalProtection(bt.Strategy):
    def __init__(self):
        self.sma50 = bt.indicators.SMA(self.data.close, period=50)
        self.sma200 = bt.indicators.SMA(self.data.close, period=200)
        self.atr = bt.indicators.ATR(self.data, period=14)

        self.order = None
        self.take_profit = None
        self.halted = False
        self.starting_equity = None

    def start(self):
        self.starting_equity = self.broker.getvalue()

    def notify_order(self, order):
        if order.status == order.Completed:
            if order.isbuy():
                self.take_profit = min(
                    order.executed.price * 1.05,
                    order.executed.price + 2.5 * self.atr[0],
                )
            elif order.issell():
                self.take_profit = None
            self.order = None

        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        if self.broker.getvalue() <= self.starting_equity * 0.90:
            self.halted = True
            if self.order is not None and self.order.isbuy():
                self.cancel(self.order)
            if self.position and self.order is None:
                self.order = self.close()
            return

        if self.order is not None:
            return

        if self.position:
            if self.data.close[0] >= self.take_profit:
                self.order = self.close()
            elif self.data.close[0] < self.sma50[0]:
                self.order = self.close()
            return

        if (
            not self.halted
            and self.data.close[0] > self.sma50[0]
            and self.data.close[0] > self.sma200[0]
            and self.atr[0] / self.data.close[0] <= 0.015
        ):
            size = math.floor(
                0.05 * self.broker.getvalue() / self.data.close[0]
            )
            if size > 0:
                self.order = self.buy(size=size)
