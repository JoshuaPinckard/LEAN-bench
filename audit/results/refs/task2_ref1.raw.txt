import backtrader as bt
import math


class SlippageAssessmentDaily(bt.Strategy):
    def next(self):
        high = self.data.high[0]
        low = self.data.low[0]
        volume = self.data.volume[0]
        total_volume = volume

        self.price_change_per_volume = (
            (math.log(high) - math.log(low)) / volume
        )

        self.volume_filter_passed = volume <= 0.05 * total_volume

        if self.volume_filter_passed:
            self.optimal_position_size = 1.0 / self.price_change_per_volume
            self.estimated_closing_slippage = (
                abs(self.position.size) * self.price_change_per_volume
            )
