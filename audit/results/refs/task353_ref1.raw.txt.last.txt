import backtrader as bt


class PulseWaveStrategy(bt.Strategy):
    params = (("vwap_period", 20),)

    def __init__(self):
        typical_price = (self.data.high + self.data.low + self.data.close) / 3.0
        self.vwap = (
            bt.ind.SumN(typical_price * self.data.volume, period=self.p.vwap_period)
            / bt.ind.SumN(self.data.volume, period=self.p.vwap_period)
        )

        self.macd = bt.ind.MACD(
            self.data.close,
            period_me1=12,
            period_me2=26,
            period_signal=9,
        )
        self.rsi = bt.ind.RSI(self.data.close, period=14)
        self.macd_crossover = bt.ind.CrossOver(self.macd.macd, self.macd.signal)

        self.order = None

    def next(self):
        if self.order:
            return

        if not self.position:
            if (
                self.data.close[0] > self.vwap[0]
                and self.macd_crossover[0] > 0
                and self.rsi[0] < 70
            ):
                self.order = self.buy()
        else:
            if (
                self.data.close[0] < self.vwap[0]
                and self.macd_crossover[0] < 0
                and self.rsi[0] > 30
            ):
                self.order = self.close()

    def notify_order(self, order):
        if order.status in (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            self.order = None
