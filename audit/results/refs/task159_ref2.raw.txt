import backtrader as bt


class LongTermMeanReversion(bt.Strategy):
    params = (
        ("bb_period", 120),
        ("bb_dev", 2.0),
        ("rsi_period", 14),
        ("rsi_low", 30),
        ("rsi_high", 70),
        ("vol_sma_period", 20),
        ("atr_period", 14),
        ("atr_sma_period", 50),
        ("atr_mult", 1.5),
        ("holding_bars", 20),
        ("risk_pct", 0.02),
        ("stop_atr_mult", 2.0),
    )

    def __init__(self):
        self.bb = bt.indicators.BollingerBands(
            self.data.close, period=self.p.bb_period, devfactor=self.p.bb_dev
        )
        self.rsi = bt.indicators.RSI(self.data.close, period=self.p.rsi_period)
        self.vol_sma = bt.indicators.SMA(self.data.volume, period=self.p.vol_sma_period)
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.atr_sma = bt.indicators.SMA(self.atr, period=self.p.atr_sma_period)

        self.close_cross_bb_bot = bt.indicators.CrossUp(self.data.close, self.bb.lines.bot)
        self.rsi_cross_low = bt.indicators.CrossUp(self.rsi, self.p.rsi_low)

        self.close_cross_bb_top = bt.indicators.CrossDown(self.data.close, self.bb.lines.top)
        self.rsi_cross_high = bt.indicators.CrossDown(self.rsi, self.p.rsi_high)

        self.order = None
        self.entry_bar = 0
        self.entry_price = 0.0
        self.stop_price = 0.0

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            if order.isbuy():
                if self.position.size > 0:
                    self.entry_bar = len(self)
                    self.entry_price = order.executed.price
                    self.stop_price = self.entry_price - (self.p.stop_atr_mult * self.atr[0])
            elif order.issell():
                if self.position.size < 0:
                    self.entry_bar = len(self)
                    self.entry_price = order.executed.price
                    self.stop_price = self.entry_price + (self.p.stop_atr_mult * self.atr[0])
            self.order = None
        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        if self.order:
            return

        vol_filter = self.data.volume[0] > self.vol_sma[0]
        volatility_filter = self.atr[0] < (self.atr_sma[0] * self.p.atr_mult)
        filters_pass = vol_filter and volatility_filter

        long_signal = (self.close_cross_bb_bot[0] > 0) or (self.rsi_cross_low[0] > 0)
        short_signal = (self.close_cross_bb_top[0] > 0) or (self.rsi_cross_high[0] > 0)

        if not self.position:
            if filters_pass:
                risk_amount = self.broker.getvalue() * self.p.risk_pct
                risk_per_share = self.p.stop_atr_mult * self.atr[0]
                if risk_per_share > 0:
                    size = int(risk_amount / risk_per_share)
                    if size > 0:
                        if long_signal:
                            self.order = self.buy(size=size)
                        elif short_signal:
                            self.order = self.sell(size=size)
        else:
            bars_held = len(self) - self.entry_bar
            if self.position.size > 0:
                if self.data.low[0] <= self.stop_price:
                    self.order = self.close()
                elif self.data.close[0] >= self.bb.lines.mid[0]:
                    self.order = self.close()
                elif bars_held >= self.p.holding_bars:
                    self.order = self.close()
                elif short_signal and filters_pass:
                    self.order = self.close()
            elif self.position.size < 0:
                if self.data.high[0] >= self.stop_price:
                    self.order = self.close()
                elif self.data.close[0] <= self.bb.lines.mid[0]:
                    self.order = self.close()
                elif bars_held >= self.p.holding_bars:
                    self.order = self.close()
                elif long_signal and filters_pass:
                    self.order = self.close()
