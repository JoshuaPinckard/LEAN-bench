import backtrader as bt
import collections


class SupertrendStrategy(bt.Strategy):
    params = (
        ('atr_period', 32),
        ('atr_multiplier', 4.35),
        ('sma_period', 200),
    )

    def __init__(self):
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.order = None
        self.final_upper = None
        self.final_lower = None
        self.supertrend = None
        self.direction = None
        self.previous_close = None
        self.closes = collections.deque(maxlen=self.p.sma_period)

    def notify_order(self, order):
        if order.status in (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            self.order = None

    def next(self):
        close = self.data.close[0]
        high = self.data.high[0]
        low = self.data.low[0]

        self.closes.append(close)
        sma = (
            sum(self.closes) / self.p.sma_period
            if len(self.closes) == self.p.sma_period
            else None
        )

        if len(self) <= self.p.atr_period or self.atr[0] != self.atr[0]:
            return

        basic_upper = ((high + low) / 2.0) + (self.p.atr_multiplier * self.atr[0])
        basic_lower = ((high + low) / 2.0) - (self.p.atr_multiplier * self.atr[0])

        if self.final_upper is None:
            self.final_upper = basic_upper
            self.final_lower = basic_lower
            self.supertrend = self.final_upper
            self.direction = 'down'
            self.previous_close = close
            return

        previous_final_upper = self.final_upper
        previous_final_lower = self.final_lower
        previous_supertrend = self.supertrend
        previous_direction = self.direction

        self.final_upper = (
            basic_upper
            if basic_upper < previous_final_upper
            or self.previous_close > previous_final_upper
            else previous_final_upper
        )
        self.final_lower = (
            basic_lower
            if basic_lower > previous_final_lower
            or self.previous_close < previous_final_lower
            else previous_final_lower
        )

        if previous_supertrend == previous_final_upper:
            self.supertrend = (
                self.final_lower if close > self.final_upper else self.final_upper
            )
        else:
            self.supertrend = (
                self.final_upper if close < self.final_lower else self.final_lower
            )

        self.direction = 'up' if self.supertrend == self.final_lower else 'down'
        self.previous_close = close

        if self.order:
            return

        if (
            not self.position
            and previous_direction == 'down'
            and self.direction == 'up'
            and sma is not None
            and close > sma
        ):
            self.order = self.order_target_percent(target=1.0)

        elif (
            self.position
            and previous_direction == 'up'
            and self.direction == 'down'
        ):
            self.order = self.close()
