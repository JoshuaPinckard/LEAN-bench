import math
import backtrader as bt


class OpeningRangeBreakout0DTE(bt.Strategy):
    def __init__(self):
        self.rsi = bt.indicators.RSI(self.data.close, period=14)
        self.order = None
        self.current_date = None
        self.or_highs = []
        self.or_lows = []
        self.range_high = None
        self.range_low = None
        self.range_ready = False

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return
        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            self.order = None

    def next(self):
        dt = self.data.datetime.date(0)
        t = self.data.datetime.time(0)

        # Reset state at the start of a new trading session
        if self.current_date != dt:
            self.current_date = dt
            self.or_highs = []
            self.or_lows = []
            self.range_high = None
            self.range_low = None
            self.range_ready = False

        # Define Opening Range during 9:30 to 10:10 EST
        if (t.hour == 9 and t.minute >= 30) or (t.hour == 10 and t.minute <= 10):
            self.or_highs.append(self.data.high[0])
            self.or_lows.append(self.data.low[0])
        elif (t.hour == 10 and t.minute > 10) or (t.hour > 10):
            if not self.range_ready and len(self.or_highs) > 0:
                self.range_high = max(self.or_highs)
                self.range_low = min(self.or_lows)
                self.range_ready = True

        # End of day mandatory exit (0DTE filter)
        is_eod = (t.hour == 15 and t.minute >= 55) or (t.hour >= 16)
        if is_eod:
            if self.position:
                self.close()
            return

        # Position Exits (Stop-loss and Take-profit)
        if self.position:
            if self.position.size > 0:
                # Long Exit: TP when returning to initial range (close <= range_high), SL when low <= range_low
                if (
                    self.data.close[0] <= self.range_high
                    or self.data.low[0] <= self.range_low
                ):
                    self.close()
            elif self.position.size < 0:
                # Short Exit: TP when returning to initial range (close >= range_low), SL when high >= range_high
                if (
                    self.data.close[0] >= self.range_low
                    or self.data.high[0] >= self.range_high
                ):
                    self.close()
            return

        # Entry logic
        if not self.range_ready or self.order:
            return

        # Do not initiate new entries near session close
        if t.hour == 15 and t.minute >= 45:
            return

        rsi_val = self.rsi[0]
        vol_inc = self.data.volume[0] > self.data.volume[-1]
        close_val = self.data.close[0]

        capital = self.broker.getvalue()
        risk_pct = 0.01

        # Long Entry condition
        if close_val > self.range_high and vol_inc and (50.0 < rsi_val <= 70.0):
            risk_per_share = close_val - self.range_low
            if risk_per_share > 0:
                size = math.floor((capital * risk_pct) / risk_per_share)
                if size > 0:
                    self.order = self.buy(size=size)

        # Short Entry condition
        elif close_val < self.range_low and vol_inc and (30.0 <= rsi_val < 50.0):
            risk_per_share = self.range_high - close_val
            if risk_per_share > 0:
                size = math.floor((capital * risk_pct) / risk_per_share)
                if size > 0:
                    self.order = self.sell(size=size)
