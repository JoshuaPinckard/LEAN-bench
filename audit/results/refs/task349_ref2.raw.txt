import backtrader as bt


class DonchianBreakout(bt.Strategy):
    params = (
        ("channel_period", 20),
        ("vol_sma_period", 20),
        ("stop_pct", 0.02),
    )

    def __init__(self):
        self.donchian_high = bt.ind.Highest(
            self.data.high(-1), period=self.p.channel_period
        )
        self.donchian_low = bt.ind.Lowest(
            self.data.low(-1), period=self.p.channel_period
        )

        self.crossover_high = bt.ind.CrossOver(self.data.close, self.donchian_high)
        self.crossover_low = bt.ind.CrossOver(self.data.close, self.donchian_low)

        self.vol_sma = bt.ind.SMA(self.data.volume, period=self.p.vol_sma_period)

        self.order = None
        self.stop_order = None
        self.stop_price = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            if order == self.order:
                self.order = None
                if self.position.size > 0:
                    self.stop_order = self.sell(
                        exectype=bt.Order.Stop, price=self.stop_price
                    )
                elif self.position.size < 0:
                    self.stop_order = self.buy(
                        exectype=bt.Order.Stop, price=self.stop_price
                    )
            elif order == self.stop_order:
                self.stop_order = None

        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            if order == self.order:
                self.order = None
            if order == self.stop_order:
                self.stop_order = None

    def next(self):
        if self.order:
            return

        if not self.position:
            vol_confirm = (self.data.volume[0] > self.data.volume[-1]) and (
                self.data.volume[0] > self.vol_sma[0]
            )
            if self.crossover_high[0] > 0 and vol_confirm:
                self.stop_price = self.donchian_high[0] * (1.0 - self.p.stop_pct)
                self.order = self.buy()
            elif self.crossover_low[0] < 0 and vol_confirm:
                self.stop_price = self.donchian_low[0] * (1.0 + self.p.stop_pct)
                self.order = self.sell()
        else:
            if self.position.size > 0:
                if (
                    self.data.close[0] < self.donchian_high[0]
                    or self.data.volume[0] < self.vol_sma[0]
                ):
                    if self.stop_order:
                        self.cancel(self.stop_order)
                    self.order = self.close()
            elif self.position.size < 0:
                if (
                    self.data.close[0] > self.donchian_low[0]
                    or self.data.volume[0] < self.vol_sma[0]
                ):
                    if self.stop_order:
                        self.cancel(self.stop_order)
                    self.order = self.close()
