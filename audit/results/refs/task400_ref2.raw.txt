import backtrader as bt


class AdaptiveChannelStrategy(bt.Strategy):
    params = (
        ('lookback', 10),
        ('stop_mult', 2.0),
        ('fast_ema', 2),
        ('slow_ema', 30),
        ('atr_period', 14),
    )

    def __init__(self):
        self.fast_ema = bt.indicators.EMA(self.data.close, period=self.p.fast_ema)
        self.slow_ema = bt.indicators.EMA(self.data.close, period=self.p.slow_ema)

        highest = bt.indicators.Highest(self.data.high, period=self.p.lookback)
        lowest = bt.indicators.Lowest(self.data.low, period=self.p.lookback)
        self.upper_band = highest(-1)
        self.lower_band = lowest(-1)

        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)

        self.order = None
        self.stop_order = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order == self.stop_order:
            if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
                self.stop_order = None

        if order == self.order:
            if order.status == order.Completed:
                if self.position.size > 0 and not self.stop_order:
                    stop_price = order.executed.price - self.p.stop_mult * self.atr[0]
                    self.stop_order = self.sell(exectype=bt.Order.Stop, price=stop_price)
                elif self.position.size < 0 and not self.stop_order:
                    stop_price = order.executed.price + self.p.stop_mult * self.atr[0]
                    self.stop_order = self.buy(exectype=bt.Order.Stop, price=stop_price)
            self.order = None

    def next(self):
        if self.order:
            return

        long_cond = (self.data.close[0] > self.upper_band[0]) and (self.fast_ema[0] > self.slow_ema[0])
        short_cond = (self.data.close[0] < self.lower_band[0]) and (self.fast_ema[0] < self.slow_ema[0])

        if self.position.size > 0:
            if short_cond:
                if self.stop_order:
                    self.cancel(self.stop_order)
                self.order = self.close()
        elif self.position.size < 0:
            if long_cond:
                if self.stop_order:
                    self.cancel(self.stop_order)
                self.order = self.close()
        else:
            if long_cond:
                self.order = self.buy()
            elif short_cond:
                self.order = self.sell()
