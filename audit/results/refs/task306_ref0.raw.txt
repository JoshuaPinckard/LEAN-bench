# complete strategy class
import backtrader as bt
import math
import numpy as np
from hmmlearn.hmm import GaussianHMM


class MacdHmmAtrStrategy(bt.Strategy):
    params = dict(
        macd_fast=12,
        macd_slow=26,
        macd_signal=9,
        atr_period=14,
        hmm_window=390,      # last 390 bars of log returns
        hmm_states=2,        # 2-state Gaussian HMM
        base_k=1.0,
        size=1,              # fixed position size: 1 share
    )

    def __init__(self):
        # --- Indicators ---
        self.macd = bt.indicators.MACD(
            self.data.close,
            period_me1=self.p.macd_fast,
            period_me2=self.p.macd_slow,
            period_signal=self.p.macd_signal,
        )
        # +1 on MACD crossing above signal, -1 on crossing below
        self.mcross = bt.indicators.CrossOver(self.macd.macd, self.macd.signal)
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)

        # k = base_k * (slow_period / fast_period)  -> 1.0 * 26/12
        self.k = self.p.base_k * (float(self.p.macd_slow) / float(self.p.macd_fast))

        # --- HMM state ---
        self.hmm = None            # fitted model (refit once per day)
        self.bull_state = None     # internal hmmlearn index of higher-mean state
        self.prev_regime = None    # regime (bull=1/bear=0) at the previous bar
        self.cur_date = None       # session date currently in effect

        # --- Order tracking ---
        self.order = None          # pending entry / close order
        self.stop_order = None     # working ATR stop order

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return
        if order.status in (order.Completed, order.Canceled, order.Margin,
                            order.Rejected):
            if order == self.order:
                self.order = None
            if order == self.stop_order:
                self.stop_order = None

    # --- HMM helpers -----------------------------------------------------
    def _fit_hmm(self):
        # 390 log returns ending at the PREVIOUS bar (strictly past data)
        closes = self.data.close.get(ago=1, size=self.p.hmm_window + 1)
        if len(closes) < self.p.hmm_window + 1:
            return
        rets = np.diff(np.log(np.asarray(closes, dtype=float))).reshape(-1, 1)
        model = GaussianHMM(n_components=self.p.hmm_states,
                            n_iter=100, random_state=42)
        try:
            model.fit(rets)
        except Exception:
            return  # keep the previously fitted model on numerical failure
        means = model.means_.flatten()
        self.hmm = model
        self.bull_state = int(np.argmax(means))  # higher mean return = bull

    def _infer_regime(self):
        if self.hmm is None:
            return None
        # 390 log returns ending at the CURRENT bar
        closes = self.data.close.get(ago=0, size=self.p.hmm_window + 1)
        if len(closes) < self.p.hmm_window + 1:
            return None
        rets = np.diff(np.log(np.asarray(closes, dtype=float))).reshape(-1, 1)
        try:
            state = int(self.hmm.predict(rets)[-1])
        except Exception:
            return None
        return 1 if state == self.bull_state else 0  # 1=bull, 0=bear

    # --- Main loop -------------------------------------------------------
    def next(self):
        # Refit the HMM once per day at the session open (calendar-date change)
        dt = self.data.datetime.date(0)
        if dt != self.cur_date:
            self.cur_date = dt
            self._fit_hmm()

        regime = self._infer_regime()

        # Same-bar regime transitions
        into_bull = (regime == 1 and self.prev_regime == 0)
        into_bear = (regime == 0 and self.prev_regime == 1)

        cross_up = self.mcross[0] > 0
        cross_down = self.mcross[0] < 0

        # Advance regime memory for the next bar's transition detection
        if regime is not None:
            self.prev_regime = regime

        if self.order:
            return

        pos = self.position.size

        if pos == 0:
            # Entries
            if cross_up and into_bull:
                self.order = self.buy(size=self.p.size)
            elif cross_down and into_bear:
                self.order = self.sell(size=self.p.size)
            return

        # In a position: opposite-confirmation exits take priority
        if pos > 0 and cross_down and into_bear:
            if self.stop_order:
                self.cancel(self.stop_order)
                self.stop_order = None
            self.order = self.close()
            return
        if pos < 0 and cross_up and into_bull:
            if self.stop_order:
                self.cancel(self.stop_order)
                self.stop_order = None
            self.order = self.close()
            return

        # ATR-based stop-loss, recomputed and re-submitted each bar
        stop_dist = self.atr[0] * self.k
        if pos > 0:
            stop_price = self.data.close[0] - stop_dist
            if self.stop_order:
                self.cancel(self.stop_order)
            self.stop_order = self.sell(size=abs(pos),
                                        exectype=bt.Order.Stop,
                                        price=stop_price)
        else:  # pos < 0
            stop_price = self.data.close[0] + stop_dist
            if self.stop_order:
                self.cancel(self.stop_order)
            self.stop_order = self.buy(size=abs(pos),
                                       exectype=bt.Order.Stop,
                                       price=stop_price)
