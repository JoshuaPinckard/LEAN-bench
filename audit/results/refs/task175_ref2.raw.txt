import backtrader as bt


class CashSecuredPutProxy(bt.Strategy):
    def __init__(self):
        self.ema = bt.indicators.EMA(self.data.close, period=20)
        self.rsi = bt.indicators.RSI(self.data.close, period=14)
        self.atr = bt.indicators.ATR(self.data, period=14)

        self.order = None
        self.entry_bar = 0

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            if order.isbuy():
                self.entry_bar = len(self)

        self.order = None

    def next(self):
        if self.order:
            return

        if not self.position:
            cond1 = self.data.close[0] < self.ema[0]
            cond2 = self.rsi[0] < 40
            cond3 = (self.atr[0] / self.data.close[0]) < 0.04
            cond4 = (self.data.close[0] - self.data.close[-1]) / self.data.close[-1] >= -0.01

            if cond1 and cond2 and cond3 and cond4:
                entry_price_est = self.data.close[0]
                risk_per_share = min(1.0 * self.atr[0], 0.02 * entry_price_est)
                if risk_per_share > 0:
                    capital = self.broker.getvalue()
                    max_risk = capital * 0.01
                    size = int(max_risk / risk_per_share)
                    if size > 0:
                        self.order = self.buy(size=size)
        else:
            entry_price = self.position.price

            profit_target = entry_price + 0.5 * self.atr[0]
            profit_exit = (self.data.close[0] >= profit_target) or (self.data.close[0] >= self.ema[0])

            stop_distance = min(1.0 * self.atr[0], 0.02 * entry_price)
            stop_exit = self.data.close[0] <= (entry_price - stop_distance)

            bars_held = len(self) - self.entry_bar
            time_exit = bars_held >= 5

            if profit_exit or stop_exit or time_exit:
                self.order = self.close()
