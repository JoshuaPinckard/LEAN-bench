import backtrader as bt


class PreMarketGapRiskStrategy(bt.Strategy):
    params = dict(
        atr_period=14,
        sma_period=20,
        atr_filter_short=5,
        atr_filter_long=20,
        atr_filter_mult=1.5,
        gap_pct=0.02,
        stop_atr_mult=1.5,
        trail_trigger_atr_mult=1.0,
        trail_atr_mult=1.0,
        risk_pct=0.015,
    )

    def __init__(self):
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.sma = bt.indicators.SMA(self.data.close, period=self.p.sma_period)
        self.atr_avg_short = bt.indicators.SMA(self.atr, period=self.p.atr_filter_short)
        self.atr_avg_long = bt.indicators.SMA(self.atr, period=self.p.atr_filter_long)
        self.crossover = bt.indicators.CrossOver(self.data.close, self.sma)

        self.order = None
        self.entry_order_pending = False
        self.pending_atr = None

        self.entry_price = None
        self.atr_at_entry = None
        self.stop_price = None
        self.trail_active = False
        self.trail_extreme = None

    def notify_order(self, order):
        if order.status == order.Completed:
            if self.entry_order_pending:
                self.entry_price = order.executed.price
                self.atr_at_entry = self.pending_atr
                if order.isbuy():
                    self.stop_price = self.entry_price - self.p.stop_atr_mult * self.atr_at_entry
                else:
                    self.stop_price = self.entry_price + self.p.stop_atr_mult * self.atr_at_entry
                self.trail_active = False
                self.trail_extreme = self.entry_price
                self.entry_order_pending = False
            if not self.position:
                self.entry_price = None
                self.atr_at_entry = None
                self.stop_price = None
                self.trail_active = False
                self.trail_extreme = None
            self.order = None
        elif order.status in (order.Canceled, order.Margin, order.Rejected):
            self.entry_order_pending = False
            self.order = None

    def next(self):
        if self.order:
            return

        if self.position:
            self._manage_open_position()
            return

        vol_ok = self.atr_avg_short[0] <= self.p.atr_filter_mult * self.atr_avg_long[0]
        gap_down = self.data.open[0] <= self.data.close[-1] * (1.0 - self.p.gap_pct)
        gap_up = self.data.open[0] >= self.data.close[-1] * (1.0 + self.p.gap_pct)

        if self.crossover[0] > 0:
            trend_ok = self.data.close[0] > self.sma[0]
            if vol_ok and not gap_down and trend_ok:
                self._enter(is_long=True)
        elif self.crossover[0] < 0:
            trend_ok = self.data.close[0] < self.sma[0]
            if vol_ok and not gap_up and trend_ok:
                self._enter(is_long=False)

    def _enter(self, is_long):
        stop_distance = self.p.stop_atr_mult * self.atr[0]
        if stop_distance <= 0:
            return
        risk_amount = self.p.risk_pct * self.broker.getvalue()
        size = int(risk_amount / stop_distance)
        if size < 1:
            return
        self.pending_atr = self.atr[0]
        self.entry_order_pending = True
        if is_long:
            self.order = self.buy(size=size)
        else:
            self.order = self.sell(size=size)

    def _manage_open_position(self):
        atr = self.atr_at_entry
        is_long = self.position.size > 0

        if is_long:
            profit = self.data.close[0] - self.entry_price
            if not self.trail_active and profit >= self.p.trail_trigger_atr_mult * atr:
                self.trail_active = True
                self.trail_extreme = self.data.high[0]
                self.stop_price = self.trail_extreme - self.p.trail_atr_mult * atr
            elif self.trail_active and self.data.high[0] > self.trail_extreme:
                self.trail_extreme = self.data.high[0]
                self.stop_price = self.trail_extreme - self.p.trail_atr_mult * atr

            if self.data.low[0] <= self.stop_price:
                self.order = self.close()
        else:
            profit = self.entry_price - self.data.close[0]
            if not self.trail_active and profit >= self.p.trail_trigger_atr_mult * atr:
                self.trail_active = True
                self.trail_extreme = self.data.low[0]
                self.stop_price = self.trail_extreme + self.p.trail_atr_mult * atr
            elif self.trail_active and self.data.low[0] < self.trail_extreme:
                self.trail_extreme = self.data.low[0]
                self.stop_price = self.trail_extreme + self.p.trail_atr_mult * atr

            if self.data.high[0] >= self.stop_price:
                self.order = self.close()
