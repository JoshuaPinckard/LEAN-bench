import backtrader as bt


class ParabolicSarReversalStrategy(bt.Strategy):
    params = dict(
        af=0.02,
        afmax=0.2,
    )

    def __init__(self):
        self.sar = bt.indicators.ParabolicSAR(self.data, af=self.p.af, afmax=self.p.afmax)
        self.crossover = bt.indicators.CrossOver(self.data.close, self.sar)
        self.order = None

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        if self.order:
            return

        if self.crossover > 0:
            if self.position.size < 0:
                self.close()
            self.order = self.buy()
        elif self.crossover < 0:
            if self.position.size > 0:
                self.close()
            self.order = self.sell()
