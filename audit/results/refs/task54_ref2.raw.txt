import backtrader as bt


class MultiTimeframeRSI(bt.Strategy):
    params = (
        ('rsi_period', 21),
        ('rsi_low', 30),
        ('rsi_high', 70),
        ('stake', 100),
    )

    def __init__(self):
        self.order = None

        # Primary M1 RSI indicator
        self.rsi_m1 = bt.indicators.RSI(self.data0, period=self.p.rsi_period)

        # Multi-timeframe indicators: process M5 (datas[1]) and M15 (datas[2]) if provided
        if len(self.datas) > 1:
            self.rsi_m5 = bt.indicators.RSI(self.datas[1], period=self.p.rsi_period)
        else:
            self.rsi_m5 = None

        if len(self.datas) > 2:
            self.rsi_m15 = bt.indicators.RSI(self.datas[2], period=self.p.rsi_period)
        else:
            self.rsi_m15 = None

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        if self.order:
            return

        rsi = self.rsi_m1

        if not self.position:
            # Long entry: RSI falls below 30
            if rsi[-1] >= self.p.rsi_low and rsi[0] < self.p.rsi_low:
                self.order = self.buy(size=self.p.stake)
            # Short entry: RSI rises above 70
            elif rsi[-1] <= self.p.rsi_high and rsi[0] > self.p.rsi_high:
                self.order = self.sell(size=self.p.stake)
        else:
            # Long exit: RSI crosses above 70
            if self.position.size > 0:
                if rsi[-1] <= self.p.rsi_high and rsi[0] > self.p.rsi_high:
                    self.order = self.close()
            # Short exit: RSI crosses below 30
            elif self.position.size < 0:
                if rsi[-1] >= self.p.rsi_low and rsi[0] < self.p.rsi_low:
                    self.order = self.close()
