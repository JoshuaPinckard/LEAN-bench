import backtrader as bt


class MeanReversionZScore(bt.Strategy):
    params = (
        ('ma_period', 20),        # SMA period
        ('stddev_period', 20),    # StdDev period
        ('entry_threshold', 2.0), # Z-Score entry level (+/-)
        ('exit_threshold', 0.0),  # Z-Score exit level (near 0)
        ('stop_loss', 0.02),      # 2% hard stop-loss from entry
    )

    def __init__(self):
        self.sma = bt.indicators.SimpleMovingAverage(
            self.data.close, period=self.p.ma_period)
        self.stddev = bt.indicators.StandardDeviation(
            self.data.close, period=self.p.stddev_period)
        self.order = None
        self.entry_price = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return
        if order.status == order.Completed:
            # If a position is now open, record its fill (entry) price;
            # if the fill flattened us, clear the reference.
            if self.position:
                self.entry_price = order.executed.price
            else:
                self.entry_price = None
        if order.status in [order.Completed, order.Canceled,
                            order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        # Guard pending orders.
        if self.order:
            return

        # Avoid division by zero on a flat window.
        if self.stddev[0] == 0:
            return

        zscore = (self.data.close[0] - self.sma[0]) / self.stddev[0]

        if not self.position:
            # Entry: long below -2, short above +2.
            if zscore < -self.p.entry_threshold:
                self.order = self.buy()
            elif zscore > self.p.entry_threshold:
                self.order = self.sell()
        else:
            if self.entry_price is None:
                return

            if self.position.size > 0:
                # Long: 2% hard stop, else exit when Z returns near 0.
                stop_level = self.entry_price * (1.0 - self.p.stop_loss)
                stop_hit = self.data.low[0] <= stop_level
                if stop_hit or zscore >= self.p.exit_threshold:
                    self.order = self.close()
            elif self.position.size < 0:
                # Short: 2% hard stop, else cover when Z returns near 0.
                stop_level = self.entry_price * (1.0 + self.p.stop_loss)
                stop_hit = self.data.high[0] >= stop_level
                if stop_hit or zscore <= self.p.exit_threshold:
                    self.order = self.close()
