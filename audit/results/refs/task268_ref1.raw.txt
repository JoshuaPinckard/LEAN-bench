import backtrader as bt


class BullTrendBounceVolatilityFilter(bt.Strategy):
    def __init__(self):
        self.sma = bt.indicators.SMA(self.data.close, period=20)
        self.rsi = bt.indicators.RSI(self.data.close, period=14)
        self.bb = bt.indicators.BollingerBands(
            self.data.close, period=20, devfactor=2
        )

        self.entry_order = None
        self.lots = []
        self.exit_order_lots = {}
        self.forced_order_lots = {}

    def next(self):
        for lot in list(self.lots):
            if not lot["closing"] and len(self) - lot["entry_bar"] >= 5:
                lot["closing"] = True
                self.cancel(lot["take_profit_order"])
                self.cancel(lot["stop_loss_order"])
                forced_order = self.close(size=lot["size"])
                self.forced_order_lots[forced_order.ref] = lot

        current_time = self.data.datetime.time(0)
        in_session = (
            current_time >= __import__("datetime").time(10, 0)
            and current_time <= __import__("datetime").time(16, 0)
        )

        relative_distance = (
            (self.data.close[0] - self.sma[0]) / self.sma[0]
        )
        relative_bb_width = (
            (self.bb.top[0] - self.bb.bot[0]) / self.bb.mid[0]
        )

        entry_signal = (
            in_session
            and self.data.close[0] > self.sma[0]
            and self.rsi[-1] < 40
            and self.rsi[0] >= 40
            and 0.05 <= relative_distance <= 0.10
            and relative_bb_width < 0.02
        )

        if (
            entry_signal
            and self.entry_order is None
            and len(self.lots) < 3
        ):
            size = int(
                0.02 * self.broker.getvalue() / self.data.close[0]
            )
            if size > 0:
                self.entry_order = self.buy(
                    size=size,
                    exectype=bt.Order.Limit,
                    price=self.data.close[0] * (1 - 0.005),
                )

    def notify_order(self, order):
        if order.status in [
            order.Submitted,
            order.Accepted,
        ]:
            return

        if order.ref == getattr(self.entry_order, "ref", None):
            if order.status == order.Completed:
                entry_price = order.executed.price
                lot = {
                    "size": order.executed.size,
                    "entry_bar": len(self),
                    "closing": False,
                }
                take_profit = self.sell(
                    size=lot["size"],
                    exectype=bt.Order.Limit,
                    price=entry_price * 1.02,
                )
                stop_loss = self.sell(
                    size=lot["size"],
                    exectype=bt.Order.Stop,
                    price=entry_price * (1 - 0.015),
                    oco=take_profit,
                )
                lot["take_profit_order"] = take_profit
                lot["stop_loss_order"] = stop_loss
                self.lots.append(lot)

            if order.status in [
                order.Completed,
                order.Canceled,
                order.Margin,
                order.Rejected,
            ]:
                self.entry_order = None

        if order.ref in self.exit_order_lots:
            lot = self.exit_order_lots[order.ref]
            if order.status == order.Completed and lot in self.lots:
                self.lots.remove(lot)
            if order.status in [
                order.Completed,
                order.Canceled,
                order.Margin,
                order.Rejected,
            ]:
                self.exit_order_lots.pop(order.ref, None)

        for lot in self.lots:
            if order.ref == lot.get("take_profit_order", None).ref:
                self.exit_order_lots[order.ref] = lot
            if order.ref == lot.get("stop_loss_order", None).ref:
                self.exit_order_lots[order.ref] = lot

        if order.ref in self.forced_order_lots:
            lot = self.forced_order_lots[order.ref]
            if order.status == order.Completed and lot in self.lots:
                self.lots.remove(lot)
            if order.status in [
                order.Completed,
                order.Canceled,
                order.Margin,
                order.Rejected,
            ]:
                self.forced_order_lots.pop(order.ref, None)
