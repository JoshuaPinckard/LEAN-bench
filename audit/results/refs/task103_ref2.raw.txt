import backtrader as bt


class DirectionalMovementEMA200(bt.Strategy):
    """
    Reference implementation for 'Directional movement filtering by EMA200'.
    Performs statistical analysis of daily price movement relative to EMA200.
    """
    params = (
        ('ema_period', 200),
    )

    def __init__(self):
        self.ema200 = bt.indicators.EMA(self.data.close, period=self.p.ema_period)

        # Main statistical counters
        self.Totaldays = 0
        self.Totalup = 0
        self.Totaldown = 0
        self.Totaldojis = 0
        self.Totalabove = 0
        self.Totalbelow = 0

        # Sub-counters for Close > EMA200
        self.Upabove = 0
        self.Downabove = 0
        self.Dojisabove = 0

        # Sub-counters for Close < EMA200
        self.Upbelow = 0
        self.Downbelow = 0
        self.Dojisbelow = 0

    def next(self):
        close = self.data.close[0]
        open_ = self.data.open[0]
        ema = self.ema200[0]

        self.Totaldays += 1

        is_up = close > open_
        is_down = close < open_
        is_doji = close == open_

        if is_up:
            self.Totalup += 1
        elif is_down:
            self.Totaldown += 1
        elif is_doji:
            self.Totaldojis += 1

        if close > ema:
            self.Totalabove += 1
            if is_up:
                self.Upabove += 1
            elif is_down:
                self.Downabove += 1
            elif is_doji:
                self.Dojisabove += 1
        elif close < ema:
            self.Totalbelow += 1
            if is_up:
                self.Upbelow += 1
            elif is_down:
                self.Downbelow += 1
            elif is_doji:
                self.Dojisbelow += 1

    def stop(self):
        def pct(val, total):
            return (val / total * 100.0) if total > 0 else 0.0

        print("=== Statistical Analysis Report: Directional Movement Filtering by EMA200 ===")
        print(f"Total days analyzed: {self.Totaldays}")
        print(f"Total Up days: {self.Totalup} ({pct(self.Totalup, self.Totaldays):.2f}%)")
        print(f"Total Down days: {self.Totaldown} ({pct(self.Totaldown, self.Totaldays):.2f}%)")
        print(f"Total Doji days: {self.Totaldojis} ({pct(self.Totaldojis, self.Totaldays):.2f}%)")
        print(f"Total days Above EMA200: {self.Totalabove} ({pct(self.Totalabove, self.Totaldays):.2f}%)")
        print(f"Total days Below EMA200: {self.Totalbelow} ({pct(self.Totalbelow, self.Totaldays):.2f}%)")
        print("-" * 50)
        print("Days Above EMA200 breakdown:")
        print(f"  Up days: {self.Upabove} ({pct(self.Upabove, self.Totalabove):.2f}%)")
        print(f"  Down days: {self.Downabove} ({pct(self.Downabove, self.Totalabove):.2f}%)")
        print(f"  Doji days: {self.Dojisabove} ({pct(self.Dojisabove, self.Totalabove):.2f}%)")
        print("-" * 50)
        print("Days Below EMA200 breakdown:")
        print(f"  Up days: {self.Upbelow} ({pct(self.Upbelow, self.Totalbelow):.2f}%)")
        print(f"  Down days: {self.Downbelow} ({pct(self.Downbelow, self.Totalbelow):.2f}%)")
        print(f"  Doji days: {self.Dojisbelow} ({pct(self.Dojisbelow, self.Totalbelow):.2f}%)")
