import backtrader as bt


class DonchianBreakoutStrategy(bt.Strategy):
    def __init__(self):
        self.order = None

        highest = bt.indicators.Highest(self.data.high, period=20)
        lowest = bt.indicators.Lowest(self.data.low, period=20)

        self.upper = highest(-1)
        self.lower = lowest(-1)
        self.mid = (self.upper + self.lower) / 2.0

    def notify_order(self, order):
        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            self.order = None

    def next(self):
        if self.order:
            return

        if not self.position:
            if self.data.close[0] > self.upper[0]:
                self.order = self.buy()
            elif self.data.close[0] < self.lower[0]:
                self.order = self.sell()
        elif self.position.size > 0:
            if self.data.close[0] <= self.mid[0]:
                self.order = self.close()
        elif self.position.size < 0:
            if self.data.close[0] >= self.mid[0]:
                self.order = self.close()
