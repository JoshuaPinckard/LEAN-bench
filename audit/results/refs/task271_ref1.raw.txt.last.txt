import backtrader as bt
import math


class BreakoutStrategy(bt.Strategy):
    params = (
        ("bb_period", 7),
        ("bb_devfactor", 2),
        ("volume_fast_period", 30),
        ("volume_slow_period", 90),
        ("atr_period", 14),
    )

    def __init__(self):
        self.initial_capital = self.broker.getvalue()
        self.order = None
        self.exit_order = None
        self.stop_order = None
        self.limit_order = None

        self.volume_sma_30 = bt.indicators.SMA(
            self.data.volume, period=self.p.volume_fast_period
        )
        self.volume_sma_90 = bt.indicators.SMA(
            self.data.volume, period=self.p.volume_slow_period
        )
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.bollinger = bt.indicators.BollingerBands(
            self.data.close,
            period=self.p.bb_period,
            devfactor=self.p.bb_devfactor,
        )
        self.upper_cross = bt.indicators.CrossOver(
            self.data.close, self.bollinger.top
        )
        self.lower_cross = bt.indicators.CrossOver(
            self.bollinger.bot, self.data.close
        )

    def notify_order(self, order):
        if order.status in (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            if order is self.order:
                self.order = None
            if order is self.exit_order:
                self.exit_order = None
            if order is self.stop_order:
                self.stop_order = None
            if order is self.limit_order:
                self.limit_order = None

    def next(self):
        if self.position:
            if self.lower_cross[0] > 0 and self.exit_order is None:
                if self.stop_order is not None:
                    self.cancel(self.stop_order)
                if self.limit_order is not None:
                    self.cancel(self.limit_order)
                self.exit_order = self.close()
            return

        if self.order is not None or self.exit_order is not None:
            return

        if self.data.datetime.date(0).weekday() not in (0, 2, 4):
            return

        volume_ok = (
            self.data.volume[0] > self.volume_sma_30[0]
            and self.volume_sma_30[0] > self.volume_sma_90[0]
        )
        return_7d = self.data.close[0] / self.data.close[-7] - 1.0
        return_ok = -0.10 <= return_7d <= 0.20
        volatility_ok = self.atr[0] / self.data.close[0] > 0.01
        breakout_ok = self.upper_cross[0] > 0

        if volume_ok and return_ok and volatility_ok and breakout_ok:
            entry_price = self.data.close[0]
            size = math.floor(
                (self.initial_capital * 0.01) / (entry_price * 0.02)
            )

            if size > 0:
                bracket = self.buy_bracket(
                    size=size,
                    exectype=bt.Order.Market,
                    stopprice=entry_price * 0.98,
                    limitprice=entry_price * 1.05,
                )
                self.order, self.stop_order, self.limit_order = bracket
