import backtrader as bt


class SuperTrend(bt.Indicator):
    lines = ('supertrend', 'direction', 'up', 'dn')
    params = (
        ('period', 12),
        ('multiplier', 3.0),
    )

    def __init__(self):
        self.hl2 = (self.data.high + self.data.low) / 2.0
        self.atr = bt.indicators.ATR(self.data, period=self.p.period)
        self._initialized = False

    def next(self):
        basic_up = self.hl2[0] - self.p.multiplier * self.atr[0]
        basic_dn = self.hl2[0] + self.p.multiplier * self.atr[0]

        if not self._initialized:
            self.lines.up[0] = basic_up
            self.lines.dn[0] = basic_dn
            if self.data.close[0] < basic_up:
                self.lines.direction[0] = -1.0
                self.lines.supertrend[0] = basic_dn
            else:
                self.lines.direction[0] = 1.0
                self.lines.supertrend[0] = basic_up
            self._initialized = True
        else:
            prev_up = self.lines.up[-1]
            prev_dn = self.lines.dn[-1]
            prev_close = self.data.close[-1]
            prev_dir = self.lines.direction[-1]

            if prev_close > prev_up:
                up = max(basic_up, prev_up)
            else:
                up = basic_up

            if prev_close < prev_dn:
                dn = min(basic_dn, prev_dn)
            else:
                dn = basic_dn

            if prev_dir == -1.0 and self.data.close[0] > prev_dn:
                direction = 1.0
            elif prev_dir == 1.0 and self.data.close[0] < prev_up:
                direction = -1.0
            else:
                direction = prev_dir

            self.lines.up[0] = up
            self.lines.dn[0] = dn
            self.lines.direction[0] = direction
            self.lines.supertrend[0] = up if direction == 1.0 else dn


class DEMASuperTrendStrategy(bt.Strategy):
    params = (
        ('dema_period', 200),
        ('st_period', 12),
        ('st_multiplier', 3.0),
    )

    def __init__(self):
        self.dema = bt.indicators.DEMA(self.data.close, period=self.p.dema_period)
        self.supertrend = SuperTrend(
            self.data,
            period=self.p.st_period,
            multiplier=self.p.st_multiplier
        )
        self.order = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        if self.order:
            return

        supertrend_buy_signal = (
            self.supertrend.direction[0] == 1.0 and self.supertrend.direction[-1] == -1.0
        )
        supertrend_sell_signal = (
            self.supertrend.direction[0] == -1.0 and self.supertrend.direction[-1] == 1.0
        )

        if not self.position:
            close_above_dema = self.data.close[0] > self.dema[0]
            if close_above_dema and supertrend_buy_signal:
                cash = self.broker.getcash()
                price = self.data.close[0]
                size = int(cash / (price * 1.001))
                if size > 0:
                    self.order = self.buy(size=size)
        else:
            if supertrend_sell_signal:
                self.order = self.close()
