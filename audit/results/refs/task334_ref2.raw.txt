import backtrader as bt


class BBVolatilityZScoreStrategy(bt.Strategy):
    def __init__(self):
        # Indicators
        self.bb = bt.indicators.BollingerBands(
            self.data.close, period=20, devfactor=2.0
        )
        self.bandwidth = (self.bb.top - self.bb.bot) / self.bb.mid
        self.bw_sma = bt.indicators.SMA(self.bandwidth, period=20)
        self.bw_std = bt.indicators.StdDev(self.bandwidth, period=20)
        self.bw_z = (self.bandwidth - self.bw_sma) / self.bw_std
        self.atr = bt.indicators.ATR(self.data, period=14)

        # Order guards
        self.order = None
        self.stop_order = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            if order == self.order:
                self.order = None
                if self.position:
                    entry_price = order.executed.price
                    atr_val = self.atr[0]
                    if self.position.size > 0:
                        stop_price = entry_price - 2.0 * atr_val
                        self.stop_order = self.sell(
                            exectype=bt.Order.Stop, price=stop_price
                        )
                    elif self.position.size < 0:
                        stop_price = entry_price + 2.0 * atr_val
                        self.stop_order = self.buy(
                            exectype=bt.Order.Stop, price=stop_price
                        )
                else:
                    if self.stop_order:
                        self.cancel(self.stop_order)
                        self.stop_order = None

            elif order == self.stop_order:
                self.stop_order = None
                if self.order:
                    self.cancel(self.order)
                    self.order = None

        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            if order == self.order:
                self.order = None
            if order == self.stop_order:
                self.stop_order = None

    def next(self):
        if self.order is not None:
            return

        if self.position:
            cross_zero = (self.bw_z[-1] > 0 and self.bw_z[0] <= 0) or (
                self.bw_z[-1] < 0 and self.bw_z[0] >= 0
            )
            abs_small = abs(self.bw_z[0]) < 0.5

            if cross_zero or abs_small:
                if self.stop_order:
                    self.cancel(self.stop_order)
                    self.stop_order = None
                self.order = self.close()
        else:
            long_cond = (
                self.bw_z[-1] > 2.0
                and self.bw_z[0] <= 2.0
                and self.bandwidth[0] < self.bandwidth[-1]
            )
            short_cond = (
                self.bw_z[-1] < -2.0
                and self.bw_z[0] >= -2.0
                and self.bandwidth[0] > self.bandwidth[-1]
            )

            if long_cond:
                self.order = self.buy()
            elif short_cond:
                self.order = self.sell()
