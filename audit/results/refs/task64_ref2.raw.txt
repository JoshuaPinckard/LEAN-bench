import backtrader as bt


class EMARibbonStrategy(bt.Strategy):
    params = (
        ("p_fast1", 8),
        ("p_fast2", 13),
        ("p_fast3", 21),
        ("p_slow1", 34),
        ("p_slow2", 55),
        ("stop_loss", 0.02),
        ("take_profit", 0.04),
        ("capital_pct", 0.02),
    )

    def __init__(self):
        self.ema8 = bt.indicators.EMA(self.data.close, period=self.p.p_fast1)
        self.ema13 = bt.indicators.EMA(self.data.close, period=self.p.p_fast2)
        self.ema21 = bt.indicators.EMA(self.data.close, period=self.p.p_fast3)
        self.ema34 = bt.indicators.EMA(self.data.close, period=self.p.p_slow1)
        self.ema55 = bt.indicators.EMA(self.data.close, period=self.p.p_slow2)

        self.entry_order = None
        self.sl_order = None
        self.tp_order = None
        self.exit_order = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            if order == self.entry_order:
                self.entry_order = None
                entry_price = order.executed.price
                size = order.executed.size
                sl_price = entry_price * (1.0 - self.p.stop_loss)
                tp_price = entry_price * (1.0 + self.p.take_profit)
                self.sl_order = self.sell(
                    exectype=bt.Order.Stop, price=sl_price, size=size
                )
                self.tp_order = self.sell(
                    exectype=bt.Order.Limit,
                    price=tp_price,
                    size=size,
                    oco=self.sl_order,
                )
            elif order == self.sl_order:
                self.sl_order = None
                if self.tp_order:
                    self.cancel(self.tp_order)
                    self.tp_order = None
            elif order == self.tp_order:
                self.tp_order = None
                if self.sl_order:
                    self.cancel(self.sl_order)
                    self.sl_order = None
            elif order == self.exit_order:
                self.exit_order = None

        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            if order == self.entry_order:
                self.entry_order = None
            elif order == self.sl_order:
                self.sl_order = None
            elif order == self.tp_order:
                self.tp_order = None
            elif order == self.exit_order:
                self.exit_order = None

    def next(self):
        is_ordered = (
            self.ema8[0]
            > self.ema13[0]
            > self.ema21[0]
            > self.ema34[0]
            > self.ema55[0]
        )
        was_ordered = (
            self.ema8[-1]
            > self.ema13[-1]
            > self.ema21[-1]
            > self.ema34[-1]
            > self.ema55[-1]
        )

        if not self.position:
            if self.entry_order is None:
                if is_ordered and not was_ordered:
                    value = self.broker.getvalue()
                    target_value = value * self.p.capital_pct
                    price = self.data.close[0]
                    size = int(target_value / price)
                    if size < 1:
                        size = 1 if self.broker.getcash() >= price else 0
                    if size > 0:
                        self.entry_order = self.buy(size=size)
        else:
            if not is_ordered and self.exit_order is None:
                if self.sl_order:
                    self.cancel(self.sl_order)
                    self.sl_order = None
                if self.tp_order:
                    self.cancel(self.tp_order)
                    self.tp_order = None
                self.exit_order = self.close()
