import backtrader as bt


class ConsolidationBreakout(bt.Strategy):
    params = (
        ("ema_period", 8),
        ("entry_level", 8.47),
        ("first_exit_level", 9.08),
        ("stop_loss_level", 8.02),
    )

    def __init__(self):
        self.ema = bt.indicators.EMA(self.data.close, period=self.p.ema_period)
        self.order = None
        self.first_stage_exited = False

    def notify_order(self, order):
        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            self.order = None

        if order.status == order.Completed:
            if order.isexit() or order.issell():
                if self.position.size > 0:
                    self.first_stage_exited = True
                else:
                    self.first_stage_exited = False

    def next(self):
        if self.order:
            return

        if not self.position:
            self.first_stage_exited = False
            if self.data.close[0] > self.p.entry_level:
                self.order = self.buy()
        else:
            # Stop-loss check based on candle closing price
            if self.data.close[0] < self.p.stop_loss_level:
                self.order = self.close()
                return

            # Two-stage exit strategy
            if not self.first_stage_exited:
                if self.data.close[0] >= self.p.first_exit_level:
                    half_size = self.position.size / 2.0
                    self.order = self.sell(size=half_size)
            else:
                if self.data.close[0] < self.ema[0]:
                    self.order = self.close()
