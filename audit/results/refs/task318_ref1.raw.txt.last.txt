import backtrader as bt


class ThreeBlackCrowsStrategy(bt.Strategy):
    def __init__(self):
        self.ema20 = bt.indicators.EMA(self.data.close, period=20)
        self.ema50 = bt.indicators.EMA(self.data.close, period=50)
        self.rsi = bt.indicators.RSI(self.data.close, period=14)
        self.resistance = bt.indicators.Highest(self.data.high(-1), period=20)

        self.entry_order = None
        self.exit_order = None
        self.stop_order = None
        self.pending_pattern_high = None
        self.pattern_high = None

    def notify_order(self, order):
        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            if order == self.entry_order:
                if order.status == order.Completed:
                    self.pattern_high = self.pending_pattern_high
                    self.stop_order = self.buy(
                        exectype=bt.Order.Stop,
                        price=self.pattern_high * 1.02,
                    )
                self.entry_order = None
                self.pending_pattern_high = None

            elif order == self.exit_order:
                self.exit_order = None
                if order.status == order.Completed:
                    self.pattern_high = None

            elif order == self.stop_order:
                self.stop_order = None
                if order.status == order.Completed:
                    self.pattern_high = None

    def next(self):
        if self.position.size < 0 and self.exit_order is None:
            rsi_crosses_above_50 = self.rsi[-1] <= 50 and self.rsi[0] > 50

            bullish_engulfing = (
                self.data.close[-1] < self.data.open[-1]
                and self.data.close[0] > self.data.open[0]
                and self.data.open[0] <= self.data.close[-1]
                and self.data.close[0] >= self.data.open[-1]
            )

            close_above_pattern_high = (
                self.pattern_high is not None
                and self.data.close[0] > self.pattern_high
            )

            if (
                close_above_pattern_high
                or rsi_crosses_above_50
                or bullish_engulfing
            ):
                if self.stop_order is not None:
                    self.cancel(self.stop_order)
                self.exit_order = self.close()
            return

        if (
            self.position.size != 0
            or self.entry_order is not None
            or self.exit_order is not None
        ):
            return

        upswing = (
            self.ema20[0] > self.ema50[0]
            and self.ema20[0] > self.ema20[-1]
        )

        bearish_1 = self.data.close[-2] < self.data.open[-2]
        bearish_2 = self.data.close[-1] < self.data.open[-1]
        bearish_3 = self.data.close[0] < self.data.open[0]

        open_2_in_body_1 = (
            min(self.data.open[-2], self.data.close[-2])
            <= self.data.open[-1]
            <= max(self.data.open[-2], self.data.close[-2])
        )
        open_3_in_body_2 = (
            min(self.data.open[-1], self.data.close[-1])
            <= self.data.open[0]
            <= max(self.data.open[-1], self.data.close[-1])
        )

        progressively_lower_closes = (
            self.data.close[-2] > self.data.close[-1] > self.data.close[0]
        )

        three_black_crows = (
            bearish_1
            and bearish_2
            and bearish_3
            and open_2_in_body_1
            and open_3_in_body_2
            and progressively_lower_closes
        )

        near_resistance = self.data.close[0] >= 0.995 * self.resistance[0]

        if upswing and three_black_crows and near_resistance:
            if self.position.size > 0:
                self.exit_order = self.close()
            else:
                self.pending_pattern_high = max(
                    self.data.high[-2],
                    self.data.high[-1],
                    self.data.high[0],
                )
                self.entry_order = self.sell()
