import backtrader as bt


class MeanReversionROC21Strategy(bt.Strategy):
    def __init__(self):
        self.roc21 = bt.indicators.RateOfChange(self.data.close, period=21)
        self.bbands = bt.indicators.BollingerBands(self.roc21, period=252, devfactor=2)
        self.atr = bt.indicators.ATR(self.data, period=14)
        self.order = None
        self.entry_bar = None

    def notify_order(self, order):
        if order.status == order.Completed:
            if self.position:
                self.entry_bar = len(self)
            self.order = None
        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        if len(self) < 273:
            return

        if self.order:
            return

        if self.position:
            if len(self) - self.entry_bar >= 5:
                self.order = self.close()
            return

        dt = self.data.datetime.date(0)
        if dt.weekday() != 0:
            return

        atr_val = self.atr[0]
        if atr_val <= 0:
            return

        roc = self.roc21[0]
        upper = self.bbands.top[0]
        lower = self.bbands.bot[0]

        risk_amount = 0.01 * self.broker.getvalue()
        stop_distance = 2 * atr_val
        size = int(risk_amount / stop_distance)
        if size <= 0:
            return

        if roc < lower:
            self.order = self.buy(size=size)
        elif roc > upper:
            self.order = self.sell(size=size)
