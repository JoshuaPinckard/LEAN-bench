import backtrader as bt


class TimeSeriesMomentum(bt.Strategy):
    params = (
        ("N", 90),
        ("R", 21),
        ("w", 0.20),
        ("dollar_vol_period", 21),
        ("dollar_vol_thresh", 1000000.0),
        ("eps", 1e-6),
    )

    def __init__(self):
        self.order = None

        # Momentum M = Close / Close[-N] - 1
        self.close_N = self.data.close(-self.p.N)
        self.momentum = (self.data.close / self.close_N) - 1.0

        # CrossOver(close, close(-N)) to track changes in the sign of momentum
        self.crossover = bt.indicators.CrossOver(self.data.close, self.close_N)

        # Liquidity filter: SMA(21) of dollar volume (Close * Volume)
        dollar_volume = self.data.close * self.data.volume
        self.sma_dv = bt.indicators.SMA(
            dollar_volume, period=self.p.dollar_vol_period
        )

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return
        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            self.order = None

    def next(self):
        if self.order:
            return

        # Liquidity filter & history requirement
        liquidity_ok = (self.sma_dv[0] >= self.p.dollar_vol_thresh) and (
            len(self) >= self.p.N
        )

        # Periodic rebalancing schedule
        is_rebal_day = len(self) % self.p.R == 0

        if is_rebal_day:
            if liquidity_ok:
                m_val = self.momentum[0]
                alloc = min(
                    self.p.w if self.p.w <= 1.0 else self.p.w / 100.0, 0.10
                )
                capital = self.broker.getvalue()
                price = self.data.close[0]

                s_target = 0
                if price > 0:
                    raw_size = int((capital * alloc) / price)
                    if m_val > self.p.eps:
                        s_target = raw_size
                    elif m_val < -self.p.eps:
                        s_target = -raw_size
                    else:
                        s_target = 0

                pos_size = self.position.size
                if s_target != pos_size:
                    if s_target == 0:
                        self.order = self.close()
                    elif s_target > pos_size:
                        self.order = self.buy(size=s_target - pos_size)
                    else:
                        self.order = self.sell(size=pos_size - s_target)
            else:
                # Liquidity filter failed: stay out / close position
                if self.position.size != 0:
                    self.order = self.close()
        else:
            # Between rebalancings: close open position on zero crossing
            if self.position.size != 0:
                if self.crossover[0] != 0:
                    self.order = self.close()
