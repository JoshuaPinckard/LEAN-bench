import backtrader as bt


class PartialTPTrailingStop(bt.Strategy):
    """
    Reference implementation of a trading strategy with partial take-profit
    and trailing stop loss.
    """

    params = (
        ("risk_per_trade", 0.01),  # 1% of total capital
        ("r_pct", 0.02),  # Risk distance R = 2% of entry price
        ("tp_r_multiple", 1.5),  # Partial TP target at Entry + 1.5 * R
        ("partial_tp_ratio", 0.50),  # Close 50% of position size at TP target
        ("trailing_stop_pct", 0.0075),  # Trailing stop offset 0.75%
    )

    def __init__(self):
        self.order = None
        self.entry_price = 0.0
        self.R = 0.0
        self.initial_sl = 0.0
        self.tp_target = 0.0
        self.tp_done = False
        self.trailing_active = False
        self.max_price = 0.0
        self.trailing_stop_level = 0.0

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            if order.isbuy() and self.position.size > 0:
                # Long entry completed
                self.entry_price = order.executed.price
                self.R = self.entry_price * self.p.r_pct
                self.initial_sl = self.entry_price - self.R
                self.tp_target = self.entry_price + (
                    self.p.tp_r_multiple * self.R
                )
                self.tp_done = False
                self.trailing_active = False
                self.max_price = 0.0
                self.trailing_stop_level = 0.0

        self.order = None

    def next(self):
        # Do not place new orders if an order is pending
        if self.order:
            return

        # Entry logic: Enter long if flat
        if not self.position:
            capital = self.broker.getvalue()
            risk_amount = capital * self.p.risk_per_trade
            est_entry = self.data.close[0]
            est_R = est_entry * self.p.r_pct
            if est_R > 0:
                size = int(risk_amount / est_R)
                if size > 0:
                    self.order = self.buy(size=size)
            return

        # Position management for long positions
        if self.position.size > 0:
            if not self.tp_done:
                # 1. Check initial stop-loss
                if self.data.low[0] <= self.initial_sl:
                    self.order = self.close()
                    return

                # 2. Check partial take-profit
                if self.data.high[0] >= self.tp_target:
                    close_size = int(
                        self.position.size * self.p.partial_tp_ratio
                    )
                    if close_size > 0:
                        self.order = self.sell(size=close_size)
                    self.tp_done = True
                    self.trailing_active = True
                    # Starting point of trailing stop is activation price / target price
                    self.max_price = max(self.tp_target, self.data.high[0])
                    self.trailing_stop_level = self.max_price * (
                        1.0 - self.p.trailing_stop_pct
                    )
                    return
            else:
                # 3. Trailing stop management for remaining position
                if self.trailing_active:
                    self.max_price = max(self.max_price, self.data.high[0])
                    self.trailing_stop_level = self.max_price * (
                        1.0 - self.p.trailing_stop_pct
                    )

                    if self.data.low[0] <= self.trailing_stop_level:
                        self.order = self.close()
                        return
