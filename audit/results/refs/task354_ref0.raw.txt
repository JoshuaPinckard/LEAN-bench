import backtrader as bt


class IntradayBreakoutBandsStrategy(bt.Strategy):
    params = (
        ('period', 20),
        ('stop_pct', 0.02),
    )

    def __init__(self):
        self.sma = bt.indicators.SMA(self.data.close, period=self.p.period)
        self.stddev = bt.indicators.StdDev(self.data.close, period=self.p.period)
        self.upper_band = self.sma + self.stddev
        self.lower_band = self.sma - self.stddev

        self.cross_upper = bt.indicators.CrossOver(self.data.close, self.upper_band)
        self.cross_lower = bt.indicators.CrossOver(self.data.close, self.lower_band)
        self.cross_sma = bt.indicators.CrossOver(self.data.close, self.sma)

        self.order = None

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        if self.order:
            return

        if not self.position:
            if self.cross_upper[0] > 0:
                self.order = self.buy()
            elif self.cross_lower[0] < 0:
                self.order = self.sell()
        else:
            entry_price = self.position.price
            if self.position.size > 0:
                stop_price = entry_price * (1 - self.p.stop_pct)
                if self.cross_sma[0] < 0 or self.data.low[0] <= stop_price:
                    self.order = self.close()
            else:
                stop_price = entry_price * (1 + self.p.stop_pct)
                if self.cross_sma[0] > 0 or self.data.high[0] >= stop_price:
                    self.order = self.close()
