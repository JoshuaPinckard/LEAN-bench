import backtrader as bt


class VolatilitySpikeStrategy(bt.Strategy):
    params = dict(
        x_threshold=20.0,
        y_period=10,
        z_holding=5,
        stop_loss_pct=0.05,
        take_profit_pct=0.10,
        min_avg_dollar_volume=1_000_000,
    )

    def __init__(self):
        self.order = None
        self.entry_price = None
        self.entry_bar = None
        self.stop_price = None
        self.tp_price = None

        dollar_volume = self.data.close * self.data.volume
        self.avg_dollar_volume = bt.indicators.SMA(dollar_volume, period=self.p.y_period)

    def notify_order(self, order):
        if order.status == order.Completed:
            if order.isbuy():
                self.entry_price = order.executed.price
                self.entry_bar = len(self)
                self.stop_price = self.entry_price * (1 - self.p.stop_loss_pct)
                self.tp_price = self.entry_price * (1 + self.p.take_profit_pct)

        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        if self.order:
            return

        if self.position:
            if self.data.low[0] <= self.stop_price:
                self.order = self.close()
                return
            if self.data.high[0] >= self.tp_price:
                self.order = self.close()
                return
            if (len(self) - self.entry_bar) >= self.p.z_holding:
                self.order = self.close()
                return
            return

        if len(self) <= self.p.y_period:
            return

        period_open_price = self.data.close[-self.p.y_period]
        period_close_price = self.data.close[0]
        pct_change = (period_close_price / period_open_price - 1) * 100

        if abs(pct_change) <= self.p.x_threshold:
            return

        if self.avg_dollar_volume[0] <= self.p.min_avg_dollar_volume:
            return

        cash = self.broker.getcash()
        price = self.data.close[0]
        size = int(cash / (price * 1.001))
        if size <= 0:
            return

        self.order = self.buy(size=size)
