import backtrader as bt


class DonchianContractionBreakout(bt.Strategy):
    params = dict(
        donchian_period=20,
        atr_period=14,
        contraction_mult=2.0,
        stop_atr_mult=2.0,
    )

    def __init__(self):
        self.donchian_high = bt.indicators.Highest(self.data.high, period=self.p.donchian_period)
        self.donchian_low = bt.indicators.Lowest(self.data.low, period=self.p.donchian_period)
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)

        # Reference the channel/ATR as of the prior completed bar so the
        # current close can be tested against a band that does not already
        # embed the current bar's own high/low.
        self.upper_ref = self.donchian_high(-1)
        self.lower_ref = self.donchian_low(-1)
        self.atr_ref = self.atr(-1)

        self.width_ref = self.upper_ref - self.lower_ref
        self.contraction = self.width_ref < (self.p.contraction_mult * self.atr_ref)

        self.cross_up = bt.indicators.CrossOver(self.data.close, self.upper_ref)
        self.cross_down = bt.indicators.CrossOver(self.data.close, self.lower_ref)

        self.order = None
        self.stop_price = None

    def notify_order(self, order):
        if order.status == order.Completed:
            if self.position:
                if order.isbuy():
                    self.stop_price = order.executed.price - self.p.stop_atr_mult * self.atr[0]
                else:
                    self.stop_price = order.executed.price + self.p.stop_atr_mult * self.atr[0]
            else:
                self.stop_price = None
            self.order = None
        elif order.status in (order.Canceled, order.Margin, order.Rejected):
            self.order = None

    def next(self):
        if self.order:
            return

        if not self.position:
            if self.contraction[0] and self.cross_up[0] > 0:
                self.order = self.buy()
            elif self.contraction[0] and self.cross_down[0] < 0:
                self.order = self.sell()
            return

        if self.position.size > 0:
            if self.data.low[0] <= self.stop_price:
                self.order = self.close()
            elif self.contraction[0] and self.cross_down[0] < 0:
                self.order = self.close()
        elif self.position.size < 0:
            if self.data.high[0] >= self.stop_price:
                self.order = self.close()
            elif self.contraction[0] and self.cross_up[0] > 0:
                self.order = self.close()
