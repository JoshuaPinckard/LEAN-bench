import backtrader as bt
import math
import statistics
from collections import deque


class VaROverlappingIntervalsStrategy(bt.Strategy):
    params = (
        ('vol_period', 250),
        ('confidence_level', 0.95),
        ('horizon', 1),
        ('black_swan_z', 5.0),
    )

    def __init__(self):
        self.returns = deque(maxlen=self.p.vol_period + 1)
        self.z = statistics.NormalDist().inv_cdf(self.p.confidence_level)

        self.hits = []
        self.total_observations = 0
        self.total_exceedances = 0
        self.total_excluded_black_swan = 0
        self.current_run = 0
        self.max_run = 0
        self.num_clusters = 0
        self.exceedance_indices = []

    def next(self):
        if len(self.data) < 2:
            return

        ret = (self.data.close[0] - self.data.close[-1]) / self.data.close[-1]
        self.returns.append(ret)

        if len(self.returns) <= self.p.vol_period:
            return

        hist = list(self.returns)[-self.p.vol_period - 1:-1]
        n = len(hist)
        mean = sum(hist) / n
        variance = sum((r - mean) ** 2 for r in hist) / (n - 1)
        sigma = math.sqrt(variance)

        var_pct = self.z * sigma * math.sqrt(self.p.horizon)
        portfolio_value = self.broker.getvalue()
        var_dollar = var_pct * portfolio_value
        loss = -ret * portfolio_value

        is_black_swan = abs(ret) > self.p.black_swan_z * sigma

        self.total_observations += 1

        if is_black_swan:
            self.total_excluded_black_swan += 1
            self.current_run = 0
            self.hits.append({'excluded': True, 'exceedance': False})
            return

        is_exceedance = loss > var_dollar
        self.hits.append({'excluded': False, 'exceedance': is_exceedance})

        if is_exceedance:
            self.total_exceedances += 1
            self.exceedance_indices.append(len(self.hits) - 1)
            self.current_run += 1
            if self.current_run == 2:
                self.num_clusters += 1
            self.max_run = max(self.max_run, self.current_run)
        else:
            self.current_run = 0

    def stop(self):
        n = self.total_observations - self.total_excluded_black_swan
        p_hat = self.total_exceedances / n if n > 0 else 0.0
        expected_p = 1.0 - self.p.confidence_level

        gaps = []
        for i in range(1, len(self.exceedance_indices)):
            gaps.append(self.exceedance_indices[i] - self.exceedance_indices[i - 1])
        avg_gap = sum(gaps) / len(gaps) if gaps else None

        conditional_hits = 0
        conditional_total = 0
        for i in range(1, len(self.hits)):
            prev = self.hits[i - 1]
            cur = self.hits[i]
            if prev['excluded'] or cur['excluded']:
                continue
            if prev['exceedance']:
                conditional_total += 1
                if cur['exceedance']:
                    conditional_hits += 1
        p_conditional = conditional_hits / conditional_total if conditional_total > 0 else None

        print('=== VaR Overlapping-Interval Exceedance Report ===')
        print('Observations analyzed (excl. black swans): %d' % n)
        print('Black-swan periods excluded: %d' % self.total_excluded_black_swan)
        print('Total VaR exceedances: %d' % self.total_exceedances)
        print('Empirical exceedance rate: %.4f (model-implied rate: %.4f)' % (p_hat, expected_p))
        print('Max consecutive-exceedance run length: %d' % self.max_run)
        print('Number of exceedance clusters (runs >= 2): %d' % self.num_clusters)
        if avg_gap is not None:
            print('Average gap between exceedances (bars): %.2f' % avg_gap)
        if p_conditional is not None:
            print('P(exceedance | previous bar exceedance) = %.4f vs unconditional rate = %.4f' % (p_conditional, p_hat))

        if p_hat > expected_p * 1.5:
            print('Conclusion: empirical exceedance rate materially exceeds the model-implied rate; VaR model appears to understate risk.')
        elif p_hat < expected_p * 0.5:
            print('Conclusion: empirical exceedance rate is materially below the model-implied rate; VaR model appears conservative.')
        else:
            print('Conclusion: empirical exceedance rate is broadly consistent with the model-implied rate.')

        if p_conditional is not None and p_conditional > p_hat * 1.5:
            print('Conclusion: exceedances cluster well beyond what independence would predict, indicating dependence introduced by overlapping intervals and likely VaR model misspecification.')
        elif self.num_clusters == 0 and self.total_exceedances > 1:
            print('Conclusion: exceedances are isolated rather than clustered, consistent with the independence assumption holding despite overlapping intervals.')
        else:
            print('Conclusion: insufficient evidence to distinguish clustering from randomness given the observed exceedance count.')
