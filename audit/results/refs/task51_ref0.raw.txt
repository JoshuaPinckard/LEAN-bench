import backtrader as bt


class WeeklySessionRectangle(bt.Strategy):
    params = dict(
        rectcolor='gray',
        rectalpha=0.15,
    )

    class _WeekBox(bt.Indicator):
        lines = ('top', 'bottom')

        params = dict(
            rectcolor='gray',
            rectalpha=0.15,
        )

        plotinfo = dict(subplot=False, plotname='Weekly Session')
        plotlines = dict(
            top=dict(color='gray', ls='-', _fill_gt=('bottom', 0.15)),
            bottom=dict(color='gray', ls='-'),
        )

        def __init__(self):
            self.plotlines.top.color = self.p.rectcolor
            self.plotlines.bottom.color = self.p.rectcolor
            self.plotlines.top._fill_gt = ('bottom', self.p.rectalpha)

            self._week_key = None
            self._top_val = None
            self._bottom_val = None

        def next(self):
            dt = self.data.datetime.datetime(0)
            weekday = dt.weekday()

            if weekday > 4:
                self.lines.top[0] = float('nan')
                self.lines.bottom[0] = float('nan')
                self._week_key = None
                return

            week_key = dt.isocalendar()[:2]
            high = self.data.high[0]
            low = self.data.low[0]

            if week_key != self._week_key:
                self._week_key = week_key
                self._top_val = high
                self._bottom_val = low
            else:
                if high > self._top_val:
                    self._top_val = high
                if low < self._bottom_val:
                    self._bottom_val = low

            self.lines.top[0] = self._top_val
            self.lines.bottom[0] = self._bottom_val

    def __init__(self):
        self.week_box = self._WeekBox(
            self.data,
            rectcolor=self.p.rectcolor,
            rectalpha=self.p.rectalpha,
        )

    def next(self):
        pass
