import backtrader as bt
import math


class LowVolatilityStrategy(bt.Strategy):
    def __init__(self):
        self.bbands = bt.indicators.BollingerBands(
            self.data.close, period=20, devfactor=2
        )
        self.atr = bt.indicators.ATR(self.data, period=14)
        self.atr_sma = bt.indicators.SMA(self.atr, period=252)
        self.volume_sma = bt.indicators.SMA(self.data.volume, period=252)
        self.order = None

    def notify_order(self, order):
        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            self.order = None

    def next(self):
        if self.order is not None or len(self.data) < 126:
            return

        # A decision on the final available bar cannot be acted on next month.
        if len(self.data) >= self.data.buflen():
            return

        current_date = self.data.datetime.date(0)
        next_date = bt.num2date(self.data.datetime[1]).date()

        if current_date.month == next_date.month:
            return

        widths = []
        for index in range(252):
            mid = self.bbands.mid[-index]
            widths.append(
                (self.bbands.top[-index] - self.bbands.bot[-index]) / mid
            )

        widths.sort()
        percentile_20 = widths[math.ceil(0.20 * len(widths)) - 1]
        current_width = (
            (self.bbands.top[0] - self.bbands.bot[0]) / self.bbands.mid[0]
        )

        conditions_met = (
            current_width <= percentile_20
            and self.atr[0] < self.atr_sma[0]
            and self.volume_sma[0] >= 1000000
        )

        if self.position:
            if not conditions_met:
                self.order = self.close()
        elif conditions_met:
            self.order = self.order_target_percent(target=1.0)
