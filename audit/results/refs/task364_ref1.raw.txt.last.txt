import backtrader as bt


class IntradayMeanReversionStrategy(bt.Strategy):
    params = (
        ('bb_period', 20),
        ('bb_devfactor', 2.0),
        ('stoch_period', 14),
        ('stoch_period_dfast', 3),
        ('stoch_period_dslow', 3),
    )

    def __init__(self):
        self.bbands = bt.indicators.BollingerBands(
            self.data.close,
            period=self.p.bb_period,
            devfactor=self.p.bb_devfactor,
        )
        self.stochastic = bt.indicators.Stochastic(
            self.data,
            period=self.p.stoch_period,
            period_dfast=self.p.stoch_period_dfast,
            period_dslow=self.p.stoch_period_dslow,
        )
        self.order = None
        self.entry_price = None

    def notify_order(self, order):
        if order.status == order.Completed:
            if order.isbuy() and self.position.size > 0:
                self.entry_price = order.executed.price
            elif order.issell() and self.position.size < 0:
                self.entry_price = order.executed.price
            elif self.position.size == 0:
                self.entry_price = None

        if order.status in (order.Completed, order.Canceled,
                            order.Margin, order.Rejected):
            self.order = None

    def next(self):
        if self.order:
            return

        long_signal = (
            self.data.low[0] <= self.bbands.bot[0]
            and self.stochastic.percK[0] < 20
        )
        short_signal = (
            self.data.high[0] >= self.bbands.top[0]
            and self.stochastic.percK[0] > 80
        )

        if not self.position:
            if long_signal:
                self.order = self.buy()
            elif short_signal:
                self.order = self.sell()
            return

        if self.position.size > 0:
            stop_loss = (
                self.entry_price is not None
                and self.data.low[0] <= self.entry_price * 0.98
            )
            if short_signal or stop_loss:
                self.order = self.close()

        elif self.position.size < 0:
            stop_loss = (
                self.entry_price is not None
                and self.data.high[0] >= self.entry_price * 1.02
            )
            if long_signal or stop_loss:
                self.order = self.close()
