import backtrader as bt


class IntradayVWAP(bt.Indicator):
    lines = ('vwap',)

    def __init__(self):
        self.addminperiod(1)

    def next(self):
        dt = self.data.datetime.date(0)
        prev_dt = self.data.datetime.date(-1) if len(self) > 1 else None

        tp = (self.data.high[0] + self.data.low[0] + self.data.close[0]) / 3.0
        vol = self.data.volume[0]

        if prev_dt is None or dt != prev_dt:
            self.cum_tp_vol = tp * vol
            self.cum_vol = vol
        else:
            self.cum_tp_vol += tp * vol
            self.cum_vol += vol

        self.lines.vwap[0] = self.cum_tp_vol / self.cum_vol if self.cum_vol > 0 else self.data.close[0]


class TSI(bt.Indicator):
    lines = ('tsi',)
    params = (('r', 25), ('s', 13),)

    def __init__(self):
        m = self.data.close - self.data.close(-1)
        abs_m = bt.abs(m)
        ema_r_m = bt.indicators.EMA(m, period=self.p.r)
        ema_s_r_m = bt.indicators.EMA(ema_r_m, period=self.p.s)
        ema_r_abs = bt.indicators.EMA(abs_m, period=self.p.r)
        ema_s_r_abs = bt.indicators.EMA(ema_r_abs, period=self.p.s)
        self.lines.tsi = 100.0 * (ema_s_r_m / (ema_s_r_abs + 1e-9))


class PVT(bt.Indicator):
    lines = ('pvt',)

    def __init__(self):
        close_diff = self.data.close - self.data.close(-1)
        ret = close_diff / self.data.close(-1)
        self.lines.pvt = bt.indicators.Cumulative(self.data.volume * ret)


class TrendChannelBreakout(bt.Strategy):
    params = (
        ('risk_pct', 0.01),
        ('stop_loss_pct', 0.01),
        ('rsi_threshold', 50.0),
        ('max_volatility_pct', 0.0),
    )

    def __init__(self):
        self.rsi = bt.indicators.RSI(self.data.close, period=14)
        self.tsi = TSI(self.data.close, r=self.p.r if hasattr(self.p, 'r') else 25, s=13)
        self.sma20 = bt.indicators.SMA(self.data.close, period=20)
        self.sma50 = bt.indicators.SMA(self.data.close, period=50)
        self.vol_sma = bt.indicators.SMA(self.data.volume, period=20)
        self.vwap = IntradayVWAP(self.data)
        self.pvt = PVT(self.data)

        self.order = None
        self.buy_price = None

        self.current_date = None
        self.upper_channel = None
        self.lower_channel = None
        self.day_skipped = False
        self.entry_upper_channel = None
        self.stop_price = None

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            if order.status == order.Completed:
                if order.isbuy():
                    self.buy_price = order.executed.price
            self.order = None

    def next(self):
        dt = self.data.datetime.date(0)
        t = self.data.datetime.time(0)

        if dt != self.current_date:
            self.current_date = dt
            self.upper_channel = None
            self.lower_channel = None
            self.day_skipped = False

        if (t.hour == 9 and t.minute >= 55) and self.upper_channel is None:
            self.upper_channel = self.data.high[0]

        if (t.hour > 9 or (t.hour == 10 and t.minute >= 0)) and self.lower_channel is None and self.upper_channel is not None:
            self.lower_channel = self.data.low[0]
            if self.p.max_volatility_pct > 0.0 and self.lower_channel > 0:
                ch_range = (self.upper_channel - self.lower_channel) / self.lower_channel
                if ch_range > self.p.max_volatility_pct:
                    self.day_skipped = True

        if self.order:
            return

        if self.position:
            if self.stop_price is not None and self.data.low[0] <= self.stop_price:
                self.order = self.close()
                return

            if self.data.low[0] <= self.sma50[0]:
                self.order = self.close()
                return

            target_channel = self.entry_upper_channel if self.entry_upper_channel is not None else self.upper_channel
            if target_channel is not None and self.data.close[0] <= target_channel:
                self.order = self.close()
                return

        else:
            if self.upper_channel is None or self.lower_channel is None or self.day_skipped:
                return

            if not (10 <= t.hour < 15):
                return

            if t.minute % 5 != 0:
                return

            cond_breakout = self.data.close[0] > self.upper_channel
            cond_trend = self.sma20[0] > self.sma50[0]
            cond_momentum = (self.rsi[0] > self.p.rsi_threshold) and (self.tsi[0] > 0)
            cond_volume = self.data.volume[0] > self.vol_sma[0]

            if cond_breakout and cond_trend and cond_momentum and cond_volume:
                stop_calc = min(self.data.close[0] * (1.0 - self.p.stop_loss_pct), self.lower_channel)
                risk_per_share = self.data.close[0] - stop_calc
                if risk_per_share <= 0:
                    risk_per_share = self.data.close[0] * self.p.stop_loss_pct

                account_val = self.broker.getvalue()
                cash = self.broker.getcash()
                risk_amount = account_val * self.p.risk_pct
                size = int(risk_amount / risk_per_share)
                max_size = int(cash / self.data.close[0])
                size = min(size, max_size)

                if size > 0:
                    self.stop_price = stop_calc
                    self.entry_upper_channel = self.upper_channel
                    self.order = self.buy(size=size)
