```python
import math
import backtrader as bt


class SequentialBuyOrderStrategy(bt.Strategy):
    params = dict(
        first_size_pct=0.5,
        second_size_pct=0.5,
        max_total_risk_pct=0.01,
        window_start_hour=10,
        window_start_minute=0,
        window_end_hour=16,
        window_end_minute=0,
    )

    def __init__(self):
        self.order = None
        self.pending_first_order = None
        self.second_size = 0

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return

        if order.status == order.Completed:
            if order.isbuy() and order is self.pending_first_order:
                self.pending_first_order = None
                self.order = self.buy(exectype=bt.Order.Market, size=self.second_size)
            else:
                self.order = None
        else:
            self.pending_first_order = None
            self.order = None

    def next(self):
        if self.order is not None:
            return
        if self.position:
            return

        bar_dt = self.data.datetime.datetime(0)
        bar_minutes = bar_dt.hour * 60 + bar_dt.minute
        start_minutes = self.p.window_start_hour * 60 + self.p.window_start_minute
        end_minutes = self.p.window_end_hour * 60 + self.p.window_end_minute
        if not (start_minutes <= bar_minutes <= end_minutes):
            return

        if (self.p.first_size_pct + self.p.second_size_pct) > self.p.max_total_risk_pct:
            return

        price = self.data.close[0]
        total_capital = self.broker.getvalue()
        first_size = int(math.floor((total_capital * self.p.first_size_pct) / price))
        second_size = int(math.floor((total_capital * self.p.second_size_pct) / price))

        if first_size <= 0 or second_size <= 0:
            return

        estimated_cost = (first_size + second_size) * price
        if self.broker.getcash() < estimated_cost:
            return

        self.second_size = second_size
        self.pending_first_order = self.buy(exectype=bt.Order.Limit, price=price, size=first_size)
        self.order = self.pending_first_order
```
