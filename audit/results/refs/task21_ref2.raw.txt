import backtrader as bt


class EmaStochasticScalping(bt.Strategy):
    params = (
        ("ema_period", 20),
        ("stoch_period", 14),
        ("stoch_dfast", 3),
        ("stoch_dslow", 3),
        ("tp_pct", 0.005),
        ("sl_pct", 0.003),
        ("position_pct", 0.01),
    )

    def __init__(self):
        self.ema = bt.indicators.ExponentialMovingAverage(
            self.data.close, period=self.p.ema_period
        )
        self.stoch = bt.indicators.Stochastic(
            self.data,
            period=self.p.stoch_period,
            period_dfast=self.p.stoch_dfast,
            period_dslow=self.p.stoch_dslow,
        )
        self.crossover = bt.indicators.CrossOver(
            self.stoch.percK, self.stoch.percD
        )

        self.order = None
        self.sl_order = None
        self.tp_order = None
        self.entry_price = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            if order == self.order:
                self.order = None
                if self.position:
                    self.entry_price = order.executed.price
                    size = abs(order.executed.size)
                    if self.position.size > 0:
                        sl_price = self.entry_price * (1.0 - self.p.sl_pct)
                        tp_price = self.entry_price * (1.0 + self.p.tp_pct)
                        self.sl_order = self.sell(
                            exectype=bt.Order.Stop, price=sl_price, size=size
                        )
                        self.tp_order = self.sell(
                            exectype=bt.Order.Limit,
                            price=tp_price,
                            size=size,
                            oco=self.sl_order,
                        )
                    elif self.position.size < 0:
                        sl_price = self.entry_price * (1.0 + self.p.sl_pct)
                        tp_price = self.entry_price * (1.0 - self.p.tp_pct)
                        self.sl_order = self.buy(
                            exectype=bt.Order.Stop, price=sl_price, size=size
                        )
                        self.tp_order = self.buy(
                            exectype=bt.Order.Limit,
                            price=tp_price,
                            size=size,
                            oco=self.sl_order,
                        )
                else:
                    self.entry_price = None
            elif order in [self.sl_order, self.tp_order]:
                self.sl_order = None
                self.tp_order = None
                self.entry_price = None

        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            if order == self.order:
                self.order = None
            if order == self.sl_order:
                self.sl_order = None
            if order == self.tp_order:
                self.tp_order = None

    def next(self):
        if self.order:
            return

        if not self.position:
            # Long entry condition
            if (
                self.data.close[0] > self.ema[0]
                and self.crossover[0] > 0
                and self.stoch.percK[0] < 20
                and self.stoch.percD[0] < 20
            ):
                size = (
                    self.broker.getvalue() * self.p.position_pct
                ) / self.data.close[0]
                self.order = self.buy(size=size)

            # Short entry condition
            elif (
                self.data.close[0] < self.ema[0]
                and self.crossover[0] < 0
                and self.stoch.percK[0] > 80
                and self.stoch.percD[0] > 80
            ):
                size = (
                    self.broker.getvalue() * self.p.position_pct
                ) / self.data.close[0]
                self.order = self.sell(size=size)

        else:
            if self.position.size > 0:
                if self.crossover[0] < 0:
                    if self.sl_order:
                        self.cancel(self.sl_order)
                        self.sl_order = None
                    if self.tp_order:
                        self.cancel(self.tp_order)
                        self.tp_order = None
                    self.order = self.close()

            elif self.position.size < 0:
                if self.crossover[0] > 0:
                    if self.sl_order:
                        self.cancel(self.sl_order)
                        self.sl_order = None
                    if self.tp_order:
                        self.cancel(self.tp_order)
                        self.tp_order = None
                    self.order = self.close()
