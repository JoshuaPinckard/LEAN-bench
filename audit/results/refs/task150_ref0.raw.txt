import backtrader as bt


class RenkoTrendFollowingStrategy(bt.Strategy):
    params = dict(
        brick_size=50.0,
    )

    def __init__(self):
        self.renko_level = None
        self.brick_directions = []
        self.order = None
        self.stop_order = None
        self.limit_order = None

    def _cancel_exit_orders(self):
        if self.stop_order is not None:
            self.cancel(self.stop_order)
            self.stop_order = None
        if self.limit_order is not None:
            self.cancel(self.limit_order)
            self.limit_order = None

    def _process_brick(self, direction):
        prev_direction = self.brick_directions[-1] if self.brick_directions else None
        self.brick_directions.append(direction)

        if self.order:
            return

        if self.position:
            if self.position.size > 0 and direction == -1:
                self._cancel_exit_orders()
                self.order = self.close()
            elif self.position.size < 0 and direction == 1:
                self._cancel_exit_orders()
                self.order = self.close()
            return

        if prev_direction is not None and direction == prev_direction:
            cash = self.broker.getcash()
            price = self.data.close[0]
            size = int(cash / price)
            if size > 0:
                if direction == 1:
                    self.order = self.buy(size=size)
                else:
                    self.order = self.sell(size=size)

    def next(self):
        if self.renko_level is None:
            self.renko_level = self.data.close[0]
            return

        close = self.data.close[0]

        while close - self.renko_level >= self.p.brick_size:
            self.renko_level += self.p.brick_size
            self._process_brick(1)

        while self.renko_level - close >= self.p.brick_size:
            self.renko_level -= self.p.brick_size
            self._process_brick(-1)

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order == self.stop_order:
            if order.status == order.Completed:
                self.stop_order = None
                if self.limit_order is not None:
                    self.cancel(self.limit_order)
                    self.limit_order = None
            elif order.status in [order.Canceled, order.Margin, order.Rejected]:
                self.stop_order = None
            return

        if order == self.limit_order:
            if order.status == order.Completed:
                self.limit_order = None
                if self.stop_order is not None:
                    self.cancel(self.stop_order)
                    self.stop_order = None
            elif order.status in [order.Canceled, order.Margin, order.Rejected]:
                self.limit_order = None
            return

        if order == self.order:
            if order.status == order.Completed:
                self.order = None
                if order.isbuy() and self.position.size > 0:
                    entry_price = order.executed.price
                    size = self.position.size
                    self.stop_order = self.sell(
                        exectype=bt.Order.Stop,
                        price=entry_price - self.p.brick_size,
                        size=size,
                    )
                    self.limit_order = self.sell(
                        exectype=bt.Order.Limit,
                        price=entry_price + 2 * self.p.brick_size,
                        size=size,
                    )
                elif order.issell() and self.position.size < 0:
                    entry_price = order.executed.price
                    size = abs(self.position.size)
                    self.stop_order = self.buy(
                        exectype=bt.Order.Stop,
                        price=entry_price + self.p.brick_size,
                        size=size,
                    )
                    self.limit_order = self.buy(
                        exectype=bt.Order.Limit,
                        price=entry_price - 2 * self.p.brick_size,
                        size=size,
                    )
            elif order.status in [order.Canceled, order.Margin, order.Rejected]:
                self.order = None
