import math
import backtrader as bt


class MSTRSqueezeMomentum(bt.Strategy):
    def __init__(self):
        # Indicators
        self.bb = bt.ind.BollingerBands(self.data, period=20, devfactor=2.0)
        self.w = self.bb.top - self.bb.bot
        self.sma_w = bt.ind.SMA(self.w, period=20)

        self.rsi = bt.ind.RSI(self.data.close, period=14)

        self.atr = bt.ind.ATR(self.data, period=14)
        self.sma_atr = bt.ind.SMA(self.atr, period=20)

        self.sma_vol20 = bt.ind.SMA(self.data.volume, period=20)
        self.sma_vol3 = bt.ind.SMA(self.data.volume, period=3)

        self.crossover = bt.ind.CrossOver(self.data.close, self.bb.mid)

        # Order & Trade Management
        self.order = None

        # Cycle state machine: 'ALLOW_ENTRY', 'WAIT_EXPANSION', 'WAIT_SQUEEZE'
        self.cycle_state = 'ALLOW_ENTRY'

        # Signal bar snapshots for trade risk calculation
        self.signal_bb_bot = None
        self.signal_atr = None

        # Position tracking levels
        self.entry_price = None
        self.stop_level = None
        self.tp_level = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            if order.isbuy():
                self.entry_price = order.executed.price
                bb_bot = (
                    self.signal_bb_bot
                    if self.signal_bb_bot is not None
                    else self.bb.bot[0]
                )
                atr_val = (
                    self.signal_atr
                    if self.signal_atr is not None
                    else self.atr[0]
                )
                self.stop_level = min(
                    bb_bot, self.entry_price - 1.5 * atr_val
                )
                self.tp_level = self.entry_price + 2.0 * atr_val
            elif order.issell():
                self.entry_price = None
                self.stop_level = None
                self.tp_level = None

            self.order = None

        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def notify_trade(self, trade):
        if trade.isclosed:
            self.cycle_state = 'WAIT_EXPANSION'

    def next(self):
        # Do not place new orders if an order is pending
        if self.order:
            return

        # Handle Exits if in position
        if self.position:
            if self.stop_level is not None and self.data.low[0] <= self.stop_level:
                self.order = self.close()
            elif self.tp_level is not None and self.data.high[0] >= self.tp_level:
                self.order = self.close()
            elif self.data.close[0] > self.bb.top[0] and self.rsi[0] > 50:
                self.order = self.close()
            return

        # Update Squeeze Cycle State Machine when flat
        if self.cycle_state == 'WAIT_EXPANSION':
            if self.w[0] > self.sma_w[0]:
                self.cycle_state = 'WAIT_SQUEEZE'

        if self.cycle_state == 'WAIT_SQUEEZE':
            # Check if squeeze condition (item 1) is met again
            if (self.w[0] < 0.75 * self.sma_w[0]) and (
                self.atr[0] < self.sma_atr[0]
            ):
                self.cycle_state = 'ALLOW_ENTRY'

        # Entry logic
        if self.cycle_state == 'ALLOW_ENTRY':
            # Entry condition 1: Volatility squeeze
            cond1 = (self.w[0] < 0.75 * self.sma_w[0]) and (
                self.atr[0] < self.sma_atr[0]
            )

            # Entry condition 2: Price in narrow range
            mid = self.bb.mid[0]
            w_val = self.w[0]
            cond2 = (mid - 0.25 * w_val) <= self.data.close[0] <= (mid + 0.25 * w_val)

            # Entry condition 3: Oversold
            cond3 = self.rsi[0] < 35

            # Entry condition 4: Abnormal volume
            cond4 = self.sma_vol3[0] > 1.5 * self.sma_vol20[0]

            # Entry condition 5: Momentum trigger
            cond5 = self.crossover[0] > 0

            if cond1 and cond2 and cond3 and cond4 and cond5:
                est_entry_price = self.data.close[0]
                est_stop_level = min(
                    self.bb.bot[0], est_entry_price - 1.5 * self.atr[0]
                )
                risk_per_share = est_entry_price - est_stop_level

                if risk_per_share > 0:
                    equity = self.broker.getvalue()
                    size = math.floor((0.02 * equity) / risk_per_share)
                    if size > 0:
                        self.signal_bb_bot = self.bb.bot[0]
                        self.signal_atr = self.atr[0]
                        self.order = self.buy(size=size)
