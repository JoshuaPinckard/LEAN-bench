import backtrader as bt


class TrendPullbackScalingStrategy(bt.Strategy):
    params = (
        ('atr_period', 14),
        ('ema_period', 50),
        ('pullback_atr_mult', 1.5),
        ('stop_atr_mult', 1.5),
        ('target_atr_mult', 2.5),
        ('max_entries', 4),
        ('position_pct', 0.01),
    )

    def __init__(self):
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.ema = bt.indicators.EMA(self.data, period=self.p.ema_period)
        self.entries = []
        self.swing_high = None
        self.order = None
        self.pending_action = None
        self.pending_leg = None

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return

        if order.status == order.Completed:
            if self.pending_action == 'entry':
                leg = dict(
                    price=order.executed.price,
                    atr=self.atr[0],
                    size=order.executed.size,
                )
                leg['stop'] = leg['price'] - self.p.stop_atr_mult * leg['atr']
                self.entries.append(leg)
            elif self.pending_action == 'leg_stop':
                self.entries = [e for e in self.entries if e is not self.pending_leg]
            elif self.pending_action == 'close_all':
                self.entries = []

        self.order = None
        self.pending_action = None
        self.pending_leg = None

    def _enter_leg(self):
        value = self.broker.getvalue()
        size = (self.p.position_pct * value) / self.data.close[0]
        self.pending_action = 'entry'
        self.order = self.buy(size=size)

    def _exit_leg(self, leg):
        self.pending_action = 'leg_stop'
        self.pending_leg = leg
        self.order = self.sell(size=leg['size'])

    def _close_all(self):
        self.pending_action = 'close_all'
        self.order = self.close()

    def next(self):
        if self.order:
            return

        close = self.data.close[0]
        high = self.data.high[0]
        low = self.data.low[0]
        ema = self.ema[0]
        atr = self.atr[0]

        if self.swing_high is None:
            self.swing_high = high

        if self.entries:
            first_entry = self.entries[0]

            if close < ema:
                self._close_all()
                self.swing_high = high
                return

            if high >= first_entry['price'] + self.p.target_atr_mult * first_entry['atr']:
                self._close_all()
                self.swing_high = high
                return

            for leg in self.entries:
                if low <= leg['stop']:
                    self._exit_leg(leg)
                    self.swing_high = high
                    return

            if len(self.entries) < self.p.max_entries and close > ema:
                self.swing_high = max(self.swing_high, high)
                if self.swing_high - close >= self.p.pullback_atr_mult * atr:
                    self._enter_leg()
                    self.swing_high = high
            else:
                self.swing_high = max(self.swing_high, high)
        else:
            if close > ema:
                self.swing_high = max(self.swing_high, high)
                if self.swing_high - close >= self.p.pullback_atr_mult * atr:
                    self._enter_leg()
                    self.swing_high = high
            else:
                self.swing_high = high
