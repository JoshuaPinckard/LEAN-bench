import backtrader as bt
import math
import numpy as np
from hmmlearn.hmm import GaussianHMM


class QCBAuditStrategy(bt.Strategy):

    params = (
        ('macd_fast', 12),
        ('macd_slow', 26),
        ('macd_signal', 9),
        ('atr_period', 14),
        ('hmm_window', 390),
        ('base_k', 1.0),
        ('size', 1),
    )

    def __init__(self):
        self.macd_ind = bt.indicators.MACD(
            self.data.close,
            period1=self.p.macd_fast,
            period2=self.p.macd_slow,
            period_signal=self.p.macd_signal,
        )
        self.atr_ind = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.macd_cross = bt.indicators.CrossOver(
            self.macd_ind.macd, self.macd_ind.signal
        )

        # k = base_k * (slow_period / fast_period) = 1.0 * (26/12)
        self.k = self.p.base_k * (self.p.macd_slow / self.p.macd_fast)

        self.order = None
        self.stop_order = None

        self.hmm_model = None
        self.bull_hmm_state = None  # raw HMM state index that maps to "bull"

        self.cur_logical = None   # 1 = bull, 0 = bear
        self.prev_logical = None

        self.last_fit_date = None

    # ------------------------------------------------------------------
    # Order lifecycle
    # ------------------------------------------------------------------

    def notify_order(self, order):
        if order.status in [
            order.Completed, order.Canceled, order.Margin, order.Rejected
        ]:
            if order is self.order:
                self.order = None
            if order is self.stop_order:
                self.stop_order = None

    # ------------------------------------------------------------------
    # HMM helpers
    # ------------------------------------------------------------------

    def _get_past_closes(self, n):
        """Up to n closes ending at previous bar (no current bar), chronological."""
        available = len(self.data) - 1
        n_use = min(n, available)
        if n_use < 1:
            return []
        return [self.data.close[-n_use + i] for i in range(n_use)]

    def _log_returns(self, closes):
        if len(closes) < 2:
            return []
        return [math.log(closes[i] / closes[i - 1]) for i in range(1, len(closes))]

    def _fit_hmm(self):
        """Refit on past data only; called once per day at session open."""
        closes = self._get_past_closes(self.p.hmm_window + 1)
        rets = self._log_returns(closes)
        if len(rets) < 4:
            return
        X = np.array(rets, dtype=float).reshape(-1, 1)
        model = GaussianHMM(
            n_components=2, covariance_type='diag',
            n_iter=100, random_state=42,
        )
        try:
            model.fit(X)
        except Exception:
            return
        self.hmm_model = model
        self.bull_hmm_state = int(np.argmax(model.means_.flatten()))

    def _infer_state(self):
        """Viterbi-decode rolling window ending at current bar; update logical state."""
        if self.hmm_model is None:
            return
        n_use = min(self.p.hmm_window + 1, len(self.data))
        if n_use < 2:
            return
        # Chronological closes from -(n_use-1) bars ago through current bar (index 0)
        closes = [self.data.close[-(n_use - 1) + i] for i in range(n_use)]
        rets = self._log_returns(closes)
        if not rets:
            return
        X = np.array(rets, dtype=float).reshape(-1, 1)
        try:
            states = self.hmm_model.predict(X)
        except Exception:
            return
        logical = 1 if int(states[-1]) == self.bull_hmm_state else 0
        self.prev_logical = self.cur_logical
        self.cur_logical = logical

    # ------------------------------------------------------------------
    # ATR stop management
    # ------------------------------------------------------------------

    def _update_stop(self):
        """Cancel existing stop and place refreshed ATR stop for current bar."""
        if self.stop_order is not None:
            self.cancel(self.stop_order)
            self.stop_order = None
        dist = self.atr_ind[0] * self.k
        if self.position.size > 0:
            self.stop_order = self.sell(
                exectype=bt.Order.Stop,
                price=self.data.close[0] - dist,
                size=abs(self.position.size),
            )
        elif self.position.size < 0:
            self.stop_order = self.buy(
                exectype=bt.Order.Stop,
                price=self.data.close[0] + dist,
                size=abs(self.position.size),
            )

    # ------------------------------------------------------------------
    # Main logic
    # ------------------------------------------------------------------

    def next(self):
        current_date = self.data.datetime.date(0)

        # Session open: refit HMM on past data only, reset state tracking
        if current_date != self.last_fit_date:
            self.last_fit_date = current_date
            self.prev_logical = None
            self.cur_logical = None
            self._fit_hmm()

        # Infer current bar's hidden state
        self._infer_state()

        cross_up = self.macd_cross[0] > 0
        cross_dn = self.macd_cross[0] < 0

        # "Transitions into" = prior state was NOT the target AND current IS
        to_bull = (
            self.prev_logical is not None
            and self.prev_logical != 1
            and self.cur_logical == 1
        )
        to_bear = (
            self.prev_logical is not None
            and self.prev_logical != 0
            and self.cur_logical == 0
        )

        pos = self.position.size

        if pos > 0:
            # Exit long on opposite confirmation
            if self.order is None and cross_dn and to_bear:
                if self.stop_order is not None:
                    self.cancel(self.stop_order)
                    self.stop_order = None
                self.order = self.close()
            elif self.order is None:
                self._update_stop()

        elif pos < 0:
            # Exit short on opposite confirmation
            if self.order is None and cross_up and to_bull:
                if self.stop_order is not None:
                    self.cancel(self.stop_order)
                    self.stop_order = None
                self.order = self.close()
            elif self.order is None:
                self._update_stop()

        else:
            # Entry: require HMM fitted and no pending order
            if self.order is None and self.hmm_model is not None:
                if cross_up and to_bull:
                    self.order = self.buy(size=self.p.size)
                elif cross_dn and to_bear:
                    self.order = self.sell(size=self.p.size)
