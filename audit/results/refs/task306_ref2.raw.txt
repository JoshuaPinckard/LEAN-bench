import math
import collections
import statistics
import numpy as np
from hmmlearn import hmm
import backtrader as bt


class HMMMACDStrategy(bt.Strategy):
    params = (
        ('fast_period', 12),
        ('slow_period', 26),
        ('signal_period', 9),
        ('atr_period', 14),
        ('base_k', 1.0),
        ('window_size', 390),
    )

    def __init__(self):
        self.macd = bt.indicators.MACD(
            self.data.close,
            period_me1=self.p.fast_period,
            period_me2=self.p.slow_period,
            period_signal=self.p.signal_period
        )
        self.crossover = bt.indicators.CrossOver(self.macd.macd, self.macd.signal)
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)

        self.k = self.p.base_k * (self.p.slow_period / self.p.fast_period)

        self.order = None
        self.stop_order = None

        self.log_returns = []
        self.model = None
        self.bull_state_idx = None
        self.bear_state_idx = None
        self.prev_regime = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            if order == self.order:
                self.order = None
            if order == self.stop_order:
                self.stop_order = None

    def next(self):
        if len(self) > 1:
            log_ret = math.log(self.data.close[0] / self.data.close[-1])
        else:
            log_ret = 0.0
        self.log_returns.append(log_ret)

        is_session_open = (len(self) == 1) or (self.data.datetime.date(0) != self.data.datetime.date(-1))

        if is_session_open and len(self.log_returns) >= self.p.window_size:
            X_train = np.array(self.log_returns[-self.p.window_size:]).reshape(-1, 1)
            model = hmm.GaussianHMM(n_components=2, covariance_type="diag", n_iter=100, random_state=42)
            try:
                model.fit(X_train)
                self.model = model
                if self.model.means_[1][0] > self.model.means_[0][0]:
                    self.bull_state_idx = 1
                    self.bear_state_idx = 0
                else:
                    self.bull_state_idx = 0
                    self.bear_state_idx = 1
            except Exception:
                pass

        current_regime = None
        if self.model is not None and len(self.log_returns) >= self.p.window_size:
            X_seq = np.array(self.log_returns[-self.p.window_size:]).reshape(-1, 1)
            try:
                states = self.model.predict(X_seq)
                last_state = states[-1]
                if last_state == self.bull_state_idx:
                    current_regime = 'bull'
                else:
                    current_regime = 'bear'
            except Exception:
                current_regime = None

        hmm_trans_bull = False
        hmm_trans_bear = False
        if current_regime is not None and self.prev_regime is not None:
            if current_regime == 'bull' and self.prev_regime == 'bear':
                hmm_trans_bull = True
            elif current_regime == 'bear' and self.prev_regime == 'bull':
                hmm_trans_bear = True

        if current_regime is not None:
            self.prev_regime = current_regime

        macd_cross_up = self.crossover[0] > 0
        macd_cross_down = self.crossover[0] < 0

        if self.order is not None:
            return

        if not self.position:
            if macd_cross_up and hmm_trans_bull:
                self.order = self.buy()
            elif macd_cross_down and hmm_trans_bear:
                self.order = self.sell()
        else:
            if self.position.size > 0:
                if macd_cross_down and hmm_trans_bear:
                    if self.stop_order:
                        self.cancel(self.stop_order)
                        self.stop_order = None
                    self.order = self.close()
                    return
            elif self.position.size < 0:
                if macd_cross_up and hmm_trans_bull:
                    if self.stop_order:
                        self.cancel(self.stop_order)
                        self.stop_order = None
                    self.order = self.close()
                    return

            if self.stop_order:
                self.cancel(self.stop_order)
                self.stop_order = None

            stop_dist = self.atr[0] * self.k
            if self.position.size > 0:
                stop_price = self.position.price - stop_dist
                self.stop_order = self.sell(exectype=bt.Order.Stop, price=stop_price)
            elif self.position.size < 0:
                stop_price = self.position.price + stop_dist
                self.stop_order = self.buy(exectype=bt.Order.Stop, price=stop_price)
