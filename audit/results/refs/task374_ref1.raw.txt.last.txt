import backtrader as bt


class IntradayMeanReversionStrategy(bt.Strategy):
    params = (
        ('sma_period', 20),
        ('atr_period', 14),
        ('atr_multiple', 2.0),
    )

    def __init__(self):
        self.sma = bt.indicators.SimpleMovingAverage(
            self.data.close, period=self.p.sma_period
        )
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)

        self.pending_order = None
        self.stop_order = None
        self.entry_atr = None

    def notify_order(self, order):
        if order.status in (
            order.Completed, order.Canceled, order.Margin, order.Rejected
        ):
            if order == self.pending_order:
                self.pending_order = None

            if order == self.stop_order:
                self.stop_order = None

            if order.status == order.Completed and order.isbuy() and order == self.pending_order:
                pass

        if order.status == order.Completed:
            if order == self.pending_order:
                self.pending_order = None

            if order.isbuy() and self.position.size > 0 and self.entry_atr is not None:
                self.stop_order = self.sell(
                    exectype=bt.Order.Stop,
                    price=order.executed.price - self.p.atr_multiple * self.entry_atr,
                )
                self.entry_atr = None
            elif order.issell() and self.position.size < 0 and self.entry_atr is not None:
                self.stop_order = self.buy(
                    exectype=bt.Order.Stop,
                    price=order.executed.price + self.p.atr_multiple * self.entry_atr,
                )
                self.entry_atr = None

        elif order.status in (order.Canceled, order.Margin, order.Rejected):
            if order == self.pending_order:
                self.pending_order = None
                self.entry_atr = None
            if order == self.stop_order:
                self.stop_order = None

    def next(self):
        if self.pending_order:
            return

        close = self.data.close[0]

        if self.position.size > 0:
            if close >= self.sma[0]:
                if self.stop_order:
                    self.cancel(self.stop_order)
                self.pending_order = self.close()
            return

        if self.position.size < 0:
            if close <= self.sma[0]:
                if self.stop_order:
                    self.cancel(self.stop_order)
                self.pending_order = self.close()
            return

        if close < self.sma[0] - self.p.atr_multiple * self.atr[0]:
            self.entry_atr = self.atr[0]
            self.pending_order = self.buy()
        elif close > self.sma[0] + self.p.atr_multiple * self.atr[0]:
            self.entry_atr = self.atr[0]
            self.pending_order = self.sell()
