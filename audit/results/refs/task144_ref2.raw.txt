import backtrader as bt


class RangeControlStrategy(bt.Strategy):
    params = (
        ("consolidation_period", 10),
        ("range_pct", 0.02),
        ("vol_sma_period", 20),
        ("vol_mult", 3.0),
        ("tp_pct", 0.05),
        ("sl_pct", 0.02),
        ("pos_size_pct", 0.01),
    )

    def __init__(self):
        self.vol_sma = bt.indicators.SimpleMovingAverage(
            self.data.volume, period=self.p.vol_sma_period
        )
        self.order = None
        self.entry_price = None
        self.consolidation_upper = None
        self.consolidation_lower = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            if self.position.size != 0:
                if self.entry_price is None:
                    self.entry_price = order.executed.price
            else:
                self.entry_price = None
                self.consolidation_upper = None
                self.consolidation_lower = None

        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            pass

        self.order = None

    def next(self):
        if self.order:
            return

        if self.position:
            if self.entry_price is None:
                return

            if self.position.size > 0:  # Long position
                # Take profit: 5% from entry
                if self.data.high[0] >= self.entry_price * (1.0 + self.p.tp_pct):
                    self.order = self.close()
                # Stop loss: 2% from entry
                elif self.data.low[0] <= self.entry_price * (1.0 - self.p.sl_pct):
                    self.order = self.close()
                # Close if price closes below lower boundary of breakout consolidation
                elif (
                    self.consolidation_lower is not None
                    and self.data.close[0] < self.consolidation_lower
                ):
                    self.order = self.close()

            elif self.position.size < 0:  # Short position
                # Take profit: 5% from entry
                if self.data.low[0] <= self.entry_price * (1.0 - self.p.tp_pct):
                    self.order = self.close()
                # Stop loss: 2% from entry
                elif self.data.high[0] >= self.entry_price * (1.0 + self.p.sl_pct):
                    self.order = self.close()
                # Close if price closes above upper boundary of breakout consolidation
                elif (
                    self.consolidation_upper is not None
                    and self.data.close[0] > self.consolidation_upper
                ):
                    self.order = self.close()

        else:
            # Evaluate consolidation over the 10 candles prior to current breakout bar
            highs = [
                self.data.high[-i]
                for i in range(1, self.p.consolidation_period + 1)
            ]
            lows = [
                self.data.low[-i]
                for i in range(1, self.p.consolidation_period + 1)
            ]
            upper_boundary = max(highs)
            lower_boundary = min(lows)
            price_range = upper_boundary - lower_boundary

            # Condition 1: Range <= 2% of current price
            is_consolidating = price_range <= (
                self.p.range_pct * self.data.close[0]
            )

            # Condition 2: Volume exceeds 20 SMA volume by at least 3x
            volume_spike = self.data.volume[0] >= (
                self.p.vol_mult * self.vol_sma[0]
            )

            if is_consolidating and volume_spike:
                capital = self.broker.getvalue()
                size = (capital * self.p.pos_size_pct) / self.data.close[0]

                if size > 0:
                    # Condition 3 (Buy): Close above upper boundary
                    if self.data.close[0] > upper_boundary:
                        self.consolidation_upper = upper_boundary
                        self.consolidation_lower = lower_boundary
                        self.order = self.buy(size=size)

                    # Condition 3 (Sell): Close below lower boundary
                    elif self.data.close[0] < lower_boundary:
                        self.consolidation_upper = upper_boundary
                        self.consolidation_lower = lower_boundary
                        self.order = self.sell(size=size)
