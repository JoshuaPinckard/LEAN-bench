import backtrader as bt


class ReferenceStrategy(bt.Strategy):
    params = (
        ("sma_period", 20),
        ("commission", 0.001),
        ("stop_loss_pct", 0.02),
    )

    def __init__(self):
        self.sma = bt.indicators.SMA(self.data.close, period=self.p.sma_period)
        self.order = None
        self.stop_order = None
        self.entry_ym = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            if order.isbuy():
                self.order = None
                entry_price = order.executed.price
                stop_price = entry_price * (1.0 - self.p.stop_loss_pct)
                self.stop_order = self.sell(
                    exectype=bt.Order.Stop,
                    price=stop_price,
                    size=order.executed.size,
                )
                dt = self.data.datetime.date(0)
                self.entry_ym = (dt.year, dt.month)
            else:
                if order == self.stop_order:
                    self.stop_order = None
                if order == self.order:
                    self.order = None

        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            if order == self.order:
                self.order = None
            if order == self.stop_order:
                self.stop_order = None

    def next(self):
        curr_date = self.data.datetime.date(0)
        prev_date = self.data.datetime.date(-1)

        # Exit rule evaluation
        if self.position:
            is_first_on_or_after_5 = (curr_date.day >= 5) and (
                prev_date.month != curr_date.month or prev_date.day < 5
            )
            curr_ym = (curr_date.year, curr_date.month)
            if (
                self.entry_ym
                and curr_ym > self.entry_ym
                and is_first_on_or_after_5
            ):
                if self.order is None:
                    if self.stop_order:
                        self.cancel(self.stop_order)
                        self.stop_order = None
                    self.order = self.close()
            return

        # Entry rule evaluation
        if not self.position and self.order is None:
            is_first_on_or_after_25 = (curr_date.day >= 25) and (
                prev_date.month != curr_date.month or prev_date.day < 25
            )
            if is_first_on_or_after_25:
                if self.data.close[0] > self.sma[0]:
                    cash = self.broker.getcash()
                    price = self.data.close[0]
                    size = int(cash / (price * (1.0 + self.p.commission)))
                    if size > 0:
                        self.order = self.buy(size=size)
