import backtrader as bt


class ATHExitStrategy(bt.Strategy):
    params = dict(
        sma_fast_period=20,
        sma_slow_period=50,
        rsi_period=14,
        volume_period=20,
        volume_threshold=1.2,
        rsi_overbought=70,
        risk_pct=0.02,
    )

    def __init__(self):
        self.sma20 = bt.indicators.SMA(self.data.close, period=self.p.sma_fast_period)
        self.sma50 = bt.indicators.SMA(self.data.close, period=self.p.sma_slow_period)
        self.rsi = bt.indicators.RSI(self.data.close, period=self.p.rsi_period)
        self.vol_avg = bt.indicators.SMA(self.data.volume, period=self.p.volume_period)

        self.order = None
        self.ath = None

        self.in_overbought = False
        self.cur_peak_price = None
        self.cur_peak_rsi = None
        self.prev_peak_price = None
        self.prev_peak_rsi = None
        self.divergence_signal = False

    def _update_ath(self):
        is_new_ath = self.ath is None or self.data.close[0] > self.ath
        if is_new_ath:
            self.ath = self.data.close[0]
        return is_new_ath

    def _update_divergence_state(self):
        self.divergence_signal = False
        if self.rsi[0] > self.p.rsi_overbought:
            if not self.in_overbought:
                self.in_overbought = True
                self.cur_peak_price = self.data.close[0]
                self.cur_peak_rsi = self.rsi[0]
            else:
                if self.data.close[0] > self.cur_peak_price:
                    self.cur_peak_price = self.data.close[0]
                if self.rsi[0] > self.cur_peak_rsi:
                    self.cur_peak_rsi = self.rsi[0]
        else:
            if self.in_overbought:
                self.in_overbought = False
                if (self.prev_peak_price is not None
                        and self.cur_peak_price > self.prev_peak_price
                        and self.cur_peak_rsi < self.prev_peak_rsi):
                    self.divergence_signal = True
                self.prev_peak_price = self.cur_peak_price
                self.prev_peak_rsi = self.cur_peak_rsi

    def prenext(self):
        self._update_ath()

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        is_new_ath = self._update_ath()
        self._update_divergence_state()

        if self.order:
            return

        if not self.position:
            if is_new_ath:
                cash_alloc = self.p.risk_pct * self.broker.getvalue()
                size = int(cash_alloc / self.data.close[0])
                if size > 0:
                    self.order = self.buy(size=size)
            return

        volume_confirmed = self.data.volume[0] > self.p.volume_threshold * self.vol_avg[0]
        stop_loss = self.data.close[0] < self.sma50[0]
        main_signal = self.data.close[0] < self.sma20[0] and volume_confirmed
        reversal_signal = self.divergence_signal

        if stop_loss or main_signal or reversal_signal:
            self.order = self.close()
            return

        if is_new_ath:
            partial_size = int(self.position.size / 2)
            if partial_size > 0:
                self.order = self.sell(size=partial_size)
