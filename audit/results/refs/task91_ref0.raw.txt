Implementing the significant-move breakout/trend strategy with ATR-based sizing and the same-direction re-entry lock.

```python
import backtrader as bt


class SignificantMoveStrategy(bt.Strategy):
    params = dict(
        atr_period=14,
        ema_period=50,
        move_mult=1.5,
        tp_mult=2.0,
        sl_mult=1.0,
        risk_pct=0.01,
    )

    def __init__(self):
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.ema = bt.indicators.EMA(self.data, period=self.p.ema_period)

        self.order = None
        self.pending_direction = None
        self.pending_atr = None

        self.entry_price = None
        self.entry_atr = None

        self.last_entry_direction = None
        self.direction_unlocked = True

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status == order.Completed:
            if self.pending_direction is not None:
                self.entry_price = order.executed.price
                self.entry_atr = self.pending_atr
                self.last_entry_direction = self.pending_direction
                self.direction_unlocked = False

        self.order = None
        self.pending_direction = None
        self.pending_atr = None

    def next(self):
        atr = self.atr[0]
        ema = self.ema[0]
        close = self.data.close[0]
        prev_close = self.data.close[-1]

        if self.last_entry_direction == 'long' and close < ema:
            self.direction_unlocked = True
        elif self.last_entry_direction == 'short' and close > ema:
            self.direction_unlocked = True

        if self.order:
            return

        if not self.position:
            up_move = (close - prev_close) >= self.p.move_mult * atr
            down_move = (prev_close - close) >= self.p.move_mult * atr

            long_signal = up_move and close > ema
            short_signal = down_move and close < ema

            long_allowed = self.last_entry_direction != 'long' or self.direction_unlocked
            short_allowed = self.last_entry_direction != 'short' or self.direction_unlocked

            risk_amount = self.p.risk_pct * self.broker.getvalue()
            stop_distance = self.p.sl_mult * atr

            if long_signal and long_allowed:
                size = int(risk_amount / stop_distance)
                if size > 0:
                    self.pending_direction = 'long'
                    self.pending_atr = atr
                    self.order = self.buy(size=size)
            elif short_signal and short_allowed:
                size = int(risk_amount / stop_distance)
                if size > 0:
                    self.pending_direction = 'short'
                    self.pending_atr = atr
                    self.order = self.sell(size=size)
        else:
            if self.position.size > 0:
                profit = close - self.entry_price
                loss = self.entry_price - close
                if profit >= self.p.tp_mult * self.entry_atr:
                    self.order = self.close()
                elif loss >= self.p.sl_mult * self.entry_atr:
                    self.order = self.close()
                elif close < ema:
                    self.order = self.close()
            elif self.position.size < 0:
                profit = self.entry_price - close
                loss = close - self.entry_price
                if profit >= self.p.tp_mult * self.entry_atr:
                    self.order = self.close()
                elif loss >= self.p.sl_mult * self.entry_atr:
                    self.order = self.close()
                elif close > ema:
                    self.order = self.close()
```
