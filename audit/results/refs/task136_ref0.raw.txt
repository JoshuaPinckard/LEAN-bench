import backtrader as bt


class NvdaBreakoutEmaTrail(bt.Strategy):
    params = (
        ('ema_period', 8),
        ('resistance', 179.33),
        ('tp1', 184.89),
        ('tp2', 192.95),
        ('risk_pct', 0.02),
    )

    def __init__(self):
        # Indicator: 8-period Exponential Moving Average
        self.ema = bt.indicators.ExponentialMovingAverage(
            self.data.close, period=self.p.ema_period)
        self.order = None
        self.trail_stop = None      # ratcheting trailing-stop level
        self.took_partial = False   # whether the 50% take-profit was taken

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled,
                            order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        # Guard against pending orders
        if self.order:
            return

        # ---- Entry ----
        if not self.position:
            # Buy if the current bar's close exceeds the resistance level
            if self.data.close[0] > self.p.resistance:
                # Trailing-stop distance = value of the 8-period EMA, so the
                # per-share risk (entry - stop) equals that same distance.
                risk_per_share = self.ema[0]
                if risk_per_share > 0:
                    cash_risk = self.p.risk_pct * self.broker.getvalue()
                    # Position size so that potential loss <= 2% of capital
                    size = int(cash_risk / risk_per_share)
                    if size > 0:
                        self.order = self.buy(size=size)
                        self.trail_stop = None
                        self.took_partial = False
            return

        # ---- In a position ----
        # Update the trailing stop: it lags the price by a distance equal to
        # the EMA value and never falls below its previous value.
        new_stop = self.data.close[0] - self.ema[0]
        if self.trail_stop is None or new_stop > self.trail_stop:
            self.trail_stop = new_stop

        # Trailing stop hit -> exit the whole remaining position
        if self.data.low[0] <= self.trail_stop:
            self.order = self.close()
            return

        # Final take-profit -> close the remaining part of the position
        if self.data.high[0] >= self.p.tp2:
            self.order = self.close()
            return

        # First take-profit -> close 50% of the position
        if not self.took_partial and self.data.high[0] >= self.p.tp1:
            half = int(abs(self.position.size) * 0.5)
            if half > 0:
                self.order = self.sell(size=half)
            self.took_partial = True
            return
