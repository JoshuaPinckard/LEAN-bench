import backtrader as bt


class NvdaEmaBreakoutStrategy(bt.Strategy):
    params = dict(
        ema_period=8,
        entry_level=179.33,
        tp1_price=184.89,
        tp2_price=192.95,
        risk_pct=0.02,
    )

    def __init__(self):
        self.ema = bt.indicators.EMA(self.data.close, period=self.p.ema_period)
        self.order = None
        self.stop_price = None
        self.tp1_done = False
        self.entry_size = None

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return
        if order.status == order.Completed and order.isbuy():
            self.entry_size = order.executed.size
        self.order = None

    def next(self):
        if self.order:
            return

        if not self.position:
            self.stop_price = None
            self.tp1_done = False
            self.entry_size = None
            if self.data.close[0] > self.p.entry_level:
                # per-share risk equals the trailing stop's initial distance (the EMA value)
                risk_amount = self.p.risk_pct * self.broker.getvalue()
                size = int(risk_amount / self.ema[0])
                if size > 0:
                    self.order = self.buy(size=size)
            return

        candidate_stop = self.data.close[0] - self.ema[0]
        self.stop_price = candidate_stop if self.stop_price is None else max(self.stop_price, candidate_stop)

        if not self.tp1_done and self.data.high[0] >= self.p.tp1_price:
            self.tp1_done = True
            close_size = self.entry_size // 2
            if close_size > 0:
                self.order = self.sell(size=close_size)
        elif self.tp1_done and self.data.high[0] >= self.p.tp2_price:
            self.order = self.close()
        elif self.data.low[0] <= self.stop_price:
            self.order = self.close()
