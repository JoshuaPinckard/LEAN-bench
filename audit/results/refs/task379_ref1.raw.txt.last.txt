import backtrader as bt


class SMASlopeReversalStrategy(bt.Strategy):
    params = (
        ("sma_period", 20),
        ("slope_lag", 20),
        ("std_lookback", 100),
        ("extreme_mult", 2.0),
        ("stop_pct", 0.02),
        ("neutral_mult", 0.2),
    )

    def __init__(self):
        self.order = None
        self.stop_order = None

        self.sma20 = bt.indicators.SMA(
            self.data.close, period=self.p.sma_period
        )
        self.slope = (
            self.sma20 - self.sma20(-self.p.slope_lag)
        ) / self.p.slope_lag
        self.slope_std = bt.indicators.StdDev(
            self.slope, period=self.p.std_lookback
        )

        self.pos_th = self.p.extreme_mult * self.slope_std
        self.neg_th = -self.p.extreme_mult * self.slope_std
        self.neutral_band = self.p.neutral_mult * self.slope_std

    def notify_order(self, order):
        if order.status in (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            if order == self.order:
                self.order = None

            if order == self.stop_order:
                self.stop_order = None

            if order.status == order.Completed and order.isbuy():
                self.stop_order = self.sell(
                    exectype=bt.Order.Stop,
                    price=order.executed.price * (1.0 - self.p.stop_pct),
                )

    def next(self):
        if self.order:
            return

        slope_crosses_zero = (
            (self.slope[-1] <= 0 and self.slope[0] > 0)
            or (self.slope[-1] >= 0 and self.slope[0] < 0)
        )

        if self.position:
            if (
                slope_crosses_zero
                or abs(self.slope[0]) < self.neutral_band[0]
            ):
                if self.stop_order:
                    self.cancel(self.stop_order)
                self.order = self.close()
            return

        long_signal = (
            self.slope[-1] <= self.neg_th[-1]
            and self.slope[0] > self.neg_th[0]
        )
        short_signal = (
            self.slope[-1] >= self.pos_th[-1]
            and self.slope[0] < self.pos_th[0]
        )

        if long_signal:
            self.order = self.buy()
        elif short_signal:
            self.order = self.sell()
