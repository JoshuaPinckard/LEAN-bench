import backtrader as bt


class EMA5BreakoutStrategy(bt.Strategy):
    params = (
        ('ema_period', 5),
        ('risk_pct', 0.05),
        ('rr_ratio', 3.0),
    )

    def __init__(self):
        self.ema = bt.indicators.EMA(self.data.close, period=self.p.ema_period)
        self.entry_order = None
        self.sl_order = None
        self.tp_order = None
        self.pending_setup_high = None
        self.pending_setup_low = None
        self.pending_dir = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            if order == self.entry_order:
                self.entry_order = None
                entry_price = order.executed.price
                exec_size = order.executed.size

                if self.pending_dir == 'LONG':
                    stop_price = self.pending_setup_low
                    r_dist = entry_price - stop_price
                    tp_price = entry_price + self.p.rr_ratio * r_dist
                    self.sl_order = self.sell(
                        exectype=bt.Order.Stop,
                        price=stop_price,
                        size=exec_size
                    )
                    self.tp_order = self.sell(
                        exectype=bt.Order.Limit,
                        price=tp_price,
                        size=exec_size,
                        oco=self.sl_order
                    )
                elif self.pending_dir == 'SHORT':
                    stop_price = self.pending_setup_high
                    r_dist = stop_price - entry_price
                    tp_price = entry_price - self.p.rr_ratio * r_dist
                    self.sl_order = self.buy(
                        exectype=bt.Order.Stop,
                        price=stop_price,
                        size=abs(exec_size)
                    )
                    self.tp_order = self.buy(
                        exectype=bt.Order.Limit,
                        price=tp_price,
                        size=abs(exec_size),
                        oco=self.sl_order
                    )
            elif order in [self.sl_order, self.tp_order]:
                self.sl_order = None
                self.tp_order = None

        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            if order == self.entry_order:
                self.entry_order = None
            elif order in [self.sl_order, self.tp_order]:
                self.sl_order = None
                self.tp_order = None

    def next(self):
        if self.position:
            if self.entry_order:
                self.cancel(self.entry_order)
                self.entry_order = None
            return

        is_long_setup = self.data.high[0] < self.ema[0]
        is_short_setup = self.data.low[0] > self.ema[0]

        if is_long_setup:
            if self.entry_order:
                self.cancel(self.entry_order)
                self.entry_order = None

            setup_high = self.data.high[0]
            setup_low = self.data.low[0]
            stop_dist = setup_high - setup_low

            if stop_dist > 0:
                equity = self.broker.getvalue()
                size = (self.p.risk_pct * equity) / stop_dist
                self.pending_setup_high = setup_high
                self.pending_setup_low = setup_low
                self.pending_dir = 'LONG'
                self.entry_order = self.buy(
                    exectype=bt.Order.Stop,
                    price=setup_high,
                    size=size
                )

        elif is_short_setup:
            if self.entry_order:
                self.cancel(self.entry_order)
                self.entry_order = None

            setup_high = self.data.high[0]
            setup_low = self.data.low[0]
            stop_dist = setup_high - setup_low

            if stop_dist > 0:
                equity = self.broker.getvalue()
                size = (self.p.risk_pct * equity) / stop_dist
                self.pending_setup_high = setup_high
                self.pending_setup_low = setup_low
                self.pending_dir = 'SHORT'
                self.entry_order = self.sell(
                    exectype=bt.Order.Stop,
                    price=setup_low,
                    size=size
                )
