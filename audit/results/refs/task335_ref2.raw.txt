import backtrader as bt


class IntradayVWAP(bt.Indicator):
    lines = ('vwap',)

    def __init__(self):
        self.addminperiod(1)

    def next(self):
        tp = (self.data.high[0] + self.data.low[0] + self.data.close[0]) / 3.0
        v = self.data.volume[0]

        if len(self) == 1 or self.data.datetime.date(0) != self.data.datetime.date(-1):
            self.cum_pv = tp * v
            self.cum_v = v
        else:
            self.cum_pv += tp * v
            self.cum_v += v

        self.lines.vwap[0] = self.cum_pv / self.cum_v if self.cum_v != 0 else tp


class VWAPMeanReversion(bt.Strategy):
    params = (
        ('threshold_multiplier', 2.0),
        ('stop_loss_pct', 0.02),
        ('period', 20),
    )

    def __init__(self):
        self.vwap = IntradayVWAP(self.data)
        self.slope = self.vwap - self.vwap(-self.p.period)
        self.slope_std = bt.ind.StdDev(self.slope, period=self.p.period)
        self.threshold = self.p.threshold_multiplier * self.slope_std
        self.vwap_base = bt.ind.SMA(self.vwap, period=self.p.period)
        self.crossover = bt.ind.CrossOver(self.vwap, self.vwap_base)

        self.order = None
        self.entry_price = None

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

        if order.status == order.Completed:
            self.entry_price = order.executed.price

    def next(self):
        if self.order:
            return

        if not self.position:
            if self.slope[-1] < -self.threshold[-1] and self.slope[0] > self.slope[-1]:
                self.order = self.buy()
            elif self.slope[-1] > self.threshold[-1] and self.slope[0] < self.slope[-1]:
                self.order = self.sell()
        else:
            if self.position.size > 0:
                if self.entry_price and self.data.low[0] <= self.entry_price * (1.0 - self.p.stop_loss_pct):
                    self.order = self.close()
                elif self.crossover[0] > 0:
                    self.order = self.close()
            elif self.position.size < 0:
                if self.entry_price and self.data.high[0] >= self.entry_price * (1.0 + self.p.stop_loss_pct):
                    self.order = self.close()
                elif self.crossover[0] < 0:
                    self.order = self.close()
