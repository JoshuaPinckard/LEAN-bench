import backtrader as bt


class BollingerATRBreakoutStrategy(bt.Strategy):
    params = (
        ('bb_period', 20),
        ('bb_devfactor', 2.0),
        ('atr_period', 14),
        ('atr_sma_period', 20),
        ('atr_mult', 2.0),
    )

    def __init__(self):
        self.bb = bt.indicators.BollingerBands(
            self.data.close,
            period=self.p.bb_period,
            devfactor=self.p.bb_devfactor
        )
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.atr_sma = bt.indicators.SMA(self.atr, period=self.p.atr_sma_period)

        self.order = None
        self.entry_price = None
        self.stop_price = None
        self.highest_close = None
        self.lowest_close = None

    def notify_order(self, order):
        if order.status in [order.Completed]:
            if self.position.size > 0:
                if self.stop_price is None:
                    self.entry_price = order.executed.price
                    self.stop_price = self.entry_price - self.p.atr_mult * self.atr[0]
                    self.highest_close = self.data.close[0]
            elif self.position.size < 0:
                if self.stop_price is None:
                    self.entry_price = order.executed.price
                    self.stop_price = self.entry_price + self.p.atr_mult * self.atr[0]
                    self.lowest_close = self.data.close[0]
            else:
                self.entry_price = None
                self.stop_price = None
                self.highest_close = None
                self.lowest_close = None
            self.order = None
        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        if self.order:
            return

        vol_filter = self.atr[0] > self.atr_sma[0]

        close_cross_above_upper = (self.data.close[-1] <= self.bb.lines.top[-1]) and (self.data.close[0] > self.bb.lines.top[0])
        close_cross_below_lower = (self.data.close[-1] >= self.bb.lines.bot[-1]) and (self.data.close[0] < self.bb.lines.bot[0])

        if not self.position:
            if close_cross_above_upper and vol_filter:
                self.order = self.buy()
            elif close_cross_below_lower and vol_filter:
                self.order = self.sell()
        else:
            if self.position.size > 0:
                if self.highest_close is None:
                    self.highest_close = self.data.close[0]
                else:
                    self.highest_close = max(self.highest_close, self.data.close[0])

                candidate_stop = self.highest_close - self.p.atr_mult * self.atr[0]
                if self.stop_price is None:
                    self.stop_price = candidate_stop
                else:
                    self.stop_price = max(self.stop_price, candidate_stop)

                stop_hit = self.data.low[0] <= self.stop_price
                opposite_signal = close_cross_below_lower and vol_filter

                if stop_hit or opposite_signal:
                    self.order = self.close()

            elif self.position.size < 0:
                if self.lowest_close is None:
                    self.lowest_close = self.data.close[0]
                else:
                    self.lowest_close = min(self.lowest_close, self.data.close[0])

                candidate_stop = self.lowest_close + self.p.atr_mult * self.atr[0]
                if self.stop_price is None:
                    self.stop_price = candidate_stop
                else:
                    self.stop_price = min(self.stop_price, candidate_stop)

                stop_hit = self.data.high[0] >= self.stop_price
                opposite_signal = close_cross_above_upper and vol_filter

                if stop_hit or opposite_signal:
                    self.order = self.close()
