import backtrader as bt


class FibHeikinAshiStrategy(bt.Strategy):
    params = (
        ('lookback', 20),
    )

    def __init__(self):
        self.order = None
        self.ha_open_prev = None
        self.ha_close_prev = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        o = self.data.open[0]
        h = self.data.high[0]
        l = self.data.low[0]
        c = self.data.close[0]

        if self.ha_open_prev is None:
            ha_open = (o + c) / 2.0
        else:
            ha_open = (self.ha_open_prev + self.ha_close_prev) / 2.0

        ha_close = (o + h + l + c) / 4.0
        ha_high = max(h, ha_open, ha_close)
        ha_low = min(l, ha_open, ha_close)

        self.ha_open_prev = ha_open
        self.ha_close_prev = ha_close

        if len(self) < self.p.lookback:
            return

        highs = [self.data.high[-i] for i in range(self.p.lookback)]
        lows = [self.data.low[-i] for i in range(self.p.lookback)]

        max_high = max(highs)
        min_low = min(lows)

        max_idx = highs.index(max_high)
        min_idx = lows.index(min_low)

        if max_high == min_low:
            return

        rng = max_high - min_low

        if max_idx > min_idx:
            # Downward move prior to current bar (max_high occurred before min_low)
            swing_high = max_high
            swing_low = min_low
            tp_price = swing_high
            sl_price = swing_high - 0.618 * rng

            upper_zone = swing_high - 0.236 * rng

            if sl_price < c <= upper_zone:
                # Bullish Heikin Ashi candle with no lower wick
                if ha_close > ha_open and abs(ha_low - ha_open) < 1e-5:
                    if not self.position and self.order is None:
                        orders = self.buy_bracket(limitprice=tp_price, stopprice=sl_price)
                        self.order = orders[0]

        elif min_idx > max_idx:
            # Upward move prior to current bar (min_low occurred before max_high)
            swing_high = max_high
            swing_low = min_low
            tp_price = swing_low
            sl_price = swing_low + 0.618 * rng

            lower_zone = swing_low + 0.236 * rng

            if lower_zone <= c < sl_price:
                # Bearish Heikin Ashi candle with no upper wick
                if ha_close < ha_open and abs(ha_high - ha_open) < 1e-5:
                    if not self.position and self.order is None:
                        orders = self.sell_bracket(limitprice=tp_price, stopprice=sl_price)
                        self.order = orders[0]
