import backtrader as bt
import math
from collections import deque


class FibonacciRSIStrategy(bt.Strategy):
    params = (
        ("rsi_period", 14),
        ("swing_lookback", 50),
        ("risk_fraction", 0.01),
    )

    def __init__(self):
        self.rsi = bt.indicators.RSI(self.data.close, period=self.p.rsi_period)

        self.order = None
        self.stop_order = None
        self.target_order = None
        self.entry_direction = None
        self.stop_price = None
        self.target_price = None

        self.highs = deque(maxlen=self.p.swing_lookback)
        self.lows = deque(maxlen=self.p.swing_lookback)

        self.current_day = None
        self.current_day_close = None
        self.last_daily_close = None
        self.previous_daily_close = None

    def _update_daily_closes(self):
        bar_day = self.data.datetime.date(0)

        if self.current_day is None:
            self.current_day = bar_day
        elif bar_day != self.current_day:
            self.previous_daily_close = self.last_daily_close
            self.last_daily_close = self.current_day_close
            self.current_day = bar_day

        self.current_day_close = self.data.close[0]

    def _daily_trend(self):
        if self.last_daily_close is None or self.previous_daily_close is None:
            return 0
        if self.last_daily_close > self.previous_daily_close:
            return 1
        if self.last_daily_close < self.previous_daily_close:
            return -1
        return 0

    def _fib_levels(self):
        high_values = list(self.highs)
        low_values = list(self.lows)

        swing_high = max(high_values)
        swing_low = min(low_values)
        price_range = swing_high - swing_low

        if price_range <= 0:
            return None

        high_index = high_values.index(swing_high)
        low_index = low_values.index(swing_low)
        levels = {
            0.0: swing_high,
            0.236: swing_high - price_range * 0.236,
            0.382: swing_high - price_range * 0.382,
            0.5: swing_high - price_range * 0.5,
            0.618: swing_high - price_range * 0.618,
            0.786: swing_high - price_range * 0.786,
            1.0: swing_low,
        }
        return levels, low_index < high_index, high_index < low_index

    def _nearest_low_below(self, entry_price):
        eligible = [value for value in self.lows if value < entry_price]
        return max(eligible) if eligible else None

    def _nearest_high_above(self, entry_price):
        eligible = [value for value in self.highs if value > entry_price]
        return min(eligible) if eligible else None

    def _cancel_exit_orders(self):
        for exit_order in (self.stop_order, self.target_order):
            if exit_order is not None and exit_order.alive():
                self.cancel(exit_order)

    def _size_for_risk(self, entry_price, stop_price):
        risk_per_share = abs(entry_price - stop_price)
        if risk_per_share <= 0:
            return 0

        risk_capital = self.broker.getvalue() * self.p.risk_fraction
        return int(math.floor(risk_capital / risk_per_share))

    def next(self):
        self._update_daily_closes()

        self.highs.append(self.data.high[0])
        self.lows.append(self.data.low[0])

        if len(self.highs) < self.p.swing_lookback:
            return

        if self.order is not None:
            return

        if self.position:
            if self.position.size > 0:
                if self.rsi[-1] >= 50 and self.rsi[0] < 50:
                    self._cancel_exit_orders()
                    self.order = self.close(exectype=bt.Order.Close)
            else:
                if self.rsi[-1] <= 50 and self.rsi[0] > 50:
                    self._cancel_exit_orders()
                    self.order = self.close(exectype=bt.Order.Close)
            return

        fib_data = self._fib_levels()
        if fib_data is None:
            return

        levels, upward_move, downward_move = fib_data
        trend = self._daily_trend()
        bar_low = self.data.low[0]
        bar_high = self.data.high[0]
        close = self.data.close[0]

        if upward_move and trend == 1 and self.rsi[0] < 30 and self.rsi[0] > self.rsi[-1]:
            for ratio in (0.382, 0.618):
                level = levels[ratio]
                if bar_low <= level <= bar_high and close > level:
                    candidates = sorted(
                        value for value in levels.values() if value > level
                    )
                    target = candidates[0] if candidates else levels[0.0]
                    nearest_low = self._nearest_low_below(close)
                    fib_stop = levels[0.786]
                    stop_candidates = [
                        value for value in (nearest_low, fib_stop) if value < close
                    ]
                    if not stop_candidates:
                        return
                    stop = max(stop_candidates)
                    size = self._size_for_risk(close, stop)
                    if size > 0:
                        self.entry_direction = 1
                        self.stop_price = stop
                        self.target_price = target
                        self.order = self.buy(size=size, exectype=bt.Order.Close)
                    return

        if downward_move and trend == -1 and self.rsi[0] > 70 and self.rsi[0] < self.rsi[-1]:
            for ratio in (0.382, 0.618):
                level = levels[ratio]
                if bar_low <= level <= bar_high and close < level:
                    candidates = sorted(
                        (value for value in levels.values() if value < level),
                        reverse=True,
                    )
                    target = candidates[0] if candidates else levels[1.0]
                    nearest_high = self._nearest_high_above(close)
                    fib_stop = levels[0.236]
                    stop_candidates = [
                        value for value in (nearest_high, fib_stop) if value > close
                    ]
                    if not stop_candidates:
                        return
                    stop = min(stop_candidates)
                    size = self._size_for_risk(close, stop)
                    if size > 0:
                        self.entry_direction = -1
                        self.stop_price = stop
                        self.target_price = target
                        self.order = self.sell(size=size, exectype=bt.Order.Close)
                    return

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return

        if order.status == order.Completed:
            if order == self.order:
                self.order = None

                if order.isbuy() and self.entry_direction == 1:
                    self.stop_order = self.sell(
                        size=order.executed.size,
                        exectype=bt.Order.Stop,
                        price=self.stop_price,
                    )
                    self.target_order = self.sell(
                        size=order.executed.size,
                        exectype=bt.Order.Limit,
                        price=self.target_price,
                    )
                elif order.issell() and self.entry_direction == -1:
                    self.stop_order = self.buy(
                        size=abs(order.executed.size),
                        exectype=bt.Order.Stop,
                        price=self.stop_price,
                    )
                    self.target_order = self.buy(
                        size=abs(order.executed.size),
                        exectype=bt.Order.Limit,
                        price=self.target_price,
                    )
                else:
                    self.entry_direction = None
                    self.stop_price = None
                    self.target_price = None

            elif order == self.stop_order or order == self.target_order:
                other_order = (
                    self.target_order if order == self.stop_order else self.stop_order
                )
                if other_order is not None and other_order.alive():
                    self.cancel(other_order)
                self.stop_order = None
                self.target_order = None
                self.entry_direction = None
                self.stop_price = None
                self.target_price = None

        elif order.status in (order.Canceled, order.Margin, order.Rejected):
            if order == self.order:
                self.order = None
                self.entry_direction = None
                self.stop_price = None
                self.target_price = None
            elif order == self.stop_order:
                self.stop_order = None
            elif order == self.target_order:
                self.target_order = None
