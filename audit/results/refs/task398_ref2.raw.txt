import math
import backtrader as bt
from sklearn.ensemble import RandomForestClassifier


class MLRsiStrategy(bt.Strategy):
    params = (
        ("rsi_period", 2),
        ("warmup_bars", 50),
        ("train_window", 200),
        ("random_state", 42),
        ("n_estimators", 100),
    )

    def __init__(self):
        self.rsi = bt.indicators.RSI(self.data.close, period=self.p.rsi_period)
        self.order = None

    def notify_order(self, order):
        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            self.order = None

    def next(self):
        if self.order:
            return

        if len(self) < self.p.warmup_bars:
            # Fallback RSI rule
            if self.rsi[0] < 30:
                order = self.order_target_percent(target=1.0)
                if order:
                    self.order = order
            elif self.rsi[0] > 70:
                order = self.order_target_percent(target=0.0)
                if order:
                    self.order = order
        else:
            # ML Model training and prediction
            L = len(self)
            start_offset = max(-(L - 2), -self.p.train_window)

            X_train = []
            y_train = []

            for i in range(start_offset, 0):
                rsi_val = float(self.rsi[i])
                ret_val = (
                    self.data.close[i] - self.data.close[i - 1]
                ) / self.data.close[i - 1]

                if math.isnan(rsi_val) or math.isnan(ret_val):
                    continue

                target = 1 if self.data.close[i + 1] > self.data.close[i] else 0
                X_train.append([rsi_val, ret_val])
                y_train.append(target)

            if not X_train:
                return

            clf = RandomForestClassifier(
                n_estimators=self.p.n_estimators,
                random_state=self.p.random_state,
            )
            clf.fit(X_train, y_train)

            current_rsi = float(self.rsi[0])
            current_ret = (
                self.data.close[0] - self.data.close[-1]
            ) / self.data.close[-1]

            pred = clf.predict([[current_rsi, current_ret]])[0]

            if pred == 1:
                order = self.order_target_percent(target=1.0)
                if order:
                    self.order = order
            elif pred == 0:
                order = self.order_target_percent(target=0.0)
                if order:
                    self.order = order
