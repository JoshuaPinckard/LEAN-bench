import backtrader as bt


class MeanReversionDynamicThreshold(bt.Strategy):
    params = (
        ('bb_period', 30),
        ('bb_dev', 2.0),
        ('atr_period', 14),
        ('ema_atr_period', 50),
        ('base_threshold', 2.0),
        ('vr_min', 0.7),
        ('vr_max', 1.6),
        ('rsi_period', 14),
        ('rsi_low', 45.0),
        ('rsi_high', 55.0),
        ('vol_sma_period', 20),
        ('risk_pct', 0.01),
        ('stop_mult', 1.5),
        ('time_stop_bars', 7),
        ('use_optional_filters', True),
    )

    def __init__(self):
        self.bb = bt.indicators.BollingerBands(
            self.data.close,
            period=self.p.bb_period,
            devfactor=self.p.bb_dev
        )
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.ema_atr = bt.indicators.EMA(self.atr, period=self.p.ema_atr_period)
        self.rsi = bt.indicators.RSI(self.data.close, period=self.p.rsi_period)
        self.vol_sma = bt.indicators.SMA(self.data.volume, period=self.p.vol_sma_period)

        self.order = None
        self.entry_price = 0.0
        self.entry_bar = 0
        self.stop_dist = 0.0

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            if self.position:
                self.entry_price = order.executed.price
                self.entry_bar = len(self)

        self.order = None

    def next(self):
        if self.order:
            return

        mid = self.bb.mid[0]
        top = self.bb.top[0]
        std = (top - mid) / 2.0
        if std == 0:
            return

        z = (self.data.close[0] - mid) / std

        atr_val = self.atr[0]
        ema_atr_val = self.ema_atr[0]
        if ema_atr_val == 0:
            return

        vr = atr_val / ema_atr_val
        vr_clipped = min(max(vr, self.p.vr_min), self.p.vr_max)
        threshold = self.p.base_threshold * vr_clipped

        if not self.position:
            rsi_val = self.rsi[0]
            vol_sma_val = self.vol_sma[0]

            long_cond = z < -threshold
            short_cond = z > threshold

            if self.p.use_optional_filters:
                long_cond = long_cond and (rsi_val < self.p.rsi_low) and (self.data.volume[0] > vol_sma_val)
                short_cond = short_cond and (rsi_val > self.p.rsi_high) and (self.data.volume[0] > vol_sma_val)

            if long_cond:
                risk_amount = self.broker.getvalue() * self.p.risk_pct
                risk_per_share = self.p.stop_mult * atr_val
                if risk_per_share > 0:
                    size = int(risk_amount / risk_per_share)
                    if size > 0:
                        self.stop_dist = risk_per_share
                        self.order = self.buy(size=size)

            elif short_cond:
                risk_amount = self.broker.getvalue() * self.p.risk_pct
                risk_per_share = self.p.stop_mult * atr_val
                if risk_per_share > 0:
                    size = int(risk_amount / risk_per_share)
                    if size > 0:
                        self.stop_dist = risk_per_share
                        self.order = self.sell(size=size)

        else:
            bars_held = len(self) - self.entry_bar
            time_exit = bars_held >= self.p.time_stop_bars

            if self.position.size > 0:
                stop_loss = self.data.close[0] <= (self.entry_price - self.stop_dist)
                mean_reversion = z >= 0
                if mean_reversion or stop_loss or time_exit:
                    self.order = self.close()

            elif self.position.size < 0:
                stop_loss = self.data.close[0] >= (self.entry_price + self.stop_dist)
                mean_reversion = z <= 0
                if mean_reversion or stop_loss or time_exit:
                    self.order = self.close()
