import backtrader as bt


class PseudoForecastTrendStrategy(bt.Strategy):
    params = (
        ("macd_fast", 12),
        ("macd_slow", 26),
        ("macd_signal", 9),
        ("sma_fast", 20),
        ("sma_slow", 50),
        ("ema_period", 10),
        ("rsi_period", 14),
        ("bb_period", 20),
        ("bb_dev", 2.0),
        ("stoch_period", 14),
        ("stoch_dfast", 3),
        ("stoch_dslow", 3),
    )

    def __init__(self):
        self.order = None

        # 1. MACD(12, 26, 9)
        self.macd = bt.indicators.MACDHisto(
            self.data,
            period_me1=self.p.macd_fast,
            period_me2=self.p.macd_slow,
            period_signal=self.p.macd_signal,
        )

        # 2. Moving Averages: SMA(20), SMA(50), EMA(10)
        self.sma20 = bt.indicators.SMA(self.data, period=self.p.sma_fast)
        self.sma50 = bt.indicators.SMA(self.data, period=self.p.sma_slow)
        self.ema10 = bt.indicators.EMA(self.data, period=self.p.ema_period)

        # 3. RSI(14)
        self.rsi = bt.indicators.RSI(self.data, period=self.p.rsi_period)

        # 4. Bollinger Bands(20, 2)
        self.bb = bt.indicators.BollingerBands(
            self.data, period=self.p.bb_period, devfactor=self.p.bb_dev
        )

        # 5. Stochastic(14, 3, 3)
        self.stoch = bt.indicators.Stochastic(
            self.data,
            period=self.p.stoch_period,
            period_dfast=self.p.stoch_dfast,
            period_dslow=self.p.stoch_dslow,
        )
        self.stoch_cross = bt.indicators.CrossOver(
            self.stoch.percK, self.stoch.percD
        )

    def notify_order(self, order):
        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            self.order = None

    def next(self):
        if self.order:
            return

        # Score 1: MACD Histogram
        if self.macd.histo[0] > 0:
            c1 = 1
        elif self.macd.histo[0] < 0:
            c1 = -1
        else:
            c1 = 0

        # Score 2: Moving Average Slopes and Alignment
        sma20_rising = self.sma20[0] > self.sma20[-1]
        ema10_rising = self.ema10[0] > self.ema10[-1]
        sma20_falling = self.sma20[0] < self.sma20[-1]
        ema10_falling = self.ema10[0] < self.ema10[-1]

        if self.sma20[0] > self.sma50[0] and sma20_rising and ema10_rising:
            c2 = 1
        elif self.sma20[0] < self.sma50[0] and sma20_falling and ema10_falling:
            c2 = -1
        else:
            c2 = 0

        # Score 3: RSI
        if self.rsi[0] > 55:
            c3 = 1
        elif self.rsi[0] < 45:
            c3 = -1
        else:
            c3 = 0

        # Score 4: Bollinger Bands
        width_today = self.bb.top[0] - self.bb.bot[0]
        width_yesterday = self.bb.top[-1] - self.bb.bot[-1]
        width_increasing = width_today > width_yesterday

        if self.data.close[0] > self.bb.mid[0] and width_increasing:
            c4 = 1
        elif self.data.close[0] < self.bb.mid[0] and width_increasing:
            c4 = -1
        else:
            c4 = 0

        # Score 5: Stochastic
        if self.stoch_cross[0] > 0 and self.stoch.percK[0] > 20:
            c5 = 1
        elif self.stoch_cross[0] < 0 and self.stoch.percK[0] < 80:
            c5 = -1
        else:
            c5 = 0

        score = c1 + c2 + c3 + c4 + c5

        pos_size = self.position.size
        capital = self.broker.get_value()
        target_qty = int(capital / (self.data.close[0] * 1.001))

        if target_qty <= 0:
            return

        if score >= 2:
            if pos_size < 0:
                self.order = self.buy(size=abs(pos_size) + target_qty)
            elif pos_size == 0:
                self.order = self.buy(size=target_qty)
        elif score <= -2:
            if pos_size > 0:
                self.order = self.sell(size=pos_size + target_qty)
            elif pos_size == 0:
                self.order = self.sell(size=target_qty)
