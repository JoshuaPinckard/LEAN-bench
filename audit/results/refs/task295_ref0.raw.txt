import backtrader as bt


class DividendPitchforkStrategist(bt.Strategy):

    def __init__(self):
        self.bb = bt.indicators.BollingerBands(self.data.close, period=20, devfactor=2)
        self.rsi = bt.indicators.RSI(self.data.close, period=14)
        self.atr = bt.indicators.ATR(self.data, period=14)
        self.sma50 = bt.indicators.SMA(self.data.close, period=50)

        self.order = None
        self.state = None
        self.entry_bar = None
        self.stop_price = None
        self.target_price = None

    def notify_order(self, order):
        if order.status in (order.Completed, order.Canceled, order.Margin, order.Rejected):
            self.order = None

    def _calc_entry_size(self):
        value = self.broker.getvalue()
        cash = self.broker.getcash()
        price = self.data.close[0]
        commission = 0.001
        cap = 0.5 * value
        size = int(cap // price)
        while size > 0:
            invested = size * price
            cost = invested * (1 + commission)
            if invested <= cap and (cash - cost) >= cap:
                break
            size -= 1
        return max(size, 0)

    def _enter_position(self):
        size = self._calc_entry_size()
        if size <= 0:
            return
        price = self.data.close[0]
        atr_now = self.atr[0]
        self.order = self.buy(size=size)
        self.stop_price = price - 1.5 * atr_now
        self.target_price = price + 1.0 * atr_now
        self.entry_bar = len(self)
        self.state = 'putsell'

    def next(self):
        if self.order:
            return

        if not self.position:
            self.state = None
            if self.data.close[0] <= self.bb.bot[0] and 25 <= self.rsi[0] <= 45:
                self._enter_position()
            return

        close = self.data.close[0]

        if self.state == 'core':
            if close > self.sma50[0] or self.rsi[0] > 60:
                self.order = self.close()
                self.state = None
            return

        bars_held = len(self) - self.entry_bar

        if close <= self.stop_price:
            self.order = self.close()
            self.state = None
            return

        if close >= self.bb.mid[0] or close >= self.target_price:
            self.order = self.close()
            self.state = None
            return

        if close <= (self.bb.bot[0] - self.atr[0]) or self.rsi[0] < 25:
            position_share = (self.position.size * close) / self.broker.getvalue()
            if position_share <= 0.5:
                self.state = 'core'
                return

        if 25 <= bars_held <= 27:
            self.order = self.close()
            self.state = None
            if close <= self.bb.bot[0]:
                self._enter_position()
            return

        if bars_held >= 30:
            self.order = self.close()
            self.state = None
