import backtrader as bt


class BollingerMATrendSystem(bt.Strategy):
    """
    Bollinger Bands & MA Trend-Following System
    
    A trend-following strategy combining Bollinger Bands for entry timing
    with a 44-period SMA for trend direction and a consolidation filter.
    """
    params = (
        ('bb_period', 20),
        ('bb_dev', 2.0),
        ('sma_period', 44),
        ('risk_pct', 0.02),
        ('consolidation_threshold', 0.20),
    )

    def __init__(self):
        self.order = None

        # Indicators
        self.bb = bt.indicators.BollingerBands(
            self.data.close,
            period=self.p.bb_period,
            devfactor=self.p.bb_dev
        )
        self.sma = bt.indicators.SMA(
            self.data.close,
            period=self.p.sma_period
        )

        # Consolidation filter line: BB Width and its 20-period average
        self.bb_width = self.bb.lines.top - self.bb.lines.bot
        self.bb_width_sma = bt.indicators.SMA(self.bb_width, period=self.p.bb_period)

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        if self.order:
            return

        # Consolidation filter: BB width < 20% of its 20-period average
        in_consolidation = self.bb_width[0] < (self.p.consolidation_threshold * self.bb_width_sma[0])

        if not self.position:
            # Long Entry
            if (self.data.close[0] > self.sma[0] and
                    self.data.close[0] > self.bb.lines.top[0] and
                    not in_consolidation):
                
                capital = self.broker.getvalue()
                risk_amount = capital * self.p.risk_pct
                entry_price = self.data.close[0]
                stop_price = self.bb.lines.bot[0]
                risk_per_share = entry_price - stop_price

                if risk_per_share > 0:
                    size = int(risk_amount / risk_per_share)
                else:
                    size = int(capital / entry_price)

                cash = self.broker.getcash()
                max_size = int(cash / entry_price)
                size = min(size, max_size)

                if size > 0:
                    self.order = self.buy(size=size)

            # Short Entry
            elif (self.data.close[0] < self.sma[0] and
                  self.data.close[0] < self.bb.lines.bot[0] and
                  not in_consolidation):

                capital = self.broker.getvalue()
                risk_amount = capital * self.p.risk_pct
                entry_price = self.data.close[0]
                stop_price = self.bb.lines.top[0]
                risk_per_share = stop_price - entry_price

                if risk_per_share > 0:
                    size = int(risk_amount / risk_per_share)
                else:
                    size = int(capital / entry_price)

                cash = self.broker.getcash()
                max_size = int(cash / entry_price)
                size = min(size, max_size)

                if size > 0:
                    self.order = self.sell(size=size)

        elif self.position.size > 0:
            # Long Exit:
            # 1. Close back below upper BB
            # 2. Cross below MA-44
            # 3. Stop loss: price reaches or crosses lower BB
            if (self.data.close[0] < self.bb.lines.top[0] or
                    self.data.close[0] < self.sma[0] or
                    self.data.low[0] <= self.bb.lines.bot[0]):
                self.order = self.close()

        elif self.position.size < 0:
            # Short Exit:
            # 1. Close back above lower BB
            # 2. Cross above MA-44
            # 3. Stop loss: price reaches or crosses upper BB
            if (self.data.close[0] > self.bb.lines.bot[0] or
                    self.data.close[0] > self.sma[0] or
                    self.data.high[0] >= self.bb.lines.top[0]):
                self.order = self.close()
