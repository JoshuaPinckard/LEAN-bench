import backtrader as bt


class IntradayVolatilityBandStrategy(bt.Strategy):
    params = (
        ("sma_period", 20),
        ("atr_period", 14),
        ("atr_factor", 2.0),
    )

    def __init__(self):
        self.sma = bt.indicators.SMA(
            self.data.close, period=self.p.sma_period
        )
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.upper_band = self.sma + self.p.atr_factor * self.atr
        self.lower_band = self.sma - self.p.atr_factor * self.atr
        self.order = None

    def notify_order(self, order):
        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            self.order = None

    def next(self):
        if self.order:
            return

        if not self.position:
            if self.data.close[0] > self.upper_band[0]:
                self.order = self.buy()
            elif self.data.close[0] < self.lower_band[0]:
                self.order = self.sell()
        else:
            if self.position.size > 0:
                if (
                    self.data.close[0] < self.sma[0]
                    or self.data.low[0] <= self.lower_band[0]
                ):
                    self.order = self.close()
            elif self.position.size < 0:
                if (
                    self.data.close[0] > self.sma[0]
                    or self.data.high[0] >= self.upper_band[0]
                ):
                    self.order = self.close()
