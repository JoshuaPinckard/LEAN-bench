import backtrader as bt


class GapFadeStrategy(bt.Strategy):
    params = (
        ('X', 1.0),
    )

    def __init__(self):
        self.broker.set_coc(True)
        self.order = None
        self.entry_order = None
        self.stop_order = None
        self.session_date = None
        self.prior_regular_close = None
        self.session_open = None
        self.entered_today = False
        self.stopped_today = False

    def next(self):
        dt = self.data.datetime.datetime(0)
        current_date = dt.date()
        current_time = dt.time()

        if current_time == dt.time().replace(hour=9, minute=30, second=0, microsecond=0):
            self.session_date = current_date
            self.session_open = self.data.open[0]
            self.entered_today = False
            self.stopped_today = False

            if self.prior_regular_close is not None:
                gap_pct = (
                    (self.session_open - self.prior_regular_close)
                    / self.prior_regular_close
                    * 100
                )

                signal = 0
                if gap_pct >= self.p.X:
                    signal = -1
                elif gap_pct <= -self.p.X:
                    signal = 1

                if signal:
                    if self.position and (
                        (self.position.size > 0 and signal < 0)
                        or (self.position.size < 0 and signal > 0)
                    ):
                        if self.stop_order is not None:
                            self.cancel(self.stop_order)
                        self.close()
                        self.order = self.sell() if signal < 0 else self.buy()
                        self.entry_order = self.order
                        self.entered_today = True
                    elif not self.position and self.order is None:
                        self.order = self.sell() if signal < 0 else self.buy()
                        self.entry_order = self.order
                        self.entered_today = True

        if current_time == dt.time().replace(hour=15, minute=45, second=0, microsecond=0):
            if self.position and self.order is None:
                if self.stop_order is not None:
                    self.cancel(self.stop_order)
                self.order = self.close()

            self.prior_regular_close = self.data.close[0]

    def notify_order(self, order):
        if order.status not in (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            return

        if order is self.entry_order:
            self.entry_order = None

            if order.status == order.Completed:
                if order.executed.size < 0:
                    stop_price = self.session_open * 1.02
                else:
                    stop_price = self.session_open * 0.98

                self.stop_order = self.close(
                    exectype=bt.Order.Stop,
                    price=stop_price,
                )

        if order is self.stop_order:
            if order.status == order.Completed:
                self.stopped_today = True
            self.stop_order = None

        if order is self.order:
            self.order = None
