import backtrader as bt


class HmaBbwAtrTrendStrategy(bt.Strategy):
    params = dict(
        hma_period=9,
        bb_period=20,
        bb_devfactor=2.0,
        bbwidth_sma_period=50,
        atr_period=14,
        contraction_factor=0.8,
        stop_atr_mult=2.0,
    )

    def __init__(self):
        self.hma = bt.indicators.HullMovingAverage(
            self.data.close, period=self.p.hma_period
        )
        self.bb = bt.indicators.BollingerBands(
            self.data.close, period=self.p.bb_period, devfactor=self.p.bb_devfactor
        )
        self.bbwidth = (self.bb.top - self.bb.bot) / self.bb.mid
        self.bbwidth_sma = bt.indicators.SMA(
            self.bbwidth, period=self.p.bbwidth_sma_period
        )
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.crossover = bt.indicators.CrossOver(self.data.close, self.hma)

        self.order = None
        self.stop_price = None
        self.entering_long = False
        self.entering_short = False

    def notify_order(self, order):
        if order.status == order.Completed:
            if self.entering_long:
                self.stop_price = order.executed.price - self.atr[0] * self.p.stop_atr_mult
            elif self.entering_short:
                self.stop_price = order.executed.price + self.atr[0] * self.p.stop_atr_mult

        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None
            self.entering_long = False
            self.entering_short = False

    def next(self):
        if self.order:
            return

        contraction = self.bbwidth[0] < self.bbwidth_sma[0] * self.p.contraction_factor
        long_signal = self.crossover[0] > 0 and self.hma[0] > self.hma[-1] and contraction
        short_signal = self.crossover[0] < 0 and self.hma[0] < self.hma[-1] and contraction

        if self.position:
            if self.position.size > 0:
                if self.data.close[0] <= self.stop_price or short_signal:
                    self.order = self.close()
            elif self.position.size < 0:
                if self.data.close[0] >= self.stop_price or long_signal:
                    self.order = self.close()
        else:
            if long_signal:
                self.order = self.buy()
                self.entering_long = True
            elif short_signal:
                self.order = self.sell()
                self.entering_short = True
