import backtrader as bt


class StopLossProtection52WeekHigh(bt.Strategy):
    params = (("commission_rate", 0.001),)

    def __init__(self):
        self.high_52_week = bt.indicators.Highest(self.data.close, period=252)
        self.prior_52_week_high = bt.indicators.Highest(
            self.data.close(-1), period=252
        )
        self.breakout = bt.indicators.CrossOver(
            self.data.close, self.prior_52_week_high
        )

        self.entry_order = None
        self.stop_order = None
        self.stop_canceling = False
        self.stop_prices = None

    def _place_stop_loss(self):
        high = self.high_52_week[0]
        stop_price = high * 0.95
        limit_price = high * 0.90

        self.stop_prices = (stop_price, limit_price)
        self.stop_order = self.sell(
            size=self.position.size,
            exectype=bt.Order.StopLimit,
            price=stop_price,
            plimit=limit_price,
        )

    def next(self):
        if self.entry_order:
            return

        if not self.position:
            if self.breakout[0] > 0:
                size = self.broker.getcash() / (
                    self.data.close[0] * (1.0 + self.p.commission_rate)
                )
                if size > 0:
                    self.entry_order = self.buy(size=size)
            return

        desired_prices = (
            self.high_52_week[0] * 0.95,
            self.high_52_week[0] * 0.90,
        )

        if self.stop_order:
            if self.stop_prices != desired_prices and not self.stop_canceling:
                self.stop_canceling = True
                self.cancel(self.stop_order)
        elif not self.stop_canceling:
            self._place_stop_loss()

    def notify_order(self, order):
        if order is self.entry_order:
            if order.status in (
                order.Completed,
                order.Canceled,
                order.Margin,
                order.Rejected,
            ):
                self.entry_order = None
            return

        if order is self.stop_order:
            if order.status == order.Canceled:
                self.stop_order = None
                self.stop_canceling = False
                if self.position:
                    self._place_stop_loss()
            elif order.status in (
                order.Completed,
                order.Margin,
                order.Rejected,
            ):
                self.stop_order = None
                self.stop_canceling = False
