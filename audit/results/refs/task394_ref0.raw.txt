import backtrader as bt


class MidTermSarVolumeBreakout(bt.Strategy):
    params = dict(
        sar_af=0.02,
        sar_afmax=0.2,
        vol_period=20,
    )

    def __init__(self):
        self.sar = bt.indicators.ParabolicSAR(
            self.data, af=self.p.sar_af, afmax=self.p.sar_afmax
        )
        self.vol_sma = bt.indicators.SMA(self.data.volume, period=self.p.vol_period)
        self.order = None

    def notify_order(self, order):
        if order.status in (order.Completed, order.Canceled, order.Margin, order.Rejected):
            self.order = None

    def next(self):
        if self.order:
            return

        close = self.data.close[0]
        sar = self.sar[0]
        vol_above_avg = self.data.volume[0] > self.vol_sma[0]

        if not self.position:
            if close > sar and vol_above_avg:
                self.order = self.buy()
            elif close < sar and vol_above_avg:
                self.order = self.sell()
        else:
            if self.position.size > 0 and close < sar:
                self.order = self.close()
            elif self.position.size < 0 and close > sar:
                self.order = self.close()
