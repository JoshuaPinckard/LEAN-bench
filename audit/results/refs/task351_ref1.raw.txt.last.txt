import backtrader as bt


class VolatilityStochasticStrategy(bt.Strategy):
    params = (
        ("volatility_period", 20),
        ("stochastic_period", 14),
        ("period_dfast", 3),
        ("period_dslow", 3),
        ("atr_period", 14),
    )

    def __init__(self):
        self.returns = bt.indicators.PercentChange(self.data.close, period=1)
        self.volatility = bt.indicators.StdDev(
            self.returns, period=self.p.volatility_period
        )
        self.stochastic = bt.indicators.StochasticGeneric(
            self.volatility,
            period=self.p.stochastic_period,
            period_dfast=self.p.period_dfast,
            period_dslow=self.p.period_dslow,
        )
        self.crossover = bt.indicators.CrossOver(
            self.stochastic.percK, self.stochastic.percD
        )
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)

        self.entry_order = None
        self.stop_order = None
        self.exit_order = None
        self.stop_cancel_pending = False

    def next(self):
        if self.entry_order or self.exit_order or self.stop_cancel_pending:
            return

        if self.position:
            if ((self.position.size > 0 and self.crossover[0] < 0) or
                    (self.position.size < 0 and self.crossover[0] > 0)):
                if self.stop_order:
                    self.cancel(self.stop_order)
                    self.stop_cancel_pending = True
                else:
                    self.exit_order = self.close()
            return

        if (self.crossover[0] > 0 and
                self.stochastic.percK[-1] < 20 and
                self.volatility[0] > self.volatility[-1]):
            self.entry_order = self.buy()

        elif (self.crossover[0] < 0 and
                self.stochastic.percK[-1] > 80 and
                self.volatility[0] < self.volatility[-1]):
            self.entry_order = self.sell()

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return

        if order is self.entry_order:
            if order.status == order.Completed:
                if order.isbuy():
                    stop_price = order.executed.price - (2 * self.atr[0])
                    self.stop_order = self.sell(
                        exectype=bt.Order.Stop,
                        price=stop_price,
                    )
                else:
                    stop_price = order.executed.price + (2 * self.atr[0])
                    self.stop_order = self.buy(
                        exectype=bt.Order.Stop,
                        price=stop_price,
                    )
            self.entry_order = None

        elif order is self.stop_order:
            if order.status == order.Canceled and self.stop_cancel_pending:
                self.stop_order = None
                self.stop_cancel_pending = False
                if self.position:
                    self.exit_order = self.close()
            elif order.status in (
                order.Completed,
                order.Canceled,
                order.Margin,
                order.Rejected,
            ):
                self.stop_order = None
                self.stop_cancel_pending = False

        elif order is self.exit_order:
            if order.status in (
                order.Completed,
                order.Canceled,
                order.Margin,
                order.Rejected,
            ):
                self.exit_order = None
