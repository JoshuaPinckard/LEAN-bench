import backtrader as bt


class RollingVWAP(bt.Indicator):
    lines = ('vwap',)
    params = (('period', 20),)

    def __init__(self):
        typical_price = (self.data.high + self.data.low + self.data.close) / 3.0
        tp_vol = typical_price * self.data.volume
        sum_tp_vol = bt.indicators.SumN(tp_vol, period=self.p.period)
        sum_vol = bt.indicators.SumN(self.data.volume, period=self.p.period)
        self.lines.vwap = sum_tp_vol / sum_vol


class PulseWaveStrategy(bt.Strategy):
    def __init__(self):
        self.macd = bt.indicators.MACD(
            self.data,
            period_me1=12,
            period_me2=26,
            period_signal=9,
        )
        self.rsi = bt.indicators.RSI(self.data, period=14)
        self.vwap = RollingVWAP(self.data, period=20)
        self.crossover = bt.indicators.CrossOver(self.macd.macd, self.macd.signal)
        self.order = None

    def notify_order(self, order):
        if order.status in (order.Completed, order.Canceled, order.Margin, order.Rejected):
            self.order = None

    def next(self):
        if self.order:
            return

        close = self.data.close[0]
        vwap = self.vwap.vwap[0]

        if not self.position:
            if close > vwap and self.crossover[0] > 0 and self.rsi[0] < 70:
                self.order = self.buy()
        else:
            if close < vwap and self.crossover[0] < 0 and self.rsi[0] > 30:
                self.order = self.close()
