import backtrader as bt


class EmaForecastingStrategy(bt.Strategy):
    params = (
        ("ema_period", 21),
        ("trend_period", 50),
        ("threshold", 0.005),
        ("stop_loss", 0.02),
        ("take_profit", 0.03),
    )

    def __init__(self):
        self.k = 2.0 / (self.p.ema_period + 1.0)
        self.ema21 = bt.indicators.ExponentialMovingAverage(
            self.data.close, period=self.p.ema_period
        )
        self.ema50 = bt.indicators.ExponentialMovingAverage(
            self.data.close, period=self.p.trend_period
        )

        # EMA_forecast(N+1) = (Close(N) * k) + (EMA(N) * (1 - k))
        self.ema_forecast = (
            self.data.close * self.k + self.ema21 * (1.0 - self.k)
        )

        # CrossOver indicator: +1 when close crosses forecast from below to above, -1 when above to below
        self.crossover = bt.indicators.CrossOver(
            self.data.close, self.ema_forecast
        )

        self.entry_order = None
        self.sl_order = None
        self.tp_order = None
        self.exit_order = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            if order == self.entry_order:
                self.entry_order = None
                exec_price = order.executed.price
                exec_size = abs(order.executed.size)
                if order.isbuy():
                    sl_price = exec_price * (1.0 - self.p.stop_loss)
                    tp_price = exec_price * (1.0 + self.p.take_profit)
                    self.sl_order = self.sell(
                        exectype=bt.Order.Stop, price=sl_price, size=exec_size
                    )
                    self.tp_order = self.sell(
                        exectype=bt.Order.Limit,
                        price=tp_price,
                        size=exec_size,
                        oco=self.sl_order,
                    )
                else:
                    sl_price = exec_price * (1.0 + self.p.stop_loss)
                    tp_price = exec_price * (1.0 - self.p.take_profit)
                    self.sl_order = self.buy(
                        exectype=bt.Order.Stop, price=sl_price, size=exec_size
                    )
                    self.tp_order = self.buy(
                        exectype=bt.Order.Limit,
                        price=tp_price,
                        size=exec_size,
                        oco=self.sl_order,
                    )
            elif order in [self.sl_order, self.tp_order]:
                self.sl_order = None
                self.tp_order = None
            elif order == self.exit_order:
                self.exit_order = None

        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            if order == self.entry_order:
                self.entry_order = None
            if order == self.sl_order:
                self.sl_order = None
            if order == self.tp_order:
                self.tp_order = None
            if order == self.exit_order:
                self.exit_order = None

    def next(self):
        if self.entry_order or self.exit_order:
            return

        if self.position:
            if self.position.size > 0 and self.crossover[0] < 0:
                if self.sl_order:
                    self.cancel(self.sl_order)
                    self.sl_order = None
                if self.tp_order:
                    self.cancel(self.tp_order)
                    self.tp_order = None
                self.exit_order = self.close()
            elif self.position.size < 0 and self.crossover[0] > 0:
                if self.sl_order:
                    self.cancel(self.sl_order)
                    self.sl_order = None
                if self.tp_order:
                    self.cancel(self.tp_order)
                    self.tp_order = None
                self.exit_order = self.close()
        else:
            close_val = self.data.close[0]
            forecast_val = self.ema_forecast[0]

            long_diff = (forecast_val - close_val) / close_val
            long_signal = (close_val < forecast_val) and (
                long_diff >= self.p.threshold
            )
            if self.ema50[0] <= self.ema50[-1]:
                long_signal = False

            short_diff = (close_val - forecast_val) / close_val
            short_signal = (close_val > forecast_val) and (
                short_diff >= self.p.threshold
            )
            if self.ema50[0] >= self.ema50[-1]:
                short_signal = False

            if long_signal:
                size = int((self.broker.getcash() * 0.99) / close_val)
                if size > 0:
                    self.entry_order = self.buy(size=size)
            elif short_signal:
                size = int((self.broker.getcash() * 0.99) / close_val)
                if size > 0:
                    self.entry_order = self.sell(size=size)
