import backtrader as bt


class EMAForecastStrategy(bt.Strategy):

    params = (
        ('ema_period', 21),
        ('ema_trend_period', 50),
        ('threshold_pct', 0.005),
        ('stop_loss_pct', 0.02),
        ('take_profit_pct', 0.03),
    )

    def __init__(self):
        self.ema21 = bt.indicators.EMA(self.data.close, period=self.p.ema_period)
        self.ema50 = bt.indicators.EMA(self.data.close, period=self.p.ema_trend_period)
        self.k = 2.0 / (self.p.ema_period + 1)

        self.entry_order = None
        self.sl_order = None
        self.tp_order = None
        self.close_order = None

        self.position_type = None       # 'long' or 'short'
        self.prev_ema_forecast = None   # EMA_forecast computed on the previous bar

    def notify_order(self, order):
        if order.status == order.Completed:
            if order is self.entry_order:
                fill = order.executed.price
                size = abs(self.position.size)
                if self.position_type == 'long':
                    self.sl_order = self.sell(
                        exectype=bt.Order.Stop,
                        price=fill * (1.0 - self.p.stop_loss_pct),
                        size=size,
                    )
                    self.tp_order = self.sell(
                        exectype=bt.Order.Limit,
                        price=fill * (1.0 + self.p.take_profit_pct),
                        size=size,
                    )
                else:
                    self.sl_order = self.buy(
                        exectype=bt.Order.Stop,
                        price=fill * (1.0 + self.p.stop_loss_pct),
                        size=size,
                    )
                    self.tp_order = self.buy(
                        exectype=bt.Order.Limit,
                        price=fill * (1.0 - self.p.take_profit_pct),
                        size=size,
                    )
                self.entry_order = None

            elif order is self.sl_order:
                self.sl_order = None
                if self.tp_order:
                    self.cancel(self.tp_order)
                    self.tp_order = None
                self.position_type = None

            elif order is self.tp_order:
                self.tp_order = None
                if self.sl_order:
                    self.cancel(self.sl_order)
                    self.sl_order = None
                self.position_type = None

            elif order is self.close_order:
                self.close_order = None
                self.position_type = None

        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            if order is self.entry_order:
                self.entry_order = None
                self.position_type = None
            elif order is self.sl_order:
                self.sl_order = None
            elif order is self.tp_order:
                self.tp_order = None
            elif order is self.close_order:
                self.close_order = None

    def next(self):
        close = self.data.close[0]
        # EMA_forecast(N+1) = Close(N) * k + EMA(N) * (1 - k)
        ema_forecast = close * self.k + self.ema21[0] * (1.0 - self.k)

        if not self.position:
            if self.entry_order is None:
                ema50_up = self.ema50[0] > self.ema50[-1]
                ema50_dn = self.ema50[0] < self.ema50[-1]

                diff_long = (ema_forecast - close) / close
                diff_short = (close - ema_forecast) / close

                if diff_long >= self.p.threshold_pct and ema50_up:
                    size = int(self.broker.getcash() / close)
                    if size > 0:
                        self.entry_order = self.buy(size=size)
                        self.position_type = 'long'

                elif diff_short >= self.p.threshold_pct and ema50_dn:
                    size = int(self.broker.getcash() / close)
                    if size > 0:
                        self.entry_order = self.sell(size=size)
                        self.position_type = 'short'

        elif self.close_order is None and self.prev_ema_forecast is not None:
            prev_close = self.data.close[-1]
            pef = self.prev_ema_forecast

            if self.position_type == 'long' and prev_close > pef and close < pef:
                if self.sl_order:
                    self.cancel(self.sl_order)
                    self.sl_order = None
                if self.tp_order:
                    self.cancel(self.tp_order)
                    self.tp_order = None
                self.close_order = self.close()

            elif self.position_type == 'short' and prev_close < pef and close > pef:
                if self.sl_order:
                    self.cancel(self.sl_order)
                    self.sl_order = None
                if self.tp_order:
                    self.cancel(self.tp_order)
                    self.tp_order = None
                self.close_order = self.close()

        self.prev_ema_forecast = ema_forecast
