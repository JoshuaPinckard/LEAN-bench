import backtrader as bt


class FibonacciGoldenPocketStrategy(bt.Strategy):
    params = (
        ("fib_period", 200),
        ("exit_period", 20),
    )

    def __init__(self):
        self.addminperiod(self.p.fib_period)
        self.local_high = bt.indicators.Highest(
            self.data.high, period=self.p.exit_period
        )

        self.order = None
        self.entry_bar = None
        self.range_key = None
        self.setup_state = 0  # 0: below .618, 1: touch .786, 2: recapture .618, 3: used

    def notify_order(self, order):
        if order.status == order.Completed:
            if order.isbuy():
                self.entry_bar = len(self)
            self.order = None
        elif order.status in (order.Canceled, order.Margin, order.Rejected):
            self.order = None

    def next(self):
        highs = list(self.data.high.get(size=self.p.fib_period))
        lows = list(self.data.low.get(size=self.p.fib_period))

        swing_high = max(highs)
        swing_low = min(lows)

        high_index = max(
            index for index, value in enumerate(highs) if value == swing_high
        )
        low_index = max(
            index for index, value in enumerate(lows) if value == swing_low
        )

        range_key = (
            swing_high,
            swing_low,
            len(self) - self.p.fib_period + high_index,
            len(self) - self.p.fib_period + low_index,
        )

        if range_key != self.range_key:
            self.range_key = range_key
            self.setup_state = 0

        price_range = swing_high - swing_low
        if price_range == 0:
            return

        if high_index < low_index:
            level_618 = swing_high - price_range * 0.618
            level_786 = swing_high - price_range * 0.786
        else:
            level_618 = swing_low + price_range * 0.618
            level_786 = swing_low + price_range * 0.786

        if (
            self.position
            and self.order is None
            and self.entry_bar is not None
            and len(self) > self.entry_bar
            and self.data.close[0] >= self.local_high[0]
        ):
            self.order = self.close()
            return

        if self.setup_state == 0:
            if self.data.close[0] < level_618 or self.data.low[0] < level_618:
                self.setup_state = 1
        elif self.setup_state == 1:
            if self.data.low[0] <= level_786:
                self.setup_state = 2
        elif self.setup_state == 2:
            if self.data.close[0] > level_618:
                self.setup_state = 3
                if not self.position and self.order is None:
                    self.order = self.buy()
