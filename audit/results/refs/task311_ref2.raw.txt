import backtrader as bt


class SmaAtrTrailingStopStrategy(bt.Strategy):
    params = (
        ("sma_period", 20),
        ("atr_period", 14),
        ("atr_mult", 3.0),
    )

    def __init__(self):
        self.sma = bt.indicators.SimpleMovingAverage(
            self.data.close, period=self.p.sma_period
        )
        self.atr = bt.indicators.AverageTrueRange(
            self.data, period=self.p.atr_period
        )
        self.crossover = bt.indicators.CrossOver(self.data.close, self.sma)

        self.order = None
        self.trail_stop = None
        self.highest_close = None
        self.lowest_close = None

        # Commission and slippage placeholders
        # self.broker.setcommission(commission=0.001)
        # self.broker.set_slippage_fixed(0.0)

    def notify_order(self, order):
        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            if self.order == order:
                self.order = None

    def next(self):
        if not self.position:
            self.trail_stop = None
            self.highest_close = None
            self.lowest_close = None

            if self.order is not None:
                return

            if self.crossover[0] > 0:
                self.order = self.buy()
            elif self.crossover[0] < 0:
                self.order = self.sell()

        elif self.position.size > 0:
            if self.highest_close is None:
                self.highest_close = self.data.close[0]
            else:
                self.highest_close = max(
                    self.highest_close, self.data.close[0]
                )

            candidate_stop = self.highest_close - self.p.atr_mult * self.atr[0]
            if self.trail_stop is None:
                self.trail_stop = candidate_stop
            else:
                self.trail_stop = max(self.trail_stop, candidate_stop)

            if self.crossover[0] < 0:
                if self.order is not None:
                    self.cancel(self.order)
                self.order = self.close()
            else:
                if self.order is not None:
                    self.cancel(self.order)
                self.order = self.sell(
                    exectype=bt.Order.Stop, price=self.trail_stop
                )

        elif self.position.size < 0:
            if self.lowest_close is None:
                self.lowest_close = self.data.close[0]
            else:
                self.lowest_close = min(self.lowest_close, self.data.close[0])

            candidate_stop = self.lowest_close + self.p.atr_mult * self.atr[0]
            if self.trail_stop is None:
                self.trail_stop = candidate_stop
            else:
                self.trail_stop = min(self.trail_stop, candidate_stop)

            if self.crossover[0] > 0:
                if self.order is not None:
                    self.cancel(self.order)
                self.order = self.close()
            else:
                if self.order is not None:
                    self.cancel(self.order)
                self.order = self.buy(
                    exectype=bt.Order.Stop, price=self.trail_stop
                )
