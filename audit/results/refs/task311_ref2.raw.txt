import backtrader as bt


class QCBStrategy(bt.Strategy):

    def __init__(self):
        self.sma = bt.indicators.SMA(self.data.close, period=20)
        self.atr = bt.indicators.ATR(self.data, period=14)
        self.order = None
        self.trail_stop = None
        self.highest_close = None
        self.lowest_close = None
        self._prev_pos = 0

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled,
                             order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        if self.order:
            return

        pos = self.position.size
        close = self.data.close[0]
        sma = self.sma[0]
        atr = self.atr[0]
        prev_close = self.data.close[-1]
        prev_sma = self.sma[-1]

        if pos == 0:
            cross_above = prev_close < prev_sma and close > sma
            cross_below = prev_close > prev_sma and close < sma

            if cross_above:
                self.order = self.buy()
            elif cross_below:
                self.order = self.sell()

        elif pos > 0:
            if self._prev_pos <= 0:
                self.highest_close = close
                self.trail_stop = close - 3.0 * atr
            else:
                if close > self.highest_close:
                    self.highest_close = close
                new_stop = self.highest_close - 3.0 * atr
                if new_stop > self.trail_stop:
                    self.trail_stop = new_stop

            cross_below = prev_close > prev_sma and close < sma
            stop_hit = close <= self.trail_stop

            if cross_below or stop_hit:
                self.order = self.close()

        elif pos < 0:
            if self._prev_pos >= 0:
                self.lowest_close = close
                self.trail_stop = close + 3.0 * atr
            else:
                if close < self.lowest_close:
                    self.lowest_close = close
                new_stop = self.lowest_close + 3.0 * atr
                if new_stop < self.trail_stop:
                    self.trail_stop = new_stop

            cross_above = prev_close < prev_sma and close > sma
            stop_hit = close >= self.trail_stop

            if cross_above or stop_hit:
                self.order = self.close()

        self._prev_pos = pos
