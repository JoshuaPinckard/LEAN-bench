class BullishEngulfingStrategy(bt.Strategy):
    def __init__(self):
        self.entry_order = None
        self.stop_order = None
        self.stop_price = None

    def next(self):
        current_time = self.data.datetime.time(0)

        if self.position and current_time.hour == 15 and current_time.minute == 45:
            if self.stop_order is not None:
                self.cancel(self.stop_order)
            self.close()
            return

        if self.position or self.entry_order is not None:
            return

        bearish_preceding = (
            self.data.close[-2] < self.data.open[-2] and
            self.data.close[-3] < self.data.open[-3] and
            self.data.close[-4] < self.data.open[-4]
        )

        bullish_engulfing = (
            self.data.close[-1] < self.data.open[-1] and
            self.data.close[0] > self.data.open[0] and
            self.data.open[0] <= self.data.close[-1] and
            self.data.close[0] >= self.data.open[-1]
        )

        if bearish_preceding and bullish_engulfing:
            pattern_low = min(self.data.low[0], self.data.low[-1])
            self.stop_price = pattern_low * 0.99

            self.entry_order = self.buy(transmit=False)
            self.stop_order = self.sell(
                exectype=bt.Order.Stop,
                price=self.stop_price,
                parent=self.entry_order,
                transmit=True,
            )

    def notify_order(self, order):
        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            if order is self.entry_order:
                self.entry_order = None
            elif order is self.stop_order:
                self.stop_order = None
