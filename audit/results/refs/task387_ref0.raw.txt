import backtrader as bt


class ADXStochasticStrategy(bt.Strategy):
    def __init__(self):
        self.adx = bt.indicators.ADX(self.data, period=14)
        self.stoch = bt.indicators.StochasticFull(
            self.data, period=14, period_dfast=3, period_dslow=3)

        self.cross_k_down20 = bt.indicators.CrossOver(self.stoch.percK, 20.0)
        self.cross_k_up80 = bt.indicators.CrossOver(self.stoch.percK, 80.0)
        self.cross_adx_down25 = bt.indicators.CrossOver(self.adx.adx, 25.0)

        self.order = None
        self.exit_order = None
        self.stop_order = None

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return

        if order.status == order.Completed:
            if order is self.order:
                if order.isbuy():
                    stop_price = self.position.price * (1 - 0.02)
                    self.stop_order = self.sell(exectype=bt.Order.Stop, price=stop_price)
                else:
                    stop_price = self.position.price * (1 + 0.02)
                    self.stop_order = self.buy(exectype=bt.Order.Stop, price=stop_price)
                self.order = None
            elif order is self.stop_order:
                self.stop_order = None
            elif order is self.exit_order:
                self.exit_order = None
        elif order.status in (order.Canceled, order.Margin, order.Rejected):
            if order is self.order:
                self.order = None
            elif order is self.stop_order:
                self.stop_order = None
            elif order is self.exit_order:
                self.exit_order = None

    def next(self):
        if self.order or self.exit_order:
            return

        adx_cross_below_25 = self.cross_adx_down25[0] < 0

        if self.position:
            if adx_cross_below_25:
                if self.stop_order is not None:
                    self.cancel(self.stop_order)
                    self.stop_order = None
                self.exit_order = self.close()
            return

        if self.adx.adx[0] <= 25:
            return

        bullish = self.adx.plusDI[0] > self.adx.minusDI[0]
        bearish = self.adx.minusDI[0] > self.adx.plusDI[0]

        if bullish and self.cross_k_down20[0] < 0:
            self.order = self.buy()
        elif bearish and self.cross_k_up80[0] > 0:
            self.order = self.sell()
