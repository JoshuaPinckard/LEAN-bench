import backtrader as bt


class LongTermMeanReversion(bt.Strategy):
    def __init__(self):
        self.bb = bt.indicators.BollingerBands(
            self.data.close,
            period=120,
            devfactor=2.0,
            movav=bt.indicators.SimpleMovingAverage,
        )
        self.middle = bt.indicators.SimpleMovingAverage(self.data.close, period=120)
        self.rsi = bt.indicators.RSI(self.data.close, period=14)
        self.volume_sma = bt.indicators.SimpleMovingAverage(self.data.volume, period=20)
        self.atr = bt.indicators.ATR(self.data, period=14)
        self.atr_sma = bt.indicators.SimpleMovingAverage(self.atr, period=50)

        self.pending_order = None
        self.entry_order = None
        self.exit_order = None
        self.stop_order = None
        self.entry_atr = None
        self.entry_bar = None

    def notify_order(self, order):
        if order.status not in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            return

        if order is self.entry_order:
            self.pending_order = None
            self.entry_order = None

            if order.status == order.Completed:
                self.entry_bar = len(self)
                size = abs(self.position.size)

                if self.position.size > 0:
                    self.stop_order = self.sell(
                        size=size,
                        exectype=bt.Order.Stop,
                        price=order.executed.price - (2.0 * self.entry_atr),
                    )
                else:
                    self.stop_order = self.buy(
                        size=size,
                        exectype=bt.Order.Stop,
                        price=order.executed.price + (2.0 * self.entry_atr),
                    )

        elif order is self.exit_order:
            self.pending_order = None
            self.exit_order = None
            if order.status == order.Completed:
                self.entry_bar = None

        elif order is self.stop_order:
            self.stop_order = None
            if order.status == order.Completed:
                self.entry_bar = None

    def next(self):
        long_signal = (
            (
                self.data.close[-1] < self.bb.bot[-1]
                and self.data.close[0] >= self.bb.bot[0]
            )
            or (self.rsi[-1] < 30.0 and self.rsi[0] >= 30.0)
        )
        short_signal = (
            (
                self.data.close[-1] > self.bb.top[-1]
                and self.data.close[0] <= self.bb.top[0]
            )
            or (self.rsi[-1] > 70.0 and self.rsi[0] <= 70.0)
        )

        filters_pass = (
            self.data.volume[0] > self.volume_sma[0]
            and self.atr[0] < (self.atr_sma[0] * 1.5)
        )

        if not self.position:
            if self.pending_order is not None:
                return

            self.entry_atr = float(self.atr[0])
            if long_signal and filters_pass:
                self.entry_order = self.buy()
                self.pending_order = self.entry_order
            elif short_signal and filters_pass:
                self.entry_order = self.sell()
                self.pending_order = self.entry_order
            return

        if self.pending_order is not None:
            return

        holding_limit_reached = (
            self.entry_bar is not None and (len(self) - self.entry_bar) >= 20
        )

        if self.position.size > 0:
            should_exit = (
                self.data.close[0] >= self.middle[0]
                or holding_limit_reached
                or short_signal
            )
        else:
            should_exit = (
                self.data.close[0] <= self.middle[0]
                or holding_limit_reached
                or long_signal
            )

        if should_exit:
            self.exit_order = self.close(oco=self.stop_order)
            self.pending_order = self.exit_order
