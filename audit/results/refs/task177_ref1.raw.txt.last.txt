import backtrader as bt


class BollingerGapStrategy(bt.Strategy):
    params = (
        ("stake", 2),
    )

    def __init__(self):
        self.bbands = bt.indicators.BollingerBands(
            self.data.close, period=20, devfactor=2
        )
        self.bandwidth = (
            (self.bbands.top - self.bbands.bot) / self.bbands.mid
        )
        self.bandwidth_low = bt.indicators.Lowest(
            self.bandwidth, period=100
        )
        self.bandwidth_high = bt.indicators.Highest(
            self.bandwidth, period=100
        )
        self.order = None
        self.exit_order = None

    def next(self):
        if self.order or self.exit_order:
            return

        if self.position:
            return

        bandwidth_range = self.bandwidth_high[0] - self.bandwidth_low[0]
        if bandwidth_range == 0:
            iv_rank = 0.0
        else:
            iv_rank = (
                (self.bandwidth[0] - self.bandwidth_low[0])
                / bandwidth_range
            )

        gap = (self.data.open[0] - self.data.close[-1]) / self.data.close[-1]

        if iv_rank <= 0.20 and gap >= -0.004:
            self.order = self.buy(size=self.p.stake)

    def notify_order(self, order):
        if order.status in (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            if order == self.order:
                self.order = None
                if order.status == order.Completed:
                    self.exit_order = self.close(exectype=bt.Order.Close)
            elif order == self.exit_order:
                self.exit_order = None
