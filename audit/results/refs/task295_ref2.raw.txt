import backtrader as bt


class DividendPitchforkStrategist(bt.Strategy):
    def __init__(self):
        self.bb = bt.indicators.BollingerBands(self.data.close, period=20, devfactor=2)
        self.rsi = bt.indicators.RSI(self.data.close, period=14)
        self.atr = bt.indicators.ATR(self.data, period=14)
        self.sma50 = bt.indicators.SMA(self.data.close, period=50)

        self.order = None
        self.is_core = False
        self.rolling = False
        self.entry_bar = 0
        self.entry_price = 0.0
        self.entry_atr = 0.0
        self.stop_price = 0.0

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            if order.isbuy():
                self.entry_bar = len(self)
                self.entry_price = order.executed.price
                self.entry_atr = self.atr[0]
                self.stop_price = self.entry_price - 1.5 * self.entry_atr
                self.is_core = False
            elif order.issell():
                if self.rolling:
                    self.rolling = False
                    total_val = self.broker.getvalue()
                    comm_rate = 0.001
                    max_spend = 0.50 * total_val
                    size = int(max_spend / (self.data.close[0] * (1.0 + comm_rate)))
                    if size > 0:
                        self.order = self.buy(size=size)
                        return

        self.rolling = False
        self.order = None

    def next(self):
        if self.order:
            return

        if not self.position:
            if self.data.close[0] <= self.bb.lines.bot[0] and 25 <= self.rsi[0] <= 45:
                total_val = self.broker.getvalue()
                comm_rate = 0.001
                max_spend = 0.50 * total_val
                size = int(max_spend / (self.data.close[0] * (1.0 + comm_rate)))
                if size > 0:
                    self.order = self.buy(size=size)
        else:
            if self.is_core:
                if self.data.close[0] > self.sma50[0] or self.rsi[0] > 60:
                    self.order = self.close()
            else:
                if self.data.close[0] <= self.stop_price:
                    self.order = self.close()
                elif (self.data.close[0] <= self.bb.lines.bot[0] - 1.0 * self.atr[0]) or (self.rsi[0] < 25):
                    pos_val = self.position.size * self.data.close[0]
                    tot_val = self.broker.getvalue()
                    if pos_val <= 0.50 * tot_val:
                        self.is_core = True
                elif self.data.close[0] >= self.bb.lines.mid[0] or self.data.close[0] >= (self.entry_price + 1.0 * self.entry_atr):
                    self.order = self.close()
                else:
                    bars_held = len(self) - self.entry_bar
                    if 25 <= bars_held <= 27:
                        if self.data.close[0] <= self.bb.lines.bot[0]:
                            self.rolling = True
                            self.order = self.close()
                        else:
                            self.order = self.close()
                    elif bars_held >= 30:
                        self.order = self.close()
