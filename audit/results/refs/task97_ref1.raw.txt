import backtrader as bt


class EMAForecastStrategy(bt.Strategy):
    params = (
        ('ema_period', 21),     # medium-term EMA
        ('threshold', 0.005),   # 0.5% significance threshold
        ('stop_pct', 0.02),     # 2% stop-loss
        ('take_pct', 0.03),     # 3% take-profit
        ('capital_pct', 0.98),  # ~100% of available capital (buffered)
    )

    def __init__(self):
        # EMA(21); backtrader's EMA uses k = 2/(period+1) internally,
        # matching the spec's weighting multiplier (2/22 for period 21).
        self.ema = bt.indicators.ExponentialMovingAverage(
            self.data.close, period=self.p.ema_period
        )
        self.k = 2.0 / (self.p.ema_period + 1)

        self.order = None        # pending entry / crossover-close order
        self.stop_order = None   # protective stop-loss order
        self.limit_order = None  # protective take-profit order
        self.entering = False    # True while an entry order is in flight

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return

        if order.status == order.Completed:
            if order is self.order:
                # Entry order or crossover-close order finished.
                if self.entering and self.position:
                    entry_price = order.executed.price
                    size = abs(self.position.size)
                    if self.position.size > 0:  # long -> protective SELLs
                        stop_price = entry_price * (1.0 - self.p.stop_pct)
                        limit_price = entry_price * (1.0 + self.p.take_pct)
                        self.stop_order = self.sell(
                            size=size, exectype=bt.Order.Stop, price=stop_price)
                        self.limit_order = self.sell(
                            size=size, exectype=bt.Order.Limit, price=limit_price,
                            oco=self.stop_order)
                    else:                        # short -> protective BUYs
                        stop_price = entry_price * (1.0 + self.p.stop_pct)
                        limit_price = entry_price * (1.0 - self.p.take_pct)
                        self.stop_order = self.buy(
                            size=size, exectype=bt.Order.Stop, price=stop_price)
                        self.limit_order = self.buy(
                            size=size, exectype=bt.Order.Limit, price=limit_price,
                            oco=self.stop_order)
                self.entering = False
                self.order = None
            elif order is self.stop_order or order is self.limit_order:
                # A protective order filled; OCO cancels the sibling.
                self.stop_order = None
                self.limit_order = None

        elif order.status in (order.Canceled, order.Margin, order.Rejected):
            if order is self.order:
                self.entering = False
                self.order = None
            elif order is self.stop_order:
                self.stop_order = None
            elif order is self.limit_order:
                self.limit_order = None

    def _cancel_protective(self):
        if self.stop_order is not None:
            self.cancel(self.stop_order)
        if self.limit_order is not None:
            self.cancel(self.limit_order)
        self.stop_order = None
        self.limit_order = None

    def next(self):
        # Do not act while an entry/close order is pending.
        if self.order is not None:
            return

        k = self.k
        close0 = self.data.close[0]                       # Close(N)
        ema0 = self.ema[0]                                # EMA(N)
        forecast_next = close0 * k + ema0 * (1.0 - k)     # EMA_forecast(N+1)

        if not self.position:
            # ---------------- Entry logic ----------------
            long_gap = (forecast_next - close0) / close0   # Close below forecast
            short_gap = (close0 - forecast_next) / close0  # Close above forecast

            if long_gap >= self.p.threshold:
                size = int((self.broker.getcash() * self.p.capital_pct) / close0)
                if size > 0:
                    self.entering = True
                    self.order = self.buy(size=size)
            elif short_gap >= self.p.threshold:
                size = int((self.broker.getcash() * self.p.capital_pct) / close0)
                if size > 0:
                    self.entering = True
                    self.order = self.sell(size=size)
        else:
            # ---------------- Exit logic ----------------
            # Rolling forecast line: value applicable to a given day is the
            # forecast made from the previous day's Close and EMA.
            fc_today = self.data.close[-1] * k + self.ema[-1] * (1.0 - k)
            fc_yest = self.data.close[-2] * k + self.ema[-2] * (1.0 - k)
            c0 = self.data.close[0]
            c1 = self.data.close[-1]

            if self.position.size > 0:
                # Close crosses forecast from above to below -> exit long.
                if c1 > fc_yest and c0 < fc_today:
                    self._cancel_protective()
                    self.order = self.close()
            elif self.position.size < 0:
                # Close crosses forecast from below to above -> exit short.
                if c1 < fc_yest and c0 > fc_today:
                    self._cancel_protective()
                    self.order = self.close()
