import collections

import backtrader as bt


class MeanReversionBollingerStrategy(bt.Strategy):
    params = dict(
        period=50,
        devfactor=2,
        trade_size=100,
    )

    def __init__(self):
        self.boll = bt.indicators.BollingerBands(
            self.data.close,
            period=self.p.period,
            devfactor=self.p.devfactor,
            movav=bt.indicators.EMA,
        )
        self.order = None
        self.trade_log = []

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status == order.Completed:
            ticker = self.data._name if self.data._name else 'AAPL'
            self.trade_log.append(collections.OrderedDict([
                ('date', self.data.datetime.date(0).isoformat()),
                ('ticker', ticker),
                ('type', 'bought' if order.isbuy() else 'sold'),
                ('price', order.executed.price),
                ('ema', self.boll.mid[0]),
                ('shares', abs(order.executed.size)),
            ]))

        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        if self.order:
            return

        price = self.data.close[0]
        upper = self.boll.top[0]
        lower = self.boll.bot[0]

        if self.position:
            if self.position.size > 0 and price >= upper:
                self.order = self.close()
            elif self.position.size < 0 and price <= lower:
                self.order = self.close()
        else:
            if price <= lower:
                self.order = self.buy(size=self.p.trade_size)
            elif price >= upper:
                self.order = self.sell(size=self.p.trade_size)
