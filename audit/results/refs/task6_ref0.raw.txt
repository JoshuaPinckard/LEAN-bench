    def _size_for_entry(self, price):
        stop_distance = price * self.p.stop_pct
        if stop_distance <= 0:
            return 0
        return int((self.broker.getvalue() * self.p.risk_pct) / stop_distance)

    def _enter_long(self):
        size = self._size_for_entry(self.data.close[0])
        if size > 0:
            self.order = self.buy(size=size)

    def _enter_short(self):
        size = self._size_for_entry(self.data.close[0])
        if size > 0:
            self.order = self.sell(size=size)

    def _exit_position(self):
        # cancel any live stop before closing, else it can fire against a flat
        # position and open an unintended reverse position
        if self.stop_order is not None:
            self.cancel(self.stop_order)
        self.close()

        if self.current_date is None or bar_date != self.current_date:

            self.day_high = self.data.high[0]

        else:
            self.day_high = max(self.day_high, self.data.high[0])
