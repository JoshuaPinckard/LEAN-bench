import backtrader as bt


class VolatilityBalancedStrategy(bt.Strategy):
    params = (
        ('risk_coefficient', 1.0),
    )

    def __init__(self):
        self.atr = bt.indicators.ATR(self.data, period=20)
        self.sma = bt.indicators.SMA(self.data.close, period=20)
        self.bbands = bt.indicators.BollingerBands(
            self.data.close, period=20, devfactor=2
        )
        self.crossover = bt.indicators.CrossOver(self.data.close, self.sma)

        self.order = None
        self.entry_price = None
        self.entry_atr = None

    def notify_order(self, order):
        if order.status == order.Completed:
            if self.position.size:
                if self.entry_price is None:
                    self.entry_price = order.executed.price
                    self.entry_atr = self.atr[0]
            else:
                self.entry_price = None
                self.entry_atr = None

        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            self.order = None

    def _target_size(self):
        capital = self.broker.getvalue()
        risk_size = (capital * 0.01) / (2.0 * self.atr[0])
        cash_size = capital / self.data.close[0]
        return max(0.01, min(risk_size, cash_size) * self.p.risk_coefficient)

    def next(self):
        if self.order:
            return

        close = self.data.close[0]
        outside_bands = abs(close - self.sma[0]) > (
            2.0 * self.bbands.stddev[0]
        )

        if self.position:
            if outside_bands:
                self.order = self.close()
                return

            if self.position.size > 0:
                if close <= self.entry_price - (2.0 * self.entry_atr):
                    self.order = self.close()
                    return
                if close >= self.entry_price + (4.0 * self.entry_atr):
                    self.order = self.close()
                    return
            else:
                if close >= self.entry_price + (2.0 * self.entry_atr):
                    self.order = self.close()
                    return
                if close <= self.entry_price - (4.0 * self.entry_atr):
                    self.order = self.close()
                    return

            target = self._target_size()
            current = abs(self.position.size)

            if target > current:
                self.order = self.buy(size=target - current) if self.position.size > 0 else self.sell(size=target - current)
            elif target < current:
                self.order = self.sell(size=current - target) if self.position.size > 0 else self.buy(size=current - target)
            return

        size = self._target_size()

        if self.crossover[0] > 0:
            self.order = self.buy(size=size)
        elif self.crossover[0] < 0:
            self.order = self.sell(size=size)
