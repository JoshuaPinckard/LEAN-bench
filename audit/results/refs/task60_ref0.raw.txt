import backtrader as bt


class SmallCapMomentumStrategy(bt.Strategy):
    params = (
        ('sma_period', 50),
        ('macd_fast', 12),
        ('macd_slow', 26),
        ('macd_signal', 9),
        ('vol_sma_period', 50),
        ('rsi_period', 14),
        ('vol_accum_days', 5),
        ('vol_spike_lookback', 30),
        ('vol_spike_mult', 2.0),
        ('divergence_lookback', 20),
        ('divergence_exclude', 3),
        ('curl_min_decline', 2),
        ('take_profit_pct', 1.0),
        ('stop_loss_pct', 0.15),
        ('position_pct', 0.03),
    )

    def __init__(self):
        self.sma = bt.indicators.SMA(self.data.close, period=self.p.sma_period)

        self.ema_fast = bt.indicators.EMA(self.data.close, period=self.p.macd_fast)
        self.ema_slow = bt.indicators.EMA(self.data.close, period=self.p.macd_slow)
        self.macd_line = self.ema_fast - self.ema_slow
        self.signal_line = bt.indicators.SMA(self.macd_line, period=self.p.macd_signal)
        self.macd_hist = self.macd_line - self.signal_line

        self.macd_bear_cross = bt.indicators.CrossDown(self.macd_line, self.signal_line)
        self.hist_zero_cross = bt.indicators.CrossDown(self.macd_hist, 0.0)

        self.vol_sma = bt.indicators.SMA(self.data.volume, period=self.p.vol_sma_period)
        self.vol_avg30 = bt.indicators.SMA(self.data.volume, period=self.p.vol_spike_lookback)

        self.rsi = bt.indicators.RSI(self.data.close, period=self.p.rsi_period)

        self.order = None
        self.buyprice = None

        self._needed_bars = max(
            self.p.sma_period,
            self.p.macd_slow + self.p.macd_signal,
            self.p.vol_sma_period,
            self.p.vol_spike_lookback,
            self.p.divergence_lookback,
            self.p.rsi_period,
        ) + 5

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return
        if order.status == order.Completed:
            if order.isbuy():
                self.buyprice = order.executed.price
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        if self.order:
            return
        if len(self.data) < self._needed_bars:
            return

        if not self.position:
            trend_up = self.data.close[0] > self.sma[0]

            hist_vals = self.macd_hist.get(size=self.p.curl_min_decline + 2)
            declining = all(
                hist_vals[i] > hist_vals[i + 1] for i in range(self.p.curl_min_decline)
            )
            turning_up = hist_vals[-1] > hist_vals[-2]
            curling_up = declining and turning_up

            bullish_divergence = False
            lows = self.data.low.get(size=self.p.divergence_lookback)
            hists = self.macd_hist.get(size=self.p.divergence_lookback)
            search_lows = lows[:-self.p.divergence_exclude]
            search_hists = hists[:-self.p.divergence_exclude]
            if search_lows:
                min_idx = min(range(len(search_lows)), key=lambda i: search_lows[i])
                prior_low = search_lows[min_idx]
                prior_hist = search_hists[min_idx]
                if self.data.low[0] < prior_low and self.macd_hist[0] > prior_hist:
                    bullish_divergence = True

            macd_momentum = curling_up or bullish_divergence

            vols = self.data.volume.get(size=self.p.vol_accum_days)
            vsmas = self.vol_sma.get(size=self.p.vol_accum_days)
            volume_accumulation = all(v > s for v, s in zip(vols, vsmas))

            volume_spike = self.data.volume[0] > self.p.vol_spike_mult * self.vol_avg30[0]

            if trend_up and macd_momentum and volume_accumulation and volume_spike:
                cash = self.broker.getvalue()
                size = int((cash * self.p.position_pct) / self.data.close[0])
                if size > 0:
                    self.order = self.buy(size=size)

        else:
            close = self.data.close[0]
            exit_signal = False

            if self.buyprice is not None and close >= self.buyprice * (1 + self.p.take_profit_pct):
                exit_signal = True

            if self.buyprice is not None and close <= self.buyprice * (1 - self.p.stop_loss_pct):
                exit_signal = True

            if close < self.sma[0]:
                exit_signal = True

            swing_lows = self.data.low.get(size=self.p.divergence_lookback)
            significant_low = min(swing_lows[:-1])
            if close < significant_low:
                exit_signal = True

            if self.hist_zero_cross[0] > 0 or self.macd_bear_cross[0] > 0:
                exit_signal = True

            if exit_signal:
                self.order = self.close()
