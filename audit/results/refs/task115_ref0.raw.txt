```python
import backtrader as bt
import collections


class InverseHeadShouldersRSIDivergence(bt.Strategy):
    params = (
        ('daily_pivot_window', 5),      # bars each side to confirm a daily swing high/low
        ('shoulder_tolerance', 0.03),   # max relative diff between left/right shoulder lows
        ('rsi_period', 14),             # weekly RSI period, per spec
        ('weekly_pivot_window', 1),     # weeks each side to confirm a weekly swing low
        ('monthly_trend_period', 10),   # months for monthly SMA trend filter
        ('risk_pct', 0.01),             # fraction of equity risked per trade
        ('rr_ratio', 2.0),              # take-profit : stop-loss ratio, per spec
    )

    def __init__(self):
        self.order = None

        # weekly aggregation / RSI / divergence state
        self.week_key = None
        self.week_high = None
        self.week_low = None
        self.week_close = None
        self.weekly_closes = []
        self.weekly_lows = []
        self.weekly_rsi = []
        self.weekly_avg_gain = None
        self.weekly_avg_loss = None
        self.weekly_pivot_lows = []
        self.rsi_divergence_active = False

        # monthly aggregation / trend state
        self.month_key = None
        self.month_close = None
        self.monthly_closes = []

        # daily H&S pattern state
        self.daily_pivots = collections.deque(maxlen=5)
        self.pattern_ready = False
        self.neckline_p1 = None
        self.neckline_p2 = None
        self.right_shoulder_low = None

    # -------------------------------------------------- daily pattern helpers
    def _is_pivot_low(self, w):
        if len(self.data) <= 2 * w:
            return False
        center = self.data.low[-w]
        for i in range(-2 * w, 1):
            if i != -w and self.data.low[i] <= center:
                return False
        return True

    def _is_pivot_high(self, w):
        if len(self.data) <= 2 * w:
            return False
        center = self.data.high[-w]
        for i in range(-2 * w, 1):
            if i != -w and self.data.high[i] >= center:
                return False
        return True

    def _neckline_value(self, idx):
        i1, p1 = self.neckline_p1
        i2, p2 = self.neckline_p2
        if i2 == i1:
            return max(p1, p2)
        return p1 + (p2 - p1) * (idx - i1) / (i2 - i1)

    def _update_daily_pattern(self, w):
        idx = len(self.data) - 1 - w
        if self._is_pivot_low(w):
            self.daily_pivots.append(('L', idx, self.data.low[-w]))
            self._evaluate_pattern()
        elif self._is_pivot_high(w):
            self.daily_pivots.append(('H', idx, self.data.high[-w]))
            self._evaluate_pattern()

    def _evaluate_pattern(self):
        if len(self.daily_pivots) < 5:
            return
        seq = list(self.daily_pivots)
        if [p[0] for p in seq] != ['L', 'H', 'L', 'H', 'L']:
            return
        (_, i_sl, sl), (_, i_p1, p1), (_, i_head, head), (_, i_p2, p2), (_, i_sr, sr) = seq
        if not (head < sl and head < sr):
            return
        if sl == 0 or abs(sr - sl) / sl > self.p.shoulder_tolerance:
            return
        self.neckline_p1 = (i_p1, p1)
        self.neckline_p2 = (i_p2, p2)
        self.right_shoulder_low = sr
        self.pattern_ready = True

    def _check_hs_breakout(self):
        if not self.pattern_ready:
            return False
        idx = len(self.data) - 1
        neckline_now = self._neckline_value(idx)
        neckline_prev = self._neckline_value(idx - 1)
        return self.data.close[-1] <= neckline_prev and self.data.close[0] > neckline_now

    # -------------------------------------------------- weekly RSI / divergence
    def _update_weekly(self):
        dt = self.data.datetime.date(0)
        key = (dt.isocalendar()[0], dt.isocalendar()[1])
        if self.week_key is None or key != self.week_key:
            if self.week_key is not None:
                self._finalize_week()
            self.week_key = key
            self.week_high = self.data.high[0]
            self.week_low = self.data.low[0]
        else:
            self.week_high = max(self.week_high, self.data.high[0])
            self.week_low = min(self.week_low, self.data.low[0])
        self.week_close = self.data.close[0]

    def _finalize_week(self):
        self.weekly_closes.append(self.week_close)
        self.weekly_lows.append(self.week_low)
        self._update_weekly_rsi()
        self._update_weekly_divergence()

    def _update_weekly_rsi(self):
        n = len(self.weekly_closes)
        if n < 2:
            self.weekly_rsi.append(None)
            return
        change = self.weekly_closes[-1] - self.weekly_closes[-2]
        gain, loss = max(change, 0.0), max(-change, 0.0)
        period = self.p.rsi_period
        num_changes = n - 1
        if num_changes < period:
            self.weekly_rsi.append(None)
        elif num_changes == period:
            gains, losses = [], []
            for i in range(1, period + 1):
                ch = self.weekly_closes[i] - self.weekly_closes[i - 1]
                gains.append(max(ch, 0.0))
                losses.append(max(-ch, 0.0))
            self.weekly_avg_gain = sum(gains) / period
            self.weekly_avg_loss = sum(losses) / period
            self.weekly_rsi.append(self._rsi_from_avg(self.weekly_avg_gain, self.weekly_avg_loss))
        else:
            self.weekly_avg_gain = (self.weekly_avg_gain * (period - 1) + gain) / period
            self.weekly_avg_loss = (self.weekly_avg_loss * (period - 1) + loss) / period
            self.weekly_rsi.append(self._rsi_from_avg(self.weekly_avg_gain, self.weekly_avg_loss))

    @staticmethod
    def _rsi_from_avg(avg_gain, avg_loss):
        if avg_loss == 0:
            return 100.0
        rs = avg_gain / avg_loss
        return 100.0 - 100.0 / (1.0 + rs)

    def _update_weekly_divergence(self):
        w = self.p.weekly_pivot_window
        n = len(self.weekly_lows)
        center = n - 1 - w
        if center < w:
            return
        window = self.weekly_lows[center - w:center + w + 1]
        if self.weekly_lows[center] != min(window) or self.weekly_rsi[center] is None:
            return
        self.weekly_pivot_lows.append((center, self.weekly_lows[center], self.weekly_rsi[center]))
        if len(self.weekly_pivot_lows) < 2:
            return
        (_, prev_price, prev_rsi) = self.weekly_pivot_lows[-2]
        (_, new_price, new_rsi) = self.weekly_pivot_lows[-1]
        self.rsi_divergence_active = new_price < prev_price and new_rsi > prev_rsi

    # -------------------------------------------------- monthly trend filter
    def _update_monthly(self):
        dt = self.data.datetime.date(0)
        key = (dt.year, dt.month)
        if self.month_key is None or key != self.month_key:
            if self.month_key is not None:
                self.monthly_closes.append(self.month_close)
            self.month_key = key
        self.month_close = self.data.close[0]

    def _monthly_trend_bullish(self):
        n = self.p.monthly_trend_period
        if len(self.monthly_closes) < n:
            return False
        recent = self.monthly_closes[-n:]
        return self.monthly_closes[-1] > sum(recent) / n

    # -------------------------------------------------- backtrader hooks
    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return
        if self.order is not None and order.ref == self.order.ref:
            if order.status in (order.Completed, order.Canceled, order.Margin, order.Rejected):
                self.order = None

    def next(self):
        self._update_weekly()
        self._update_monthly()
        self._update_daily_pattern(self.p.daily_pivot_window)

        if self.order or self.position:
            return
        if not self.pattern_ready or not self._check_hs_breakout():
            return
        if not self.rsi_divergence_active or not self._monthly_trend_bullish():
            self.pattern_ready = False
            return

        entry_price = self.data.close[0]
        stop_price = self.right_shoulder_low
        risk_per_share = entry_price - stop_price
        if risk_per_share <= 0:
            self.pattern_ready = False
            return
        take_profit_price = entry_price + self.p.rr_ratio * risk_per_share

        risk_amount = self.broker.getvalue() * self.p.risk_pct
        size = risk_amount / risk_per_share
        if size <= 0:
            self.pattern_ready = False
            return

        orders = self.buy_bracket(
            size=size,
            exectype=bt.Order.Market,
            stopprice=stop_price,
            limitprice=take_profit_price,
        )
        self.order = orders[0]
        self.pattern_ready = False
```
