import backtrader as bt


class PulseWave(bt.Strategy):
    params = (
        ('vwap_period', 20),
    )

    def __init__(self):
        self.order = None

        self.macd = bt.indicators.MACD(
            self.data.close,
            period_me1=12,
            period_me2=26,
            period_signal=9
        )
        self.rsi = bt.indicators.RSI(self.data.close, period=14)

        typical_price = (self.data.high + self.data.low + self.data.close) / 3.0
        tp_volume = typical_price * self.data.volume
        sum_tp_volume = bt.indicators.SumN(tp_volume, period=self.p.vwap_period)
        sum_volume = bt.indicators.SumN(self.data.volume, period=self.p.vwap_period)
        self.vwap = sum_tp_volume / sum_volume

        self.crossover = bt.indicators.CrossOver(self.macd.macd, self.macd.signal)

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        if self.order:
            return

        if not self.position:
            if (self.data.close[0] > self.vwap[0] and
                self.crossover[0] > 0 and
                self.rsi[0] < 70):
                self.order = self.buy()
        else:
            if (self.data.close[0] < self.vwap[0] and
                self.crossover[0] < 0 and
                self.rsi[0] > 30):
                self.order = self.close()
