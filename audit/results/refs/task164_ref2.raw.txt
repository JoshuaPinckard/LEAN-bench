import backtrader as bt


class CSILiteMomentum(bt.Strategy):
    params = (
        ("atr_period", 14),
        ("macd_fast", 12),
        ("macd_slow", 26),
        ("macd_signal", 9),
        ("sma_period", 50),
        ("rsi_period", 14),
        ("s_threshold", 1.5),
        ("tranche_pct", 0.05),
        ("max_tranches", 3),
        ("max_exposure", 0.50),
        ("peak_drop_pct", 0.70),
        ("rsi_exit_threshold", 45.0),
    )

    def __init__(self):
        self.order = None
        self.tranches = 0
        self.peak_S = 0.0

        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.sma_atr = bt.indicators.SMA(self.atr, period=self.p.sma_period)
        self.macd = bt.indicators.MACDHisto(
            self.data,
            period_me1=self.p.macd_fast,
            period_me2=self.p.macd_slow,
            period_signal=self.p.macd_signal,
        )
        self.sma_price = bt.indicators.SMA(
            self.data.close, period=self.p.sma_period
        )
        self.sma_vol = bt.indicators.SMA(
            self.data.volume, period=self.p.sma_period
        )
        self.rsi = bt.indicators.RSI(self.data.close, period=self.p.rsi_period)

    def notify_order(self, order):
        if order.status in [order.Completed]:
            if not self.position:
                self.tranches = 0
                self.peak_S = 0.0
            self.order = None
        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            if order.isbuy():
                if not self.position:
                    self.tranches = 0
                    self.peak_S = 0.0
                else:
                    self.tranches = max(0, self.tranches - 1)
            self.order = None

    def next(self):
        macd_hist = self.macd.histo[0]
        atr14 = self.atr[0]
        sma_atr50 = self.sma_atr[0]
        vol = self.data.volume[0]
        sma_vol50 = self.sma_vol[0]

        if sma_atr50 > 0 and sma_vol50 > 0:
            S = (
                max(0.0, macd_hist)
                * (atr14 / sma_atr50)
                * (vol / sma_vol50)
                * 100.0
            )
        else:
            S = 0.0

        macd_line = self.macd.macd[0]
        macd_signal = self.macd.signal[0]

        if not self.position:
            if self.order is not None:
                return

            entry_cond = (
                S > self.p.s_threshold
                and self.data.close[0] > self.sma_price[0]
                and macd_line > macd_signal
            )

            if entry_cond:
                capital = self.broker.getvalue()
                size = int((capital * self.p.tranche_pct) / self.data.close[0])
                if size > 0:
                    self.peak_S = S
                    self.tranches = 1
                    self.order = self.buy(size=size)
        else:
            self.peak_S = max(self.peak_S, S)

            macd_cross_down = (
                self.macd.macd[-1] >= self.macd.signal[-1]
            ) and (macd_line < macd_signal)
            rsi_exit = self.rsi[0] < self.p.rsi_exit_threshold
            peak_drop_exit = S < (self.p.peak_drop_pct * self.peak_S)

            if peak_drop_exit or macd_cross_down or rsi_exit:
                if self.order is None:
                    self.order = self.close()
            else:
                if self.order is not None:
                    return

                capital = self.broker.getvalue()
                exposure = (self.position.size * self.data.close[0]) / capital

                if (
                    self.tranches < self.p.max_tranches
                    and exposure < self.p.max_exposure
                    and S > self.p.s_threshold
                ):
                    size = int(
                        (capital * self.p.tranche_pct) / self.data.close[0]
                    )
                    if size > 0:
                        self.tranches += 1
                        self.order = self.buy(size=size)
