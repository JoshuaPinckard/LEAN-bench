import backtrader as bt


class ZScoreMeanReversion(bt.Strategy):
    params = (
        ("period", 120),
        ("z_long_exit", -1.5),
        ("z_short_exit", 1.5),
        ("stop_loss_pct", 0.05),
        ("trail_stop_pct", 0.01),
    )

    def __init__(self):
        self.ma = bt.ind.OLS_LinearRegressionCurve(
            self.data.close, period=self.params.period
        )
        self.std = bt.ind.StdDev(self.data.close, period=self.params.period)
        self.zscore = (self.data.close - self.ma) / self.std

        self.order = None
        self.entry_price = None
        self.highest_price = None
        self.lowest_price = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            if self.position.size != 0:
                self.entry_price = order.executed.price
                self.highest_price = self.data.high[0]
                self.lowest_price = self.data.low[0]
            else:
                self.entry_price = None
                self.highest_price = None
                self.lowest_price = None
            self.order = None
        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        if self.order:
            return

        if not self.position:
            # Entry conditions based on Z-Score crossing 0
            if self.zscore[0] > 0 and self.zscore[-1] <= 0:
                self.order = self.buy()
            elif self.zscore[0] < 0 and self.zscore[-1] >= 0:
                self.order = self.sell()
        else:
            if self.entry_price is None:
                self.entry_price = self.position.price
                self.highest_price = self.data.high[0]
                self.lowest_price = self.data.low[0]

            if self.position.size > 0:
                self.highest_price = max(self.highest_price, self.data.high[0])
                fixed_stop = self.entry_price * (1.0 - self.params.stop_loss_pct)
                trail_stop = self.highest_price * (1.0 - self.params.trail_stop_pct)

                if (
                    self.zscore[0] <= self.params.z_long_exit
                    or self.data.low[0] <= fixed_stop
                    or self.data.low[0] <= trail_stop
                ):
                    self.order = self.close()

            elif self.position.size < 0:
                self.lowest_price = min(self.lowest_price, self.data.low[0])
                fixed_stop = self.entry_price * (1.0 + self.params.stop_loss_pct)
                trail_stop = self.lowest_price * (1.0 + self.params.trail_stop_pct)

                if (
                    self.zscore[0] >= self.params.z_short_exit
                    or self.data.high[0] >= fixed_stop
                    or self.data.high[0] >= trail_stop
                ):
                    self.order = self.close()
