import math
import backtrader as bt


class DonchianBreakoutStrategy(bt.Strategy):
    params = (
        ('donchian_period', 20),
        ('adx_period', 14),
        ('atr_period', 14),
        ('adx_threshold', 25),
        ('risk_pct', 0.01),
        ('stop_atr_mult', 2.0),
        ('stop_pct', 0.02),
    )

    def __init__(self):
        self.donchian_top = bt.indicators.Highest(self.data.high, period=self.p.donchian_period)
        self.donchian_bottom = bt.indicators.Lowest(self.data.low, period=self.p.donchian_period)
        self.adx = bt.indicators.ADX(self.data, period=self.p.adx_period)
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)

        self.order = None
        self.entry_pending = False
        self.stop_dist = None
        self.stop_order = None

    def notify_order(self, order):
        if order.status == order.Completed:
            if order is self.stop_order:
                self.stop_order = None
                self.order = None
                return
            if self.entry_pending:
                self.entry_pending = False
                entry_price = order.executed.price
                size = abs(order.executed.size)
                if order.isbuy():
                    stop_price = entry_price - self.stop_dist
                    self.stop_order = self.sell(exectype=bt.Order.Stop, price=stop_price, size=size)
                else:
                    stop_price = entry_price + self.stop_dist
                    self.stop_order = self.buy(exectype=bt.Order.Stop, price=stop_price, size=size)
                self.order = None
                return
            self.order = None
            return

        if order.status in (order.Canceled, order.Margin, order.Rejected):
            if order is self.stop_order:
                self.stop_order = None
            if order is self.order:
                self.order = None
                self.entry_pending = False

    def next(self):
        if self.order:
            return

        close = self.data.close[0]

        if not self.position:
            long_signal = close > self.donchian_top[0] and self.adx[0] > self.p.adx_threshold
            short_signal = close < self.donchian_bottom[0] and self.adx[0] > self.p.adx_threshold

            if long_signal or short_signal:
                stop_dist = max(self.p.stop_atr_mult * self.atr[0], self.p.stop_pct * close)
                equity = self.broker.getvalue()
                size = math.floor((equity * self.p.risk_pct) / stop_dist)
                if size > 0:
                    self.stop_dist = stop_dist
                    self.entry_pending = True
                    if long_signal:
                        self.order = self.buy(size=size)
                    else:
                        self.order = self.sell(size=size)
        else:
            exit_signal = False
            if self.position.size > 0 and close < self.donchian_bottom[0]:
                exit_signal = True
            elif self.position.size < 0 and close > self.donchian_top[0]:
                exit_signal = True

            if exit_signal:
                if self.stop_order is not None:
                    self.cancel(self.stop_order)
                    self.stop_order = None
                self.order = self.close()
