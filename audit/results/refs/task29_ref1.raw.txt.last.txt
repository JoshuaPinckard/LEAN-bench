```python
import backtrader as bt
from collections import deque


class GapTradingStrategy(bt.Strategy):
    def __init__(self):
        self.order = None
        self.current_session = None
        self.session_close = None

        self.gaps = deque(maxlen=20)
        self.h4_closes = deque(maxlen=20)
        self.h4_bar_count = 0
        self.h4_close_sum = 0.0
        self.h4_sma = None

    def notify_order(self, order):
        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            self.order = None

    def next(self):
        session_date = self.data.datetime.datetime(0).date()

        if self.current_session is None:
            self.current_session = session_date
        elif session_date != self.current_session:
            if self.session_close is not None:
                gap = self.data.open[0] - self.session_close

                if len(self.gaps) == 20 and self.h4_sma is not None:
                    average_gap = sum(self.gaps) / 20.0
                    gap_size = abs(gap)
                    small_gap = gap_size <= 0.70 * average_gap
                    big_gap = gap_size > 1.50 * average_gap
                    uptrend = self.data.open[0] > self.h4_sma
                    downtrend = self.data.open[0] < self.h4_sma

                    if self.order is None and not self.position:
                        if gap > 0 and uptrend and (small_gap or big_gap):
                            self.order = self.buy()
                        elif gap < 0 and downtrend and (small_gap or big_gap):
                            self.order = self.sell()

                self.gaps.append(abs(gap))

            self.current_session = session_date
            self.h4_bar_count = 0

        self.h4_bar_count += 1
        if self.h4_bar_count == 4:
            self.h4_closes.append(self.data.close[0])
            self.h4_bar_count = 0

            if len(self.h4_closes) == 20:
                self.h4_sma = sum(self.h4_closes) / 20.0

        self.session_close = self.data.close[0]
```
