import math
import backtrader as bt


class PseudoProtectivePutATR(bt.Strategy):
    params = (
        ('base_price', 100.0),
        ('main_shares', 100),
    )

    def __init__(self):
        self.atr = bt.indicators.ATR(self.data, period=14)
        self.order = None
        self.in_position = False
        self.entry_price = None
        self.entry_bar = None
        self.stop_price = None
        self.expiration_bars = None
        self.next_expiration = 90
        self.pending_roll = False
        self.highest_k = 0

    def _submit_open(self, expiration):
        close = self.data.close[0]
        atr = self.atr[0]
        qty = math.floor(0.02 * (self.p.main_shares * close) / atr)
        if qty < 1:
            return
        self.next_expiration = expiration
        self.order = self.sell(size=qty)

    def _submit_close(self, roll):
        self.pending_roll = roll
        self.order = self.close()

    def next(self):
        if self.order:
            return

        close = self.data.close[0]
        atr = self.atr[0]
        base_price = self.p.base_price

        k_current = max(0, math.floor((close - base_price) / (0.10 * base_price)))
        new_breach = k_current > self.highest_k

        if self.in_position:
            bars_held = len(self) - self.entry_bar
            if close >= self.stop_price:
                self._submit_close(roll=False)
            elif close <= self.entry_price * 0.85:
                self._submit_close(roll=False)
            elif bars_held >= self.expiration_bars:
                self._submit_close(roll=False)
            elif new_breach and (atr / close) <= 0.04:
                self._submit_close(roll=True)
            if new_breach:
                self.highest_k = k_current
        else:
            if new_breach:
                self.highest_k = k_current
            if close >= 3 * base_price and (atr / close) <= 0.04:
                self._submit_open(90)

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status == order.Completed:
            if order.issell():
                self.entry_price = order.executed.price
                self.entry_bar = len(self)
                self.stop_price = self.entry_price + self.atr[0]
                self.expiration_bars = self.next_expiration
                self.in_position = True
                self.order = None
            else:
                self.in_position = False
                self.entry_price = None
                self.entry_bar = None
                self.stop_price = None
                self.expiration_bars = None
                self.order = None
                if self.pending_roll:
                    self.pending_roll = False
                    close = self.data.close[0]
                    atr = self.atr[0]
                    if atr / close <= 0.04:
                        self._submit_open(180)
        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            self.pending_roll = False
            self.order = None
