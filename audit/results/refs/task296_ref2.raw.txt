import backtrader as bt


class DailyRSISMAStrategy(bt.Strategy):
    def __init__(self):
        self.sma = bt.indicators.SMA(self.data.close, period=20)
        self.crossover = bt.indicators.CrossOver(self.data.close, self.sma)
        self.order = None
        self.daily_closes = []
        self.day_allows_trading = False
        self.current_date = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return
        self.order = None

    def _compute_daily_rsi(self):
        if len(self.daily_closes) < 15:
            return None

        gains = []
        losses = []
        for i in range(1, len(self.daily_closes)):
            change = self.daily_closes[i] - self.daily_closes[i - 1]
            if change > 0:
                gains.append(change)
                losses.append(0.0)
            else:
                gains.append(0.0)
                losses.append(-change)

        if len(gains) < 14:
            return None

        avg_gain = sum(gains[:14]) / 14.0
        avg_loss = sum(losses[:14]) / 14.0

        for i in range(14, len(gains)):
            avg_gain = (avg_gain * 13.0 + gains[i]) / 14.0
            avg_loss = (avg_loss * 13.0 + losses[i]) / 14.0

        if avg_loss == 0:
            return 100.0

        rs = avg_gain / avg_loss
        return 100.0 - (100.0 / (1.0 + rs))

    def next(self):
        bar_date = self.data.datetime.date(0)
        if self.current_date is None:
            self.current_date = bar_date
        elif bar_date != self.current_date:
            self.daily_closes.append(self.data.close[-1])
            self.current_date = bar_date
            rsi = self._compute_daily_rsi()
            self.day_allows_trading = rsi is not None and rsi < 30.0

        if self.order:
            return

        t = self.data.datetime.time(0)
        is_exit_time = (t.hour > 15) or (t.hour == 15 and t.minute >= 45)

        if self.position:
            entry_price = self.position.price
            stop_price = entry_price * 0.98
            tp_price = entry_price * 1.04

            if self.data.low[0] <= stop_price:
                self.order = self.close()
            elif self.data.high[0] >= tp_price:
                self.order = self.close()
            elif self.crossover[0] < 0:
                self.order = self.close()
            elif is_exit_time:
                self.order = self.close()
        else:
            if self.day_allows_trading and not is_exit_time:
                if self.crossover[0] > 0:
                    capital = self.broker.getvalue()
                    allocated = capital * 0.05
                    size = int(allocated / self.data.close[0])
                    if size > 0:
                        self.order = self.buy(size=size)
