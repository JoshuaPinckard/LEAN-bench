import backtrader as bt


class MonthlySeasonalityStrategy(bt.Strategy):
    def __init__(self):
        self.order = None
        self.entry_order = None
        self.entry_price = None
        self.reenter_after_close = None

        self.current_month = None
        self.month_first_open = None
        self.month_last_close = None
        self.monthly_returns = {month: [] for month in range(1, 13)}

    def _seasonal_bias(self, month):
        returns = self.monthly_returns[month]
        if len(returns) < 1:
            return 0

        average_return = sum(returns) / len(returns)
        if average_return > 0:
            return 1
        if average_return < 0:
            return -1
        return 0

    def _submit_entry(self, target):
        if target > 0:
            self.order = self.buy()
            self.entry_order = self.order
        elif target < 0:
            self.order = self.sell()
            self.entry_order = self.order

    def _apply_monthly_target(self, target):
        position_size = self.position.size

        if position_size > 0:
            if target != 1:
                self.reenter_after_close = target if target == -1 else None
                self.entry_price = None
                self.order = self.close()
        elif position_size < 0:
            if target != -1:
                self.reenter_after_close = target if target == 1 else None
                self.entry_price = None
                self.order = self.close()
        elif target != 0:
            self._submit_entry(target)

    def notify_order(self, order):
        if order.status not in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            return

        if order is self.entry_order:
            if order.status == order.Completed:
                self.entry_price = order.executed.price
            self.entry_order = None

        if order is self.order:
            self.order = None

        if (
            order.status == order.Completed
            and self.reenter_after_close is not None
            and not self.position
            and self.order is None
        ):
            target = self.reenter_after_close
            self.reenter_after_close = None
            self._submit_entry(target)
        elif order.status != order.Completed:
            self.reenter_after_close = None

    def next(self):
        dt = self.data.datetime.datetime(0)
        month_key = (dt.year, dt.month)

        if self.current_month is None:
            self.current_month = month_key
            self.month_first_open = self.data.open[0]
            self.month_last_close = self.data.close[0]
            self._apply_monthly_target(self._seasonal_bias(dt.month))
            return

        if month_key != self.current_month:
            completed_month = self.current_month[1]
            monthly_return = (
                self.month_last_close / self.month_first_open
            ) - 1.0
            self.monthly_returns[completed_month].append(monthly_return)

            self.current_month = month_key
            self.month_first_open = self.data.open[0]
            self.month_last_close = self.data.close[0]

            if self.order is None:
                self._apply_monthly_target(self._seasonal_bias(dt.month))
            return

        self.month_last_close = self.data.close[0]

        if self.order is None and self.entry_price is not None:
            if self.position.size > 0 and self.data.low[0] <= self.entry_price * 0.98:
                self.entry_price = None
                self.reenter_after_close = None
                self.order = self.close()
            elif self.position.size < 0 and self.data.high[0] >= self.entry_price * 1.02:
                self.entry_price = None
                self.reenter_after_close = None
                self.order = self.close()
