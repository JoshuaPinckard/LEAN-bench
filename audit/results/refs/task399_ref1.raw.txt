import backtrader as bt
import math


class SupertrendIntradayStrategy(bt.Strategy):
    params = (
        ("atr_period", 10),
        ("multiplier", 3.0),
        ("volume_sma_period", 20),
        ("equity_percent", 0.25),
    )

    def __init__(self):
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.volume_sma = bt.indicators.SMA(
            self.data.volume, period=self.p.volume_sma_period
        )

        self.order = None
        self.final_upper = None
        self.final_lower = None
        self.trend = None

    def notify_order(self, order):
        if order == self.order and order.status in (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            self.order = None

    def _size(self):
        return math.floor(
            (self.broker.getvalue() * self.p.equity_percent) / self.data.close[0]
        )

    def next(self):
        if self.order:
            return

        hl2 = (self.data.high[0] + self.data.low[0]) / 2.0
        basic_upper = hl2 + (self.p.multiplier * self.atr[0])
        basic_lower = hl2 - (self.p.multiplier * self.atr[0])

        if self.final_upper is None:
            self.final_upper = basic_upper
            self.final_lower = basic_lower
            self.trend = "up"
            return

        prior_final_upper = self.final_upper
        prior_final_lower = self.final_lower
        prior_trend = self.trend
        prior_close = self.data.close[-1]

        self.final_upper = (
            basic_upper
            if basic_upper < prior_final_upper or prior_close > prior_final_upper
            else prior_final_upper
        )
        self.final_lower = (
            basic_lower
            if basic_lower > prior_final_lower or prior_close < prior_final_lower
            else prior_final_lower
        )

        if self.data.close[0] > prior_final_upper:
            self.trend = "up"
        elif self.data.close[0] < prior_final_lower:
            self.trend = "down"
        else:
            self.trend = prior_trend

        flipped_up = prior_trend == "down" and self.trend == "up"
        flipped_down = prior_trend == "up" and self.trend == "down"

        if self.position.size > 0 and flipped_down:
            self.order = self.close()
        elif self.position.size < 0 and flipped_up:
            self.order = self.close()
        elif self.position.size == 0 and self.data.volume[0] > self.volume_sma[0]:
            size = self._size()
            if size > 0:
                if flipped_up:
                    self.order = self.buy(size=size)
                elif flipped_down:
                    self.order = self.sell(size=size)
