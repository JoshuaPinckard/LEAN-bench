import backtrader as bt


class CVaRMomentumStrategy(bt.Strategy):
    params = (
        ('ema_fast', 20),
        ('ema_slow', 100),
        ('atr_period', 14),
        ('vol_sma_period', 20),
        ('c', 2.0),
        ('r', 0.02),
        ('min_volume', 0.0),
    )

    def __init__(self):
        self.order = None
        self.ema20 = bt.indicators.EMA(self.data.close, period=self.p.ema_fast)
        self.ema100 = bt.indicators.EMA(self.data.close, period=self.p.ema_slow)
        self.macd = bt.indicators.MACD(self.data.close)
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.vol_sma = bt.indicators.SMA(self.data.volume, period=self.p.vol_sma_period)
        self.crossover = bt.indicators.CrossOver(self.ema20, self.ema100)

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        if self.order:
            return

        bullish = (self.ema20[0] > self.ema100[0]) and (self.macd.macd[0] > 0)
        liquidity_ok = self.vol_sma[0] >= self.p.min_volume

        if bullish and liquidity_ok:
            close_price = self.data.close[0]
            if close_price > 0:
                es95 = self.p.c * (self.atr[0] / close_price)
                if es95 > 0:
                    w = min(1.0, self.p.r / es95)
                else:
                    w = 1.0
            else:
                w = 0.0
        else:
            w = 0.0

        self.order = self.order_target_percent(target=w)
