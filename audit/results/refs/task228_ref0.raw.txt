import backtrader as bt


class SwingTrendRsiPullbackStrategy(bt.Strategy):
    def __init__(self):
        self.ema_fast = bt.indicators.EMA(self.data.close, period=12)
        self.ema_slow = bt.indicators.EMA(self.data.close, period=26)
        self.rsi = bt.indicators.RSI(self.data.close, period=14)
        self.atr = bt.indicators.ATR(self.data, period=14)
        self.tr = bt.indicators.TrueRange(self.data)

        self.order = None
        self.pending_action = None

        self.entry_price = None
        self.stop_price = None
        self.breakeven_moved = False

        self.long_dip_seen = False
        self.long_armed = False
        self.long_signal_high = None

        self.short_dip_seen = False
        self.short_armed = False
        self.short_signal_low = None

        self.equity_peak = self.broker.getvalue()

    def _reset_long_setup(self):
        self.long_dip_seen = False
        self.long_armed = False
        self.long_signal_high = None

    def _reset_short_setup(self):
        self.short_dip_seen = False
        self.short_armed = False
        self.short_signal_low = None

    def _enter_long(self, ref_price):
        equity = self.broker.getvalue()
        risk_amount = 0.01 * equity
        stop_distance = ref_price * 0.015
        size = risk_amount / stop_distance
        if size <= 0:
            return
        self.pending_action = 'enter_long'
        self.order = self.buy(size=size)
        self._reset_long_setup()
        self._reset_short_setup()

    def _enter_short(self, ref_price):
        equity = self.broker.getvalue()
        risk_amount = 0.01 * equity
        stop_distance = ref_price * 0.015
        size = risk_amount / stop_distance
        if size <= 0:
            return
        self.pending_action = 'enter_short'
        self.order = self.sell(size=size)
        self._reset_long_setup()
        self._reset_short_setup()

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return

        if order.status == order.Completed:
            if self.pending_action == 'enter_long':
                self.entry_price = order.executed.price
                self.stop_price = self.entry_price * (1 - 0.015)
                self.breakeven_moved = False
            elif self.pending_action == 'enter_short':
                self.entry_price = order.executed.price
                self.stop_price = self.entry_price * (1 + 0.015)
                self.breakeven_moved = False
            elif self.pending_action == 'exit':
                self.entry_price = None
                self.stop_price = None
                self.breakeven_moved = False

        self.order = None
        self.pending_action = None

    def next(self):
        equity = self.broker.getvalue()
        if equity > self.equity_peak:
            self.equity_peak = equity

        if self.order:
            return

        close = self.data.close[0]
        open_ = self.data.open[0]
        high = self.data.high[0]
        low = self.data.low[0]
        prev_close = self.data.close[-1]

        ema_fast = self.ema_fast[0]
        ema_slow = self.ema_slow[0]
        rsi_now = self.rsi[0]
        rsi_prev = self.rsi[-1]

        drawdown = (self.equity_peak - equity) / self.equity_peak if self.equity_peak > 0 else 0.0

        if self.position:
            if self.position.size > 0:
                profit_frac = (close - self.entry_price) / self.entry_price
                if not self.breakeven_moved and profit_frac >= 0.01:
                    self.stop_price = self.entry_price
                    self.breakeven_moved = True

                if rsi_now > 70:
                    self.pending_action = 'exit'
                    self.order = self.close()
                elif low <= self.stop_price:
                    self.pending_action = 'exit'
                    self.order = self.close()
            elif self.position.size < 0:
                profit_frac = (self.entry_price - close) / self.entry_price
                if not self.breakeven_moved and profit_frac >= 0.01:
                    self.stop_price = self.entry_price
                    self.breakeven_moved = True

                if rsi_now < 30:
                    self.pending_action = 'exit'
                    self.order = self.close()
                elif high >= self.stop_price:
                    self.pending_action = 'exit'
                    self.order = self.close()
            return

        if ema_fast <= ema_slow:
            self._reset_long_setup()
        else:
            if rsi_prev >= 40 and rsi_now < 40:
                self.long_dip_seen = (rsi_now >= 30)
                self.long_armed = False
                self.long_signal_high = None
            if self.long_dip_seen and rsi_now < 30:
                self._reset_long_setup()
            if self.long_dip_seen and not self.long_armed and rsi_prev <= 35 and rsi_now > 35:
                self.long_armed = True
                self.long_signal_high = high

        if ema_fast >= ema_slow:
            self._reset_short_setup()
        else:
            if rsi_prev <= 60 and rsi_now > 60:
                self.short_dip_seen = (rsi_now <= 70)
                self.short_armed = False
                self.short_signal_low = None
            if self.short_dip_seen and rsi_now > 70:
                self._reset_short_setup()
            if self.short_dip_seen and not self.short_armed and rsi_prev >= 65 and rsi_now < 65:
                self.short_armed = True
                self.short_signal_low = low

        daily_range_ok = (high - low) >= 0.005 * open_
        news_spike = (self.tr[0] > 2 * self.atr[0]) or (abs(open_ - prev_close) / prev_close > 0.01)
        can_enter = daily_range_ok and (not news_spike) and (drawdown <= 0.05)

        if can_enter:
            if self.long_armed and ema_fast > ema_slow and close > self.long_signal_high:
                self._enter_long(close)
            elif self.short_armed and ema_fast < ema_slow and close < self.short_signal_low:
                self._enter_short(close)
