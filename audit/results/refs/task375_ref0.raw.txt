import backtrader as bt


class KeltnerWillRStrategy(bt.Strategy):
    params = (
        ('ema_period', 20),
        ('atr_period', 14),
        ('keltner_mult', 2.0),
        ('willr_period', 14),
        ('stop_atr_mult', 1.5),
    )

    def __init__(self):
        self.ema = bt.indicators.EMA(self.data.close, period=self.p.ema_period)
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.willr = bt.indicators.WilliamsR(self.data, period=self.p.willr_period)
        self.order = None
        self.stop_order = None

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return

        if order.status == order.Completed:
            if order is self.stop_order:
                self.stop_order = None
            elif self.position.size != 0:
                entry_price = order.executed.price
                if self.position.size > 0:
                    stop_price = entry_price - self.p.stop_atr_mult * self.atr[0]
                    self.stop_order = self.sell(
                        exectype=bt.Order.Stop,
                        price=stop_price,
                        size=self.position.size,
                    )
                else:
                    stop_price = entry_price + self.p.stop_atr_mult * self.atr[0]
                    self.stop_order = self.buy(
                        exectype=bt.Order.Stop,
                        price=stop_price,
                        size=abs(self.position.size),
                    )
            self.order = None
        elif order.status in (order.Canceled, order.Margin, order.Rejected):
            if order is self.stop_order:
                self.stop_order = None
            else:
                self.order = None

    def next(self):
        upper = self.ema[0] + self.p.keltner_mult * self.atr[0]
        lower = self.ema[0] - self.p.keltner_mult * self.atr[0]
        close = self.data.close[0]

        if self.order:
            return

        if not self.position:
            if close < lower and self.willr[0] < -80:
                self.order = self.buy()
            elif close > upper and self.willr[0] > -20:
                self.order = self.sell()
        else:
            if self.position.size > 0 and close >= self.ema[0]:
                if self.stop_order:
                    self.cancel(self.stop_order)
                self.order = self.close()
            elif self.position.size < 0 and close <= self.ema[0]:
                if self.stop_order:
                    self.cancel(self.stop_order)
                self.order = self.close()
