import backtrader as bt
import math


class MeanVarianceStrategy(bt.Strategy):
    params = (
        ("lambda_", 2.5),
        ("wmin", 0.0),
        ("wmax", 1.0),
    )

    def __init__(self):
        self.log_return = bt.ind.LogN(self.data.close / self.data.close(-1))
        self.mu_hat = bt.ind.SMA(self.log_return, period=252)
        self.sigma_hat = bt.ind.ATR(self.data, period=63) / self.data.close
        self.order = None
        self.last_rebalance_month = None

    def next(self):
        current_month = (self.data.datetime.date(0).year,
                         self.data.datetime.date(0).month)

        if self.order or current_month == self.last_rebalance_month:
            return

        self.last_rebalance_month = current_month

        mu_hat = self.mu_hat[0]
        sigma_hat = self.sigma_hat[0]

        if sigma_hat < 0.005:
            weight = 0.0
        else:
            weight = mu_hat / (self.p.lambda_ * sigma_hat * sigma_hat)
            weight = max(self.p.wmin, min(self.p.wmax, weight))

        if math.isfinite(weight):
            self.order = self.order_target_percent(target=weight)

    def notify_order(self, order):
        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            self.order = None
