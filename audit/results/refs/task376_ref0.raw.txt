import backtrader as bt


class RsiDivergenceStrategy(bt.Strategy):
    params = (
        ('rsi_period', 14),
        ('pivot_n', 5),
        ('rsi_overbought', 70),
        ('rsi_oversold', 30),
        ('stop_loss_pct', 0.02),
    )

    def __init__(self):
        self.rsi = bt.indicators.RSI(self.data.close, period=self.p.rsi_period)
        self.order = None
        self.entering = False
        self.entry_price = None
        self.pivot_lows = []
        self.pivot_highs = []
        self.pending_long_entry = False
        self.pending_short_entry = False

    def notify_order(self, order):
        if order.status == order.Completed:
            if self.entering:
                self.entry_price = order.executed.price
                self.entering = False
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        n = self.p.pivot_n

        if self.order is None and not self.position:
            if self.pending_long_entry:
                cash = self.broker.getcash()
                price = self.data.close[0]
                size = cash / price
                self.entering = True
                self.order = self.buy(size=size)
            elif self.pending_short_entry:
                cash = self.broker.getcash()
                price = self.data.close[0]
                size = cash / price
                self.entering = True
                self.order = self.sell(size=size)
        self.pending_long_entry = False
        self.pending_short_entry = False

        if self.position:
            if self.position.size > 0:
                stop_price = self.entry_price * (1 - self.p.stop_loss_pct)
                crossed_up = self.rsi[-1] <= self.p.rsi_overbought and self.rsi[0] > self.p.rsi_overbought
                if crossed_up or self.data.low[0] <= stop_price:
                    if self.order is None:
                        self.order = self.close()
            elif self.position.size < 0:
                stop_price = self.entry_price * (1 + self.p.stop_loss_pct)
                crossed_down = self.rsi[-1] >= self.p.rsi_oversold and self.rsi[0] < self.p.rsi_oversold
                if crossed_down or self.data.high[0] >= stop_price:
                    if self.order is None:
                        self.order = self.close()

        if len(self.data) > 2 * n:
            window_lows = [self.data.low[-k] for k in range(0, 2 * n + 1)]
            window_highs = [self.data.high[-k] for k in range(0, 2 * n + 1)]
            pivot_low_val = self.data.low[-n]
            pivot_high_val = self.data.high[-n]

            if pivot_low_val == min(window_lows):
                rsi_val = self.rsi[-n]
                self.pivot_lows.append((pivot_low_val, rsi_val))
                if len(self.pivot_lows) >= 2:
                    prev_price, prev_rsi = self.pivot_lows[-2]
                    curr_price, curr_rsi = self.pivot_lows[-1]
                    if curr_price < prev_price and curr_rsi > prev_rsi:
                        if not self.position:
                            self.pending_long_entry = True

            if pivot_high_val == max(window_highs):
                rsi_val = self.rsi[-n]
                self.pivot_highs.append((pivot_high_val, rsi_val))
                if len(self.pivot_highs) >= 2:
                    prev_price, prev_rsi = self.pivot_highs[-2]
                    curr_price, curr_rsi = self.pivot_highs[-1]
                    if curr_price > prev_price and curr_rsi < prev_rsi:
                        if not self.position:
                            self.pending_short_entry = True
