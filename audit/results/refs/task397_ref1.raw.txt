import backtrader as bt


class WilliamsRBandsStrategy(bt.Strategy):
    params = (
        ("stop_loss_pct", 0.01),
    )

    def __init__(self):
        self.williams_r = bt.indicators.WilliamsR(self.data, period=14)
        self.wr_mean = bt.indicators.SimpleMovingAverage(self.williams_r, period=20)
        self.wr_std = bt.indicators.StandardDeviation(self.williams_r, period=20)
        self.upper_band = self.wr_mean + 2 * self.wr_std
        self.lower_band = self.wr_mean - 2 * self.wr_std

        self.cross_upper = bt.indicators.CrossOver(self.williams_r, self.upper_band)
        self.cross_lower = bt.indicators.CrossOver(self.williams_r, self.lower_band)
        self.cross_mean = bt.indicators.CrossOver(self.williams_r, self.wr_mean)

        self.entry_order = None
        self.exit_order = None
        self.stop_order = None

    def next(self):
        if self.entry_order or self.exit_order:
            return

        if not self.position:
            if self.stop_order:
                return

            if self.cross_upper[0] > 0:
                self.entry_order = self.buy()
            elif self.cross_lower[0] < 0:
                self.entry_order = self.sell()
            return

        if self.position.size > 0 and self.cross_mean[0] < 0:
            if self.stop_order:
                self.cancel(self.stop_order)
            self.exit_order = self.close()

        elif self.position.size < 0 and self.cross_mean[0] > 0:
            if self.stop_order:
                self.cancel(self.stop_order)
            self.exit_order = self.close()

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return

        if order is self.entry_order:
            self.entry_order = None

            if order.status == order.Completed:
                if self.position.size > 0:
                    self.stop_order = self.sell(
                        exectype=bt.Order.Stop,
                        price=self.position.price * (1.0 - self.p.stop_loss_pct),
                    )
                elif self.position.size < 0:
                    self.stop_order = self.buy(
                        exectype=bt.Order.Stop,
                        price=self.position.price * (1.0 + self.p.stop_loss_pct),
                    )

        elif order is self.exit_order:
            self.exit_order = None

        elif order is self.stop_order:
            self.stop_order = None

        if order.status in (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            if order is self.entry_order:
                self.entry_order = None
            elif order is self.exit_order:
                self.exit_order = None
            elif order is self.stop_order:
                self.stop_order = None
