```python
import math
import backtrader as bt


class WeeklyORB(bt.Strategy):
    params = (
        ('ema_period', 20),
        ('sma_period', 50),
        ('risk_pct', 0.01),
    )

    def __init__(self):
        self.ema20 = bt.indicators.EMA(self.data.close, period=self.p.ema_period)
        self.sma50 = bt.indicators.SMA(self.data.close, period=self.p.sma_period)

        self.order = None
        self.current_week = None

        self.monday_high = None
        self.monday_low = None
        self.level_05 = None

        self.breakout_occurred = False
        self.breakout_days_count = 0
        self.traded_this_week = False

        self.sl_price = None
        self.tp_price = None

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        dt = self.data.datetime.date(0)
        week_id = dt.isocalendar()[:2]

        # Reset range parameters on the first trading day of a new week
        if week_id != self.current_week:
            self.current_week = week_id
            self.monday_high = self.data.high[0]
            self.monday_low = self.data.low[0]
            self.level_05 = (self.monday_high + self.monday_low) / 2.0
            self.breakout_occurred = False
            self.breakout_days_count = 0
            self.traded_this_week = False
            return

        # Position exit management
        if self.position:
            if self.data.high[0] >= self.sl_price or self.data.low[0] <= self.tp_price:
                if self.order is None:
                    self.order = self.close()
            return

        if self.order is not None:
            return

        # Breakout state update
        if self.breakout_occurred:
            self.breakout_days_count += 1
            if self.breakout_days_count > 2:
                self.breakout_occurred = False
        else:
            if self.data.close[0] < self.monday_low:
                self.breakout_occurred = True
                self.breakout_days_count = 0

        # Short signal evaluation
        if self.breakout_occurred and 1 <= self.breakout_days_count <= 2 and not self.traded_this_week:
            price_returned = self.data.high[0] >= self.level_05

            trend_filter = (
                self.data.close[0] < self.ema20[0] and
                self.data.close[0] < self.sma50[0] and
                self.ema20[0] < self.sma50[0] and
                self.ema20[0] < self.ema20[-1]
            )

            if price_returned and trend_filter:
                risk_per_share = self.monday_high - self.level_05
                if risk_per_share > 0:
                    capital = self.broker.getvalue()
                    risk_amount = capital * self.p.risk_pct
                    size = math.floor(risk_amount / risk_per_share)
                    if size > 0:
                        self.sl_price = self.monday_high
                        self.tp_price = self.monday_low
                        self.order = self.sell(size=size)
                        self.traded_this_week = True
```
