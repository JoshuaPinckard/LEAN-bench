import backtrader as bt
import math


class BreakoutAfterDoubleBottom(bt.Strategy):
    params = (
        ("support", 135.0),
        ("support_tolerance", 0.1),
        ("resistance", 191.8),
        ("stop_loss", 132.3),
        ("volume_period", 20),
        ("risk_fraction", 0.015),
    )

    def __init__(self):
        self.entry_order = None
        self.stop_order = None
        self.target_order = None
        self.alternative_exit_order = None

        self.first_bottom = None
        self.local_high_after_first = False
        self.pattern_low = None
        self.take_profit = None

    def _prior_consolidation(self):
        return all(
            134.2 <= self.data.close[-offset] <= self.p.resistance
            for offset in range(2, 22)
        )

    def _is_local_low(self):
        return (
            self.p.support - self.p.support_tolerance
            <= self.data.low[-1]
            <= self.p.support + self.p.support_tolerance
            and self.data.low[-1] <= self.data.low[-2]
            and self.data.low[-1] <= self.data.low[0]
        )

    def _is_local_high(self):
        return (
            self.data.high[-1] >= self.data.high[-2]
            and self.data.high[-1] >= self.data.high[0]
        )

    def next(self):
        if len(self.data) < 22:
            return

        if self.position:
            if (
                self.alternative_exit_order is None
                and self.data.close[0] < self.p.support
            ):
                if self.stop_order is not None:
                    self.cancel(self.stop_order)
                if self.target_order is not None:
                    self.cancel(self.target_order)
                self.alternative_exit_order = self.close()
            return

        if self.entry_order is not None or self.alternative_exit_order is not None:
            return

        if self.pattern_low is None:
            if self.first_bottom is not None and self._is_local_high():
                self.local_high_after_first = True

            if self._is_local_low() and self._prior_consolidation():
                if (
                    self.first_bottom is not None
                    and self.local_high_after_first
                ):
                    self.pattern_low = min(self.first_bottom, self.data.low[-1])
                    self.take_profit = (
                        self.p.resistance
                        + (self.p.resistance - self.pattern_low)
                    )
                    self.first_bottom = None
                    self.local_high_after_first = False
                else:
                    self.first_bottom = self.data.low[-1]
                    self.local_high_after_first = False
            return

        average_volume = sum(
            self.data.volume[-offset] for offset in range(1, 21)
        ) / self.p.volume_period

        breakout = (
            self.data.close[-1] <= self.p.resistance
            and self.data.close[0] > self.p.resistance
        )
        volume_confirmed = self.data.volume[0] >= 1.2 * average_volume

        if breakout and volume_confirmed:
            risk_per_share = self.data.close[0] - self.p.stop_loss
            risk_capital = self.broker.getvalue() * self.p.risk_fraction
            size = math.floor(risk_capital / risk_per_share)

            if size > 0:
                self.entry_order = self.buy(size=size)
                self.pattern_low = None

    def notify_order(self, order):
        if order.status in (
            order.Submitted,
            order.Accepted,
        ):
            return

        was_entry = order is self.entry_order

        if order.status == order.Completed and was_entry:
            self.entry_order = None
            size = self.position.size
            self.stop_order = self.sell(
                size=size,
                exectype=bt.Order.Stop,
                price=self.p.stop_loss,
            )
            self.target_order = self.sell(
                size=size,
                exectype=bt.Order.Limit,
                price=self.take_profit,
                oco=self.stop_order,
            )
            return

        if order.status in (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            if order is self.entry_order:
                self.entry_order = None
            if order is self.stop_order:
                self.stop_order = None
            if order is self.target_order:
                self.target_order = None
            if order is self.alternative_exit_order:
                self.alternative_exit_order = None
