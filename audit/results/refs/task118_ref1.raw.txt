# complete strategy class
import backtrader as bt
import math


class EnhancedDCAStrategy(bt.Strategy):
    def __init__(self):
        self.rsi = bt.indicators.RSI(self.data.close, period=14)
        self.sma = bt.indicators.SMA(self.data.close, period=50)
        self.bb = bt.indicators.BollingerBands(
            self.data.close, period=20, devfactor=2
        )
        self.macd = bt.indicators.MACD(
            self.data.close,
            period_me1=12,
            period_me2=26,
            period_signal=9,
        )
        self.atr = bt.indicators.ATR(self.data, period=14)

        self.order = None
        self.entry_price = None

    def notify_order(self, order):
        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            if order.status == order.Completed:
                if order.isbuy():
                    self.entry_price = order.executed.price
                elif order.issell():
                    self.entry_price = None
            self.order = None

    def next(self):
        if self.order:
            return

        macd_cross_up = (
            self.macd.macd[-1] < self.macd.signal[-1]
            and self.macd.macd[0] > self.macd.signal[0]
        )
        macd_cross_down = (
            self.macd.macd[-1] > self.macd.signal[-1]
            and self.macd.macd[0] < self.macd.signal[0]
        )

        if self.position:
            stop_loss = self.entry_price - 2 * self.atr[0]
            if (
                self.rsi[0] > 70
                or macd_cross_down
                or self.data.close[0] < stop_loss
            ):
                self.order = self.close()
            return

        rsi_buy = self.rsi[0] < 30
        sma_buy = self.data.close[0] > self.sma[0]
        bb_buy = self.data.close[0] < self.bb.bot[0]
        macd_buy = macd_cross_up

        signal_count = sum([rsi_buy, sma_buy, bb_buy, macd_buy])

        trending = self.data.close[0] > self.sma[0]
        if trending:
            rsi_weight = 1
            sma_weight = 2
            bb_weight = 1
            macd_weight = 2
        else:
            rsi_weight = 2
            sma_weight = 1
            bb_weight = 2
            macd_weight = 1

        weighted_score = (
            rsi_weight * rsi_buy
            + sma_weight * sma_buy
            + bb_weight * bb_buy
            + macd_weight * macd_buy
        )

        if signal_count >= 3 and weighted_score >= 5 and self.atr[0] > 0:
            total_capital = self.broker.getvalue()
            close_price = self.data.close[0]

            base_position_size = 0.05 * total_capital
            atr_based_size = base_position_size / self.atr[0]

            current_exposure = abs(self.position.size) * close_price
            aggregate_capacity = max(
                0.0, 0.20 * total_capital - current_exposure
            )
            position_capital = min(0.05 * total_capital, aggregate_capacity)
            capital_capped_size = position_capital / close_price

            size = math.floor(min(atr_based_size, capital_capped_size))
            if size > 0:
                self.order = self.buy(size=size)
