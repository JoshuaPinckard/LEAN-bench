import backtrader as bt


class RSIDivergenceStrategy(bt.Strategy):
    params = (
        ("rsi_period", 14),
        ("pivot_n", 5),
        ("stop_loss_pct", 0.02),
    )

    def __init__(self):
        self.rsi = bt.indicators.RSI(self.data.close, period=self.p.rsi_period)
        self.order = None
        self.prev_pivot_low = None
        self.prev_pivot_high = None

    def notify_order(self, order):
        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            self.order = None

    def next(self):
        if self.order:
            return

        # Handle exits for existing positions
        if self.position:
            if self.position.size > 0:  # Long
                rsi_exit = self.rsi[-1] <= 70 and self.rsi[0] > 70
                sl_exit = self.data.low[0] <= self.position.price * (
                    1.0 - self.p.stop_loss_pct
                )
                if rsi_exit or sl_exit:
                    self.order = self.close()
            elif self.position.size < 0:  # Short
                rsi_exit = self.rsi[-1] >= 30 and self.rsi[0] < 30
                sl_exit = self.data.high[0] >= self.position.price * (
                    1.0 + self.p.stop_loss_pct
                )
                if rsi_exit or sl_exit:
                    self.order = self.close()

        # Pivot Detection Logic
        n = self.p.pivot_n
        window_size = 2 * n + 1
        if len(self) < window_size:
            return

        lows = [self.data.low[-i] for i in range(window_size)]
        highs = [self.data.high[-i] for i in range(window_size)]

        pivot_low_val = self.data.low[-n]
        pivot_high_val = self.data.high[-n]

        is_pivot_low = pivot_low_val == min(lows)
        is_pivot_high = pivot_high_val == max(highs)

        bullish_divergence = False
        bearish_divergence = False

        if is_pivot_low:
            curr_low_rsi = self.rsi[-n]
            if self.prev_pivot_low is not None:
                prev_low_val, prev_low_rsi = self.prev_pivot_low
                if pivot_low_val < prev_low_val and curr_low_rsi > prev_low_rsi:
                    bullish_divergence = True
            self.prev_pivot_low = (pivot_low_val, curr_low_rsi)

        if is_pivot_high:
            curr_high_rsi = self.rsi[-n]
            if self.prev_pivot_high is not None:
                prev_high_val, prev_high_rsi = self.prev_pivot_high
                if (
                    pivot_high_val > prev_high_val
                    and curr_high_rsi < prev_high_rsi
                ):
                    bearish_divergence = True
            self.prev_pivot_high = (pivot_high_val, curr_high_rsi)

        # Signal Execution for New Entries
        if not self.position:
            cash = self.broker.getcash()
            size = int(cash / self.data.close[0])
            if size > 0:
                if bullish_divergence:
                    self.order = self.buy(size=size)
                elif bearish_divergence:
                    self.order = self.sell(size=size)
