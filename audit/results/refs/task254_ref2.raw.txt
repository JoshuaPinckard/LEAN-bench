import backtrader as bt


class DynamicSequenceStrategy(bt.Strategy):
    params = (
        ('sma_period', 10),
        ('period_n', 100),
    )

    def __init__(self):
        self.sma = bt.indicators.SMA(self.data.close, period=self.p.sma_period)
        self.addminperiod(self.p.sma_period + 1)
        self.order = None
        self.current_seq = 0
        self.seq_history = []

    def notify_order(self, order):
        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            self.order = None

    def next(self):
        cond = (self.data.close[0] >= self.data.close[-1]) and (
            self.sma[0] >= self.sma[-1]
        )

        if cond:
            self.current_seq += 1
        else:
            self.current_seq = 0

        lookback = (
            self.seq_history[-self.p.period_n :] if self.seq_history else []
        )
        is_record = (self.current_seq > max(lookback)) if lookback else True

        self.seq_history.append(self.current_seq)

        if self.order is not None:
            return

        if not self.position:
            if self.current_seq >= 2 and is_record:
                self.order = self.buy()
        else:
            if not cond:
                self.order = self.close()
