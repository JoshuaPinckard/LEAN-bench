import backtrader as bt
from collections import deque


class ArcherChannelBreakout(bt.Strategy):
    params = dict(
        resistance=9.99,
        take_profit=13.92,
        support_primary=8.78,
        support_secondary=8.23,
        sma_period=50,
        vol_ma_period=20,
        vol_filter_pct=0.30,
        risk_pct=0.01,
    )

    def __init__(self):
        self.order = None

        self.weekly_closes = deque(maxlen=self.p.sma_period)
        self.weekly_volumes = deque(maxlen=self.p.vol_ma_period)

        self.cur_week_key = None
        self.week_close = None
        self.week_volume = 0.0

        self.prev_weekly_close = None
        self.prev_sma50 = None
        self.trend_blocked = False

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def _weekly_sma(self):
        if len(self.weekly_closes) < self.p.sma_period:
            return None
        return sum(self.weekly_closes) / len(self.weekly_closes)

    def _weekly_volma(self):
        if len(self.weekly_volumes) < self.p.vol_ma_period:
            return None
        return sum(self.weekly_volumes) / len(self.weekly_volumes)

    def _on_weekly_close(self, w_close, w_volume):
        sma50 = self._weekly_sma()
        volma20 = self._weekly_volma()

        if sma50 is not None and w_close < sma50:
            if self.position and not self.order:
                self.order = self.close()
            self.trend_blocked = True
        elif (sma50 is not None and self.prev_sma50 is not None
              and self.prev_weekly_close is not None
              and self.prev_weekly_close < self.prev_sma50 and w_close > sma50):
            self.trend_blocked = False

        if self.position and not self.order and w_close < self.p.support_primary:
            self.order = self.close()

        if not self.position and not self.order and not self.trend_blocked:
            if w_close > self.p.resistance and sma50 is not None and w_close > sma50:
                volume_ok = True
                if volma20 is not None and w_volume < volma20 * (1.0 - self.p.vol_filter_pct):
                    volume_ok = False
                if volume_ok:
                    stop_diff = w_close - self.p.support_primary
                    if stop_diff > 0:
                        risk_amount = self.broker.getvalue() * self.p.risk_pct
                        size = risk_amount / stop_diff
                        if size > 0:
                            self.order = self.buy(size=size)

        if sma50 is not None:
            self.prev_sma50 = sma50
        self.prev_weekly_close = w_close

    def next(self):
        dt = self.data.datetime.date(0)
        week_key = dt.isocalendar()[:2]

        if self.cur_week_key is None:
            self.cur_week_key = week_key
            self.week_close = self.data.close[0]
            self.week_volume = self.data.volume[0]
        elif week_key == self.cur_week_key:
            self.week_close = self.data.close[0]
            self.week_volume += self.data.volume[0]
        else:
            completed_close = self.week_close
            completed_volume = self.week_volume
            self.weekly_closes.append(completed_close)
            self.weekly_volumes.append(completed_volume)
            self._on_weekly_close(completed_close, completed_volume)

            self.cur_week_key = week_key
            self.week_close = self.data.close[0]
            self.week_volume = self.data.volume[0]

        if self.position and not self.order:
            if self.data.low[0] <= self.p.support_secondary:
                self.order = self.close()
            elif self.data.high[0] >= self.p.take_profit:
                self.order = self.close()
