import backtrader as bt


class IntradayMeanReversionStrategy(bt.Strategy):
    params = (
        ('bb_period', 20),
        ('bb_devfactor', 2),
        ('rsi_period', 14),
        ('atr_period', 14),
        ('stop_atr_multiple', 2.0),
    )

    def __init__(self):
        self.bbands = bt.indicators.BollingerBands(
            self.data.close,
            period=self.p.bb_period,
            devfactor=self.p.bb_devfactor,
        )
        self.rsi = bt.indicators.RSI(self.data.close, period=self.p.rsi_period)
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)

        self.order = None
        self.stop_order = None
        self.stop_distance = None
        self.exit_requested = False

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return

        if order == self.order:
            if order.status == order.Completed:
                if order.isbuy() and self.position.size > 0:
                    self.stop_distance = self.p.stop_atr_multiple * self.atr[0]
                    self.stop_order = self.sell(
                        exectype=bt.Order.Stop,
                        price=order.executed.price - self.stop_distance,
                    )
                elif order.issell() and self.position.size < 0:
                    self.stop_distance = self.p.stop_atr_multiple * self.atr[0]
                    self.stop_order = self.buy(
                        exectype=bt.Order.Stop,
                        price=order.executed.price + self.stop_distance,
                    )
            if order.status in (
                order.Completed,
                order.Canceled,
                order.Margin,
                order.Rejected,
            ):
                self.order = None

        if order == self.stop_order:
            if order.status in (
                order.Completed,
                order.Canceled,
                order.Margin,
                order.Rejected,
            ):
                self.stop_order = None
                self.stop_distance = None
                if self.exit_requested and self.position:
                    self.exit_requested = False
                    self.order = self.close()

    def next(self):
        if self.order:
            return

        close = self.data.close[0]
        upper = self.bbands.top[0]
        lower = self.bbands.bot[0]

        short_signal = close > upper and self.rsi[0] > 70
        long_signal = close < lower and self.rsi[0] < 30
        in_channel = lower <= close <= upper

        if not self.position:
            if short_signal:
                self.order = self.sell()
            elif long_signal:
                self.order = self.buy()
            return

        if self.position.size > 0:
            exit_signal = in_channel or short_signal
        else:
            exit_signal = in_channel or long_signal

        if exit_signal and not self.exit_requested:
            if self.stop_order:
                self.exit_requested = True
                self.cancel(self.stop_order)
            else:
                self.order = self.close()
