import backtrader as bt


class BearishWeeklyShortStrategyWithATRTarget(bt.Strategy):
    params = (
        ('risk_per_trade', 0.01),
        ('k', 2.0),
    )

    def __init__(self):
        self.atr = bt.indicators.ATR(self.data, period=14)
        self.pending_order = None
        self.entry_order = None
        self.take_profit_order = None
        self.stop_loss_order = None
        self.entry_bar = None
        self.entry_price = None
        self.force_exit_requested = False

    def next(self):
        if self.position:
            if (not self.force_exit_requested and
                    len(self) >= self.entry_bar + 20):
                if self.take_profit_order is not None:
                    self.cancel(self.take_profit_order)
                if self.stop_loss_order is not None:
                    self.cancel(self.stop_loss_order)
                self.pending_order = self.close()
                self.force_exit_requested = True
            return

        if self.pending_order is not None:
            return

        if len(self) % 7 == 0:
            risk_amount = self.broker.getvalue() * self.p.risk_per_trade
            size = risk_amount / (self.p.k * self.atr[0])
            self.entry_order = self.sell(size=size)
            self.pending_order = self.entry_order

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return

        if order.status == order.Completed:
            if order == self.entry_order:
                self.entry_price = order.executed.price
                self.entry_bar = len(self)
                size = abs(self.position.size)

                take_profit = self.entry_price - (0.5 * self.p.k * self.atr[0])
                stop_loss = self.entry_price + (self.p.k * self.atr[0])

                self.take_profit_order = self.buy(
                    size=size,
                    exectype=bt.Order.Limit,
                    price=take_profit,
                )
                self.stop_loss_order = self.buy(
                    size=size,
                    exectype=bt.Order.Stop,
                    price=stop_loss,
                    oco=self.take_profit_order,
                )

            elif order == self.take_profit_order or order == self.stop_loss_order:
                self.take_profit_order = None
                self.stop_loss_order = None
                self.entry_bar = None
                self.entry_price = None
                self.force_exit_requested = False

            elif self.force_exit_requested:
                self.take_profit_order = None
                self.stop_loss_order = None
                self.entry_bar = None
                self.entry_price = None
                self.force_exit_requested = False

        if order.status in (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            if order == self.pending_order:
                self.pending_order = None

            if order == self.entry_order and order.status != order.Completed:
                self.entry_order = None

            if order == self.take_profit_order and order.status != order.Completed:
                self.take_profit_order = None

            if order == self.stop_loss_order and order.status != order.Completed:
                self.stop_loss_order = None
