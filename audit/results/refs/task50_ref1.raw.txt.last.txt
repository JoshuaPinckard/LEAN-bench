import backtrader as bt
import math


class ATRIntradayRangeReversal(bt.Strategy):
    params = (
        ("atr_period", 14),
        ("range_fraction", 0.80),
        ("risk_fraction", 0.01),
        ("zone_atr_fraction", 0.10),
        ("stop_offset", 0.01),
    )

    def __init__(self):
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.order = None
        self.current_day = None
        self.hod = None
        self.lod = None
        self.range_activated = False
        self.trade_stop = None
        self.trade_target = None
        self.pending_stop = None
        self.pending_target = None

    def prenext(self):
        self._update_day_range()

    def next(self):
        self._update_day_range()

        if self.order:
            return

        dt = self.data.datetime.datetime(0)
        hour_minute = (dt.hour, dt.minute)

        if self.position:
            if self.position.size > 0:
                if self.data.low[0] <= self.trade_stop:
                    self.order = self.close()
                elif self.data.high[0] >= self.trade_target:
                    self.order = self.close()
            else:
                if self.data.high[0] >= self.trade_stop:
                    self.order = self.close()
                elif self.data.low[0] <= self.trade_target:
                    self.order = self.close()

            if hour_minute == (15, 55) and not self.order:
                self.order = self.close()
            return

        if hour_minute > (12, 30):
            return

        atr_value = self.atr[0]
        if atr_value <= 0:
            return

        if self.hod - self.lod >= self.p.range_fraction * atr_value:
            self.range_activated = True

        if not self.range_activated:
            return

        open_price = self.data.open[0]
        high_price = self.data.high[0]
        low_price = self.data.low[0]
        close_price = self.data.close[0]
        body = abs(close_price - open_price)
        lower_wick = min(open_price, close_price) - low_price
        upper_wick = high_price - max(open_price, close_price)
        zone_width = self.p.zone_atr_fraction * atr_value

        bullish_pin_bar = (
            close_price > open_price
            and lower_wick >= body
            and upper_wick <= body
        )
        bearish_pin_bar = (
            close_price < open_price
            and upper_wick >= body
            and lower_wick <= body
        )

        if close_price <= self.lod + zone_width and bullish_pin_bar:
            stop_price = self.lod - self.p.stop_offset
            risk_per_share = close_price - stop_price
            size = math.floor(
                (self.broker.getvalue() * self.p.risk_fraction) / risk_per_share
            )
            if size > 0:
                self.pending_stop = stop_price
                self.pending_target = self.hod
                self.order = self.buy(size=size)

        elif close_price >= self.hod - zone_width and bearish_pin_bar:
            stop_price = self.hod + self.p.stop_offset
            risk_per_share = stop_price - close_price
            size = math.floor(
                (self.broker.getvalue() * self.p.risk_fraction) / risk_per_share
            )
            if size > 0:
                self.pending_stop = stop_price
                self.pending_target = self.lod
                self.order = self.sell(size=size)

    def _update_day_range(self):
        bar_day = self.data.datetime.date(0)

        if bar_day != self.current_day:
            self.current_day = bar_day
            self.hod = self.data.high[0]
            self.lod = self.data.low[0]
            self.range_activated = False
        else:
            self.hod = max(self.hod, self.data.high[0])
            self.lod = min(self.lod, self.data.low[0])

    def notify_order(self, order):
        if order.status == order.Completed:
            if order == self.order and not self.position:
                self.trade_stop = None
                self.trade_target = None
            elif order.isbuy() or order.issell():
                if self.position:
                    self.trade_stop = self.pending_stop
                    self.trade_target = self.pending_target
            self.order = None

        elif order.status in (order.Canceled, order.Margin, order.Rejected):
            self.order = None
            self.pending_stop = None
            self.pending_target = None
