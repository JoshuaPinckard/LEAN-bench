import backtrader as bt


class SessionVWAP(bt.Indicator):
    lines = ('vwap',)

    def __init__(self):
        self.addminperiod(1)

    def next(self):
        d0 = self.data.datetime.date(0)
        if len(self) == 1 or d0 != self.data.datetime.date(-1):
            self.cum_tp_vol = 0.0
            self.cum_vol = 0.0

        tp = (self.data.high[0] + self.data.low[0] + self.data.close[0]) / 3.0
        vol = self.data.volume[0]

        self.cum_tp_vol += tp * vol
        self.cum_vol += vol

        if self.cum_vol > 0:
            self.lines.vwap[0] = self.cum_tp_vol / self.cum_vol
        else:
            self.lines.vwap[0] = self.data.close[0]


class IntradayMeanReversion(bt.Strategy):
    def __init__(self):
        self.vwap = SessionVWAP(self.data)
        self.williams = bt.indicators.WilliamsR(self.data, period=14)

        self.entry_order = None
        self.exit_order = None
        self.stop_order = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            if order == self.entry_order:
                self.entry_order = None
                exec_price = order.executed.price
                entry_vwap = self.vwap[0]
                if order.isbuy():
                    stop_price = max(exec_price - 2.0, entry_vwap - 2.0)
                    self.stop_order = self.sell(
                        exectype=bt.Order.Stop,
                        price=stop_price
                    )
                elif order.issell():
                    stop_price = min(exec_price + 2.0, entry_vwap + 2.0)
                    self.stop_order = self.buy(
                        exectype=bt.Order.Stop,
                        price=stop_price
                    )
            elif order == self.stop_order:
                self.stop_order = None
                if self.exit_order:
                    self.cancel(self.exit_order)
                    self.exit_order = None
            elif order == self.exit_order:
                self.exit_order = None
                if self.stop_order:
                    self.cancel(self.stop_order)
                    self.stop_order = None

        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            if order == self.entry_order:
                self.entry_order = None
            if order == self.stop_order:
                self.stop_order = None
            if order == self.exit_order:
                self.exit_order = None

    def next(self):
        if self.entry_order or self.exit_order:
            return

        if self.position:
            if self.position.size > 0:
                if self.data.close[-1] <= self.vwap[-1] and self.data.close[0] > self.vwap[0]:
                    if self.stop_order:
                        self.cancel(self.stop_order)
                        self.stop_order = None
                    self.exit_order = self.close()
            elif self.position.size < 0:
                if self.data.close[-1] >= self.vwap[-1] and self.data.close[0] < self.vwap[0]:
                    if self.stop_order:
                        self.cancel(self.stop_order)
                        self.stop_order = None
                    self.exit_order = self.close()
        else:
            long_cond = (
                self.data.close[0] < self.vwap[0] and
                self.williams[-1] >= -80.0 and
                self.williams[0] < -80.0
            )
            short_cond = (
                self.data.close[0] > self.vwap[0] and
                self.williams[-1] <= -20.0 and
                self.williams[0] > -20.0
            )

            if long_cond:
                self.entry_order = self.buy()
            elif short_cond:
                self.entry_order = self.sell()
