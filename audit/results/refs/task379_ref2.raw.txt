import backtrader as bt


class SlopeReversalStrategy(bt.Strategy):
    params = (
        ('sma_period', 20),
        ('slope_lag', 20),
        ('std_lookback', 100),
        ('extreme_mult', 2.0),
        ('stop_pct', 0.02),
        ('neutral_mult', 0.2),
    )

    def __init__(self):
        self.sma = bt.ind.SMA(self.data.close, period=self.p.sma_period)
        self.slope = (self.sma - self.sma(-self.p.slope_lag)) / self.p.slope_lag
        self.slope_std = bt.ind.StdDev(self.slope, period=self.p.std_lookback)
        self.pos_th = self.p.extreme_mult * self.slope_std
        self.neg_th = -self.p.extreme_mult * self.slope_std
        self.neutral_band = self.p.neutral_mult * self.slope_std

        self.order = None
        self.entry_price = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            if self.position:
                self.entry_price = self.position.price
            else:
                self.entry_price = None

        self.order = None

    def next(self):
        if self.order:
            return

        if not self.position:
            long_condition = (self.slope[-1] <= self.neg_th[-1]) and (self.slope[0] > self.neg_th[0])
            short_condition = (self.slope[-1] >= self.pos_th[-1]) and (self.slope[0] < self.pos_th[0])

            if long_condition:
                self.order = self.buy()
            elif short_condition:
                self.order = self.sell()
        else:
            entry = self.entry_price if self.entry_price is not None else self.position.price
            if self.position.size > 0:
                stop_price = entry * (1.0 - self.p.stop_pct)
                if self.data.low[0] <= stop_price:
                    self.order = self.close()
                    return
            elif self.position.size < 0:
                stop_price = entry * (1.0 + self.p.stop_pct)
                if self.data.high[0] >= stop_price:
                    self.order = self.close()
                    return

            cross_zero = (self.slope[-1] <= 0 and self.slope[0] >= 0) or (self.slope[-1] >= 0 and self.slope[0] <= 0)
            below_neutral = abs(self.slope[0]) < self.neutral_band[0]

            if cross_zero or below_neutral:
                self.order = self.close()
