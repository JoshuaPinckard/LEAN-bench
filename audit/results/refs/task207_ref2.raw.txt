import backtrader as bt


class BbBreakoutStrategy(bt.Strategy):
    params = (
        ("period", 10),
        ("devfactor", 0.674),
    )

    def __init__(self):
        self.order = None
        self.boll = bt.indicators.BollingerBands(
            self.data.close,
            period=self.p.period,
            devfactor=self.p.devfactor,
        )
        self.crossover = bt.indicators.CrossOver(
            self.data.close, self.boll.top
        )

    def notify_order(self, order):
        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            self.order = None

    def next(self):
        if self.order:
            return

        if not self.position:
            if self.crossover[0] > 0:
                cash = self.broker.getcash()
                price = self.data.close[0]
                comm_rate = 0.001
                size = int(cash / (price * (1.0 + comm_rate)))
                if size > 0:
                    self.order = self.buy(size=size)
        else:
            if self.crossover[0] < 0 or self.data.close[0] < self.boll.top[0]:
                self.order = self.close()
