import backtrader as bt


class TradingStrategy(bt.Strategy):
    params = (("sma_period", 50), ("lot_size", 100), ("transaction_cost", 0.002))

    def __init__(self):
        self.sma = bt.indicators.SimpleMovingAverage(
            self.data.close, period=self.p.sma_period
        )
        self.order = None
        self.broker.setcommission(commission=self.p.transaction_cost)
        self.broker.set_coc(True)

    def next(self):
        current_date = self.data.datetime.date(0)

        if current_date.year < 2000 or current_date.year > 2012:
            return

        if self.order:
            return

        if not self.position and self.data.close[0] > self.sma[0]:
            self.order = self.buy(size=self.p.lot_size)
        elif self.position and self.data.close[0] < self.sma[0]:
            self.order = self.close()

    def notify_order(self, order):
        if order.status in (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            self.order = None
