import backtrader as bt


class BollingerBandwidthReversalStrategy(bt.Strategy):
    def __init__(self):
        self.bb = bt.indicators.BollingerBands(self.data.close, period=20, devfactor=2)
        self.bandwidth = (self.bb.top - self.bb.bot) / self.bb.mid
        self.bw_sma = bt.indicators.SMA(self.bandwidth, period=20)
        self.bw_std = bt.indicators.StandardDeviation(self.bandwidth, period=20)
        self.bwz = (self.bandwidth - self.bw_sma) / self.bw_std
        self.atr = bt.indicators.ATR(self.data, period=14)

        self.order = None
        self.order_intent = None
        self.stop_price = None
        self.pending_atr = None

    def notify_order(self, order):
        if order.status == order.Completed:
            if self.order_intent == 'entry':
                entry_price = order.executed.price
                if order.isbuy():
                    self.stop_price = entry_price - 2 * self.pending_atr
                else:
                    self.stop_price = entry_price + 2 * self.pending_atr
            self.order = None
            self.order_intent = None
        elif order.status in (order.Canceled, order.Margin, order.Rejected):
            self.order = None
            self.order_intent = None

    def next(self):
        if self.order:
            return

        bwz0 = self.bwz[0]
        bwz1 = self.bwz[-1]
        bw0 = self.bandwidth[0]
        bw1 = self.bandwidth[-1]

        if self.position:
            cross_zero = (bwz1 > 0 and bwz0 <= 0) or (bwz1 < 0 and bwz0 >= 0)
            mean_revert = abs(bwz0) < 0.5

            stop_hit = False
            if self.position.size > 0 and self.data.close[0] <= self.stop_price:
                stop_hit = True
            elif self.position.size < 0 and self.data.close[0] >= self.stop_price:
                stop_hit = True

            if cross_zero or mean_revert or stop_hit:
                self.order = self.close()
                self.order_intent = 'exit'
            return

        if bwz1 > 2 and bwz0 <= 2 and bw0 < bw1:
            self.pending_atr = self.atr[0]
            self.order = self.buy()
            self.order_intent = 'entry'
        elif bwz1 < -2 and bwz0 >= -2 and bw0 > bw1:
            self.pending_atr = self.atr[0]
            self.order = self.sell()
            self.order_intent = 'entry'
