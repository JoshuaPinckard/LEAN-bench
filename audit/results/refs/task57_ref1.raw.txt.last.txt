import backtrader as bt
import math
from collections import deque


class GapRiskStrategy(bt.Strategy):
    params = (
        ("atr_period", 14),
        ("sma_period", 20),
    )

    def __init__(self):
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.sma = bt.indicators.SMA(self.data.close, period=self.p.sma_period)

        self.order = None
        self._entry_order = False
        self._atr_history = deque(maxlen=20)

        self.entry_price = None
        self.entry_atr = None
        self.stop_price = None
        self.trailing_active = False
        self.best_price = None

    def notify_order(self, order):
        if order.status in (bt.Order.Completed, bt.Order.Canceled,
                            bt.Order.Margin, bt.Order.Rejected):
            if order.status == bt.Order.Completed and self._entry_order:
                self.entry_price = order.executed.price
                self.entry_atr = float(self.atr[0])
                self.trailing_active = False
                self.best_price = self.entry_price

                if order.isbuy():
                    self.stop_price = self.entry_price - 1.5 * self.entry_atr
                else:
                    self.stop_price = self.entry_price + 1.5 * self.entry_atr

            self.order = None
            self._entry_order = False

    def _volatility_allowed(self):
        if len(self._atr_history) < 20:
            return False

        prior_five_average = sum(list(self._atr_history)[-5:]) / 5.0
        prior_twenty_average = sum(self._atr_history) / 20.0
        return prior_five_average <= 1.5 * prior_twenty_average

    def _submit_entry(self, is_long):
        atr_value = float(self.atr[0])
        if atr_value <= 0.0:
            return

        capital = self.broker.getvalue()
        risk_per_share = 1.5 * atr_value
        size = math.floor((capital * 0.015) / risk_per_share)

        if size < 1:
            return

        self._entry_order = True
        self.order = self.buy(size=size) if is_long else self.sell(size=size)

    def _manage_stops(self):
        if not self.position or self.order is not None:
            return

        atr_value = float(self.atr[0])

        if self.position.size > 0:
            self.best_price = max(self.best_price, float(self.data.high[0]))

            if (not self.trailing_active and
                    self.best_price >= self.entry_price + atr_value):
                self.trailing_active = True

            if self.trailing_active:
                self.stop_price = max(
                    self.stop_price,
                    self.best_price - atr_value
                )

            if float(self.data.low[0]) <= self.stop_price:
                self.order = self.close()
                return

        else:
            self.best_price = min(self.best_price, float(self.data.low[0]))

            if (not self.trailing_active and
                    self.best_price <= self.entry_price - atr_value):
                self.trailing_active = True

            if self.trailing_active:
                self.stop_price = min(
                    self.stop_price,
                    self.best_price + atr_value
                )

            if float(self.data.high[0]) >= self.stop_price:
                self.order = self.close()

    def next(self):
        self._manage_stops()

        if self.position and self.order is None:
            self.order = self.close()

        if not self.position and self.order is None and self._volatility_allowed():
            close_now = float(self.data.close[0])
            close_previous = float(self.data.close[-1])
            open_now = float(self.data.open[0])

            crossed_above_sma = (
                close_now > float(self.sma[0]) and
                close_previous <= float(self.sma[-1])
            )
            crossed_below_sma = (
                close_now < float(self.sma[0]) and
                close_previous >= float(self.sma[-1])
            )

            long_gap_prohibited = open_now <= close_previous * 0.98
            short_gap_prohibited = open_now >= close_previous * 1.02

            if crossed_above_sma and not long_gap_prohibited:
                self._submit_entry(is_long=True)
            elif crossed_below_sma and not short_gap_prohibited:
                self._submit_entry(is_long=False)

        self._atr_history.append(float(self.atr[0]))
