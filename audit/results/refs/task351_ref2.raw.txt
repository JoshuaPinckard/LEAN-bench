import backtrader as bt


class VolatilityStochasticStrategy(bt.Strategy):
    def __init__(self):
        # 20-bar Standard Deviation of close price
        self.vol = bt.indicators.StdDev(self.data.close, period=20)

        # Attach high/low/close attributes to vol line to pass to StochasticGeneric
        self.vol.high = self.vol
        self.vol.low = self.vol
        self.vol.close = self.vol

        # StochasticGeneric on volatility line
        self.stoch = bt.indicators.StochasticGeneric(
            self.vol,
            period=14,
            period_dfast=3,
            period_dslow=3
        )

        # 14-period ATR for stop-loss calculation
        self.atr = bt.indicators.ATR(self.data, period=14)

        # Order tracking attributes
        self.order = None
        self.stop_order = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            if order == self.order:
                self.order = None
                # Place ATR stop loss upon entry execution
                if self.position.size > 0:
                    stop_price = order.executed.price - 2.0 * self.atr[0]
                    self.stop_order = self.sell(
                        exectype=bt.Order.Stop,
                        price=stop_price
                    )
                elif self.position.size < 0:
                    stop_price = order.executed.price + 2.0 * self.atr[0]
                    self.stop_order = self.buy(
                        exectype=bt.Order.Stop,
                        price=stop_price
                    )
            elif order == self.stop_order:
                self.stop_order = None

        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            if order == self.order:
                self.order = None
            if order == self.stop_order:
                self.stop_order = None

    def next(self):
        if self.order:
            return

        k0 = self.stoch.percK[0]
        k1 = self.stoch.percK[-1]
        d0 = self.stoch.percD[0]
        d1 = self.stoch.percD[-1]

        v0 = self.vol[0]
        v1 = self.vol[-1]

        if not self.position:
            # Long entry condition
            long_cond = (k0 > d0) and (k1 <= d1) and (k1 < 20) and (v0 > v1)
            # Short entry condition
            short_cond = (k0 < d0) and (k1 >= d1) and (k1 > 80) and (v0 < v1)

            if long_cond:
                self.order = self.buy()
            elif short_cond:
                self.order = self.sell()

        else:
            if self.position.size > 0:
                # Exit long on opposite stochastic crossover (%K crossing below %D)
                if (k0 < d0) and (k1 >= d1):
                    if self.stop_order:
                        self.cancel(self.stop_order)
                        self.stop_order = None
                    self.order = self.close()

            elif self.position.size < 0:
                # Exit short on opposite stochastic crossover (%K crossing above %D)
                if (k0 > d0) and (k1 <= d1):
                    if self.stop_order:
                        self.cancel(self.stop_order)
                        self.stop_order = None
                    self.order = self.close()
