import backtrader as bt


class TimeSeriesMomentum(bt.Strategy):
    params = (
        ('period', 252),
    )

    def __init__(self):
        self.roc = bt.indicators.RateOfChange(
            self.data.close,
            period=self.p.period,
        )
        self.order = None

    def notify_order(self, order):
        if order.status in (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            self.order = None

    def next(self):
        if self.order:
            return

        momentum = self.roc[0]

        if not self.position:
            if momentum > 0:
                self.order = self.order_target_percent(target=1.0)
            elif momentum < 0:
                self.order = self.order_target_percent(target=-1.0)
        elif self.position.size > 0 and momentum < 0:
            self.order = self.close()
        elif self.position.size < 0 and momentum > 0:
            self.order = self.close()
