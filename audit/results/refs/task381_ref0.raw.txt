import backtrader as bt


class EmaBreakoutStrategy(bt.Strategy):
    def __init__(self):
        self.ema = bt.indicators.EMA(self.data.close, period=5)

        self.entry_order = None
        self.entry_direction = None
        self.setup_high = None
        self.setup_low = None

        self.stop_order = None
        self.target_order = None

    def next(self):
        if self.position:
            return

        high = self.data.high[0]
        low = self.data.low[0]
        ema = self.ema[0]

        if self.entry_order is not None:
            if self.entry_direction == 'long' and low > ema:
                self.cancel(self.entry_order)
                self._enter_short(high, low)
            elif self.entry_direction == 'short' and high < ema:
                self.cancel(self.entry_order)
                self._enter_long(high, low)
            return

        if high < ema:
            self._enter_long(high, low)
        elif low > ema:
            self._enter_short(high, low)

    def _enter_long(self, setup_high, setup_low):
        stop_distance = setup_high - setup_low
        if stop_distance <= 0:
            return
        size = (0.05 * self.broker.getvalue()) / stop_distance
        self.entry_order = self.buy(exectype=bt.Order.Stop, price=setup_high, size=size)
        self.entry_direction = 'long'
        self.setup_high = setup_high
        self.setup_low = setup_low

    def _enter_short(self, setup_high, setup_low):
        stop_distance = setup_high - setup_low
        if stop_distance <= 0:
            return
        size = (0.05 * self.broker.getvalue()) / stop_distance
        self.entry_order = self.sell(exectype=bt.Order.Stop, price=setup_low, size=size)
        self.entry_direction = 'short'
        self.setup_high = setup_high
        self.setup_low = setup_low

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return

        if order.status == order.Completed:
            if order is self.entry_order:
                entry_price = order.executed.price
                size = abs(order.executed.size)
                if order.isbuy():
                    stop_price = self.setup_low
                    target_price = entry_price + 3 * (entry_price - stop_price)
                    self.stop_order = self.sell(exectype=bt.Order.Stop, price=stop_price, size=size)
                    self.target_order = self.sell(exectype=bt.Order.Limit, price=target_price, size=size)
                else:
                    stop_price = self.setup_high
                    target_price = entry_price - 3 * (stop_price - entry_price)
                    self.stop_order = self.buy(exectype=bt.Order.Stop, price=stop_price, size=size)
                    self.target_order = self.buy(exectype=bt.Order.Limit, price=target_price, size=size)
                self.entry_order = None
                self.entry_direction = None
            elif order is self.stop_order:
                if self.target_order is not None:
                    self.cancel(self.target_order)
                self.stop_order = None
                self.target_order = None
                self.setup_high = None
                self.setup_low = None
            elif order is self.target_order:
                if self.stop_order is not None:
                    self.cancel(self.stop_order)
                self.stop_order = None
                self.target_order = None
                self.setup_high = None
                self.setup_low = None
        elif order.status in (order.Canceled, order.Margin, order.Rejected):
            if order is self.entry_order:
                self.entry_order = None
                self.entry_direction = None
            elif order is self.stop_order:
                self.stop_order = None
            elif order is self.target_order:
                self.target_order = None
