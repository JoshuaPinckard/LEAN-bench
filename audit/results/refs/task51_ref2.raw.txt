import math
import backtrader as bt


class TradingSessionVisualizer(bt.Strategy):
    """
    Reference implementation of a trading strategy specification that visualizes
    trading sessions (Monday to Friday) as expanding rectangles on the chart.
    """
    params = (
        ('color', 'gray'),
        ('alpha', 0.2),
    )

    @staticmethod
    def _get_rgba(color, alpha):
        if isinstance(color, (tuple, list)):
            if len(color) == 3:
                return (color[0], color[1], color[2], alpha)
            return tuple(color)
        if isinstance(color, str):
            c = color.lower().strip()
            named_colors = {
                'gray': (0.5, 0.5, 0.5),
                'grey': (0.5, 0.5, 0.5),
                'red': (1.0, 0.0, 0.0),
                'green': (0.0, 1.0, 0.0),
                'blue': (0.0, 0.0, 1.0),
                'yellow': (1.0, 1.0, 0.0),
                'cyan': (0.0, 1.0, 1.0),
                'magenta': (1.0, 0.0, 1.0),
                'black': (0.0, 0.0, 0.0),
                'white': (1.0, 1.0, 1.0),
            }
            if c in named_colors:
                r, g, b = named_colors[c]
                return (r, g, b, alpha)
            if c.startswith('#'):
                hex_str = c.lstrip('#')
                if len(hex_str) == 6:
                    r = int(hex_str[0:2], 16) / 255.0
                    g = int(hex_str[2:4], 16) / 255.0
                    b = int(hex_str[4:6], 16) / 255.0
                    return (r, g, b, alpha)
        return color

    class _WeeklyRect(bt.Indicator):
        lines = ('rect_top', 'rect_bottom')
        params = (
            ('color', 'gray'),
            ('alpha', 0.2),
        )
        plotinfo = dict(subplot=False)

        def __init__(self):
            rgba = TradingSessionVisualizer._get_rgba(self.p.color, self.p.alpha)
            self.plotlines = dict(
                rect_top=dict(
                    _fill_gt=('rect_bottom', rgba),
                    color=self.p.color,
                    ls='--',
                    linewidth=1,
                ),
                rect_bottom=dict(
                    color=self.p.color,
                    ls='--',
                    linewidth=1,
                ),
            )

        def next(self):
            dt = self.data.datetime.date(0)
            weekday = dt.weekday()

            if weekday > 4:
                self.lines.rect_top[0] = float('nan')
                self.lines.rect_bottom[0] = float('nan')
                return

            current_week = dt.isocalendar()[:2]

            if len(self) == 1:
                self.lines.rect_top[0] = self.data.high[0]
                self.lines.rect_bottom[0] = self.data.low[0]
            else:
                prev_dt = self.data.datetime.date(-1)
                prev_weekday = prev_dt.weekday()
                prev_week = prev_dt.isocalendar()[:2]

                if prev_weekday > 4 or current_week != prev_week:
                    self.lines.rect_top[0] = self.data.high[0]
                    self.lines.rect_bottom[0] = self.data.low[0]
                else:
                    prev_top = self.lines.rect_top[-1]
                    prev_bottom = self.lines.rect_bottom[-1]
                    if math.isnan(prev_top) or math.isnan(prev_bottom):
                        self.lines.rect_top[0] = self.data.high[0]
                        self.lines.rect_bottom[0] = self.data.low[0]
                    else:
                        self.lines.rect_top[0] = max(prev_top, self.data.high[0])
                        self.lines.rect_bottom[0] = min(prev_bottom, self.data.low[0])

    def __init__(self):
        self.order = None
        self.rect = self._WeeklyRect(
            self.data, color=self.p.color, alpha=self.p.alpha
        )

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        pass
