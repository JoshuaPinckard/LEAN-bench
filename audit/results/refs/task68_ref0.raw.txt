import backtrader as bt


class AITradingStrategy(bt.Strategy):
    params = dict(
        sma_fast_period=10,
        sma_slow_period=30,
        rsi_period=14,
        atr_period=14,
        atr_multiplier=2.0,
        risk_reward_ratio=2.0,
        risk_per_trade=0.01,
        max_daily_risk=0.02,
        max_drawdown=0.05,
    )

    def __init__(self):
        self.sma_fast = bt.indicators.SMA(self.data.close, period=self.p.sma_fast_period)
        self.sma_slow = bt.indicators.SMA(self.data.close, period=self.p.sma_slow_period)
        self.rsi = bt.indicators.RSI(self.data.close, period=self.p.rsi_period)
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)

        self.order = None
        self.direction = None
        self.stop_price = None
        self.tp_price = None

        self.current_day = None
        self.day_start_equity = None
        self.day_peak_equity = None
        self.daily_pnl = 0.0
        self.daily_risk_halt = False
        self.drawdown_halt = False

        self.trade_log = []

    def log(self, txt):
        dt = self.data.datetime.datetime(0)
        print('%s %s' % (dt.isoformat(), txt))

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return
        if order.status in (order.Completed, order.Canceled, order.Margin, order.Rejected):
            self.order = None

    def notify_trade(self, trade):
        if trade.isclosed:
            self.daily_pnl += trade.pnlcomm

    def get_signal(self):
        if self.sma_fast[0] > self.sma_slow[0] and self.rsi[0] < 70:
            return 'BUY'
        elif self.sma_fast[0] < self.sma_slow[0] and self.rsi[0] > 30:
            return 'SELL'
        return 'HOLD'

    def next(self):
        current_date = self.data.datetime.date(0)
        current_equity = self.broker.getvalue()

        if self.current_day != current_date:
            self.current_day = current_date
            self.day_start_equity = current_equity
            self.day_peak_equity = current_equity
            self.daily_pnl = 0.0
            self.daily_risk_halt = False
            self.drawdown_halt = False

        if current_equity > self.day_peak_equity:
            self.day_peak_equity = current_equity

        if self.day_peak_equity > 0:
            drawdown = (self.day_peak_equity - current_equity) / self.day_peak_equity
            if drawdown >= self.p.max_drawdown:
                self.drawdown_halt = True

        if self.day_start_equity > 0 and -self.daily_pnl >= self.p.max_daily_risk * self.day_start_equity:
            self.daily_risk_halt = True

        if self.order:
            return

        signal = self.get_signal()

        if self.position:
            price = self.data.close[0]
            if self.direction == 'long':
                if self.data.low[0] <= self.stop_price:
                    self.order = self.close()
                    self.trade_log.append(dict(decision=signal, action='SL_EXIT', price=self.stop_price))
                elif self.data.high[0] >= self.tp_price:
                    self.order = self.close()
                    self.trade_log.append(dict(decision=signal, action='TP_EXIT', price=self.tp_price))
                elif signal == 'SELL':
                    self.order = self.close()
                    self.trade_log.append(dict(decision=signal, action='OPPOSITE_EXIT', price=price))
                elif signal == 'BUY':
                    new_stop = price - self.p.atr_multiplier * self.atr[0]
                    if new_stop > self.stop_price:
                        self.stop_price = new_stop
                        self.trade_log.append(dict(decision=signal, action='TRAIL_SL', price=self.stop_price))
            else:
                if self.data.high[0] >= self.stop_price:
                    self.order = self.close()
                    self.trade_log.append(dict(decision=signal, action='SL_EXIT', price=self.stop_price))
                elif self.data.low[0] <= self.tp_price:
                    self.order = self.close()
                    self.trade_log.append(dict(decision=signal, action='TP_EXIT', price=self.tp_price))
                elif signal == 'BUY':
                    self.order = self.close()
                    self.trade_log.append(dict(decision=signal, action='OPPOSITE_EXIT', price=price))
                elif signal == 'SELL':
                    new_stop = price + self.p.atr_multiplier * self.atr[0]
                    if new_stop < self.stop_price:
                        self.stop_price = new_stop
                        self.trade_log.append(dict(decision=signal, action='TRAIL_SL', price=self.stop_price))
            return

        if self.daily_risk_halt or self.drawdown_halt:
            self.trade_log.append(dict(decision=signal, action='HALTED'))
            return

        if signal == 'HOLD':
            self.trade_log.append(dict(decision=signal, action='NONE'))
            return

        atr_val = self.atr[0]
        if atr_val <= 0:
            return

        price = self.data.close[0]
        stop_distance = self.p.atr_multiplier * atr_val
        risk_amount = current_equity * self.p.risk_per_trade
        size = int(risk_amount / stop_distance)
        if size < 1:
            return

        expected_risk = stop_distance * size
        self.log('signal=%s size=%d risk_points=%.4f expected_risk=%.2f' % (signal, size, stop_distance, expected_risk))

        if signal == 'BUY':
            self.direction = 'long'
            self.stop_price = price - stop_distance
            self.tp_price = price + stop_distance * self.p.risk_reward_ratio
            self.order = self.buy(size=size)
        else:
            self.direction = 'short'
            self.stop_price = price + stop_distance
            self.tp_price = price - stop_distance * self.p.risk_reward_ratio
            self.order = self.sell(size=size)

        self.trade_log.append(dict(decision=signal, action='ENTRY', price=price, size=size,
                                    sl=self.stop_price, tp=self.tp_price, risk=expected_risk))
