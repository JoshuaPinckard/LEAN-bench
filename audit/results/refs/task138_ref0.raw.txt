import backtrader as bt


class RSISessionStrategy(bt.Strategy):
    params = dict(
        rsi_period=14,
        oversold=30,
        overbought=70,
    )

    def __init__(self):
        self.rsi = bt.indicators.RSI(self.data.close, period=self.p.rsi_period)
        self.cross_30 = bt.indicators.CrossOver(self.rsi, self.p.oversold)
        self.cross_70 = bt.indicators.CrossOver(self.rsi, self.p.overbought)
        self.order = None
        self.armed_long = False
        self.armed_short = False

    def notify_order(self, order):
        if order.status in (order.Completed, order.Canceled, order.Margin, order.Rejected):
            self.order = None

    def _in_session(self):
        dt = self.data.datetime.datetime(0)
        if dt.weekday() >= 5:
            return False
        hm = (dt.hour, dt.minute)
        return (9, 30) <= hm <= (16, 0)

    def _target_size(self, price):
        cash = self.broker.getcash()
        comminfo = self.broker.getcommissioninfo(self.data)
        commission = comminfo.p.commission
        return int(cash / (price * (1 + commission)))

    def next(self):
        if not self._in_session():
            return
        if self.order:
            return

        cross_down_30 = self.cross_30[0] == -1
        cross_up_30 = self.cross_30[0] == 1
        cross_up_70 = self.cross_70[0] == 1
        cross_down_70 = self.cross_70[0] == -1

        pos = self.position.size

        if cross_down_30:
            self.armed_long = True
            if pos < 0:
                self.order = self.close()
                return

        if cross_up_70:
            self.armed_short = True
            if pos > 0:
                self.order = self.close()
                return

        if pos == 0:
            if cross_up_30 and self.armed_long:
                size = self._target_size(self.data.close[0])
                if size > 0:
                    self.order = self.buy(size=size)
                self.armed_long = False
            elif cross_down_70 and self.armed_short:
                size = self._target_size(self.data.close[0])
                if size > 0:
                    self.order = self.sell(size=size)
                self.armed_short = False
