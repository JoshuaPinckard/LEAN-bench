import backtrader as bt


class MacdAdaptiveBandsStrategy(bt.Strategy):

    def __init__(self):
        self.macd = bt.indicators.MACD(
            self.data.close,
            period_me1=12,
            period_me2=26,
            period_signal=9,
        )
        self.hist = self.macd.macd - self.macd.signal
        self.hist_sma = bt.indicators.SMA(self.hist, period=20)
        self.hist_std = bt.indicators.StdDev(self.hist, period=20)
        self.upper_band = self.hist_sma + 2 * self.hist_std
        self.lower_band = self.hist_sma - 2 * self.hist_std
        self.atr = bt.indicators.ATR(self.data, period=14)

        self.macd_cross = bt.indicators.CrossOver(self.macd.macd, self.macd.signal)
        self.hist_cross_upper = bt.indicators.CrossOver(self.hist, self.upper_band)
        self.hist_cross_lower = bt.indicators.CrossOver(self.hist, self.lower_band)

        self.order = None
        self.stop_order = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status == order.Completed:
            if order is self.stop_order:
                self.stop_order = None
                self.order = None
                return

            if self.position:
                entry_price = order.executed.price
                atr_val = self.atr[0]
                if order.isbuy():
                    stop_price = max(entry_price * 0.98, entry_price - 2 * atr_val)
                    self.stop_order = self.sell(
                        exectype=bt.Order.Stop,
                        price=stop_price,
                        size=order.executed.size,
                    )
                else:
                    stop_price = min(entry_price * 1.02, entry_price + 2 * atr_val)
                    self.stop_order = self.buy(
                        exectype=bt.Order.Stop,
                        price=stop_price,
                        size=abs(order.executed.size),
                    )
            else:
                if self.stop_order is not None:
                    self.cancel(self.stop_order)
                    self.stop_order = None
            self.order = None

        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            if order is self.stop_order:
                self.stop_order = None
            else:
                self.order = None

    def next(self):
        if self.order:
            return

        if not self.position:
            if self.macd_cross > 0 and self.hist_cross_upper > 0:
                self.order = self.buy()
            elif self.macd_cross < 0 and self.hist_cross_lower < 0:
                self.order = self.sell()
        else:
            if self.position.size > 0:
                if self.macd_cross < 0 and self.hist_cross_lower < 0:
                    if self.stop_order is not None:
                        self.cancel(self.stop_order)
                    self.order = self.close()
            elif self.position.size < 0:
                if self.macd_cross > 0 and self.hist_cross_upper > 0:
                    if self.stop_order is not None:
                        self.cancel(self.stop_order)
                    self.order = self.close()
