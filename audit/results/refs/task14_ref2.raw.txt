import backtrader as bt


class EmaCrossover(bt.Strategy):
    params = (
        ('FAST', 21),
        ('SLOW', 50),
        ('UNITS', 10000),
    )

    def log(self, txt, dt=None):
        dt = dt or self.data.datetime.datetime(0)
        print(f'{dt.isoformat()} - {txt}')

    def __init__(self):
        self.fast_ema = bt.ind.EMA(self.data.close, period=self.p.FAST)
        self.slow_ema = bt.ind.EMA(self.data.close, period=self.p.SLOW)
        self.crossover = bt.ind.CrossOver(self.fast_ema, self.slow_ema)
        self.order = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            if order.isbuy():
                self.log(
                    f'BUY EXECUTED - Price: {order.executed.price:.2f}, '
                    f'Cost: {order.executed.value:.2f}, Comm: {order.executed.comm:.2f}'
                )
            else:
                self.log(
                    f'SELL EXECUTED - Price: {order.executed.price:.2f}, '
                    f'Cost: {order.executed.value:.2f}, Comm: {order.executed.comm:.2f}'
                )
        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            self.log(f'ORDER FAILED - Status: {order.getstatusname()}')

        self.order = None

    def notify_trade(self, trade):
        if trade.isclosed:
            dt = self.data.datetime.datetime(0)
            self.log(
                f'TRADE CLOSED - Entry Price: {trade.price:.2f}, '
                f'PnL Gross: {trade.pnl:.2f}, PnL Net: {trade.pnlcomm:.2f}',
                dt=dt,
            )

    def next(self):
        if self.order:
            return

        if self.crossover > 0:
            if self.position.size < 0:
                size = abs(self.position.size) + self.p.UNITS
                self.order = self.buy(size=size)
            elif self.position.size == 0:
                self.order = self.buy(size=self.p.UNITS)

        elif self.crossover < 0:
            if self.position.size > 0:
                size = self.position.size + self.p.UNITS
                self.order = self.sell(size=size)
            elif self.position.size == 0:
                self.order = self.sell(size=self.p.UNITS)
