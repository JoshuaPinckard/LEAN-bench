import backtrader as bt


class MACDBollingerATRStrategy(bt.Strategy):
    params = (
        ('fast', 12),
        ('slow', 26),
        ('signal', 9),
        ('bb_period', 20),
        ('bb_dev', 2),
        ('atr_period', 14),
        ('atr_multiplier', 2),
    )

    def __init__(self):
        self.macd = bt.indicators.MACD(
            self.data.close,
            period_me1=self.p.fast,
            period_me2=self.p.slow,
            period_signal=self.p.signal,
        )
        self.macd_cross = bt.indicators.CrossOver(self.macd.macd, self.macd.signal)
        self.bbands = bt.indicators.BollingerBands(
            self.data.close,
            period=self.p.bb_period,
            devfactor=self.p.bb_dev,
        )
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)

        self.order = None
        self.stop_order = None

    def next(self):
        if self.order:
            return

        if not self.position:
            if self.macd_cross[0] > 0 and self.data.close[0] <= self.bbands.bot[0]:
                self.order = self.buy()
            elif self.macd_cross[0] < 0 and self.data.close[0] >= self.bbands.top[0]:
                self.order = self.sell()

        elif self.position.size > 0 and self.data.close[0] >= self.bbands.mid[0]:
            self.order = self.close()

        elif self.position.size < 0 and self.data.close[0] <= self.bbands.mid[0]:
            self.order = self.close()

    def notify_order(self, order):
        if order.status in (order.Completed, order.Canceled, order.Margin, order.Rejected):
            if order is self.order:
                self.order = None

            if order is self.stop_order:
                self.stop_order = None

            if order.status == order.Completed:
                if order.isbuy() and self.position.size > 0 and self.stop_order is None:
                    self.stop_order = self.sell(
                        exectype=bt.Order.Stop,
                        price=order.executed.price - self.p.atr_multiplier * self.atr[0],
                    )
                elif order.issell() and self.position.size < 0 and self.stop_order is None:
                    self.stop_order = self.buy(
                        exectype=bt.Order.Stop,
                        price=order.executed.price + self.p.atr_multiplier * self.atr[0],
                    )
                elif order is self.order and not self.position and self.stop_order:
                    self.cancel(self.stop_order)
