import backtrader as bt


class ReferenceStrategy(bt.Strategy):
    params = (
        ('hma_period', 9),
        ('atr_period', 14),
        ('atr_mult', 2.0),
    )

    def __init__(self):
        self.hma = bt.indicators.HMA(self.data.close, period=self.p.hma_period)
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)

        self.order = None
        self.stop_order = None
        self.stop_price = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            if order == self.order:
                self.order = None
                if self.position.size > 0:
                    self.stop_order = self.sell(
                        exectype=bt.Order.Stop,
                        price=self.stop_price,
                        size=self.position.size
                    )
                elif self.position.size < 0:
                    self.stop_order = self.buy(
                        exectype=bt.Order.Stop,
                        price=self.stop_price,
                        size=abs(self.position.size)
                    )
            elif order == self.stop_order:
                self.stop_order = None
                self.stop_price = None

        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            if order == self.order:
                self.order = None
            if order == self.stop_order:
                self.stop_order = None

    def next(self):
        if self.order or self.position:
            return

        hma_falling_prev = self.hma[-1] < self.hma[-2]
        hma_rising_curr = self.hma[0] > self.hma[-1]

        hma_rising_prev = self.hma[-1] > self.hma[-2]
        hma_falling_curr = self.hma[0] < self.hma[-1]

        # Long Entry: HMA shifts from falling to rising
        if hma_falling_prev and hma_rising_curr:
            self.stop_price = self.data.low[0] - self.p.atr_mult * self.atr[0]
            self.order = self.buy()

        # Short Entry: HMA shifts from rising to falling
        elif hma_rising_prev and hma_falling_curr:
            self.stop_price = self.data.high[0] + self.p.atr_mult * self.atr[0]
            self.order = self.sell()
