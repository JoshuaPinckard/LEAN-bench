import backtrader as bt


class VolatilityBalancedStrategy(bt.Strategy):
    params = (
        ("atr_period", 20),
        ("sma_period", 20),
        ("bb_period", 20),
        ("bb_dev", 2.0),
        ("risk_pct", 0.01),
        ("risk_coeff", 1.0),
        ("min_lot", 0.01),
    )

    def __init__(self):
        self.order = None
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.sma = bt.indicators.SMA(self.data.close, period=self.p.sma_period)
        self.bb = bt.indicators.BollingerBands(
            self.data.close, period=self.p.bb_period, devfactor=self.p.bb_dev
        )
        self.crossover = bt.indicators.CrossOver(self.data.close, self.sma)

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return
        self.order = None

    def _get_target_size(self):
        capital = self.broker.getvalue()
        close = self.data.close[0]
        atr = self.atr[0]

        if atr <= 0 or close <= 0:
            return 0.0

        risk_size = (capital * self.p.risk_pct) / (2.0 * atr)
        cap_size = capital / close
        size = min(risk_size, cap_size) * self.p.risk_coeff
        return max(self.p.min_lot, size)

    def next(self):
        if self.order:
            return

        size = self._get_target_size()

        if not self.position:
            if self.crossover[0] > 0:
                self.order = self.buy(size=size)
            elif self.crossover[0] < 0:
                self.order = self.sell(size=size)
        else:
            entry_price = self.position.price
            close = self.data.close[0]
            atr = self.atr[0]

            # Abnormal deviation exit: |Close - SMA(20)| > 2*Std (outside Bollinger Bands)
            dev_exceeded = close > self.bb.top[0] or close < self.bb.bot[0]

            exit_signal = False

            if self.position.size > 0:
                sl_price = entry_price - 2.0 * atr
                tp_price = entry_price + 4.0 * atr
                if close <= sl_price or close >= tp_price or dev_exceeded:
                    exit_signal = True
            else:
                sl_price = entry_price + 2.0 * atr
                tp_price = entry_price - 4.0 * atr
                if close >= sl_price or close <= tp_price or dev_exceeded:
                    exit_signal = True

            if exit_signal:
                self.order = self.close()
            else:
                target = size if self.position.size > 0 else -size
                if target != self.position.size:
                    self.order = self.order_target_size(target=target)
