import backtrader as bt


class DonchianVolumeBreakoutStrategy(bt.Strategy):
    params = (
        ("channel_period", 20),
        ("volume_sma_period", 20),
        ("stop_percent", 0.02),
    )

    def __init__(self):
        self.upper_band = bt.ind.Highest(
            self.data.high(-1), period=self.p.channel_period
        )
        self.lower_band = bt.ind.Lowest(
            self.data.low(-1), period=self.p.channel_period
        )
        self.volume_sma = bt.ind.SMA(
            self.data.volume, period=self.p.volume_sma_period
        )

        self.entry_order = None
        self.exit_order = None
        self.stop_order = None
        self.cancel_stop_for_exit = False

    def next(self):
        if self.entry_order or self.exit_order or self.cancel_stop_for_exit:
            return

        volume_confirmed = (
            self.data.volume[0] > self.data.volume[-1]
            and self.data.volume[0] > self.volume_sma[0]
        )

        if not self.position:
            if volume_confirmed and self.data.close[-1] <= self.upper_band[-1] and self.data.close[0] > self.upper_band[0]:
                self.entry_order = self.buy()
            elif volume_confirmed and self.data.close[-1] >= self.lower_band[-1] and self.data.close[0] < self.lower_band[0]:
                self.entry_order = self.sell()
            return

        if self.position.size > 0:
            should_exit = (
                self.data.close[0] < self.upper_band[0]
                or self.data.volume[0] < self.volume_sma[0]
            )
        else:
            should_exit = (
                self.data.close[0] > self.lower_band[0]
                or self.data.volume[0] < self.volume_sma[0]
            )

        if should_exit:
            if self.stop_order:
                self.cancel_stop_for_exit = True
                self.cancel(self.stop_order)
            else:
                self.exit_order = self.close()

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return

        if order == self.entry_order:
            self.entry_order = None
            if order.status == order.Completed:
                if order.isbuy():
                    stop_price = self.upper_band[0] * (1 - self.p.stop_percent)
                    self.stop_order = self.sell(
                        exectype=bt.Order.Stop, price=stop_price
                    )
                else:
                    stop_price = self.lower_band[0] * (1 + self.p.stop_percent)
                    self.stop_order = self.buy(
                        exectype=bt.Order.Stop, price=stop_price
                    )
            return

        if order == self.stop_order:
            self.stop_order = None
            if self.cancel_stop_for_exit:
                self.cancel_stop_for_exit = False
                if self.position:
                    self.exit_order = self.close()
            return

        if order == self.exit_order:
            self.exit_order = None
