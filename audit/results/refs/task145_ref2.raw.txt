import backtrader as bt


class VolatilitySpikeStrategy(bt.Strategy):
    """Recent Strong Price Move Detection Strategy (Volatility Spike Strategy)

    Identifies stocks exhibiting a significant price swing (X%) over Y days,
    filtered by average daily dollar trading volume ($N) and market cap ($M).
    Exits positions after Z days or upon hitting Stop-Loss (5%) or Take-Profit (10%).
    """

    params = (
        ("X", 20.0),  # Price change threshold (%)
        ("Y", 10),  # Volatility lookback period (days)
        ("Z", 5),  # Holding period (days)
        ("stop_loss", 0.05),  # Stop-loss percentage (5%)
        ("take_profit", 0.10),  # Take-profit percentage (10%)
        ("N", 1000000.0),  # Min average daily dollar trading volume ($)
        ("M", 500000000.0),  # Min market capitalization ($)
    )

    def __init__(self):
        self.order = None
        self.entry_price = None
        self.entry_bar = 0

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            if order.isbuy():
                self.entry_price = order.executed.price
                self.entry_bar = len(self)
            self.order = None
        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        if len(self) < self.p.Y:
            return

        if self.order:
            return

        if not self.position:
            # 1. Volatility Indicator: (price_on_period_close_day / price_on_period_open_day - 1) * 100
            price_close = self.data.close[0]
            price_open = self.data.open[-(self.p.Y - 1)]

            pct_change = ((price_close / price_open) - 1.0) * 100.0

            # 2. Volume Filter: Average daily dollar volume over Y days > $N
            dollar_vols = [
                self.data.close[-i] * self.data.volume[-i] for i in range(self.p.Y)
            ]
            avg_dollar_vol = sum(dollar_vols) / float(self.p.Y)

            volatility_signal = abs(pct_change) > self.p.X
            volume_signal = avg_dollar_vol > self.p.N

            # 3. Entry Signal Verification
            if volatility_signal and volume_signal:
                cash = self.broker.getcash()
                size = int((cash * 0.99) / price_close)
                if size > 0:
                    self.order = self.buy(size=size)
        else:
            # Exit Conditions: Stop-Loss, Take-Profit, or Z-day holding period
            bars_held = len(self) - self.entry_bar
            stop_price = self.entry_price * (1.0 - self.p.stop_loss)
            tp_price = self.entry_price * (1.0 + self.p.take_profit)

            if self.data.low[0] <= stop_price:
                self.order = self.close()
            elif self.data.high[0] >= tp_price:
                self.order = self.close()
            elif bars_held >= self.p.Z:
                self.order = self.close()
