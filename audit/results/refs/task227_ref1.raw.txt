import backtrader as bt


class UniversalPortfolioStrategy(bt.Strategy):
    params = ()

    def __init__(self):
        self.order = None
        self.fractions = [i / 10.0 for i in range(11)]

    def next(self):
        if self.order:
            return

        # 252 returns require 253 closing prices.
        if len(self.data) <= 252:
            return

        wealths = []
        for w in self.fractions:
            wealth = 1.0
            for i in range(252):
                daily_return = (self.data.close[-i] / self.data.close[-i - 1]) - 1.0
                wealth *= 1.0 + (w * daily_return)
            wealths.append(wealth)

        total_wealth = sum(wealths)
        target_weight = sum(
            w * wealth for w, wealth in zip(self.fractions, wealths)
        ) / total_wealth

        self.order = self.order_target_percent(
            data=self.data,
            target=target_weight,
        )

    def notify_order(self, order):
        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            self.order = None
