import backtrader as bt


class VolatilitySpikeStrategy(bt.Strategy):
    def __init__(self):
        self.sma20 = bt.indicators.SMA(self.data.close, period=20)
        self.sma50 = bt.indicators.SMA(self.data.close, period=50)

        self.bb = bt.indicators.BollingerBands(
            self.data.close, period=20, devfactor=2
        )
        self.bbwidth = self.bb.top - self.bb.bot
        self.bbwidth_sma = bt.indicators.SMA(self.bbwidth, period=20)

        self.atr = bt.indicators.ATR(self.data, period=14)
        self.volume_sma = bt.indicators.SMA(self.data.volume, period=20)

        self.long_cross = bt.indicators.CrossOver(self.data.close, self.bb.top)
        self.short_cross = bt.indicators.CrossOver(self.bb.bot, self.data.close)

        self.order = None
        self.order_role = None
        self.stop_order = None
        self.take_order = None
        self.entry_bar = None

    def notify_order(self, order):
        if order.status not in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            return

        role = None
        if order == self.order:
            role = self.order_role
            self.order = None
            self.order_role = None

        if order == self.stop_order:
            self.stop_order = None

        if order == self.take_order:
            self.take_order = None

        if order.status != order.Completed:
            return

        if role == "entry":
            self.entry_bar = len(self)
            atr_value = self.atr[0]

            if order.isbuy():
                self.stop_order = self.sell(
                    exectype=bt.Order.Stop,
                    price=order.executed.price - atr_value,
                )
                self.take_order = self.sell(
                    exectype=bt.Order.Limit,
                    price=order.executed.price + (2.0 * atr_value),
                    oco=self.stop_order,
                )
            else:
                self.stop_order = self.buy(
                    exectype=bt.Order.Stop,
                    price=order.executed.price + atr_value,
                )
                self.take_order = self.buy(
                    exectype=bt.Order.Limit,
                    price=order.executed.price - (2.0 * atr_value),
                    oco=self.stop_order,
                )

        elif role == "time_exit":
            self.entry_bar = None

        elif order == self.stop_order or order == self.take_order:
            self.entry_bar = None

        if order == self.stop_order and self.take_order is not None:
            self.cancel(self.take_order)
        elif order == self.take_order and self.stop_order is not None:
            self.cancel(self.stop_order)

    def next(self):
        if self.position:
            if (
                self.entry_bar is not None
                and len(self) - self.entry_bar >= 3
                and self.order is None
            ):
                if self.stop_order is not None:
                    self.cancel(self.stop_order)
                if self.take_order is not None:
                    self.cancel(self.take_order)

                self.order = self.close()
                self.order_role = "time_exit"
            return

        if (
            self.order is not None
            or self.stop_order is not None
            or self.take_order is not None
        ):
            return

        volatility_expansion = self.bbwidth[0] > self.bbwidth_sma[0] * 1.3
        volume_spike = self.data.volume[0] > 1.5 * self.volume_sma[0]

        if (
            self.sma20[0] > self.sma50[0]
            and volatility_expansion
            and volume_spike
            and self.long_cross[0] > 0
        ):
            self.order = self.buy()
            self.order_role = "entry"

        elif (
            self.sma20[0] < self.sma50[0]
            and volatility_expansion
            and volume_spike
            and self.short_cross[0] < 0
        ):
            self.order = self.sell()
            self.order_role = "entry"
