import datetime
import backtrader as bt


class BullishEngulfingStrategy(bt.Strategy):
    def __init__(self):
        self.order = None
        self.stop_order = None
        self.stop_price = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            if order.isbuy():
                self.order = None
                if self.stop_price is not None:
                    self.stop_order = self.sell(
                        exectype=bt.Order.Stop, price=self.stop_price
                    )
            elif order.issell():
                if order == self.stop_order:
                    self.stop_order = None
                if order == self.order:
                    self.order = None

        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            if order == self.order:
                self.order = None
            if order == self.stop_order:
                self.stop_order = None

    def next(self):
        if len(self) < 5:
            return

        current_time = self.data.datetime.time(0)
        is_session_end = current_time >= datetime.time(15, 45)

        if is_session_end and self.position:
            if self.stop_order:
                self.cancel(self.stop_order)
                self.stop_order = None
            self.order = self.close()
            return

        if not self.position and self.order is None:
            bar_m1_bearish = self.data.close[-1] < self.data.open[-1]
            bar_0_bullish = self.data.close[0] > self.data.open[0]
            engulfs = (self.data.open[0] <= self.data.close[-1]) and (
                self.data.close[0] >= self.data.open[-1]
            )

            filter_bearish = (
                (self.data.close[-2] < self.data.open[-2])
                and (self.data.close[-3] < self.data.open[-3])
                and (self.data.close[-4] < self.data.open[-4])
            )

            if bar_m1_bearish and bar_0_bullish and engulfs and filter_bearish:
                pattern_low = min(self.data.low[0], self.data.low[-1])
                self.stop_price = pattern_low * 0.99
                self.order = self.buy()
