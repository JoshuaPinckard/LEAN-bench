import backtrader as bt


class EnergySectorETFStrategy(bt.Strategy):
    params = dict(
        sma_period=50,
        rsi_period=14,
        daily_gain_threshold=0.02,
        rsi_low=40,
        rsi_high=65,
        take_profit=0.10,
        stop_loss=0.05,
        max_hold_days=90,
        position_pct=0.05,
    )

    def __init__(self):
        self.sma = bt.indicators.SimpleMovingAverage(
            self.data.close, period=self.p.sma_period
        )
        self.rsi = bt.indicators.RSI(self.data.close, period=self.p.rsi_period)
        self.order = None
        self.entry_price = None
        self.entry_date = None

    def notify_order(self, order):
        if order.status == order.Completed:
            if order.isbuy():
                self.entry_price = order.executed.price
                self.entry_date = self.data.datetime.date(0)
            else:
                self.entry_price = None
                self.entry_date = None
        if order.status in (order.Completed, order.Canceled, order.Margin, order.Rejected):
            self.order = None

    def next(self):
        if self.order:
            return

        if not self.position:
            if len(self.data) < self.p.sma_period + 1:
                return

            daily_gain = (self.data.close[0] - self.data.close[-1]) / self.data.close[-1]
            trend_up = self.sma[0] > self.data.close[-self.p.sma_period]
            rsi_ok = self.p.rsi_low <= self.rsi[0] <= self.p.rsi_high

            if daily_gain >= self.p.daily_gain_threshold and trend_up and rsi_ok:
                size = (self.broker.getvalue() * self.p.position_pct) / self.data.close[0]
                # exectype=Close fills using the *next* bar's closing price, per spec
                self.order = self.buy(size=size, exectype=bt.Order.Close)
        else:
            gain = (self.data.close[0] - self.entry_price) / self.entry_price
            days_held = (self.data.datetime.date(0) - self.entry_date).days

            if gain >= self.p.take_profit or gain <= -self.p.stop_loss or days_held > self.p.max_hold_days:
                self.order = self.close(exectype=bt.Order.Close)
