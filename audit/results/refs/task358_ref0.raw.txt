import backtrader as bt


class WeekdaySmaStopStrategy(bt.Strategy):
    params = (
        ('sma_period', 20),
        ('stop_loss_pct', 0.02),
        ('cash_buffer', 0.99),
    )

    def __init__(self):
        self.sma = bt.indicators.SimpleMovingAverage(
            self.data.close, period=self.p.sma_period
        )
        self.order = None
        self.stop_order = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status == order.Completed:
            if order is self.stop_order:
                self.stop_order = None
                self.order = None
            elif order is self.order:
                if self.position:
                    entry_price = order.executed.price
                    size = abs(self.position.size)
                    if order.isbuy():
                        stop_price = entry_price * (1 - self.p.stop_loss_pct)
                        self.stop_order = self.sell(
                            exectype=bt.Order.Stop, price=stop_price, size=size
                        )
                    else:
                        stop_price = entry_price * (1 + self.p.stop_loss_pct)
                        self.stop_order = self.buy(
                            exectype=bt.Order.Stop, price=stop_price, size=size
                        )
                self.order = None
        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            if order is self.stop_order:
                self.stop_order = None
            if order is self.order:
                self.order = None

    def next(self):
        if self.order:
            return

        weekday = self.data.datetime.date(0).weekday()
        price = self.data.close[0]
        pos_size = self.position.size

        if pos_size == 0:
            if weekday == 0 and price > self.sma[0]:
                cash = self.broker.getcash()
                size = int((cash * self.p.cash_buffer) / price)
                if size > 0:
                    self.order = self.buy(size=size)
            elif weekday == 4 and price < self.sma[0]:
                cash = self.broker.getcash()
                size = int((cash * self.p.cash_buffer) / price)
                if size > 0:
                    self.order = self.sell(size=size)
        elif pos_size > 0 and weekday == 4:
            self.cancel(self.stop_order)
            self.stop_order = None
            self.order = self.close()
        elif pos_size < 0 and weekday == 0:
            self.cancel(self.stop_order)
            self.stop_order = None
            self.order = self.close()
