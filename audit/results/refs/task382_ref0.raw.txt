import backtrader as bt


class CCISlopeZScoreStrategy(bt.Strategy):
    def __init__(self):
        self.cci = bt.indicators.CCI(self.data, period=20)
        self.slope = bt.indicators.LinearRegressionSlope(self.cci, period=20)
        self.mean_slope = bt.indicators.SMA(self.slope, period=20)
        self.std_slope = bt.indicators.StdDev(self.slope, period=20)

        self.order = None
        self.stop_order = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order == self.stop_order:
            self.stop_order = None
            return

        if order.status == order.Completed:
            if order.isbuy() and self.position.size > 0:
                stop_price = order.executed.price * 0.98
                self.stop_order = self.sell(exectype=bt.Order.Stop, price=stop_price)
            elif order.issell() and self.position.size < 0:
                stop_price = order.executed.price * 1.02
                self.stop_order = self.buy(exectype=bt.Order.Stop, price=stop_price)

        self.order = None

    def next(self):
        if self.order:
            return

        slope = self.slope[0]
        mean_slope = self.mean_slope[0]
        std_slope = self.std_slope[0]

        if not self.position:
            if std_slope == 0:
                return
            z = (slope - mean_slope) / std_slope
            if z > 2:
                self.order = self.buy()
            elif z < -2:
                self.order = self.sell()
        else:
            if self.position.size > 0 and slope <= mean_slope:
                if self.stop_order is not None:
                    self.cancel(self.stop_order)
                self.order = self.close()
            elif self.position.size < 0 and slope >= mean_slope:
                if self.stop_order is not None:
                    self.cancel(self.stop_order)
                self.order = self.close()
