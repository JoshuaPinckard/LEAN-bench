import datetime
import backtrader as bt


class ExchangeScheduleFilterStrategy(bt.Strategy):
    """Reference implementation of a trading strategy that filters entries and

    exits based on U.S. exchange schedules (weekends and holidays).
    """

    def __init__(self):
        self.order = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return
        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            self.order = None

    @staticmethod
    def get_easter(year):
        """Calculates Easter Sunday for a given year using the Meeus/Jones/Butcher algorithm."""
        a = year % 19
        b = year // 100
        c = year % 100
        d = b // 4
        e = b % 4
        f = (b + 8) // 25
        g = (b - f + 1) // 3
        h = (19 * a + b - d - g + 15) % 30
        i = c // 4
        k = c % 4
        l = (32 + 2 * e + 2 * i - h - k) % 7
        m = (a + 11 * h + 22 * l) // 451
        month = (h + l - 7 * m + 114) // 31
        day = ((h + l - 7 * m + 114) % 31) + 1
        return datetime.date(year, month, day)

    @classmethod
    def is_us_business_day(cls, dt):
        """Checks if an arbitrary date (datetime.date) is a business day for U.S.

        exchanges (NYSE/NASDAQ), taking into account weekends and official
        holidays.
        """
        if dt.weekday() >= 5:
            return False

        year = dt.year
        month = dt.month
        day = dt.day

        # New Year's Day
        if month == 1 and day == 1:
            return False
        if month == 1 and day == 2 and dt.weekday() == 0:
            return False

        # Martin Luther King Jr. Day (3rd Monday in January)
        if month == 1 and dt.weekday() == 0 and 15 <= day <= 21:
            return False

        # Washington's Birthday / Presidents' Day (3rd Monday in February)
        if month == 2 and dt.weekday() == 0 and 15 <= day <= 21:
            return False

        # Good Friday
        easter = cls.get_easter(year)
        good_friday = easter - datetime.timedelta(days=2)
        if dt == good_friday:
            return False

        # Memorial Day (last Monday in May)
        if month == 5 and dt.weekday() == 0 and day >= 25:
            return False

        # Juneteenth National Independence Day (June 19, observed starting 2022)
        if year >= 2022:
            if month == 6 and day == 19:
                return False
            if month == 6 and day == 20 and dt.weekday() == 0:
                return False
            if month == 6 and day == 18 and dt.weekday() == 4:
                return False

        # Independence Day (July 4)
        if month == 7 and day == 4:
            return False
        if month == 7 and day == 5 and dt.weekday() == 0:
            return False
        if month == 7 and day == 3 and dt.weekday() == 4:
            return False

        # Labor Day (1st Monday in September)
        if month == 9 and dt.weekday() == 0 and day <= 7:
            return False

        # Thanksgiving Day (4th Thursday in November)
        if month == 11 and dt.weekday() == 3 and 22 <= day <= 28:
            return False

        # Christmas Day (December 25)
        if month == 12 and day == 25:
            return False
        if month == 12 and day == 26 and dt.weekday() == 0:
            return False
        if month == 12 and day == 24 and dt.weekday() == 4:
            return False

        return True

    @classmethod
    def is_last_business_day_of_week(cls, dt):
        """Determines if the given date is the last business day of its calendar

        week.
        """
        if not cls.is_us_business_day(dt):
            return False
        days_to_sunday = 6 - dt.weekday()
        for i in range(1, days_to_sunday + 1):
            future_day = dt + datetime.timedelta(days=i)
            if cls.is_us_business_day(future_day):
                return False
        return True

    def next(self):
        if self.order:
            return

        current_date = self.data.datetime.date(0)
        is_biz_day = self.is_us_business_day(current_date)
        is_last_biz_day = self.is_last_business_day_of_week(current_date)

        if self.position:
            # Exit rule: Forcibly close positions on the last business day of the week to avoid weekend holdings
            if is_last_biz_day:
                self.order = self.close()
        else:
            # Entry rule: Execute buy signal only on valid business days that are not the last business day of the week
            if is_biz_day and not is_last_biz_day:
                self.order = self.buy()
