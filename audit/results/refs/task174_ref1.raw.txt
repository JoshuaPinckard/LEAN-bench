import backtrader as bt
import math


class ProfitBackedExpansion(bt.Strategy):
    params = (
        ('rsi_period', 14),
        ('high_period', 252),
        ('volume_fast_period', 20),
        ('volume_slow_period', 100),
    )

    def __init__(self):
        self.high_52w = bt.indicators.Highest(self.data.high, period=self.p.high_period)
        self.rsi = bt.indicators.RSI(self.data.close, period=self.p.rsi_period)
        self.volume_sma_20 = bt.indicators.SMA(
            self.data.volume, period=self.p.volume_fast_period
        )
        self.volume_sma_100 = bt.indicators.SMA(
            self.data.volume, period=self.p.volume_slow_period
        )

        self.order = None
        self.order_type = None

        self.base_entry_price = None
        self.base_shares = 0

        self.tranche_entry_price = None
        self.tranche_shares = 0
        self.tranche_entry_bar = None

    def notify_order(self, order):
        if order.status == order.Completed:
            if self.order_type == 'base_buy':
                self.base_entry_price = order.executed.price
                self.base_shares = order.executed.size

            elif self.order_type == 'tranche_buy':
                self.tranche_entry_price = order.executed.price
                self.tranche_shares = order.executed.size
                self.tranche_entry_bar = len(self)

            elif self.order_type == 'tranche_exit':
                self.tranche_entry_price = None
                self.tranche_shares = 0
                self.tranche_entry_bar = None

        if order.status in (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            self.order = None
            self.order_type = None

    def next(self):
        if self.order:
            return

        if self.base_entry_price is None:
            self.order = self.buy()
            self.order_type = 'base_buy'
            return

        close = self.data.close[0]

        if self.tranche_shares > 0:
            tp = self.tranche_entry_price * 1.12
            time_stop = len(self) - self.tranche_entry_bar >= 126

            if (
                close >= tp
                or close <= 0.8 * self.base_entry_price
                or time_stop
            ):
                self.order = self.sell(size=self.tranche_shares)
                self.order_type = 'tranche_exit'
            return

        unrealized_profit_base = (
            (close - self.base_entry_price) * self.base_shares
        )
        buy_shares = math.floor(0.5 * unrealized_profit_base / close)

        if (
            close >= 0.9 * self.high_52w[0]
            and self.rsi[0] > 50
            and close / self.base_entry_price - 1 >= 0.4
            and self.volume_sma_20[0] > self.volume_sma_100[0]
            and buy_shares > 0
        ):
            self.order = self.buy(size=buy_shares)
            self.order_type = 'tranche_buy'
