import backtrader as bt


class MACDSignalCrossover(bt.Strategy):
    def __init__(self):
        self.order = None
        self.macd = bt.indicators.MACD(
            self.data.close, period_me1=12, period_me2=26, period_signal=9
        )
        self.crossover = bt.indicators.CrossOver(
            self.macd.macd, self.macd.signal
        )

    def notify_order(self, order):
        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            self.order = None

    def next(self):
        if self.order:
            return

        if self.crossover > 0:
            if self.position.size < 0:
                self.close()
            if self.position.size <= 0:
                self.order = self.buy()

        elif self.crossover < 0:
            if self.position.size > 0:
                self.close()
            if self.position.size >= 0:
                self.order = self.sell()
