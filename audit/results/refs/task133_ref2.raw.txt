import backtrader as bt


class EuropeanEquitiesStopLossStrategy(bt.Strategy):
    params = (
        ("sma_period", 20),
        ("atr_period", 14),
        ("position_size_pct", 0.01),
        ("take_profit_pct", 0.02),
        ("stop_loss_atr_mult", 2.0),
    )

    def __init__(self):
        self.sma = bt.indicators.SimpleMovingAverage(
            self.data.close, period=self.p.sma_period
        )
        self.atr = bt.indicators.AverageTrueRange(
            self.data, period=self.p.atr_period
        )

        self.order = None
        self.stop_order = None
        self.limit_order = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            if order.isbuy():
                entry_price = order.executed.price
                atr_val = self.atr[0]
                sl_price = entry_price - self.p.stop_loss_atr_mult * atr_val
                sl_price = max(0.01, sl_price)
                tp_price = entry_price * (1.0 + self.p.take_profit_pct)

                self.stop_order = self.sell(
                    size=order.executed.size,
                    price=sl_price,
                    exectype=bt.Order.Stop,
                )
                self.limit_order = self.sell(
                    size=order.executed.size,
                    price=tp_price,
                    exectype=bt.Order.Limit,
                    oco=self.stop_order,
                )
                self.order = None
            else:
                self.order = None
                self.stop_order = None
                self.limit_order = None

        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            self.order = None
            if order == self.stop_order or order == self.limit_order:
                self.stop_order = None
                self.limit_order = None

    def next(self):
        if self.order or self.position:
            return

        if self.data.close[0] > self.sma[0]:
            capital = self.broker.getvalue()
            position_value = capital * self.p.position_size_pct
            size = position_value / self.data.close[0]
            if size > 0:
                self.order = self.buy(size=size)
