import backtrader as bt


class KalmanAtrTrendStrategy(bt.Strategy):
    params = (
        ("atr_period", 14),
        ("atr_stop_multiple", 2.0),
        ("process_variance", 0.0001),
        ("measurement_variance", 0.01),
    )

    def __init__(self):
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.order = None
        self.stop_order = None
        self.kalman_estimate = None
        self.kalman_variance = 1.0
        self.previous_close = None
        self.previous_estimate = None

    def prenext(self):
        self._update_kalman()

    def next(self):
        self._update_kalman()

        if self.order:
            return

        crossed_above = (
            self.previous_close is not None
            and self.previous_close <= self.previous_estimate
            and self.data.close[0] > self.kalman_estimate
        )
        crossed_below = (
            self.previous_close is not None
            and self.previous_close >= self.previous_estimate
            and self.data.close[0] < self.kalman_estimate
        )

        if self.position.size == 0:
            if crossed_above:
                self.order = self.buy()
            elif crossed_below:
                self.order = self.sell()
        elif self.position.size > 0 and crossed_below:
            self._cancel_stop()
            self.order = self.close()
        elif self.position.size < 0 and crossed_above:
            self._cancel_stop()
            self.order = self.close()

        self.previous_close = self.data.close[0]
        self.previous_estimate = self.kalman_estimate

    def _update_kalman(self):
        price = self.data.close[0]

        if self.kalman_estimate is None:
            self.kalman_estimate = price
            self.previous_close = price
            self.previous_estimate = price
            return

        predicted_variance = self.kalman_variance + self.p.process_variance
        gain = predicted_variance / (
            predicted_variance + self.p.measurement_variance
        )
        self.kalman_estimate += gain * (price - self.kalman_estimate)
        self.kalman_variance = (1.0 - gain) * predicted_variance

    def _cancel_stop(self):
        if self.stop_order:
            self.cancel(self.stop_order)
            self.stop_order = None

    def notify_order(self, order):
        if order.status in (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            if order == self.order:
                self.order = None

            if order == self.stop_order:
                self.stop_order = None

        if order.status == order.Completed and order == self.order:
            if order.isbuy():
                self.stop_order = self.sell(
                    exectype=bt.Order.Stop,
                    price=order.executed.price
                    - self.p.atr_stop_multiple * self.atr[0],
                )
            elif order.issell():
                self.stop_order = self.buy(
                    exectype=bt.Order.Stop,
                    price=order.executed.price
                    + self.p.atr_stop_multiple * self.atr[0],
                )
