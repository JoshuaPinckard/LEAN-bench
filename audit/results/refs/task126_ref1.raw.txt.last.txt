import backtrader as bt


class DemaSuperTrendStrategy(bt.Strategy):
    class SuperTrend(bt.Indicator):
        lines = ("supertrend", "direction")
        params = (("period", 12), ("multiplier", 3.0))

        def __init__(self):
            self.atr = bt.ind.ATR(self.data, period=self.p.period)
            self.addminperiod(self.atr._minperiod)

        def nextstart(self):
            midpoint = (self.data.high[0] + self.data.low[0]) / 2.0
            upper = midpoint + self.p.multiplier * self.atr[0]
            lower = midpoint - self.p.multiplier * self.atr[0]

            self.lines.supertrend[0] = upper
            self.lines.direction[0] = -1

        def next(self):
            midpoint = (self.data.high[0] + self.data.low[0]) / 2.0
            basic_upper = midpoint + self.p.multiplier * self.atr[0]
            basic_lower = midpoint - self.p.multiplier * self.atr[0]

            previous_supertrend = self.lines.supertrend[-1]
            previous_close = self.data.close[-1]

            if previous_supertrend > previous_close:
                final_upper = min(basic_upper, previous_supertrend)
                if self.data.close[0] > final_upper:
                    self.lines.supertrend[0] = basic_lower
                    self.lines.direction[0] = 1
                else:
                    self.lines.supertrend[0] = final_upper
                    self.lines.direction[0] = -1
            else:
                final_lower = max(basic_lower, previous_supertrend)
                if self.data.close[0] < final_lower:
                    self.lines.supertrend[0] = basic_upper
                    self.lines.direction[0] = -1
                else:
                    self.lines.supertrend[0] = final_lower
                    self.lines.direction[0] = 1

    def __init__(self):
        self.order = None
        self.dema = bt.ind.DEMA(self.data.close, period=200)
        self.supertrend = self.SuperTrend(
            self.data,
            period=12,
            multiplier=3.0,
        )

    def next(self):
        if self.order:
            return

        buy_signal = (
            self.supertrend.direction[-1] < 0
            and self.supertrend.direction[0] > 0
        )
        sell_signal = (
            self.supertrend.direction[-1] > 0
            and self.supertrend.direction[0] < 0
        )

        if not self.position:
            if self.data.close[0] > self.dema[0] and buy_signal:
                self.order = self.order_target_percent(target=1.0)
        elif sell_signal:
            self.order = self.close()

    def notify_order(self, order):
        if order.status in (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            self.order = None
