import backtrader as bt


class MeanReversionBollingerZScore(bt.Strategy):
    params = (
        ('lookback', 20),
        ('half_life', 10),
        ('entry_threshold', 2.0),
        ('stoploss_factor', 1.5),
        ('narrow_band_epsilon', 1e-12),
    )

    def __init__(self):
        self.order = None
        self.bbands = bt.indicators.BollingerBands(
            self.data.close, period=self.p.lookback, devfactor=2
        )
        self.band_width = self.bbands.top - self.bbands.mid
        self.z = 2.0 * (self.data.close - self.bbands.mid) / self.band_width
        self.z_ema = bt.indicators.EMA(self.z, period=self.p.half_life)
        self.z_cross_up = bt.indicators.CrossUp(self.z_ema, 0.0)
        self.z_cross_down = bt.indicators.CrossDown(self.z_ema, 0.0)
        self.stoploss_threshold = (
            self.p.entry_threshold * self.p.stoploss_factor
        )

    def next(self):
        if self.order:
            return

        if abs(self.band_width[0]) <= self.p.narrow_band_epsilon:
            return

        if not self.position:
            if self.z[0] < -self.p.entry_threshold:
                self.order = self.buy()
            elif self.z[0] > self.p.entry_threshold:
                self.order = self.sell()
            return

        if self.position.size > 0:
            if self.z_cross_up[0] or self.z[0] >= self.stoploss_threshold:
                self.order = self.close()
        elif self.position.size < 0:
            if self.z_cross_down[0] or self.z[0] <= -self.stoploss_threshold:
                self.order = self.close()

    def notify_order(self, order):
        if order.status in (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            self.order = None
