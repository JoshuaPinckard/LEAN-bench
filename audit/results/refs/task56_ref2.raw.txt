import datetime
import math
import statistics
import backtrader as bt


class DailyVolatilityRanker(bt.Strategy):
    params = (
        ("start_time", datetime.time(8, 0)),
        ("end_time", datetime.time(9, 0)),
    )

    def __init__(self):
        self.current_date = None
        self.daily_log_returns = []
        self.daily_volatility = {}
        self.ranked_days = []

    def _finalize_day(self):
        if self.current_date is not None and self.daily_log_returns:
            if len(self.daily_log_returns) > 1:
                vol = statistics.stdev(self.daily_log_returns)
            else:
                vol = 0.0
            self.daily_volatility[self.current_date] = vol

    def next(self):
        bar_dt = self.data.datetime.datetime(0)
        bar_date = bar_dt.date()
        bar_time = bar_dt.time()

        if self.current_date != bar_date:
            if self.current_date is not None:
                self._finalize_day()
            self.current_date = bar_date
            self.daily_log_returns = []

        if self.p.start_time <= bar_time <= self.p.end_time:
            open_price = self.data.open[0]
            close_price = self.data.close[0]
            if open_price > 0 and close_price > 0:
                log_ret = math.log(close_price / open_price)
                self.daily_log_returns.append(log_ret)

    def stop(self):
        self._finalize_day()
        self.ranked_days = sorted(
            self.daily_volatility.items(), key=lambda x: x[1], reverse=True
        )
