import backtrader as bt


class SingleAssetRiskParity(bt.Strategy):
    params = (
        ("vtgt", 0.01),
        ("period", 252),
        ("min_w", 0.01),
        ("max_w", 0.10),
    )

    def __init__(self):
        self.atr = bt.indicators.ATR(self.data, period=self.p.period)
        self.sma = bt.indicators.SMA(self.data.close, period=self.p.period)
        self.order = None

    def notify_order(self, order):
        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            self.order = None

    def next(self):
        if self.order:
            return

        dt_today = self.data.datetime.date(0)
        is_last_day = False
        if len(self) < len(self.data):
            dt_next = self.data.datetime.date(1)
            if dt_today.month != dt_next.month:
                is_last_day = True
        else:
            is_last_day = True

        if not is_last_day:
            return

        sma_val = self.sma[0]
        atr_val = self.atr[0]

        if sma_val > 0:
            vol = atr_val / sma_val
        else:
            vol = 0.0

        if vol > 0:
            w_star = self.p.vtgt / vol
        else:
            w_star = self.p.min_w

        w = min(max(w_star, self.p.min_w), self.p.max_w)
        self.order = self.order_target_percent(target=w)
