import backtrader as bt


class APlusReversalStrategy(bt.Strategy):
    """Reference implementation of the A+ Reversal Strategy."""

    def __init__(self):
        self.order = None
        self.sl_price = None
        self.tp_price = None

    def notify_order(self, order):
        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            self.order = None

    def next(self):
        if self.order:
            return

        if not self.position:
            if len(self.data) < 2:
                return

            # Condition 1: Current day open above previous day high (Outside Day condition)
            cond1 = self.data.open[0] > self.data.high[-1]
            # Condition 2: Current day open above previous day close (Gap Up condition)
            cond2 = self.data.open[0] > self.data.close[-1]
            # Condition 3: Current session open above ICT Midnight Open level (approximated as previous close)
            ict_midnight_open = self.data.close[-1]
            cond3 = self.data.open[0] > ict_midnight_open

            if cond1 and cond2 and cond3:
                self.order = self.sell()
                # SL placed above the initial spike high (signal day's high)
                self.sl_price = self.data.high[0]
                # TP set to previous day high (TP2 / Outside Day target level)
                self.tp_price = self.data.high[-1]
        else:
            if self.sl_price is not None and self.data.high[0] >= self.sl_price:
                self.order = self.close()
            elif (
                self.tp_price is not None and self.data.low[0] <= self.tp_price
            ):
                self.order = self.close()
