import backtrader as bt


class TweezerBottomStrategy(bt.Strategy):
    params = (
        ("sma_period", 20),
        ("tolerance", 0.001),
        ("stop_pct", 0.02),
    )

    def __init__(self):
        self.sma = bt.indicators.SimpleMovingAverage(
            self.data.close, period=self.p.sma_period
        )
        self.order = None
        self.stop_order = None
        self.stop_price = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            if order == self.order:
                self.order = None
                if self.position and self.stop_price is not None:
                    self.stop_order = self.sell(
                        exectype=bt.Order.Stop, price=self.stop_price
                    )
            elif order == self.stop_order:
                self.stop_order = None

        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            if order == self.order:
                self.order = None
            if order == self.stop_order:
                self.stop_order = None

    def next(self):
        if self.position:
            is_bearish = self.data.close[0] < self.data.open[0]
            engulfs = (self.data.open[0] >= self.data.close[-1]) and (
                self.data.close[0] <= self.data.open[-1]
            )

            if is_bearish and engulfs:
                if self.order is None:
                    if self.stop_order is not None:
                        self.cancel(self.stop_order)
                        self.stop_order = None
                    self.order = self.close()
            return

        if self.order is not None or self.stop_order is not None:
            return

        # (1) Decline condition
        decline = (self.data.close[0] < self.sma[0]) and (
            self.data.close[0] < self.data.close[-5]
        )

        # (2) Tweezer Bottom setup (bars -2 and -1)
        lows_equal = (
            abs(self.data.low[-2] - self.data.low[-1])
            <= self.p.tolerance * self.data.close[-1]
        )
        bar_minus2_bearish = self.data.close[-2] < self.data.open[-2]
        bar_minus1_bullish = self.data.close[-1] > self.data.open[-1]

        tweezer_setup = lows_equal and bar_minus2_bearish and bar_minus1_bullish

        # (3) Confirmation (bar 0)
        confirmation = self.data.close[0] > self.data.high[-1]

        if decline and tweezer_setup and confirmation:
            common_low = min(self.data.low[-2], self.data.low[-1])
            self.stop_price = common_low * (1.0 - self.p.stop_pct)
            self.order = self.buy()
