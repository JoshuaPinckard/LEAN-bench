import backtrader as bt


class AbsoluteMomentumStrategy(bt.Strategy):
    params = (
        ("period", 252),
        ("threshold", 100.0),
        ("capital_pct", 0.95),
    )

    def __init__(self):
        self.mom = bt.indicators.MomentumOscillator(
            self.data.close, period=self.p.period
        )
        self.order = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            self.order = None

    def next(self):
        if self.order:
            return

        if not self.position:
            if self.mom[0] > self.p.threshold:
                available_cash = self.broker.getcash()
                target_value = available_cash * self.p.capital_pct
                size = int(target_value / self.data.close[0])
                if size > 0:
                    self.order = self.buy(size=size)
        else:
            if self.mom[0] < self.p.threshold:
                self.order = self.close()
