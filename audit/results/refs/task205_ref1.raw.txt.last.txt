import backtrader as bt


class ReferenceStrategy(bt.Strategy):
    params = (
        ('ema_period', 20),
        ('volume_sma_period', 20),
        ('stop_loss', 0.015),
        ('take_profit', 0.03),
    )

    def __init__(self):
        self.ema = bt.indicators.EMA(self.data.close, period=self.p.ema_period)
        self.volume_sma = bt.indicators.SMA(
            self.data.volume, period=self.p.volume_sma_period
        )
        self.crossover = bt.indicators.CrossOver(self.data.close, self.ema)
        self.order = None
        self.entry_price = None

    def notify_order(self, order):
        if order.status == order.Completed:
            if self.position.size:
                self.entry_price = order.executed.price
            else:
                self.entry_price = None

        if order.status in (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            self.order = None

    def _size(self):
        commission = self.broker.getcommissioninfo(self.data).p.commission
        return self.broker.getcash() / (
            self.data.close[0] * (1.0 + commission)
        )

    def next(self):
        if self.order or len(self.data) < 20:
            return

        if self.position.size > 0:
            if (
                self.crossover[0] < 0
                or self.data.close[0] <= self.entry_price * (1.0 - self.p.stop_loss)
                or self.data.close[0] >= self.entry_price * (1.0 + self.p.take_profit)
            ):
                self.order = self.close()

        elif self.position.size < 0:
            if (
                self.crossover[0] > 0
                or self.data.close[0] >= self.entry_price * (1.0 + self.p.stop_loss)
                or self.data.close[0] <= self.entry_price * (1.0 - self.p.take_profit)
            ):
                self.order = self.close()

        elif self.data.volume[0] > self.volume_sma[0]:
            size = self._size()

            if self.crossover[0] > 0:
                self.order = self.buy(size=size)
            elif self.crossover[0] < 0:
                self.order = self.sell(size=size)
