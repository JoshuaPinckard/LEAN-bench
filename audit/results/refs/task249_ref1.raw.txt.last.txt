import backtrader as bt


class MomentumIchimokuSMA(bt.Strategy):
    params = (
        ('commission', 0.001),
    )

    def __init__(self):
        self.sma9 = bt.indicators.SMA(self.data.close, period=9)
        self.sma26 = bt.indicators.SMA(self.data.close, period=26)
        self.sma52 = bt.indicators.SMA(self.data.close, period=52)
        self.crossover = bt.indicators.CrossOver(self.sma9, self.sma26)
        self.order = None

    def notify_order(self, order):
        if order.status in (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            self.order = None

    def _size(self):
        return self.broker.getcash() / (self.data.close[0] * (1.0 + self.p.commission))

    def next(self):
        if self.order:
            return

        close = self.data.close[0]
        cross = self.crossover[0]

        if self.position.size > 0:
            if cross < 0 or close < min(self.sma26[0], self.sma52[0]):
                self.order = self.close()
            return

        if self.position.size < 0:
            if cross > 0 or close > max(self.sma26[0], self.sma52[0]):
                self.order = self.close()
            return

        if (cross > 0 and close > self.sma26[0] and close > self.sma52[0]
                and self.sma26[0] > self.sma52[0]):
            self.order = self.buy(size=self._size())
        elif (cross < 0 and close < self.sma26[0] and close < self.sma52[0]
              and self.sma26[0] < self.sma52[0]):
            self.order = self.sell(size=self._size())
