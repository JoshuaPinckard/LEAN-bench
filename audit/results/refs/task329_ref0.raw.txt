import backtrader as bt


class ATRVolatilityFadeStrategy(bt.Strategy):
    params = (
        ('atr_period', 14),
        ('atr_sma_period', 20),
        ('atr_std_period', 20),
        ('close_sma_period', 20),
        ('stop_loss_pct', 0.02),
    )

    def __init__(self):
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.atr_sma = bt.indicators.SMA(self.atr, period=self.p.atr_sma_period)
        self.atr_std = bt.indicators.StandardDeviation(self.atr, period=self.p.atr_std_period)
        self.close_sma = bt.indicators.SMA(self.data.close, period=self.p.close_sma_period)
        self.atr_cross = bt.indicators.CrossOver(self.atr, self.atr_sma)

        self.order = None
        self.stop_order = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order == self.stop_order:
            if order.status == order.Completed:
                self.stop_order = None
                self.order = None
            elif order.status in [order.Canceled, order.Margin, order.Rejected]:
                self.stop_order = None
            return

        if order.status == order.Completed:
            if self.position:
                price = order.executed.price
                if order.isbuy():
                    stop_price = price * (1 - self.p.stop_loss_pct)
                    self.stop_order = self.sell(exectype=bt.Order.Stop, price=stop_price)
                else:
                    stop_price = price * (1 + self.p.stop_loss_pct)
                    self.stop_order = self.buy(exectype=bt.Order.Stop, price=stop_price)
            else:
                if self.stop_order is not None:
                    self.cancel(self.stop_order)
            self.order = None
        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        if self.order:
            return

        if not self.position:
            upper_band = self.atr_sma[0] + 2 * self.atr_std[0]
            lower_band = self.atr_sma[0] - 2 * self.atr_std[0]

            if self.atr[0] < lower_band and self.data.close[0] < self.close_sma[0]:
                self.order = self.buy()
            elif self.atr[0] > upper_band and self.data.close[0] > self.close_sma[0]:
                self.order = self.sell()
        else:
            if self.position.size > 0 and self.atr_cross[0] > 0:
                self.order = self.close()
            elif self.position.size < 0 and self.atr_cross[0] < 0:
                self.order = self.close()
