import backtrader as bt


class CupAndHandleStrategy(bt.Strategy):
    params = (
        ("volume_period", 50),
        ("pattern_min_len", 250),
        ("risk_pct", 0.01),
    )

    def __init__(self):
        self.vol_sma = bt.indicators.SMA(
            self.data.volume, period=self.p.volume_period
        )
        self.order = None
        self.stop_loss = None

    def notify_order(self, order):
        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            self.order = None

    def _detect_cup_and_handle(self):
        if len(self) < self.p.pattern_min_len + 1:
            return False, None, None

        highs = [
            self.data.high[-i] for i in range(self.p.pattern_min_len, 0, -1)
        ]
        lows = [self.data.low[-i] for i in range(self.p.pattern_min_len, 0, -1)]

        for h_len in range(10, 41):
            c_len = self.p.pattern_min_len - h_len
            cup_highs = highs[:c_len]
            cup_lows = lows[:c_len]

            left_idx = int(c_len * 0.3)
            right_idx = int(c_len * 0.7)

            if left_idx < 1 or right_idx <= left_idx or right_idx >= c_len:
                continue

            left_rim = max(cup_highs[:left_idx])
            right_rim = max(cup_highs[right_idx:])
            bottom = min(cup_lows[left_idx:right_idx])

            max_rim = max(left_rim, right_rim)
            min_rim = min(left_rim, right_rim)

            if max_rim <= 0:
                continue

            if (max_rim - min_rim) / max_rim > 0.15:
                continue

            if bottom >= min_rim * 0.92:
                continue

            cup_depth = max_rim - bottom
            handle_highs = highs[c_len:]
            handle_lows = lows[c_len:]

            handle_low = min(handle_lows)
            handle_high = max(handle_highs)

            if handle_low <= bottom + 0.5 * cup_depth:
                continue

            if handle_high > max_rim * 1.02:
                continue

            return True, max_rim, handle_low

        return False, None, None

    def next(self):
        if self.order:
            return

        if not self.position:
            is_pattern, resistance, handle_low = self._detect_cup_and_handle()
            if is_pattern:
                if (
                    self.data.close[0] > resistance
                    and self.data.volume[0] > self.vol_sma[0]
                ):
                    entry_price = self.data.close[0]
                    risk_per_share = entry_price - handle_low
                    if risk_per_share > 0:
                        capital = self.broker.getvalue()
                        risk_amount = capital * self.p.risk_pct
                        size = int(risk_amount / risk_per_share)

                        cash = self.broker.getcash()
                        max_shares = int(cash / entry_price)
                        size = min(size, max_shares)

                        if size > 0:
                            self.stop_loss = handle_low
                            self.order = self.buy(size=size)
        else:
            if self.data.close[0] < self.stop_loss:
                self.order = self.close()
