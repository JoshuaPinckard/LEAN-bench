import backtrader as bt


class CupAndHandleStrategy(bt.Strategy):
    params = (
        ('cup_period', 200),          # bars for cup formation
        ('handle_period', 50),        # bars for handle formation (cup+handle >= 250)
        ('vol_avg_period', 50),       # spec states 50 candles for average volume
        ('risk_pct', 0.01),           # 1% risk per trade (spec: 1-2%)
        ('cup_sym_tol', 0.05),        # max fractional difference between left/right rim heights
        ('handle_pullback_max', 0.30),# handle depth must not exceed 30% of cup height
    )

    def __init__(self):
        self.order = None
        self.stop_price = None
        self.vol_avg = bt.indicators.SimpleMovingAverage(
            self.data.volume, period=self.p.vol_avg_period
        )

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def _detect_pattern(self):
        """Return (neckline, handle_low) if a valid Cup-and-Handle is found, else (None, None)."""
        cp = self.p.cup_period
        hp = self.p.handle_period

        # Need cp+hp bars of history plus current bar
        if len(self.data) < cp + hp + 1:
            return None, None

        # Cup: cp bars ending at -(hp+1), ordered oldest-to-newest
        # Index -(cp+hp)+i goes from -(cp+hp) (oldest) to -(hp+1) (most recent cup bar)
        cup_highs = [self.data.high[-(cp + hp) + i] for i in range(cp)]
        cup_lows  = [self.data.low[ -(cp + hp) + i] for i in range(cp)]

        # Divide cup into thirds to locate left rim, bottom, right rim
        third = cp // 3
        left_rim  = max(cup_highs[:third])
        cup_bot   = min(cup_lows[third: 2 * third])
        right_rim = max(cup_highs[2 * third:])

        # U-shape: bottom must be below both rims
        if cup_bot >= left_rim or cup_bot >= right_rim:
            return None, None

        # Rims must be approximately equal in height
        rim_diff = abs(left_rim - right_rim) / max(left_rim, right_rim)
        if rim_diff > self.p.cup_sym_tol:
            return None, None

        neckline   = max(left_rim, right_rim)
        cup_height = neckline - cup_bot

        # Handle: hp bars ending at -1, ordered oldest-to-newest
        # Index -hp+i goes from -hp (oldest) to -1 (bar just before current)
        handle_highs = [self.data.high[-hp + i] for i in range(hp)]
        handle_lows  = [self.data.low[ -hp + i] for i in range(hp)]

        handle_high = max(handle_highs)
        handle_low  = min(handle_lows)

        # Handle must not break meaningfully above resistance (2% tolerance)
        if handle_high > neckline * 1.02:
            return None, None

        # Handle depth must be a modest retracement of the cup
        handle_depth = neckline - handle_low
        if cup_height > 0 and handle_depth / cup_height > self.p.handle_pullback_max:
            return None, None

        return neckline, handle_low

    def next(self):
        if self.order:
            return

        # Exit: close below handle low (stop-loss)
        if self.position:
            if self.data.close[0] < self.stop_price:
                self.order = self.close()
            return

        neckline, handle_low = self._detect_pattern()
        if neckline is None:
            return

        # Entry condition: closing price above resistance (neckline) — satisfies rules 1, 2, and 4
        if self.data.close[0] <= neckline:
            return

        # Entry condition: volume above 50-bar average (rule 3)
        if self.data.volume[0] <= self.vol_avg[0]:
            return

        # Position sizing: risk 1% of capital, stop at handle low
        entry_price    = self.data.close[0]
        risk_per_share = entry_price - handle_low
        if risk_per_share <= 0:
            return

        capital     = self.broker.getvalue()
        risk_amount = capital * self.p.risk_pct
        size        = int(risk_amount / risk_per_share)
        if size <= 0:
            return

        self.stop_price = handle_low
        self.order      = self.buy(size=size)
