import backtrader as bt


class SmaAtrTrendTrailStrategy(bt.Strategy):
    def __init__(self):
        self.sma = bt.indicators.SMA(self.data.close, period=20)
        self.atr = bt.indicators.ATR(self.data, period=14)
        self.order = None
        self.trail_stop = None
        self.highest_close = None
        self.lowest_close = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return
        if order.status == order.Completed:
            if self.position.size > 0 and self.trail_stop is None:
                self.highest_close = self.data.close[0]
                self.trail_stop = self.highest_close - 3 * self.atr[0]
            elif self.position.size < 0 and self.trail_stop is None:
                self.lowest_close = self.data.close[0]
                self.trail_stop = self.lowest_close + 3 * self.atr[0]
            elif self.position.size == 0:
                self.trail_stop = None
                self.highest_close = None
                self.lowest_close = None
        self.order = None

    def next(self):
        if self.order:
            return

        if not self.position:
            cross_up = self.data.close[-1] <= self.sma[-1] and self.data.close[0] > self.sma[0]
            cross_down = self.data.close[-1] >= self.sma[-1] and self.data.close[0] < self.sma[0]
            if cross_up:
                self.order = self.buy()
            elif cross_down:
                self.order = self.sell()
            return

        if self.position.size > 0:
            self.highest_close = max(self.highest_close, self.data.close[0])
            candidate = self.highest_close - 3 * self.atr[0]
            self.trail_stop = max(self.trail_stop, candidate)
            cross_below = self.data.close[-1] >= self.sma[-1] and self.data.close[0] < self.sma[0]
            stop_hit = self.data.low[0] <= self.trail_stop
            if cross_below or stop_hit:
                self.order = self.close()
        elif self.position.size < 0:
            self.lowest_close = min(self.lowest_close, self.data.close[0])
            candidate = self.lowest_close + 3 * self.atr[0]
            self.trail_stop = min(self.trail_stop, candidate)
            cross_above = self.data.close[-1] <= self.sma[-1] and self.data.close[0] > self.sma[0]
            stop_hit = self.data.high[0] >= self.trail_stop
            if cross_above or stop_hit:
                self.order = self.close()
