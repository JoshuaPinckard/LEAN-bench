import backtrader as bt


class PremarketGapStrategy(bt.Strategy):
    def __init__(self):
        self.sma = bt.ind.SMA(self.data.close, period=20)
        self.atr = bt.ind.ATR(self.data, period=14)
        self.atr_highest_5 = bt.ind.Highest(self.atr, period=5)
        self.atr_sma_20 = bt.ind.SMA(self.atr, period=20)
        self.crossover = bt.ind.CrossOver(self.data.close, self.sma)

        self.order = None
        self.entry_price = 0.0
        self.entry_atr = 0.0
        self.sl_price = 0.0
        self.trailing_active = False

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            if self.position:
                self.entry_price = order.executed.price
                self.entry_atr = self.atr[0]
                self.trailing_active = False
                if self.position.size > 0:
                    self.sl_price = self.entry_price - 1.5 * self.entry_atr
                else:
                    self.sl_price = self.entry_price + 1.5 * self.entry_atr

        self.order = None

    def next(self):
        if self.order:
            return

        if self.position:
            if self.position.size > 0:
                if not self.trailing_active:
                    if self.data.high[0] - self.entry_price >= 1.0 * self.entry_atr:
                        self.trailing_active = True

                if self.trailing_active:
                    trail_sl = self.data.high[0] - 1.0 * self.atr[0]
                    self.sl_price = max(self.sl_price, trail_sl)

                if self.data.low[0] <= self.sl_price:
                    self.order = self.close()

            elif self.position.size < 0:
                if not self.trailing_active:
                    if self.entry_price - self.data.low[0] >= 1.0 * self.entry_atr:
                        self.trailing_active = True

                if self.trailing_active:
                    trail_sl = self.data.low[0] + 1.0 * self.atr[0]
                    self.sl_price = min(self.sl_price, trail_sl)

                if self.data.high[0] >= self.sl_price:
                    self.order = self.close()
        else:
            vol_ok = self.atr_highest_5[0] <= 1.5 * self.atr_sma_20[0]
            gap = (self.data.open[0] - self.data.close[-1]) / self.data.close[-1]

            if self.crossover[0] > 0:
                if vol_ok and gap > -0.02 and self.data.close[0] > self.sma[0]:
                    risk_amount = self.broker.getvalue() * 0.015
                    risk_per_share = 1.5 * self.atr[0]
                    if risk_per_share > 0:
                        size = int(risk_amount / risk_per_share)
                        max_size = int(self.broker.getcash() / self.data.close[0])
                        size = min(size, max_size)
                        if size > 0:
                            self.order = self.buy(size=size)

            elif self.crossover[0] < 0:
                if vol_ok and gap < 0.02 and self.data.close[0] < self.sma[0]:
                    risk_amount = self.broker.getvalue() * 0.015
                    risk_per_share = 1.5 * self.atr[0]
                    if risk_per_share > 0:
                        size = int(risk_amount / risk_per_share)
                        max_size = int(self.broker.getcash() / self.data.close[0])
                        size = min(size, max_size)
                        if size > 0:
                            self.order = self.sell(size=size)
