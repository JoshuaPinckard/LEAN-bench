import backtrader as bt


class ORBMeanReversionStrategy(bt.Strategy):
    RISK_PCT = 0.01

    NY_OPEN_MIN = 13 * 60
    ORB_END_MIN = 14 * 60
    ENTRY_CUTOFF_MIN = 16 * 60
    DAY_END_MIN = 22 * 60

    def __init__(self):
        self.current_day = None

        self.orb_ready = False
        self.orb_high = None
        self.orb_low = None
        self.orb_mid = None
        self._orb_h_temp = None
        self._orb_l_temp = None

        self.swept_low = False
        self.swept_high = False

        self.order = None
        self.stop_order = None
        self.limit_order = None

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return
        if order.status in (order.Completed, order.Canceled, order.Margin, order.Rejected):
            if order is self.order:
                self.order = None
            if order is self.stop_order:
                self.stop_order = None
            if order is self.limit_order:
                self.limit_order = None

    def next(self):
        dt = self.data.datetime.datetime(0)
        day = dt.date()
        t_min = dt.hour * 60 + dt.minute

        if self.current_day != day:
            self.current_day = day
            self.orb_ready = False
            self.orb_high = None
            self.orb_low = None
            self.orb_mid = None
            self._orb_h_temp = None
            self._orb_l_temp = None
            self.swept_low = False
            self.swept_high = False

        if not self.orb_ready:
            if self.NY_OPEN_MIN <= t_min < self.ORB_END_MIN:
                h = self.data.high[0]
                l = self.data.low[0]
                if self._orb_h_temp is None:
                    self._orb_h_temp = h
                    self._orb_l_temp = l
                else:
                    if h > self._orb_h_temp:
                        self._orb_h_temp = h
                    if l < self._orb_l_temp:
                        self._orb_l_temp = l
            elif t_min >= self.ORB_END_MIN and self._orb_h_temp is not None:
                self.orb_high = self._orb_h_temp
                self.orb_low = self._orb_l_temp
                self.orb_mid = (self.orb_high + self.orb_low) / 2.0
                self.orb_ready = True

        if self.position and t_min >= self.DAY_END_MIN:
            if self.stop_order is not None and self.stop_order.alive():
                self.cancel(self.stop_order)
            if self.limit_order is not None and self.limit_order.alive():
                self.cancel(self.limit_order)
            self.close()
            return

        if (self.orb_ready and not self.position and self.order is None
                and t_min < self.ENTRY_CUTOFF_MIN):
            close = self.data.close[0]

            if not self.swept_low and close < self.orb_low:
                self.swept_low = True
            elif self.swept_low and close > self.orb_low:
                stop_price = self.orb_low
                target_price = self.orb_mid
                stop_distance = close - stop_price
                if stop_distance > 0:
                    risk_amount = self.broker.getvalue() * self.RISK_PCT
                    size = int(risk_amount / stop_distance)
                    if size > 0:
                        orders = self.buy_bracket(
                            size=size,
                            exectype=bt.Order.Market,
                            stopprice=stop_price,
                            limitprice=target_price,
                        )
                        self.order = orders[0]
                        self.stop_order = orders[1]
                        self.limit_order = orders[2]
                self.swept_low = False
                self.swept_high = False

            if self.order is None:
                if not self.swept_high and close > self.orb_high:
                    self.swept_high = True
                elif self.swept_high and close < self.orb_high:
                    stop_price = self.orb_high
                    target_price = self.orb_mid
                    stop_distance = stop_price - close
                    if stop_distance > 0:
                        risk_amount = self.broker.getvalue() * self.RISK_PCT
                        size = int(risk_amount / stop_distance)
                        if size > 0:
                            orders = self.sell_bracket(
                                size=size,
                                exectype=bt.Order.Market,
                                stopprice=stop_price,
                                limitprice=target_price,
                            )
                            self.order = orders[0]
                            self.stop_order = orders[1]
                            self.limit_order = orders[2]
                    self.swept_low = False
                    self.swept_high = False
