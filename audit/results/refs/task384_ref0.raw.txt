import backtrader as bt


class GapFadeStrategy(bt.Strategy):
    params = (
        ('gap_pct_threshold', 1.0),
    )

    def __init__(self):
        self.prior_close = None
        self.day_open = None
        self.entry_order = None
        self.stop_order = None
        self.exit_order = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order is self.entry_order:
            if order.status == order.Completed:
                if order.issell():
                    stop_price = self.day_open * 1.02
                    self.stop_order = self.buy(exectype=bt.Order.Stop, price=stop_price)
                else:
                    stop_price = self.day_open * 0.98
                    self.stop_order = self.sell(exectype=bt.Order.Stop, price=stop_price)
            self.entry_order = None
        elif order is self.stop_order:
            self.stop_order = None
        elif order is self.exit_order:
            self.exit_order = None

    def next(self):
        dt = self.data.datetime.time(0)

        if dt.hour == 9 and dt.minute == 30:
            self.day_open = self.data.open[0]

            if (not self.position and self.entry_order is None
                    and self.stop_order is None and self.exit_order is None
                    and self.prior_close is not None):
                gap_pct = (self.day_open - self.prior_close) / self.prior_close * 100
                if gap_pct >= self.p.gap_pct_threshold:
                    self.entry_order = self.sell(exectype=bt.Order.Close)
                elif gap_pct <= -self.p.gap_pct_threshold:
                    self.entry_order = self.buy(exectype=bt.Order.Close)

        if dt.hour == 15 and dt.minute == 45:
            self.prior_close = self.data.close[0]

            if self.position:
                if self.stop_order is not None:
                    self.cancel(self.stop_order)
                    self.stop_order = None
                self.exit_order = self.close(exectype=bt.Order.Close)
