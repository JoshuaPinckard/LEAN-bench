import math
import backtrader as bt


class PseudoSpreadMeanReversion(bt.Strategy):
    params = (
        ("sma200_period", 200),
        ("sma50_period", 50),
        ("atr_period", 14),
        ("atr_sma_period", 30),
        ("vol_threshold", 0.05),
        ("volume_sma_period", 20),
        ("liquidity_threshold", 10000000.0),
        ("limit_pct", 0.005),
        ("risk_pct", 0.02),
        ("k", 1.0),
        ("stop_dev_mult", 1.1),
    )

    def __init__(self):
        self.sma200 = bt.ind.SMA(self.data.close, period=self.p.sma200_period)
        self.sma50 = bt.ind.SMA(self.data.close, period=self.p.sma50_period)
        self.atr = bt.ind.ATR(self.data, period=self.p.atr_period)
        self.sma_atr = bt.ind.SMA(self.atr, period=self.p.atr_sma_period)

        self.dollar_volume = self.data.volume * self.data.close
        self.sma_dollar_vol = bt.ind.SMA(
            self.dollar_volume, period=self.p.volume_sma_period
        )

        self.crossover = bt.ind.CrossOver(self.data.close, self.sma50)

        self.order = None
        self.entry_price = None
        self.entry_sma50 = None
        self.d0 = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            if order.isbuy():
                if self.position.size > 0:
                    self.entry_price = order.executed.price
                    self.entry_sma50 = self.sma50[0]
                    self.d0 = self.entry_sma50 - self.entry_price
            elif order.issell():
                if self.position.size < 0:
                    self.entry_price = order.executed.price
                    self.entry_sma50 = self.sma50[0]
                    self.d0 = self.entry_price - self.entry_sma50

        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            self.order = None

    def next(self):
        if self.order:
            return

        if not self.position:
            vol_ok = (
                self.sma_atr[0] / self.data.close[0]
            ) <= self.p.vol_threshold
            liq_ok = self.sma_dollar_vol[0] > self.p.liquidity_threshold

            if not (vol_ok and liq_ok):
                return

            if (
                self.data.close[0] > self.sma200[0]
                and self.data.close[0] <= self.sma50[0] * 0.95
            ):
                if self.atr[0] > 0:
                    equity = self.broker.getvalue()
                    qty = math.floor(
                        (self.p.risk_pct * equity) / (self.p.k * self.atr[0])
                    )
                    if qty > 0:
                        limit_price = self.data.close[0] * (
                            1.0 - self.p.limit_pct
                        )
                        self.order = self.buy(
                            exectype=bt.Order.Limit,
                            price=limit_price,
                            size=qty,
                        )

            elif (
                self.data.close[0] < self.sma200[0]
                and self.data.close[0] >= self.sma50[0] * 1.05
            ):
                if self.atr[0] > 0:
                    equity = self.broker.getvalue()
                    qty = math.floor(
                        (self.p.risk_pct * equity) / (self.p.k * self.atr[0])
                    )
                    if qty > 0:
                        limit_price = self.data.close[0] * (
                            1.0 + self.p.limit_pct
                        )
                        self.order = self.sell(
                            exectype=bt.Order.Limit,
                            price=limit_price,
                            size=qty,
                        )

        elif self.position.size > 0:
            if self.crossover[0] > 0:
                self.order = self.close()
                return

            if self.d0 is not None:
                d = self.sma50[0] - self.data.close[0]
                if d >= self.p.stop_dev_mult * self.d0:
                    self.order = self.close()

        elif self.position.size < 0:
            if self.crossover[0] < 0:
                self.order = self.close()
                return

            if self.d0 is not None:
                d = self.data.close[0] - self.sma50[0]
                if d >= self.p.stop_dev_mult * self.d0:
                    self.order = self.close()
