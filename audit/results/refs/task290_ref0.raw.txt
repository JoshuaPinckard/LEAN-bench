import backtrader as bt


class MeanReversionZScoreStrategy(bt.Strategy):
    params = dict(
        bb_period=252,
        bb_devfactor=2,
        z_threshold=1.5,
        exit_bars=63,
        stop_pct=0.015,
        risk_pct=0.01,
        std_eps=1e-6,
    )

    def __init__(self):
        self.bb = bt.indicators.BollingerBands(
            self.data.close, period=self.p.bb_period, devfactor=self.p.bb_devfactor
        )
        self.crossover = bt.indicators.CrossOver(self.data.close, self.bb.mid)
        self.order = None
        self.stop_order = None
        self.exit_order = None
        self.bar_executed = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order == self.order:
            if order.status == order.Completed:
                self.bar_executed = len(self)
                entry_price = order.executed.price
                size = abs(order.executed.size)
                if order.isbuy():
                    stop_price = entry_price * (1 - self.p.stop_pct)
                    self.stop_order = self.sell(
                        exectype=bt.Order.Stop, price=stop_price, size=size
                    )
                else:
                    stop_price = entry_price * (1 + self.p.stop_pct)
                    self.stop_order = self.buy(
                        exectype=bt.Order.Stop, price=stop_price, size=size
                    )
            self.order = None

        elif order == self.stop_order:
            if order.status == order.Completed:
                self.bar_executed = None
            self.stop_order = None

        elif order == self.exit_order:
            if order.status == order.Completed:
                self.bar_executed = None
            self.exit_order = None

    def next(self):
        if self.order:
            return

        if self.position:
            if self.exit_order:
                return
            exit_by_cross = self.crossover[0] != 0
            exit_by_timer = (
                self.bar_executed is not None
                and (len(self) - self.bar_executed) >= self.p.exit_bars
            )
            if exit_by_cross or exit_by_timer:
                if self.stop_order:
                    self.cancel(self.stop_order)
                self.exit_order = self.close()
            return

        std = (self.bb.top[0] - self.bb.mid[0]) / 2.0
        if std <= self.p.std_eps:
            return

        z = (self.data.close[0] - self.bb.mid[0]) / std

        capital = self.broker.getvalue()
        price = self.data.close[0]
        size = (capital * self.p.risk_pct) / (price * self.p.stop_pct)

        if z <= -self.p.z_threshold:
            self.order = self.buy(size=size)
        elif z >= self.p.z_threshold:
            self.order = self.sell(size=size)
