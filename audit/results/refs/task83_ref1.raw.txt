import backtrader as bt
import datetime


class OpeningRangeBreakoutStrategy(bt.Strategy):
    def __init__(self):
        self.entry_order = None
        self.close_order = None
        self.stop_order = None
        self.target_order = None

        self.opening_high = None
        self.opening_low = None
        self.current_date = None
        self.blocked_long = False
        self.blocked_short = False
        self.force_close_submitted = False

    def _reset_day(self, date):
        self.current_date = date
        self.opening_high = None
        self.opening_low = None
        self.blocked_long = False
        self.blocked_short = False
        self.force_close_submitted = False

    def _cancel_protective_orders(self):
        for order in (self.stop_order, self.target_order):
            if order is not None and order.alive():
                self.cancel(order)

    def next(self):
        dt = self.data.datetime.datetime(0)
        current_time = dt.time()

        if self.current_date != dt.date():
            self._reset_day(dt.date())

        if current_time < datetime.time(9, 30) or current_time > datetime.time(16, 0):
            return

        if datetime.time(9, 30) <= current_time < datetime.time(9, 35):
            if self.opening_high is None:
                self.opening_high = self.data.high[0]
                self.opening_low = self.data.low[0]
            else:
                self.opening_high = max(self.opening_high, self.data.high[0])
                self.opening_low = min(self.opening_low, self.data.low[0])
            return

        if current_time >= datetime.time(16, 0):
            if self.position and self.close_order is None:
                self._cancel_protective_orders()
                self.close_order = self.close()
            return

        if self.opening_high is None or self.entry_order is not None:
            return

        if self.position or self.close_order is not None:
            return

        size = (self.broker.getvalue() * 0.01) / self.data.close[0]

        if self.data.high[0] >= self.opening_high and not self.blocked_long:
            self.entry_order = self.buy(size=size)
        elif self.data.low[0] <= self.opening_low and not self.blocked_short:
            self.entry_order = self.sell(size=size)

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return

        if order.status == order.Completed:
            if order is self.entry_order:
                self.entry_order = None
                entry_price = order.executed.price

                if order.isbuy():
                    self.stop_order = self.sell(
                        exectype=bt.Order.Stop,
                        price=entry_price * 0.995,
                    )
                    self.target_order = self.sell(
                        exectype=bt.Order.Limit,
                        price=entry_price * 1.01,
                        oco=self.stop_order,
                    )
                else:
                    self.stop_order = self.buy(
                        exectype=bt.Order.Stop,
                        price=entry_price * 1.005,
                    )
                    self.target_order = self.buy(
                        exectype=bt.Order.Limit,
                        price=entry_price * 0.99,
                        oco=self.stop_order,
                    )

            elif order is self.stop_order:
                if self.position.size > 0:
                    self.blocked_long = True
                else:
                    self.blocked_short = True
                self.stop_order = None
                self.target_order = None

            elif order is self.target_order:
                self.stop_order = None
                self.target_order = None

            elif order is self.close_order:
                self.close_order = None
                self.stop_order = None
                self.target_order = None

        elif order.status in (order.Canceled, order.Margin, order.Rejected):
            if order is self.entry_order:
                self.entry_order = None
            elif order is self.close_order:
                self.close_order = None
            elif order is self.stop_order:
                self.stop_order = None
            elif order is self.target_order:
                self.target_order = None
