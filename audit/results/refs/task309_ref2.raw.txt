import backtrader as bt


class MeanReversionZScore(bt.Strategy):
    params = (
        ('ma_period', 20),
        ('stddev_period', 20),
        ('entry_threshold', 2.0),
        ('exit_threshold', 0.0),
        ('stop_loss_pct', 0.02),
    )

    def __init__(self):
        self.sma = bt.indicators.SMA(self.data.close, period=self.p.ma_period)
        self.stddev = bt.indicators.StdDev(self.data.close, period=self.p.stddev_period)
        self.order = None
        self.stop_order = None

    def notify_order(self, order):
        if order.status == order.Completed:
            if order == self.order:
                self.order = None
                executed_price = order.executed.price
                if order.isbuy() and self.position.size > 0:
                    stop_price = executed_price * (1.0 - self.p.stop_loss_pct)
                    self.stop_order = self.sell(exectype=bt.Order.Stop, price=stop_price)
                elif order.issell() and self.position.size < 0:
                    stop_price = executed_price * (1.0 + self.p.stop_loss_pct)
                    self.stop_order = self.buy(exectype=bt.Order.Stop, price=stop_price)
            elif order == self.stop_order:
                self.stop_order = None
        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            if order == self.order:
                self.order = None
            elif order == self.stop_order:
                self.stop_order = None

    def next(self):
        if self.order:
            return

        zscore = (self.data.close[0] - self.sma[0]) / self.stddev[0]

        if not self.position:
            if zscore < -self.p.entry_threshold:
                self.order = self.buy()
            elif zscore > self.p.entry_threshold:
                self.order = self.sell()
        elif self.position.size > 0:
            if zscore >= self.p.exit_threshold:
                if self.stop_order:
                    self.cancel(self.stop_order)
                    self.stop_order = None
                self.order = self.close()
        else:
            if zscore <= self.p.exit_threshold:
                if self.stop_order:
                    self.cancel(self.stop_order)
                    self.stop_order = None
                self.order = self.close()
