import backtrader as bt


class ZScoreMeanReversion(bt.Strategy):
    params = (
        ("ma_period", 20),
        ("std_period", 20),
        ("entry_threshold", 2.0),
        ("exit_threshold", 0.0),
        ("stop_loss_pct", 0.02),
    )

    def __init__(self):
        self.sma = bt.indicators.SimpleMovingAverage(
            self.data.close, period=self.params.ma_period
        )
        self.std = bt.indicators.StdDev(
            self.data.close, period=self.params.std_period
        )
        self.zscore = (self.data.close - self.sma) / self.std

        self.order = None
        self.stop_order = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order == self.order:
            self.order = None
            if order.status == order.Completed:
                if self.position.size > 0:
                    stop_price = order.executed.price * (
                        1.0 - self.params.stop_loss_pct
                    )
                    self.stop_order = self.sell(
                        exectype=bt.Order.Stop, price=stop_price
                    )
                elif self.position.size < 0:
                    stop_price = order.executed.price * (
                        1.0 + self.params.stop_loss_pct
                    )
                    self.stop_order = self.buy(
                        exectype=bt.Order.Stop, price=stop_price
                    )
                else:
                    if self.stop_order:
                        self.cancel(self.stop_order)
                        self.stop_order = None

        elif order == self.stop_order:
            if order.status in [
                order.Completed,
                order.Canceled,
                order.Margin,
                order.Rejected,
            ]:
                self.stop_order = None

    def next(self):
        if self.order:
            return

        z = self.zscore[0]

        if not self.position:
            if z < -self.params.entry_threshold:
                self.order = self.buy()
            elif z > self.params.entry_threshold:
                self.order = self.sell()
        else:
            if self.position.size > 0:
                if z >= self.params.exit_threshold:
                    if self.stop_order:
                        self.cancel(self.stop_order)
                        self.stop_order = None
                    self.order = self.close()
            elif self.position.size < 0:
                if z <= -self.params.exit_threshold:
                    if self.stop_order:
                        self.cancel(self.stop_order)
                        self.stop_order = None
                    self.order = self.close()
