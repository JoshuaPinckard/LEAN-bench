import backtrader as bt


class FairValueGapStrategy(bt.Strategy):
    params = (
        ('risk_pct', 0.01),                      # base risk per trade as fraction of equity
        ('bear_size_multiplier', 63.2 / 60.71),  # scale short size vs long per stated fill-rate stats
        ('stop_buffer', 0.10),                   # "a few points" beyond FVG boundary
        ('vol_lookback', 20),
        ('vol_spike_factor', 1.5),
        ('wick_body_mult', 2.0),
        ('long_wick_frac', 0.5),
    )

    def __init__(self):
        self.order = None
        self.fvgs = []
        self.current_session_date = None
        self.session_high = None
        self.session_low = None
        self.prev_session_high = None
        self.prev_session_low = None
        # minutes-since-midnight bounds for the 9:30-16:00 ET NY session
        self.ny_open_min = 9 * 60 + 30
        self.ny_close_min = 16 * 60

    def _volume_spike(self):
        lb = self.p.vol_lookback
        if len(self) <= lb:
            return False
        total = 0.0
        for i in range(1, lb + 1):
            total += self.data.volume[-i]
        avg_vol = total / lb
        return avg_vol > 0 and self.data.volume[0] >= self.p.vol_spike_factor * avg_vol

    def _bullish_signal(self):
        o, h, l, c = self.data.open[0], self.data.high[0], self.data.low[0], self.data.close[0]
        body = abs(c - o)
        upper_wick = h - max(o, c)
        lower_wick = min(o, c) - l
        rng = h - l
        hammer = rng > 0 and lower_wick >= self.p.wick_body_mult * body and upper_wick <= body
        long_lower_wick = rng > 0 and lower_wick >= self.p.long_wick_frac * rng
        return hammer or long_lower_wick or self._volume_spike()

    def _bearish_signal(self):
        o, h, l, c = self.data.open[0], self.data.high[0], self.data.low[0], self.data.close[0]
        body = abs(c - o)
        upper_wick = h - max(o, c)
        lower_wick = min(o, c) - l
        rng = h - l
        shooting_star = rng > 0 and upper_wick >= self.p.wick_body_mult * body and lower_wick <= body
        long_upper_wick = rng > 0 and upper_wick >= self.p.long_wick_frac * rng
        return shooting_star or long_upper_wick or self._volume_spike()

    def _calc_size(self, stop_dist, fvg_type):
        equity = self.broker.getvalue()
        risk_pct = self.p.risk_pct
        if fvg_type == 'bear':
            risk_pct *= self.p.bear_size_multiplier
        size = int((equity * risk_pct) / stop_dist)
        if size < 1:
            return 0
        price = self.data.close[0]
        if price <= 0:
            return 0
        max_affordable = int(self.broker.getcash() / price)
        return max(0, min(size, max_affordable))

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return
        if self.order is not None and order.ref == self.order.ref:
            if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
                self.order = None

    def next(self):
        dt0 = self.data.datetime.datetime(0)
        bar_date = dt0.date()
        bar_min = dt0.hour * 60 + dt0.minute
        in_session = self.ny_open_min <= bar_min <= self.ny_close_min

        close = self.data.close[0]
        high = self.data.high[0]
        low = self.data.low[0]

        if self.current_session_date is None or bar_date != self.current_session_date:
            if self.current_session_date is not None:
                self.prev_session_high = self.session_high
                self.prev_session_low = self.session_low
            self.current_session_date = bar_date
            self.session_high = high
            self.session_low = low
        else:
            if high > self.session_high:
                self.session_high = high
            if low < self.session_low:
                self.session_low = low

        if in_session and len(self) >= 3:
            dtm1 = self.data.datetime.datetime(-1)
            dtm2 = self.data.datetime.datetime(-2)
            if dtm1.date() == bar_date and dtm2.date() == bar_date:
                m1 = dtm1.hour * 60 + dtm1.minute
                m2 = dtm2.hour * 60 + dtm2.minute
                if self.ny_open_min <= m1 <= self.ny_close_min and self.ny_open_min <= m2 <= self.ny_close_min:
                    c1_high, c1_low = self.data.high[-2], self.data.low[-2]
                    c3_high, c3_low = self.data.high[0], self.data.low[0]
                    if c1_high < c3_low:
                        self.fvgs.append({'type': 'bull', 'top': c3_low, 'bottom': c1_high,
                                           'date': bar_date, 'mitigated': False})
                    elif c1_low > c3_high:
                        self.fvgs.append({'type': 'bear', 'top': c1_low, 'bottom': c3_high,
                                           'date': bar_date, 'mitigated': False})

        if in_session and self.order is None and not self.position:
            for fvg in self.fvgs:
                if fvg['mitigated'] or fvg['date'] != bar_date:
                    continue
                if fvg['type'] == 'bull':
                    if low <= fvg['top'] and self._bullish_signal():
                        tp = self.prev_session_high
                        if tp is not None and tp > close:
                            stop_price = fvg['bottom'] - self.p.stop_buffer
                            stop_dist = close - stop_price
                            if stop_dist > 0:
                                size = self._calc_size(stop_dist, 'bull')
                                if size >= 1:
                                    orders = self.buy_bracket(size=size, exectype=bt.Order.Market,
                                                               stopprice=stop_price, limitprice=tp)
                                    self.order = orders[0]
                                    break
                else:
                    if high >= fvg['bottom'] and self._bearish_signal():
                        tp = self.prev_session_low
                        if tp is not None and tp < close:
                            stop_price = fvg['top'] + self.p.stop_buffer
                            stop_dist = stop_price - close
                            if stop_dist > 0:
                                size = self._calc_size(stop_dist, 'bear')
                                if size >= 1:
                                    orders = self.sell_bracket(size=size, exectype=bt.Order.Market,
                                                                stopprice=stop_price, limitprice=tp)
                                    self.order = orders[0]
                                    break

        if in_session:
            for fvg in self.fvgs:
                if fvg['mitigated']:
                    continue
                if fvg['type'] == 'bull' and close <= fvg['top']:
                    fvg['mitigated'] = True
                elif fvg['type'] == 'bear' and close >= fvg['bottom']:
                    fvg['mitigated'] = True
