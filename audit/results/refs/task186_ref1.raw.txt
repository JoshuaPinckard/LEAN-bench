import backtrader as bt


class SentimentProxyWeakSignalAmplifier(bt.Strategy):
    params = (
        ("rsi_period", 14),
        ("atr_period", 14),
        ("volume_sma_period", 20),
        ("atr_sma_period", 100),
        ("bb_period", 20),
        ("bb_devfactor", 2.0),
    )

    def __init__(self):
        self.rsi = bt.indicators.RSI(self.data.close, period=self.p.rsi_period)
        self.macd = bt.indicators.MACD(
            self.data.close,
            period_me1=12,
            period_me2=26,
            period_signal=9,
        )
        self.macd_cross = bt.indicators.CrossOver(
            self.macd.macd, self.macd.signal
        )
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.atr_sma = bt.indicators.SMA(
            self.atr, period=self.p.atr_sma_period
        )
        self.volume_sma = bt.indicators.SMA(
            self.data.volume, period=self.p.volume_sma_period
        )
        self.bollinger = bt.indicators.BollingerBands(
            self.data.close,
            period=self.p.bb_period,
            devfactor=self.p.bb_devfactor,
        )

        self.entry_order = None
        self.stop_order = None
        self.target_order = None
        self.exit_order = None
        self.rsi_extreme = None

    def _size(self):
        return (self.broker.getvalue() * 0.01) / self.data.close[0]

    def _cancel_protection(self):
        if self.stop_order is not None:
            self.cancel(self.stop_order)
        if self.target_order is not None:
            self.cancel(self.target_order)

    def next(self):
        if self.position:
            if self.exit_order is not None:
                return

            if self.position.size > 0:
                if self.rsi_extreme is None or self.rsi[0] > self.rsi_extreme:
                    self.rsi_extreme = self.rsi[0]
                elif self.rsi[0] <= self.rsi_extreme - 10:
                    self._cancel_protection()
                    self.exit_order = self.close()
            else:
                if self.rsi_extreme is None or self.rsi[0] < self.rsi_extreme:
                    self.rsi_extreme = self.rsi[0]
                elif self.rsi[0] >= self.rsi_extreme + 10:
                    self._cancel_protection()
                    self.exit_order = self.close()
            return

        if self.entry_order is not None:
            return

        high_volatility = self.atr[0] > 1.5 * self.atr_sma[0]
        delta_rsi = self.rsi[0] - self.rsi[-3]
        band_width = self.bollinger.top[0] - self.bollinger.bot[0]

        no_upper_break = all(
            self.data.close[-i] <= self.bollinger.top[-i] for i in range(3)
        )
        no_lower_break = all(
            self.data.close[-i] >= self.bollinger.bot[-i] for i in range(3)
        )

        long_price_unreacted = (
            no_upper_break
            and self.data.close[0]
            <= self.bollinger.mid[0] + 0.5 * band_width
        )
        short_price_unreacted = (
            no_lower_break
            and self.data.close[0]
            >= self.bollinger.mid[0] - 0.5 * band_width
        )

        long_general = (
            delta_rsi >= (10 if high_volatility else 15)
            and long_price_unreacted
            and (
                not high_volatility
                or (
                    self.macd_cross[0] == 1
                    and self.data.volume[0] > 1.3 * self.volume_sma[0]
                )
            )
        )
        short_general = (
            delta_rsi <= (-10 if high_volatility else -15)
            and short_price_unreacted
            and (
                not high_volatility
                or (
                    self.macd_cross[0] == -1
                    and self.data.volume[0] > 1.3 * self.volume_sma[0]
                )
            )
        )

        long_policy = (
            self.data.open[0] > self.data.high[-1] + 0.5 * self.atr[0]
            and self.data.volume[0] > 1.5 * self.volume_sma[0]
        )
        short_policy = (
            self.data.open[0] < self.data.low[-1] - 0.5 * self.atr[0]
            and self.data.volume[0] > 1.5 * self.volume_sma[0]
        )

        size = self._size()
        if long_general or long_policy:
            self.entry_order = self.buy(size=size)
        elif short_general or short_policy:
            self.entry_order = self.sell(size=size)

    def notify_order(self, order):
        if order.status in (
            order.Submitted,
            order.Accepted,
        ):
            return

        if order.status == order.Completed:
            if order is self.entry_order:
                entry_price = order.executed.price
                self.rsi_extreme = self.rsi[0]

                if order.isbuy():
                    self.stop_order = self.sell(
                        size=order.executed.size,
                        exectype=bt.Order.Stop,
                        price=entry_price * 0.97,
                    )
                    self.target_order = self.sell(
                        size=order.executed.size,
                        exectype=bt.Order.Limit,
                        price=entry_price * 1.05,
                        oco=self.stop_order,
                    )
                else:
                    size = abs(order.executed.size)
                    self.stop_order = self.buy(
                        size=size,
                        exectype=bt.Order.Stop,
                        price=entry_price * 1.03,
                    )
                    self.target_order = self.buy(
                        size=size,
                        exectype=bt.Order.Limit,
                        price=entry_price * 0.95,
                        oco=self.stop_order,
                    )
                self.entry_order = None
                return

            if order is self.stop_order or order is self.target_order:
                self._cancel_protection()
                self.stop_order = None
                self.target_order = None
                self.rsi_extreme = None
                return

            if order is self.exit_order:
                self._cancel_protection()
                self.stop_order = None
                self.target_order = None
                self.exit_order = None
                self.rsi_extreme = None
                return

        if order.status in (
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            if order is self.entry_order:
                self.entry_order = None
            elif order is self.stop_order:
                self.stop_order = None
            elif order is self.target_order:
                self.target_order = None
            elif order is self.exit_order:
                self.exit_order = None
