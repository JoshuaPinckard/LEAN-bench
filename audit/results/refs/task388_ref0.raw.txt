import backtrader as bt


class WilliamsRMomentumATR(bt.Strategy):
    params = dict(
        wr_period=14,          # Williams %R period
        mom_period=14,         # Momentum period
        atr_period=14,         # ATR period
        wr_cross_long=-80,     # %R level crossed UP through for a long entry
        wr_cross_short=-20,    # %R level crossed DOWN through for a short entry
        atr_mult=2.0,          # adjustable ATR multiple for the protective stop
        trailing=False,        # False = static stop from entry (default);
                               # True  = stop updated each bar (trailing/volatility stop)
    )

    def __init__(self):
        # Built-in Backtrader indicators, exactly as specified
        self.wr = bt.indicators.WilliamsR(self.data, period=self.p.wr_period)
        self.mom = bt.indicators.Momentum(self.data, period=self.p.mom_period)
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)

        self.order = None        # pending-order guard
        self.stop_price = None    # active protective stop level

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return
        if order.status == order.Completed:
            if self.position:
                # A fresh entry just filled -> set the protective stop from entry price.
                entry = order.executed.price
                if self.position.size > 0:      # long
                    self.stop_price = entry - self.p.atr_mult * self.atr[0]
                else:                           # short
                    self.stop_price = entry + self.p.atr_mult * self.atr[0]
            else:
                # A close just filled -> flat, no stop.
                self.stop_price = None
        # Completed / Canceled / Margin / Rejected -> release the guard
        self.order = None

    def next(self):
        # Entry signals (evaluated every bar; also used as opposite-signal exits)
        long_signal = (self.wr[-1] <= self.p.wr_cross_long and
                       self.wr[0] > self.p.wr_cross_long and
                       self.mom[0] > 0)
        short_signal = (self.wr[-1] >= self.p.wr_cross_short and
                        self.wr[0] < self.p.wr_cross_short and
                        self.mom[0] < 0)

        if self.order:
            return

        # ---- Flat: look for an entry ----
        if not self.position:
            if long_signal:
                self.order = self.buy()
            elif short_signal:
                self.order = self.sell()
            return

        # ---- In a position: protective stop first, then opposite-signal exit ----
        if self.position.size > 0:  # long
            if self.p.trailing:
                # trailing volatility stop: ratchet up, never down
                cand = self.data.close[0] - self.p.atr_mult * self.atr[0]
                self.stop_price = cand if self.stop_price is None else max(self.stop_price, cand)
            if self.stop_price is not None and self.data.low[0] <= self.stop_price:
                self.order = self.close()
            elif short_signal:
                self.order = self.close()
        else:                       # short
            if self.p.trailing:
                # trailing volatility stop: ratchet down, never up
                cand = self.data.close[0] + self.p.atr_mult * self.atr[0]
                self.stop_price = cand if self.stop_price is None else min(self.stop_price, cand)
            if self.stop_price is not None and self.data.high[0] >= self.stop_price:
                self.order = self.close()
            elif long_signal:
                self.order = self.close()
