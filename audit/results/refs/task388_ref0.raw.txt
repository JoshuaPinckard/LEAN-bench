import backtrader as bt


class WilliamsMomentumATRStrategy(bt.Strategy):
    params = (
        ('wr_period', 14),
        ('mom_period', 14),
        ('atr_period', 14),
        ('atr_mult', 2.0),
        ('trailing_stop', False),
    )

    def __init__(self):
        self.williams = bt.indicators.WilliamsR(self.data, period=self.p.wr_period)
        self.momentum = bt.indicators.Momentum(self.data, period=self.p.mom_period)
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.order = None
        self.stop_price = None

    def notify_order(self, order):
        if order.status == order.Completed:
            if order.isbuy():
                self.stop_price = order.executed.price - self.p.atr_mult * self.atr[0]
            elif order.issell():
                self.stop_price = order.executed.price + self.p.atr_mult * self.atr[0]

        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        if self.order:
            return

        long_signal = self.williams[-1] <= -80 and self.williams[0] > -80 and self.momentum[0] > 0
        short_signal = self.williams[-1] >= -20 and self.williams[0] < -20 and self.momentum[0] < 0

        if not self.position:
            if long_signal:
                self.order = self.buy()
            elif short_signal:
                self.order = self.sell()
            return

        if self.position.size > 0:
            if short_signal:
                self.order = self.close()
                return
            if self.data.close[0] <= self.stop_price:
                self.order = self.close()
                return
            if self.p.trailing_stop:
                new_stop = self.data.close[0] - self.p.atr_mult * self.atr[0]
                if new_stop > self.stop_price:
                    self.stop_price = new_stop

        elif self.position.size < 0:
            if long_signal:
                self.order = self.close()
                return
            if self.data.close[0] >= self.stop_price:
                self.order = self.close()
                return
            if self.p.trailing_stop:
                new_stop = self.data.close[0] + self.p.atr_mult * self.atr[0]
                if new_stop < self.stop_price:
                    self.stop_price = new_stop
