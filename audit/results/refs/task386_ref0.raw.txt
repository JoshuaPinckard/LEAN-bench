import backtrader as bt


class BreakoutStrategy(bt.Strategy):
    params = dict(
        period=20,
        stop_pct=0.02,
    )

    def __init__(self):
        self.highest_high = bt.indicators.Highest(self.data.high(-1), period=self.p.period)
        self.lowest_low = bt.indicators.Lowest(self.data.low(-1), period=self.p.period)
        self.sma = bt.indicators.SimpleMovingAverage(self.data.close, period=self.p.period)
        self.order = None
        self.entry_price = None

    def notify_order(self, order):
        if order.status == order.Completed:
            if self.position:
                self.entry_price = order.executed.price
            else:
                self.entry_price = None
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        if self.order:
            return

        close = self.data.close[0]
        long_signal = close > self.highest_high[0] and close > self.sma[0]
        short_signal = close < self.lowest_low[0] and close < self.sma[0]

        if not self.position:
            if long_signal:
                self.order = self.buy()
            elif short_signal:
                self.order = self.sell()
            return

        if self.position.size > 0:
            stop_hit = close <= self.entry_price * (1 - self.p.stop_pct)
            if short_signal or stop_hit:
                self.order = self.close()
        elif self.position.size < 0:
            stop_hit = close >= self.entry_price * (1 + self.p.stop_pct)
            if long_signal or stop_hit:
                self.order = self.close()
