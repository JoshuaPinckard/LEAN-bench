import backtrader as bt


class IchimokuSmaApproxStrategy(bt.Strategy):
    params = (
        ('tenkan_period', 9),
        ('kijun_period', 26),
        ('spanb_period', 52),
        ('stake', 10),
    )

    def __init__(self):
        self.tenkan = bt.indicators.SMA(self.data.close, period=self.p.tenkan_period)
        self.kijun = bt.indicators.SMA(self.data.close, period=self.p.kijun_period)
        self.spanb = bt.indicators.SMA(self.data.close, period=self.p.spanb_period)
        self.crossover = bt.indicators.CrossOver(self.tenkan, self.kijun)
        self.order = None

    def notify_order(self, order):
        if order.status in (order.Completed, order.Canceled, order.Margin, order.Rejected):
            self.order = None

    def next(self):
        if self.order:
            return

        if self.position:
            stop_price = self.kijun[0]
            if self.position.size > 0:
                if self.crossover[0] < 0 or self.data.low[0] <= stop_price:
                    self.order = self.close()
            else:
                if self.crossover[0] > 0 or self.data.high[0] >= stop_price:
                    self.order = self.close()
        else:
            spana = (self.tenkan[0] + self.kijun[0]) / 2.0
            cloud_top = max(spana, self.spanb[0])
            cloud_bottom = min(spana, self.spanb[0])
            if self.crossover[0] > 0 and self.data.close[0] > cloud_top:
                self.order = self.buy(size=self.p.stake)
            elif self.crossover[0] < 0 and self.data.close[0] < cloud_bottom:
                self.order = self.sell(size=self.p.stake)
