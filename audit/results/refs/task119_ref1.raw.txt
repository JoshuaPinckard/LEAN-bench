import backtrader as bt
import math


class SupplyDemandSwingStrategy(bt.Strategy):
    params = ()

    def __init__(self):
        self.order = None
        self._pending_kind = None
        self._pending_context = None
        self.active_trade = None

        self._four_hour = None
        self._daily = None
        self._four_hour_history = []
        self._daily_history = []
        self.demand_zones = []
        self.supply_zones = []
        self.daily_trend = None
        self.previous_close = None

    def _new_bar(self, dt):
        return {
            "open": float(self.data.open[0]),
            "high": float(self.data.high[0]),
            "low": float(self.data.low[0]),
            "close": float(self.data.close[0]),
            "volume": float(self.data.volume[0]),
            "dt": dt,
        }

    def _update_bar(self, bar):
        bar["high"] = max(bar["high"], float(self.data.high[0]))
        bar["low"] = min(bar["low"], float(self.data.low[0]))
        bar["close"] = float(self.data.close[0])
        bar["volume"] += float(self.data.volume[0])

    def _add_zones(self, history, timeframe):
        if len(history) < 4:
            return

        consolidation = history[-4:-1]
        impulse = history[-1]
        zone_low = min(bar["low"] for bar in consolidation)
        zone_high = max(bar["high"] for bar in consolidation)
        zone_height = zone_high - zone_low

        if zone_height < 0.005 * impulse["close"]:
            return

        if (impulse["close"] > zone_high and
                impulse["close"] - impulse["open"] >= zone_height):
            self.demand_zones.append({
                "lower": zone_low,
                "upper": zone_high,
                "timeframe": timeframe,
                "created": impulse["dt"],
            })

        if (impulse["close"] < zone_low and
                impulse["open"] - impulse["close"] >= zone_height):
            self.supply_zones.append({
                "lower": zone_low,
                "upper": zone_high,
                "timeframe": timeframe,
                "created": impulse["dt"],
            })

    def _complete_four_hour(self):
        self._four_hour_history.append(self._four_hour)
        self._add_zones(self._four_hour_history, "4H")

    def _complete_daily(self):
        self._daily_history.append(self._daily)
        self._add_zones(self._daily_history, "1D")

        if len(self._daily_history) >= 2:
            if self._daily_history[-1]["close"] > self._daily_history[-2]["close"]:
                self.daily_trend = "up"
            elif self._daily_history[-1]["close"] < self._daily_history[-2]["close"]:
                self.daily_trend = "down"
            else:
                self.daily_trend = None

    def _update_timeframes(self, dt):
        four_hour_key = (dt.date(), dt.hour // 4)
        daily_key = dt.date()

        if self._four_hour is None:
            self._four_hour = self._new_bar(dt)
            self._four_hour_key = four_hour_key
        elif four_hour_key != self._four_hour_key:
            self._complete_four_hour()
            self._four_hour = self._new_bar(dt)
            self._four_hour_key = four_hour_key
        else:
            self._update_bar(self._four_hour)

        if self._daily is None:
            self._daily = self._new_bar(dt)
            self._daily_key = daily_key
        elif daily_key != self._daily_key:
            self._complete_daily()
            self._daily = self._new_bar(dt)
            self._daily_key = daily_key
        else:
            self._update_bar(self._daily)

    def _nearest_higher_supply(self, price):
        candidates = [z for z in self.supply_zones if z["lower"] > price]
        return min(candidates, key=lambda z: z["lower"]) if candidates else None

    def _nearest_lower_demand(self, price):
        candidates = [z for z in self.demand_zones if z["upper"] < price]
        return max(candidates, key=lambda z: z["upper"]) if candidates else None

    def _enter(self, side, zone):
        price = float(self.data.close[0])
        stop = zone["lower"] * 0.995 if side == "long" else zone["upper"] * 1.005
        per_unit_risk = abs(price - stop)

        if per_unit_risk <= 0:
            return

        size = math.floor((self.broker.getvalue() * 0.01) / per_unit_risk)
        if size <= 0:
            return

        target_zone = (
            self._nearest_higher_supply(price)
            if side == "long"
            else self._nearest_lower_demand(price)
        )

        self._pending_kind = "entry"
        self._pending_context = {
            "side": side,
            "zone": zone,
            "stop": stop,
            "target_zone": target_zone,
        }
        self.order = self.buy(size=size) if side == "long" else self.sell(size=size)

    def next(self):
        dt = self.data.datetime.datetime(0)
        self._update_timeframes(dt)

        close = float(self.data.close[0])
        high = float(self.data.high[0])
        low = float(self.data.low[0])

        if self.position and self.order is None and self.active_trade is not None:
            trade = self.active_trade

            if self.position.size > 0:
                if low <= trade["stop"] or high >= trade["target"]:
                    self._pending_kind = "exit"
                    self._pending_context = None
                    self.order = self.close()
            else:
                if high >= trade["stop"] or low <= trade["target"]:
                    self._pending_kind = "exit"
                    self._pending_context = None
                    self.order = self.close()

        if self.position or self.order is not None or self.previous_close is None:
            self.previous_close = close
            return

        if self.daily_trend == "up":
            candidates = [
                zone for zone in self.demand_zones
                if zone["lower"] <= close <= zone["upper"]
                and low <= (zone["lower"] + zone["upper"]) / 2.0
            ]
            if candidates and close > self.previous_close:
                zone = max(candidates, key=lambda z: z["lower"])
                self._enter("long", zone)

        elif self.daily_trend == "down":
            candidates = [
                zone for zone in self.supply_zones
                if zone["lower"] <= close <= zone["upper"]
                and high >= (zone["lower"] + zone["upper"]) / 2.0
            ]
            if candidates and close < self.previous_close:
                zone = min(candidates, key=lambda z: z["upper"])
                self._enter("short", zone)

        self.previous_close = close

    def notify_order(self, order):
        if order.status in (bt.Order.Submitted, bt.Order.Accepted):
            return

        kind = self._pending_kind
        context = self._pending_context

        if order.status == bt.Order.Completed:
            self.order = None
            self._pending_kind = None
            self._pending_context = None

            if kind == "entry":
                entry = order.executed.price
                if context["target_zone"] is not None:
                    target = (
                        context["target_zone"]["lower"]
                        if context["side"] == "long"
                        else context["target_zone"]["upper"]
                    )
                else:
                    risk = abs(entry - context["stop"])
                    target = (
                        entry + 2.0 * risk
                        if context["side"] == "long"
                        else entry - 2.0 * risk
                    )

                self.active_trade = {
                    "side": context["side"],
                    "stop": context["stop"],
                    "target": target,
                }
            elif kind == "exit":
                self.active_trade = None

        elif order.status in (
            bt.Order.Canceled,
            bt.Order.Margin,
            bt.Order.Rejected,
            bt.Order.Expired,
        ):
            self.order = None
            if kind == "entry":
                self._pending_context = None
            self._pending_kind = None
