import backtrader as bt


class GapTradingStrategy(bt.Strategy):
    params = (
        ("gap_avg_period", 20),
        ("h4_sma_period", 20),
        ("small_gap_pct", 0.70),
        ("big_gap_pct", 1.50),
    )

    def __init__(self):
        self.order = None
        self.prev_date = None
        self.session_gaps = []
        self.h4_closes = []
        self.h1_bar_count = 0
        self.target_tp = None
        self.active_scenario = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return
        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            self.order = None
            if not self.position:
                self.target_tp = None
                self.active_scenario = None

    def next(self):
        self.h1_bar_count += 1

        current_date = self.data.datetime.date(0)
        is_new_session = False
        if self.prev_date is not None and current_date != self.prev_date:
            is_new_session = True

        h4_sma = None
        if len(self.h4_closes) >= self.p.h4_sma_period:
            h4_sma = (
                sum(self.h4_closes[-self.p.h4_sma_period :])
                / self.p.h4_sma_period
            )

        if self.position:
            if self.order is None:
                if self.active_scenario == 1 and self.target_tp is not None:
                    if self.position.size > 0:
                        if self.data.low[0] <= self.target_tp:
                            self.order = self.close()
                    elif self.position.size < 0:
                        if self.data.high[0] >= self.target_tp:
                            self.order = self.close()

                if self.position and self.order is None and h4_sma is not None:
                    if (
                        self.position.size > 0
                        and self.data.close[0] < h4_sma
                    ):
                        self.order = self.close()
                    elif (
                        self.position.size < 0
                        and self.data.close[0] > h4_sma
                    ):
                        self.order = self.close()

        if is_new_session:
            prev_close = self.data.close[-1]
            current_open = self.data.open[0]
            gap = current_open - prev_close
            gap_size = abs(gap)

            if (
                not self.position
                and self.order is None
                and len(self.session_gaps) >= self.p.gap_avg_period
                and h4_sma is not None
            ):
                avg_gap = (
                    sum(self.session_gaps[-self.p.gap_avg_period :])
                    / self.p.gap_avg_period
                )

                if gap_size <= self.p.small_gap_pct * avg_gap:
                    if gap > 0 and current_open > h4_sma:
                        self.order = self.buy()
                        self.active_scenario = 1
                        self.target_tp = prev_close
                    elif gap < 0 and current_open < h4_sma:
                        self.order = self.sell()
                        self.active_scenario = 1
                        self.target_tp = prev_close

                elif gap_size > self.p.big_gap_pct * avg_gap:
                    if gap > 0 and current_open > h4_sma:
                        self.order = self.buy()
                        self.active_scenario = 2
                        self.target_tp = None
                    elif gap < 0 and current_open < h4_sma:
                        self.order = self.sell()
                        self.active_scenario = 2
                        self.target_tp = None

            self.session_gaps.append(gap_size)

        if self.h1_bar_count % 4 == 0:
            self.h4_closes.append(self.data.close[0])

        self.prev_date = current_date
