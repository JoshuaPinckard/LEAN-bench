import backtrader as bt


class LongTermBreakoutStrategy(bt.Strategy):
    params = (
        ("risk_fraction", 0.01),
        ("stop_atr_multiple", 3.0),
        ("take_profit_r", 3.0),
        ("breakeven_r", 1.5),
        ("min_holding_days", 60),
        ("cooldown_days", 20),
    )

    def __init__(self):
        self.sma200 = bt.indicators.SMA(self.data.close, period=200)
        self.bbands = bt.indicators.BollingerBands(
            self.data.close, period=20, devfactor=2
        )
        self.atr = bt.indicators.ATR(self.data, period=14)

        self.band_width = self.bbands.top - self.bbands.bot
        self.band_width_sma = bt.indicators.SMA(self.band_width, period=100)
        self.atr_sma = bt.indicators.SMA(self.atr, period=100)

        self.upper_cross = bt.indicators.CrossOver(
            self.data.close, self.bbands.top
        )
        self.lower_cross = bt.indicators.CrossOver(
            self.data.close, self.bbands.bot
        )

        self.order = None
        self.stop_order = None
        self.take_order = None
        self.pending_stop_price = None

        self.entry_price = None
        self.entry_atr = None
        self.entry_bar = None
        self.entry_direction = None
        self.risk_distance = None
        self.stop_price = None
        self.take_price = None
        self.peak_close = None
        self.trough_close = None
        self.breakeven_moved = False
        self.cooldown_exit_bar = None

    def _consolidating(self):
        return all(
            self.band_width[-offset] < self.band_width_sma[-offset] * 0.8
            for offset in range(10)
        )

    def _clear_trade_state(self):
        self.stop_order = None
        self.take_order = None
        self.pending_stop_price = None
        self.entry_price = None
        self.entry_atr = None
        self.entry_bar = None
        self.entry_direction = None
        self.risk_distance = None
        self.stop_price = None
        self.take_price = None
        self.peak_close = None
        self.trough_close = None
        self.breakeven_moved = False
        self.cooldown_exit_bar = len(self)

    def _place_stop(self, price):
        if self.entry_direction == 1:
            self.stop_order = self.sell(
                size=abs(self.position.size),
                exectype=bt.Order.Stop,
                price=price,
                oco=self.take_order,
            )
        else:
            self.stop_order = self.buy(
                size=abs(self.position.size),
                exectype=bt.Order.Stop,
                price=price,
                oco=self.take_order,
            )
        self.stop_price = price

    def _place_exit_orders(self):
        if self.entry_direction == 1:
            self.stop_order = self.sell(
                size=abs(self.position.size),
                exectype=bt.Order.Stop,
                price=self.stop_price,
            )
            self.take_order = self.sell(
                size=abs(self.position.size),
                exectype=bt.Order.Limit,
                price=self.take_price,
                oco=self.stop_order,
            )
        else:
            self.stop_order = self.buy(
                size=abs(self.position.size),
                exectype=bt.Order.Stop,
                price=self.stop_price,
            )
            self.take_order = self.buy(
                size=abs(self.position.size),
                exectype=bt.Order.Limit,
                price=self.take_price,
                oco=self.stop_order,
            )

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return

        if order is self.order:
            if order.status == order.Completed:
                self.entry_price = order.executed.price
                self.entry_atr = order.info.atr
                self.entry_bar = len(self)
                self.entry_direction = 1 if order.isbuy() else -1
                self.risk_distance = self.p.stop_atr_multiple * self.entry_atr
                self.breakeven_moved = False

                if self.entry_direction == 1:
                    self.stop_price = self.entry_price - self.risk_distance
                    self.take_price = (
                        self.entry_price
                        + self.p.take_profit_r * self.risk_distance
                    )
                    self.peak_close = self.data.close[0]
                else:
                    self.stop_price = self.entry_price + self.risk_distance
                    self.take_price = (
                        self.entry_price
                        - self.p.take_profit_r * self.risk_distance
                    )
                    self.trough_close = self.data.close[0]

                self._place_exit_orders()

            if order.status in (
                order.Completed,
                order.Canceled,
                order.Margin,
                order.Rejected,
            ):
                self.order = None
            return

        if order is self.stop_order and order.status == order.Canceled:
            self.stop_order = None
            if self.pending_stop_price is not None and self.position:
                new_stop = self.pending_stop_price
                self.pending_stop_price = None
                self._place_stop(new_stop)
            return

        if order is self.stop_order and order.status == order.Completed:
            self._clear_trade_state()
            return

        if order is self.take_order and order.status == order.Completed:
            self._clear_trade_state()
            return

        if order is self.take_order and order.status in (
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            self.take_order = None

    def next(self):
        if self.position:
            if len(self) == self.entry_bar:
                return

            if self.entry_direction == 1:
                self.peak_close = max(self.peak_close, self.data.close[0])

                if self.data.high[0] >= (
                    self.entry_price + self.p.breakeven_r * self.risk_distance
                ):
                    self.breakeven_moved = True

                desired_stop = (
                    self.peak_close - self.p.stop_atr_multiple * self.atr[0]
                )
                if self.breakeven_moved:
                    desired_stop = max(desired_stop, self.entry_price)
            else:
                self.trough_close = min(self.trough_close, self.data.close[0])

                if self.data.low[0] <= (
                    self.entry_price - self.p.breakeven_r * self.risk_distance
                ):
                    self.breakeven_moved = True

                desired_stop = (
                    self.trough_close + self.p.stop_atr_multiple * self.atr[0]
                )
                if self.breakeven_moved:
                    desired_stop = min(desired_stop, self.entry_price)

            if (
                self.stop_order is not None
                and self.stop_order.alive()
                and desired_stop != self.stop_price
            ):
                self.pending_stop_price = desired_stop
                self.cancel(self.stop_order)

            return

        if self.order is not None:
            return

        if (
            self.cooldown_exit_bar is not None
            and len(self) <= self.cooldown_exit_bar + self.p.cooldown_days
        ):
            return

        if not self._consolidating():
            return

        if self.atr[0] == 0:
            return

        atr_ratio = self.atr[0] / self.atr_sma[0]
        if not 0.7 <= atr_ratio <= 1.3:
            return

        stop_distance = self.p.stop_atr_multiple * self.atr[0]
        size = (self.broker.getvalue() * self.p.risk_fraction) / stop_distance

        if self.data.close[0] > self.sma200[0] and self.upper_cross[0] > 0:
            self.order = self.buy(size=size)
            self.order.addinfo(atr=self.atr[0])
        elif self.data.close[0] < self.sma200[0] and self.lower_cross[0] < 0:
            self.order = self.sell(size=size)
            self.order.addinfo(atr=self.atr[0])
