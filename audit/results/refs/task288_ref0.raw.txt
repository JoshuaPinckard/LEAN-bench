import backtrader as bt


class ScalpingMomentumStrategy(bt.Strategy):
    def __init__(self):
        self.sma = bt.indicators.SMA(self.data.close, period=20)
        self.atr1 = bt.indicators.ATR(self.data, period=1)
        self.bb = bt.indicators.BollingerBands(self.data, period=20, devfactor=2.0)
        self.crossover = bt.indicators.CrossOver(self.data.close, self.sma)

        self.order = None
        self.entry_price = 0.0
        self.max_since_entry = 0.0
        self.trailing_active = False
        self.pending_proxy_spread = 0.0
        self.entry_proxy_spread = 0.0
        self.trades_today = 0
        self.current_day = None

        self.total_trades = 0
        self.total_pnl = 0.0
        self.total_proxy_spread_sum = 0.0
        self.avg_proxy_spread = 0.0
        self.total_slippage_penalty = 0.0
        self.net_profit_after_penalty = 0.0

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return
        if order.status == order.Completed and order.isbuy():
            self.entry_price = order.executed.price
            self.max_since_entry = self.entry_price
            self.trailing_active = False
            self.trades_today += 1
            self.entry_proxy_spread = self.pending_proxy_spread
        self.order = None

    def notify_trade(self, trade):
        if trade.isclosed:
            self.total_trades += 1
            self.total_pnl += trade.pnlcomm
            self.total_proxy_spread_sum += self.entry_proxy_spread
            self.avg_proxy_spread = self.total_proxy_spread_sum / self.total_trades
            self.net_profit_after_penalty = self.total_pnl - self.total_slippage_penalty

    def next(self):
        dt = self.data.datetime.datetime(0)
        if self.current_day is None or dt.date() != self.current_day:
            self.current_day = dt.date()
            self.trades_today = 0

        if self.position:
            price = self.data.close[0]
            if price > self.max_since_entry:
                self.max_since_entry = price
            if not self.trailing_active and price >= self.entry_price * 1.002:
                self.trailing_active = True
            if self.trailing_active and not self.order:
                trailing_stop_price = self.max_since_entry * (1 - 0.0005)
                if price <= trailing_stop_price:
                    proxy_spread_exit = max(self.data.high[0] - self.data.low[0], self.atr1[0])
                    if proxy_spread_exit > 0.001 * price:
                        self.total_slippage_penalty += 0.0005 * price * self.position.size
                    self.order = self.close()
            return

        if self.order:
            return

        if self.trades_today >= 5:
            return

        minutes = dt.hour * 60 + dt.minute
        if not (585 <= minutes <= 660):
            return

        if self.crossover[0] <= 0:
            return

        price = self.data.close[0]
        proxy_spread = max(self.data.high[0] - self.data.low[0], self.atr1[0])
        if proxy_spread > 0.001 * price:
            return

        sma_val = self.sma[0]
        if abs(price - sma_val) / price > 0.005:
            return

        bb_width = (self.bb.top[0] - self.bb.bot[0]) / self.bb.mid[0]
        if bb_width > 0.015:
            return

        self.pending_proxy_spread = proxy_spread
        self.order = self.buy(size=100)
