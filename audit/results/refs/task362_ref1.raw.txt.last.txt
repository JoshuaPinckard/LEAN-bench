import backtrader as bt


class ChandelierADXStrategy(bt.Strategy):
    params = (
        ("lookback_n", 22),
        ("atr_mult", 3.0),
        ("adx_threshold", 25),
    )

    class _ChandelierStops(bt.Indicator):
        lines = ("long_stop", "short_stop")
        params = (
            ("lookback_n", 22),
            ("atr_mult", 3.0),
            ("adx_threshold", 25),
            ("strategy", None),
            ("adx", None),
        )

        def __init__(self):
            self.atr = bt.indicators.ATR(self.data, period=14)
            self.highest = bt.indicators.Highest(
                self.data.high, period=self.p.lookback_n
            )
            self.lowest = bt.indicators.Lowest(
                self.data.low, period=self.p.lookback_n
            )

        def next(self):
            long_stop_raw = self.highest[0] - self.p.atr_mult * self.atr[0]
            short_stop_raw = self.lowest[0] + self.p.atr_mult * self.atr[0]

            position_size = self.p.strategy.position.size
            long_eligible = position_size > 0 or (
                position_size == 0 and self.p.adx[0] > self.p.adx_threshold
            )
            short_eligible = position_size < 0 or (
                position_size == 0 and self.p.adx[0] > self.p.adx_threshold
            )

            if long_eligible and len(self) > 1:
                self.lines.long_stop[0] = max(
                    self.lines.long_stop[-1], long_stop_raw
                )
            else:
                self.lines.long_stop[0] = long_stop_raw

            if short_eligible and len(self) > 1:
                self.lines.short_stop[0] = min(
                    self.lines.short_stop[-1], short_stop_raw
                )
            else:
                self.lines.short_stop[0] = short_stop_raw

    def __init__(self):
        self.order = []
        self.atr = bt.indicators.ATR(self.data, period=14)
        self.adx = bt.indicators.ADX(self.data, period=14)
        self.stops = self._ChandelierStops(
            self.data,
            lookback_n=self.p.lookback_n,
            atr_mult=self.p.atr_mult,
            adx_threshold=self.p.adx_threshold,
            strategy=self,
            adx=self.adx,
        )

    def next(self):
        if self.order:
            return

        crossed_above_long_stop = (
            self.data.close[-1] <= self.stops.long_stop[-1]
            and self.data.close[0] > self.stops.long_stop[0]
        )
        crossed_below_long_stop = (
            self.data.close[-1] >= self.stops.long_stop[-1]
            and self.data.close[0] < self.stops.long_stop[0]
        )
        crossed_below_short_stop = (
            self.data.close[-1] >= self.stops.short_stop[-1]
            and self.data.close[0] < self.stops.short_stop[0]
        )
        crossed_above_short_stop = (
            self.data.close[-1] <= self.stops.short_stop[-1]
            and self.data.close[0] > self.stops.short_stop[0]
        )

        if self.position.size > 0:
            if crossed_below_long_stop:
                self.order.append(self.close())
                if self.adx[0] > self.p.adx_threshold and crossed_below_short_stop:
                    self.order.append(self.sell())

        elif self.position.size < 0:
            if crossed_above_short_stop:
                self.order.append(self.close())
                if self.adx[0] > self.p.adx_threshold and crossed_above_long_stop:
                    self.order.append(self.buy())

        elif self.adx[0] > self.p.adx_threshold:
            if crossed_above_long_stop:
                self.order.append(self.buy())
            elif crossed_below_short_stop:
                self.order.append(self.sell())

    def notify_order(self, order):
        if order.status in (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            if order in self.order:
                self.order.remove(order)
