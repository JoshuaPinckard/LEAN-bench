import backtrader as bt


class DeepDipLeaderPyramiding(bt.Strategy):
    def __init__(self):
        self.sma200 = bt.ind.SMA(self.data.close, period=200)
        self.ema20 = bt.ind.EMA(self.data.close, period=20)
        self.ema10 = bt.ind.EMA(self.data.close, period=10)
        self.rsi = bt.ind.RSI(self.data.close, period=14)
        self.macd = bt.ind.MACD(
            self.data.close,
            period_me1=12,
            period_me2=26,
            period_signal=9,
        )
        self.bbands = bt.ind.BollingerBands(
            self.data.close,
            period=20,
            devfactor=2,
        )
        self.atr = bt.ind.ATR(self.data, period=14)
        self.highest_close_252 = bt.ind.Highest(self.data.close, period=252)

        self.close_ema20_cross = bt.ind.CrossOver(self.data.close, self.ema20)
        self.rsi_30_cross = bt.ind.CrossOver(self.rsi, 30)
        self.macd_signal_cross = bt.ind.CrossOver(
            self.macd.macd,
            self.macd.signal,
        )

        self.order = None
        self.in_trade = False
        self.entry_price = None
        self.initial_stop = None
        self.maximum_price_since_entry = None
        self.add_count = 0
        self.partial_taken = False
        self.last_exit_bar = None

    def notify_order(self, order):
        if order.status in (
            order.Submitted,
            order.Accepted,
        ):
            return

        if order.status == order.Completed:
            kind = order.info.get("kind")

            if order.isbuy():
                if not self.in_trade:
                    self.in_trade = True
                    self.entry_price = order.executed.price
                    self.initial_stop = (
                        self.entry_price - (1.5 * self.atr[0])
                    )
                    self.maximum_price_since_entry = max(
                        self.entry_price,
                        self.data.high[0],
                    )
                    self.add_count = 0
                    self.partial_taken = False

            elif kind == "partial":
                self.partial_taken = True

            if order.issell() and self.position.size == 0:
                self.in_trade = False
                self.entry_price = None
                self.initial_stop = None
                self.maximum_price_since_entry = None
                self.add_count = 0
                self.partial_taken = False
                self.last_exit_bar = len(self)

        if order.status in (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            self.order = None

    def _submit_buy(self, size=None, kind=None):
        if size is None:
            self.order = self.buy()
        else:
            self.order = self.buy(size=size)
        self.order.addinfo(kind=kind)

    def _submit_close(self, size=None, kind=None):
        if size is None:
            self.order = self.close()
        else:
            self.order = self.close(size=size)
        self.order.addinfo(kind=kind)

    def next(self):
        if self.order:
            return

        if self.position.size:
            self.maximum_price_since_entry = max(
                self.maximum_price_since_entry,
                self.data.high[0],
            )

            growth_phase = (
                self.data.close[0] > self.sma200[0]
                and self.macd.macd[0] > 0
            )

            if growth_phase:
                active_stop = (
                    self.maximum_price_since_entry - (3 * self.atr[0])
                )
            else:
                active_stop = self.initial_stop

            if self.data.low[0] <= active_stop:
                self._submit_close(kind="stop")
                return

            at_fifty_percent = (
                self.data.close[0] >= self.position.price * 1.50
            )
            at_hundred_percent = (
                self.data.close[0] >= self.position.price * 2.00
            )
            upper_band_exit = (
                self.data.high[0] >= self.bbands.top[0]
                and self.rsi[0] > 75
            )
            trend_exit = (
                self.macd_signal_cross[0] < 0
                or (
                    self.data.close[0] < self.ema20[0]
                    and self.data.close[-1] < self.ema20[-1]
                )
            )

            if at_hundred_percent or upper_band_exit or trend_exit:
                self._submit_close(kind="exit")
                return

            if not self.partial_taken and at_fifty_percent:
                self._submit_close(
                    size=self.position.size / 2.0,
                    kind="partial",
                )
                return

            pullback = (
                self.data.low[0] <= self.ema20[0]
                or self.data.low[0] <= self.bbands.bot[0]
            )
            add_signal = (
                growth_phase
                and pullback
                and 45 <= self.rsi[0] <= 60
                and self.macd.macd[0] > 0
                and self.add_count < 3
                and self.data.close[0] > self.position.price
            )

            if add_signal:
                self.add_count += 1
                self._submit_buy(kind="add")

            return

        if (
            self.last_exit_bar is not None
            and len(self) <= self.last_exit_bar + 30
        ):
            return

        crash_filter = (
            self.data.close[0] <= 0.30 * self.highest_close_252[0]
        )
        crossover_entry = (
            self.close_ema20_cross[0] > 0
            and self.rsi_30_cross[0] > 0
        )
        lower_band_entry = (
            self.data.low[0] <= self.bbands.bot[0]
            and self.data.close[0] > self.bbands.bot[0]
            and self.rsi[0] < 35
        )

        if crash_filter and (crossover_entry or lower_band_entry):
            risk_per_share = 1.5 * self.atr[0]
            if risk_per_share > 0:
                size = (0.01 * self.broker.getvalue()) / risk_per_share
                self._submit_buy(size=size, kind="initial")
