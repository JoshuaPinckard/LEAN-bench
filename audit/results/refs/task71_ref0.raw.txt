import backtrader as bt
import collections


class ORBEmaTrendFuturesStrategy(bt.Strategy):
    params = dict(
        ema_fast_period=9,
        ema_slow_period=21,
        orb_minutes=30,
        take_profit_pct=0.005,
        risk_pct=0.01,
        vol_lookback_days=20,
        vol_threshold_mult=1.2,
        time_exit_minutes_before_close=15,
        session_open_minute=9 * 60 + 30,
        session_close_minute=16 * 60,
    )

    def __init__(self):
        self.ema_fast = bt.indicators.EMA(self.data.close, period=self.p.ema_fast_period)
        self.ema_slow = bt.indicators.EMA(self.data.close, period=self.p.ema_slow_period)
        self.crossover = bt.indicators.CrossOver(self.ema_fast, self.ema_slow)

        self.order = None
        self.stop_order = None
        self.limit_order = None

        self.current_day = None
        self.orb_high = None
        self.orb_low = None
        self.orb_locked = False
        self.orb_end_minute = self.p.session_open_minute + self.p.orb_minutes
        self.exit_cutoff_minute = self.p.session_close_minute - self.p.time_exit_minutes_before_close

        self.day_high = None
        self.day_low = None
        self.day_close = None
        self.prev_day_close = None
        self.daily_tr_history = collections.deque(maxlen=self.p.vol_lookback_days)
        self.high_vol_day = False

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            if order == self.order:
                self.order = None
            if order == self.stop_order:
                self.stop_order = None
            if order == self.limit_order:
                self.limit_order = None

    def next(self):
        dt = self.data.datetime.datetime(0)
        bar_date = dt.date()
        minute_of_day = dt.hour * 60 + dt.minute

        if self.current_day != bar_date:
            if self.current_day is not None and self.day_high is not None:
                if self.prev_day_close is not None:
                    tr = max(
                        self.day_high - self.day_low,
                        abs(self.day_high - self.prev_day_close),
                        abs(self.day_low - self.prev_day_close),
                    )
                else:
                    tr = self.day_high - self.day_low
                self.daily_tr_history.append(tr)
                self.prev_day_close = self.day_close

            self.current_day = bar_date
            self.day_high = self.data.high[0]
            self.day_low = self.data.low[0]
            self.day_close = self.data.close[0]
            self.orb_high = self.data.high[0]
            self.orb_low = self.data.low[0]
            self.orb_locked = False

            if len(self.daily_tr_history) >= self.p.vol_lookback_days:
                prev_tr = self.daily_tr_history[-1]
                avg_tr = sum(self.daily_tr_history) / len(self.daily_tr_history)
                self.high_vol_day = prev_tr > self.p.vol_threshold_mult * avg_tr
            else:
                self.high_vol_day = False
        else:
            self.day_high = max(self.day_high, self.data.high[0])
            self.day_low = min(self.day_low, self.data.low[0])
            self.day_close = self.data.close[0]

        if not self.orb_locked:
            if minute_of_day < self.orb_end_minute:
                self.orb_high = max(self.orb_high, self.data.high[0])
                self.orb_low = min(self.orb_low, self.data.low[0])
                return
            self.orb_locked = True

        if self.position and minute_of_day >= self.exit_cutoff_minute:
            if self.stop_order:
                self.cancel(self.stop_order)
            if self.limit_order:
                self.cancel(self.limit_order)
            self.close()
            return

        if self.order or self.position or self.high_vol_day:
            return

        price = self.data.close[0]

        if self.crossover[0] > 0 and price > self.orb_high:
            stop_price = self.orb_low
            stop_distance = price - stop_price
            if stop_distance <= 0:
                return
            size = int((self.p.risk_pct * self.broker.getvalue()) / stop_distance)
            if size < 1:
                return
            take_profit = self.orb_high + self.p.take_profit_pct * price
            orders = self.buy_bracket(size=size, exectype=bt.Order.Market,
                                       stopprice=stop_price, limitprice=take_profit)
            self.order, self.stop_order, self.limit_order = orders

        elif self.crossover[0] < 0 and price < self.orb_low:
            stop_price = self.orb_high
            stop_distance = stop_price - price
            if stop_distance <= 0:
                return
            size = int((self.p.risk_pct * self.broker.getvalue()) / stop_distance)
            if size < 1:
                return
            take_profit = self.orb_low - self.p.take_profit_pct * price
            orders = self.sell_bracket(size=size, exectype=bt.Order.Market,
                                        stopprice=stop_price, limitprice=take_profit)
            self.order, self.stop_order, self.limit_order = orders
