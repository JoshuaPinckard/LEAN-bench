import datetime
import backtrader as bt


class CreditSpreadProxyStrategy(bt.Strategy):
    def __init__(self):
        self.sma = bt.indicators.SMA(self.data.close, period=20)
        self.bb = bt.indicators.BollingerBands(self.data.close, period=20, devfactor=2.0)
        self.atr = bt.indicators.ATR(self.data, period=14)
        self.crossover_bot = bt.indicators.CrossOver(self.data.close, self.bb.lines.bot)
        self.crossover_top = bt.indicators.CrossOver(self.data.close, self.bb.lines.top)

        self.order = None
        self.tp_order = None
        self.sl_order = None
        self.tp_price = None
        self.sl_price = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            if order == self.order:
                size = order.executed.size
                if order.isbuy():
                    self.tp_order = self.sell(
                        size=size, price=self.tp_price, exectype=bt.Order.Limit
                    )
                    self.sl_order = self.sell(
                        size=size,
                        price=self.sl_price,
                        exectype=bt.Order.Stop,
                        oco=self.tp_order,
                    )
                else:
                    self.tp_order = self.buy(
                        size=abs(size), price=self.tp_price, exectype=bt.Order.Limit
                    )
                    self.sl_order = self.buy(
                        size=abs(size),
                        price=self.sl_price,
                        exectype=bt.Order.Stop,
                        oco=self.tp_order,
                    )
                self.order = None
            elif order in [self.tp_order, self.sl_order]:
                self.tp_order = None
                self.sl_order = None
        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            if order == self.order:
                self.order = None

    def next(self):
        cur_time = self.data.datetime.time()
        start_time = datetime.time(11, 0)
        end_time = datetime.time(15, 30)

        # Close open positions near session end
        if cur_time >= end_time:
            if self.position:
                if self.tp_order:
                    self.cancel(self.tp_order)
                    self.tp_order = None
                if self.sl_order:
                    self.cancel(self.sl_order)
                    self.sl_order = None
                self.close()
            return

        # Do not trade prior to 90 minutes after open (09:30 + 90m = 11:00)
        if cur_time < start_time:
            return

        # Guard against pending orders or active positions
        if self.order or self.position:
            return

        # Ensure sufficient history for 30 bars ago SMA(20)
        if len(self.sma) <= 30:
            return

        close = self.data.close[0]
        sma_curr = self.sma[0]
        sma_prev = self.sma[-30]

        is_bullish = (close > sma_curr) and (sma_curr > sma_prev)
        is_bearish = (close < sma_curr) and (sma_curr < sma_prev)

        # Volatility filter: skip if ATR(14)/Close > 2%
        if close <= 0:
            return
        if (self.atr[0] / close) > 0.02:
            return

        capital = self.broker.getvalue()
        risk_amount = 0.01 * capital

        if is_bullish and self.crossover_bot[0] > 0:
            entry_price = close
            dist_mid = abs(self.bb.lines.mid[0] - entry_price)
            tp_dist = min(0.25 * self.atr[0], dist_mid)
            self.tp_price = entry_price + tp_dist

            dist_opp = abs(entry_price - self.bb.lines.top[0])
            sl_dist = max(1.25 * self.atr[0], dist_opp)
            self.sl_price = entry_price - sl_dist

            if sl_dist > 0:
                size = int(risk_amount / sl_dist)
                if size >= 1 and (size * entry_price) <= self.broker.getcash():
                    self.order = self.buy(size=size)

        elif is_bearish and self.crossover_top[0] < 0:
            entry_price = close
            dist_mid = abs(entry_price - self.bb.lines.mid[0])
            tp_dist = min(0.25 * self.atr[0], dist_mid)
            self.tp_price = entry_price - tp_dist

            dist_opp = abs(entry_price - self.bb.lines.bot[0])
            sl_dist = max(1.25 * self.atr[0], dist_opp)
            self.sl_price = entry_price + sl_dist

            if sl_dist > 0:
                size = int(risk_amount / sl_dist)
                if size >= 1 and (size * entry_price) <= self.broker.getcash():
                    self.order = self.sell(size=size)
