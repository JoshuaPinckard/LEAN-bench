import backtrader as bt


class InitialPositionSMAStrategy(bt.Strategy):
    params = (("sma_period", 20), ("position_pct", 0.05), ("max_positions", 3))

    def __init__(self):
        self.sma = bt.indicators.SimpleMovingAverage(
            self.data.close, period=self.p.sma_period
        )
        self.order = None
        self.entries_count = 0

    def start(self):
        # Set initial position of 10 shares bought at $150 per share
        initial_shares = 10
        initial_price = 150.0
        self.broker.positions[self.data] = bt.position.Position(
            size=initial_shares, price=initial_price
        )

        # Deduct initial position cost and 0.1% commission from initial cash
        cost = initial_shares * initial_price
        commission = cost * 0.001
        self.broker.cash -= cost + commission

        # Count initial position as 1 entry leg
        self.entries_count = 1

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            if order.isbuy():
                self.entries_count += 1
            elif order.issell():
                if not self.position or self.position.size == 0:
                    self.entries_count = 0

        self.order = None

    def next(self):
        if self.order:
            return

        if self.data.close[0] < self.sma[0]:
            if self.position.size > 0:
                self.order = self.close()
        elif self.data.close[0] > self.sma[0]:
            if self.entries_count < self.p.max_positions:
                capital = self.broker.getvalue()
                target_value = capital * self.p.position_pct
                size = int(target_value / self.data.close[0])
                if size > 0:
                    self.order = self.buy(size=size)
