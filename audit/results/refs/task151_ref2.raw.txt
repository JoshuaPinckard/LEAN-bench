import datetime
import backtrader as bt


class ReferenceStrategy(bt.Strategy):
    def __init__(self):
        self.sma = bt.indicators.SMA(self.data.close, period=10)
        self.rsi = bt.indicators.RSI(self.data.close, period=14)
        self.sma_vol = bt.indicators.SMA(self.data.volume, period=50)

        self.order = None
        self.pos_count = 0
        self.last_entry_bar = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            if self.position.size == 0:
                self.pos_count = 0
        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            if self.position.size == 0:
                self.pos_count = 0

        self.order = None

    def next(self):
        if self.order:
            return

        price = self.data.close[0]
        volume = self.data.volume[0]
        dt = self.data.datetime.datetime(0)

        # Handle exit conditions for open positions
        if self.position:
            entry_price = self.position.price

            if self.position.size > 0:
                sl_hit = price <= entry_price * (1.0 - 0.005)
                tp_hit = price >= entry_price * (1.0 + 0.008)
                reversal = price < self.sma[0]
                rsi_exit = self.rsi[0] < 30 or self.rsi[0] > 70

                if sl_hit or tp_hit or reversal or rsi_exit:
                    self.order = self.close()
                    return

            elif self.position.size < 0:
                sl_hit = price >= entry_price * (1.0 + 0.005)
                tp_hit = price <= entry_price * (1.0 - 0.008)
                reversal = price > self.sma[0]
                rsi_exit = self.rsi[0] < 30 or self.rsi[0] > 70

                if sl_hit or tp_hit or reversal or rsi_exit:
                    self.order = self.close()
                    return

        # Ignore trading signals in the first 5 minutes of the trading day (09:30 - 09:34)
        if dt.time() < datetime.time(9, 35):
            return

        # Check interval of at least 3 ticks between entry signals
        if self.last_entry_bar is not None and (len(self) - self.last_entry_bar) < 3:
            return

        buy_cond = (
            price > self.sma[0] * 1.001 and
            40 <= self.rsi[0] <= 60 and
            volume > self.sma_vol[0] * 1.20
        )

        sell_cond = (
            price < self.sma[0] * 0.999 and
            40 <= self.rsi[0] <= 60 and
            volume > self.sma_vol[0] * 1.20
        )

        size = (0.01 * self.broker.getvalue()) / price

        if buy_cond:
            if self.position.size >= 0 and self.pos_count < 3:
                self.order = self.buy(size=size)
                self.pos_count += 1
                self.last_entry_bar = len(self)
        elif sell_cond:
            if self.position.size <= 0 and self.pos_count < 3:
                self.order = self.sell(size=size)
                self.pos_count += 1
                self.last_entry_bar = len(self)
