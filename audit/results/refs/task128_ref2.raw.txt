import backtrader as bt


class StatisticalCapitulationBounce(bt.Strategy):
    """
    Reference implementation of 'Statistical Capitulation Bounce' strategy.
    
    Identifies extreme capitulation events using 9 oversold conditions, with a
    mandatory requirement that price deviation from the 100-day SMA is in the 
    1st or 2nd percentile rank historically. Exits when price crosses down below
    the 100-day SMA.
    """
    params = (
        ('sma_period', 100),
        ('rsi_long_period', 14),
        ('rsi_short_period', 7),
        ('stoch_period', 14),
        ('stoch_dfast', 3),
        ('stoch_dslow', 3),
        ('williams_period', 14),
        ('cci_period', 20),
        ('roc_period', 10),
        ('boll_period', 20),
        ('boll_dev', 2.0),
    )

    def __init__(self):
        self.order = None
        self.dev_history = []

        # Key baseline indicator: 100-day SMA
        self.sma100 = bt.ind.SMA(self.data.close, period=self.p.sma_period)

        # 8 additional oversold indicators
        self.rsi14 = bt.ind.RSI(self.data.close, period=self.p.rsi_long_period)
        self.rsi7 = bt.ind.RSI(self.data.close, period=self.p.rsi_short_period)
        self.stoch = bt.ind.Stochastic(
            self.data,
            period=self.p.stoch_period,
            period_dfast=self.p.stoch_dfast,
            period_dslow=self.p.stoch_dslow
        )
        self.williams = bt.ind.WilliamsR(self.data, period=self.p.williams_period)
        self.cci = bt.ind.CCI(self.data, period=self.p.cci_period)
        self.roc = bt.ind.ROC(self.data.close, period=self.p.roc_period)
        self.boll = bt.ind.BollingerBands(
            self.data.close,
            period=self.p.boll_period,
            devfactor=self.p.boll_dev
        )

        # Primary exit signal: Price crosses 100-day SMA from above to below
        self.crossdown = bt.ind.CrossDown(self.data.close, self.sma100)

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        # Calculate price deviation from 100-day SMA
        dev = self.data.close[0] - self.sma100[0]
        self.dev_history.append(dev)

        # Percentile rank of current deviation over available historical period
        count_less_equal = sum(1 for d in self.dev_history if d <= dev)
        pct_rank = (count_less_equal / len(self.dev_history)) * 100.0

        # Mandatory Condition: Price deviation from 100-day SMA is at 1st or 2nd percentile
        c1 = (pct_rank <= 2.0)

        # 8 Additional Oversold Conditions
        c2 = (self.rsi14[0] <= 30.0)
        c3 = (self.rsi7[0] <= 20.0)
        c4 = (self.stoch.percK[0] <= 20.0)
        c5 = (self.stoch.percD[0] <= 20.0)
        c6 = (self.williams[0] <= -80.0)
        c7 = (self.cci[0] <= -100.0)
        c8 = (self.roc[0] <= -5.0)
        c9 = (self.data.close[0] <= self.boll.lines.bot[0])

        conditions = [c1, c2, c3, c4, c5, c6, c7, c8, c9]
        total_met = sum(1 for c in conditions if c)

        if self.order:
            return

        if not self.position:
            # Entry condition: Extreme selloff signal (at least 9 conditions met including mandatory c1)
            if c1 and total_met >= 9:
                self.order = self.buy()
        else:
            # Exit condition: Price crosses 100-day SMA from above to below
            if self.crossdown[0] > 0:
                self.order = self.close()
