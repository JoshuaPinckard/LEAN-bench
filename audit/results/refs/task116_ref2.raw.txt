import datetime
import backtrader as bt


class OrbStrategy(bt.Strategy):
    params = (
        ('or_start_time', datetime.time(9, 30)),
        ('or_end_time', datetime.time(10, 0)),
        ('entry_cutoff_time', datetime.time(15, 0)),
        ('risk_pct', 0.01),
        ('reward_risk_ratio', 1.5),
    )

    def __init__(self):
        self.order = None
        self.current_date = None
        self.or_high = None
        self.or_low = None
        self.or_prices_high = []
        self.or_prices_low = []
        self.breakout_occurred = False
        self.pullback_occurred = False
        self.wick_seen = False
        self.stop_loss = None
        self.take_profit = None

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        dt = self.data.datetime.datetime(0)
        current_date = dt.date()
        current_time = dt.time()

        if self.current_date != current_date:
            self.current_date = current_date
            self.or_high = None
            self.or_low = None
            self.or_prices_high = []
            self.or_prices_low = []
            self.breakout_occurred = False
            self.pullback_occurred = False
            self.wick_seen = False
            self.stop_loss = None
            self.take_profit = None

        if self.p.or_start_time <= current_time < self.p.or_end_time:
            self.or_prices_high.append(self.data.high[0])
            self.or_prices_low.append(self.data.low[0])
            return

        if self.or_high is None:
            if self.or_prices_high and self.or_prices_low:
                self.or_high = max(self.or_prices_high)
                self.or_low = min(self.or_prices_low)
            else:
                return

        if self.position:
            if self.stop_loss and self.data.low[0] <= self.stop_loss:
                self.close()
            elif self.take_profit and self.data.high[0] >= self.take_profit:
                self.close()
            return

        if self.order or current_time >= self.p.entry_cutoff_time:
            return

        or_range = self.or_high - self.or_low
        if or_range <= 0:
            return

        fib_618 = self.or_low + 0.618 * or_range
        fib_382 = self.or_low + 0.382 * or_range
        fib_236 = self.or_low + 0.236 * or_range

        if not self.breakout_occurred:
            if self.data.close[0] > self.or_high:
                self.breakout_occurred = True
            return

        if self.breakout_occurred and not self.pullback_occurred:
            if self.data.low[0] <= fib_618 and self.data.high[0] >= fib_382:
                self.pullback_occurred = True

        if not self.pullback_occurred:
            return

        upper_wick = self.data.high[0] - max(self.data.open[0], self.data.close[0])
        body = abs(self.data.close[0] - self.data.open[0])
        if upper_wick >= body:
            self.wick_seen = True

        if self.pullback_occurred and self.wick_seen:
            if self.data.close[0] > self.data.open[0]:
                entry_price = self.data.close[0]
                sl = fib_236
                risk_per_share = entry_price - sl
                if risk_per_share > 0:
                    capital = self.broker.getvalue()
                    risk_amount = capital * self.p.risk_pct
                    size = int(risk_amount / risk_per_share)
                    if size > 0:
                        self.stop_loss = sl
                        self.take_profit = entry_price + self.p.reward_risk_ratio * risk_per_share
                        self.order = self.buy(size=size)
