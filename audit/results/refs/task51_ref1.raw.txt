import backtrader as bt
import datetime


class WeeklySessionRectangle(bt.Strategy):
    params = (
        ("rectangle_color", "gray"),
        ("rectangle_alpha", 0.30),
    )

    def __init__(self):
        self.week_rectangle = None

    def next(self):
        current_time = self.data.datetime.datetime(0)

        if current_time.weekday() >= 5:
            self.week_rectangle = None
            return

        monday_date = current_time.date() - datetime.timedelta(
            days=current_time.weekday()
        )
        week_start = datetime.datetime.combine(monday_date, datetime.time.min)

        self.week_rectangle = {
            "start": week_start,
            "end": current_time,
            "color": self.p.rectangle_color,
            "alpha": self.p.rectangle_alpha,
        }
