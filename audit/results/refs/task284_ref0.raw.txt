import backtrader as bt


class BBBreakoutTrailingStop(bt.Strategy):
    params = dict(
        bb_period=20,
        bb_devfactor=2,
        rsi_period=14,
        rsi_threshold=55,
        trail_fraction=0.01 / 100,
    )

    def __init__(self):
        self.bb = bt.indicators.BollingerBands(
            self.data.close, period=self.p.bb_period, devfactor=self.p.bb_devfactor
        )
        self.rsi = bt.indicators.RSI(self.data.close, period=self.p.rsi_period)
        self.crossover = bt.indicators.CrossOver(self.data.close, self.bb.top)

        self.order = None
        self.highest_price = None
        self.trailing_stop_loss = None
        self.exit_signal = 0

    def notify_order(self, order):
        if order.status in (order.Completed, order.Canceled, order.Margin, order.Rejected):
            self.order = None

    def next(self):
        self.exit_signal = 0

        if self.order:
            return

        if not self.position:
            self.trailing_stop_loss = None
            self.highest_price = None
            if self.crossover[0] > 0 and self.rsi[0] > self.p.rsi_threshold:
                self.order = self.buy(size=1)
            return

        if self.highest_price is None:
            entry_price = self.position.price
            self.highest_price = entry_price
            self.trailing_stop_loss = entry_price * (1 - self.p.trail_fraction)

        self.highest_price = max(self.highest_price, self.data.high[0])
        self.trailing_stop_loss = max(
            self.trailing_stop_loss,
            self.highest_price * (1 - self.p.trail_fraction),
        )

        if self.data.close[0] <= self.trailing_stop_loss:
            self.exit_signal = 1
            self.order = self.close()
