import backtrader as bt


class VolatilityTrendStrategy(bt.Strategy):
    def __init__(self):
        self.sma = bt.indicators.SMA(self.data.close, period=20)
        self.rsi = bt.indicators.RSI(self.data.close, period=14)
        self.atr = bt.indicators.ATR(self.data, period=14)
        self.atr_ema = bt.indicators.EMA(self.atr, period=14)
        self.vol_sma = bt.indicators.SMA(self.data.volume, period=20)

        self.order = None
        self.entry_pending = False
        self.pending_atr = None
        self.stop_price = None
        self.take_price = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status == order.Completed:
            if self.entry_pending:
                if order.isbuy():
                    self.stop_price = order.executed.price - 1.5 * self.pending_atr
                    self.take_price = order.executed.price + 2.0 * self.pending_atr
                else:
                    self.stop_price = order.executed.price + 1.5 * self.pending_atr
                    self.take_price = order.executed.price - 2.0 * self.pending_atr
                self.entry_pending = False
            else:
                self.stop_price = None
                self.take_price = None

        if order.status in [order.Canceled, order.Margin, order.Rejected]:
            self.entry_pending = False

        self.order = None

    def next(self):
        if self.order:
            return

        close = self.data.close[0]
        rsi_now = self.rsi[0]
        rsi_prev = self.rsi[-1]
        rsi_cross_below_50 = rsi_prev >= 50 and rsi_now < 50
        rsi_cross_above_50 = rsi_prev <= 50 and rsi_now > 50

        if not self.position:
            if (close > self.sma[0] and rsi_now > 55 and
                    self.atr[0] > self.atr_ema[0] and
                    self.data.volume[0] > self.vol_sma[0]):
                self.entry_pending = True
                self.pending_atr = self.atr[0]
                self.order = self.buy()
            elif (close < self.sma[0] and rsi_now < 45 and
                    self.atr[0] > self.atr_ema[0] and
                    self.data.volume[0] > self.vol_sma[0]):
                self.entry_pending = True
                self.pending_atr = self.atr[0]
                self.order = self.sell()
        else:
            if self.position.size > 0:
                sl_hit = self.data.low[0] <= self.stop_price
                tp_hit = self.data.high[0] >= self.take_price
                rule_exit = close < self.sma[0] or rsi_cross_below_50
                if sl_hit or tp_hit or rule_exit:
                    self.order = self.close()
            elif self.position.size < 0:
                sl_hit = self.data.high[0] >= self.stop_price
                tp_hit = self.data.low[0] <= self.take_price
                rule_exit = close > self.sma[0] or rsi_cross_above_50
                if sl_hit or tp_hit or rule_exit:
                    self.order = self.close()
