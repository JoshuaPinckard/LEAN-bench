class MartingaleRsiBbStrategy(bt.Strategy):
    params = dict(
        rsi_period=14,
        bb_period=20,
        bb_devfactor=2.0,
        rsi_oversold=30,
        rsi_overbought=70,
        rsi_exit_mid=50,
        initial_lot=0.01,
        lotexp=1.2345,
        max_levels=6,
        max_dd=350.0,
    )

    def __init__(self):
        self.rsi = bt.indicators.RSI(self.data, period=self.p.rsi_period)
        self.bb = bt.indicators.BollingerBands(
            self.data, period=self.p.bb_period, devfactor=self.p.bb_devfactor
        )

        self.order = None
        self.direction = None
        self.level = 0
        self.peak = self.broker.getvalue()

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        if self.order:
            return

        equity = self.broker.getvalue()
        if equity > self.peak:
            self.peak = equity
        drawdown = self.peak - equity

        if self.position and drawdown >= self.p.max_dd:
            self.order = self.close()
            self.direction = None
            self.level = 0
            return

        close = self.data.close[0]
        upper = self.bb.lines.top[0]
        lower = self.bb.lines.bot[0]
        rsi = self.rsi[0]

        if self.position:
            if self.direction == 'long':
                if close > upper or rsi > self.p.rsi_exit_mid:
                    self.order = self.close()
                    self.direction = None
                    self.level = 0
                    return
                if self.level < self.p.max_levels and close < lower and rsi < self.p.rsi_oversold:
                    lot = self.p.initial_lot * (self.p.lotexp ** self.level)
                    self.order = self.buy(size=lot)
                    self.level += 1
            elif self.direction == 'short':
                if close < lower or rsi < self.p.rsi_exit_mid:
                    self.order = self.close()
                    self.direction = None
                    self.level = 0
                    return
                if self.level < self.p.max_levels and close > upper and rsi > self.p.rsi_overbought:
                    lot = self.p.initial_lot * (self.p.lotexp ** self.level)
                    self.order = self.sell(size=lot)
                    self.level += 1
        else:
            if close < lower and rsi < self.p.rsi_oversold:
                self.order = self.buy(size=self.p.initial_lot)
                self.direction = 'long'
                self.level = 1
            elif close > upper and rsi > self.p.rsi_overbought:
                self.order = self.sell(size=self.p.initial_lot)
                self.direction = 'short'
                self.level = 1
