import backtrader as bt


class KeltnerMeanReversionStrategy(bt.Strategy):
    params = dict(
        ema_period=20,
        atr_period=14,
        atr_mult=2.0,
    )

    def __init__(self):
        self.ema = bt.indicators.EMA(self.data.close, period=self.p.ema_period)
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.upper = self.ema + self.p.atr_mult * self.atr
        self.lower = self.ema - self.p.atr_mult * self.atr
        self.crossover = bt.indicators.CrossOver(self.data.close, self.ema)

        self.order = None
        self.pending_entry_atr = None
        self.stop_price = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return
        if order.status == order.Completed:
            if self.position:
                if self.position.size > 0:
                    self.stop_price = order.executed.price - self.p.atr_mult * self.pending_entry_atr
                else:
                    self.stop_price = order.executed.price + self.p.atr_mult * self.pending_entry_atr
            else:
                self.stop_price = None
        self.order = None

    def next(self):
        if self.order:
            return

        if self.position:
            if self.position.size > 0:
                if self.data.low[0] <= self.stop_price:
                    self.order = self.close()
                    return
                if self.crossover[0] > 0:
                    self.order = self.close()
                    return
            else:
                if self.data.high[0] >= self.stop_price:
                    self.order = self.close()
                    return
                if self.crossover[0] < 0:
                    self.order = self.close()
                    return
            return

        bullish = self.data.close[0] > self.data.open[0]
        bearish = self.data.close[0] < self.data.open[0]

        if bullish and self.data.close[0] < self.lower[0]:
            self.pending_entry_atr = self.atr[0]
            self.order = self.buy()
        elif bearish and self.data.close[0] > self.upper[0]:
            self.pending_entry_atr = self.atr[0]
            self.order = self.sell()
