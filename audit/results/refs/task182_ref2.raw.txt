import backtrader as bt


class CAGR3YSMA200Strategy(bt.Strategy):
    def __init__(self):
        self.order = None
        self.sma = bt.indicators.SMA(self.data.close, period=200)
        self.crossover = bt.indicators.CrossOver(self.data.close, self.sma)
        
        # TotalReturn_3Y = Close / Close[-756]
        self.total_return = self.data.close / self.data.close(-756)
        # CAGR_3Y = (TotalReturn_3Y)^(1/3) - 1
        self.cagr = (self.total_return ** (1.0 / 3.0)) - 1.0

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        if self.order:
            return

        # Explicit history guard per spec: History < 756 days — no trades
        if len(self) <= 756:
            return

        if not self.position:
            if self.cagr[0] >= 0.05 and self.crossover[0] > 0:
                self.order = self.buy()
        else:
            if self.crossover[0] < 0 or self.cagr[0] < 0.0:
                self.order = self.close()
