import backtrader as bt
import math


class InitialPositionSMAStrategy(bt.Strategy):
    params = (
        ("sma_period", 20),
        ("initial_size", 10),
        ("initial_price", 150.0),
        ("max_positions", 3),
        ("position_fraction", 0.05),
    )

    def __init__(self):
        self.sma = bt.indicators.SMA(self.data.close, period=self.p.sma_period)
        self.order = None
        self.open_positions = 1

    def start(self):
        initial_cost = self.p.initial_size * self.p.initial_price
        initial_commission = initial_cost * 0.001

        position = self.broker.positions[self.data]
        position.size = self.p.initial_size
        position.price = self.p.initial_price
        self.broker.setcash(self.broker.getcash() - initial_cost - initial_commission)

    def next(self):
        if self.order:
            return

        if self.position.size and self.data.close[0] < self.sma[0]:
            self.order = self.close()
        elif (self.data.close[0] > self.sma[0] and
              not self.position.size and self.open_positions < self.p.max_positions):
            capital = self.broker.getvalue()
            size = math.floor((capital * self.p.position_fraction) / self.data.close[0])
            if size > 0:
                self.order = self.buy(size=size)
        elif (self.data.close[0] > self.sma[0] and
              self.position.size and self.open_positions < self.p.max_positions):
            capital = self.broker.getvalue()
            size = math.floor((capital * self.p.position_fraction) / self.data.close[0])
            if size > 0:
                self.order = self.buy(size=size)

    def notify_order(self, order):
        if order.status == order.Completed:
            if order.isbuy():
                self.open_positions += 1
            elif order.issell() and not self.position.size:
                self.open_positions = 0
            self.order = None
        elif order.status in (order.Canceled, order.Margin, order.Rejected):
            self.order = None
