import backtrader as bt


class HeuristicStrategy(bt.Strategy):
    params = (
        ('fast_period', 20),
        ('slow_period', 50),
        ('ma_type', 'sma'),
        ('rsi_period', 14),
        ('macd_fast', 12),
        ('macd_slow', 26),
        ('macd_signal', 9),
        ('take_profit', 0.02),
        ('stop_loss', 0.01),
        ('position_pct', 0.02),
        ('use_trend_filter', True),
        ('trend_period', 200),
        ('use_volatility_filter', False),
        ('atr_period', 14),
        ('atr_min', 0.0),
        ('atr_max', float('inf')),
    )

    def __init__(self):
        self.order = None

        ma_class = bt.indicators.EMA if str(self.p.ma_type).lower() == 'ema' else bt.indicators.SMA
        self.fast_ma = ma_class(self.data.close, period=self.p.fast_period)
        self.slow_ma = ma_class(self.data.close, period=self.p.slow_period)
        self.crossover = bt.indicators.CrossOver(self.fast_ma, self.slow_ma)

        self.rsi = bt.indicators.RSI(self.data.close, period=self.p.rsi_period)
        self.macd = bt.indicators.MACD(
            self.data.close,
            period_me1=self.p.macd_fast,
            period_me2=self.p.macd_slow,
            period_signal=self.p.macd_signal
        )

        self.trend_ma = bt.indicators.SMA(self.data.close, period=self.p.trend_period)
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        if self.order:
            return

        if not self.position:
            if self.p.use_volatility_filter:
                if not (self.p.atr_min <= self.atr[0] <= self.p.atr_max):
                    return

            if self.crossover[0] > 0:
                if not self.p.use_trend_filter or (self.data.close[0] > self.trend_ma[0]):
                    cash = self.broker.getcash()
                    size = int((cash * self.p.position_pct) / self.data.close[0])
                    if size > 0:
                        self.order = self.buy(size=size)

            elif self.crossover[0] < 0:
                if not self.p.use_trend_filter or (self.data.close[0] < self.trend_ma[0]):
                    cash = self.broker.getcash()
                    size = int((cash * self.p.position_pct) / self.data.close[0])
                    if size > 0:
                        self.order = self.sell(size=size)

        else:
            entry_price = self.position.price
            if self.position.size > 0:
                tp_price = entry_price * (1.0 + self.p.take_profit)
                sl_price = entry_price * (1.0 - self.p.stop_loss)
                if (self.data.close[0] >= tp_price or
                    self.data.close[0] <= sl_price or
                    self.crossover[0] < 0):
                    self.order = self.close()

            elif self.position.size < 0:
                tp_price = entry_price * (1.0 - self.p.take_profit)
                sl_price = entry_price * (1.0 + self.p.stop_loss)
                if (self.data.close[0] <= tp_price or
                    self.data.close[0] >= sl_price or
                    self.crossover[0] > 0):
                    self.order = self.close()
