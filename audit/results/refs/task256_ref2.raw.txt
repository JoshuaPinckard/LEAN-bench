import math
import backtrader as bt


class ManagedIncomeCapitalProtection(bt.Strategy):
    def __init__(self):
        self.sma50 = bt.indicators.SMA(self.data.close, period=50)
        self.sma200 = bt.indicators.SMA(self.data.close, period=200)
        self.atr = bt.indicators.ATR(self.data, period=14)

        self.order = None
        self.entry_price = None
        self.entry_atr = None
        self.tp_price = None
        self.halted = False
        self.starting_capital = None

    def start(self):
        self.starting_capital = self.broker.getvalue()

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status == order.Completed:
            if order.isbuy():
                self.entry_price = order.executed.price
                self.tp_price = min(
                    self.entry_price * 1.05,
                    self.entry_price + 2.5 * self.entry_atr,
                )
            elif order.issell():
                self.entry_price = None
                self.tp_price = None
                self.entry_atr = None

        self.order = None

    def next(self):
        equity = self.broker.getvalue()

        if (
            self.starting_capital
            and (self.starting_capital - equity) / self.starting_capital >= 0.10
        ):
            self.halted = True
            if self.position and not self.order:
                self.order = self.close()
            return

        if self.halted or self.order:
            return

        if not self.position:
            if (
                self.data.close[0] > self.sma50[0]
                and self.data.close[0] > self.sma200[0]
                and (self.atr[0] / self.data.close[0]) <= 0.015
            ):
                size = math.floor(0.05 * equity / self.data.close[0])
                if size > 0:
                    self.entry_atr = self.atr[0]
                    self.order = self.buy(size=size)
        else:
            if self.data.close[0] < self.sma50[0]:
                self.order = self.close()
            elif self.tp_price and self.data.close[0] >= self.tp_price:
                self.order = self.close()
