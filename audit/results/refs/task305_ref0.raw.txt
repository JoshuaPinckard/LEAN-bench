import backtrader as bt


class BullishEngulfingIntradayStrategy(bt.Strategy):
    def __init__(self):
        self.entry_order = None
        self.stop_order = None
        self.stop_price = None

    def next(self):
        try:
            last_bar_of_day = self.data.datetime.date(1) != self.data.datetime.date(0)
        except IndexError:
            last_bar_of_day = True

        if self.position and last_bar_of_day:
            if self.stop_order is not None:
                self.cancel(self.stop_order)
                self.stop_order = None
            self.close()
            self.stop_price = None
            return

        if self.position or self.entry_order is not None:
            return

        if len(self.data) < 5:
            return

        bearish_filter = (
            self.data.close[-2] < self.data.open[-2]
            and self.data.close[-3] < self.data.open[-3]
            and self.data.close[-4] < self.data.open[-4]
        )
        bar_minus1_bearish = self.data.close[-1] < self.data.open[-1]
        bar0_bullish = self.data.close[0] > self.data.open[0]
        engulfing = (
            self.data.open[0] <= self.data.close[-1]
            and self.data.close[0] >= self.data.open[-1]
        )

        if bearish_filter and bar_minus1_bearish and bar0_bullish and engulfing:
            pattern_low = min(self.data.low[0], self.data.low[-1])
            self.stop_price = pattern_low * 0.99
            self.entry_order = self.buy(exectype=bt.Order.Market)

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return

        if order.status == order.Completed:
            if order == self.entry_order:
                self.entry_order = None
                self.stop_order = self.close(
                    exectype=bt.Order.Stop, price=self.stop_price
                )
            elif order == self.stop_order:
                self.stop_order = None
                self.stop_price = None
        elif order.status in (order.Canceled, order.Margin, order.Rejected):
            if order == self.entry_order:
                self.entry_order = None
                self.stop_price = None
            elif order == self.stop_order:
                self.stop_order = None
