import backtrader as bt


class RollingCoveredCalls(bt.Strategy):
    params = (
        ('bb_period', 20),
        ('bb_dev', 2.0),
        ('rsi_period', 14),
        ('atr_period', 14),
        ('vol_period', 20),
    )

    def __init__(self):
        self.bb = bt.indicators.BollingerBands(
            self.data.close,
            period=self.p.bb_period,
            devfactor=self.p.bb_dev
        )
        self.rsi = bt.indicators.RSI(self.data.close, period=self.p.rsi_period)
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.vol_sma = bt.indicators.SMA(self.data.volume, period=self.p.vol_period)

        self.order = None
        self.tranches = []
        self.waiting_for_reentry = False

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def get_core_size(self):
        cash = self.broker.getcash()
        price = self.data.close[0]
        if price <= 0:
            return 0
        max_shares = int(cash // price)
        if max_shares >= 100:
            return (max_shares // 100) * 100
        return max_shares

    def get_tranche_size(self):
        pos_size = self.position.size
        if pos_size <= 0:
            return 0
        target_20pct = int(round(0.20 * pos_size))
        if pos_size >= 100:
            tranche = max(100, target_20pct)
            return min(pos_size, tranche)
        else:
            return max(1, target_20pct)

    def next(self):
        if self.order:
            return

        # Re-entry after forced assignment
        if self.waiting_for_reentry:
            if self.position.size == 0:
                if self.data.close[0] < self.bb.mid[0] and self.rsi[0] < 50:
                    size = self.get_core_size()
                    if size > 0:
                        self.order = self.buy(size=size)
                        self.waiting_for_reentry = False
                        self.tranches = []
            return

        # Initial core entry
        if self.position.size == 0 and not self.waiting_for_reentry:
            size = self.get_core_size()
            if size > 0:
                self.order = self.buy(size=size)
            return

        # Forced assignment check when active tranches exist
        if len(self.tranches) > 0:
            if self.data.close[0] > (self.bb.top[0] + 1.0 * self.atr[0]):
                self.order = self.close()
                self.tranches = []
                self.waiting_for_reentry = True
                return

        # Expiration without exercise (mean reversion buyback of active tranches)
        if len(self.tranches) > 0:
            if self.data.close[0] < self.bb.mid[0] or self.rsi[0] < 45:
                total_tranche_shares = sum(t['shares'] for t in self.tranches)
                if total_tranche_shares > 0:
                    self.order = self.buy(size=total_tranche_shares)
                self.tranches = []
                return

        # Update term age for active tranches and handle rolling on the 7th bar
        tranches_to_keep = []
        shares_to_buyback = 0
        rolled_any = False

        for t in self.tranches:
            t['age'] += 1
            if t['age'] == 7:
                if self.data.close[0] > self.bb.top[0]:
                    t['strike'] = self.bb.top[0]
                    t['age'] = 0
                    tranches_to_keep.append(t)
                    rolled_any = True
                else:
                    shares_to_buyback += t['shares']
            else:
                tranches_to_keep.append(t)

        self.tranches = tranches_to_keep

        if shares_to_buyback > 0:
            self.order = self.buy(size=shares_to_buyback)
            return

        # Add one more 20% tranche if rolled on 7th bar and RSI > 60
        if rolled_any and self.rsi[0] > 60:
            if self.position.size > 0:
                t_size = self.get_tranche_size()
                if t_size > 0:
                    self.order = self.sell(size=t_size)
                    self.tranches.append({'shares': t_size, 'strike': self.bb.top[0], 'age': 0})
            return

        # Initial call sale on upper band crossover + RSI > 60 + volume > SMA(volume, 20)
        crossover = (self.data.close[-1] <= self.bb.top[-1]) and (self.data.close[0] > self.bb.top[0])
        if crossover and self.rsi[0] > 60 and self.data.volume[0] > self.vol_sma[0]:
            if self.position.size > 0:
                t_size = self.get_tranche_size()
                if t_size > 0:
                    self.order = self.sell(size=t_size)
                    self.tranches.append({'shares': t_size, 'strike': self.bb.top[0], 'age': 0})
