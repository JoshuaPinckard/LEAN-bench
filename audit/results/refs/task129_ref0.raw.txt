import backtrader as bt


class FallingChannelBreakoutStrategy(bt.Strategy):
    params = (
        ('bb_period', 20),
        ('bb_dev', 2),
        ('vol_period', 10),
        ('risk_pct', 0.02),
        ('dd_threshold', 0.10),
    )

    def __init__(self):
        self.bb = bt.indicators.BollingerBands(
            self.data.close, period=self.p.bb_period, devfactor=self.p.bb_dev
        )
        self.vol_sma = bt.indicators.SimpleMovingAverage(
            self.data.volume, period=self.p.vol_period
        )
        self.order = None
        self.tp_order = None
        self.sl_order = None
        self.pending_channel_height = None
        self.pending_stop_price = None
        self.peak_value = self.broker.getvalue()

    def _falling_structure(self):
        for i in range(1, 4):
            if not (self.data.high[-i] < self.data.high[-i - 1]):
                return False
            if not (self.data.low[-i] < self.data.low[-i - 1]):
                return False
        return True

    def _place_exit_orders(self, fill_price):
        size = self.position.size
        tp_price = fill_price + self.pending_channel_height
        sl_price = self.pending_stop_price
        self.tp_order = self.sell(exectype=bt.Order.Limit, price=tp_price, size=size)
        self.sl_order = self.sell(
            exectype=bt.Order.Stop, price=sl_price, size=size, oco=self.tp_order
        )

    def next(self):
        current_value = self.broker.getvalue()
        if current_value > self.peak_value:
            self.peak_value = current_value
        drawdown = (
            (self.peak_value - current_value) / self.peak_value
            if self.peak_value > 0 else 0.0
        )

        if self.position or self.order:
            return

        if drawdown > self.p.dd_threshold:
            return

        upper = self.bb.lines.top[0]
        lower = self.bb.lines.bot[0]

        if self.data.close[0] <= upper:
            return

        if not self._falling_structure():
            return

        if self.data.volume[0] <= self.vol_sma[-1]:
            return

        channel_height = upper - lower
        stop_price = lower
        entry_price_est = self.data.close[0]
        risk_per_share = entry_price_est - stop_price
        if risk_per_share <= 0:
            return

        risk_amount = self.broker.getvalue() * self.p.risk_pct
        size = int(risk_amount / risk_per_share)
        if size < 1:
            return

        self.pending_channel_height = channel_height
        self.pending_stop_price = stop_price
        self.order = self.buy(size=size)

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order == self.order:
            if order.status == order.Completed and order.isbuy():
                self._place_exit_orders(order.executed.price)
            self.order = None
        elif order == self.tp_order:
            self.tp_order = None
        elif order == self.sl_order:
            self.sl_order = None
