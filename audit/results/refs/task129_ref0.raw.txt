# complete strategy class
import backtrader as bt


class FallingChannelBreakout(bt.Strategy):
    params = (
        ('bb_period', 20),      # Bollinger Bands period
        ('bb_dev', 2.0),        # Bollinger Bands standard deviation
        ('vol_period', 10),     # lookback for average-volume filter
        ('struct_len', 3),      # required consecutive lower highs / lower lows
        ('risk_pct', 0.01),     # fixed risk per trade (1% of equity)
        ('dd_threshold', 0.10), # max drawdown allowed before taking a new trade
    )

    def __init__(self):
        # Channel = Bollinger Bands(20, 2): mid = midline, top = resistance, bot = support
        self.bb = bt.indicators.BollingerBands(
            self.data, period=self.p.bb_period, devfactor=self.p.bb_dev)

        self.order = None
        self.stop_price = None
        self.target_price = None

        # peak equity for the drawdown gate
        self.peak_value = self.broker.getvalue()

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return
        if order.status in [order.Completed, order.Canceled,
                            order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        # --- track peak equity for drawdown control ---
        value = self.broker.getvalue()
        if value > self.peak_value:
            self.peak_value = value

        # pending order guard
        if self.order:
            return

        # --- exit management for an open position ---
        if self.position:
            # stop-loss takes priority if both levels are touched in the same bar
            if self.data.low[0] <= self.stop_price:
                self.order = self.close()
            elif self.data.high[0] >= self.target_price:
                self.order = self.close()
            return

        # --- drawdown gate: no new trade if drawdown exceeds threshold ---
        drawdown = (self.peak_value - value) / self.peak_value
        if drawdown > self.p.dd_threshold:
            return

        # --- falling channel structure: >= 3 consecutive lower highs & lower lows ---
        h = self.data.high
        l = self.data.low
        lower_highs = True
        lower_lows = True
        for i in range(1, self.p.struct_len + 1):
            if not (h[-i] < h[-i - 1]):
                lower_highs = False
                break
        if lower_highs:
            for i in range(1, self.p.struct_len + 1):
                if not (l[-i] < l[-i - 1]):
                    lower_lows = False
                    break

        if not (lower_highs and lower_lows):
            return

        # --- breakout: close above the upper channel boundary ---
        if not (self.data.close[0] > self.bb.top[0]):
            return

        # --- volume filter: breakout volume above average of previous 10 periods ---
        avg_vol = 0.0
        for i in range(1, self.p.vol_period + 1):
            avg_vol += self.data.volume[-i]
        avg_vol /= self.p.vol_period
        if not (self.data.volume[0] > avg_vol):
            return

        # --- levels captured at the time of the breakout ---
        upper = self.bb.top[0]
        lower = self.bb.bot[0]
        entry_ref = self.data.close[0]
        channel_height = upper - lower

        stop_price = lower                          # stop below/at the lower band
        target_price = entry_ref + channel_height   # TP a channel-height away

        risk_per_unit = entry_ref - stop_price
        if risk_per_unit <= 0:
            return

        # --- risk-based position size ---
        risk_amount = self.p.risk_pct * value
        size = risk_amount / risk_per_unit
        if size <= 0:
            return

        self.stop_price = stop_price
        self.target_price = target_price
        self.order = self.buy(size=size)
