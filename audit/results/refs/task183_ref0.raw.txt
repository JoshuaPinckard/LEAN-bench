import backtrader as bt


class BreakoutStrategy(bt.Strategy):
    params = dict(
        lookback=10,
        size_pct=0.01,
        stop_loss_pct=0.02,
    )

    def __init__(self):
        self.highest_close = bt.indicators.Highest(
            self.data.close(-1), period=self.p.lookback
        )
        self.lowest_close = bt.indicators.Lowest(
            self.data.close(-1), period=self.p.lookback
        )
        self.order = None
        self.stop_order = None
        self.entry_price = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status == order.Completed:
            if order.isbuy():
                self.entry_price = order.executed.price
                stop_price = self.entry_price * (1 - self.p.stop_loss_pct)
                self.stop_order = self.sell(
                    exectype=bt.Order.Stop,
                    price=stop_price,
                    size=order.executed.size,
                )
            elif order.issell():
                if order is self.stop_order:
                    self.stop_order = None
                self.entry_price = None
            self.order = None
        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            if order is self.stop_order:
                self.stop_order = None
            self.order = None

    def next(self):
        if self.order:
            return

        if not self.position:
            if self.data.close[0] > self.highest_close[0]:
                size = (self.broker.getvalue() * self.p.size_pct) / self.data.close[0]
                self.order = self.buy(size=size)
        else:
            if self.data.close[0] < self.lowest_close[0]:
                if self.stop_order is not None:
                    self.cancel(self.stop_order)
                    self.stop_order = None
                self.order = self.close()
