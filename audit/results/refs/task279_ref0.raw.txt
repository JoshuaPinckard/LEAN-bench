import backtrader as bt


class IntradayMeanReversionVolAdjStrategy(bt.Strategy):

    def __init__(self):
        self.boll = bt.indicators.BollingerBands(self.data.close, period=20, devfactor=2)
        self.sma = bt.indicators.SMA(self.data.close, period=20)
        self.rsi = bt.indicators.RSI(self.data.close, period=14)
        self.atr = bt.indicators.ATR(self.data, period=14)
        self.crossover = bt.indicators.CrossOver(self.data.close, self.boll.lines.bot)

        self.order = None
        self.trail_high = None
        self.tp_done = False
        self.base_size = 0
        self.eod_reduced_today = False
        self.no_trade_today = False
        self.last_date = None
        self.last_atr_pct = None

    def notify_order(self, order):
        if order.status in (order.Completed, order.Canceled, order.Margin, order.Rejected):
            self.order = None
            if order.status == order.Completed and self.position.size == 0:
                self.trail_high = None
                self.tp_done = False

    def next(self):
        cur_date = self.data.datetime.date(0)
        t = self.data.datetime.time(0)
        minute_of_day = t.hour * 60 + t.minute

        if cur_date != self.last_date:
            bb_width = (self.boll.lines.top[0] - self.boll.lines.bot[0]) / self.boll.lines.mid[0]
            no_trade = bb_width > 0.015
            if self.last_atr_pct is not None and self.last_atr_pct > 0.015:
                no_trade = True
            self.no_trade_today = no_trade
            self.eod_reduced_today = False
            self.last_date = cur_date

        self.last_atr_pct = self.atr[0] / self.data.close[0]

        if self.order:
            return

        in_entry_window = 570 <= minute_of_day < 600
        in_eod_window = 900 <= minute_of_day < 930

        if self.position:
            self.trail_high = self.data.high[0] if self.trail_high is None else max(self.trail_high, self.data.high[0])

            if self.data.close[0] <= self.trail_high * 0.98:
                self.order = self.close()
                return

            if not self.tp_done and self.data.close[0] >= self.position.price * 1.003:
                sell_size = self.position.size // 2
                if sell_size > 0:
                    self.tp_done = True
                    self.order = self.sell(size=sell_size)
                    return

            if in_eod_window and not self.eod_reduced_today:
                target_size = int(0.5 * self.base_size)
                self.eod_reduced_today = True
                if self.position.size > target_size:
                    self.order = self.sell(size=self.position.size - target_size)
                    return
        else:
            if self.no_trade_today or not in_entry_window:
                return

            bb_width = (self.boll.lines.top[0] - self.boll.lines.bot[0]) / self.boll.lines.mid[0]
            atr_pct = self.atr[0] / self.data.close[0]

            if self.crossover[0] > 0 and self.rsi[0] < 45 and (bb_width <= 0.005 or atr_pct <= 0.008):
                price = self.data.close[0]
                value = self.broker.getvalue()
                target_pct = 1.25 if atr_pct > 0.01 else 1.50
                target_value = target_pct * value
                commission = self.broker.getcommissioninfo(self.data).p.commission
                max_affordable_value = self.broker.getcash() / (1 + commission)
                actual_value = min(target_value, max_affordable_value)
                size = int(actual_value / price)
                if size > 0:
                    self.base_size = int(value / price)
                    self.order = self.buy(size=size)
