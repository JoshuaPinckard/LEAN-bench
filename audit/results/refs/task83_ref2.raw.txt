import datetime
import math
import backtrader as bt


class OpeningRangeBreakout(bt.Strategy):
    def __init__(self):
        self.order = None
        self.current_date = None
        self.or_high = None
        self.or_low = None
        self.long_traded = False
        self.short_traded = False

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        if self.order:
            return

        dt = self.data.datetime.datetime(0)
        bar_date = dt.date()
        bar_time = dt.time()

        # First 5-minute bar of a new trading day establishes the Opening Range
        if bar_date != self.current_date:
            self.current_date = bar_date
            self.or_high = self.data.high[0]
            self.or_low = self.data.low[0]
            self.long_traded = False
            self.short_traded = False
            return

        # End of day forcible exit at or near 16:00 EST
        if bar_time >= datetime.time(15, 55):
            if self.position:
                self.order = self.close()
            return

        # Position Risk Management (Stop Loss and Take Profit)
        if self.position:
            entry_price = self.position.price
            if self.position.size > 0:
                sl_price = entry_price * (1.0 - 0.005)
                tp_price = entry_price * (1.0 + 0.010)
                if self.data.low[0] <= sl_price or self.data.high[0] >= tp_price:
                    self.order = self.close()
            elif self.position.size < 0:
                sl_price = entry_price * (1.0 + 0.005)
                tp_price = entry_price * (1.0 - 0.010)
                if self.data.high[0] >= sl_price or self.data.low[0] <= tp_price:
                    self.order = self.close()
            return

        if self.or_high is None or self.or_low is None:
            return

        # Position Sizing: 1% of total portfolio capital per trade
        capital = self.broker.getvalue()
        target_value = capital * 0.01
        close_price = self.data.close[0]

        if close_price <= 0:
            return

        size = int(target_value / close_price)
        if size <= 0:
            return

        # Entry Signals
        if not self.long_traded and self.data.close[0] > self.or_high:
            self.order = self.buy(size=size)
            self.long_traded = True
        elif not self.short_traded and self.data.close[0] < self.or_low:
            self.order = self.sell(size=size)
            self.short_traded = True
