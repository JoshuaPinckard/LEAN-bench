import backtrader as bt


class SmaIchimokuStrategy(bt.Strategy):
    params = (
        ('tenkan_period', 9),
        ('kijun_period', 26),
        ('span_b_period', 52),
        ('stake', 100),
    )

    def __init__(self):
        self.tenkan = bt.indicators.SMA(self.data.close, period=self.p.tenkan_period)
        self.kijun = bt.indicators.SMA(self.data.close, period=self.p.kijun_period)
        self.span_b = bt.indicators.SMA(self.data.close, period=self.p.span_b_period)

        self.span_a = (self.tenkan + self.kijun) / 2.0

        self.cloud_top = bt.If(self.span_a > self.span_b, self.span_a, self.span_b)
        self.cloud_bottom = bt.If(self.span_a < self.span_b, self.span_a, self.span_b)

        self.crossover = bt.indicators.CrossOver(self.tenkan, self.kijun)
        self.order = None

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        if self.order:
            return

        if not self.position:
            if self.crossover[0] > 0 and self.data.close[0] > self.cloud_top[0]:
                self.order = self.buy(size=self.p.stake)
            elif self.crossover[0] < 0 and self.data.close[0] < self.cloud_bottom[0]:
                self.order = self.sell(size=self.p.stake)
        elif self.position.size > 0:
            if self.crossover[0] < 0 or self.data.low[0] <= self.kijun[0]:
                self.order = self.close()
        elif self.position.size < 0:
            if self.crossover[0] > 0 or self.data.high[0] >= self.kijun[0]:
                self.order = self.close()
