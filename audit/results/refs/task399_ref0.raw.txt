import backtrader as bt


class Supertrend(bt.Indicator):
    lines = ('supertrend', 'trend', 'final_upper', 'final_lower')
    params = (('period', 10), ('multiplier', 3.0))

    def __init__(self):
        self.atr = bt.indicators.ATR(self.data, period=self.p.period)
        self._seeded = False

    def next(self):
        hl2 = (self.data.high[0] + self.data.low[0]) / 2.0
        basic_upper = hl2 + self.p.multiplier * self.atr[0]
        basic_lower = hl2 - self.p.multiplier * self.atr[0]

        if not self._seeded:
            final_upper = basic_upper
            final_lower = basic_lower
            trend = 1
            self._seeded = True
        else:
            prior_final_upper = self.l.final_upper[-1]
            prior_final_lower = self.l.final_lower[-1]
            prior_close = self.data.close[-1]
            current_close = self.data.close[0]
            prior_trend = self.l.trend[-1]

            final_upper = (basic_upper if (basic_upper < prior_final_upper
                                            or prior_close > prior_final_upper)
                           else prior_final_upper)
            final_lower = (basic_lower if (basic_lower > prior_final_lower
                                            or prior_close < prior_final_lower)
                           else prior_final_lower)

            if current_close > prior_final_upper:
                trend = 1
            elif current_close < prior_final_lower:
                trend = -1
            else:
                trend = prior_trend

        self.l.final_upper[0] = final_upper
        self.l.final_lower[0] = final_lower
        self.l.trend[0] = trend
        self.l.supertrend[0] = final_lower if trend == 1 else final_upper


class SupertrendIntradayStrategy(bt.Strategy):
    params = (
        ('atr_period', 10),
        ('multiplier', 3.0),
        ('volume_period', 20),
        ('percent_equity', 0.25),
    )

    def __init__(self):
        self.st = Supertrend(self.data, period=self.p.atr_period, multiplier=self.p.multiplier)
        self.volume_sma = bt.indicators.SMA(self.data.volume, period=self.p.volume_period)
        self.orders = []

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            if order in self.orders:
                self.orders.remove(order)

    def _target_size(self):
        equity = self.broker.getvalue()
        price = self.data.close[0]
        return int((equity * self.p.percent_equity) / price)

    def next(self):
        if self.orders:
            return
        if len(self.st) < 2:
            return

        trend = self.st.trend[0]
        prior_trend = self.st.trend[-1]
        flip_up = prior_trend == -1 and trend == 1
        flip_down = prior_trend == 1 and trend == -1

        if not (flip_up or flip_down):
            return

        is_long = self.position.size > 0
        is_short = self.position.size < 0

        if (is_long and flip_down) or (is_short and flip_up):
            self.orders.append(self.close())

        volume_filter = self.data.volume[0] > self.volume_sma[0]
        if volume_filter:
            if flip_up and not is_long:
                size = self._target_size()
                if size > 0:
                    self.orders.append(self.buy(size=size))
            elif flip_down and not is_short:
                size = self._target_size()
                if size > 0:
                    self.orders.append(self.sell(size=size))
