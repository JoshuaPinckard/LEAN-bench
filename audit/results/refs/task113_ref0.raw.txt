# complete strategy class
import backtrader as bt
import math
import statistics
import collections


class CostAwarePortfolioOptimization(bt.Strategy):
    # Cost-aware portfolio optimization: at every step choose the asset weight w
    # that maximizes  (r - I(t))w - λ w'Σ w - Θ|w|^(3/2),  with t = |w - w0|,
    # then rebalance toward that target weight.
    params = dict(
        lookback=252,     # T = 1 year of daily bars used to estimate r and Σ
        lam=2.0,          # λ : risk-aversion coefficient
        theta=0.01,       # Θ : market-impact / transaction-cost coefficient
        w_min=0.0,        # lower box bound on the asset weight (long-only)
        w_max=1.0,        # upper box bound on the asset weight (max-weight constraint)
        grid=200,         # resolution of the 1-D weight grid used by the optimizer
    )

    def __init__(self):
        self.order = None
        self.rets = collections.deque(maxlen=self.p.lookback)
        self.prev_close = None

    def _objective(self, w, w0, r, sigma2):
        # (r - I(t)) w - λ σ² w² - Θ |w|^(3/2),  I(t) = Θ t^(3/2),  t = |w - w0|
        t = abs(w - w0)
        it = self.p.theta * (t ** 1.5)                       # impact cost I(t)
        return ((r - it) * w
                - self.p.lam * sigma2 * w * w
                - self.p.theta * (abs(w) ** 1.5))

    def next(self):
        # maintain the rolling window of historical simple returns
        c = self.data.close[0]
        if self.prev_close is not None and self.prev_close > 0.0:
            self.rets.append(c / self.prev_close - 1.0)
        self.prev_close = c

        if self.order:
            return
        if len(self.rets) < self.p.lookback:
            return

        # expected return r and risk Σ (variance) over the reinvestment period (annualized)
        mean_d = statistics.fmean(self.rets)
        var_d = statistics.pvariance(self.rets)
        r = mean_d * self.p.lookback
        sigma2 = var_d * self.p.lookback

        # current portfolio weight w0 of the asset (remainder held in cash)
        value = self.broker.getvalue()
        w0 = (self.position.size * c) / value if value > 0.0 else 0.0
        w0 = min(max(w0, self.p.w_min), self.p.w_max)

        # solve w* = argmax objective over the box [w_min, w_max] via grid search;
        # the current weight w0 is included so "do nothing" is always a candidate,
        # which is how the model declines to trade when costs exceed the benefit.
        best_w = w0
        best_f = self._objective(w0, w0, r, sigma2)
        n = self.p.grid
        span = self.p.w_max - self.p.w_min
        for i in range(n + 1):
            w = self.p.w_min + span * i / n
            f = self._objective(w, w0, r, sigma2)
            if f > best_f:
                best_f = f
                best_w = w

        # rebalance toward the optimal target weight w*
        self.order = self.order_target_percent(target=best_w)

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return
        if order.status in (order.Completed, order.Canceled,
                            order.Margin, order.Rejected):
            self.order = None
