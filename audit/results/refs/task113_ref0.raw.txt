```python
import statistics

import backtrader as bt


class PortfolioCostOptimizationStrategy(bt.Strategy):
    params = dict(
        lookback=252,      # trading days approximating the T = 1 year reinvestment period
        lam=1.0,           # risk-aversion coefficient (lambda)
        theta=1.0,         # market-impact cost scaling coefficient (Theta)
        max_weight=0.10,   # per-asset weight cap (spec example: "no more than 10%")
        grid_points=201,   # resolution of the bounded grid search for w*
    )

    def __init__(self):
        self.order = None

    def notify_order(self, order):
        if order.status in (order.Completed, order.Canceled, order.Margin, order.Rejected):
            self.order = None

    def next(self):
        if self.order:
            return

        if len(self.data) < self.p.lookback + 1:
            return

        closes = self.data.close.get(size=self.p.lookback + 1)
        daily_returns = [
            (closes[i] - closes[i - 1]) / closes[i - 1]
            for i in range(1, len(closes))
        ]

        r = statistics.mean(daily_returns) * self.p.lookback
        sigma2 = statistics.pvariance(daily_returns) * self.p.lookback

        portfolio_value = self.broker.getvalue()
        price = self.data.close[0]
        current_size = self.position.size if self.position else 0.0
        w0 = (current_size * price) / portfolio_value if portfolio_value else 0.0

        lam = self.p.lam
        theta = self.p.theta
        wmax = self.p.max_weight
        n = self.p.grid_points

        best_w = 0.0
        best_val = float('-inf')
        for i in range(n):
            w = wmax * i / (n - 1)
            t = abs(w - w0)
            impact_vec = theta * (t ** 1.5)
            obj = (r - impact_vec) * w - lam * sigma2 * (w ** 2) - theta * (abs(w) ** 1.5)
            if obj > best_val:
                best_val = obj
                best_w = w

        self.order = self.order_target_percent(target=best_w)
```
