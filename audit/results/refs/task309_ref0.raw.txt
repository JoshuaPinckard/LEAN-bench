import backtrader as bt


class MeanReversionZScoreStrategy(bt.Strategy):
    params = dict(
        period=20,
        entry_threshold=2.0,
        exit_threshold=0.0,
        stop_pct=0.02,
    )

    def __init__(self):
        self.sma = bt.indicators.SMA(self.data.close, period=self.p.period)
        self.stddev = bt.indicators.StdDev(self.data.close, period=self.p.period)
        self.order = None
        self.entry_price = None

    def notify_order(self, order):
        if order.status == order.Completed:
            self.entry_price = order.executed.price
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        if self.order:
            return

        if self.stddev[0] == 0:
            return

        zscore = (self.data.close[0] - self.sma[0]) / self.stddev[0]

        if self.position:
            if self.position.size > 0:
                if self.data.close[0] <= self.entry_price * (1 - self.p.stop_pct):
                    self.order = self.close()
                    return
                if zscore >= self.p.exit_threshold:
                    self.order = self.close()
                    return
            elif self.position.size < 0:
                if self.data.close[0] >= self.entry_price * (1 + self.p.stop_pct):
                    self.order = self.close()
                    return
                if zscore <= self.p.exit_threshold:
                    self.order = self.close()
                    return
        else:
            if zscore < -self.p.entry_threshold:
                self.order = self.buy()
            elif zscore > self.p.entry_threshold:
                self.order = self.sell()
