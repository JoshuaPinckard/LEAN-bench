import backtrader as bt


class MeanReversionZScore(bt.Strategy):
    # Default parameters for the medium-risk profile as stated in the spec.
    params = (
        ('ma_period', 20),        # 20-period SMA
        ('std_period', 20),       # 20-period StdDev
        ('entry_threshold', 2.0), # go long < -2, short > +2
        ('exit_threshold', 0.0),  # cover/reverse when Z returns near 0
        ('stop_loss', 0.02),      # 2% hard stop-loss from entry
    )

    def __init__(self):
        # 20-period SMA and 20-period StdDev of price (close).
        self.sma = bt.indicators.SimpleMovingAverage(
            self.data.close, period=self.p.ma_period)
        self.stddev = bt.indicators.StandardDeviation(
            self.data.close, period=self.p.std_period)
        # Z-Score of price versus the MA.
        self.zscore = (self.data.close - self.sma) / self.stddev

        self.order = None
        self.entry_price = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return
        if order.status == order.Completed:
            # After a closing fill the position is flat; after an entry fill
            # it is set. Track the entry price for the hard stop.
            if not self.position:
                self.entry_price = None
            else:
                self.entry_price = order.executed.price
        if order.status in [order.Completed, order.Canceled,
                             order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        if self.order:
            return

        z = self.zscore[0]

        if self.position:
            if self.position.size > 0:
                # Long: 2% hard stop-loss from entry (intrabar touch).
                if self.entry_price is not None and \
                        self.data.low[0] <= self.entry_price * (1.0 - self.p.stop_loss):
                    self.order = self.close()
                    return
                # Z-Score returned near 0 -> cover.
                if z >= self.p.exit_threshold:
                    self.order = self.close()
                    return
            elif self.position.size < 0:
                # Short: 2% hard stop-loss from entry (intrabar touch).
                if self.entry_price is not None and \
                        self.data.high[0] >= self.entry_price * (1.0 + self.p.stop_loss):
                    self.order = self.close()
                    return
                # Z-Score returned near 0 -> cover.
                if z <= self.p.exit_threshold:
                    self.order = self.close()
                    return
        else:
            # Entries at the +/- 2 thresholds.
            if z < -self.p.entry_threshold:
                self.order = self.buy()
            elif z > self.p.entry_threshold:
                self.order = self.sell()
