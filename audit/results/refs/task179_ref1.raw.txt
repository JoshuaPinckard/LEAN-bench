import backtrader as bt


class BuyingSpikeMomentum(bt.Strategy):
    params = (
        ('risk_fraction', 0.02),
        ('stop_loss_fraction', 0.02),
    )

    def __init__(self):
        self.price_sma = bt.indicators.SMA(self.data.close, period=20)
        self.volume_sma_5 = bt.indicators.SMA(self.data.volume, period=5)
        self.volume_sma_20 = bt.indicators.SMA(self.data.volume, period=20)
        self.stochastic = bt.indicators.Stochastic(
            self.data,
            period=14,
            period_dfast=3,
            period_dslow=3,
        )
        self.stoch_cross = bt.indicators.CrossOver(
            self.stochastic.percK,
            self.stochastic.percD,
        )
        self.atr = bt.indicators.ATR(self.data, period=14)

        self.order = None
        self.stop_order = None
        self.entry_bar = None

    def notify_order(self, order):
        if order.status in (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            if order is self.stop_order:
                self.stop_order = None
                if order.status == order.Completed:
                    self.entry_bar = None

            if order is self.order:
                self.order = None

                if order.status == order.Completed:
                    if order.isbuy():
                        self.entry_bar = len(self)
                        self.stop_order = self.sell(
                            size=order.executed.size,
                            exectype=bt.Order.Stop,
                            price=order.executed.price * (
                                1.0 - self.p.stop_loss_fraction
                            ),
                        )
                    else:
                        self.entry_bar = None
                        if self.stop_order is not None:
                            self.cancel(self.stop_order)

    def next(self):
        if self.position:
            held_bars = len(self) - self.entry_bar

            if (
                self.data.close[0] < self.price_sma[0]
                or held_bars >= 10
            ):
                if self.order is None:
                    if self.stop_order is not None:
                        self.cancel(self.stop_order)
                    self.order = self.close()
            return

        if self.order is not None:
            return

        volume_spike = self.volume_sma_5[0] >= (
            3.0 * self.volume_sma_20[0]
        )
        buying_pressure_confirmation = (
            self.stoch_cross[0] > 0
            or self.data.close[0] > self.data.close[-1]
        )
        sufficient_liquidity = (
            self.volume_sma_20[0] * self.price_sma[0] >= 10000000
        )

        if (
            self.data.close[0] > self.price_sma[0]
            and volume_spike
            and self.stochastic.percK[0] > 80
            and buying_pressure_confirmation
            and self.atr[0] / self.data.close[0] > 0.015
            and sufficient_liquidity
        ):
            size = (
                self.broker.getvalue() * self.p.risk_fraction
                / self.data.close[0]
            )
            if size > 0:
                self.order = self.buy(size=size)
