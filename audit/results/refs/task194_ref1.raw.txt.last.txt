import backtrader as bt


class DownwardReversalStrategy(bt.Strategy):
    params = (
        ('risk_fraction', 0.01),
    )

    def __init__(self):
        self.ema50 = bt.indicators.EMA(self.data.close, period=50)
        self.bb = bt.indicators.BollingerBands(self.data.close, period=20, devfactor=2)
        self.atr14 = bt.indicators.ATR(self.data, period=14)
        self.atr_sma50 = bt.indicators.SMA(self.atr14, period=50)
        self.volume_sma20 = bt.indicators.SMA(self.data.volume, period=20)

        self.entry_order = None
        self.stop_order = None
        self.target_order = None
        self.exit_pending = False
        self.signal_high = None
        self.signal_atr = None

    def next(self):
        if self.exit_pending:
            return

        if self.position:
            if self._opposite_signal():
                self.exit_pending = True
                if self.stop_order:
                    self.cancel(self.stop_order)
                if self.target_order:
                    self.cancel(self.target_order)
                if not self.stop_order and not self.target_order:
                    self.close()
            return

        if self.entry_order:
            return

        if self._short_signal():
            stop_price = self.data.high[0] + 0.5 * self.atr14[0]
            risk_per_share = stop_price - self.data.close[0]
            if risk_per_share > 0:
                allowed_risk = self.broker.getvalue() * self.p.risk_fraction
                size = allowed_risk / risk_per_share
                self.signal_high = self.data.high[0]
                self.signal_atr = self.atr14[0]
                self.entry_order = self.sell(size=size)

    def notify_order(self, order):
        if order.status in (order.Completed, order.Canceled, order.Margin, order.Rejected):
            if order == self.entry_order:
                self.entry_order = None

                if order.status == order.Completed:
                    stop_price = self.signal_high + 0.5 * self.signal_atr
                    risk = stop_price - order.executed.price
                    if risk > 0:
                        target_price = order.executed.price - risk
                        self.stop_order = self.buy(
                            size=abs(self.position.size),
                            exectype=bt.Order.Stop,
                            price=stop_price,
                        )
                        self.target_order = self.buy(
                            size=abs(self.position.size),
                            exectype=bt.Order.Limit,
                            price=target_price,
                        )

            elif order == self.stop_order:
                self.stop_order = None
                if order.status == order.Completed and self.target_order:
                    self.cancel(self.target_order)

            elif order == self.target_order:
                self.target_order = None
                if order.status == order.Completed and self.stop_order:
                    self.cancel(self.stop_order)

            if self.exit_pending and not self.stop_order and not self.target_order:
                self.exit_pending = False
                if self.position:
                    self.close()

    def _short_signal(self):
        high = self.data.high[0]
        low = self.data.low[0]
        open_ = self.data.open[0]
        close = self.data.close[0]
        candle_range = high - low

        if candle_range <= 0:
            return False

        upper_wick = high - max(open_, close)
        lower_wick = min(open_, close) - low
        body = abs(close - open_)

        return (
            close < self.ema50[0]
            and self.atr14[0] > self.atr_sma50[0]
            and high >= self.bb.top[0]
            and close < open_
            and (close - low) / candle_range <= 0.33
            and upper_wick >= 2 * body
            and upper_wick >= 1.5 * lower_wick
            and self.data.volume[0] >= 1.5 * self.volume_sma20[0]
            and upper_wick / candle_range >= 0.5
        )

    def _opposite_signal(self):
        high = self.data.high[0]
        low = self.data.low[0]
        open_ = self.data.open[0]
        close = self.data.close[0]
        candle_range = high - low

        if candle_range <= 0:
            return False

        return (
            close > open_
            and (high - close) / candle_range <= 0.33
            and (min(open_, close) - low) >= 2 * abs(close - open_)
            and low <= self.bb.bot[0]
            and self.data.volume[0] >= 1.5 * self.volume_sma20[0]
        )
