import backtrader as bt
import math


class SequentialBuyOrderStrategy(bt.Strategy):
    params = (
        ("first_position_size", 1),
        ("second_position_size", 1),
        ("start_hour", 10),
        ("end_hour", 16),
    )

    def __init__(self):
        self.first_limit_order = None
        self.second_market_order = None
        self.second_size = None

    def _within_time_window(self):
        dt = self.data.datetime.datetime(0)
        return self.p.start_hour <= dt.hour <= self.p.end_hour

    def _cash_required(self, size, price):
        commission_info = self.broker.getcommissioninfo(self.data)
        return (size * price) + commission_info.getcommission(size, price)

    def next(self):
        if self.position or self.first_limit_order or self.second_market_order:
            return

        if not self._within_time_window():
            return

        limit_price = self.data.close[0]
        first_size = self.p.first_position_size
        second_size = self.p.second_position_size

        required_cash = (
            self._cash_required(first_size, limit_price)
            + self._cash_required(second_size, limit_price)
        )

        if self.broker.getcash() >= required_cash:
            self.second_size = second_size
            self.first_limit_order = self.buy(
                size=first_size,
                exectype=bt.Order.Limit,
                price=limit_price,
            )

    def notify_order(self, order):
        if order.status in (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            if order is self.first_limit_order:
                self.first_limit_order = None

                if order.status == order.Completed:
                    estimated_cost = self._cash_required(
                        self.second_size, order.executed.price
                    )
                    if self.broker.getcash() >= estimated_cost:
                        self.second_market_order = self.buy(size=self.second_size)

            elif order is self.second_market_order:
                self.second_market_order = None
