import backtrader as bt
import math


class DistributedSellStrategy(bt.Strategy):
    def __init__(self):
        self.order = None
        self.initial_investment = None
        self.cost_basis = None
        self.next_threshold = None
        self.entered = False

    def notify_order(self, order):
        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            self.order = None

    def next(self):
        if not self.entered:
            self.initial_investment = 400 * self.data.close[0]
            self.cost_basis = self.initial_investment
            self.next_threshold = self.initial_investment * 2
            self.order = self.buy(size=400)
            self.entered = True
            return

        if self.order is not None or not self.position:
            return

        position_value = self.position.size * self.data.close[0]

        if position_value >= self.next_threshold:
            shares_to_sell = math.floor(self.position.size * 0.5)
            if shares_to_sell > 0:
                self.order = self.sell(size=shares_to_sell)
                self.cost_basis = self.cost_basis * 0.5
                self.next_threshold = self.next_threshold * 2
