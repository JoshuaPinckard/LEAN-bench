import backtrader as bt


class GridMartingaleStrategy(bt.Strategy):
    params = dict(
        grid_distance_pct=0.01,
        take_profit_pct=0.005,
        reinvest_multiplier=2,
        max_reinvest_levels=5,
        base_size=1,
    )

    def __init__(self):
        self.direction = None
        self.level = 0
        self.total_size = 0
        self.total_cost = 0.0
        self.last_entry_price = None

        self.grid_buy_order = None
        self.grid_sell_order = None
        self.reinvest_order = None
        self.tp_order = None

    def next(self):
        if self.direction is None:
            if self.grid_buy_order is not None:
                self.cancel(self.grid_buy_order)
            if self.grid_sell_order is not None:
                self.cancel(self.grid_sell_order)
            price = self.data.close[0]
            buy_price = price * (1 - self.p.grid_distance_pct)
            sell_price = price * (1 + self.p.grid_distance_pct)
            self.grid_buy_order = self.buy(
                exectype=bt.Order.Limit, price=buy_price, size=self.p.base_size)
            self.grid_sell_order = self.sell(
                exectype=bt.Order.Limit, price=sell_price, size=self.p.base_size)

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order == self.grid_buy_order:
            if order.status == order.Completed:
                self.grid_buy_order = None
                self._on_entry_fill(order, 'long')
                if self.grid_sell_order is not None:
                    self.cancel(self.grid_sell_order)
                    self.grid_sell_order = None
            else:
                self.grid_buy_order = None
        elif order == self.grid_sell_order:
            if order.status == order.Completed:
                self.grid_sell_order = None
                self._on_entry_fill(order, 'short')
                if self.grid_buy_order is not None:
                    self.cancel(self.grid_buy_order)
                    self.grid_buy_order = None
            else:
                self.grid_sell_order = None
        elif order == self.reinvest_order:
            if order.status == order.Completed:
                self.reinvest_order = None
                self._on_entry_fill(order, self.direction)
            else:
                self.reinvest_order = None
        elif order == self.tp_order:
            if order.status == order.Completed:
                self.tp_order = None
                self._on_position_closed()
            else:
                self.tp_order = None

    def _on_entry_fill(self, order, direction):
        price = order.executed.price
        size = abs(order.executed.size)
        if self.direction is None:
            self.direction = direction
            self.total_cost = price * size
            self.total_size = size
            self.level = 0
        else:
            self.total_cost += price * size
            self.total_size += size
            self.level += 1
        self.last_entry_price = price
        avg_price = self.total_cost / self.total_size
        self._update_tp_order(avg_price)
        self._update_reinvest_order()

    def _update_tp_order(self, avg_price):
        if self.tp_order is not None:
            self.cancel(self.tp_order)
        if self.direction == 'long':
            tp_price = avg_price * (1 + self.p.take_profit_pct)
        else:
            tp_price = avg_price * (1 - self.p.take_profit_pct)
        self.tp_order = self.close(exectype=bt.Order.Limit, price=tp_price)

    def _update_reinvest_order(self):
        if self.reinvest_order is not None:
            self.cancel(self.reinvest_order)
            self.reinvest_order = None
        if self.level >= self.p.max_reinvest_levels:
            return
        next_size = self.p.base_size * (self.p.reinvest_multiplier ** (self.level + 1))
        if self.direction == 'long':
            next_price = self.last_entry_price * (1 - self.p.grid_distance_pct)
            self.reinvest_order = self.buy(
                exectype=bt.Order.Limit, price=next_price, size=next_size)
        else:
            next_price = self.last_entry_price * (1 + self.p.grid_distance_pct)
            self.reinvest_order = self.sell(
                exectype=bt.Order.Limit, price=next_price, size=next_size)

    def _on_position_closed(self):
        if self.reinvest_order is not None:
            self.cancel(self.reinvest_order)
            self.reinvest_order = None
        self.direction = None
        self.level = 0
        self.total_size = 0
        self.total_cost = 0.0
        self.last_entry_price = None
