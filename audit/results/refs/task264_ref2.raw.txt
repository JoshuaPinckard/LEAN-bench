import backtrader as bt


class ReferenceBreakoutStrategy(bt.Strategy):
    params = (
        ("sma_period", 200),
        ("bb_period", 20),
        ("bb_dev", 2.0),
        ("atr_period", 14),
        ("sma_width_period", 100),
        ("sma_atr_period", 100),
        ("consolidation_bars", 10),
        ("consolidation_mult", 0.8),
        ("vol_min", 0.7),
        ("vol_max", 1.3),
        ("risk_pct", 0.01),
        ("stop_atr_mult", 3.0),
        ("tp_r_mult", 3.0),
        ("be_r_mult", 1.5),
        ("min_holding_period", 60),
        ("cooldown_period", 20),
    )

    def __init__(self):
        self.sma200 = bt.ind.SMA(self.data.close, period=self.p.sma_period)
        self.boll = bt.ind.BollingerBands(
            self.data.close, period=self.p.bb_period, devfactor=self.p.bb_dev
        )
        self.width = self.boll.lines.top - self.boll.lines.bot
        self.sma_width = bt.ind.SMA(self.width, period=self.p.sma_width_period)
        self.atr = bt.ind.ATR(self.data, period=self.p.atr_period)
        self.sma_atr = bt.ind.SMA(self.atr, period=self.p.sma_atr_period)

        self.crossover_top = bt.ind.CrossOver(
            self.data.close, self.boll.lines.top
        )
        self.crossover_bot = bt.ind.CrossOver(
            self.data.close, self.boll.lines.bot
        )

        self.order = None
        self.last_exit_bar = None

        self.entry_price = None
        self.entry_atr = None
        self.entry_bar = None
        self.highest_close = None
        self.lowest_close = None
        self.be_triggered = False

    def reset_trade_state(self):
        self.entry_price = None
        self.entry_atr = None
        self.entry_bar = None
        self.highest_close = None
        self.lowest_close = None
        self.be_triggered = False

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            if order.isbuy():
                if self.position.size > 0:
                    self.entry_price = order.executed.price
                    self.entry_bar = len(self)
                    self.entry_atr = self.atr[0]
                    self.highest_close = self.data.close[0]
                    self.be_triggered = False
                else:
                    self.last_exit_bar = len(self)
                    self.reset_trade_state()
            elif order.issell():
                if self.position.size < 0:
                    self.entry_price = order.executed.price
                    self.entry_bar = len(self)
                    self.entry_atr = self.atr[0]
                    self.lowest_close = self.data.close[0]
                    self.be_triggered = False
                else:
                    self.last_exit_bar = len(self)
                    self.reset_trade_state()

        self.order = None

    def next(self):
        if self.order is not None:
            return

        if self.position.size == 0:
            if (
                self.last_exit_bar is not None
                and (len(self) - self.last_exit_bar) < self.p.cooldown_period
            ):
                return

            vol_ratio = self.atr[0] / self.sma_atr[0]
            if not (self.p.vol_min <= vol_ratio <= self.p.vol_max):
                return

            consolidation = all(
                self.width[-i]
                < self.sma_width[-i] * self.p.consolidation_mult
                for i in range(self.p.consolidation_bars)
            )

            if not consolidation:
                return

            long_condition = (
                self.data.close[0] > self.sma200[0]
                and self.crossover_top[0] > 0
            )

            short_condition = (
                self.data.close[0] < self.sma200[0]
                and self.crossover_bot[0] < 0
            )

            if long_condition or short_condition:
                equity = self.broker.getvalue()
                stop_dist = self.p.stop_atr_mult * self.atr[0]
                if stop_dist <= 0:
                    return
                risk_amount = equity * self.p.risk_pct
                size = int(risk_amount / stop_dist)
                if size <= 0:
                    return

                if long_condition:
                    self.order = self.buy(size=size)
                elif short_condition:
                    self.order = self.sell(size=size)

        else:
            if self.entry_price is None:
                return

            r_dist = self.p.stop_atr_mult * self.entry_atr

            if self.position.size > 0:
                self.highest_close = max(
                    self.highest_close, self.data.close[0]
                )

                if (
                    not self.be_triggered
                    and self.highest_close
                    >= self.entry_price + self.p.be_r_mult * r_dist
                ):
                    self.be_triggered = True

                initial_stop = self.entry_price - r_dist
                chandelier_stop = (
                    self.highest_close - self.p.stop_atr_mult * self.atr[0]
                )
                effective_stop = max(initial_stop, chandelier_stop)

                if self.be_triggered:
                    effective_stop = max(effective_stop, self.entry_price)

                tp_price = self.entry_price + self.p.tp_r_mult * r_dist

                if self.data.low[0] <= effective_stop:
                    self.order = self.close()
                elif self.data.high[0] >= tp_price:
                    self.order = self.close()

            elif self.position.size < 0:
                self.lowest_close = min(self.lowest_close, self.data.close[0])

                if (
                    not self.be_triggered
                    and self.lowest_close
                    <= self.entry_price - self.p.be_r_mult * r_dist
                ):
                    self.be_triggered = True

                initial_stop = self.entry_price + r_dist
                chandelier_stop = (
                    self.lowest_close + self.p.stop_atr_mult * self.atr[0]
                )
                effective_stop = min(initial_stop, chandelier_stop)

                if self.be_triggered:
                    effective_stop = min(effective_stop, self.entry_price)

                tp_price = self.entry_price - self.p.tp_r_mult * r_dist

                if self.data.high[0] >= effective_stop:
                    self.order = self.close()
                elif self.data.low[0] <= tp_price:
                    self.order = self.close()
