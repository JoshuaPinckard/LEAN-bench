import backtrader as bt


class SessionVWAP(bt.Indicator):
    lines = ('vwap',)
    params = ()

    def __init__(self):
        self.addminperiod(1)

    def next(self):
        if len(self) == 1:
            self.cum_pv = 0.0
            self.cum_vol = 0.0
        elif self.data.datetime.date(0) != self.data.datetime.date(-1):
            self.cum_pv = 0.0
            self.cum_vol = 0.0

        tp = (self.data.high[0] + self.data.low[0] + self.data.close[0]) / 3.0
        v = self.data.volume[0]
        self.cum_pv += tp * v
        self.cum_vol += v

        if self.cum_vol > 0:
            self.lines.vwap[0] = self.cum_pv / self.cum_vol
        else:
            self.lines.vwap[0] = tp


class IntradayVWAPRSIStrategy(bt.Strategy):
    params = (
        ('vwap_threshold_pct', 0.003),
        ('rsi_oversold', 30),
        ('rsi_overbought', 70),
        ('stop_loss_pct', 0.02),
    )

    def __init__(self):
        self.vwap = SessionVWAP(self.data)
        self.rsi = bt.indicators.RSI(self.data.close, period=14)
        self.order = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        if self.order:
            return

        vwap_val = self.vwap[0]
        rsi_val = self.rsi[0]
        close_val = self.data.close[0]

        long_signal = (close_val < vwap_val * (1.0 - self.p.vwap_threshold_pct)) and (rsi_val < self.p.rsi_oversold)
        short_signal = (close_val > vwap_val * (1.0 + self.p.vwap_threshold_pct)) and (rsi_val > self.p.rsi_overbought)

        if not self.position:
            if long_signal:
                self.order = self.buy()
            elif short_signal:
                self.order = self.sell()
        else:
            if self.position.size > 0:
                stop_price = self.position.price * (1.0 - self.p.stop_loss_pct)
                if short_signal or self.data.low[0] <= stop_price:
                    self.order = self.close()
            elif self.position.size < 0:
                stop_price = self.position.price * (1.0 + self.p.stop_loss_pct)
                if long_signal or self.data.high[0] >= stop_price:
                    self.order = self.close()
