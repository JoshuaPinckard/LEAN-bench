import backtrader as bt
import collections


class SofiMeanReversionFlagStrategy(bt.Strategy):
    params = (
        ('ema_short_period', 10),
        ('ema_long_period', 20),
        ('position_filter_pct', 0.01),
        ('pullback_tolerance_pct', 0.003),
        ('flag_lookback', 5),
        ('flag_consolidation_pct', 0.015),
        ('skip_bars_after_open', 2),
        ('vrvp_lookback', 20),
        ('vrvp_thin_pocket_pctile', 0.2),
        ('vrvp_key_level', 26.0),
        ('stop_loss_pct', 0.015),
        ('take_profit_pct', 0.04),
        ('risk_pct', 0.015),
    )

    def __init__(self):
        self.ema10 = bt.indicators.EMA(self.data.close, period=self.p.ema_short_period)
        self.ema20 = bt.indicators.EMA(self.data.close, period=self.p.ema_long_period)
        self.order = None
        self.entry_price = None
        self.stop_price = None
        self.target_price = None
        self.year_high = None
        self.last_date = None
        self.last_close = None
        self.gap_fill_level = None
        self.day_bar_count = 0
        self.vrvp_window = collections.deque(maxlen=self.p.vrvp_lookback)

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return
        if order.status == order.Completed and order.isbuy():
            self.entry_price = order.executed.price
            self.stop_price = self.entry_price * (1 - self.p.stop_loss_pct)
            self.target_price = self.entry_price * (1 + self.p.take_profit_pct)
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        current_date = self.data.datetime.date(0)
        new_day = self.last_date is None or current_date != self.last_date

        if new_day:
            # "closes the day below EMA" can only be evaluated once the day has ended,
            # i.e. at the first bar of the following day, using the prior bar's close.
            if self.last_date is not None and self.position and self.order is None:
                if self.data.close[-1] < self.ema10[-1] or self.data.close[-1] < self.ema20[-1]:
                    self.order = self.close()
            if self.last_close is not None and self.data.open[0] > self.last_close:
                self.gap_fill_level = self.last_close
            else:
                self.gap_fill_level = None
            self.day_bar_count = 0
            self.last_date = current_date
        else:
            self.day_bar_count += 1

        self.last_close = self.data.close[0]

        self.year_high = self.data.high[0] if self.year_high is None else max(self.year_high, self.data.high[0])

        # VRVP proxy: volume-weighted range, ranked against its trailing distribution.
        metric = (self.data.high[0] - self.data.low[0]) * self.data.volume[0]
        is_thin_pocket = False
        if len(self.vrvp_window) == self.vrvp_window.maxlen:
            rank = sum(1 for v in self.vrvp_window if v <= metric) / len(self.vrvp_window)
            is_thin_pocket = rank <= self.p.vrvp_thin_pocket_pctile
        self.vrvp_window.append(metric)

        if self.order:
            return

        if self.position:
            if self.data.close[0] <= self.stop_price:
                self.order = self.close()
            elif self.data.close[0] >= self.target_price:
                self.order = self.close()
            return

        trend_filter = self.data.close[0] > self.ema10[0] and self.data.close[0] > self.ema20[0]
        position_filter = self.data.close[0] >= self.year_high * (1 - self.p.position_filter_pct)
        vrvp_filter = self.data.close[0] > self.p.vrvp_key_level and is_thin_pocket
        gap_open_block = (self.gap_fill_level is not None) and (self.day_bar_count == 0)

        if not (trend_filter and position_filter and vrvp_filter) or gap_open_block:
            return

        tol = self.p.pullback_tolerance_pct
        ema10_rising = self.ema10[0] > self.ema10[-1]
        ema20_rising = self.ema20[0] > self.ema20[-1]

        touched_gap = (
            self.gap_fill_level is not None
            and self.data.low[0] <= self.gap_fill_level * (1 + tol)
            and self.data.close[0] >= self.gap_fill_level * (1 - tol)
        )
        touched_ema10 = (
            ema10_rising
            and self.data.low[0] <= self.ema10[0] * (1 + tol)
            and self.data.close[0] > self.ema10[0]
        )
        touched_ema20 = (
            ema20_rising
            and self.data.low[0] <= self.ema20[0] * (1 + tol)
            and self.data.close[0] > self.ema20[0]
        )
        reversal_candle = self.data.close[0] > self.data.open[0] and self.data.close[0] > self.data.close[-1]
        option_a = (touched_gap or touched_ema10 or touched_ema20) and reversal_candle

        option_b = False
        if self.day_bar_count >= self.p.skip_bars_after_open + self.p.flag_lookback:
            window_highs = [self.data.high[-i] for i in range(1, self.p.flag_lookback + 1)]
            window_lows = [self.data.low[-i] for i in range(1, self.p.flag_lookback + 1)]
            flag_high = max(window_highs)
            flag_low = min(window_lows)
            if flag_low > 0:
                flag_range_pct = (flag_high - flag_low) / flag_low
                if flag_range_pct <= self.p.flag_consolidation_pct and self.data.close[0] > flag_high:
                    option_b = True

        if option_a or option_b:
            stop_distance = self.data.close[0] * self.p.stop_loss_pct
            risk_amount = self.broker.getvalue() * self.p.risk_pct
            size = int(risk_amount / stop_distance)
            if size > 0:
                self.order = self.buy(size=size)
