import backtrader as bt


class Impulse121Strategy(bt.Strategy):
    params = dict(
        sma_period=252,
        atr_period=14,
        atr_sma_period=252,
        atr_pct_threshold=0.05,
        position_pct=0.10,
        stop_loss_pct=0.02,
        take_profit_pct=0.05,
    )

    def __init__(self):
        self.sma = bt.indicators.SimpleMovingAverage(
            self.data.close, period=self.p.sma_period
        )
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.atr_sma = bt.indicators.SimpleMovingAverage(
            self.atr, period=self.p.atr_sma_period
        )
        self.atr_pct = self.atr_sma / self.data.close

        self.order = None
        self.stop_order = None
        self.limit_order = None

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return

        if order.status == order.Completed:
            if order.isbuy():
                entry_price = order.executed.price
                stop_price = entry_price * (1 - self.p.stop_loss_pct)
                limit_price = entry_price * (1 + self.p.take_profit_pct)
                self.stop_order = self.sell(
                    exectype=bt.Order.Stop, price=stop_price, size=self.position.size
                )
                self.limit_order = self.sell(
                    exectype=bt.Order.Limit,
                    price=limit_price,
                    size=self.position.size,
                    oco=self.stop_order,
                )
            else:
                self.stop_order = None
                self.limit_order = None
            self.order = None
        elif order.status in (order.Canceled, order.Margin, order.Rejected):
            if order is self.order:
                self.order = None
            if order is self.stop_order:
                self.stop_order = None
            if order is self.limit_order:
                self.limit_order = None

    def next(self):
        if self.order:
            return

        if not self.position:
            if (
                self.data.close[0] > self.sma[0]
                and self.atr_pct[0] <= self.p.atr_pct_threshold
            ):
                size = (self.p.position_pct * self.broker.getvalue()) / self.data.close[0]
                self.order = self.buy(size=size)
        else:
            if self.data.close[0] < self.sma[0]:
                if self.stop_order is not None and self.stop_order.alive():
                    self.cancel(self.stop_order)
                if self.limit_order is not None and self.limit_order.alive():
                    self.cancel(self.limit_order)
                self.order = self.close()
