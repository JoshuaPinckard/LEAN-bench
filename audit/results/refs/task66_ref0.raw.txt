import backtrader as bt


class DoubleBottomStrategy(bt.Strategy):
    params = (
        ('sma_period', 200),
        ('vol_avg_period', 20),
        ('swing_window', 5),
        ('bottom_tolerance', 0.01),
        ('min_bars_between_bottoms', 10),
        ('stop_loss_pct', 0.005),
        ('reward_risk_ratio', 2.0),
    )

    def __init__(self):
        self.sma200 = bt.indicators.SMA(self.data.close, period=self.p.sma_period)
        self.vol_sma = bt.indicators.SMA(self.data.volume, period=self.p.vol_avg_period)

        self.order = None
        self.stop_order = None
        self.limit_order = None
        self.pending_stop_price = None

        self.state = 'SEARCH_BOTTOM1'
        self.bottom1_price = None
        self.bottom1_bar = None
        self.peak_price = None
        self.peak_bar = None
        self.bottom2_price = None
        self.bottom2_bar = None

    def _is_swing_low(self, offset):
        w = self.p.swing_window
        if len(self) <= 2 * w:
            return False
        candidate = self.data.low[offset]
        for j in range(-2 * w, 1):
            if j == offset:
                continue
            if self.data.low[j] < candidate:
                return False
        return True

    def _is_swing_high(self, offset):
        w = self.p.swing_window
        if len(self) <= 2 * w:
            return False
        candidate = self.data.high[offset]
        for j in range(-2 * w, 1):
            if j == offset:
                continue
            if self.data.high[j] > candidate:
                return False
        return True

    def _reset_pattern(self):
        self.state = 'SEARCH_BOTTOM1'
        self.bottom1_price = None
        self.bottom1_bar = None
        self.peak_price = None
        self.peak_bar = None
        self.bottom2_price = None
        self.bottom2_bar = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status == order.Completed:
            if order.isbuy() and order is self.order:
                entry_price = order.executed.price
                risk = entry_price - self.pending_stop_price
                tp_price = entry_price + self.p.reward_risk_ratio * risk
                size = self.position.size
                self.stop_order = self.sell(size=size, exectype=bt.Order.Stop, price=self.pending_stop_price)
                self.limit_order = self.sell(size=size, exectype=bt.Order.Limit, price=tp_price)
            elif order.issell() and order in (self.stop_order, self.limit_order):
                other = self.limit_order if order is self.stop_order else self.stop_order
                if other is not None and other.status in [other.Submitted, other.Accepted]:
                    self.cancel(other)
                self.stop_order = None
                self.limit_order = None

        if order is self.order:
            self.order = None
        if order is self.stop_order:
            self.stop_order = None
        if order is self.limit_order:
            self.limit_order = None

    def next(self):
        if self.order:
            return

        if self.position:
            if self.data.close[0] < self.sma200[0]:
                if self.stop_order is not None:
                    self.cancel(self.stop_order)
                if self.limit_order is not None:
                    self.cancel(self.limit_order)
                self.order = self.close()
            return

        w = self.p.swing_window
        offset = -w
        candidate_bar = len(self) - w
        is_low = self._is_swing_low(offset)
        is_high = self._is_swing_high(offset)
        candidate_low = self.data.low[offset] if is_low else None
        candidate_high = self.data.high[offset] if is_high else None

        if self.state == 'SEARCH_BOTTOM1':
            if is_low:
                self.bottom1_price = candidate_low
                self.bottom1_bar = candidate_bar
                self.state = 'SEARCH_PEAK'

        elif self.state == 'SEARCH_PEAK':
            if is_low and candidate_low < self.bottom1_price:
                self.bottom1_price = candidate_low
                self.bottom1_bar = candidate_bar
            elif is_high and candidate_bar > self.bottom1_bar:
                self.peak_price = candidate_high
                self.peak_bar = candidate_bar
                self.state = 'SEARCH_BOTTOM2'

        elif self.state == 'SEARCH_BOTTOM2':
            if is_high and candidate_high > self.peak_price:
                self.peak_price = candidate_high
                self.peak_bar = candidate_bar
            elif is_low and candidate_bar > self.peak_bar:
                gap = candidate_bar - self.bottom1_bar
                close_enough = abs(candidate_low - self.bottom1_price) / self.bottom1_price <= self.p.bottom_tolerance
                if close_enough and gap >= self.p.min_bars_between_bottoms:
                    self.bottom2_price = candidate_low
                    self.bottom2_bar = candidate_bar
                    self.state = 'SEARCH_BREAKOUT'
                else:
                    self.bottom1_price = candidate_low
                    self.bottom1_bar = candidate_bar
                    self.peak_price = None
                    self.peak_bar = None
                    self.state = 'SEARCH_PEAK'

        elif self.state == 'SEARCH_BREAKOUT':
            if (self.data.close[0] > self.peak_price and
                    self.data.close[0] > self.sma200[0] and
                    self.data.volume[0] > self.vol_sma[0]):
                self.pending_stop_price = self.bottom2_price * (1.0 - self.p.stop_loss_pct)
                self.order = self.buy()
                self._reset_pattern()
