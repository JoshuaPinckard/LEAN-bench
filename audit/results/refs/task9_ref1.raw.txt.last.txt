import backtrader as bt


class ConsolidationBreakoutStrategy(bt.Strategy):
    params = (
        ('bb_period', 20),
        ('bb_devfactor', 2),
        ('rsi_period', 14),
        ('volume_period', 20),
        ('consolidation_bars', 10),
        ('take_profit', 0.02),
        ('stop_loss', 0.015),
    )

    def __init__(self):
        self.bb = bt.indicators.BollingerBands(
            self.data.close,
            period=self.p.bb_period,
            devfactor=self.p.bb_devfactor,
        )
        self.rsi = bt.indicators.RSI(self.data.close, period=self.p.rsi_period)
        self.volume_average = bt.indicators.SimpleMovingAverage(
            self.data.volume,
            period=self.p.volume_period,
        )

        self.order = None
        self.entry_price = None
        self.entry_bar = None

    def notify_order(self, order):
        if order.status == order.Completed:
            if order.isbuy() and self.position.size == 0:
                self.entry_price = order.executed.price
                self.entry_bar = len(self)
            elif order.issell() and self.position.size == 0:
                self.entry_price = order.executed.price
                self.entry_bar = len(self)

        if order.status in (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            self.order = None

    def next(self):
        if self.order:
            return

        if self.position:
            if self.position.size > 0:
                if (
                    self.data.close[0] <= self.entry_price * (1.0 - self.p.stop_loss)
                    or self.data.close[0] >= self.entry_price * (1.0 + self.p.take_profit)
                    or self.data.close[0] < self.bb.mid[0]
                    or (
                        self.entry_bar is not None
                        and 0 <= len(self) - self.entry_bar <= 2
                        and self.bb.bot[0] <= self.data.close[0] <= self.bb.top[0]
                    )
                ):
                    self.order = self.close()
            else:
                if (
                    self.data.close[0] >= self.entry_price * (1.0 + self.p.stop_loss)
                    or self.data.close[0] <= self.entry_price * (1.0 - self.p.take_profit)
                    or self.data.close[0] > self.bb.mid[0]
                    or (
                        self.entry_bar is not None
                        and 0 <= len(self) - self.entry_bar <= 2
                        and self.bb.bot[0] <= self.data.close[0] <= self.bb.top[0]
                    )
                ):
                    self.order = self.close()
            return

        in_consolidation = all(
            self.bb.bot[-bar] <= self.data.close[-bar] <= self.bb.top[-bar]
            for bar in range(1, self.p.consolidation_bars + 1)
        )
        volume_confirmed = self.data.volume[0] >= 1.5 * self.volume_average[-1]

        if not in_consolidation or not volume_confirmed:
            return

        size = (self.broker.getvalue() * 0.01) / self.data.close[0]

        if self.data.close[0] > self.bb.top[0] and self.rsi[0] > 50:
            self.order = self.buy(size=size)
        elif self.data.close[0] < self.bb.bot[0] and self.rsi[0] < 50:
            self.order = self.sell(size=size)
