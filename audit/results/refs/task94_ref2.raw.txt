import backtrader as bt


class NXXTMultiMATrendFollowing(bt.Strategy):
    """NXXT Multi-Moving Average Trend Following Strategy."""

    def __init__(self):
        # Moving Averages
        self.sma30 = bt.indicators.SimpleMovingAverage(
            self.data.close, period=30
        )
        self.sma60 = bt.indicators.SimpleMovingAverage(
            self.data.close, period=60
        )
        self.sma120 = bt.indicators.SimpleMovingAverage(
            self.data.close, period=120
        )

        # Volatility Filter: 14-day ATR and its 30-day SMA
        self.atr14 = bt.indicators.ATR(self.data, period=14)
        self.atr14_sma30 = bt.indicators.SimpleMovingAverage(
            self.atr14, period=30
        )

        # Pending order tracking
        self.order = None

    def notify_order(self, order):
        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            self.order = None

    def next(self):
        if self.order:
            return

        if not self.position:
            # Volatility Filter: No new position if ATR 14 > 1.5 * ATR 30-day SMA
            if self.atr14[0] > 1.5 * self.atr14_sma30[0]:
                return

            # Condition 1: Price above SMA 30, SMA 60, SMA 120
            price_above = (
                self.data.close[0] > self.sma30[0]
                and self.data.close[0] > self.sma60[0]
                and self.data.close[0] > self.sma120[0]
            )

            # Condition 1 (cont.): MAs trending upwards
            mas_trending = (
                self.sma30[0] > self.sma30[-1]
                and self.sma60[0] > self.sma60[-1]
                and self.sma120[0] > self.sma120[-1]
            )

            if not (price_above and mas_trending):
                return

            # Condition 2: Consolidation range over the prior 5 sessions (bars -5 to -1)
            highs_5 = [self.data.high[-i] for i in range(1, 6)]
            lows_5 = [self.data.low[-i] for i in range(1, 6)]
            max_h = max(highs_5)
            min_l = min(lows_5)

            is_consolidating = (max_h - min_l) / min_l <= 0.05
            is_breakout = self.data.close[0] > max_h

            if is_consolidating and is_breakout:
                entry_price = self.data.close[0]
                stop_loss_price = self.sma30[0]
                risk_per_share = entry_price - stop_loss_price

                if risk_per_share > 0:
                    capital = self.broker.getvalue()
                    max_risk_dollars = capital * 0.02
                    size = int(max_risk_dollars / risk_per_share)

                    # Cap position size by available cash
                    cash = self.broker.getcash()
                    max_size_cash = int(cash / entry_price)
                    size = min(size, max_size_cash)

                    if size > 0:
                        self.order = self.buy(size=size)
        else:
            entry_price = self.position.price

            # Exit 1: Take Profit - price reaches 10% gain from entry price
            if self.data.high[0] >= 1.10 * entry_price:
                self.order = self.close()
            # Exit 2: Stop Loss - price closes below SMA 30
            elif self.data.close[0] < self.sma30[0]:
                self.order = self.close()
