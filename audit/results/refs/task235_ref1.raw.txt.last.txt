import backtrader as bt


class ReferenceStrategy(bt.Strategy):
    def __init__(self):
        self.atr = bt.indicators.ATR(self.data, period=14)
        self.atr_sma = bt.indicators.SMA(self.atr, period=50)
        self.rsi = bt.indicators.RSI(self.data.close, period=14)
        self.boll = bt.indicators.BollingerBands(
            self.data.close, period=20, devfactor=2
        )

        self.entry_order = None
        self.stop_order = None
        self.exit_order = None
        self.entry_bar = None
        self.trailing_active = False
        self.last_entry_day = None

    def notify_order(self, order):
        if order.status not in (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            return

        if order is self.entry_order:
            self.entry_order = None
            if order.status == order.Completed:
                self.entry_bar = len(self)
                self.trailing_active = False

                if order.isbuy():
                    self.stop_order = self.sell(
                        exectype=bt.Order.Stop,
                        price=444.75,
                        size=2,
                    )
                else:
                    self.stop_order = self.buy(
                        exectype=bt.Order.Stop,
                        price=450.25,
                        size=2,
                    )

        elif order is self.stop_order:
            self.stop_order = None
            if order.status == order.Completed and self.exit_order is not None:
                self.cancel(self.exit_order)

        elif order is self.exit_order:
            self.exit_order = None

    def notify_trade(self, trade):
        if trade.isclosed:
            self.entry_bar = None
            self.trailing_active = False

    def _exit_position(self):
        if self.exit_order is None:
            if self.stop_order is not None:
                self.cancel(self.stop_order)
            self.exit_order = self.close()

    def _activate_trailing_stop(self):
        if self.stop_order is not None:
            self.cancel(self.stop_order)

        trail_amount = 0.5 * self.atr[0]
        if self.position.size > 0:
            self.stop_order = self.sell(
                exectype=bt.Order.StopTrail,
                size=2,
                trailamount=trail_amount,
            )
        else:
            self.stop_order = self.buy(
                exectype=bt.Order.StopTrail,
                size=2,
                trailamount=trail_amount,
            )
        self.trailing_active = True

    def next(self):
        if self.position:
            if self.position.size > 0:
                if (
                    not self.trailing_active
                    and self.data.high[0]
                    >= self.position.price + 0.75 * self.atr[0]
                ):
                    self._activate_trailing_stop()

                if self.data.high[0] >= 449.5 or self.data.close[0] >= 449.0:
                    self._exit_position()
                    return
            else:
                if (
                    not self.trailing_active
                    and self.data.low[0]
                    <= self.position.price - 0.75 * self.atr[0]
                ):
                    self._activate_trailing_stop()

                middle_touched = (
                    self.data.low[0] <= self.boll.mid[0] <= self.data.high[0]
                )
                if (
                    self.data.low[0] <= 447.5
                    or self.data.close[0] <= 446.0
                    or middle_touched
                ):
                    self._exit_position()
                    return

            if (
                self.entry_bar is not None
                and len(self) > self.entry_bar
                and self.exit_order is None
            ):
                self._exit_position()
            return

        today = self.data.datetime.date(0)
        if (
            self.entry_order is not None
            or self.stop_order is not None
            or self.exit_order is not None
            or self.last_entry_day == today
            or not self.atr[0] > self.atr_sma[0]
        ):
            return

        if (
            self.data.high[0] > 450.00
            and self.data.close[0] < 450.00
            and self.rsi[0] > 60
            and self.data.high[0] >= self.boll.top[0]
        ):
            self.entry_order = self.sell(size=2)
            self.last_entry_day = today

        elif (
            self.data.low[0] < 445.00
            and self.data.close[0] > 445.00
            and self.rsi[0] < 40
            and self.data.low[0] <= self.boll.bot[0]
        ):
            self.entry_order = self.buy(size=2)
            self.last_entry_day = today
