import backtrader as bt


class BreakoutBBStrategy(bt.Strategy):
    params = dict(
        bb_period=10,
        bb_devfactor=0.674,
    )

    def __init__(self):
        self.bb = bt.indicators.BollingerBands(
            self.data.close, period=self.p.bb_period, devfactor=self.p.bb_devfactor
        )
        self.crossover = bt.indicators.CrossOver(self.data.close, self.bb.lines.top)
        self.order = None

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        if self.order:
            return

        if not self.position:
            if self.crossover[0] > 0:
                cash = self.broker.getcash()
                price = self.data.close[0]
                size = int((cash * 0.999) / price)
                if size > 0:
                    self.order = self.buy(size=size)
        else:
            if self.crossover[0] < 0 or self.data.close[0] < self.bb.lines.top[0]:
                self.order = self.close()
