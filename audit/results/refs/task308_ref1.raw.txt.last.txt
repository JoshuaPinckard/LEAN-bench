```python
import backtrader as bt


class DonchianContractionStrategy(bt.Strategy):
    params = (
        ("donchian_period", 20),
        ("atr_period", 14),
    )

    def __init__(self):
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)

        # Bands exclude the current bar so a close can break out of them.
        self.upper_band = bt.indicators.Highest(
            self.data.high(-1), period=self.p.donchian_period
        )
        self.lower_band = bt.indicators.Lowest(
            self.data.low(-1), period=self.p.donchian_period
        )

        self.order = None
        self.stop_order = None
        self.entry_side = 0

    def notify_order(self, order):
        if order == self.stop_order:
            if order.status in (
                order.Completed,
                order.Canceled,
                order.Margin,
                order.Rejected,
            ):
                self.stop_order = None
            return

        if order.status in (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            self.order = None

        if order.status == order.Completed and self.entry_side:
            atr_at_entry = self.atr[0]

            if self.entry_side > 0:
                self.stop_order = self.sell(
                    exectype=bt.Order.Stop,
                    price=order.executed.price - (2 * atr_at_entry),
                )
            else:
                self.stop_order = self.buy(
                    exectype=bt.Order.Stop,
                    price=order.executed.price + (2 * atr_at_entry),
                )

            self.entry_side = 0

    def next(self):
        if self.order:
            return

        contraction = (
            (self.upper_band[-1] - self.lower_band[-1])
            < (2 * self.atr[-1])
        )

        long_breakout = (
            contraction
            and self.data.close[-1] <= self.upper_band[-1]
            and self.data.close[0] > self.upper_band[0]
        )

        short_breakout = (
            contraction
            and self.data.close[-1] >= self.lower_band[-1]
            and self.data.close[0] < self.lower_band[0]
        )

        if not self.position:
            if long_breakout:
                self.entry_side = 1
                self.order = self.buy()
            elif short_breakout:
                self.entry_side = -1
                self.order = self.sell()
            return

        if self.position.size > 0 and short_breakout:
            if self.stop_order:
                self.cancel(self.stop_order)
            self.entry_side = 0
            self.order = self.close()

        elif self.position.size < 0 and long_breakout:
            if self.stop_order:
                self.cancel(self.stop_order)
            self.entry_side = 0
            self.order = self.close()
```
