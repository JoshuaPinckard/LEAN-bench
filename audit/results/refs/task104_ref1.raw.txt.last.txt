import backtrader as bt
import math


class ConsolidationBreakout(bt.Strategy):
    params = (
        ("sma_period", 120),
        ("volume_period", 20),
        ("lower_boundary", 180.00),
        ("upper_boundary", 193.00),
        ("target_level", 200.00),
        ("risk_fraction", 0.02),
    )

    def __init__(self):
        self.sma = bt.indicators.SMA(self.data.close, period=self.p.sma_period)
        self.avg_volume = bt.indicators.SMA(
            self.data.volume, period=self.p.volume_period
        )
        self.order = None

    def notify_order(self, order):
        if order.status in (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            self.order = None

    def next(self):
        if self.order:
            return

        if self.position:
            if (
                self.data.high[0] >= self.p.target_level
                or self.data.low[0] < self.sma[0]
                or self.data.close[0] < self.p.lower_boundary
            ):
                self.order = self.close()
            return

        prior_in_consolidation = (
            self.p.lower_boundary <= self.data.close[-1] <= self.p.upper_boundary
        )
        selling_volume = (
            self.data.volume[-1] if self.data.close[-1] < self.data.open[-1] else 0.0
        )
        volume_ok = selling_volume <= (self.avg_volume[0] * 1.5)
        sma_rising = self.sma[0] > self.sma[-5]

        if (
            self.data.close[0] > self.sma[0]
            and prior_in_consolidation
            and volume_ok
            and self.data.close[0] > self.p.upper_boundary
            and sma_rising
        ):
            risk_per_share = self.data.close[0] - self.sma[0]
            risk_capital = self.broker.getvalue() * self.p.risk_fraction
            size = math.floor(risk_capital / risk_per_share)

            if size > 0:
                self.order = self.buy(size=size)
