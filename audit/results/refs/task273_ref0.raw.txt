import math

import backtrader as bt


class FairValueArbitrage(bt.Strategy):
    params = dict(
        ema_period=50,
        atr_period=14,
        rsi_period=14,
        bb_period=20,
        bb_dev=2,
        vol_threshold=0.80,
        bb_width_threshold=0.30,
        spread_entry=0.01,
        rsi_long_th=45,
        rsi_short_th=55,
        stop_loss_pct=0.05,
        take_profit_pct=0.02,
        risk_fraction=0.02,
        spread_scale=0.02,
        max_multiplier=1.5,
    )

    def __init__(self):
        self.ema = bt.indicators.EMA(self.data.close, period=self.p.ema_period)
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.rsi = bt.indicators.RSI(self.data.close, period=self.p.rsi_period)
        self.bb = bt.indicators.BollingerBands(
            self.data.close, period=self.p.bb_period, devfactor=self.p.bb_dev
        )
        self.crossover = bt.indicators.CrossOver(self.data.close, self.ema)

        self.order = None

    def notify_order(self, order):
        if order.status in (order.Completed, order.Canceled, order.Margin, order.Rejected):
            self.order = None

    def next(self):
        if self.order:
            return

        close = self.data.close[0]
        ema = self.ema[0]
        spread = (close - ema) / ema

        if not self.position:
            atr_vol = (self.atr[0] / close) * math.sqrt(252)
            bb_width = (self.bb.top[0] - self.bb.bot[0]) / self.bb.mid[0]
            vol_ok = atr_vol <= self.p.vol_threshold and bb_width < self.p.bb_width_threshold

            if vol_ok:
                multiplier = min(
                    self.p.max_multiplier, 1 + abs(spread) / self.p.spread_scale
                )
                size = (self.broker.getvalue() * self.p.risk_fraction * multiplier) / close

                if spread <= -self.p.spread_entry and self.rsi[0] < self.p.rsi_long_th:
                    self.order = self.buy(size=size)
                elif spread >= self.p.spread_entry and self.rsi[0] > self.p.rsi_short_th:
                    self.order = self.sell(size=size)
        else:
            entry_price = self.position.price
            if self.position.size > 0:
                if (
                    self.crossover[0] > 0
                    or close <= entry_price * (1 - self.p.stop_loss_pct)
                    or close >= entry_price * (1 + self.p.take_profit_pct)
                ):
                    self.order = self.close()
            else:
                if (
                    self.crossover[0] < 0
                    or close >= entry_price * (1 + self.p.stop_loss_pct)
                    or close <= entry_price * (1 - self.p.take_profit_pct)
                ):
                    self.order = self.close()
