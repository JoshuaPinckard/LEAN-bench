import backtrader as bt


class PeriodicTargetWeightRebalance(bt.Strategy):
    params = (
        ("target_weights", {}),
        ("rebalance_period", "quarterly"),
    )

    def __init__(self):
        self.order = None
        self._last_rebalance_key = None

    def start(self):
        self.broker.set_slippage_perc(0.0005)

    def _weight_for_date(self, dt):
        schedule = self.p.target_weights
        date = dt.date()
        keys = (dt, date, dt.isoformat(), date.isoformat())

        for key in keys:
            if key in schedule:
                return schedule[key]

        return None

    def next(self):
        if self.order:
            return

        price = self.data.close[0]
        if price <= 0:
            return

        dt = self.data.datetime.datetime(0)
        if self.p.rebalance_period == "quarterly":
            rebalance_key = (dt.year, (dt.month - 1) // 3 + 1)
        elif self.p.rebalance_period == "yearly":
            rebalance_key = (dt.year,)
        else:
            raise ValueError("rebalance_period must be 'quarterly' or 'yearly'")

        if rebalance_key == self._last_rebalance_key:
            return

        weight = self._weight_for_date(dt)
        if weight is None:
            return

        self._last_rebalance_key = rebalance_key
        target_shares = int((self.broker.getvalue() * weight) / price)
        self.order = self.order_target_size(target=target_shares)

    def notify_order(self, order):
        if order.status in (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            self.order = None
