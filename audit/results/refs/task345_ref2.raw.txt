import backtrader as bt


class RSIMeanReversion(bt.Strategy):
    params = (
        ("rsi_period", 14),
        ("sma_period", 20),
        ("std_period", 20),
        ("num_std", 2.0),
        ("stop_pct", 0.02),
    )

    def __init__(self):
        self.rsi = bt.ind.RSI(self.data.close, period=self.p.rsi_period)
        self.rsi_sma = bt.ind.SMA(self.rsi, period=self.p.sma_period)
        self.rsi_std = bt.ind.StdDev(self.rsi, period=self.p.std_period)

        self.upper_band = self.rsi_sma + self.p.num_std * self.rsi_std
        self.lower_band = self.rsi_sma - self.p.num_std * self.rsi_std

        self.crossover = bt.ind.CrossOver(self.rsi, self.rsi_sma)

        self.entry_order = None
        self.stop_order = None
        self.close_order = None

        self.long_setup = False
        self.short_setup = False

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            if order == self.entry_order:
                self.entry_order = None
                if self.position.size > 0:
                    stop_price = order.executed.price * (1.0 - self.p.stop_pct)
                    self.stop_order = self.sell(
                        exectype=bt.Order.Stop, price=stop_price
                    )
                elif self.position.size < 0:
                    stop_price = order.executed.price * (1.0 + self.p.stop_pct)
                    self.stop_order = self.buy(
                        exectype=bt.Order.Stop, price=stop_price
                    )
            elif order == self.stop_order:
                self.stop_order = None
            elif order == self.close_order:
                self.close_order = None

        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            if order == self.entry_order:
                self.entry_order = None
            elif order == self.stop_order:
                self.stop_order = None
            elif order == self.close_order:
                self.close_order = None

    def next(self):
        if self.rsi[0] < self.lower_band[0]:
            self.long_setup = True
            self.short_setup = False
        elif self.rsi[0] > self.upper_band[0]:
            self.short_setup = True
            self.long_setup = False

        if self.entry_order is not None or self.close_order is not None:
            return

        if not self.position:
            if self.long_setup and self.crossover[0] > 0:
                self.entry_order = self.buy()
                self.long_setup = False
            elif self.short_setup and self.crossover[0] < 0:
                self.entry_order = self.sell()
                self.short_setup = False
        else:
            if self.position.size > 0:
                if self.crossover[0] < 0:
                    if self.stop_order is not None:
                        self.cancel(self.stop_order)
                        self.stop_order = None
                    self.close_order = self.close()
            elif self.position.size < 0:
                if self.crossover[0] > 0:
                    if self.stop_order is not None:
                        self.cancel(self.stop_order)
                        self.stop_order = None
                    self.close_order = self.close()
