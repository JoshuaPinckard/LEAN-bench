import backtrader as bt


class MeanReversionStrikeVectorStrategy(bt.Strategy):
    params = (
        ('ema_period', 20),
        ('bb_period', 20),
        ('bb_devfactor_entry', 2.0),
        ('bb_devfactor_exit', 0.5),
        ('atr_period', 14),
        ('atr_sma_period', 60),
        ('atr_filter_mult', 1.5),
        ('vol_sma_period', 20),
        ('gap_atr_mult', 1.5),
        ('stop_atr_mult', 2.0),
        ('max_bars_in_trade', 20),
        ('risk_pct', 0.05),
        ('max_drawdown_pct', 0.10),
    )

    def __init__(self):
        self.ema20 = bt.indicators.EMA(self.data.close, period=self.p.ema_period)
        self.atr14 = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.atr_sma60 = bt.indicators.SMA(self.atr14, period=self.p.atr_sma_period)
        self.vol_sma20 = bt.indicators.SMA(self.data.volume, period=self.p.vol_sma_period)

        self.bb2 = bt.indicators.BollingerBands(
            self.data.close, period=self.p.bb_period,
            devfactor=self.p.bb_devfactor_entry, movav=bt.indicators.MovAv.Exponential)
        self.bb05 = bt.indicators.BollingerBands(
            self.data.close, period=self.p.bb_period,
            devfactor=self.p.bb_devfactor_exit, movav=bt.indicators.MovAv.Exponential)

        self.cross_lower = bt.indicators.CrossOver(self.data.close, self.bb2.bot)
        self.cross_upper = bt.indicators.CrossOver(self.data.close, self.bb2.top)

        self.order = None
        self.position_stop = None
        self.entry_bar = None
        self.peak_value = None
        self.trading_stopped = False

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return
        if order.status == order.Completed:
            if self.position:
                self.entry_bar = len(self)
            else:
                self.entry_bar = None
                self.position_stop = None
        self.order = None

    def next(self):
        value = self.broker.getvalue()
        self.peak_value = value if self.peak_value is None else max(self.peak_value, value)

        if self.trading_stopped:
            return

        if self.order:
            return

        drawdown = (self.peak_value - value) / self.peak_value if self.peak_value > 0 else 0.0
        if drawdown >= self.p.max_drawdown_pct:
            self.trading_stopped = True
            if self.position:
                self.order = self.close()
            return

        close = self.data.close[0]
        ema = self.ema20[0]
        atr = self.atr14[0]

        if self.position:
            if self.position.size > 0:
                stop_hit = close <= self.position_stop
            else:
                stop_hit = close >= self.position_stop
            reentered_band = self.bb05.bot[0] < close < self.bb05.top[0]
            bars_held = (len(self) - self.entry_bar) if self.entry_bar is not None else 0
            time_expired = bars_held >= self.p.max_bars_in_trade

            if reentered_band or time_expired or stop_hit:
                self.order = self.close()
            return

        filter_atr_ok = atr <= self.p.atr_filter_mult * self.atr_sma60[0]
        filter_vol_ok = self.data.volume[0] >= self.vol_sma20[0]
        filter_gap_ok = abs(self.data.open[0] - self.data.close[-1]) <= self.p.gap_atr_mult * atr

        if not (filter_atr_ok and filter_vol_ok and filter_gap_ok):
            return

        if self.cross_lower[0] == 1:
            stop_price = ema - self.p.stop_atr_mult * atr
            risk = close - stop_price
            if risk > 0:
                size = int((self.p.risk_pct * value) / risk)
                if size > 0:
                    self.position_stop = stop_price
                    self.order = self.buy(size=size)
        elif self.cross_upper[0] == -1:
            stop_price = ema + self.p.stop_atr_mult * atr
            risk = stop_price - close
            if risk > 0:
                size = int((self.p.risk_pct * value) / risk)
                if size > 0:
                    self.position_stop = stop_price
                    self.order = self.sell(size=size)
