import backtrader as bt


class RiskParityVolTargetStrategy(bt.Strategy):
    params = (("vtgt", 0.01),)

    def __init__(self):
        self.atr = bt.indicators.ATR(self.data, period=252)
        self.sma = bt.indicators.SMA(self.data.close, period=252)
        self.order = None

    def notify_order(self, order):
        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            self.order = None

    def next(self):
        if self.order:
            return

        current_date = self.data.datetime.date(0)
        try:
            next_date = self.data.datetime.date(1)
            is_last_trading_day = next_date.month != current_date.month
        except IndexError:
            is_last_trading_day = True

        if not is_last_trading_day:
            return

        vol = self.atr[0] / self.sma[0]
        target_weight = self.p.vtgt / vol
        weight = min(max(target_weight, 0.01), 0.10)
        self.order = self.order_target_percent(target=weight)
