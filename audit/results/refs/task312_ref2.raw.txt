import backtrader as bt


class VolumeClimaxReversal(bt.Strategy):
    def __init__(self):
        self.sma50 = bt.indicators.SMA(self.data.close, period=50)
        self.sma_vol20 = bt.indicators.SMA(self.data.volume, period=20)
        self.atr14 = bt.indicators.ATR(self.data, period=14)

        self.order = None
        self.stop_order = None

        self.blowoff_active = False
        self.blowoff_type = None  # 'UP' or 'DOWN'
        self.blowoff_high = 0.0
        self.blowoff_low = 0.0
        self.blowoff_close = 0.0
        self.bars_since_blowoff = 0
        self.pullback_met = False

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            if order == self.order:
                self.order = None
                if self.position.size > 0:
                    stop_price = order.executed.price * 0.98
                    self.stop_order = self.sell(
                        exectype=bt.Order.Stop, price=stop_price
                    )
                elif self.position.size < 0:
                    stop_price = order.executed.price * 1.02
                    self.stop_order = self.buy(
                        exectype=bt.Order.Stop, price=stop_price
                    )
                else:
                    self.blowoff_active = False
            elif order == self.stop_order:
                self.stop_order = None
                self.blowoff_active = False

        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            if order == self.order:
                self.order = None
            elif order == self.stop_order:
                self.stop_order = None

    def next(self):
        if self.order:
            return

        if self.position.size != 0:
            if self.position.size < 0:
                if (
                    self.data.close[0] > self.blowoff_high
                    and self.data.volume[0] >= 1.5 * self.sma_vol20[0]
                ):
                    if self.stop_order:
                        self.cancel(self.stop_order)
                        self.stop_order = None
                    self.order = self.close()
            elif self.position.size > 0:
                if (
                    self.data.close[0] < self.blowoff_low
                    and self.data.volume[0] >= 1.5 * self.sma_vol20[0]
                ):
                    if self.stop_order:
                        self.cancel(self.stop_order)
                        self.stop_order = None
                    self.order = self.close()
            return

        is_uptrend = (self.data.close[0] > self.sma50[0]) and (
            self.sma50[0] > self.sma50[-1]
        )
        is_downtrend = (self.data.close[0] < self.sma50[0]) and (
            self.sma50[0] < self.sma50[-1]
        )

        strong_move_up = (
            self.data.close[0] - self.data.close[-6]
        ) > 1.5 * self.atr14[0]
        strong_move_down = (
            self.data.close[-6] - self.data.close[0]
        ) > 1.5 * self.atr14[0]

        vol_climax = self.data.volume[0] > 2.0 * self.sma_vol20[0]
        bar_range = self.data.high[0] - self.data.low[0]
        range_climax = bar_range > 1.2 * self.atr14[0]

        is_blowoff_up = False
        is_blowoff_down = False

        if (
            is_uptrend
            and strong_move_up
            and vol_climax
            and range_climax
            and bar_range > 0
        ):
            if ((self.data.high[0] - self.data.close[0]) / bar_range) < 0.25:
                is_blowoff_up = True

        if (
            is_downtrend
            and strong_move_down
            and vol_climax
            and range_climax
            and bar_range > 0
        ):
            if ((self.data.close[0] - self.data.low[0]) / bar_range) < 0.25:
                is_blowoff_down = True

        if is_blowoff_up:
            self.blowoff_active = True
            self.blowoff_type = "UP"
            self.blowoff_high = self.data.high[0]
            self.blowoff_low = self.data.low[0]
            self.blowoff_close = self.data.close[0]
            self.bars_since_blowoff = 0
            self.pullback_met = False
            return

        if is_blowoff_down:
            self.blowoff_active = True
            self.blowoff_type = "DOWN"
            self.blowoff_high = self.data.high[0]
            self.blowoff_low = self.data.low[0]
            self.blowoff_close = self.data.close[0]
            self.bars_since_blowoff = 0
            self.pullback_met = False
            return

        if self.blowoff_active:
            self.bars_since_blowoff += 1
            if self.bars_since_blowoff > 8:
                self.blowoff_active = False
                return

            if not self.pullback_met:
                if self.blowoff_type == "UP":
                    if (
                        self.blowoff_close - self.data.low[0]
                    ) >= 0.5 * self.atr14[0]:
                        self.pullback_met = True
                elif self.blowoff_type == "DOWN":
                    if (
                        self.data.high[0] - self.blowoff_close
                    ) >= 0.5 * self.atr14[0]:
                        self.pullback_met = True

            if self.pullback_met:
                if self.blowoff_type == "UP":
                    if self.data.close[0] < self.data.low[-1]:
                        self.order = self.sell()
                elif self.blowoff_type == "DOWN":
                    if self.data.close[0] > self.data.high[-1]:
                        self.order = self.buy()
