import math
import backtrader as bt


class InverseHeadAndShouldersStrategy(bt.Strategy):
    def __init__(self):
        self._order = None
        self._entry_order = None
        self._daily = []
        self._daily_work = None
        self._session_date = None
        self._four_hour = []
        self._previous_four_close = None
        self._previous_red_volume = None
        self._pattern = None
        self._trade_pattern = None
        self._stop_price = None
        self._target_price = None

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            if order is self._order:
                self._order = None

            if order is self._entry_order:
                if order.status != order.Completed:
                    self._stop_price = None
                    self._target_price = None
                    self._trade_pattern = None
                self._entry_order = None

            if order.status == order.Completed and order.issell() and not self.position:
                self._stop_price = None
                self._target_price = None
                self._trade_pattern = None

    def _finish_daily_bar(self):
        if self._daily_work is None:
            return

        self._daily.append(self._daily_work)
        self._daily_work = None
        self._update_pattern()

    def _update_pattern(self):
        self._pattern = None
        if len(self._daily) < 21:
            return

        sma_now = sum(bar["close"] for bar in self._daily[-20:]) / 20.0
        sma_previous = sum(bar["close"] for bar in self._daily[-21:-1]) / 20.0
        if sma_now < sma_previous:
            return

        pivot_lows = []
        for index in range(1, len(self._daily) - 1):
            low = self._daily[index]["low"]
            if low < self._daily[index - 1]["low"] and low <= self._daily[index + 1]["low"]:
                pivot_lows.append(index)

        if len(pivot_lows) < 3:
            return

        left, head, right = pivot_lows[-3:]
        if head - left < 2 or right - head < 2:
            return

        left_low = self._daily[left]["low"]
        head_low = self._daily[head]["low"]
        right_low = self._daily[right]["low"]

        shoulder_average = (left_low + right_low) / 2.0
        if shoulder_average <= 0.0:
            return
        if abs(left_low - right_low) > shoulder_average * 0.03:
            return
        if not (head_low < left_low and head_low < right_low):
            return

        first_peak = max(
            range(left + 1, head),
            key=lambda index: self._daily[index]["high"],
        )
        second_peak = max(
            range(head + 1, right),
            key=lambda index: self._daily[index]["high"],
        )

        first_high = self._daily[first_peak]["high"]
        second_high = self._daily[second_peak]["high"]
        slope = (second_high - first_high) / float(second_peak - first_peak)

        self._pattern = {
            "first_peak": first_peak,
            "first_high": first_high,
            "slope": slope,
            "head_low": head_low,
        }

    def _neckline(self, pattern, daily_index):
        return (
            pattern["first_high"]
            + pattern["slope"] * (daily_index - pattern["first_peak"])
        )

    def _on_four_hour_close(self, candle):
        daily_index = len(self._daily)

        if self.position and self._trade_pattern is not None and self._order is None:
            neckline = self._neckline(self._trade_pattern, daily_index)
            if candle["close"] < neckline:
                self._order = self.close()

        if (
            not self.position
            and self._order is None
            and self._pattern is not None
            and self._previous_red_volume is not None
        ):
            neckline = self._neckline(self._pattern, daily_index)
            crosses_neckline = (
                self._previous_four_close is not None
                and self._previous_four_close <= neckline
                and candle["close"] > neckline
            )
            volume_confirmed = candle["volume"] >= self._previous_red_volume * 1.5

            if crosses_neckline and volume_confirmed:
                stop_price = self._pattern["head_low"] - 0.01
                per_share_risk = candle["close"] - stop_price

                if per_share_risk > 0.0:
                    risk_capital = self.broker.getvalue() * 0.01
                    risk_size = math.floor(risk_capital / per_share_risk)
                    affordable_size = math.floor(self.broker.getcash() / candle["close"])
                    size = min(risk_size, affordable_size)

                    if size > 0:
                        self._stop_price = stop_price
                        self._target_price = (
                            candle["close"]
                            + (neckline - self._pattern["head_low"])
                        )
                        self._trade_pattern = dict(self._pattern)
                        self._order = self.buy(size=size)
                        self._entry_order = self._order

        if candle["close"] < candle["open"]:
            self._previous_red_volume = candle["volume"]

        self._previous_four_close = candle["close"]

    def next(self):
        dt = self.data.datetime.datetime(0)
        current_date = dt.date()

        if self._session_date is None:
            self._session_date = current_date
            self._daily_work = {
                "high": self.data.high[0],
                "low": self.data.low[0],
                "close": self.data.close[0],
            }
        elif current_date != self._session_date:
            self._finish_daily_bar()
            self._session_date = current_date
            self._four_hour = []
            self._daily_work = {
                "high": self.data.high[0],
                "low": self.data.low[0],
                "close": self.data.close[0],
            }
        else:
            self._daily_work["high"] = max(self._daily_work["high"], self.data.high[0])
            self._daily_work["low"] = min(self._daily_work["low"], self.data.low[0])
            self._daily_work["close"] = self.data.close[0]

        if self.position and self._order is None:
            if self.data.low[0] <= self._stop_price:
                self._order = self.close()
            elif self.data.high[0] >= self._target_price:
                self._order = self.close()

        self._four_hour.append({
            "open": self.data.open[0],
            "high": self.data.high[0],
            "low": self.data.low[0],
            "close": self.data.close[0],
            "volume": self.data.volume[0],
        })

        if len(self._four_hour) == 4:
            candle = {
                "open": self._four_hour[0]["open"],
                "high": max(bar["high"] for bar in self._four_hour),
                "low": min(bar["low"] for bar in self._four_hour),
                "close": self._four_hour[-1]["close"],
                "volume": sum(bar["volume"] for bar in self._four_hour),
            }
            self._four_hour = []
            self._on_four_hour_close(candle)
