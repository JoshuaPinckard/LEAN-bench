import math
import backtrader as bt


class FairValueArbitrage(bt.Strategy):
    def __init__(self):
        self.order = None

        # Indicators specified in the strategy
        self.ema = bt.ind.EMA(self.data.close, period=50)
        self.rsi = bt.ind.RSI(self.data.close, period=14)
        self.atr = bt.ind.ATR(self.data, period=14)
        self.bb = bt.ind.BollingerBands(self.data.close, period=20, devfactor=2)
        self.crossover = bt.ind.CrossOver(self.data.close, self.ema)

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return
        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            self.order = None

    def next(self):
        if self.order:
            return

        close = self.data.close[0]
        ema_val = self.ema[0]
        spread = (close - ema_val) / ema_val

        # Volatility Filter
        annual_vol = (self.atr[0] / close) * math.sqrt(252)
        bb_width = (self.bb.lines.top[0] - self.bb.lines.bot[0]) / self.bb.lines.mid[0]
        vol_filter_passed = (annual_vol <= 0.80) and (bb_width < 0.30)

        if self.position:
            entry_price = self.position.price
            basic_exit = self.crossover[0] != 0

            if self.position.size > 0:  # Long
                pnl_pct = (close - entry_price) / entry_price
                stop_loss = pnl_pct <= -0.05
                take_profit = pnl_pct >= 0.02
                if basic_exit or stop_loss or take_profit:
                    self.order = self.close()

            elif self.position.size < 0:  # Short
                pnl_pct = (entry_price - close) / entry_price
                stop_loss = pnl_pct <= -0.05
                take_profit = pnl_pct >= 0.02
                if basic_exit or stop_loss or take_profit:
                    self.order = self.close()
        else:
            if not vol_filter_passed:
                return

            rsi_val = self.rsi[0]

            # Long Entry: Spread <= -1% and RSI(14) < 45
            if spread <= -0.01 and rsi_val < 45:
                multiplier = min(1.5, 1.0 + abs(spread) / 0.02)
                base_amount = self.broker.getvalue() * 0.02
                target_amount = base_amount * multiplier
                size = int(target_amount / close)
                if size < 1:
                    size = 1
                self.order = self.buy(size=size)

            # Short Entry: Spread >= +1% and RSI(14) > 55
            elif spread >= 0.01 and rsi_val > 55:
                multiplier = min(1.5, 1.0 + abs(spread) / 0.02)
                base_amount = self.broker.getvalue() * 0.02
                target_amount = base_amount * multiplier
                size = int(target_amount / close)
                if size < 1:
                    size = 1
                self.order = self.sell(size=size)
