import backtrader as bt


class StreakReversalStrategy(bt.Strategy):
    params = (
        ("method", "swing"),          # "swing" or "day"
        ("stop_loss_pct", 0.01),
    )

    def __init__(self):
        self.order = None
        self.entry_order = None
        self.entry_price = None

        self.day_date = None
        self.day_open = None
        self.day_close = None
        self.previous_day_close = None
        self.red_streak = 0
        self.green_streak = 0

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return

        if order.status == order.Completed:
            if order is self.entry_order:
                self.entry_price = order.executed.price
                self.entry_order = None

        if order.status in (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            if order is self.order:
                self.order = None

    def _record_completed_day(self):
        color = None

        if self.p.method == "day":
            if self.day_close > self.day_open:
                color = "green"
            elif self.day_close < self.day_open:
                color = "red"
        else:
            if self.previous_day_close is not None:
                if self.day_close > self.previous_day_close:
                    color = "green"
                elif self.day_close < self.previous_day_close:
                    color = "red"

        if color == "green":
            self.green_streak += 1
            self.red_streak = 0
        elif color == "red":
            self.red_streak += 1
            self.green_streak = 0
        else:
            self.green_streak = 0
            self.red_streak = 0

        self.previous_day_close = self.day_close
        return color

    def next(self):
        current_date = self.data.datetime.date(0)
        current_time = self.data.datetime.time(0)
        close = self.data.close[0]

        if self.day_date is None:
            self.day_date = current_date
            self.day_open = self.data.open[0]
            self.day_close = close
            return

        completed_day_color = None
        if current_date != self.day_date:
            completed_day_color = self._record_completed_day()
            self.day_date = current_date
            self.day_open = self.data.open[0]
            self.day_close = close
        else:
            self.day_close = close

        if self.order:
            return

        if self.position and self.entry_price is not None:
            if self.position.size > 0:
                if close <= self.entry_price * (1.0 - self.p.stop_loss_pct):
                    self.order = self.close()
                    return
            else:
                if close >= self.entry_price * (1.0 + self.p.stop_loss_pct):
                    self.order = self.close()
                    return

        if self.p.method == "swing" and self.position:
            if self.position.size > 0 and completed_day_color == "red":
                self.order = self.close()
                return
            if self.position.size < 0 and completed_day_color == "green":
                self.order = self.close()
                return

        if self.p.method == "day" and self.position:
            if current_time.hour == 15 and current_time.minute == 55:
                self.order = self.close()
                return

        if self.position:
            return

        if len(self.data) < 2:
            return

        bullish_confirmation = close > self.data.high[-1]
        bearish_confirmation = close < self.data.low[-1]

        if self.red_streak >= 2 and bullish_confirmation:
            self.order = self.buy()
            self.entry_order = self.order
        elif self.green_streak >= 2 and bearish_confirmation:
            self.order = self.sell()
            self.entry_order = self.order
