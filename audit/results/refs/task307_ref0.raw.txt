import backtrader as bt


class AdxMomentumAtrStrategy(bt.Strategy):
    params = (
        ('AdxPeriod', 14),
        ('AdxThreshold', 25),
        ('MomentumPeriod', 5),
        ('ATRMultiplier', 2.0),
        ('AtrPeriod', 14),
    )

    def __init__(self):
        self.adx = bt.ind.ADX(self.data, period=self.p.AdxPeriod)
        self.mom = bt.ind.Momentum(self.data.close, period=self.p.MomentumPeriod)
        self.zero_line = self.mom - self.mom
        self.mom_cross = bt.ind.CrossOver(self.mom, self.zero_line)
        self.atr = bt.ind.ATR(self.data, period=self.p.AtrPeriod)

        self.order = None
        self.entry_price = None
        self.stop_price = None

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return

        if order.status == order.Completed:
            if self.position:
                self.entry_price = order.executed.price
                if self.position.size > 0:
                    self.stop_price = self.entry_price - self.p.ATRMultiplier * self.atr[0]
                else:
                    self.stop_price = self.entry_price + self.p.ATRMultiplier * self.atr[0]
            else:
                self.entry_price = None
                self.stop_price = None

        self.order = None

    def next(self):
        if self.order:
            return

        if not self.position:
            if self.adx[0] > self.p.AdxThreshold:
                if self.mom_cross[0] > 0:
                    self.order = self.buy()
                elif self.mom_cross[0] < 0:
                    self.order = self.sell()
        else:
            if self.position.size > 0:
                if self.mom_cross[0] < 0 or self.data.close[0] <= self.stop_price:
                    self.order = self.close()
            elif self.position.size < 0:
                if self.mom_cross[0] > 0 or self.data.close[0] >= self.stop_price:
                    self.order = self.close()
