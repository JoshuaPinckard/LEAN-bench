```python
import backtrader as bt
import math


class ElliottWavePatternStrategy(bt.Strategy):
    params = (
        ("pivot_span", 2),
        ("similarity", 0.02),
        ("minimum_pattern_size", 0.02),
        ("volume_period", 5),
        ("target_ratio", 1.272),
    )

    def __init__(self):
        self.order = None
        self.stop_order = None
        self.pivots = []
        self.entry_price = None
        self.stop_price = None
        self.target_price = None
        self.pattern_support = None
        self.pattern_resistance = None
        self.pattern_height = None
        self.last_wave_pivot_bar = None

    def _volume_confirmed(self):
        if len(self.data) < self.p.volume_period:
            return False
        average = sum(self.data.volume[-i] for i in range(self.p.volume_period)) / self.p.volume_period
        return self.data.volume[0] > average

    def _add_pivot(self):
        span = self.p.pivot_span
        if len(self.data) < span * 2 + 1:
            return False

        candidate_high = self.data.high[-span]
        candidate_low = self.data.low[-span]
        highs = [self.data.high[-i] for i in range(span * 2 + 1)]
        lows = [self.data.low[-i] for i in range(span * 2 + 1)]
        bar = len(self.data) - span - 1

        pivot = None
        if candidate_high == max(highs):
            pivot = ("H", candidate_high, bar)
        elif candidate_low == min(lows):
            pivot = ("L", candidate_low, bar)

        if pivot is None:
            return False

        if self.pivots and self.pivots[-1][0] == pivot[0]:
            if ((pivot[0] == "H" and pivot[1] > self.pivots[-1][1]) or
                    (pivot[0] == "L" and pivot[1] < self.pivots[-1][1])):
                self.pivots[-1] = pivot
            return False

        self.pivots.append(pivot)
        if len(self.pivots) > 20:
            self.pivots.pop(0)
        return True

    def _similar(self, a, b):
        return abs(a - b) / max(abs(a), abs(b), 1e-12) <= self.p.similarity

    def _wave(self):
        if len(self.pivots) < 6:
            return None

        p = self.pivots[-6:]
        kinds = "".join(x[0] for x in p)
        prices = [x[1] for x in p]

        if kinds == "LHLHLH":
            wave1 = prices[1] - prices[0]
            wave3 = prices[3] - prices[2]
            wave4 = prices[3] - prices[4]
            if wave1 > 0 and wave3 > 0 and wave4 > 0:
                return ("up", prices[5], prices[4], wave1, wave3, wave4, p[-1][2])

        if kinds == "HLHLHL":
            wave1 = prices[0] - prices[1]
            wave3 = prices[2] - prices[3]
            wave4 = prices[4] - prices[3]
            if wave1 > 0 and wave3 > 0 and wave4 > 0:
                return ("down", prices[5], prices[4], wave1, wave3, wave4, p[-1][2])

        return None

    def _pattern(self):
        if len(self.pivots) < 3 or not self._volume_confirmed():
            return None

        p = self.pivots
        close = self.data.close[0]

        # Double bottom / Double top
        if len(p) >= 3:
            a, b, c = p[-3:]
            if (a[0], b[0], c[0]) == ("L", "H", "L") and self._similar(a[1], c[1]):
                height = b[1] - min(a[1], c[1])
                if height / b[1] >= self.p.minimum_pattern_size and close > b[1]:
                    return ("bullish", b[1], min(a[1], c[1]), height)
            if (a[0], b[0], c[0]) == ("H", "L", "H") and self._similar(a[1], c[1]):
                height = max(a[1], c[1]) - b[1]
                if height / b[1] >= self.p.minimum_pattern_size and close < b[1]:
                    return ("bearish", b[1], max(a[1], c[1]), height)

        # Triple bottom / Triple top and inverse/regular Head and Shoulders
        if len(p) >= 5:
            a, b, c, d, e = p[-5:]
            kinds = "".join(x[0] for x in (a, b, c, d, e))

            if kinds == "LHLHL":
                neckline = (b[1] + d[1]) / 2.0
                lows = (a[1], c[1], e[1])
                height = neckline - min(lows)

                if self._similar(a[1], c[1]) and self._similar(c[1], e[1]):
                    if height / neckline >= self.p.minimum_pattern_size and close > neckline:
                        return ("bullish", neckline, min(lows), height)

                if c[1] < a[1] and c[1] < e[1] and self._similar(a[1], e[1]):
                    if height / neckline >= self.p.minimum_pattern_size and close > neckline:
                        return ("bullish", neckline, c[1], height)

            if kinds == "HLHLH":
                neckline = (b[1] + d[1]) / 2.0
                highs = (a[1], c[1], e[1])
                height = max(highs) - neckline

                if self._similar(a[1], c[1]) and self._similar(c[1], e[1]):
                    if height / neckline >= self.p.minimum_pattern_size and close < neckline:
                        return ("bearish", neckline, max(highs), height)

                if c[1] > a[1] and c[1] > e[1] and self._similar(a[1], e[1]):
                    if height / neckline >= self.p.minimum_pattern_size and close < neckline:
                        return ("bearish", neckline, c[1], height)

        # Ascending / descending Triangle
        if len(p) >= 4:
            a, b, c, d = p[-4:]
            kinds = "".join(x[0] for x in (a, b, c, d))
            if kinds == "HLHL" and self._similar(a[1], c[1]) and d[1] > b[1]:
                height = a[1] - b[1]
                if height / a[1] >= self.p.minimum_pattern_size and close > max(a[1], c[1]):
                    return ("bullish", max(a[1], c[1]), d[1], height)
            if kinds == "LHLH" and self._similar(a[1], c[1]) and d[1] < b[1]:
                height = b[1] - a[1]
                if height / b[1] >= self.p.minimum_pattern_size and close < min(a[1], c[1]):
                    return ("bearish", min(a[1], c[1]), d[1], height)

        # Flag and Pennant: short counter-trend consolidation followed by breakout.
        if len(self.data) >= 8:
            pole = self.data.close[-7] - self.data.close[-4]
            consolidation_high = max(self.data.high[-3], self.data.high[-2], self.data.high[-1])
            consolidation_low = min(self.data.low[-3], self.data.low[-2], self.data.low[-1])
            height = abs(pole)

            if height / max(self.data.close[-7], 1e-12) >= self.p.minimum_pattern_size:
                if pole > 0 and close > consolidation_high:
                    return ("bullish", consolidation_high, consolidation_low, height)
                if pole < 0 and close < consolidation_low:
                    return ("bearish", consolidation_low, consolidation_high, height)

        return None

    def _submit_entry(self, direction, pattern):
        _, neckline, extreme, height = pattern
        self.pattern_height = height
        self.pattern_support = extreme if direction == "bullish" else neckline
        self.pattern_resistance = neckline if direction == "bullish" else extreme
        self.stop_price = extreme

        if direction == "bullish":
            self.order = self.buy()
        else:
            self.order = self.sell()

    def _submit_stop(self):
        if not self.position or self.stop_price is None:
            return
        if self.position.size > 0:
            self.stop_order = self.sell(exectype=bt.Order.Stop, price=self.stop_price)
        else:
            self.stop_order = self.buy(exectype=bt.Order.Stop, price=self.stop_price)

    def _close_position(self):
        if self.stop_order is not None:
            self.cancel(self.stop_order)
            self.stop_order = None
        self.order = self.close()

    def next(self):
        pivot_added = self._add_pivot()
        wave = self._wave() if pivot_added else None
        pattern = self._pattern()

        bullish_entry = False
        bearish_entry = False

        if wave is not None and wave[6] != self.last_wave_pivot_bar:
            direction, endpoint, prior_extreme, wave1, wave3, wave4, pivot_bar = wave
            fib_3_target = wave1 * 1.618
            fib_4_level = wave3 * 0.618
            fib_bounce = abs(wave4 - fib_4_level) <= max(fib_4_level * self.p.similarity, 1e-12)

            if direction == "down" and pattern is not None and pattern[0] == "bullish":
                bullish_entry = True
            if direction == "up" and pattern is not None and pattern[0] == "bearish":
                bearish_entry = True

            if direction == "down" and fib_bounce and self.data.close[0] > endpoint:
                bullish_entry = True
                if pattern is None:
                    pattern = ("bullish", self.data.close[0], endpoint, abs(wave1))
            if direction == "up" and fib_bounce and self.data.close[0] < endpoint:
                bearish_entry = True
                if pattern is None:
                    pattern = ("bearish", self.data.close[0], endpoint, abs(wave1))

            self.last_wave_pivot_bar = pivot_bar

        if self.order is not None:
            return

        if not self.position:
            if bullish_entry and pattern is not None:
                self._submit_entry("bullish", pattern)
            elif bearish_entry and pattern is not None:
                self._submit_entry("bearish", pattern)
            return

        if self.position.size > 0:
            if self.target_price is not None and self.data.close[0] >= self.target_price:
                self._close_position()
            elif self.pattern_support is not None and self.data.close[0] < self.pattern_support:
                self._close_position()
            elif bearish_entry:
                self._close_position()
        else:
            if self.target_price is not None and self.data.close[0] <= self.target_price:
                self._close_position()
            elif self.pattern_resistance is not None and self.data.close[0] > self.pattern_resistance:
                self._close_position()
            elif bullish_entry:
                self._close_position()

    def notify_order(self, order):
        if order.status in (order.Completed, order.Canceled, order.Margin, order.Rejected):
            if order == self.order:
                self.order = None

            if order.status == order.Completed and order.isbuy() and self.position.size > 0:
                self.entry_price = order.executed.price
                self.target_price = self.entry_price + self.pattern_height * self.p.target_ratio
                self._submit_stop()
            elif order.status == order.Completed and order.issell() and self.position.size < 0:
                self.entry_price = order.executed.price
                self.target_price = self.entry_price - self.pattern_height * self.p.target_ratio
                self._submit_stop()

            if order == self.stop_order:
                self.stop_order = None
                if not self.position:
                    self.entry_price = None
                    self.stop_price = None
                    self.target_price = None
```
