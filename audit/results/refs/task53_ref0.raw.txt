import backtrader as bt


class SMTCPatternStrategy(bt.Strategy):
    params = dict(
        ema_period=9,
        entry_level=62.17,
        stop_loss_level=59.16,
        take_profit_level=65.33,
    )

    def __init__(self):
        self.ema = bt.indicators.ExponentialMovingAverage(
            self.data.close, period=self.p.ema_period
        )
        self.order = None
        self.tp_triggered = False

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def _is_piercing_pattern(self):
        open0, close0 = self.data.open[0], self.data.close[0]
        open1, close1, low1 = self.data.open[-1], self.data.close[-1], self.data.low[-1]

        first_bearish = close1 < open1
        second_bullish = close0 > open0
        opens_below_low = open0 < low1
        midpoint = (open1 + close1) / 2.0
        closes_within_body = midpoint <= close0 <= open1

        return first_bearish and second_bullish and opens_below_low and closes_within_body

    def _is_bullish_engulfing(self):
        open0, close0 = self.data.open[0], self.data.close[0]
        open1, close1 = self.data.open[-1], self.data.close[-1]

        first_bearish = close1 < open1
        second_bullish = close0 > open0
        engulfs = open0 <= close1 and close0 >= open1

        return first_bearish and second_bullish and engulfs

    def next(self):
        if self.order:
            return

        if not self.position:
            self.tp_triggered = False
            pattern_formed = self._is_piercing_pattern() or self._is_bullish_engulfing()
            if pattern_formed and self.data.close[0] > self.p.entry_level:
                self.order = self.buy()
            return

        if self.data.close[0] < self.p.stop_loss_level:
            self.order = self.close()
            return

        if not self.tp_triggered:
            if self.data.close[0] >= self.p.take_profit_level:
                size_to_close = self.position.size * 0.5
                self.order = self.sell(size=size_to_close)
                self.tp_triggered = True
            return

        if self.data.close[0] < self.ema[0]:
            self.order = self.close()
