import backtrader as bt


class RegimeTradingStrategy(bt.Strategy):
    params = (
        ("bb_period", 20),
        ("bb_devfactor", 2),
        ("ema_fast_period", 50),
        ("ema_slow_period", 200),
        ("width_sma_period", 200),
    )

    def __init__(self):
        self.order = None
        self.trade_regime = None

        self.bbands = bt.indicators.BollingerBands(
            self.data.close,
            period=self.p.bb_period,
            devfactor=self.p.bb_devfactor,
        )
        self.ema50 = bt.indicators.EMA(
            self.data.close, period=self.p.ema_fast_period
        )
        self.ema200 = bt.indicators.EMA(
            self.data.close, period=self.p.ema_slow_period
        )
        self.width = self.bbands.top - self.bbands.bot
        self.width_sma = bt.indicators.SMA(
            self.width, period=self.p.width_sma_period
        )

    def notify_order(self, order):
        if order.status == order.Completed:
            if order.info.get("kind") == "entry":
                self.trade_regime = order.info["regime"]
            elif order.info.get("kind") == "exit":
                self.trade_regime = None

        if order.status in (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            self.order = None

    def _crossed_above(self, line):
        return self.data.close[0] > line[0] and self.data.close[-1] <= line[-1]

    def _crossed_below(self, line):
        return self.data.close[0] < line[0] and self.data.close[-1] >= line[-1]

    def _enter(self, target, regime):
        self.order = self.order_target_percent(target=target)
        self.order.addinfo(kind="entry", regime=regime)

    def _exit(self):
        self.order = self.close()
        self.order.addinfo(kind="exit")

    def next(self):
        if self.order:
            return

        mean_reversion = self.width[0] <= self.width_sma[0]
        trend = self.width[0] > self.width_sma[0]

        if self.position:
            if self.position.size > 0:
                if (
                    self.trade_regime == "mean_reversion"
                    and self._crossed_above(self.bbands.mid)
                ):
                    self._exit()
                elif (
                    self.trade_regime == "trend"
                    and self._crossed_below(self.bbands.mid)
                ):
                    self._exit()
            else:
                if (
                    self.trade_regime == "mean_reversion"
                    and self._crossed_below(self.bbands.mid)
                ):
                    self._exit()
                elif (
                    self.trade_regime == "trend"
                    and self._crossed_above(self.bbands.mid)
                ):
                    self._exit()
            return

        if mean_reversion:
            if self.data.close[0] <= self.bbands.bot[0]:
                self._enter(1.0, "mean_reversion")
            elif self.data.close[0] >= self.bbands.top[0]:
                self._enter(-1.0, "mean_reversion")
        elif trend:
            if (
                self.ema50[0] > self.ema200[0]
                and self._crossed_above(self.bbands.top)
            ):
                self._enter(1.0, "trend")
            elif (
                self.ema50[0] < self.ema200[0]
                and self._crossed_below(self.bbands.bot)
            ):
                self._enter(-1.0, "trend")
