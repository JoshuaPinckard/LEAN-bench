import backtrader as bt


class MonthlySmaStopStrategy(bt.Strategy):
    def __init__(self):
        self.sma = bt.indicators.SimpleMovingAverage(self.data.close, period=20)
        self.order = None
        self.stop_order = None
        self.current_month = None
        self.trading_day_of_month = 0

    def next(self):
        current_date = self.data.datetime.date(0)
        month_key = (current_date.year, current_date.month)

        if month_key != self.current_month:
            self.current_month = month_key
            self.trading_day_of_month = 1
        else:
            self.trading_day_of_month += 1

        if self.trading_day_of_month == 1:
            if not self.position and self.order is None and self.stop_order is None:
                if self.data.close[0] > self.sma[0]:
                    self.order = self.buy()

        if self.trading_day_of_month == 5 and self.position:
            if self.stop_order is not None:
                self.cancel(self.stop_order)
            if self.order is None:
                self.order = self.close()

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            if order == self.stop_order:
                self.stop_order = None
                return

            if order == self.order:
                self.order = None

            if order.status == order.Completed and order.isbuy():
                self.stop_order = self.sell(
                    exectype=bt.Order.Stop,
                    price=order.executed.price * 0.98,
                )
