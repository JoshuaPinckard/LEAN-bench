import backtrader as bt


class ScalpingStrategy(bt.Strategy):
    params = (
        ("stake", 100),
        ("move_threshold", 0.002),
        ("vol_threshold", 0.003),
        ("tp_pct", 0.003),
        ("sl_pct", 0.003),
    )

    def __init__(self):
        self.order = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            self.order = None

    def next(self):
        if self.order:
            return

        if not self.position:
            if self.data.close[0] == 0 or self.data.close[-1] == 0:
                return

            bar_range = (
                self.data.high[0] - self.data.low[0]
            ) / self.data.close[0]
            if bar_range < self.p.vol_threshold:
                return

            price_change = (
                self.data.close[0] - self.data.close[-1]
            ) / self.data.close[-1]

            if price_change >= self.p.move_threshold:
                self.order = self.buy(size=self.p.stake)
            elif price_change <= -self.p.move_threshold:
                self.order = self.sell(size=self.p.stake)
        else:
            entry_price = self.position.price
            if self.position.size > 0:
                tp_price = entry_price * (1.0 + self.p.tp_pct)
                sl_price = entry_price * (1.0 - self.p.sl_pct)

                if (
                    self.data.low[0] <= sl_price
                    or self.data.high[0] >= tp_price
                ):
                    self.order = self.close()
            elif self.position.size < 0:
                tp_price = entry_price * (1.0 - self.p.tp_pct)
                sl_price = entry_price * (1.0 + self.p.sl_pct)

                if (
                    self.data.high[0] >= sl_price
                    or self.data.low[0] <= tp_price
                ):
                    self.order = self.close()
