import backtrader as bt
import math
from collections import deque


class FinancialTurbulenceStrategy(bt.Strategy):
    params = (
        ("lookback_months", 24),
        ("ma_months", 12),
        ("capital_fraction", 0.05),
        ("stop_loss", 0.02),
    )

    def __init__(self):
        self.order = None
        self.stop_order = None
        self.month_closes = deque()
        self.monthly_returns = deque()
        self.turbulence_values = deque()
        self.previous_turbulence = None
        self.entry_price = None

    def _percentile_95(self, values):
        ordered = sorted(values)
        index = math.ceil(0.95 * len(ordered)) - 1
        return ordered[index]

    def _monthly_bar(self):
        return len(self.data) == 1 or self.data.datetime.date(0).month != self.data.datetime.date(1).month

    def next(self):
        if not self._monthly_bar():
            return

        close = self.data.close[0]

        if self.month_closes:
            self.monthly_returns.append((close / self.month_closes[-1]) - 1.0)
        self.month_closes.append(close)

        n = self.p.lookback_months
        if len(self.monthly_returns) <= n:
            return

        prior_returns = list(self.monthly_returns)[-n-1:-1]
        current_return = self.monthly_returns[-1]
        mean_return = sum(prior_returns) / n
        covariance = sum(
            (value - mean_return) ** 2 for value in prior_returns
        ) / (n - 1)

        if covariance == 0:
            return

        turbulence = ((current_return - mean_return) ** 2) / covariance

        threshold = None
        if len(self.turbulence_values) >= n:
            threshold = self._percentile_95(list(self.turbulence_values)[-n:])

        self.turbulence_values.append(turbulence)

        if self.order:
            self.previous_turbulence = turbulence
            return

        if self.position:
            if len(self.turbulence_values) >= self.p.ma_months:
                turbulence_ma = sum(
                    list(self.turbulence_values)[-self.p.ma_months:]
                ) / self.p.ma_months
                if turbulence < turbulence_ma:
                    if self.stop_order:
                        self.cancel(self.stop_order)
                    self.order = self.close()

        elif (
            threshold is not None
            and self.previous_turbulence is not None
            and self.previous_turbulence <= threshold
            and turbulence > threshold
        ):
            size = (self.broker.getvalue() * self.p.capital_fraction) / close
            self.order = self.buy(size=size)

        self.previous_turbulence = turbulence

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            if order == self.order:
                self.order = None

            if order.status == order.Completed and order.isbuy():
                self.entry_price = order.executed.price
                self.stop_order = self.sell(
                    exectype=bt.Order.Stop,
                    price=self.entry_price * (1.0 - self.p.stop_loss),
                    size=order.executed.size,
                )

            if order == self.stop_order:
                self.stop_order = None
