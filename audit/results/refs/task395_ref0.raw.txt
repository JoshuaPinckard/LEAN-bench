import backtrader as bt


class HmaBandMeanReversion(bt.Strategy):
    params = dict(
        hma_period=9,
        band_period=20,
        band_dev=2.0,
        atr_period=14,
        atr_mult=2.0,
    )

    def __init__(self):
        self.hma = bt.indicators.HullMovingAverage(
            self.data.close, period=self.p.hma_period)
        self.sma_hma = bt.indicators.SMA(self.hma, period=self.p.band_period)
        self.std_hma = bt.indicators.StdDev(self.hma, period=self.p.band_period)
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)

        self.upper_band = self.sma_hma + self.p.band_dev * self.std_hma
        self.lower_band = self.sma_hma - self.p.band_dev * self.std_hma

        self.cross_lower = bt.indicators.CrossOver(self.hma, self.lower_band)
        self.cross_upper = bt.indicators.CrossOver(self.hma, self.upper_band)
        self.cross_mid = bt.indicators.CrossOver(self.hma, self.sma_hma)

        self.order = None
        self.entry_order = False
        self.stop_price = None

    def notify_order(self, order):
        if order.status == order.Completed and self.entry_order:
            if order.isbuy():
                self.stop_price = order.executed.price - self.p.atr_mult * self.atr[0]
            else:
                self.stop_price = order.executed.price + self.p.atr_mult * self.atr[0]

        if order.status in (order.Completed, order.Canceled, order.Margin, order.Rejected):
            self.order = None
            self.entry_order = False

    def next(self):
        if self.order:
            return

        if self.position:
            if self.position.size > 0:
                if self.data.low[0] <= self.stop_price:
                    self.order = self.close()
                elif self.cross_mid[0] > 0:
                    self.order = self.close()
            elif self.position.size < 0:
                if self.data.high[0] >= self.stop_price:
                    self.order = self.close()
                elif self.cross_mid[0] < 0:
                    self.order = self.close()
        else:
            if self.cross_lower[0] > 0:
                self.entry_order = True
                self.order = self.buy()
            elif self.cross_upper[0] < 0:
                self.entry_order = True
                self.order = self.sell()
