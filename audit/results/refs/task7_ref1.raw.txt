import backtrader as bt


class RSITradingStrategy(bt.Strategy):
    params = (
        ('rsi_period', 14),
        ('oversold', 30),
        ('overbought', 70),
        ('stop_loss_pct', 0.02),
        ('take_profit_pct', 0.04),
    )

    def __init__(self):
        self.rsi = bt.indicators.RSI(
            self.data.close,
            period=self.p.rsi_period,
            movav=bt.indicators.SmoothedMovingAverage,
        )
        self.entry_order = None
        self.exit_order = None
        self.stop_order = None
        self.take_profit_order = None

    def next(self):
        if self.entry_order or self.exit_order:
            return

        if self.position:
            if self.position.size > 0 and self.rsi[0] > self.p.overbought:
                if self.stop_order:
                    self.cancel(self.stop_order)
                if self.take_profit_order:
                    self.cancel(self.take_profit_order)
                self.exit_order = self.close()
            elif self.position.size < 0 and self.rsi[0] < self.p.oversold:
                if self.stop_order:
                    self.cancel(self.stop_order)
                if self.take_profit_order:
                    self.cancel(self.take_profit_order)
                self.exit_order = self.close()
            return

        if self.rsi[-1] < self.p.oversold and self.rsi[0] > self.p.oversold:
            self.entry_order = self.buy()
        elif self.rsi[-1] > self.p.overbought and self.rsi[0] < self.p.overbought:
            self.entry_order = self.sell()

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return

        if order.status == order.Completed:
            if order == self.entry_order:
                entry_price = order.executed.price

                if order.isbuy():
                    self.stop_order = self.sell(
                        exectype=bt.Order.Stop,
                        price=entry_price * (1.0 - self.p.stop_loss_pct),
                    )
                    self.take_profit_order = self.sell(
                        exectype=bt.Order.Limit,
                        price=entry_price * (1.0 + self.p.take_profit_pct),
                        oco=self.stop_order,
                    )
                else:
                    self.stop_order = self.buy(
                        exectype=bt.Order.Stop,
                        price=entry_price * (1.0 + self.p.stop_loss_pct),
                    )
                    self.take_profit_order = self.buy(
                        exectype=bt.Order.Limit,
                        price=entry_price * (1.0 - self.p.take_profit_pct),
                        oco=self.stop_order,
                    )

                self.entry_order = None
            elif order == self.exit_order:
                self.exit_order = None
            elif order == self.stop_order:
                self.stop_order = None
            elif order == self.take_profit_order:
                self.take_profit_order = None

        elif order.status in (order.Canceled, order.Margin, order.Rejected):
            if order == self.entry_order:
                self.entry_order = None
            elif order == self.exit_order:
                self.exit_order = None
            elif order == self.stop_order:
                self.stop_order = None
            elif order == self.take_profit_order:
                self.take_profit_order = None
