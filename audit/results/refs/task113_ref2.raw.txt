```python
import backtrader as bt
import math
import collections
import statistics


class CostAwarePortfolioStrategy(bt.Strategy):
    params = (
        ('lookback', 252),       # bars for r and sigma estimation (1 year)
        ('lambda_risk', 1.0),    # risk-aversion coefficient lambda
        ('theta', 0.1),          # market-impact coefficient Theta
        ('n_grid', 1000),        # grid points for 1-D optimisation
    )

    def __init__(self):
        self.order = None
        self._returns = collections.deque(maxlen=self.p.lookback)
        self._prev_close = None

    def notify_order(self, order):
        if order.status in [
            order.Completed, order.Canceled,
            order.Margin, order.Rejected,
        ]:
            self.order = None

    def _objective(self, w, r, sigma2, w0):
        """
        Scalar form of (r - I(t))^T w - lambda w^T Sigma w - Theta |w|^(3/2)
        I(t) = Theta * sqrt(|w - w0|), t = |w - w0|
        """
        t = abs(w - w0)
        I_t = self.p.theta * math.sqrt(t) if t > 0.0 else 0.0
        return (
            (r - I_t) * w
            - self.p.lambda_risk * sigma2 * w * w
            - self.p.theta * math.pow(w, 1.5)
        )

    def next(self):
        if self._prev_close is not None:
            self._returns.append(
                (self.data.close[0] - self._prev_close) / self._prev_close
            )
        self._prev_close = self.data.close[0]

        if len(self._returns) < self.p.lookback:
            return
        if self.order:
            return

        rets = list(self._returns)
        r = statistics.mean(rets) * 252          # annualised expected return
        sigma2 = statistics.variance(rets) * 252 # annualised variance

        portfolio_value = self.broker.get_value()
        if portfolio_value <= 0.0:
            return
        w0 = (self.position.size * self.data.close[0]) / portfolio_value

        # Solve: maximise f(w) over long-only w in [0, 1] via grid search
        best_w = 0.0
        best_f = self._objective(0.0, r, sigma2, w0)
        for i in range(1, self.p.n_grid + 1):
            w_cand = i / self.p.n_grid
            fv = self._objective(w_cand, r, sigma2, w0)
            if fv > best_f:
                best_f = fv
                best_w = w_cand

        self.order = self.order_target_percent(self.data, best_w)
```
