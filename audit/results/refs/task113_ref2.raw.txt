import statistics
import backtrader as bt


class PortfolioCostOptimizationStrategy(bt.Strategy):
    params = (
        ("lookback", 252),
        ("risk_aversion", 1.0),  # lambda
        ("impact_coef", 0.001),  # Theta
        ("min_weight", 0.0),  # l_b
        ("max_weight", 1.0),  # u_b
    )

    def __init__(self):
        self.order = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return
        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            self.order = None

    def next(self):
        if len(self.data) <= self.p.lookback:
            return

        if self.order:
            return

        returns = [
            (self.data.close[-i] - self.data.close[-i - 1])
            / self.data.close[-i - 1]
            for i in range(self.p.lookback)
        ]

        r = statistics.mean(returns)
        var = statistics.variance(returns) if len(returns) > 1 else 0.0

        total_value = self.broker.getvalue()
        price = self.data.close[0]

        if total_value <= 0 or price <= 0:
            return

        current_size = self.position.size
        w0 = (current_size * price) / total_value
        w0 = max(self.p.min_weight, min(self.p.max_weight, w0))

        best_w = w0
        best_utility = -float("inf")
        steps = 1000
        step_size = (self.p.max_weight - self.p.min_weight) / steps

        for i in range(steps + 1):
            w = self.p.min_weight + i * step_size
            utility = (
                r * w
                - self.p.risk_aversion * var * (w**2)
                - self.p.impact_coef * (abs(w - w0) ** 1.5)
            )
            if utility > best_utility:
                best_utility = utility
                best_w = w

        target_value = best_w * total_value
        target_size = int(target_value / price)
        size_diff = target_size - current_size

        if size_diff > 0:
            self.order = self.buy(size=size_diff)
        elif size_diff < 0:
            self.order = self.sell(size=abs(size_diff))
