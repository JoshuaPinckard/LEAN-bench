import backtrader as bt


class TimeSeriesMomentumStrategy(bt.Strategy):
    params = dict(
        N=90,
        R=21,
        w=0.20,
        max_alloc=0.10,
        vol_period=21,
        min_dollar_vol=1_000_000,
        flat_band=1e-4,
    )

    def __init__(self):
        dollar_volume = self.data.close * self.data.volume
        self.avg_dollar_vol = bt.indicators.SMA(dollar_volume, period=self.p.vol_period)
        self.crossover = bt.indicators.CrossOver(self.data.close, self.data.close(-self.p.N))
        self.orders = []

    def notify_order(self, order):
        if order.status in (order.Completed, order.Canceled, order.Margin, order.Rejected):
            if order in self.orders:
                self.orders.remove(order)

    def next(self):
        if self.orders:
            return

        is_rebalance_day = len(self) % self.p.R == 0

        if not is_rebalance_day:
            if self.position and self.crossover[0] != 0:
                order = self.close()
                if order:
                    self.orders.append(order)
            return

        if self.position:
            order = self.close()
            if order:
                self.orders.append(order)

        has_history = len(self) >= self.p.N + 1
        liquid = self.avg_dollar_vol[0] >= self.p.min_dollar_vol
        if not (has_history and liquid):
            return

        momentum = self.data.close[0] / self.data.close[-self.p.N] - 1.0
        if abs(momentum) < self.p.flat_band:
            return

        alloc_fraction = min(self.p.w, self.p.max_alloc)
        target_value = self.broker.getvalue() * alloc_fraction
        size = int(target_value / self.data.close[0])
        if size <= 0:
            return

        if momentum > 0:
            order = self.buy(size=size)
        else:
            order = self.sell(size=size)
        if order:
            self.orders.append(order)
