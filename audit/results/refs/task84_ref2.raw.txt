import backtrader as bt


class SuperTrend(bt.Indicator):
    lines = ('supertrend', 'direction')
    params = (
        ('period', 10),
        ('multiplier', 3.0),
    )

    def __init__(self):
        self.addminperiod(self.p.period + 1)
        self.atr = bt.indicators.ATR(self.data, period=self.p.period)

    def next(self):
        hl2 = (self.data.high[0] + self.data.low[0]) / 2.0
        basic_ub = hl2 + self.p.multiplier * self.atr[0]
        basic_lb = hl2 - self.p.multiplier * self.atr[0]

        if not hasattr(self, '_final_ub'):
            self._final_ub = basic_ub
            self._final_lb = basic_lb
            if self.data.close[0] > basic_ub:
                self.lines.supertrend[0] = basic_lb
                self.lines.direction[0] = 1
            else:
                self.lines.supertrend[0] = basic_ub
                self.lines.direction[0] = -1
            return

        prev_final_ub = self._final_ub
        prev_final_lb = self._final_lb
        prev_close = self.data.close[-1]
        prev_dir = self.lines.direction[-1]

        if basic_ub < prev_final_ub or prev_close > prev_final_ub:
            final_ub = basic_ub
        else:
            final_ub = prev_final_ub

        if basic_lb > prev_final_lb or prev_close < prev_final_lb:
            final_lb = basic_lb
        else:
            final_lb = prev_final_lb

        self._final_ub = final_ub
        self._final_lb = final_lb

        if prev_dir == 1:
            if self.data.close[0] < final_lb:
                curr_dir = -1
                st = final_ub
            else:
                curr_dir = 1
                st = final_lb
        else:
            if self.data.close[0] > final_ub:
                curr_dir = 1
                st = final_lb
            else:
                curr_dir = -1
                st = final_ub

        self.lines.supertrend[0] = st
        self.lines.direction[0] = curr_dir


class SupertrendStrategy(bt.Strategy):
    params = (
        ('period', 10),
        ('multiplier', 3.0),
    )

    def __init__(self):
        self.st = SuperTrend(self.data, period=self.p.period, multiplier=self.p.multiplier)
        self.pending_orders = []

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            if order in self.pending_orders:
                self.pending_orders.remove(order)

    def next(self):
        if self.pending_orders:
            return

        if len(self.st.direction) < 2:
            return

        prev_dir = self.st.direction[-1]
        curr_dir = self.st.direction[0]

        if prev_dir == -1 and curr_dir == 1:
            if self.position.size < 0:
                o1 = self.close()
                self.pending_orders.append(o1)
            if self.position.size <= 0:
                cash_or_val = self.broker.getcash() if self.position.size == 0 else self.broker.getvalue()
                size = int((cash_or_val * 0.98) / self.data.close[0])
                if size > 0:
                    o2 = self.buy(size=size)
                    self.pending_orders.append(o2)

        elif prev_dir == 1 and curr_dir == -1:
            if self.position.size > 0:
                o1 = self.close()
                self.pending_orders.append(o1)
            if self.position.size >= 0:
                cash_or_val = self.broker.getcash() if self.position.size == 0 else self.broker.getvalue()
                size = int((cash_or_val * 0.98) / self.data.close[0])
                if size > 0:
                    o2 = self.sell(size=size)
                    self.pending_orders.append(o2)
