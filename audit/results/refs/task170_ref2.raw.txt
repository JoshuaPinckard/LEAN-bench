import backtrader as bt


class SwingBreakoutStrategy(bt.Strategy):
    def __init__(self):
        # Indicators
        self.sma200 = bt.ind.SMA(self.data.close, period=200)
        self.bb = bt.ind.BollingerBands(self.data.close, period=20, devfactor=2.0)
        self.atr = bt.ind.ATR(self.data, period=14)
        self.sma_vol = bt.ind.SMA(self.data.volume, period=20)
        self.hhv20 = bt.ind.Highest(self.data.high, period=20)
        self.llv20 = bt.ind.Lowest(self.data.low, period=20)

        # Pending entry order tracking
        self.entry_order = None
        self.pending_entry_price = None
        self.pending_stop_price = None

        # Position exit order tracking
        self.t1_limit_order = None
        self.t1_stop_order = None
        self.t2_limit_order = None
        self.t2_stop_order = None
        self.t3_stop_order = None

        # Position state tracking
        self.in_position = False
        self.actual_entry_price = None
        self.highest_close_since_entry = None
        self.t3_stop_price = None

    def reset_entry_state(self):
        self.entry_order = None
        self.pending_entry_price = None
        self.pending_stop_price = None

    def reset_position_orders(self):
        self.t1_limit_order = None
        self.t1_stop_order = None
        self.t2_limit_order = None
        self.t2_stop_order = None
        self.t3_stop_order = None
        self.in_position = False
        self.actual_entry_price = None
        self.highest_close_since_entry = None
        self.t3_stop_price = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order == self.entry_order:
            if order.status == order.Completed:
                self.entry_order = None
                entry_p = self.pending_entry_price
                stop_p = self.pending_stop_price

                # Check breakout bar validation conditions
                gap_up = self.data.open[0] > entry_p
                stop_hit = self.data.low[0] <= stop_p
                vol_ok = self.data.volume[0] > (self.sma_vol[0] * 1.3)

                if gap_up or stop_hit or not vol_ok:
                    self.close()
                    self.reset_entry_state()
                    self.reset_position_orders()
                else:
                    self.in_position = True
                    self.actual_entry_price = order.executed.price
                    self.highest_close_since_entry = self.data.close[0]
                    self.t3_stop_price = stop_p

                    total_size = self.position.size
                    s1 = total_size // 3
                    s2 = total_size // 3
                    s3 = total_size - s1 - s2

                    if s1 > 0:
                        t1_target = self.actual_entry_price * 1.02
                        self.t1_limit_order = self.sell(size=s1, exectype=bt.Order.Limit, price=t1_target)
                        self.t1_stop_order = self.sell(size=s1, exectype=bt.Order.Stop, price=stop_p, oco=self.t1_limit_order)

                    if s2 > 0:
                        t2_target = self.actual_entry_price * 1.07
                        self.t2_limit_order = self.sell(size=s2, exectype=bt.Order.Limit, price=t2_target)
                        self.t2_stop_order = self.sell(size=s2, exectype=bt.Order.Stop, price=stop_p, oco=self.t2_limit_order)

                    if s3 > 0:
                        self.t3_stop_order = self.sell(size=s3, exectype=bt.Order.Stop, price=stop_p)

                    self.reset_entry_state()
            else:
                self.reset_entry_state()

        else:
            if order == self.t1_limit_order and order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
                self.t1_limit_order = None
            elif order == self.t1_stop_order and order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
                self.t1_stop_order = None
            elif order == self.t2_limit_order and order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
                self.t2_limit_order = None
            elif order == self.t2_stop_order and order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
                self.t2_stop_order = None
            elif order == self.t3_stop_order and order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
                self.t3_stop_order = None

    def next(self):
        # Reset position tracking if position was completely closed
        if self.position.size == 0 and self.in_position:
            self.reset_position_orders()

        # Manage open position
        if self.position.size > 0 and self.in_position:
            self.highest_close_since_entry = max(self.highest_close_since_entry, self.data.close[0])
            chandelier_trail = self.highest_close_since_entry - 3.0 * self.atr[0]

            if chandelier_trail > self.t3_stop_price:
                self.t3_stop_price = chandelier_trail

                rem_size = self.position.size
                if self.t1_limit_order and self.t1_limit_order.status in [bt.Order.Submitted, bt.Order.Accepted]:
                    rem_size -= self.t1_limit_order.created.size
                if self.t2_limit_order and self.t2_limit_order.status in [bt.Order.Submitted, bt.Order.Accepted]:
                    rem_size -= self.t2_limit_order.created.size

                if rem_size > 0:
                    if self.t3_stop_order and self.t3_stop_order.status in [bt.Order.Submitted, bt.Order.Accepted]:
                        self.cancel(self.t3_stop_order)
                    self.t3_stop_order = self.sell(size=rem_size, exectype=bt.Order.Stop, price=self.t3_stop_price)

            return

        # Evaluate cancellation of pending entry order before fill
        if self.entry_order is not None:
            atr_ratio = self.atr[0] / self.data.close[0]
            bb_mid = self.bb.mid[0]
            bb_width = (self.bb.top[0] - self.bb.bot[0]) / bb_mid if bb_mid != 0 else 1.0

            volatility_jump = (atr_ratio >= 0.025) or (bb_width >= 0.06)
            stop_level_hit = self.data.low[0] <= self.pending_stop_price

            if volatility_jump or stop_level_hit:
                self.cancel(self.entry_order)
                self.reset_entry_state()
            return

        # Signal generation when flat and no order is pending
        if self.position.size == 0 and self.entry_order is None:
            # 1) Trend filter
            if not (self.data.close[0] > self.sma200[0]):
                return

            # 2) Consolidation filter (at least 8 of last 10 bars meet low volatility criteria)
            low_vol_count = 0
            for i in range(10):
                c_price = self.data.close[-i]
                c_atr = self.atr[-i]
                c_mid = self.bb.mid[-i]
                c_width = (self.bb.top[-i] - self.bb.bot[-i]) / c_mid if c_mid != 0 else 1.0

                if (c_atr / c_price < 0.02) and (c_width < 0.04):
                    low_vol_count += 1

            if low_vol_count < 8:
                return

            # 3 & 4) Resistance/Support & Buy Stop order
            hhv20 = self.hhv20[0]
            llv20 = self.llv20[0]

            entry_price = hhv20 + max(0.001 * self.data.close[0], 0.25 * self.atr[0])
            stop_price = llv20 - 0.25 * self.atr[0]

            self.pending_entry_price = entry_price
            self.pending_stop_price = stop_price
            self.entry_order = self.buy(exectype=bt.Order.Stop, price=entry_price)
