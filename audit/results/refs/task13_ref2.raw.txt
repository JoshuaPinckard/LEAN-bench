import backtrader as bt


class PyramidingInTrend(bt.Strategy):
    params = (
        ('ema_period', 50),
        ('rsi_period', 14),
    )

    def __init__(self):
        self.ema = bt.ind.EMA(self.data.close, period=self.p.ema_period)
        self.rsi = bt.ind.RSI(self.data.close, period=self.p.rsi_period)

        self.order = None
        self.unit_size = None
        self.entries_count = 0

        self.rsi_below_50_for_entry1 = False
        self.rsi_reached_70_for_pyramid = False
        self.rsi_above_60_for_exit = False

        self.breakeven_done = False
        self.tp_done = False

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            if order.isbuy():
                self.entries_count += 1
                self.breakeven_done = False
                self.tp_done = False
            elif order.issell():
                if self.position.size == 0:
                    self.entries_count = 0
                    self.rsi_above_60_for_exit = False
                    self.rsi_reached_70_for_pyramid = False
                    self.breakeven_done = False
                    self.tp_done = False

        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        if self.unit_size is None:
            initial_cash = self.broker.get_cash()
            self.unit_size = (0.01 * initial_cash) / self.data.close[0]

        if self.order:
            return

        rsi_val = self.rsi[0]
        close_val = self.data.close[0]
        ema_val = self.ema[0]

        if self.position.size == 0:
            if rsi_val < 50:
                self.rsi_below_50_for_entry1 = True
        else:
            if rsi_val > 60:
                self.rsi_above_60_for_exit = True
            if rsi_val >= 70:
                self.rsi_reached_70_for_pyramid = True

        if self.position.size > 0:
            if close_val < ema_val:
                self.order = self.close()
                return

            if self.rsi_above_60_for_exit and rsi_val < 40:
                self.order = self.close()
                return

            pnl_pct = (close_val - self.position.price) / self.position.price

            if not self.breakeven_done and not self.tp_done:
                if pnl_pct >= 0.05:
                    close_qty = self.position.size * 0.625
                    self.order = self.sell(size=close_qty)
                    self.breakeven_done = True
                    self.tp_done = True
                    return
                elif pnl_pct >= 0.0:
                    close_qty = self.position.size * 0.25
                    self.order = self.sell(size=close_qty)
                    self.breakeven_done = True
                    return
            elif self.breakeven_done and not self.tp_done:
                if pnl_pct >= 0.05:
                    close_qty = self.position.size * 0.50
                    self.order = self.sell(size=close_qty)
                    self.tp_done = True
                    return

            if self.entries_count < 4 and self.rsi_reached_70_for_pyramid and rsi_val <= 60 and close_val > ema_val:
                self.order = self.buy(size=self.unit_size)
                self.rsi_reached_70_for_pyramid = False
                return

        else:
            if self.rsi_below_50_for_entry1 and rsi_val > 50 and close_val > ema_val:
                self.order = self.buy(size=self.unit_size)
                self.rsi_below_50_for_entry1 = False
                return
