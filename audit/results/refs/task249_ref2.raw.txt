import backtrader as bt


class IchimokuSmaMomentum(bt.Strategy):
    def __init__(self):
        self.sma9 = bt.ind.SMA(self.data.close, period=9)
        self.sma26 = bt.ind.SMA(self.data.close, period=26)
        self.sma52 = bt.ind.SMA(self.data.close, period=52)
        self.crossover = bt.ind.CrossOver(self.sma9, self.sma26)
        self.order = None

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        if self.order:
            return

        close = self.data.close[0]
        sma26 = self.sma26[0]
        sma52 = self.sma52[0]

        if not self.position:
            # Long Entry
            if (self.crossover[0] > 0 and
                close > sma26 and
                close > sma52 and
                sma26 > sma52):
                
                cash = self.broker.getcash()
                comm = self.broker.getcommissioninfo(self.data).p.commission
                size = int(cash / (close * (1.0 + comm)))
                if size > 0:
                    self.order = self.buy(size=size)

            # Short Entry
            elif (self.crossover[0] < 0 and
                  close < sma26 and
                  close < sma52 and
                  sma26 < sma52):
                
                cash = self.broker.getcash()
                comm = self.broker.getcommissioninfo(self.data).p.commission
                size = int(cash / (close * (1.0 + comm)))
                if size > 0:
                    self.order = self.sell(size=size)

        elif self.position.size > 0:
            # Long Exit
            if self.crossover[0] < 0 or close < min(sma26, sma52):
                self.order = self.close()

        elif self.position.size < 0:
            # Short Exit
            if self.crossover[0] > 0 or close > max(sma26, sma52):
                self.order = self.close()
