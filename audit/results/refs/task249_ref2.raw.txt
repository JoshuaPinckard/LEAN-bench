import backtrader as bt
import math

class MomentumIchimokuSMA(bt.Strategy):

    def __init__(self):
        self.sma9  = bt.indicators.SMA(self.data.close, period=9)
        self.sma26 = bt.indicators.SMA(self.data.close, period=26)
        self.sma52 = bt.indicators.SMA(self.data.close, period=52)
        self.crossover = bt.indicators.CrossOver(self.sma9, self.sma26)
        self.order = None

    def notify_order(self, order):
        if order.status in [
            order.Completed, order.Canceled, order.Margin, order.Rejected
        ]:
            self.order = None

    def next(self):
        if self.order:
            return

        close  = self.data.close[0]
        sma26  = self.sma26[0]
        sma52  = self.sma52[0]
        cross  = self.crossover[0]

        if not self.position:
            # Long entry: SMA9 crosses above SMA26, price above both, bullish cloud
            if cross > 0 and close > sma26 and close > sma52 and sma26 > sma52:
                size = int(self.broker.getcash() / close)
                if size > 0:
                    self.order = self.buy(size=size)
            # Short entry: SMA9 crosses below SMA26, price below both, bearish cloud
            elif cross < 0 and close < sma26 and close < sma52 and sma26 < sma52:
                size = int(self.broker.getcash() / close)
                if size > 0:
                    self.order = self.sell(size=size)

        elif self.position.size > 0:
            # Exit long: reverse cross OR price closes below lower of SMA26/SMA52
            if cross < 0 or close < min(sma26, sma52):
                self.order = self.close()

        elif self.position.size < 0:
            # Exit short: reverse cross OR price closes above higher of SMA26/SMA52
            if cross > 0 or close > max(sma26, sma52):
                self.order = self.close()
