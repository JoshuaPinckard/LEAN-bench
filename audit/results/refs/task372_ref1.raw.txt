import backtrader as bt


class KeltnerMeanReversionStrategy(bt.Strategy):
    params = (
        ("ema_period", 20),
        ("atr_period", 14),
        ("atr_multiplier", 2.0),
    )

    def __init__(self):
        self.middle = bt.indicators.EMA(self.data.close, period=self.p.ema_period)
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.upper = self.middle + (self.atr * self.p.atr_multiplier)
        self.lower = self.middle - (self.atr * self.p.atr_multiplier)
        self.middle_cross = bt.indicators.CrossOver(self.data.close, self.middle)

        self.entry_order = None
        self.exit_order = None
        self.stop_order = None
        self.entry_atr = None

    def next(self):
        if self.entry_order or self.exit_order:
            return

        if not self.position:
            if self.data.close[0] > self.data.open[0] and self.data.close[0] < self.lower[0]:
                self.entry_order = self.buy()
            elif self.data.close[0] < self.data.open[0] and self.data.close[0] > self.upper[0]:
                self.entry_order = self.sell()
            return

        if self.position.size > 0 and self.middle_cross[0] > 0:
            if self.stop_order:
                self.cancel(self.stop_order)
            self.exit_order = self.close()
        elif self.position.size < 0 and self.middle_cross[0] < 0:
            if self.stop_order:
                self.cancel(self.stop_order)
            self.exit_order = self.close()

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return

        if order.status == order.Completed:
            if order == self.entry_order:
                self.entry_atr = self.atr[0]
                if order.isbuy():
                    self.stop_order = self.sell(
                        exectype=bt.Order.Stop,
                        price=order.executed.price - (2.0 * self.entry_atr),
                    )
                else:
                    self.stop_order = self.buy(
                        exectype=bt.Order.Stop,
                        price=order.executed.price + (2.0 * self.entry_atr),
                    )
                self.entry_order = None
            elif order == self.exit_order:
                self.exit_order = None
                self.entry_atr = None
            elif order == self.stop_order:
                self.stop_order = None
                self.entry_atr = None

        elif order.status in (order.Canceled, order.Margin, order.Rejected):
            if order == self.entry_order:
                self.entry_order = None
            elif order == self.exit_order:
                self.exit_order = None
            elif order == self.stop_order:
                self.stop_order = None
