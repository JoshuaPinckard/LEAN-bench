```python
import backtrader as bt
import math


class SniperPullback(bt.Strategy):
    params = (
        ('bb_period', 20),
        ('bb_dev', 2),
        ('atr_period', 14),
        ('vol_period', 20),
        ('rsi_period', 14),
        ('lookback', 6),
        ('drop_min', 0.09),
        ('drop_max', 0.13),
        ('vol_mult', 1.5),
        ('tp_pct', 0.05),
        ('sl_pct', 0.03),
        ('size_pct', 0.02),
    )

    def __init__(self):
        self.bb = bt.indicators.BollingerBands(
            self.data.close, period=self.p.bb_period, devfactor=self.p.bb_dev)
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.vol_sma = bt.indicators.SMA(self.data.volume, period=self.p.vol_period)
        self.rsi = bt.indicators.RSI(self.data.close, period=self.p.rsi_period)

        self.order = None
        self.href = None
        self.href_age = None
        self.entry_price = None
        self.tp_price = None
        self.sl_price = None

    def notify_order(self, order):
        if order.status in (order.Completed, order.Canceled, order.Margin, order.Rejected):
            if order.status == order.Completed and order.isbuy():
                self.entry_price = order.executed.price
                self.tp_price = self.entry_price * (1 + self.p.tp_pct)
                self.sl_price = self.entry_price * (1 - self.p.sl_pct)
            self.order = None

    def next(self):
        # H_ref is persistent state updated on every new BB(20,2) breakout close
        # (rule 5); recency for entry eligibility is tracked via an age counter
        # so it reflects "within the last 6 bars" (rule 1) without rescanning.
        if self.data.close[0] > self.bb.top[0]:
            self.href = self.data.close[0]
            self.href_age = 0
        elif self.href_age is not None:
            self.href_age += 1

        if self.order:
            return

        if self.position:
            if self.data.close[0] >= self.tp_price or self.data.close[0] <= self.sl_price:
                self.order = self.close()
            return

        if self.href is None or self.href_age > self.p.lookback - 1:
            return

        drop = (self.href - self.data.close[0]) / self.href
        if not (self.p.drop_min <= drop <= self.p.drop_max):
            return

        if self.data.volume[0] <= self.p.vol_mult * self.vol_sma[0]:
            return

        value = self.broker.getvalue()
        size = math.floor((value * self.p.size_pct) / self.data.close[0])
        if size > 0:
            self.order = self.buy(size=size)
```
