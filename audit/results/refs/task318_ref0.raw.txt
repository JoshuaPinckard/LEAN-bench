import backtrader as bt


class ThreeBlackCrowsStrategy(bt.Strategy):
    def __init__(self):
        self.ema20 = bt.indicators.EMA(self.data.close, period=20)
        self.ema50 = bt.indicators.EMA(self.data.close, period=50)
        self.resistance = bt.indicators.Highest(self.data.high(-1), period=20)
        self.rsi = bt.indicators.RSI(self.data.close, period=14)

        self.order = None
        self.entry_order = None
        self.stop_order = None
        self.pattern_high = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order is self.stop_order:
            self.stop_order = None
            return

        if order is self.entry_order:
            self.entry_order = None
            if order.status == order.Completed:
                self.stop_order = self.buy(
                    exectype=bt.Order.Stop,
                    price=self.pattern_high * 1.02,
                    size=abs(order.executed.size),
                )
            self.order = None
            return

        if order is self.order:
            self.order = None

    def next(self):
        if self.order:
            return

        o2, c2 = self.data.open[-2], self.data.close[-2]
        o1, c1 = self.data.open[-1], self.data.close[-1]
        o0, c0 = self.data.open[0], self.data.close[0]

        bearish1 = c2 < o2
        bearish2 = c1 < o1
        bearish3 = c0 < o0
        open_in_body2 = min(o2, c2) <= o1 <= max(o2, c2)
        open_in_body3 = min(o1, c1) <= o0 <= max(o1, c1)
        progressively_lower = c2 > c1 > c0

        pattern_ok = (bearish1 and bearish2 and bearish3
                      and open_in_body2 and open_in_body3
                      and progressively_lower)

        upswing = self.ema20[0] > self.ema50[0] and self.ema20[0] > self.ema20[-1]
        resistance_ok = c0 >= 0.995 * self.resistance[0]
        entry_signal = pattern_ok and upswing and resistance_ok

        if self.position:
            if self.position.size < 0:
                exit_on_close = c0 > self.pattern_high
                rsi_cross_up = self.rsi[0] > 50 and self.rsi[-1] <= 50
                bullish_engulf = (c0 > o0 and c1 < o1
                                   and o0 <= c1 and c0 >= o1)
                if exit_on_close or rsi_cross_up or bullish_engulf:
                    if self.stop_order is not None:
                        self.cancel(self.stop_order)
                    self.order = self.close()
            else:
                if entry_signal:
                    self.order = self.close()
        else:
            if entry_signal:
                self.pattern_high = max(self.data.high[-2], self.data.high[-1], self.data.high[0])
                self.order = self.entry_order = self.sell()
