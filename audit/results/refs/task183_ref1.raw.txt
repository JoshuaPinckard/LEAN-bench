import backtrader as bt


class BreakoutStrategy(bt.Strategy):
    params = (
        ('lookback', 10),
        ('capital_fraction', 0.01),
        ('stop_loss_fraction', 0.02),
    )

    def __init__(self):
        self.previous_max_close = bt.ind.Highest(self.data.close(-1), period=self.p.lookback)
        self.previous_min_close = bt.ind.Lowest(self.data.close(-1), period=self.p.lookback)
        self.entry_order = None
        self.exit_order = None
        self.stop_order = None

    def next(self):
        if self.entry_order or self.exit_order:
            return

        if not self.position:
            if self.data.close[0] > self.previous_max_close[0]:
                size = (self.broker.getvalue() * self.p.capital_fraction) / self.data.close[0]
                self.entry_order = self.buy(size=size)
        elif self.data.close[0] < self.previous_min_close[0]:
            if self.stop_order:
                self.cancel(self.stop_order)
            self.exit_order = self.close()

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            if order is self.entry_order:
                self.entry_order = None
                if order.status == order.Completed:
                    self.stop_order = self.sell(
                        size=order.executed.size,
                        exectype=bt.Order.Stop,
                        price=order.executed.price * (1.0 - self.p.stop_loss_fraction),
                    )
            elif order is self.exit_order:
                self.exit_order = None
                if order.status == order.Completed and self.stop_order:
                    self.cancel(self.stop_order)
            elif order is self.stop_order:
                self.stop_order = None
