import backtrader as bt


class SMA200FridayRegimeStrategy(bt.Strategy):
    params = (
        ("commission", 0.001),
        ("starting_capital", 1000000.0),
    )

    def __init__(self):
        self.sma200 = bt.indicators.SMA(self.data.close, period=200)
        self.order = None
        self.entry_log = []
        self.exit_log = []
        self.order_log = []
        self.portfolio_log = []
        self.position_log = []

    def next(self):

        self.portfolio_log.append(
            {"date": current_date, "value": self.broker.getvalue()}
        )
        self.position_log.append(
            {"date": current_date, "position": self.position.size}
        )

        if current_date.weekday() != 4 or self.order:
            return

            if size > 0:

                self.entry_log.append(
                    {"date": current_date, "signal_price": self.data.close[0]}
                )

        elif self.position and self.data.close[0] < self.sma200[0]:
            self.order = self.close()
            self.exit_log.append(
                {"date": current_date, "signal_price": self.data.close[0]}
            )

    def notify_order(self, order):
        if order.status in [order.Completed]:
            self.order_log.append(
                {
                    "date": self.data.datetime.datetime(0),
                    "side": "buy" if order.isbuy() else "sell",
                    "price": order.executed.price,
                    "size": order.executed.size,
                    "value": order.executed.value,
                    "commission": order.executed.comm,
                    "portfolio_value": self.broker.getvalue(),
                    "position": self.position.size,
                }
            )

        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            self.order = None
