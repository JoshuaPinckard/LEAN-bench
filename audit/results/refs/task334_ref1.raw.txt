import backtrader as bt


class BollingerBandwidthReversalStrategy(bt.Strategy):
    def __init__(self):
        self.bb = bt.indicators.BollingerBands(
            self.data.close, period=20, devfactor=2
        )
        self.bandwidth = (self.bb.top - self.bb.bot) / self.bb.mid
        self.bw_sma = bt.indicators.SMA(self.bandwidth, period=20)
        self.bw_std = bt.indicators.StdDev(self.bandwidth, period=20)
        self.atr = bt.indicators.ATR(self.data, period=14)

        self.entry_order = None
        self.exit_order = None
        self.stop_order = None

    def next(self):
        bw_std = self.bw_std[0]
        prior_bw_z = (self.bandwidth[-1] - self.bw_sma[-1]) / self.bw_std[-1]

        if bw_std == 0 or self.bw_std[-1] == 0:
            return

        bw_z = (self.bandwidth[0] - self.bw_sma[0]) / bw_std

        if self.position:
            crosses_zero = (
                (prior_bw_z > 0 and bw_z <= 0) or
                (prior_bw_z < 0 and bw_z >= 0)
            )

            if (crosses_zero or abs(bw_z) < 0.5) and self.exit_order is None:
                if self.stop_order is not None:
                    self.cancel(self.stop_order)
                self.exit_order = self.close()
            return

        if self.entry_order is not None or self.exit_order is not None:
            return

        if (
            prior_bw_z > 2 and
            bw_z <= 2 and
            self.bandwidth[0] < self.bandwidth[-1]
        ):
            self.entry_order = self.buy()
        elif (
            prior_bw_z < -2 and
            bw_z >= -2 and
            self.bandwidth[0] > self.bandwidth[-1]
        ):
            self.entry_order = self.sell()

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            if order is self.entry_order:
                self.entry_order = None

                if order.status == order.Completed:
                    if order.isbuy():
                        stop_price = order.executed.price - 2 * self.atr[0]
                        self.stop_order = self.sell(
                            exectype=bt.Order.Stop,
                            price=stop_price
                        )
                    else:
                        stop_price = order.executed.price + 2 * self.atr[0]
                        self.stop_order = self.buy(
                            exectype=bt.Order.Stop,
                            price=stop_price
                        )

            if order is self.exit_order:
                self.exit_order = None

            if order is self.stop_order:
                self.stop_order = None
