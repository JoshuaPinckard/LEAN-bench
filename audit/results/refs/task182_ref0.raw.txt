import backtrader as bt


class LongTermCAGRStrategy(bt.Strategy):
    def __init__(self):
        self.sma200 = bt.indicators.SMA(self.data.close, period=200)
        self.crossover = bt.indicators.CrossOver(self.data.close, self.sma200)
        self.order = None

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        if self.order:
            return

        if len(self) < 757:
            return

        total_return_3y = self.data.close[0] / self.data.close[-756]
        cagr_3y = total_return_3y ** (1.0 / 3.0) - 1.0

        if not self.position:
            if cagr_3y >= 0.05 and self.crossover > 0:
                self.order = self.buy()
        else:
            if self.crossover < 0 or cagr_3y < 0:
                self.order = self.close()
