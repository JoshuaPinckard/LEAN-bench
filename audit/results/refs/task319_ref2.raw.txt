import backtrader as bt


class RsiDonchianStrategy(bt.Strategy):
    params = (
        ("rsi_period", 14),
        ("rsi_low", 30),
        ("rsi_high", 70),
        ("donchian_period", 20),
        ("stop_loss", 0.02),
    )

    def __init__(self):
        self.order = None

        self.rsi = bt.indicators.RSI(self.data.close, period=self.p.rsi_period)

        # Compute Donchian Channel using previous 20 bars (excluding current bar)
        # so close[0] > upper_band is mathematically possible.
        self.donchian_high = bt.indicators.Highest(
            self.data.high(-1), period=self.p.donchian_period
        )
        self.donchian_low = bt.indicators.Lowest(
            self.data.low(-1), period=self.p.donchian_period
        )
        self.donchian_mid = (self.donchian_high + self.donchian_low) / 2.0

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        self.order = None

    def next(self):
        if self.order:
            return

        if not self.position:
            # Long entry condition: RSI < 30 and Close > Donchian Upper Band
            if (
                self.rsi[0] < self.p.rsi_low
                and self.data.close[0] > self.donchian_high[0]
            ):
                self.order = self.buy()
            # Short entry condition: RSI > 70 and Close < Donchian Lower Band
            elif (
                self.rsi[0] > self.p.rsi_high
                and self.data.close[0] < self.donchian_low[0]
            ):
                self.order = self.sell()
        else:
            entry_price = self.position.price
            if self.position.size > 0:
                # Long position exit: 2% stop loss or Close below Donchian Middle Line
                if self.data.low[0] <= entry_price * (1.0 - self.p.stop_loss):
                    self.order = self.close()
                elif self.data.close[0] < self.donchian_mid[0]:
                    self.order = self.close()
            elif self.position.size < 0:
                # Short position exit: 2% stop loss or Close above Donchian Middle Line
                if self.data.high[0] >= entry_price * (1.0 + self.p.stop_loss):
                    self.order = self.close()
                elif self.data.close[0] > self.donchian_mid[0]:
                    self.order = self.close()
