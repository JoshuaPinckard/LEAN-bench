import backtrader as bt


class OnePercentPlaybookSimplified(bt.Strategy):
    params = dict(
        ema_period=20,
        orb_minutes=5,
        stop_loss_pct=0.01,
        take_profit_pct=0.02,
        daily_loss_limit_pct=0.005,
        max_losing_trades=2,
        eod_close_hour=15,
        eod_close_minute=55,
    )

    def __init__(self):
        self.current_date = None
        self.bar_count_in_session = 0
        self.session_open_price = None

        self.orb_high = None
        self.orb_low = None

        self.session_ema = None
        self.ema_seed_sum = 0.0
        self.ema_seed_count = 0
        self.was_below_ema = False
        self.was_above_ema = False
        self.prev_ema_for_cross = None

        self.last_session_close = None
        self.prev_close_price = None
        self.gap_type = None

        self.daily_pnl = 0.0
        self.daily_losing_trades = 0
        self.trading_halted_today = False

        self.order = None
        self.main_order = None
        self.stop_order = None
        self.limit_order = None

    def next(self):
        dt = self.data.datetime.datetime(0)
        bar_date = dt.date()

        if self.current_date != bar_date:
            if self.current_date is not None:
                self.prev_close_price = self.last_session_close
            self.current_date = bar_date
            self.bar_count_in_session = 0
            self.orb_high = None
            self.orb_low = None
            self.session_ema = None
            self.ema_seed_sum = 0.0
            self.ema_seed_count = 0
            self.was_below_ema = False
            self.was_above_ema = False
            self.prev_ema_for_cross = None
            self.session_open_price = self.data.open[0]
            if self.prev_close_price is None:
                self.gap_type = None
            elif self.session_open_price > self.prev_close_price:
                self.gap_type = 'bullish'
            elif self.session_open_price < self.prev_close_price:
                self.gap_type = 'bearish'
            else:
                self.gap_type = 'none'
            self.daily_pnl = 0.0
            self.daily_losing_trades = 0
            self.trading_halted_today = False

        self.bar_count_in_session += 1
        close_now = self.data.close[0]
        self.last_session_close = close_now

        if self.bar_count_in_session <= self.p.orb_minutes:
            self.orb_high = self.data.high[0] if self.orb_high is None else max(self.orb_high, self.data.high[0])
            self.orb_low = self.data.low[0] if self.orb_low is None else min(self.orb_low, self.data.low[0])

        ema_long_signal = False
        ema_short_signal = False
        if self.ema_seed_count < self.p.ema_period:
            self.ema_seed_sum += close_now
            self.ema_seed_count += 1
            if self.ema_seed_count == self.p.ema_period:
                self.session_ema = self.ema_seed_sum / self.p.ema_period
        else:
            alpha = 2.0 / (self.p.ema_period + 1.0)
            self.session_ema = (close_now - self.session_ema) * alpha + self.session_ema

        if self.session_ema is not None:
            if self.prev_ema_for_cross is not None:
                crossover_up = self.data.close[-1] <= self.prev_ema_for_cross and close_now > self.session_ema
                crossover_down = self.data.close[-1] >= self.prev_ema_for_cross and close_now < self.session_ema
                if crossover_up and self.was_below_ema and close_now > self.session_ema:
                    ema_long_signal = True
                if crossover_down and self.was_above_ema and close_now < self.session_ema:
                    ema_short_signal = True
            self.was_below_ema = self.was_below_ema or (close_now < self.session_ema)
            self.was_above_ema = self.was_above_ema or (close_now > self.session_ema)
            self.prev_ema_for_cross = self.session_ema

        if (dt.hour, dt.minute) >= (self.p.eod_close_hour, self.p.eod_close_minute):
            if self.position:
                if self.stop_order is not None:
                    self.cancel(self.stop_order)
                if self.limit_order is not None:
                    self.cancel(self.limit_order)
                self.close()
                self.order = None
                self.main_order = None
                self.stop_order = None
                self.limit_order = None
            elif self.main_order is not None:
                self.cancel(self.main_order)
                self.order = None
                self.main_order = None
                self.stop_order = None
                self.limit_order = None
            return

        if self.order is not None or self.position or self.trading_halted_today:
            return

        long_signal = False
        short_signal = False

        if self.bar_count_in_session > self.p.orb_minutes and self.orb_high is not None:
            if self.data.close[-1] <= self.orb_high and close_now > self.orb_high:
                long_signal = True
            elif self.data.close[-1] >= self.orb_low and close_now < self.orb_low:
                short_signal = True

        if not long_signal and not short_signal:
            if ema_long_signal:
                long_signal = True
            elif ema_short_signal:
                short_signal = True

        if not long_signal and not short_signal and self.gap_type in ('bullish', 'bearish') and self.prev_close_price is not None:
            if self.gap_type == 'bullish' and self.data.close[-1] <= self.prev_close_price and close_now > self.prev_close_price:
                long_signal = True
            elif self.gap_type == 'bearish' and self.data.close[-1] >= self.prev_close_price and close_now < self.prev_close_price:
                short_signal = True

        if not long_signal and not short_signal:
            return

        entry_est = close_now
        if long_signal:
            sl = entry_est * (1.0 - self.p.stop_loss_pct)
            tp = entry_est * (1.0 + self.p.take_profit_pct)
            orders = self.buy_bracket(price=entry_est, exectype=bt.Order.Market,
                                       stopprice=sl, limitprice=tp)
        else:
            sl = entry_est * (1.0 + self.p.stop_loss_pct)
            tp = entry_est * (1.0 - self.p.take_profit_pct)
            orders = self.sell_bracket(price=entry_est, exectype=bt.Order.Market,
                                        stopprice=sl, limitprice=tp)

        self.main_order, self.stop_order, self.limit_order = orders
        self.order = self.main_order

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return

        if order == self.main_order:
            if order.status == order.Completed:
                self.order = None
            elif order.status in (order.Canceled, order.Margin, order.Rejected):
                self.order = None
                self.main_order = None
                self.stop_order = None
                self.limit_order = None
        elif order in (self.stop_order, self.limit_order):
            if order.status == order.Completed:
                self.main_order = None
                self.stop_order = None
                self.limit_order = None

    def notify_trade(self, trade):
        if trade.isclosed:
            self.daily_pnl += trade.pnlcomm
            if trade.pnlcomm < 0:
                self.daily_losing_trades += 1
            if (self.daily_pnl <= -self.p.daily_loss_limit_pct * self.broker.startingcash
                    or self.daily_losing_trades >= self.p.max_losing_trades):
                self.trading_halted_today = True
