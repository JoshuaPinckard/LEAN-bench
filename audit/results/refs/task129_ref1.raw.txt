import backtrader as bt


class FallingChannelBreakout(bt.Strategy):
    params = (
        ("risk_pct", 0.01),
        ("max_drawdown", 0.10),
    )

    def __init__(self):
        self.bb = bt.indicators.BollingerBands(
            self.data.close, period=20, devfactor=2.0
        )
        self.volume_average = bt.indicators.SimpleMovingAverage(
            self.data.volume, period=10
        )

        self.entry_order = None
        self.stop_order = None
        self.target_order = None
        self.breakout_lower_band = None
        self.breakout_height = None
        self.peak_value = self.broker.getvalue()

    def next(self):
        current_value = self.broker.getvalue()
        if current_value > self.peak_value:
            self.peak_value = current_value

        drawdown = (self.peak_value - current_value) / self.peak_value

        if self.position or self.entry_order or self.stop_order or self.target_order:
            return

        falling_channel = (
            self.data.high[0] < self.data.high[-1]
            and self.data.high[-1] < self.data.high[-2]
            and self.data.high[-2] < self.data.high[-3]
            and self.data.low[0] < self.data.low[-1]
            and self.data.low[-1] < self.data.low[-2]
            and self.data.low[-2] < self.data.low[-3]
        )

        breakout = self.data.close[0] > self.bb.top[0]
        volume_spike = self.data.volume[0] > self.volume_average[0]

        if falling_channel and breakout and volume_spike and drawdown <= self.p.max_drawdown:
            stop_price = self.bb.bot[0]
            channel_height = self.bb.top[0] - self.bb.bot[0]
            risk_per_share = self.data.close[0] - stop_price

            if risk_per_share > 0 and channel_height > 0:
                size = (current_value * self.p.risk_pct) / risk_per_share
                self.breakout_lower_band = stop_price
                self.breakout_height = channel_height
                self.entry_order = self.buy(size=size)

    def notify_order(self, order):
        if order.status in [order.Completed]:
            if order == self.entry_order:
                entry_price = order.executed.price
                self.stop_order = self.sell(
                    size=order.executed.size,
                    exectype=bt.Order.Stop,
                    price=self.breakout_lower_band,
                )
                self.target_order = self.sell(
                    size=order.executed.size,
                    exectype=bt.Order.Limit,
                    price=entry_price + self.breakout_height,
                    oco=self.stop_order,
                )
                self.entry_order = None
            elif order == self.stop_order:
                self.stop_order = None
            elif order == self.target_order:
                self.target_order = None

        elif order.status in [
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            if order == self.entry_order:
                self.entry_order = None
            elif order == self.stop_order:
                self.stop_order = None
            elif order == self.target_order:
                self.target_order = None
