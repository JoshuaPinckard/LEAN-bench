import backtrader as bt


class FallingChannelBreakout(bt.Strategy):
    params = dict(
        bb_period=20,        # Bollinger Bands period (spec)
        bb_dev=2.0,          # Bollinger Bands standard deviation (spec)
        vol_period=10,       # average-volume lookback for spike filter (spec)
        risk_pct=0.01,       # fixed 1% equity risk per trade (spec "e.g. 1% or 2%")
        dd_threshold=0.10,   # 10% drawdown gate (spec "e.g. 10%")
        consecutive=3,       # >= 3 consecutive lower highs and lower lows (spec)
    )

    def __init__(self):
        # Channel boundaries via Bollinger Bands (20, 2): top = resistance/upper,
        # mid = midline, bot = support/lower.
        self.bb = bt.indicators.BollingerBands(
            self.data.close, period=self.p.bb_period, devfactor=self.p.bb_dev
        )
        # Average volume of the last 10 periods for the breakout volume filter.
        self.vol_sma = bt.indicators.SimpleMovingAverage(
            self.data.volume, period=self.p.vol_period
        )
        self.order = None
        self.tp = None
        self.sl = None
        self.peak_value = None

    def notify_order(self, order):
        if order.status in (order.Completed, order.Canceled,
                             order.Margin, order.Rejected):
            self.order = None

    def _falling_structure(self):
        # At least 3 consecutive lower highs AND lower lows in the bars
        # immediately preceding the breakout bar, confirming the falling channel.
        n = self.p.consecutive
        for i in range(1, n + 1):
            if not (self.data.high[-i] < self.data.high[-i - 1]):
                return False
            if not (self.data.low[-i] < self.data.low[-i - 1]):
                return False
        return True

    def next(self):
        # Track peak equity for the drawdown gate.
        value = self.broker.getvalue()
        if self.peak_value is None or value > self.peak_value:
            self.peak_value = value

        if self.order:
            return

        # In a position: manage take-profit / stop-loss.
        if self.position:
            if self.data.close[0] >= self.tp or self.data.close[0] <= self.sl:
                self.order = self.close()
            return

        # Enough history for the falling-structure lookback.
        if len(self) < self.p.consecutive + 2:
            return

        upper = self.bb.top[0]
        lower = self.bb.bot[0]

        # Long entry: close breaks above the upper channel band,
        # after an established falling structure, with a volume spike.
        breakout = self.data.close[0] > upper
        falling = self._falling_structure()
        volume_spike = self.data.volume[0] > self.vol_sma[0]

        # Drawdown gate: do not enter if current drawdown exceeds threshold.
        drawdown = 0.0
        if self.peak_value:
            drawdown = (self.peak_value - value) / self.peak_value
        dd_ok = drawdown <= self.p.dd_threshold

        if breakout and falling and volume_spike and dd_ok:
            entry = self.data.close[0]
            channel_height = upper - lower           # height of the channel at breakout
            self.tp = entry + channel_height         # TP a channel-height above entry
            self.sl = lower                          # SL at the lower band
            risk_per_share = entry - self.sl
            if risk_per_share <= 0:
                return
            risk_amount = self.p.risk_pct * value
            size = int(risk_amount / risk_per_share)  # fixed-risk position sizing
            if size <= 0:
                return
            self.order = self.buy(size=size)
