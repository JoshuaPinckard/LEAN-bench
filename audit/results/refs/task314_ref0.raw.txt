import backtrader as bt


class MACDATRStrategy(bt.Strategy):
    params = dict(
        macd_fast=12,
        macd_slow=26,
        macd_signal=9,
        atr_period=14,
        risk_fraction=0.01,
        stop_pct=0.02,
    )

    def __init__(self):
        self.macd = bt.indicators.MACD(
            self.data.close,
            period_me1=self.p.macd_fast,
            period_me2=self.p.macd_slow,
            period_signal=self.p.macd_signal,
        )
        self.crossover = bt.indicators.CrossOver(self.macd.macd, self.macd.signal)
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.order = None
        self.entry_price = None

    def notify_order(self, order):
        if order.status == order.Completed:
            if self.position:
                self.entry_price = order.executed.price
            else:
                self.entry_price = None
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        if self.order:
            return

        if self.position:
            if self.position.size > 0:
                stop_price = self.entry_price * (1 - self.p.stop_pct)
                stop_triggered = self.data.low[0] <= stop_price
                signal_exit = self.crossover[0] < 0
            else:
                stop_price = self.entry_price * (1 + self.p.stop_pct)
                stop_triggered = self.data.high[0] >= stop_price
                signal_exit = self.crossover[0] > 0

            if stop_triggered or signal_exit:
                self.order = self.close()
            return

        if self.atr[0] <= 0:
            return

        size = int((self.broker.getvalue() * self.p.risk_fraction) / self.atr[0])
        if size < 1:
            size = 1

        if self.crossover[0] > 0:
            self.order = self.buy(size=size)
        elif self.crossover[0] < 0:
            self.order = self.sell(size=size)
