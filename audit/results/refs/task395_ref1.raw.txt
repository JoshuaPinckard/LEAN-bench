import backtrader as bt


class MeanReversionStrategy(bt.Strategy):
    params = (
        ("hma_period", 9),
        ("band_period", 20),
        ("atr_period", 14),
    )

    def __init__(self):
        self.hma = bt.indicators.HullMovingAverage(
            self.data.close, period=self.p.hma_period
        )
        self.sma_hma = bt.indicators.SimpleMovingAverage(
            self.hma, period=self.p.band_period
        )
        self.std_hma = bt.indicators.StdDev(
            self.hma, period=self.p.band_period
        )
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)

        self.upper_band = self.sma_hma + 2.0 * self.std_hma
        self.lower_band = self.sma_hma - 2.0 * self.std_hma

        self.order = None
        self.stop_order = None
        self.pending_exit = False

    def notify_order(self, order):
        if order.status in (order.Completed, order.Canceled,
                            order.Margin, order.Rejected):
            if order == self.order:
                self.order = None

            if order == self.stop_order:
                self.stop_order = None

            if order.status == order.Completed and order.isbuy() and order == self.order:
                pass

        if order.status == order.Completed:
            if order == self.order:
                if order.isbuy():
                    self.stop_order = self.sell(
                        exectype=bt.Order.Stop,
                        price=order.executed.price - 2.0 * self.atr[0],
                    )
                else:
                    self.stop_order = self.buy(
                        exectype=bt.Order.Stop,
                        price=order.executed.price + 2.0 * self.atr[0],
                    )

            if self.pending_exit and order == self.stop_order:
                self.pending_exit = False
                if self.position:
                    self.order = self.close()

        if (order.status in (order.Canceled, order.Margin, order.Rejected)
                and self.pending_exit and order == self.stop_order):
            self.pending_exit = False
            if self.position:
                self.order = self.close()

    def next(self):
        if self.order or self.pending_exit:
            return

        if not self.position:
            if self.hma[-1] < self.lower_band[-1] and self.hma[0] > self.lower_band[0]:
                self.order = self.buy()
            elif self.hma[-1] > self.upper_band[-1] and self.hma[0] < self.upper_band[0]:
                self.order = self.sell()
            return

        if self.position.size > 0:
            if self.hma[-1] <= self.sma_hma[-1] and self.hma[0] > self.sma_hma[0]:
                if self.stop_order:
                    self.pending_exit = True
                    self.cancel(self.stop_order)
                else:
                    self.order = self.close()

        elif self.position.size < 0:
            if self.hma[-1] >= self.sma_hma[-1] and self.hma[0] < self.sma_hma[0]:
                if self.stop_order:
                    self.pending_exit = True
                    self.cancel(self.stop_order)
                else:
                    self.order = self.close()
