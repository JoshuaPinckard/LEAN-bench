import backtrader as bt
from collections import deque


class NonDecreasingSequenceRecordStrategy(bt.Strategy):
    params = dict(
        sma_period=10,
        lookback=100,
    )

    def __init__(self):
        self.sma10 = bt.indicators.SMA(self.data.close, period=self.p.sma_period)
        self.addminperiod(self.p.sma_period + 1)
        self.seq_len = 0
        self.seq_history = deque(maxlen=self.p.lookback)
        self.order = None

    def notify_order(self, order):
        if order.status in (order.Completed, order.Canceled, order.Margin, order.Rejected):
            self.order = None

    def next(self):
        cond = (self.data.close[0] >= self.data.close[-1]) and (self.sma10[0] >= self.sma10[-1])

        if cond:
            self.seq_len += 1
        else:
            self.seq_len = 0

        is_record = len(self.seq_history) > 0 and self.seq_len > max(self.seq_history)

        if self.order is None:
            if not self.position:
                if cond and self.seq_len >= 2 and is_record:
                    self.order = self.buy()
            elif not cond:
                self.order = self.close()

        self.seq_history.append(self.seq_len)
