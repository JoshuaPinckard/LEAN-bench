import backtrader as bt


class ClassicApproachesStrategy(bt.Strategy):
    params = (
        ('ticker', 'AAPL'),
    )

        self.entry_price = None
        self.sl_price = None
        self.tp_price = None

        self.sma50 = bt.ind.SMA(self.data.close, period=50)
        self.sma200 = bt.ind.SMA(self.data.close, period=200)

        self.ema10 = bt.ind.EMA(self.data.close, period=10)
        self.ema12 = bt.ind.EMA(self.data.close, period=12)
        self.ema20 = bt.ind.EMA(self.data.close, period=20)
        self.ema26 = bt.ind.EMA(self.data.close, period=26)
        self.ema50 = bt.ind.EMA(self.data.close, period=50)

        # Oscillators & Channels
        self.rsi14 = bt.ind.RSI(self.data.close, period=14)
        self.macd = bt.ind.MACD(self.data.close, period_me1=12, period_me2=26, period_signal=9)
        self.macd_histo = self.macd.macd - self.macd.signal

        self.bb = bt.ind.BollingerBands(self.data.close, period=20, devfactor=2.0)
        self.bb_width = self.bb.top - self.bb.bot
        self.sma_bb_width = bt.ind.SMA(self.bb_width, period=20)

        self.stoch = bt.ind.Stochastic(self.data, period=14, period_dfast=3, period_dslow=3)

        self.atr14 = bt.ind.ATR(self.data, period=14)
        self.sma_atr14 = bt.ind.SMA(self.atr14, period=14)

        self.signals = {}

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        close = self.data.close[0]
        atr = self.atr14[0]
        entry = close
        default_sl_buy = entry - 1.5 * atr
        default_tp_buy = entry + 2.0 * atr
        default_sl_sell = entry + 1.5 * atr
        default_tp_sell = entry - 2.0 * atr

        elif self.macd.macd[0] < self.macd.signal[0] and self.macd.macd[-1] >= self.macd.signal[-1]:
            signals[1] = {'signal': 'Sell', 'entry': entry, 'sl': default_sl_sell, 'tp': default_tp_sell}
        else:
            signals[1] = {'signal': 'Hold', 'entry': entry, 'sl': None, 'tp': None}

        # 2) MACD zero line
        if self.macd.macd[0] > 0 and self.macd.macd[-1] <= 0:
            signals[2] = {'signal': 'Buy', 'entry': entry, 'sl': default_sl_buy, 'tp': default_tp_buy}
        elif self.macd.macd[0] < 0 and self.macd.macd[-1] >= 0:
            signals[2] = {'signal': 'Sell', 'entry': entry, 'sl': default_sl_sell, 'tp': default_tp_sell}

        if curr_w <= min_w and close > self.bb.top[0]:
            signals[14] = {'signal': 'Buy', 'entry': entry, 'sl': default_sl_buy, 'tp': default_tp_buy}
