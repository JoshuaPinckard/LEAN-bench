import backtrader as bt
import math


class MeanReversionZScoreStrategy(bt.Strategy):
    params = (
        ("period", 252),
        ("devfactor", 2),
        ("entry_z", 1.5),
        ("stop_loss", 0.015),
        ("risk_fraction", 0.01),
        ("max_hold_bars", 63),
    )

    def __init__(self):
        self.order = None
        self.stop_order = None
        self.entry_price = None
        self.entry_bar = None

        self.bbands = bt.indicators.BollingerBands(
            self.data.close,
            period=self.p.period,
            devfactor=self.p.devfactor,
        )
        self.sma = self.bbands.mid
        self.std = (self.bbands.top - self.bbands.mid) / 2.0
        self.mean_cross = bt.indicators.CrossOver(self.data.close, self.sma)

    def next(self):
        if self.order:
            return

        std = self.std[0]
        if math.isclose(std, 0.0, abs_tol=1e-12):
            return

        zscore = (self.data.close[0] - self.sma[0]) / std

        if self.position:
            timed_exit = (
                self.entry_bar is not None
                and len(self) - self.entry_bar >= self.p.max_hold_bars
            )

            if self.mean_cross[0] != 0 or timed_exit:
                self.order = self.close(oco=self.stop_order)
            return

        capital = self.broker.getvalue()
        entry_price = self.data.close[0]
        size = (capital * self.p.risk_fraction) / (
            entry_price * self.p.stop_loss
        )

        if zscore <= -self.p.entry_z:
            self.order = self.buy(size=size)
        elif zscore >= self.p.entry_z:
            self.order = self.sell(size=size)

    def notify_order(self, order):
        if order.status != order.Completed:
            if order.status in (
                order.Canceled,
                order.Margin,
                order.Rejected,
            ):
                if order is self.order:
                    self.order = None
                if order is self.stop_order:
                    self.stop_order = None
            return

        if order is self.order:
            self.order = None

        if order is self.stop_order:
            self.stop_order = None
            self.entry_price = None
            self.entry_bar = None
            return

        if order.isbuy() and self.position.size > 0:
            self.entry_price = order.executed.price
            self.entry_bar = len(self)
            self.stop_order = self.sell(
                exectype=bt.Order.Stop,
                price=self.entry_price * (1.0 - self.p.stop_loss),
                size=abs(self.position.size),
            )
        elif order.issell() and self.position.size < 0:
            self.entry_price = order.executed.price
            self.entry_bar = len(self)
            self.stop_order = self.buy(
                exectype=bt.Order.Stop,
                price=self.entry_price * (1.0 + self.p.stop_loss),
                size=abs(self.position.size),
            )
        elif not self.position:
            self.entry_price = None
            self.entry_bar = None
            self.stop_order = None
