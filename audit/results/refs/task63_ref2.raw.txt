import backtrader as bt


class ROCSMAStrategy(bt.Strategy):
    """
    ROC-Based Technical Trading Strategy using SMA Crossover TTR.
    Classifies BUY signals as True/False Positives based on daily closing returns
    and computes ROC performance metrics (Confusion Matrix, TPR, FPR).
    """

    params = (
        ('fast_period', 20),
        ('slow_period', 50),
        ('size', 1.0),
    )

    def __init__(self):
        self.fast_sma = bt.ind.SMA(self.data.close, period=self.p.fast_period)
        self.slow_sma = bt.ind.SMA(self.data.close, period=self.p.slow_period)
        self.crossover = bt.ind.CrossOver(self.fast_sma, self.slow_sma)

        self.order = None

        # ROC metrics counters
        self.tp = 0  # True Positives (TRUE BUY)
        self.fp = 0  # False Positives (FALSE BUY)
        self.tn = 0  # True Negatives
        self.fn = 0  # False Negatives

        self.tpr = 0.0
        self.fpr = 0.0
        self.confusion_matrix = {}

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            if order.isbuy():
                # Immediately queue market close order to exit intraday on the same bar
                self.order = self.close(exectype=bt.Order.Close)
            elif order.issell():
                self.order = None
        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        is_buy_signal = self.crossover[0] > 0
        is_positive_return = self.data.close[0] > self.data.open[0]

        # Update ROC confusion matrix metrics for the current day
        if is_buy_signal:
            if is_positive_return:
                self.tp += 1
            else:
                self.fp += 1
        else:
            if is_positive_return:
                self.fn += 1
            else:
                self.tn += 1

        # Guard pending orders
        if self.order:
            return

        # Execute intraday trade on BUY signal when flat
        if not self.position and is_buy_signal:
            self.order = self.buy(size=self.p.size)

    def stop(self):
        total_positives = self.tp + self.fn
        total_negatives = self.fp + self.tn
        self.tpr = self.tp / total_positives if total_positives > 0 else 0.0
        self.fpr = self.fp / total_negatives if total_negatives > 0 else 0.0
        self.confusion_matrix = {
            'TP': self.tp,
            'FP': self.fp,
            'TN': self.tn,
            'FN': self.fn,
        }
