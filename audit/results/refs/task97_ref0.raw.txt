import backtrader as bt
import math


class EMAForecastStrategy(bt.Strategy):
    params = dict(
        ema_period=21,        # EMA period (spec: 21)
        threshold=0.005,      # 0.5% minimum deviation for an entry signal
        stop_loss=0.02,       # 2% stop-loss distance from entry
        take_profit=0.03,     # 3% take-profit distance from entry
        size_pct=0.95,        # ~100% of available capital per trade (buffered)
    )

    def __init__(self):
        # EMA(21). k = 2 / (period + 1) as stated in the spec.
        self.ema = bt.indicators.ExponentialMovingAverage(
            self.data.close, period=self.p.ema_period)
        self.k = 2.0 / (self.p.ema_period + 1)

        self.entry_order = None   # pending market entry order
        self.close_order = None   # pending market crossover-exit order
        self.stop_order = None    # resting protective stop-loss order
        self.take_order = None    # resting protective take-profit order
        self.entry_price = None

    def _has_pending(self):
        return self.entry_order is not None or self.close_order is not None

    def _place_protective_orders(self):
        # Called once an entry fills; attaches stop-loss + take-profit (OCO).
        size = abs(self.position.size)
        if size == 0:
            return
        if self.position.size > 0:  # long
            stop_price = self.entry_price * (1.0 - self.p.stop_loss)   # 2% below
            take_price = self.entry_price * (1.0 + self.p.take_profit)  # 3% above
            self.stop_order = self.sell(
                size=size, exectype=bt.Order.Stop, price=stop_price)
            self.take_order = self.sell(
                size=size, exectype=bt.Order.Limit, price=take_price,
                oco=self.stop_order)
        else:  # short
            stop_price = self.entry_price * (1.0 + self.p.stop_loss)   # 2% above
            take_price = self.entry_price * (1.0 - self.p.take_profit)  # 3% below
            self.stop_order = self.buy(
                size=size, exectype=bt.Order.Stop, price=stop_price)
            self.take_order = self.buy(
                size=size, exectype=bt.Order.Limit, price=take_price,
                oco=self.stop_order)

    def _cancel_protective(self):
        if self.stop_order is not None:
            self.cancel(self.stop_order)
        if self.take_order is not None:
            self.cancel(self.take_order)

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return

        if order.status == order.Completed:
            if order is self.entry_order:
                self.entry_price = order.executed.price
                self.entry_order = None
                self._place_protective_orders()
            elif order is self.stop_order or order is self.take_order:
                # A protective order filled; its OCO sibling is auto-cancelled.
                self.stop_order = None
                self.take_order = None
            elif order is self.close_order:
                self.close_order = None
            return

        if order.status in (order.Canceled, order.Margin, order.Rejected):
            if order is self.entry_order:
                self.entry_order = None
            elif order is self.stop_order:
                self.stop_order = None
            elif order is self.take_order:
                self.take_order = None
            elif order is self.close_order:
                self.close_order = None

    def next(self):
        close = self.data.close[0]
        ema = self.ema[0]

        # Forecast EMA for the NEXT day, built from the current bar:
        # EMA_forecast(N+1) = Close(N)*k + EMA(N)*(1-k)
        forecast_next = close * self.k + ema * (1.0 - self.k)

        # --- Manage an open position -------------------------------------
        # Stop-loss / take-profit are resting orders; here we only evaluate
        # the crossover exit against the forecast that targeted the current
        # bar (i.e. the forecast computed on the previous bar).
        if self.position:
            if self.close_order is None and not math.isnan(self.ema[-2]):
                forecast_today = (self.data.close[-1] * self.k
                                  + self.ema[-1] * (1.0 - self.k))
                forecast_prev = (self.data.close[-2] * self.k
                                 + self.ema[-2] * (1.0 - self.k))
                diff_now = close - forecast_today
                diff_prev = self.data.close[-1] - forecast_prev

                if self.position.size > 0:
                    # Long exit: close crosses forecast from above to below.
                    if diff_prev > 0 and diff_now < 0:
                        self._cancel_protective()
                        self.close_order = self.close()
                else:
                    # Short exit: close crosses forecast from below to above.
                    if diff_prev < 0 and diff_now > 0:
                        self._cancel_protective()
                        self.close_order = self.close()
            return

        # --- Flat: look for an entry -------------------------------------
        if self._has_pending():
            return

        # Signals require a >= 0.5% deviation of Close(N) from the forecast.
        long_signal = (forecast_next - close) / close >= self.p.threshold
        short_signal = (close - forecast_next) / close >= self.p.threshold

        if not (long_signal or short_signal):
            return

        cash = self.broker.get_cash()
        size = int(cash * self.p.size_pct / close)
        if size <= 0:
            return

        if long_signal:
            # Close(N) significantly below forecast -> expect upward reversion.
            self.entry_order = self.buy(size=size)
        elif short_signal:
            # Close(N) significantly above forecast -> expect downward reversion.
            self.entry_order = self.sell(size=size)
