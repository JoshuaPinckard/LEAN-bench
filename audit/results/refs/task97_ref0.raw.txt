import backtrader as bt


class EMAForecastEntryStrategy(bt.Strategy):
    params = dict(
        ema_period=21,
        trend_period=50,
        entry_threshold=0.005,
        stop_loss_long=0.02,
        take_profit_long=0.03,
        stop_loss_short=0.02,
        take_profit_short=0.03,
    )

    def __init__(self):
        self.ema = bt.indicators.EMA(self.data.close, period=self.p.ema_period)
        self.ema_trend = bt.indicators.EMA(self.data.close, period=self.p.trend_period)
        self.k = 2.0 / (self.p.ema_period + 1)

        self.entry_order = None
        self.exit_order = None
        self.stop_order = None
        self.limit_order = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status == order.Completed:
            if order == self.entry_order:
                self.entry_order = None
                entry_price = order.executed.price
                size = abs(self.position.size)
                if order.isbuy():
                    stop_price = entry_price * (1 - self.p.stop_loss_long)
                    limit_price = entry_price * (1 + self.p.take_profit_long)
                    self.stop_order = self.sell(exectype=bt.Order.Stop, price=stop_price, size=size)
                    self.limit_order = self.sell(exectype=bt.Order.Limit, price=limit_price, size=size, oco=self.stop_order)
                else:
                    stop_price = entry_price * (1 + self.p.stop_loss_short)
                    limit_price = entry_price * (1 - self.p.take_profit_short)
                    self.stop_order = self.buy(exectype=bt.Order.Stop, price=stop_price, size=size)
                    self.limit_order = self.buy(exectype=bt.Order.Limit, price=limit_price, size=size, oco=self.stop_order)
            elif order == self.exit_order:
                self.exit_order = None
            elif order == self.stop_order:
                self.stop_order = None
            elif order == self.limit_order:
                self.limit_order = None
        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            if order == self.entry_order:
                self.entry_order = None
            elif order == self.exit_order:
                self.exit_order = None
            elif order == self.stop_order:
                self.stop_order = None
            elif order == self.limit_order:
                self.limit_order = None

    def next(self):
        if self.entry_order or self.exit_order:
            return

        close0 = self.data.close[0]
        close_1 = self.data.close[-1]
        ema0 = self.ema[0]
        ema_1 = self.ema[-1]

        forecast_today = close_1 * self.k + ema_1 * (1 - self.k)
        forecast_tomorrow = close0 * self.k + ema0 * (1 - self.k)

        if self.position:
            if self.position.size > 0:
                if close_1 > forecast_today and close0 < forecast_today:
                    if self.stop_order:
                        self.cancel(self.stop_order)
                    if self.limit_order:
                        self.cancel(self.limit_order)
                    self.exit_order = self.close()
            else:
                if close_1 < forecast_today and close0 > forecast_today:
                    if self.stop_order:
                        self.cancel(self.stop_order)
                    if self.limit_order:
                        self.cancel(self.limit_order)
                    self.exit_order = self.close()
            return

        threshold = self.p.entry_threshold * close0
        diff = forecast_tomorrow - close0

        if diff >= threshold and self.ema_trend[0] > self.ema_trend[-1]:
            self.entry_order = self.order_target_percent(target=1.0)
        elif -diff >= threshold and self.ema_trend[0] < self.ema_trend[-1]:
            self.entry_order = self.order_target_percent(target=-1.0)
