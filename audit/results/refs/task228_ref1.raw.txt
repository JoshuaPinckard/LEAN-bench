import backtrader as bt


class SwingStrategy(bt.Strategy):
    def __init__(self):
        self.ema12 = bt.indicators.EMA(self.data.close, period=12)
        self.ema26 = bt.indicators.EMA(self.data.close, period=26)
        self.rsi = bt.indicators.RSI(self.data.close, period=14)
        self.atr = bt.indicators.ATR(self.data, period=14)

        self.order = None
        self.entry_price = None
        self.stop_price = None
        self.breakeven_moved = False
        self.equity_peak = self.broker.getvalue()

        self.long_pullback = False
        self.long_trigger_high = None
        self.short_pullback = False
        self.short_trigger_low = None

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            if order.status == order.Completed and order.isbuy():
                self.entry_price = order.executed.price
                self.stop_price = self.entry_price * (1.0 - 0.015)
                self.breakeven_moved = False
            elif order.status == order.Completed and order.issell() and self.position.size < 0:
                self.entry_price = order.executed.price
                self.stop_price = self.entry_price * (1.0 + 0.015)
                self.breakeven_moved = False

            if order == self.order:
                self.order = None

    def next(self):
        equity = self.broker.getvalue()
        if equity > self.equity_peak:
            self.equity_peak = equity

        if self.order:
            return

        close = self.data.close[0]
        open_price = self.data.open[0]
        high = self.data.high[0]
        low = self.data.low[0]

        if self.position:
            if self.position.size > 0:
                if not self.breakeven_moved and close >= self.entry_price * 1.01:
                    self.stop_price = self.entry_price
                    self.breakeven_moved = True

                if self.rsi[0] > 70 or close <= self.stop_price:
                    self.order = self.close()

            else:
                if not self.breakeven_moved and close <= self.entry_price * 0.99:
                    self.stop_price = self.entry_price
                    self.breakeven_moved = True

                if self.rsi[0] < 30 or close >= self.stop_price:
                    self.order = self.close()

            return

        prev_rsi = self.rsi[-1]

        if prev_rsi >= 40 and self.rsi[0] < 40 and self.rsi[0] >= 30:
            self.long_pullback = True
            self.long_trigger_high = None
        elif self.long_pullback and self.rsi[0] < 30:
            self.long_pullback = False
            self.long_trigger_high = None

        if self.long_pullback and self.long_trigger_high is None:
            if prev_rsi <= 35 and self.rsi[0] > 35:
                self.long_trigger_high = high

        if prev_rsi <= 60 and self.rsi[0] > 60 and self.rsi[0] <= 70:
            self.short_pullback = True
            self.short_trigger_low = None
        elif self.short_pullback and self.rsi[0] > 70:
            self.short_pullback = False
            self.short_trigger_low = None

        if self.short_pullback and self.short_trigger_low is None:
            if prev_rsi >= 65 and self.rsi[0] < 65:
                self.short_trigger_low = low

        daily_range_ok = (high - low) >= (0.005 * open_price)
        prev_close = self.data.close[-1]
        true_range = max(high - low, abs(high - prev_close), abs(low - prev_close))
        news_filter_ok = (
            true_range <= (2.0 * self.atr[0])
            and abs(open_price - prev_close) / prev_close <= 0.01
        )
        drawdown_ok = (self.equity_peak - equity) / self.equity_peak <= 0.05

        if not (daily_range_ok and news_filter_ok and drawdown_ok):
            return

        risk_amount = equity * 0.01
        stop_distance = close * 0.015
        size = risk_amount / stop_distance

        if (
            self.ema12[0] > self.ema26[0]
            and self.long_trigger_high is not None
            and close > self.long_trigger_high
        ):
            self.order = self.buy(size=size)
            self.long_pullback = False
            self.long_trigger_high = None
            self.short_pullback = False
            self.short_trigger_low = None

        elif (
            self.ema12[0] < self.ema26[0]
            and self.short_trigger_low is not None
            and close < self.short_trigger_low
        ):
            self.order = self.sell(size=size)
            self.long_pullback = False
            self.long_trigger_high = None
            self.short_pullback = False
            self.short_trigger_low = None
