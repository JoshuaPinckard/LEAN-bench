import backtrader as bt


class AbsoluteMomentumStrategy(bt.Strategy):
    def __init__(self):
        self.momentum = bt.indicators.MomentumOscillator(
            self.data.close,
            period=252,
        )
        self.order = None

    def notify_order(self, order):
        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            self.order = None

    def next(self):
        if self.order:
            return

        if not self.position and self.momentum[0] > 100:
            size = (self.broker.getcash() * 0.95) / self.data.close[0]
            self.order = self.buy(size=size)

        elif self.position and self.momentum[0] < 100:
            self.order = self.close()
