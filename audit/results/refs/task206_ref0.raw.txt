import backtrader as bt


class VolAdaptiveMeanReversionStrategy(bt.Strategy):
    params = (
        ('bb_period', 30),
        ('bb_devfactor', 2.0),
        ('atr_period', 14),
        ('atr_ema_period', 50),
        ('base_threshold', 2.0),
        ('vr_low', 0.7),
        ('vr_high', 1.6),
        ('rsi_period', 14),
        ('rsi_long_max', 45),
        ('rsi_short_min', 55),
        ('vol_sma_period', 20),
        ('use_filters', True),
        ('stop_atr_mult', 1.5),
        ('time_stop_bars', 7),
        ('risk_pct', 0.015),
    )

    def __init__(self):
        self.bb = bt.indicators.BollingerBands(
            self.data.close, period=self.p.bb_period, devfactor=self.p.bb_devfactor)
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.atr_ema = bt.indicators.EMA(self.atr, period=self.p.atr_ema_period)
        self.rsi = bt.indicators.RSI(self.data.close, period=self.p.rsi_period)
        self.vol_sma = bt.indicators.SMA(self.data.volume, period=self.p.vol_sma_period)

        self.order = None
        self.entry_bar = None
        self.entry_price = None
        self.stop_price = None
        self.prev_z = None

    def notify_order(self, order):
        if order.status == order.Completed:
            if self.position.size != 0:
                self.entry_price = order.executed.price
                self.entry_bar = len(self)
                if self.position.size > 0:
                    self.stop_price = self.entry_price - self.p.stop_atr_mult * self.atr[0]
                else:
                    self.stop_price = self.entry_price + self.p.stop_atr_mult * self.atr[0]
            else:
                self.entry_price = None
                self.entry_bar = None
                self.stop_price = None

        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        if self.order:
            return

        mid = self.bb.mid[0]
        upper = self.bb.top[0]
        std = (upper - mid) / 2.0
        if std == 0:
            self.prev_z = None
            return
        z = (self.data.close[0] - mid) / std

        atr_val = self.atr[0]
        atr_ema_val = self.atr_ema[0]
        vr = atr_val / atr_ema_val if atr_ema_val != 0 else 1.0
        clipped_vr = max(self.p.vr_low, min(self.p.vr_high, vr))
        T = self.p.base_threshold * clipped_vr

        if self.position:
            exit_signal = False
            if self.prev_z is not None and (
                    (self.prev_z < 0 and z >= 0) or (self.prev_z > 0 and z <= 0)):
                exit_signal = True
            if self.stop_price is not None:
                if self.position.size > 0 and self.data.close[0] <= self.stop_price:
                    exit_signal = True
                elif self.position.size < 0 and self.data.close[0] >= self.stop_price:
                    exit_signal = True
            if self.entry_bar is not None and (len(self) - self.entry_bar) >= self.p.time_stop_bars:
                exit_signal = True

            if exit_signal:
                self.order = self.close()
        else:
            long_signal = z < -T
            short_signal = z > T
            if self.p.use_filters:
                long_signal = long_signal and (self.rsi[0] < self.p.rsi_long_max) and \
                    (self.data.volume[0] > self.vol_sma[0])
                short_signal = short_signal and (self.rsi[0] > self.p.rsi_short_min) and \
                    (self.data.volume[0] > self.vol_sma[0])

            per_share_risk = self.p.stop_atr_mult * atr_val
            size = int((self.p.risk_pct * self.broker.getvalue()) / per_share_risk) if per_share_risk > 0 else 0

            if long_signal and size > 0:
                self.order = self.buy(size=size)
            elif short_signal and size > 0:
                self.order = self.sell(size=size)

        self.prev_z = z
