import backtrader as bt


class MomentumSMAStrategy(bt.Strategy):
    params = (
        ('momentum_period', 10),
        ('momentum_threshold', 5.0),
        ('sma_period', 50),
        ('atr_period', 14),
        ('stop_loss_method', 'max'),  # 'max', 'pct', or 'atr'
        ('allow_reverse', True),
    )

    def __init__(self):
        self.sma = bt.indicators.SimpleMovingAverage(
            self.data.close, period=self.p.sma_period
        )
        self.atr = bt.indicators.AverageTrueRange(
            self.data, period=self.p.atr_period
        )

        self.order = None
        self.entry_price = None
        self.entry_atr = None
        self.skip_next_entry = False

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            if self.position.size != 0:
                self.entry_price = self.position.price
                self.entry_atr = self.atr[0]
            else:
                self.entry_price = None
                self.entry_atr = None
        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            pass

        self.order = None

    def next(self):
        if self.order:
            return

        mom = (
            (self.data.close[0] / self.data.close[-self.p.momentum_period] - 1.0)
            * 100.0
        )

        if self.position.size > 0:
            entry_p = (
                self.entry_price
                if self.entry_price is not None
                else self.position.price
            )
            entry_a = (
                self.entry_atr if self.entry_atr is not None else self.atr[0]
            )

            pct_stop = entry_p * 0.98
            atr_stop = entry_p - 1.0 * entry_a

            if self.p.stop_loss_method == 'pct':
                stop_price = pct_stop
            elif self.p.stop_loss_method == 'atr':
                stop_price = atr_stop
            else:
                stop_price = max(pct_stop, atr_stop)

            if self.data.low[0] <= stop_price:
                self.order = self.close()
                return

            if mom < -self.p.momentum_threshold:
                self.order = self.close()
                if not self.p.allow_reverse:
                    self.skip_next_entry = True
                return

        elif self.position.size < 0:
            entry_p = (
                self.entry_price
                if self.entry_price is not None
                else self.position.price
            )
            entry_a = (
                self.entry_atr if self.entry_atr is not None else self.atr[0]
            )

            pct_stop = entry_p * 1.02
            atr_stop = entry_p + 1.0 * entry_a

            if self.p.stop_loss_method == 'pct':
                stop_price = pct_stop
            elif self.p.stop_loss_method == 'atr':
                stop_price = atr_stop
            else:
                stop_price = min(pct_stop, atr_stop)

            if self.data.high[0] >= stop_price:
                self.order = self.close()
                return

            if mom > self.p.momentum_threshold:
                self.order = self.close()
                if not self.p.allow_reverse:
                    self.skip_next_entry = True
                return

        else:
            if self.skip_next_entry:
                self.skip_next_entry = False
                return

            if (
                mom > self.p.momentum_threshold
                and self.data.close[0] > self.sma[0]
            ):
                self.order = self.buy()
            elif (
                mom < -self.p.momentum_threshold
                and self.data.close[0] < self.sma[0]
            ):
                self.order = self.sell()
