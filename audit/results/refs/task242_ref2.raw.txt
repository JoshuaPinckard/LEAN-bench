import backtrader as bt


class SpecMeanReversion(bt.Strategy):
    params = (
        ("roc_period", 21),
        ("bb_period", 252),
        ("bb_dev", 2.0),
        ("atr_period", 14),
        ("risk_pct", 0.01),
        ("stop_atr_mult", 2.0),
        ("hold_days", 5),
        ("min_history", 273),
    )

    def __init__(self):
        # 21-day return ROC = (close - close[-21]) / close[-21]
        self.roc = bt.indicators.ROC(self.data.close, period=self.p.roc_period)
        # Bollinger Bands on ROC21
        self.bb = bt.indicators.BollingerBands(
            self.roc, period=self.p.bb_period, devfactor=self.p.bb_dev
        )
        # ATR(14) for position sizing
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)

        self.order = None
        self.bars_held = 0

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            self.bars_held = 0

        self.order = None

    def next(self):
        # Trade only when history is >= 273 days
        if len(self) < self.p.min_history:
            return

        # Pending order guard
        if self.order:
            return

        # Manage open position (strictly close after 5 trading days)
        if self.position:
            self.bars_held += 1
            if self.bars_held >= self.p.hold_days:
                self.order = self.close()
            return

        # Entry condition evaluated only on Mondays
        if self.data.datetime.date(0).weekday() == 0:
            roc_val = self.roc[0]
            bot_val = self.bb.lines.bot[0]
            top_val = self.bb.lines.top[0]
            atr_val = self.atr[0]

            if atr_val <= 0:
                return

            risk_amount = self.broker.getvalue() * self.p.risk_pct
            stop_distance = self.p.stop_atr_mult * atr_val
            size = int(risk_amount / stop_distance)

            if size <= 0:
                return

            if roc_val < bot_val:
                self.order = self.buy(size=size)
            elif roc_val > top_val:
                self.order = self.sell(size=size)
