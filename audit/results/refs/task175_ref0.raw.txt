```python
import backtrader as bt


class CSPMeanReversionProxyStrategy(bt.Strategy):
    params = (
        ('ema_period', 20),
        ('rsi_period', 14),
        ('atr_period', 14),
        ('rsi_entry_threshold', 40),
        ('atr_close_ratio_max', 0.04),
        ('anti_change_pct', 0.01),
        ('profit_atr_mult', 0.5),
        ('stop_atr_mult', 1.0),
        ('stop_entry_pct', 0.02),
        ('time_stop_bars', 5),
        ('risk_pct', 0.01),
    )

    def __init__(self):
        self.ema = bt.indicators.EMA(self.data.close, period=self.p.ema_period)
        self.rsi = bt.indicators.RSI(self.data.close, period=self.p.rsi_period)
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)

        self.order = None
        self.entry_price = None
        self.entry_atr = None
        self.target_price = None
        self.stop_price = None
        self.bars_in_trade = 0

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status == order.Completed:
            if order.isbuy():
                self.entry_price = order.executed.price
                # Stop/target levels are fixed off the ATR observed at signal time,
                # not recomputed bar-by-bar while the trade is open.
                self.stop_price = self.entry_price - min(
                    self.p.stop_atr_mult * self.entry_atr,
                    self.p.stop_entry_pct * self.entry_price,
                )
                self.target_price = self.entry_price + self.p.profit_atr_mult * self.entry_atr
                self.bars_in_trade = 0
            else:
                self.entry_price = None
                self.entry_atr = None
                self.target_price = None
                self.stop_price = None
                self.bars_in_trade = 0
        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            if order.isbuy():
                self.entry_atr = None

        self.order = None

    def next(self):
        if self.order:
            return

        if self.position:
            self.bars_in_trade += 1
            close = self.data.close[0]

            if close >= self.target_price or close >= self.ema[0]:
                self.order = self.close()
            elif close <= self.stop_price:
                self.order = self.close()
            elif self.bars_in_trade >= self.p.time_stop_bars:
                self.order = self.close()
            return

        close = self.data.close[0]
        prev_close = self.data.close[-1]

        cond_below_ema = close < self.ema[0]
        cond_rsi = self.rsi[0] < self.p.rsi_entry_threshold
        cond_vol = (self.atr[0] / close) < self.p.atr_close_ratio_max
        cond_anti_change = close >= prev_close * (1.0 - self.p.anti_change_pct)

        if cond_below_ema and cond_rsi and cond_vol and cond_anti_change:
            atr_now = self.atr[0]
            stop_distance = min(
                self.p.stop_atr_mult * atr_now,
                self.p.stop_entry_pct * close,
            )
            if stop_distance <= 0:
                return

            capital = self.broker.getvalue()
            risk_amount = self.p.risk_pct * capital
            size = int(risk_amount / stop_distance)

            if size < 1:
                return

            self.entry_atr = atr_now
            self.order = self.buy(size=size)
```
