import math

import backtrader as bt


class OptionsProxyStrategy(bt.Strategy):
    params = (
        ('strategy_type', 1),  # 1 = Strategy 1 (CSP-proxy), 2 = Strategy 2 (CC-proxy)
        ('atr_period', 14),
        ('bb_period', 20),
        ('bb_devfactor', 2.0),
        ('sma_period', 50),
        ('ema_period', 20),
        ('rsi_period', 14),
        ('vol_sma_period', 20),
        ('k_min', 0.01),
        ('k_max', 0.05),
        ('expiration_bars', 20),
        ('rsi_entry_threshold', 35.0),
        ('size_pct_s1', 0.20),
        ('size_pct_s2', 1.00),
    )

    def __init__(self):
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.bb = bt.indicators.BollingerBands(
            self.data, period=self.p.bb_period, devfactor=self.p.bb_devfactor)
        self.sma50 = bt.indicators.SMA(self.data, period=self.p.sma_period)
        self.ema20 = bt.indicators.EMA(self.data, period=self.p.ema_period)
        self.rsi = bt.indicators.RSI(self.data, period=self.p.rsi_period)
        self.vol_sma = bt.indicators.SMA(self.data.volume, period=self.p.vol_sma_period)
        self.crossover = bt.indicators.CrossOver(self.ema20, self.sma50)

        self.order = None
        self.entry_price = None
        self.entry_bar = None
        self.k_entry = None
        self.atr_entry = None
        self.protection_active = False
        self.pending_k = None
        self.pending_atr = None

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return
        if order.status == order.Completed and order.isbuy():
            self.entry_price = order.executed.price
            self.entry_bar = len(self)
            self.protection_active = False
            if self.p.strategy_type == 1:
                self.k_entry = self.pending_k
                self.atr_entry = self.pending_atr
            else:
                self.k_entry = self._k()
                self.atr_entry = self.atr[0]
        if order.status in (order.Completed, order.Canceled, order.Margin, order.Rejected):
            self.order = None

    def _k(self):
        k_raw = self.atr[0] / self.data.close[0]
        return max(self.p.k_min, min(self.p.k_max, k_raw))

    def _liquidity_ok(self):
        return self.data.volume[0] > self.vol_sma[0]

    def next(self):
        if self.order is not None:
            return
        if self.p.strategy_type == 1:
            self._next_strategy1()
        else:
            self._next_strategy2()

    def _next_strategy1(self):
        if not self.position:
            if (self.data.close[0] < self.bb.lines.bot[0]
                    and self.rsi[0] < self.p.rsi_entry_threshold
                    and self._liquidity_ok()):
                k = self._k()
                limit_price = self.data.close[0] * (1 - k)
                alloc = self.broker.getvalue() * self.p.size_pct_s1
                size = math.floor(alloc / limit_price)
                if size > 0:
                    self.pending_k = k
                    self.pending_atr = self.atr[0]
                    self.order = self.buy(exectype=bt.Order.Limit, price=limit_price, size=size)
            return

        bars_held = len(self) - self.entry_bar
        tp_level = max(self.entry_price + 0.5 * self.k_entry * self.entry_price,
                        self.bb.lines.mid[0])
        sl_level = self.entry_price - 1.5 * self.atr_entry

        if self.data.close[0] >= tp_level:
            self.order = self.close()
        elif self.data.close[0] <= sl_level:
            self.order = self.close()
        elif bars_held >= self.p.expiration_bars:
            self.order = self.close()

    def _next_strategy2(self):
        if not self.position:
            if (self.crossover[0] > 0
                    and self.data.close[0] > self.ema20[0]
                    and self._liquidity_ok()):
                alloc = self.broker.getvalue() * self.p.size_pct_s2
                size = math.floor(alloc / self.data.close[0])
                if size > 0:
                    self.order = self.buy(size=size)
            return

        bars_held = len(self) - self.entry_bar
        protect_trigger = self.entry_price * (1 + 0.5 * self.k_entry)
        tp_pct_level = self.entry_price * (1 + self.k_entry)

        if not self.protection_active and self.data.close[0] >= protect_trigger:
            self.protection_active = True

        take_profit = (self.data.close[0] >= self.bb.lines.top[0]
                        or self.data.close[0] >= tp_pct_level)

        if self.protection_active:
            stop_trigger = self.data.close[0] < self.ema20[0]
        else:
            stop_trigger = (self.data.close[0] < (self.ema20[0] - 1.0 * self.atr[0])
                             or self.crossover[0] < 0)

        if take_profit:
            self.order = self.close()
        elif stop_trigger:
            self.order = self.close()
        elif bars_held >= self.p.expiration_bars:
            self.order = self.close()
