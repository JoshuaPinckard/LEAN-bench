import backtrader as bt


class TrendFollowingPseudoForecast(bt.Strategy):
    def __init__(self):
        self.order = None

        self.macd = bt.indicators.MACDHisto(
            self.data.close,
            period_me1=12,
            period_me2=26,
            period_signal=9,
        )
        self.sma20 = bt.indicators.SMA(self.data.close, period=20)
        self.sma50 = bt.indicators.SMA(self.data.close, period=50)
        self.ema10 = bt.indicators.EMA(self.data.close, period=10)
        self.rsi = bt.indicators.RSI(self.data.close, period=14)
        self.stochastic = bt.indicators.Stochastic(
            self.data,
            period=14,
            period_dfast=3,
            period_dslow=3,
        )
        self.bollinger = bt.indicators.BollingerBands(
            self.data.close,
            period=20,
            devfactor=2,
        )
        self.stochastic_cross = bt.indicators.CrossOver(
            self.stochastic.percK,
            self.stochastic.percD,
        )

    def notify_order(self, order):
        if order.status in (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            self.order = None

    def next(self):
        if self.order:
            return

        score = 0

        if self.macd.histo[0] > 0:
            score += 1
        elif self.macd.histo[0] < 0:
            score -= 1

        if (
            self.sma20[0] > self.sma50[0]
            and self.ema10[0] > self.ema10[-1]
            and self.sma20[0] > self.sma20[-1]
        ):
            score += 1
        elif (
            self.sma20[0] < self.sma50[0]
            and self.ema10[0] < self.ema10[-1]
            and self.sma20[0] < self.sma20[-1]
        ):
            score -= 1

        if self.rsi[0] > 55:
            score += 1
        elif self.rsi[0] < 45:
            score -= 1

        band_width = self.bollinger.top[0] - self.bollinger.bot[0]
        previous_band_width = self.bollinger.top[-1] - self.bollinger.bot[-1]
        if self.data.close[0] > self.bollinger.mid[0] and band_width > previous_band_width:
            score += 1
        elif self.data.close[0] < self.bollinger.mid[0] and band_width > previous_band_width:
            score -= 1

        if self.stochastic_cross[0] > 0 and self.stochastic.percK[0] > 20:
            score += 1
        elif self.stochastic_cross[0] < 0 and self.stochastic.percK[0] < 80:
            score -= 1

        if score >= 2 and self.position.size <= 0:
            self.order = self.order_target_percent(target=1.0)
        elif score <= -2 and self.position.size >= 0:
            self.order = self.order_target_percent(target=-1.0)
