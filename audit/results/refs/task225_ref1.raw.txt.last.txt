import backtrader as bt
import math


class BreakoutMomentumStrategy(bt.Strategy):
    params = (
        ("sma_trend_period", 200),
        ("ema_fast_period", 20),
        ("ema_slow_period", 50),
        ("bb_period", 20),
        ("bb_devfactor", 2),
        ("bb_width_sma_period", 100),
        ("volume_fast_period", 20),
        ("volume_slow_period", 50),
        ("rsi_period", 14),
        ("atr_period", 14),
    )

    def __init__(self):
        self.order = None
        self.entry_price = None
        self.entry_atr = None
        self.initial_stop = None
        self.max_close_since_entry = None
        self.pending_entry_atr = None

        self.sma200 = bt.indicators.SMA(
            self.data.close, period=self.p.sma_trend_period
        )
        self.ema20 = bt.indicators.EMA(
            self.data.close, period=self.p.ema_fast_period
        )
        self.ema50 = bt.indicators.EMA(
            self.data.close, period=self.p.ema_slow_period
        )

        self.bb = bt.indicators.BollingerBands(
            self.data.close,
            period=self.p.bb_period,
            devfactor=self.p.bb_devfactor,
        )
        self.bb_width = (self.bb.lines.top - self.bb.lines.bot) / self.bb.lines.mid
        self.bb_width_sma100 = bt.indicators.SMA(
            self.bb_width, period=self.p.bb_width_sma_period
        )

        self.volume_sma20 = bt.indicators.SMA(
            self.data.volume, period=self.p.volume_fast_period
        )
        self.volume_sma50 = bt.indicators.SMA(
            self.data.volume, period=self.p.volume_slow_period
        )
        self.rsi = bt.indicators.RSI(
            self.data.close, period=self.p.rsi_period
        )
        self.atr = bt.indicators.ATR(
            self.data, period=self.p.atr_period
        )
        self.breakout = bt.indicators.CrossOver(
            self.data.close, self.bb.lines.top
        )

    def notify_order(self, order):
        if order.status == order.Completed:
            if order.isbuy():
                self.entry_price = order.executed.price
                self.entry_atr = self.pending_entry_atr
                self.initial_stop = self.entry_price - 1.5 * self.entry_atr
                self.max_close_since_entry = self.data.close[0]
                self.pending_entry_atr = None
            elif order.issell():
                self.entry_price = None
                self.entry_atr = None
                self.initial_stop = None
                self.max_close_since_entry = None

            self.order = None

        elif order.status in (
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            self.pending_entry_atr = None
            self.order = None

    def next(self):
        if self.order:
            return

        close = self.data.close[0]
        atr = self.atr[0]

        if self.position:
            self.max_close_since_entry = max(self.max_close_since_entry, close)

            take_profit = close >= self.entry_price + 2.0 * atr
            weakening = close < self.ema20[0]
            trailing_stop = close <= self.max_close_since_entry - 2.0 * atr
            stop_loss = close <= self.initial_stop

            if take_profit or weakening or trailing_stop or stop_loss:
                self.order = self.close()
            return

        uptrend = close > self.sma200[0] and self.ema20[0] > self.ema50[0]
        squeeze = self.bb_width[0] < 0.8 * self.bb_width_sma100[0]
        volume_spike = self.data.volume[0] >= 5.0 * self.volume_sma20[0]
        breakout = self.breakout[0] > 0
        momentum = self.rsi[0] > 55
        liquidity = self.volume_sma20[0] >= self.volume_sma50[0] * 0.7

        if uptrend and squeeze and volume_spike and breakout and momentum and liquidity:
            risk_per_share = 1.5 * atr
            size = math.floor(0.02 * self.broker.getvalue() / risk_per_share)

            if size > 0:
                self.pending_entry_atr = atr
                self.order = self.buy(size=size)
