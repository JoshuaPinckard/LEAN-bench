import backtrader as bt


class EMARibbonStrategy(bt.Strategy):
    params = (
        ('stop_loss_pct', 0.02),
        ('take_profit_pct', 0.04),
        ('capital_risk_pct', 0.02),
    )

    def __init__(self):
        self.ema8 = bt.indicators.EMA(self.data.close, period=8)
        self.ema13 = bt.indicators.EMA(self.data.close, period=13)
        self.ema21 = bt.indicators.EMA(self.data.close, period=21)
        self.ema34 = bt.indicators.EMA(self.data.close, period=34)
        self.ema55 = bt.indicators.EMA(self.data.close, period=55)

        self.order = None
        self.stop_order = None
        self.take_profit_order = None
        self.exit_requested = False

    def _bullish_ordered(self, ago=0):
        return (
            self.ema8[ago] > self.ema13[ago] > self.ema21[ago] >
            self.ema34[ago] > self.ema55[ago]
        )

    def _bearish_ordered(self, ago=0):
        return (
            self.ema8[ago] < self.ema13[ago] < self.ema21[ago] <
            self.ema34[ago] < self.ema55[ago]
        )

    def _cancel_protective_orders(self):
        if self.stop_order is not None:
            self.cancel(self.stop_order)
        if self.take_profit_order is not None:
            self.cancel(self.take_profit_order)

    def next(self):
        if self.order is not None:
            return

        if not self.position:
            if self._bullish_ordered(0) and self._bearish_ordered(-1):
                capital = self.broker.getvalue()
                size = (capital * self.p.capital_risk_pct) / self.data.close[0]
                if size > 0:
                    self.order = self.buy(size=size)
            return

        if self._bearish_ordered(0) and self._bullish_ordered(-1):
            self.exit_requested = True
            self._cancel_protective_orders()
            if self.stop_order is None and self.take_profit_order is None:
                self.order = self.close()

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return

        if order.status == order.Completed:
            if order.isbuy():
                self.order = None
                entry_price = order.executed.price
                stop_price = entry_price * (1.0 - self.p.stop_loss_pct)
                take_profit_price = entry_price * (1.0 + self.p.take_profit_pct)

                self.stop_order = self.sell(
                    size=order.executed.size,
                    exectype=bt.Order.Stop,
                    price=stop_price,
                )
                self.take_profit_order = self.sell(
                    size=order.executed.size,
                    exectype=bt.Order.Limit,
                    price=take_profit_price,
                    oco=self.stop_order,
                )
            else:
                self.order = None
                self.stop_order = None
                self.take_profit_order = None
                self.exit_requested = False

        elif order.status in (order.Canceled, order.Margin, order.Rejected):
            if order == self.stop_order:
                self.stop_order = None
            elif order == self.take_profit_order:
                self.take_profit_order = None
            elif order == self.order:
                self.order = None

            if self.exit_requested and not self.position:
                self.exit_requested = False

            if (
                self.exit_requested
                and self.position
                and self.stop_order is None
                and self.take_profit_order is None
                and self.order is None
            ):
                self.order = self.close()
