import backtrader as bt


class MACDHistogramStrategy(bt.Strategy):
    params = (
        ('fastperiod', 12),
        ('slowperiod', 26),
        ('signalperiod', 9),
        ('stop_loss_pct', 0.02),
    )

    def __init__(self):
        self.macd = bt.indicators.MACDHisto(
            self.data.close,
            period_me1=self.p.fastperiod,
            period_me2=self.p.slowperiod,
            period_signal=self.p.signalperiod,
        )
        self.order = None
        self.stop_order = None

    def next(self):
        if self.order:
            return

        histogram = self.macd.histo[0]
        previous_histogram = self.macd.histo[-1]

        if not self.position:
            if previous_histogram < 0 and histogram > 0:
                self.order = self.buy()
            elif previous_histogram > 0 and histogram < 0:
                self.order = self.sell()

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            if order == self.order:
                self.order = None

            if order.status == order.Completed and order.isbuy() and order == self.order:
                self.stop_order = self.sell(
                    exectype=bt.Order.Stop,
                    price=order.executed.price * (1.0 - self.p.stop_loss_pct),
                )
            elif order.status == order.Completed and order.issell() and order == self.order:
                self.stop_order = self.buy(
                    exectype=bt.Order.Stop,
                    price=order.executed.price * (1.0 + self.p.stop_loss_pct),
                )

            if order == self.stop_order:
                self.stop_order = None
