import backtrader as bt


class GapFadeStrategy(bt.Strategy):
    params = (
        ('X', 1.0),
    )

    def __init__(self):
        self.order = None
        self.stop_order = None
        self.stop_price = None
        self.entered_today = False

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            if order == self.order:
                self.order = None
                if self.position.size != 0 and self.stop_price is not None:
                    if self.position.size > 0:
                        self.stop_order = self.sell(
                            exectype=bt.Order.Stop,
                            price=self.stop_price
                        )
                    elif self.position.size < 0:
                        self.stop_order = self.buy(
                            exectype=bt.Order.Stop,
                            price=self.stop_price
                        )
            elif order == self.stop_order:
                self.stop_order = None

        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            if order == self.order:
                self.order = None
            elif order == self.stop_order:
                self.stop_order = None

    def next(self):
        if len(self) <= 1:
            return

        is_new_day = self.data.datetime.date(0) != self.data.datetime.date(-1)

        if is_new_day:
            self.entered_today = False
            today_open = self.data.open[0]
            prior_close = self.data.close[-1]

            if prior_close != 0:
                gap_pct = (today_open - prior_close) / prior_close * 100.0

                if gap_pct >= self.p.X:
                    if self.position.size > 0:
                        self.close()
                    if not self.entered_today and self.order is None:
                        self.stop_price = today_open * 1.02
                        self.order = self.sell()
                        self.entered_today = True

                elif gap_pct <= -self.p.X:
                    if self.position.size < 0:
                        self.close()
                    if not self.entered_today and self.order is None:
                        self.stop_price = today_open * 0.98
                        self.order = self.buy()
                        self.entered_today = True

        t = self.data.datetime.time(0)
        if t.hour > 15 or (t.hour == 15 and t.minute >= 45):
            if self.position.size != 0:
                if self.stop_order:
                    self.cancel(self.stop_order)
                    self.stop_order = None
                if self.order is None:
                    self.order = self.close()
