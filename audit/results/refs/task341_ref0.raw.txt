import backtrader as bt


class SmaTrendStrategy(bt.Strategy):
    params = dict(
        fast_period=5,
        mid_period=20,
        slow_period=50,
        stop_pct=0.02,
    )

    def __init__(self):
        self.sma_fast = bt.indicators.SMA(self.data.close, period=self.p.fast_period)
        self.sma_mid = bt.indicators.SMA(self.data.close, period=self.p.mid_period)
        self.sma_slow = bt.indicators.SMA(self.data.close, period=self.p.slow_period)
        self.crossover = bt.indicators.CrossOver(self.sma_fast, self.sma_mid)
        self.order = None
        self.entry_price = None

    def notify_order(self, order):
        if order.status == order.Completed:
            self.entry_price = order.executed.price if self.position else None
        if order.status in (order.Completed, order.Canceled, order.Margin, order.Rejected):
            self.order = None

    def next(self):
        if self.order:
            return

        price = self.data.close[0]

        if not self.position:
            if self.sma_fast[0] > self.sma_mid[0] > self.sma_slow[0]:
                cash = self.broker.getcash()
                size = int(cash / price)
                if size > 0:
                    self.order = self.buy(size=size)
            elif self.sma_fast[0] < self.sma_mid[0] < self.sma_slow[0]:
                cash = self.broker.getcash()
                size = int(cash / price)
                if size > 0:
                    self.order = self.sell(size=size)
        elif self.position.size > 0:
            exit_signal = self.crossover[0] < 0
            stop_hit = self.entry_price is not None and price <= self.entry_price * (1 - self.p.stop_pct)
            if exit_signal or stop_hit:
                self.order = self.close()
        elif self.position.size < 0:
            exit_signal = self.crossover[0] > 0
            stop_hit = self.entry_price is not None and price >= self.entry_price * (1 + self.p.stop_pct)
            if exit_signal or stop_hit:
                self.order = self.close()
