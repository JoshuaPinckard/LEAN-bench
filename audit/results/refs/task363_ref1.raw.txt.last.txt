import backtrader as bt


class MeanReversionStrategy(bt.Strategy):
    def __init__(self):
        self.stoch = bt.indicators.Stochastic(
            self.data,
            period=14,
            period_dfast=3,
            period_dslow=3,
        )
        self.k = self.stoch.percK
        self.k_sma20 = bt.indicators.SimpleMovingAverage(self.k, period=20)
        self.k_std20 = bt.indicators.StandardDeviation(self.k, period=20)

        self.entry_order = None
        self.exit_order = None
        self.stop_order = None

    def _slope(self, values):
        return sum((i - 9.5) * value for i, value in enumerate(values)) / 665.0

    def next(self):
        current_std = self.k_std20[0]
        if current_std == 0:
            return

        z_score = (self.k[0] - self.k_sma20[0]) / current_std
        current_slope = self._slope(self.k.get(size=20))
        previous_slope = self._slope(self.k.get(ago=-1, size=20))

        if self.position.size > 0:
            if self.k[0] > 50 or z_score >= 0:
                if self.exit_order is None:
                    if self.stop_order is not None:
                        self.cancel(self.stop_order)
                    self.exit_order = self.close()
            return

        if self.position.size < 0:
            if self.k[0] < 50 or z_score <= 0:
                if self.exit_order is None:
                    if self.stop_order is not None:
                        self.cancel(self.stop_order)
                    self.exit_order = self.close()
            return

        if (self.entry_order is not None or self.exit_order is not None or
                self.stop_order is not None):
            return

        if z_score < -2.0 and current_slope > 0 and previous_slope <= 0:
            self.entry_order = self.buy()
        elif z_score > 2.0 and current_slope < 0 and previous_slope >= 0:
            self.entry_order = self.sell()

    def notify_order(self, order):
        terminal_statuses = (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        )

        if order is self.entry_order:
            if order.status == order.Completed:
                if self.position.size > 0:
                    self.stop_order = self.sell(
                        exectype=bt.Order.Stop,
                        price=order.executed.price * 0.98,
                        size=abs(self.position.size),
                    )
                elif self.position.size < 0:
                    self.stop_order = self.buy(
                        exectype=bt.Order.Stop,
                        price=order.executed.price * 1.02,
                        size=abs(self.position.size),
                    )

            if order.status in terminal_statuses:
                self.entry_order = None

        elif order is self.exit_order:
            if order.status in terminal_statuses:
                self.exit_order = None

        elif order is self.stop_order:
            if order.status in terminal_statuses:
                self.stop_order = None
