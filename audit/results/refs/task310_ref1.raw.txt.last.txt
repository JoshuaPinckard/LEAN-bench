import backtrader as bt
import math


class AAPLIntradayTrendStrategy(bt.Strategy):
    class HurstExponent(bt.Indicator):
        lines = ("hurst",)
        params = (("lookback", 100),)

        def __init__(self):
            self.addminperiod(self.p.lookback)

        def next(self):
            closes = list(self.data.get(size=self.p.lookback))
            returns = [
                math.log(closes[i] / closes[i - 1])
                for i in range(1, len(closes))
                if closes[i - 1] > 0 and closes[i] > 0
            ]

            if len(returns) < 2:
                self.lines.hurst[0] = 0.0
                return

            mean_return = sum(returns) / len(returns)
            cumulative = []
            total = 0.0
            for value in returns:
                total += value - mean_return
                cumulative.append(total)

            value_range = max(cumulative) - min(cumulative)
            variance = sum((value - mean_return) ** 2 for value in returns)
            std_dev = math.sqrt(variance / (len(returns) - 1))

            if value_range <= 0 or std_dev <= 0:
                self.lines.hurst[0] = 0.0
            else:
                self.lines.hurst[0] = math.log(value_range / std_dev) / math.log(len(returns))

    params = (
        ("hurst_lookback", 100),
        ("hurst_threshold", 0.55),
        ("risk_fraction", 0.01),
        ("atr_multiplier", 2.0),
    )

    def __init__(self):
        self.hurst = self.HurstExponent(self.data.close, lookback=self.p.hurst_lookback)
        self.psar = bt.indicators.ParabolicSAR(self.data, af=0.02, afmax=0.2)
        self.atr = bt.indicators.ATR(self.data, period=14)

        self.pending_order = None
        self.pending_action = None
        self.stop_order = None
        self.entry_price = None
        self.reverse_direction = None

    def _stop_price(self, direction, price):
        if direction > 0:
            return max(float(self.psar[0]), price - self.p.atr_multiplier * float(self.atr[0]))
        return min(float(self.psar[0]), price + self.p.atr_multiplier * float(self.atr[0]))

    def _size_for_trade(self, direction):
        price = float(self.data.close[0])
        stop_price = self._stop_price(direction, price)
        risk_per_share = abs(price - stop_price)

        if risk_per_share <= 0:
            return 0

        risk_budget = self.broker.getvalue() * self.p.risk_fraction
        risk_size = int(risk_budget / risk_per_share)
        cash_cap = int(self.broker.getvalue() / (price * 1.001))
        return max(0, min(risk_size, cash_cap))

    def _submit_entry(self, direction):
        if self.position or self.pending_order is not None:
            return

        size = self._size_for_trade(direction)
        if size <= 0:
            return

        self.pending_action = "entry"
        if direction > 0:
            self.pending_order = self.buy(size=size)
        else:
            self.pending_order = self.sell(size=size)

    def _update_protective_stop(self):
        if not self.position or self.pending_order is not None:
            return

        direction = 1 if self.position.size > 0 else -1
        stop_price = self._stop_price(direction, float(self.entry_price))

        if self.stop_order is not None:
            self.cancel(self.stop_order)
            self.stop_order = None

        if direction > 0:
            self.stop_order = self.sell(
                size=abs(self.position.size),
                exectype=bt.Order.Stop,
                price=stop_price,
            )
        else:
            self.stop_order = self.buy(
                size=abs(self.position.size),
                exectype=bt.Order.Stop,
                price=stop_price,
            )

    def _close_and_reverse(self, direction):
        if self.pending_order is not None:
            return

        if self.stop_order is not None:
            self.cancel(self.stop_order)
            self.stop_order = None

        self.reverse_direction = direction
        self.pending_action = "close"
        self.pending_order = self.close()

    def notify_order(self, order):
        if order is self.stop_order:
            if order.status in (order.Completed, order.Canceled, order.Margin, order.Rejected):
                if order.status == order.Completed:
                    self.entry_price = None
                    self.reverse_direction = None
                self.stop_order = None
            return

        if order is not self.pending_order:
            return

        action = self.pending_action

        if order.status == order.Completed:
            self.pending_order = None
            self.pending_action = None

            if action == "entry":
                self.entry_price = order.executed.price
            elif action == "close":
                self.entry_price = None
                direction = self.reverse_direction
                self.reverse_direction = None
                self._submit_entry(direction)

        elif order.status in (order.Canceled, order.Margin, order.Rejected):
            self.pending_order = None
            self.pending_action = None
            self.reverse_direction = None

    def next(self):
        close_crossed_above = (
            self.data.close[-1] <= self.psar[-1]
            and self.data.close[0] > self.psar[0]
        )
        close_crossed_below = (
            self.data.close[-1] >= self.psar[-1]
            and self.data.close[0] < self.psar[0]
        )

        if self.position:
            if self.position.size > 0 and close_crossed_below:
                self._close_and_reverse(-1)
                return

            if self.position.size < 0 and close_crossed_above:
                self._close_and_reverse(1)
                return

            self._update_protective_stop()
            return

        if self.pending_order is None and self.stop_order is None and self.hurst[0] > self.p.hurst_threshold:
            if close_crossed_above:
                self._submit_entry(1)
            elif close_crossed_below:
                self._submit_entry(-1)
