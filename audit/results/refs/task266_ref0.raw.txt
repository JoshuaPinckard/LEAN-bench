import backtrader as bt


class ORBFibonacciStrategy(bt.Strategy):
    params = (
        ('ema_fast_period', 9),
        ('ema_slow_period', 21),
        ('fib_38', 0.382),
        ('fib_62', 0.618),
        ('risk_pct', 0.01),
    )

    def __init__(self):
        self.ema_fast = bt.indicators.EMA(self.data.close, period=self.p.ema_fast_period)
        self.ema_slow = bt.indicators.EMA(self.data.close, period=self.p.ema_slow_period)

        self.order = None
        self.current_date = None
        self.or_set = False
        self.orh = None
        self.orl = None
        self.long_fib38 = None
        self.long_fib62 = None
        self.short_fib38 = None
        self.short_fib62 = None
        self.breakout_up = False
        self.breakout_down = False

        self.cum_pv = 0.0
        self.cum_vol = 0.0
        self.vwap = None

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        bar_date = self.data.datetime.date(0)

        if self.current_date is None or bar_date != self.current_date:
            if self.position:
                self.close()
            self.current_date = bar_date
            self.or_set = False
            self.orh = None
            self.orl = None
            self.breakout_up = False
            self.breakout_down = False
            self.cum_pv = 0.0
            self.cum_vol = 0.0
            self.vwap = None

        typical_price = (self.data.high[0] + self.data.low[0] + self.data.close[0]) / 3.0
        self.cum_pv += typical_price * self.data.volume[0]
        self.cum_vol += self.data.volume[0]
        if self.cum_vol > 0:
            self.vwap = self.cum_pv / self.cum_vol

        if not self.or_set:
            self.orh = self.data.high[0]
            self.orl = self.data.low[0]
            range_ = self.orh - self.orl
            self.long_fib38 = self.orh - self.p.fib_38 * range_
            self.long_fib62 = self.orh - self.p.fib_62 * range_
            self.short_fib38 = self.orl + self.p.fib_38 * range_
            self.short_fib62 = self.orl + self.p.fib_62 * range_
            self.or_set = True
            return

        if self.order:
            return

        close = self.data.close[0]

        if self.position:
            if self.position.size > 0:
                if close <= self.orl:
                    self.order = self.close()
            elif self.position.size < 0:
                if close >= self.orh:
                    self.order = self.close()
            return

        if close > self.orh:
            self.breakout_up = True
        if close < self.orl:
            self.breakout_down = True

        ema_fast_up = self.ema_fast[0] > self.ema_fast[-1]
        ema_slow_up = self.ema_slow[0] > self.ema_slow[-1]
        ema_fast_down = self.ema_fast[0] < self.ema_fast[-1]
        ema_slow_down = self.ema_slow[0] < self.ema_slow[-1]

        if self.breakout_up:
            pulled_back = close <= self.long_fib38 or close <= self.long_fib62
            if pulled_back and ema_fast_up and ema_slow_up and self.vwap is not None and close > self.vwap:
                stop_price = self.orl
                risk_per_share = close - stop_price
                if risk_per_share > 0:
                    size = int((self.broker.getvalue() * self.p.risk_pct) / risk_per_share)
                    if size > 0:
                        self.order = self.buy(size=size)

        if self.breakout_down:
            pulled_back = close >= self.short_fib38 or close >= self.short_fib62
            if pulled_back and ema_fast_down and ema_slow_down and self.vwap is not None and close < self.vwap:
                stop_price = self.orh
                risk_per_share = stop_price - close
                if risk_per_share > 0:
                    size = int((self.broker.getvalue() * self.p.risk_pct) / risk_per_share)
                    if size > 0:
                        self.order = self.sell(size=size)
