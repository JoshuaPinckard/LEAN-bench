import backtrader as bt


class QuantValueStrategy(bt.Strategy):
    params = (
        ("sma_fast", 50),
        ("sma_slow", 200),
        ("bb_period", 20),
        ("bb_dev", 2.0),
        ("rsi_period", 14),
        ("vol_fast", 20),
        ("vol_slow", 50),
        ("atr_period", 14),
    )

    def __init__(self):
        self.order = None

        self.sma50 = bt.indicators.SMA(self.data.close, period=self.p.sma_fast)
        self.sma200 = bt.indicators.SMA(
            self.data.close, period=self.p.sma_slow
        )
        self.crossover = bt.indicators.CrossOver(self.sma50, self.sma200)

        self.bb = bt.indicators.BollingerBands(
            self.data.close, period=self.p.bb_period, devfactor=self.p.bb_dev
        )

        self.rsi = bt.indicators.RSI(self.data.close, period=self.p.rsi_period)

        self.vol_sma20 = bt.indicators.SMA(
            self.data.volume, period=self.p.vol_fast
        )
        self.vol_sma50 = bt.indicators.SMA(
            self.data.volume, period=self.p.vol_slow
        )

        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)

    def notify_order(self, order):
        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            self.order = None

    def next(self):
        if self.order:
            return

        if not self.position:
            # 1. Trend: Close > SMA(200)
            cond_trend = self.data.close[0] > self.sma200[0]

            # 2. Golden Cross within last 5 days
            cond_gc = any(self.crossover[-i] > 0 for i in range(5))

            # 3. Value proxy: Close below BB mid and PercentB <= 0.2
            bb_top = self.bb.lines.top[0]
            bb_bot = self.bb.lines.bot[0]
            bb_mid = self.bb.lines.mid[0]
            bb_width = bb_top - bb_bot
            percent_b = (
                (self.data.close[0] - bb_bot) / bb_width if bb_width > 0 else 1.0
            )
            cond_val = (self.data.close[0] < bb_mid) and (percent_b <= 0.2)

            # 4. RSI(14) in 40–60 range
            cond_rsi = 40.0 <= self.rsi[0] <= 60.0

            # 5. Liquidity: SMA(Volume,20) > 100000 and > SMA(Volume,50)
            cond_liq = (self.vol_sma20[0] > 100000) and (
                self.vol_sma20[0] > self.vol_sma50[0]
            )

            # 6. Stability: ATR(14)/Close <= 0.04
            cond_stab = (self.atr[0] / self.data.close[0]) <= 0.04

            if (
                cond_trend
                and cond_gc
                and cond_val
                and cond_rsi
                and cond_liq
                and cond_stab
            ):
                capital = self.broker.getvalue()
                close_price = self.data.close[0]
                stop_dist = max(0.10 * close_price, 2.0 * self.atr[0])

                if stop_dist > 0 and close_price > 0:
                    risk_shares = (0.01 * capital) / stop_dist
                    max_cash_shares = (0.25 * capital) / close_price
                    size = int(min(risk_shares, max_cash_shares))

                    if size > 0:
                        self.order = self.buy(size=size)
        else:
            entry_price = self.position.price
            close_price = self.data.close[0]

            # Exit 1: Take-profit +20%
            cond_tp = close_price >= entry_price * 1.20

            # Exit 2: Stop-loss -10%
            cond_sl = close_price <= entry_price * 0.90

            # Exit 3: Death cross
            cond_dc = self.crossover[0] < 0

            # Exit 4: Overheating (RSI >= 70 or touch of upper BB)
            cond_overheat = (self.rsi[0] >= 70.0) or (
                self.data.high[0] >= self.bb.lines.top[0]
            )

            # Exit 5: Risk spike
            cond_risk = (self.atr[0] / close_price) > 0.06

            if cond_tp or cond_sl or cond_dc or cond_overheat or cond_risk:
                self.order = self.close()
