import backtrader as bt


class CPSReference(bt.Strategy):
    def __init__(self):
        self.ema = bt.indicators.EMA(self.data.close, period=20)
        self.rsi = bt.indicators.RSI(self.data.close, period=14)
        self.atr = bt.indicators.ATR(self.data, period=14)

        self.order = None
        self.entry_price = None
        self.entry_bar = None

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            if order.status == order.Completed and order.isbuy():
                self.entry_price = order.executed.price
                self.entry_bar = len(self)

            if order.status == order.Completed and order.issell():
                self.entry_price = None
                self.entry_bar = None

            if order == self.order:
                self.order = None

    def next(self):
        if self.order:
            return

        close = self.data.close[0]
        atr = self.atr[0]

        if self.position:
            profit_exit = (
                close >= self.entry_price + 0.5 * atr
                or close >= self.ema[0]
            )
            stop_exit = (
                close <= self.entry_price - min(atr, 0.02 * self.entry_price)
            )
            time_exit = len(self) - self.entry_bar >= 5

            if profit_exit or stop_exit or time_exit:
                self.order = self.close()
            return

        entry_condition = (
            close < self.ema[0]
            and self.rsi[0] < 40
            and atr / close < 0.04
            and close >= self.data.close[-1] * 0.99
        )

        if entry_condition:
            risk_per_share = min(atr, 0.02 * close)
            size = int((0.01 * self.broker.getvalue()) / risk_per_share)

            if size > 0:
                self.order = self.buy(size=size)
