import backtrader as bt


class TrailingStopLossShortStrategy(bt.Strategy):
    params = (("trail_amount", 0.19),)

    def __init__(self):
        self.order = None
        self.trailing_stop = None
        self.maximum_price = None
        self.executed_transactions = []

    def next(self):
        current_time = self.data.datetime.datetime(0).time()

        if self.order:
            return

        if not self.position:
            if current_time.hour == 13 and current_time.minute == 10:
                self.order = self.sell(size=10)
            return

        if self.position.size < 0:
            self.maximum_price = max(self.maximum_price, self.data.high[0])
            self.trailing_stop = self.maximum_price - self.p.trail_amount

            if self.data.low[0] <= self.trailing_stop:
                self.order = self.buy(size=10)

    def notify_order(self, order):
        if order.status in [order.Completed]:
            self.executed_transactions.append({
                "time": self.data.datetime.datetime(0),
                "volume": abs(order.executed.size),
                "price": order.executed.price,
            })

            if order.isbuy() and self.position.size == 0:
                self.trailing_stop = None
                self.maximum_price = None

            if order.issell() and self.position.size < 0:
                self.maximum_price = self.data.high[0]
                self.trailing_stop = self.maximum_price - self.p.trail_amount

            self.order = None

        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            self.order = None
