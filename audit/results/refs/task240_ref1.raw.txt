import backtrader as bt
import math


class MeanReversionVolTargetStrategy(bt.Strategy):
    params = (
        ('capital', 10000.0),
        ('risk_factor', 100.0),
    )

    def __init__(self):
        self.bb2 = bt.indicators.BollingerBands(
            self.data.close, period=100, devfactor=2
        )
        self.bb3 = bt.indicators.BollingerBands(
            self.data.close, period=100, devfactor=3
        )
        self.atr = bt.indicators.ATR(self.data, period=20)
        self.mid_cross = bt.indicators.CrossOver(self.data.close, self.bb2.mid)

        self.order = None

    def notify_order(self, order):
        if order.status in (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            self.order = None

    def next(self):
        if self.order:
            return

        close = self.data.close[0]
        mid = self.bb2.mid[0]
        atr = self.atr[0]

        if self.position:
            if self.position.size > 0:
                if (self.data.low[0] <= self.bb3.bot[0] or
                        self.mid_cross[0] > 0 or
                        abs(close - mid) <= 0.25 * (self.bb2.top[0] - mid)):
                    self.order = self.close()
            else:
                if (self.data.high[0] >= self.bb3.top[0] or
                        self.mid_cross[0] < 0 or
                        abs(close - mid) <= 0.25 * (self.bb2.top[0] - mid)):
                    self.order = self.close()
            return

        if atr <= 0:
            return

        shares = min(
            self.broker.getcash() / close,
            math.floor(self.p.capital / (atr * self.p.risk_factor)),
        )

        if shares <= 0:
            return

        if close < self.bb2.bot[0]:
            self.order = self.buy(size=shares)
        elif close > self.bb2.top[0]:
            self.order = self.sell(size=shares)
