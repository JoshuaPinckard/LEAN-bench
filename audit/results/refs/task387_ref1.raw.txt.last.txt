import backtrader as bt


class Strategy(bt.Strategy):
    def __init__(self):
        self.dmi = bt.indicators.DirectionalMovementIndex(self.data, period=14)
        self.stoch = bt.indicators.StochasticFull(
            self.data,
            period=14,
            period_dfast=3,
            period_dslow=3,
        )

        self.entry_order = None
        self.exit_order = None
        self.stop_order = None
        self.entry_price = None

    def next(self):
        adx = self.dmi.adx[0]
        prev_adx = self.dmi.adx[-1]
        plus_di = self.dmi.plusDI[0]
        minus_di = self.dmi.minusDI[0]
        k = self.stoch.percK[0]
        prev_k = self.stoch.percK[-1]

        if self.position:
            if prev_adx >= 25 and adx < 25 and self.exit_order is None:
                if self.stop_order is not None:
                    self.cancel(self.stop_order)
                self.exit_order = self.close()
            return

        if self.entry_order is not None or self.exit_order is not None or self.stop_order is not None:
            return

        if adx > 25 and plus_di > minus_di and prev_k > 20 and k < 20:
            self.entry_order = self.buy()
        elif adx > 25 and minus_di > plus_di and prev_k < 80 and k > 80:
            self.entry_order = self.sell()

    def notify_order(self, order):
        terminal_statuses = (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        )

        if order is self.entry_order:
            if order.status == order.Completed:
                self.entry_price = self.position.price
                if self.position.size > 0:
                    self.stop_order = self.sell(
                        exectype=bt.Order.Stop,
                        price=self.entry_price * (1 - 0.02),
                    )
                else:
                    self.stop_order = self.buy(
                        exectype=bt.Order.Stop,
                        price=self.entry_price * (1 + 0.02),
                    )

            if order.status in terminal_statuses:
                self.entry_order = None

        elif order is self.exit_order:
            if order.status == order.Completed:
                self.entry_price = None

            if order.status in terminal_statuses:
                self.exit_order = None

        elif order is self.stop_order:
            if order.status == order.Completed:
                self.entry_price = None

            if order.status in terminal_statuses:
                self.stop_order = None
