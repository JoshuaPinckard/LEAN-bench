import backtrader as bt


class BBLowerTouchVolSellStrategy(bt.Strategy):
    params = dict(
        bb_period=20,
        bb_devfactor=2.0,
        ema_period=50,
        sma_period=200,
        rsi_period=14,
        rsi_threshold=35.0,
        bb_width_low=0.02,
        bb_width_high=0.08,
        atr_period=14,
        atr_stop_mult=1.5,
        risk_pct=0.01,
        tp_r=0.5,
        soft_stop_r=0.8,
        hard_stop_r=1.0,
        vol_exit_growth=0.20,
        timebox_bars=60,
        dd_limit=0.10,
    )

    def __init__(self):
        self.bb = bt.indicators.BollingerBands(
            self.data, period=self.p.bb_period, devfactor=self.p.bb_devfactor)
        self.ema50 = bt.indicators.EMA(self.data, period=self.p.ema_period)
        self.sma200 = bt.indicators.SMA(self.data, period=self.p.sma_period)
        self.rsi = bt.indicators.RSI(self.data, period=self.p.rsi_period)
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)

        self.order = None
        self.entry_price = None
        self.entry_bar = None
        self.stop_price = None
        self.soft_stop_price = None
        self.take_profit_price = None
        self.bb_width_entry = None
        self._pending_atr = None
        self._pending_bb_width = None
        self.equity_peak = self.broker.getvalue()

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return
        if order.status == order.Completed and order.isbuy():
            self.entry_price = order.executed.price
            self.entry_bar = len(self)
            stop_distance = self.p.atr_stop_mult * self._pending_atr
            self.stop_price = self.entry_price - self.p.hard_stop_r * stop_distance
            self.soft_stop_price = self.entry_price - self.p.soft_stop_r * stop_distance
            self.take_profit_price = self.entry_price + self.p.tp_r * stop_distance
            self.bb_width_entry = self._pending_bb_width
        self.order = None

    def notify_trade(self, trade):
        if trade.isclosed:
            self.entry_price = None
            self.entry_bar = None
            self.stop_price = None
            self.soft_stop_price = None
            self.take_profit_price = None
            self.bb_width_entry = None

    def next(self):
        value = self.broker.getvalue()
        if value > self.equity_peak:
            self.equity_peak = value
        drawdown = (self.equity_peak - value) / self.equity_peak if self.equity_peak > 0 else 0.0

        if self.order is not None:
            return

        if self.position:
            if self.stop_price is not None and self.data.low[0] <= self.stop_price:
                self.order = self.close()
                return
            if self.soft_stop_price is not None and self.data.close[0] <= self.soft_stop_price:
                self.order = self.close()
                return
            if self.take_profit_price is not None and self.data.high[0] >= self.take_profit_price:
                self.order = self.close()
                return
            if self.bb_width_entry is not None:
                bb_width_now = (self.bb.lines.top[0] - self.bb.lines.bot[0]) / self.bb.lines.mid[0]
                if bb_width_now >= (1.0 + self.p.vol_exit_growth) * self.bb_width_entry:
                    self.order = self.close()
                    return
            if self.entry_bar is not None and (len(self) - self.entry_bar) >= self.p.timebox_bars:
                self.order = self.close()
                return
            return

        if drawdown >= self.p.dd_limit:
            return
        if self.data.close[0] < self.sma200[0]:
            return
        if not (self.ema50[0] > self.sma200[0]):
            return
        if not (self.rsi[0] < self.p.rsi_threshold):
            return
        if not (self.data.low[0] <= self.bb.lines.bot[0]):
            return

        bb_width = (self.bb.lines.top[0] - self.bb.lines.bot[0]) / self.bb.lines.mid[0]
        if not (self.p.bb_width_low <= bb_width <= self.p.bb_width_high):
            return

        stop_distance = self.p.atr_stop_mult * self.atr[0]
        if stop_distance <= 0:
            return
        risk_amount = value * self.p.risk_pct
        size = int(risk_amount / stop_distance)
        if size < 1:
            return

        self._pending_atr = self.atr[0]
        self._pending_bb_width = bb_width
        self.order = self.buy(size=size)
