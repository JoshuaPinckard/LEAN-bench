import backtrader as bt
import math


class BounceOffSupportInAnUptrend(bt.Strategy):
    params = (
        ("risk_fraction", 0.01),
    )

    def __init__(self):
        self.ema20 = bt.indicators.EMA(self.data.close, period=20)
        self.ema50 = bt.indicators.EMA(self.data.close, period=50)
        self.rsi = bt.indicators.RSI(self.data.close, period=14)
        self.volume_sma = bt.indicators.SMA(self.data.volume, period=20)
        self.atr = bt.indicators.ATR(self.data, period=14)
        self.hh20 = bt.indicators.Highest(self.data.high, period=20)
        self.swing_low5 = bt.indicators.Lowest(self.data.low, period=5)

        self.order = None
        self.stop_order = None
        self.target_order = None
        self.pending_stop = None
        self.pending_target = None
        self.pending_size = None

        self.addminperiod(51)

    def next(self):
        if self.order:
            return

        if self.position:
            if self.data.close[0] < self.ema20[0]:
                if self.stop_order:
                    self.cancel(self.stop_order)
                if self.target_order:
                    self.cancel(self.target_order)
                self.order = self.close()
            return

        return_20 = (self.data.close[0] / self.data.close[-20]) - 1.0
        support_test_current = self.data.low[0] <= self.ema20[0]
        support_test_previous = self.data.low[-1] <= self.ema20[-1]

        if support_test_current:
            test_low = self.data.low[0]
        else:
            test_low = self.data.low[-1]

        entry_price = self.data.close[0]
        stop_price = min(self.swing_low5[0], test_low) - (0.5 * self.atr[0])
        risk_per_share = entry_price - stop_price

        conditions = (
            return_20 >= 0.15
            and self.ema50[0] > self.ema50[-1]
            and self.data.close[0] > self.ema20[0]
            and self.ema20[0] > self.ema20[-1]
            and (support_test_current or support_test_previous)
            and self.data.close[0] > self.data.high[-1]
            and 40.0 <= self.rsi[0] <= 60.0
            and self.data.volume[0] > (1.2 * self.volume_sma[0])
            and risk_per_share > 0.0
            and (self.hh20[0] - entry_price) >= (2.0 * risk_per_share)
        )

        if conditions:
            risk_amount = self.broker.getvalue() * self.p.risk_fraction
            size = math.floor(risk_amount / risk_per_share)

            if size > 0:
                self.pending_stop = stop_price
                self.pending_target = self.hh20[0]
                self.pending_size = size
                self.order = self.buy(size=size)

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status == order.Completed:
            if order == self.order:
                if order.isbuy():
                    self.stop_order = self.sell(
                        size=self.pending_size,
                        exectype=bt.Order.Stop,
                        price=self.pending_stop,
                    )
                    self.target_order = self.sell(
                        size=self.pending_size,
                        exectype=bt.Order.Limit,
                        price=self.pending_target,
                        oco=self.stop_order,
                    )
                self.order = None
            elif order == self.stop_order or order == self.target_order:
                other = (
                    self.target_order
                    if order == self.stop_order
                    else self.stop_order
                )
                if other:
                    self.cancel(other)
                self.stop_order = None
                self.target_order = None
                self.order = None
            return

        if order.status in [
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            if order == self.order:
                self.order = None
            if order == self.stop_order:
                self.stop_order = None
            if order == self.target_order:
                self.target_order = None
