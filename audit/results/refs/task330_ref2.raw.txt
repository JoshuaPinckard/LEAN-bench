import collections
import backtrader as bt


class MonthlySeasonalityStrategy(bt.Strategy):
    params = (
        ("min_occurrences", 1),
        ("stop_loss_pct", 0.02),
    )

    def __init__(self):
        self.order = None
        self.monthly_returns = collections.defaultdict(list)
        self.current_month = None
        self.month_first_open = None
        self.month_last_close = None
        self.stopped_out = False

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return
        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            self.order = None

    def next(self):
        dt = self.data.datetime.datetime(0)
        bar_month = dt.month

        if self.current_month is None:
            self.current_month = bar_month
            self.month_first_open = self.data.open[0]
            self.month_last_close = self.data.close[0]
            self.stopped_out = False
            return

        if bar_month != self.current_month:
            # Finalize return for the completed month
            if (
                self.month_first_open is not None
                and self.month_first_open != 0
            ):
                prev_ret = (
                    self.month_last_close / self.month_first_open
                ) - 1.0
                self.monthly_returns[self.current_month].append(prev_ret)

            # Start new month
            self.current_month = bar_month
            self.month_first_open = self.data.open[0]
            self.month_last_close = self.data.close[0]
            self.stopped_out = False

            # Compute historical average return for this calendar month
            history = self.monthly_returns[bar_month]
            if len(history) >= self.p.min_occurrences:
                avg_ret = sum(history) / len(history)
            else:
                avg_ret = 0.0

            # Execute seasonal signal
            if self.order is None:
                if avg_ret > 0:
                    if self.position.size < 0:
                        self.close()
                        self.order = self.buy()
                    elif self.position.size == 0:
                        self.order = self.buy()
                elif avg_ret < 0:
                    if self.position.size > 0:
                        self.close()
                        self.order = self.sell()
                    elif self.position.size == 0:
                        self.order = self.sell()
                else:
                    if self.position.size != 0:
                        self.order = self.close()
        else:
            self.month_last_close = self.data.close[0]

            # Check intramonth stop loss
            if (
                self.order is None
                and self.position.size != 0
                and not self.stopped_out
            ):
                entry = self.position.price
                if self.position.size > 0:
                    if self.data.low[0] <= entry * (1.0 - self.p.stop_loss_pct):
                        self.order = self.close()
                        self.stopped_out = True
                elif self.position.size < 0:
                    if (
                        self.data.high[0]
                        >= entry * (1.0 + self.p.stop_loss_pct)
                    ):
                        self.order = self.close()
                        self.stopped_out = True
