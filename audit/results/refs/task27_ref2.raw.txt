import backtrader as bt


class ReferenceStrategy(bt.Strategy):
    params = (("sma_period", 20),)

    def __init__(self):
        self.sma = bt.indicators.SimpleMovingAverage(
            self.data.close, period=self.p.sma_period
        )
        self.longcondition = self.data.close > self.sma

        self.order = None
        self.previousExitPrice = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            if order.issell():
                self.previousExitPrice = order.executed.price
            self.order = None
        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        if self.order:
            return

        if not self.position:
            cond1 = bool(self.longcondition[0])
            cond2 = (self.previousExitPrice is None) or (
                self.data.close[-1] < self.previousExitPrice
            )

            if cond1 and cond2:
                cash = self.broker.getcash()
                price = self.data.close[0]
                size = int(cash / (price * 1.001))
                if size > 0:
                    self.order = self.buy(size=size)
        else:
            if not bool(self.longcondition[0]):
                self.order = self.close()
