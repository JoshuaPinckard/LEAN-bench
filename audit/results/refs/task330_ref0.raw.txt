import backtrader as bt
from collections import defaultdict


class MonthlySeasonalityStrategy(bt.Strategy):
    params = dict(
        min_occurrences=1,
        stop_loss_pct=0.02,
    )

    def __init__(self):
        self.order = None
        self.month_returns = defaultdict(list)
        self.current_month_key = None
        self.month_first_open = None
        self.month_last_close = None
        self.stopped_out = False
        self.entry_price = None

    def notify_order(self, order):
        if order.status == order.Completed:
            if order.isbuy() and self.position.size > 0:
                self.entry_price = order.executed.price
            elif order.issell() and self.position.size < 0:
                self.entry_price = order.executed.price
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        dt = self.data.datetime.datetime(0)
        month_key = (dt.year, dt.month)

        if self.current_month_key is None:
            self.current_month_key = month_key
            self.month_first_open = self.data.open[0]
            self.month_last_close = self.data.close[0]
            return

        if month_key != self.current_month_key:
            prev_month_num = self.current_month_key[1]
            completed_return = (self.month_last_close / self.month_first_open) - 1
            self.month_returns[prev_month_num].append(completed_return)

            self.current_month_key = month_key
            self.month_first_open = self.data.open[0]
            self.month_last_close = self.data.close[0]
            self.stopped_out = False

            history = self.month_returns[dt.month]
            if len(history) >= self.p.min_occurrences:
                avg_return = sum(history) / len(history)
                new_bias = 1 if avg_return > 0 else (-1 if avg_return < 0 else 0)
            else:
                new_bias = 0

            if not self.order:
                pos_dir = 0
                if self.position:
                    pos_dir = 1 if self.position.size > 0 else -1

                if new_bias != pos_dir:
                    if self.position:
                        self.order = self.close()
                    if new_bias == 1:
                        self.order = self.buy()
                    elif new_bias == -1:
                        self.order = self.sell()
        else:
            self.month_last_close = self.data.close[0]

            if self.position and not self.stopped_out and not self.order:
                if self.position.size > 0 and self.data.low[0] <= self.entry_price * (1 - self.p.stop_loss_pct):
                    self.order = self.close()
                    self.stopped_out = True
                elif self.position.size < 0 and self.data.high[0] >= self.entry_price * (1 + self.p.stop_loss_pct):
                    self.order = self.close()
                    self.stopped_out = True
