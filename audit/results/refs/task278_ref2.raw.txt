import backtrader as bt


class OptionIncomeStrategy(bt.Strategy):
    params = (("strategy_type", 1),)

    def __init__(self):
        self.order = None
        self.last_exit_date = None
        self.entry_bar = 0
        self.bought_bnh = False

        st = self.p.strategy_type
        if isinstance(st, str):
            st_lower = st.lower()
            if st_lower in ["1", "cc", "covered_call", "covered call"]:
                self.st_type = 1
            elif st_lower in [
                "2",
                "csp",
                "cash_secured_put",
                "cash secured put",
            ]:
                self.st_type = 2
            elif st_lower in ["3", "wheel"]:
                self.st_type = 3
            elif st_lower in ["4", "bnh", "buy_and_hold", "buy & hold"]:
                self.st_type = 4
            else:
                self.st_type = 1
        else:
            self.st_type = int(st)

        self.bb = bt.indicators.BollingerBands(
            self.data.close, period=20, devfactor=2
        )
        self.sma = bt.indicators.SMA(self.data.close, period=50)
        self.rsi = bt.indicators.RSI(self.data.close, period=14)
        self.atr = bt.indicators.ATR(self.data, period=14)

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            self.order = None

        if order.status == order.Completed:
            if order.isbuy():
                self.entry_bar = len(self)
            elif order.issell():
                self.last_exit_date = self.data.datetime.date(0)

    def next(self):
        if self.order is not None:
            return

        current_date = self.data.datetime.date(0)
        is_monday = current_date.weekday() == 0

        close = self.data.close[0]
        sma = self.sma[0]
        rsi = self.rsi[0]
        atr = self.atr[0]
        bb_top = self.bb.top[0]
        bb_bot = self.bb.bot[0]
        bb_mid = self.bb.mid[0]

        if self.st_type == 4:
            if not self.position and not self.bought_bnh:
                self.order = self.buy()
                self.bought_bnh = True
            return

        if not self.position:
            if not is_monday:
                return

            if self.st_type == 1:
                cooldown_ok = (
                    self.last_exit_date is None
                    or (current_date - self.last_exit_date).days >= 7
                )
                if cooldown_ok and close > sma and rsi > 50:
                    self.order = self.buy()

            elif self.st_type == 2:
                if close <= bb_bot or rsi <= 35:
                    self.order = self.buy()

            elif self.st_type == 3:
                if close <= bb_bot or rsi <= 35:
                    self.order = self.buy()

        else:
            if self.st_type == 1:
                if (
                    close >= bb_top
                    or rsi >= 70
                    or close <= (bb_mid - 1.0 * atr)
                ):
                    self.order = self.close()

            elif self.st_type == 2:
                bars_held = len(self) - self.entry_bar
                if (
                    close >= bb_mid
                    or bars_held >= 20
                    or close <= (bb_bot - 1.0 * atr)
                ):
                    self.order = self.close()

            elif self.st_type == 3:
                if (
                    close >= bb_top
                    or rsi >= 70
                    or close <= (bb_mid - 1.0 * atr)
                ):
                    self.order = self.close()
