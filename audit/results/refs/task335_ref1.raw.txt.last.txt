import backtrader as bt
import collections
import statistics


class IntradayVWAPMeanReversion(bt.Strategy):
    params = (
        ("ThresholdMultiplier", 2),
        ("StopLossPercent", 0.02),
    )

    def __init__(self):
        self.entry_order = None
        self.exit_order = None
        self.stop_order = None

        self._session_date = None
        self._cumulative_pv = 0.0
        self._cumulative_volume = 0.0

        self._vwaps = collections.deque(maxlen=21)
        self._slopes = collections.deque(maxlen=20)

        self._prev_slope = None
        self._prev_slope_std = None
        self._prev_vwap = None
        self._prev_vwap_base = None

    def notify_order(self, order):
        if order.status in (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            if order is self.entry_order:
                self.entry_order = None

                if order.status == order.Completed:
                    if self.position.size > 0:
                        self.stop_order = self.sell(
                            size=self.position.size,
                            exectype=bt.Order.Stop,
                            price=order.executed.price * (1.0 - self.p.StopLossPercent),
                        )
                    elif self.position.size < 0:
                        self.stop_order = self.buy(
                            size=abs(self.position.size),
                            exectype=bt.Order.Stop,
                            price=order.executed.price * (1.0 + self.p.StopLossPercent),
                        )

            if order is self.exit_order:
                self.exit_order = None

            if order is self.stop_order:
                self.stop_order = None

    def next(self):
        current_date = self.data.datetime.datetime(0).date()

        if current_date != self._session_date:
            self._session_date = current_date
            self._cumulative_pv = 0.0
            self._cumulative_volume = 0.0

        typical_price = (
            self.data.high[0] + self.data.low[0] + self.data.close[0]
        ) / 3.0
        volume = self.data.volume[0]

        self._cumulative_pv += typical_price * volume
        self._cumulative_volume += volume

        if self._cumulative_volume == 0:
            return

        vwap = self._cumulative_pv / self._cumulative_volume
        self._vwaps.append(vwap)

        if len(self._vwaps) < 21:
            return

        slope = vwap - self._vwaps[0]
        self._slopes.append(slope)

        if len(self._slopes) < 20:
            self._prev_slope = slope
            return

        slope_std = statistics.pstdev(self._slopes)
        threshold = self.p.ThresholdMultiplier * slope_std
        vwap_base = sum(list(self._vwaps)[-20:]) / 20.0

        if (
            self.position.size > 0
            and self.exit_order is None
            and self._prev_vwap is not None
            and self._prev_vwap_base is not None
            and self._prev_vwap <= self._prev_vwap_base
            and vwap > vwap_base
        ):
            if self.stop_order is not None:
                self.cancel(self.stop_order)
            self.exit_order = self.close()

        elif (
            self.position.size < 0
            and self.exit_order is None
            and self._prev_vwap is not None
            and self._prev_vwap_base is not None
            and self._prev_vwap >= self._prev_vwap_base
            and vwap < vwap_base
        ):
            if self.stop_order is not None:
                self.cancel(self.stop_order)
            self.exit_order = self.close()

        elif (
            not self.position
            and self.entry_order is None
            and self.exit_order is None
            and self._prev_slope is not None
            and self._prev_slope_std is not None
        ):
            previous_threshold = self.p.ThresholdMultiplier * self._prev_slope_std

            if self._prev_slope < -previous_threshold and slope > self._prev_slope:
                self.entry_order = self.buy()

            elif self._prev_slope > previous_threshold and slope < self._prev_slope:
                self.entry_order = self.sell()

        self._prev_slope = slope
        self._prev_slope_std = slope_std
        self._prev_vwap = vwap
        self._prev_vwap_base = vwap_base
