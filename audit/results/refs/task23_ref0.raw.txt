import backtrader as bt
import collections
import statistics


class IntradayEmaSmaRsiStrategy(bt.Strategy):
    def __init__(self):
        self.ema20 = bt.indicators.EMA(self.data.close, period=20)
        self.sma50 = bt.indicators.SMA(self.data.close, period=50)
        self.rsi = bt.indicators.RSI(self.data.close, period=14)
        self.crossover = bt.indicators.CrossOver(self.ema20, self.sma50)

        self.order = None
        self.stop_order = None
        self.limit_order = None
        self.entry_price = None

        self.current_day = None
        self.day_volume = 0.0
        self.daily_volumes = collections.deque(maxlen=20)

        self.entry_start = 9 * 60 + 30
        self.no_entry_before = 10 * 60
        self.entry_end = 14 * 60 + 30
        self.no_entry_after_start = 15 * 60 + 30
        self.no_entry_after_end = 16 * 60
        self.mandatory_close_after = 15 * 60 + 30

    def _update_daily_volume(self):
        dt = self.data.datetime.date(0)
        if self.current_day is None or dt != self.current_day:
            if self.current_day is not None:
                self.daily_volumes.append(self.day_volume)
            self.day_volume = 0.0
            self.current_day = dt
        self.day_volume += float(self.data.volume[0])

    def prenext(self):
        self._update_daily_volume()

    def next(self):
        self._update_daily_volume()

        if self.order:
            return

        t = self.data.datetime.time(0)
        minutes = t.hour * 60 + t.minute

        if not self.position:
            entry_window = self.entry_start <= minutes <= self.entry_end
            not_first_30 = not (self.entry_start <= minutes < self.no_entry_before)
            not_last_30 = not (self.no_entry_after_start <= minutes < self.no_entry_after_end)
            entry_time_ok = entry_window and not_first_30 and not_last_30

            entry_cross = self.crossover[0] > 0
            entry_rsi_ok = 40 <= self.rsi[0] <= 60

            avg_volume = statistics.mean(self.daily_volumes) if len(self.daily_volumes) == 20 else None
            volume_ok = avg_volume is not None and self.day_volume >= 1.2 * avg_volume

            if entry_cross and entry_rsi_ok and entry_time_ok and volume_ok:
                value = self.broker.getvalue()
                price = self.data.close[0]
                size = (0.02 * value) / price
                if size > 0:
                    self.order = self.buy(size=size)
        else:
            exit_cross = self.crossover[0] < 0
            exit_rsi = self.rsi[0] > 70
            exit_time = minutes > self.mandatory_close_after

            if exit_cross or exit_rsi or exit_time:
                if self.stop_order:
                    self.cancel(self.stop_order)
                if self.limit_order:
                    self.cancel(self.limit_order)
                self.order = self.close()

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status == order.Completed:
            if order.isbuy():
                self.entry_price = order.executed.price
                stop_price = self.entry_price * 0.99
                target_price = self.entry_price * 1.02
                size = abs(order.executed.size)
                self.stop_order = self.sell(exectype=bt.Order.Stop, price=stop_price, size=size)
                self.limit_order = self.sell(exectype=bt.Order.Limit, price=target_price, size=size, oco=self.stop_order)
            else:
                self.entry_price = None
                self.stop_order = None
                self.limit_order = None

        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            if order == self.order:
                self.order = None
