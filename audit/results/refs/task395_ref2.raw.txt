import math
import backtrader as bt


class HMA(bt.Indicator):
    lines = ('hma',)
    params = (('period', 9),)

    def __init__(self):
        p = self.p.period
        half_p = int(p / 2)
        sqrt_p = int(math.sqrt(p))
        wma_half = bt.indicators.WMA(self.data.close, period=half_p)
        wma_full = bt.indicators.WMA(self.data.close, period=p)
        diff = 2.0 * wma_half - wma_full
        self.lines.hma = bt.indicators.WMA(diff, period=sqrt_p)


class MeanReversionHMA(bt.Strategy):
    params = (
        ('hma_period', 9),
        ('sma_period', 20),
        ('std_period', 20),
        ('atr_period', 14),
        ('std_mult', 2.0),
        ('atr_mult', 2.0),
    )

    def __init__(self):
        self.hma = HMA(self.data.close, period=self.p.hma_period)
        self.sma_hma = bt.indicators.SMA(self.hma, period=self.p.sma_period)
        self.std_hma = bt.indicators.StdDev(self.hma, period=self.p.std_period)
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)

        self.upper_band = self.sma_hma + self.p.std_mult * self.std_hma
        self.lower_band = self.sma_hma - self.p.std_mult * self.std_hma

        self.order = None
        self.stop_order = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            if order == self.order:
                self.order = None
                if self.position.size > 0 and self.stop_order is None:
                    stop_price = order.executed.price - self.p.atr_mult * self.atr[0]
                    self.stop_order = self.sell(exectype=bt.Order.Stop, price=stop_price)
                elif self.position.size < 0 and self.stop_order is None:
                    stop_price = order.executed.price + self.p.atr_mult * self.atr[0]
                    self.stop_order = self.buy(exectype=bt.Order.Stop, price=stop_price)

            elif order == self.stop_order:
                self.stop_order = None

        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            if order == self.order:
                self.order = None
            if order == self.stop_order:
                self.stop_order = None

    def next(self):
        if self.order:
            return

        if self.position.size > 0:
            if self.hma[-1] < self.sma_hma[-1] and self.hma[0] > self.sma_hma[0]:
                if self.stop_order:
                    self.cancel(self.stop_order)
                    self.stop_order = None
                self.order = self.close()

        elif self.position.size < 0:
            if self.hma[-1] > self.sma_hma[-1] and self.hma[0] < self.sma_hma[0]:
                if self.stop_order:
                    self.cancel(self.stop_order)
                    self.stop_order = None
                self.order = self.close()

        else:
            if self.hma[-1] < self.lower_band[-1] and self.hma[0] > self.lower_band[0]:
                self.order = self.buy()
            elif self.hma[-1] > self.upper_band[-1] and self.hma[0] < self.upper_band[0]:
                self.order = self.sell()
