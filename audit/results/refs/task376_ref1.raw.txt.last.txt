import backtrader as bt


class RSIDivergenceStrategy(bt.Strategy):
    params = (
        ('rsi_period', 14),
        ('pivot_window', 5),
        ('stop_loss_pct', 0.02),
    )

    def __init__(self):
        self.rsi = bt.indicators.RSI(self.data.close, period=self.p.rsi_period)
        self.rsi_at_pivot = self.rsi(-self.p.pivot_window)
        self.order = None
        self.entry_price = None
        self.pivot_lows = []
        self.pivot_highs = []

    def notify_order(self, order):
        if order.status == order.Completed:
            if order.isbuy() or order.issell():
                if not self.position:
                    self.entry_price = None
                elif self.entry_price is None:
                    self.entry_price = order.executed.price

            self.order = None

        elif order.status in (order.Canceled, order.Margin, order.Rejected):
            self.order = None

    def next(self):
        if self.order:
            return

        if self.position:
            if self.position.size > 0:
                if ((self.rsi[-1] <= 70 and self.rsi[0] > 70) or
                        self.data.low[0] <= self.entry_price * (1.0 - self.p.stop_loss_pct)):
                    self.order = self.close()
            else:
                if ((self.rsi[-1] >= 30 and self.rsi[0] < 30) or
                        self.data.high[0] >= self.entry_price * (1.0 + self.p.stop_loss_pct)):
                    self.order = self.close()
            return

        n = self.p.pivot_window
        pivot_low_price = self.data.low[-n]
        pivot_high_price = self.data.high[-n]
        pivot_rsi = self.rsi_at_pivot[0]

        is_pivot_low = pivot_low_price == min(self.data.low[-i] for i in range(2 * n + 1))
        is_pivot_high = pivot_high_price == max(self.data.high[-i] for i in range(2 * n + 1))

        bullish_divergence = False
        bearish_divergence = False

        if is_pivot_low:
            self.pivot_lows.append((pivot_low_price, pivot_rsi))
            if len(self.pivot_lows) >= 2:
                prior_price, prior_rsi = self.pivot_lows[-2]
                recent_price, recent_rsi = self.pivot_lows[-1]
                bullish_divergence = recent_price < prior_price and recent_rsi > prior_rsi

        if is_pivot_high:
            self.pivot_highs.append((pivot_high_price, pivot_rsi))
            if len(self.pivot_highs) >= 2:
                prior_price, prior_rsi = self.pivot_highs[-2]
                recent_price, recent_rsi = self.pivot_highs[-1]
                bearish_divergence = recent_price > prior_price and recent_rsi < prior_rsi

        size = self.broker.getcash() / self.data.close[0]

        if bullish_divergence:
            self.order = self.buy(size=size)
        elif bearish_divergence:
            self.order = self.sell(size=size)
