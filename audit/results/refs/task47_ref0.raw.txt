import backtrader as bt


class FVGSwingFailureBreakout(bt.Strategy):
    # FVG Swing Failure Breakout Strategy
    # 4H: Fair Value Gap, Swing Failure Pattern, Market Structure Shift
    # 5m: Order Block breakout entry, partial take-profit, OB stop, max-hold, no weekends
    params = dict(
        risk_pct=0.01,        # fixed 1% risk per trade
        rr=3.3333,            # risk:reward 1:3.33 ($300 -> $1000)
        max_hold_bars=48,     # "3-4 hours" -> 4 hours on the 5m feed
        setup_window=6,       # 4H candles: FVG-entry / SFP / MSS must co-occur inside this window
        ob_lookback=20,       # 5m bars scanned for the order block
    )

    def __init__(self):
        self.order = None

        # ---- 4H aggregation (single 5m feed -> 4H candles) ----
        self.cur_period = None
        self.h4o = self.h4h = self.h4l = self.h4c = None
        self.h4 = []                 # completed 4H candles: dict(o, h, l, c)

        # ---- 4H structure ----
        self.swing_highs = []        # (idx, price)
        self.swing_lows = []         # (idx, price)
        self.fvgs = []               # active bullish FVGs: dict(lower, upper, idx)

        # ---- confirmed condition indices (in completed-4H units) ----
        self.fvg_entered_idx = None
        self.sfp_idx = None
        self.mss_idx = None

        # ---- 5m order block ----
        self.ob_top = None
        self.ob_bottom = None

        # ---- trade management ----
        self.entry_price = None
        self.stop_price = None
        self.tp_half = None
        self.tp_full = None
        self.entry_size = None
        self.partial_done = False
        self.bar_executed = None

    # ------------------------------------------------------------------ #
    #  Order / trade bookkeeping
    # ------------------------------------------------------------------ #
    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return
        if order.status in (order.Completed, order.Canceled,
                            order.Margin, order.Rejected):
            self.order = None

    def notify_trade(self, trade):
        if trade.isclosed:
            self.entry_price = None
            self.stop_price = None
            self.tp_half = None
            self.tp_full = None
            self.entry_size = None
            self.partial_done = False
            self.bar_executed = None

    # ------------------------------------------------------------------ #
    #  4H candle aggregation
    # ------------------------------------------------------------------ #
    def _start_h4(self, period):
        self.cur_period = period
        self.h4o = self.data.open[0]
        self.h4h = self.data.high[0]
        self.h4l = self.data.low[0]
        self.h4c = self.data.close[0]

    def _update_h4(self):
        self.h4h = max(self.h4h, self.data.high[0])
        self.h4l = min(self.h4l, self.data.low[0])
        self.h4c = self.data.close[0]

    def _finalize_h4(self):
        self.h4.append(dict(o=self.h4o, h=self.h4h, l=self.h4l, c=self.h4c))
        self._process_h4()

    # ------------------------------------------------------------------ #
    #  4H analysis: FVG, Swing Failure, Market Structure Shift
    # ------------------------------------------------------------------ #
    def _process_h4(self):
        n = len(self.h4)
        cur = self.h4[n - 1]

        # --- Fair Value Gap: bullish candle then bearish candle, bodies not overlapping ---
        if n >= 2:
            a = self.h4[n - 2]
            b = self.h4[n - 1]
            if a['c'] > a['o'] and b['c'] < b['o'] and b['c'] > a['c']:
                lo = min(a['h'], b['l'])
                hi = max(a['h'], b['l'])
                self.fvgs.append(dict(lower=lo, upper=hi, idx=n - 1))

        # --- Price returns into an earlier FVG zone ---
        for fvg in self.fvgs:
            if fvg['idx'] < n - 1:
                if cur['l'] <= fvg['upper'] and cur['h'] >= fvg['lower']:
                    self.fvg_entered_idx = n - 1

        # --- Confirmed 3-bar swing points (confirmed with 1-candle lag at index n-2) ---
        if n >= 3:
            m = n - 2
            if self.h4[m]['h'] > self.h4[m - 1]['h'] and self.h4[m]['h'] > self.h4[m + 1]['h']:
                self.swing_highs.append((m, self.h4[m]['h']))
                # Swing Failure Pattern (bullish): a *higher* high whose following
                # candle pulls back and does not reach it (inherent in the swing).
                if len(self.swing_highs) >= 2 and self.swing_highs[-1][1] > self.swing_highs[-2][1]:
                    self.sfp_idx = n - 1
            if self.h4[m]['l'] < self.h4[m - 1]['l'] and self.h4[m]['l'] < self.h4[m + 1]['l']:
                self.swing_lows.append((m, self.h4[m]['l']))

        # --- Market Structure Shift (bullish): break of the previous low ---
        if self.swing_lows and cur['l'] < self.swing_lows[-1][1]:
            self.mss_idx = n - 1

    # ------------------------------------------------------------------ #
    #  Setup / entry helpers
    # ------------------------------------------------------------------ #
    def _setup_ready(self):
        if self.fvg_entered_idx is None or self.sfp_idx is None or self.mss_idx is None:
            return False
        n = len(self.h4)
        w = self.p.setup_window
        return ((n - 1 - self.fvg_entered_idx) <= w and
                (n - 1 - self.sfp_idx) <= w and
                (n - 1 - self.mss_idx) <= w)

    def _find_order_block(self):
        # Order Block = body of the last strong (bearish) candle before a sharp up move.
        self.ob_top = None
        self.ob_bottom = None
        if len(self) < self.p.ob_lookback + 2:
            return
        for j in range(1, self.p.ob_lookback):
            o0, c0 = self.data.open[-j], self.data.close[-j]          # candidate move (up)
            o1, c1 = self.data.open[-j - 1], self.data.close[-j - 1]  # candidate order block (down)
            if c1 < o1 and c0 > o0:
                body_down = o1 - c1
                body_up = c0 - o0
                if body_down > 0 and body_up > body_down:
                    self.ob_top = o1     # top of bearish body
                    self.ob_bottom = c1  # bottom of bearish body
                    break

    def _enter_long(self):
        entry = self.data.close[0]
        stop = self.ob_bottom                      # stop on the opposite side of the OB
        risk_per_unit = entry - stop
        if risk_per_unit <= 0:
            return
        value = self.broker.getvalue()
        size = (value * self.p.risk_pct) / risk_per_unit
        size = min(size, (value * 0.99) / entry)   # cap to affordable notional
        if size <= 0:
            return

        self.entry_price = entry
        self.stop_price = stop
        self.tp_full = entry + self.p.rr * risk_per_unit
        self.tp_half = entry + 0.5 * self.p.rr * risk_per_unit
        self.entry_size = size
        self.partial_done = False
        self.bar_executed = len(self)
        self.order = self.buy(size=size)

        # consume the setup (no re-entry until conditions re-form)
        self.fvg_entered_idx = None
        self.sfp_idx = None
        self.mss_idx = None
        self.ob_top = None
        self.ob_bottom = None

    def _manage_position(self):
        # Maximum holding time
        if self.bar_executed is not None and (len(self) - self.bar_executed) > self.p.max_hold_bars:
            self.order = self.close()
            return
        # Stop-loss (opposite side of the order block)
        if self.data.low[0] <= self.stop_price:
            self.order = self.close()
            return
        # Take-profit: first portion at 50% of total TP
        if not self.partial_done and self.data.high[0] >= self.tp_half:
            self.order = self.sell(size=self.entry_size / 2.0)
            self.partial_done = True
            return
        # Take-profit: second portion at 100% of total TP
        if self.partial_done and self.data.high[0] >= self.tp_full:
            self.order = self.close()
            return

    # ------------------------------------------------------------------ #
    #  Main loop
    # ------------------------------------------------------------------ #
    def next(self):
        dt = self.data.datetime.datetime(0)
        period = (dt.year, dt.month, dt.day, dt.hour // 4)
        if self.cur_period is None:
            self._start_h4(period)
        elif period != self.cur_period:
            self._finalize_h4()
            self._start_h4(period)
        else:
            self._update_h4()

        if self.order:
            return

        if self.position:
            self._manage_position()
            return

        # Strategy not active on weekends (UTC Sat/Sun) -> no new entries
        if dt.weekday() >= 5:
            return

        if not self._setup_ready():
            return

        self._find_order_block()
        if self.ob_top is None:
            return

        # Entry on an upward breakout of the order block
        if self.data.close[0] > self.ob_top:
            self._enter_long()
