    def _enter_long(self):
        price = self.data.close[0]
        stop_price = max(self.psar[0], price - self.p.atr_mult * self.atr[0])
        risk_per_share = price - stop_price
        if risk_per_share <= 0:
            return
        size = self._compute_size(price, risk_per_share)
        if size <= 0:
            return
        self.order_entry = self.buy(size=size)

    def _enter_short(self):
        price = self.data.close[0]
        stop_price = min(self.psar[0], price + self.p.atr_mult * self.atr[0])
        risk_per_share = stop_price - price
        if risk_per_share <= 0:
            return
        size = self._compute_size(price, risk_per_share)
        if size <= 0:
            return
        self.order_entry = self.sell(size=size)

    def _compute_size(self, price, risk_per_share):
        equity = self.broker.getvalue()
        risk_amount = self.p.risk_pct * equity
        size = risk_amount / risk_per_share
        if price > 0:
            max_affordable = self.broker.getcash() / price
            size = min(size, max_affordable)
        return int(size)
