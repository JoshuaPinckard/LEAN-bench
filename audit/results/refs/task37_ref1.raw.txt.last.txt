import backtrader as bt


class TrendDayMACrossover(bt.Strategy):
    params = (
        ('ma20_period', 20),
        ('ma30_period', 30),
        ('ma60_period', 60),
    )

    def __init__(self):
        self.ma20 = bt.indicators.SMA(self.data.close, period=self.p.ma20_period)
        self.ma30 = bt.indicators.SMA(self.data.close, period=self.p.ma30_period)
        self.ma60 = bt.indicators.SMA(self.data.close, period=self.p.ma60_period)

        self.order = None
        self.sequence_stage = 0

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled,
                            order.Margin, order.Rejected]:
            self.order = None

            if order.status == order.Completed and order.isbuy():
                self.sequence_stage = 0

    def next(self):
        if self.order:
            return

        if self.position:
            if self.data.high[0] >= 0.278:
                self.order = self.close()
            elif (self.data.close[0] < 0.265 and
                  self.data.close[0] < self.data.open[0] and
                  self.data.volume[0] > self.data.volume[-1]):
                self.order = self.close()
            return

        crossed_ma20 = (
            self.data.close[-1] <= self.ma20[-1] and
            self.data.close[0] > self.ma20[0]
        )
        crossed_ma30 = (
            self.data.close[-1] <= self.ma30[-1] and
            self.data.close[0] > self.ma30[0]
        )

        if self.sequence_stage == 0 and crossed_ma20:
            self.sequence_stage = 1
        elif self.sequence_stage == 1 and crossed_ma30:
            self.sequence_stage = 2

        hourly_volume_increasing = (
            sum(self.data.volume.get(size=12)) >
            sum(self.data.volume.get(ago=-12, size=12))
        )

        if (self.sequence_stage == 2 and
                self.data.close[0] > self.ma60[0] and
                hourly_volume_increasing):
            self.order = self.buy()
