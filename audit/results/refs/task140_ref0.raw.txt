import backtrader as bt


class MonthlyLongStrategy(bt.Strategy):
    def __init__(self):
        self.order = None

    def start(self):
        print(f"Initial Portfolio Value: {self.broker.getvalue():.2f}")

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        if self.order:
            return

        current_date = self.data.datetime.date(0)

        is_first_trading_day = (len(self) == 1) or (
            self.data.datetime.date(-1).month != current_date.month
        )

        try:
            next_date = self.data.datetime.date(1)
            is_last_trading_day = next_date.month != current_date.month
        except IndexError:
            is_last_trading_day = True

        if is_last_trading_day and self.position:
            self.order = self.close()
        elif is_first_trading_day and not self.position:
            self.order = self.order_target_percent(target=1.0)

    def stop(self):
        print(f"Final Portfolio Value: {self.broker.getvalue():.2f}")
