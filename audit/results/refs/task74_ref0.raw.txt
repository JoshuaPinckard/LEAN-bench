import backtrader as bt


class DynamicStepTrailStrategy(bt.Strategy):
    params = (
        ('lot_size', 0.1),
        ('initial_stop_points', 10.0),
        ('take_profit_points', 50.0),
        ('trail_step_points', 10.0),
    )

    def __init__(self):
        self.order = None
        self.entering = False
        self.entry_price = None
        self.stop_price = None
        self.take_price = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status == order.Completed:
            if self.entering:
                self.entry_price = order.executed.price
                if order.isbuy():
                    self.stop_price = self.entry_price - self.p.initial_stop_points
                    self.take_price = self.entry_price + self.p.take_profit_points
                else:
                    self.stop_price = self.entry_price + self.p.initial_stop_points
                    self.take_price = self.entry_price - self.p.take_profit_points
                self.entering = False
            else:
                self.entry_price = None
                self.stop_price = None
                self.take_price = None
        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            self.entering = False

        self.order = None

    def next(self):
        if self.order:
            return

        if not self.position:
            self.entering = True
            self.order = self.buy(size=self.p.lot_size)
            return

        step = self.p.trail_step_points
        max_tier = self.p.take_profit_points - step

        if self.position.size > 0:
            if self.data.low[0] <= self.stop_price:
                self.order = self.close()
            elif self.data.high[0] >= self.take_price:
                self.order = self.close()
            else:
                profit_points = self.data.high[0] - self.entry_price
                if profit_points >= step:
                    tier = min(int(profit_points // step) * step, max_tier)
                    candidate_stop = self.entry_price + tier
                    if candidate_stop > self.stop_price:
                        self.stop_price = candidate_stop
        else:
            if self.data.high[0] >= self.stop_price:
                self.order = self.close()
            elif self.data.low[0] <= self.take_price:
                self.order = self.close()
            else:
                profit_points = self.entry_price - self.data.low[0]
                if profit_points >= step:
                    tier = min(int(profit_points // step) * step, max_tier)
                    candidate_stop = self.entry_price - tier
                    if candidate_stop < self.stop_price:
                        self.stop_price = candidate_stop
