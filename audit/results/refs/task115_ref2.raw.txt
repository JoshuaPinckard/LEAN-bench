```python
import backtrader as bt
import math
import collections
import statistics


class InverseHeadAndShouldersRSIDivergence(bt.Strategy):
    """
    Reference implementation of the Inverse Head & Shoulders + Weekly RSI Divergence strategy.
    
    Signals:
    1. Inverse Head & Shoulders pattern completed on daily timeframe (upward breakout of the neckline).
    2. Positive RSI(14) divergence on weekly timeframe (price lower low, RSI higher low).
    3. Monthly trend filter is bullish.
    
    Exits:
    - Stop-loss at the lowest price of the right shoulder.
    - Take-profit set at a 2:1 reward-to-risk ratio.
    """

    def __init__(self):
        self.order = None
        self.sl_price = None
        self.tp_price = None
        self.entry_price = None

        self.weekly_data = []
        self.monthly_data = []

    def calc_rsi(self, closes, period=14):
        if len(closes) < period + 1:
            return [None] * len(closes)

        gains = []
        losses = []
        for i in range(1, len(closes)):
            diff = closes[i] - closes[i - 1]
            gains.append(max(diff, 0.0))
            losses.append(max(-diff, 0.0))

        rsi_values = [None] * period
        avg_gain = sum(gains[:period]) / float(period)
        avg_loss = sum(losses[:period]) / float(period)

        if avg_loss == 0:
            rsi_values.append(100.0)
        else:
            rs = avg_gain / avg_loss
            rsi_values.append(100.0 - (100.0 / (1.0 + rs)))

        for i in range(period, len(gains)):
            avg_gain = (avg_gain * (period - 1) + gains[i]) / float(period)
            avg_loss = (avg_loss * (period - 1) + losses[i]) / float(period)
            if avg_loss == 0:
                rsi_values.append(100.0)
            else:
                rs = avg_gain / avg_loss
                rsi_values.append(100.0 - (100.0 / (1.0 + rs)))

        return rsi_values

    def check_ihs_pattern(self):
        N = 5
        curr_idx = len(self) - 1
        if curr_idx < 4 * N + 10:
            return False, None

        lows = list(self.data.low.get(size=len(self)))
        highs = list(self.data.high.get(size=len(self)))
        closes = list(self.data.close.get(size=len(self)))

        swing_lows = []
        for i in range(N, curr_idx - N + 1):
            window = lows[i - N : i + N + 1]
            if lows[i] == min(window):
                swing_lows.append((i, lows[i]))

        if len(swing_lows) < 3:
            return False, None

        sl1_idx, sl1_val = swing_lows[-3]
        sl2_idx, sl2_val = swing_lows[-2]
        sl3_idx, sl3_val = swing_lows[-1]

        if not (sl2_val < sl1_val and sl2_val < sl3_val):
            return False, None

        sh1_candidates = [(i, highs[i]) for i in range(sl1_idx + 1, sl2_idx) if N <= i <= curr_idx - N]
        if not sh1_candidates:
            if sl2_idx - sl1_idx <= 1:
                return False, None
            sub = highs[sl1_idx + 1 : sl2_idx]
            max_val = max(sub)
            sh1_idx = sl1_idx + 1 + sub.index(max_val)
            sh1_val = max_val
        else:
            sh1_idx, sh1_val = max(sh1_candidates, key=lambda x: x[1])

        sh2_candidates = [(i, highs[i]) for i in range(sl2_idx + 1, sl3_idx) if N <= i <= curr_idx - N]
        if not sh2_candidates:
            if sl3_idx - sl2_idx <= 1:
                return False, None
            sub = highs[sl2_idx + 1 : sl3_idx]
            max_val = max(sub)
            sh2_idx = sl2_idx + 1 + sub.index(max_val)
            sh2_val = max_val
        else:
            sh2_idx, sh2_val = max(sh2_candidates, key=lambda x: x[1])

        if not (sl1_idx < sh1_idx < sl2_idx < sh2_idx < sl3_idx):
            return False, None

        dx = sh2_idx - sh1_idx
        if dx == 0:
            return False, None
        slope = (sh2_val - sh1_val) / float(dx)

        neckline_curr = sh1_val + slope * (curr_idx - sh1_idx)
        neckline_prev = sh1_val + slope * (curr_idx - 1 - sh1_idx)

        current_close = closes[curr_idx]
        prev_close = closes[curr_idx - 1]

        if current_close > neckline_curr and prev_close <= neckline_prev:
            return True, sl3_val

        return False, None

    def check_rsi_divergence(self):
        weekly_closes = [w['close'] for w in self.weekly_data]
        weekly_lows = [w['low'] for w in self.weekly_data]

        if len(weekly_closes) < 20:
            return False

        rsi_vals = self.calc_rsi(weekly_closes, period=14)

        troughs = []
        n_weeks = len(weekly_lows)
        for i in range(1, n_weeks - 1):
            if rsi_vals[i] is not None:
                if weekly_lows[i] <= weekly_lows[i - 1] and weekly_lows[i] <= weekly_lows[i + 1]:
                    troughs.append(i)

        if len(troughs) < 2:
            return False

        t1, t2 = troughs[-2], troughs[-1]

        if (n_weeks - 1 - t2) > 8:
            return False

        price_lower_low = weekly_lows[t2] < weekly_lows[t1]
        rsi_higher_low = rsi_vals[t2] > rsi_vals[t1]

        return price_lower_low and rsi_higher_low

    def check_monthly_trend(self):
        monthly_closes = [m['close'] for m in self.monthly_data]
        if len(monthly_closes) < 2:
            return True

        if len(monthly_closes) >= 10:
            sma10 = sum(monthly_closes[-10:]) / 10.0
            return monthly_closes[-1] > sma10
        else:
            return monthly_closes[-1] > monthly_closes[-2]

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            if order.isbuy():
                self.entry_price = order.executed.price
                if self.sl_price is not None:
                    risk = self.entry_price - self.sl_price
                    if risk > 0:
                        self.tp_price = self.entry_price + 2.0 * risk
            self.order = None
        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def notify_trade(self, trade):
        if trade.isclosed:
            self.sl_price = None
            self.tp_price = None
            self.entry_price = None

    def next(self):
        dt = self.data.datetime.date(0)
        iso_year, iso_week, _ = dt.isocalendar()
        m_key = (dt.year, dt.month)

        if not self.weekly_data or (iso_year, iso_week) != (self.weekly_data[-1]['year'], self.weekly_data[-1]['week']):
            self.weekly_data.append({
                'year': iso_year,
                'week': iso_week,
                'close': self.data.close[0],
                'low': self.data.low[0],
                'high': self.data.high[0]
            })
        else:
            self.weekly_data[-1]['close'] = self.data.close[0]
            self.weekly_data[-1]['low'] = min(self.weekly_data[-1]['low'], self.data.low[0])
            self.weekly_data[-1]['high'] = max(self.weekly_data[-1]['high'], self.data.high[0])

        if not self.monthly_data or m_key != self.monthly_data[-1]['year_month']:
            self.monthly_data.append({
                'year_month': m_key,
                'close': self.data.close[0]
            })
        else:
            self.monthly_data[-1]['close'] = self.data.close[0]

        if self.order:
            return

        if self.position:
            current_low = self.data.low[0]
            current_high = self.data.high[0]

            if self.sl_price is not None and current_low <= self.sl_price:
                self.order = self.close()
                return

            if self.tp_price is not None and current_high >= self.tp_price:
                self.order = self.close()
                return

            return

        has_ihs, sl_level = self.check_ihs_pattern()
        if not has_ihs or sl_level is None:
            return

        entry_price = self.data.close[0]
        if sl_level >= entry_price:
            return

        rsi_div = self.check_rsi_divergence()
        if not rsi_div:
            return

        monthly_bullish = self.check_monthly_trend()
        if not monthly_bullish:
            return

        capital = self.broker.getvalue()
        risk_pct = 0.02
        max_risk_amount = capital * risk_pct
        risk_per_share = entry_price - sl_level

        if risk_per_share <= 0:
            return

        size = int(max_risk_amount / risk_per_share)
        max_affordable = int((capital * 0.95) / entry_price)
        size = min(size, max_affordable)

        if size > 0:
            self.sl_price = sl_level
            self.tp_price = entry_price + 2.0 * risk_per_share
            self.entry_price = entry_price
            self.order = self.buy(size=size)
```
