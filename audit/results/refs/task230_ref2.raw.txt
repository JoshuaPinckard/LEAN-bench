import backtrader as bt


class WeeklyMACDMomentum(bt.Strategy):
    def __init__(self):
        self.order = None
        self.entry_price = None
        self.weeks_held = 0

        # Resampled weekly indicators
        self.macd = bt.indicators.MACD(
            self.data,
            period_me1=12,
            period_me2=26,
            period_signal=9,
            timeframe=bt.TimeFrame.Weeks,
        )
        self.crossover = bt.indicators.CrossOver(
            self.macd.macd, self.macd.signal, timeframe=bt.TimeFrame.Weeks
        )
        self.sma50 = bt.indicators.SMA(
            self.data.close, period=50, timeframe=bt.TimeFrame.Weeks
        )
        self.atr14 = bt.indicators.ATR(
            self.data, period=14, timeframe=bt.TimeFrame.Weeks
        )
        self.weekly_volume = bt.indicators.SMA(
            self.data.volume, period=1, timeframe=bt.TimeFrame.Weeks
        )
        self.vol_sma50 = bt.indicators.SMA(
            self.data.volume, period=50, timeframe=bt.TimeFrame.Weeks
        )

    def is_weekly_close(self):
        if len(self.data) == self.data.buflen():
            return True
        cur_date = self.data.datetime.date(0)
        next_date = self.data.datetime.date(1)
        return cur_date.isocalendar()[:2] != next_date.isocalendar()[:2]

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            if order.isbuy():
                self.entry_price = order.executed.price
                self.weeks_held = 0
            elif order.issell():
                self.entry_price = None
                self.weeks_held = 0

        self.order = None

    def next(self):
        if self.order:
            return

        if not self.position:
            if self.is_weekly_close():
                cond1 = self.crossover[0] > 0
                cond2 = self.data.close[0] > self.sma50[0]
                cond3 = self.weekly_volume[0] > 1.3 * self.vol_sma50[0]
                cond4 = (self.atr14[0] / self.data.close[0]) < 0.04

                if cond1 and cond2 and cond3 and cond4:
                    self.order = self.buy(size=1)
        else:
            if self.entry_price is None:
                return

            pnl_pct = (
                self.data.close[0] - self.entry_price
            ) / self.entry_price
            if pnl_pct <= -0.08 or pnl_pct >= 0.20:
                self.order = self.close()
                return

            if self.is_weekly_close():
                self.weeks_held += 1
                if self.crossover[0] < 0 or self.weeks_held >= 6:
                    self.order = self.close()
                    return
