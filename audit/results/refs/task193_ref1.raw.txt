import backtrader as bt


class VolatilityTargetingStrategy(bt.Strategy):
    def __init__(self):
        self.atr21 = bt.indicators.ATR(self.data, period=21)
        self.atr5 = bt.indicators.ATR(self.data, period=5)
        self.order = None
        self.last_month = None

    def next(self):
        current_date = self.data.datetime.date(0)
        current_month = (current_date.year, current_date.month)

        if self.last_month == current_month:
            return

        self.last_month = current_month

        if self.order is not None:
            return

        close = self.data.close[0]

        if self.atr5[0] / close > 0.02:
            if self.position:
                self.order = self.close()
            return

        target_weight = min(0.4, 0.01 / (self.atr21[0] / close))
        self.order = self.order_target_percent(target=target_weight)

    def notify_order(self, order):
        if order.status in (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            self.order = None
