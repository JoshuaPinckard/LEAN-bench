import backtrader as bt


class MeanReversionBBStoch(bt.Strategy):
    params = (
        ('bband_period', 20),
        ('bband_dev', 2.0),
        ('stoch_period', 14),
        ('stoch_dfast', 3),
        ('stoch_dslow', 3),
        ('stoch_oversold', 20.0),
        ('stoch_overbought', 80.0),
        ('stop_loss_pct', 0.02),
    )

    def __init__(self):
        self.bband = bt.indicators.BollingerBands(
            self.data,
            period=self.params.bband_period,
            devfactor=self.params.bband_dev
        )
        self.stoch = bt.indicators.Stochastic(
            self.data,
            period=self.params.stoch_period,
            period_dfast=self.params.stoch_dfast,
            period_dslow=self.params.stoch_dslow
        )
        self.order = None
        self.entry_price = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            if self.position.size != 0:
                self.entry_price = order.executed.price
            else:
                self.entry_price = None
            self.order = None
        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        if self.order:
            return

        long_signal = (self.data.low[0] <= self.bband.lines.bot[0]) and (
            self.stoch.lines.percK[0] < self.params.stoch_oversold
        )
        short_signal = (self.data.high[0] >= self.bband.lines.top[0]) and (
            self.stoch.lines.percK[0] > self.params.stoch_overbought
        )

        if self.position.size > 0:
            stop_loss = self.data.low[0] <= self.entry_price * (1.0 - self.params.stop_loss_pct)
            if stop_loss or short_signal:
                self.order = self.close()
        elif self.position.size < 0:
            stop_loss = self.data.high[0] >= self.entry_price * (1.0 + self.params.stop_loss_pct)
            if stop_loss or long_signal:
                self.order = self.close()
        else:
            if long_signal:
                self.order = self.buy()
            elif short_signal:
                self.order = self.sell()
