import math
import backtrader as bt


class NVDABreakoutStrategy(bt.Strategy):
    params = (
        ('ema_period', 8),
        ('breakout_level', 179.33),
        ('tp1_level', 184.89),
        ('tp2_level', 192.95),
        ('risk_pct', 0.02),
    )

    def __init__(self):
        self.ema = bt.indicators.EMA(self.data.close, period=self.p.ema_period)
        self.order = None
        self.stop_price = 0.0
        self.tp1_done = False

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        if self.order:
            return

        if not self.position:
            if self.data.close[0] > self.p.breakout_level:
                entry_price = self.data.close[0]
                init_stop = self.ema[0]
                risk_per_share = entry_price - init_stop

                if risk_per_share > 0:
                    capital = self.broker.getvalue()
                    max_risk = capital * self.p.risk_pct
                    size = int(max_risk / risk_per_share)

                    cash = self.broker.getcash()
                    max_cash_size = int(cash / entry_price)
                    size = min(size, max_cash_size)

                    if size > 0:
                        self.order = self.buy(size=size)
                        self.stop_price = init_stop
                        self.tp1_done = False
        else:
            self.stop_price = max(self.stop_price, self.ema[0])

            if self.data.low[0] <= self.stop_price:
                self.order = self.close()
            elif self.data.high[0] >= self.p.tp2_level:
                self.order = self.close()
            elif not self.tp1_done and self.data.high[0] >= self.p.tp1_level:
                half_size = max(1, self.position.size // 2)
                self.order = self.sell(size=half_size)
                self.tp1_done = True
