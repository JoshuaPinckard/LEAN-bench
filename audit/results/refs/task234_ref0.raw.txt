import backtrader as bt


class RangeVolumeBreakoutStrategy(bt.Strategy):
    params = dict(
        range_period=20,
        volume_period=20,
        volume_multiplier=2.15,
        target_price=28.00,
    )

    def __init__(self):
        # Range/avg-volume are built on data shifted by one bar so today's own
        # high/volume never leak into the breakout thresholds it is compared against.
        self.highest_high = bt.indicators.Highest(
            self.data.high(-1), period=self.p.range_period
        )
        self.avg_volume = bt.indicators.SMA(
            self.data.volume(-1), period=self.p.volume_period
        )
        self.order = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return
        if order.status == order.Completed:
            if order.isbuy():
                # Resting limit order at the fixed target; stays open until
                # filled, matching "remain open until target is reached".
                self.order = self.sell(
                    exectype=bt.Order.Limit, price=self.p.target_price
                )
            else:
                self.order = None
        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        if self.order:
            return
        if not self.position:
            if (self.data.open[0] > self.highest_high[0]
                    and self.data.volume[0] > self.p.volume_multiplier * self.avg_volume[0]):
                size = int(self.broker.getcash() / self.data.open[0])
                if size > 0:
                    self.order = self.buy(size=size)
