import backtrader as bt


class ReferenceStrategy(bt.Strategy):
    def __init__(self):
        self.bb = bt.indicators.BollingerBands(
            self.data.close, period=10, devfactor=0.674
        )
        self.cross = bt.indicators.CrossOver(self.data.close, self.bb.top)
        self.order = None

    def next(self):
        if self.order:
            return

        if not self.position:
            if self.cross > 0:
                self.order = self.order_target_percent(target=1.0)
        elif self.cross < 0 or self.data.close[0] < self.bb.top[0]:
            self.order = self.close()

    def notify_order(self, order):
        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            self.order = None
