import backtrader as bt


class MultiTimeframeRSIStrategy(bt.Strategy):
    params = (("rsi_period", 21),)

    def __init__(self):
        self.rsi_m1 = bt.indicators.RSI(self.data.close, period=self.p.rsi_period)
        self.order = None

        self.m5_bars = []
        self.m15_bars = []

    def next(self):
        dt = self.data.datetime.datetime(0)
        bar = {
            "datetime": dt,
            "open": self.data.open[0],
            "high": self.data.high[0],
            "low": self.data.low[0],
            "close": self.data.close[0],
            "volume": self.data.volume[0],
        }

        self.m5_bars.append(bar)
        self.m15_bars.append(bar)

        if len(self.m5_bars) > 5:
            self.m5_bars.pop(0)
        if len(self.m15_bars) > 15:
            self.m15_bars.pop(0)

        if self.order:
            return

        if self.position:
            if self.position.size > 0 and self.rsi_m1[-1] <= 70 < self.rsi_m1[0]:
                self.order = self.close()
            elif self.position.size < 0 and self.rsi_m1[-1] >= 30 > self.rsi_m1[0]:
                self.order = self.close()
            return

        if self.rsi_m1[0] < 30:
            self.order = self.buy(size=100)
        elif self.rsi_m1[0] > 70:
            self.order = self.sell(size=100)

    def notify_order(self, order):
        if order.status in (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            self.order = None
