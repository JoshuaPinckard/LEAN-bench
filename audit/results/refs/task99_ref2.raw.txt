import backtrader as bt


class AccumulationBreakoutStrategy(bt.Strategy):
    def __init__(self):
        self.order = None
        self.august_started = False
        self.x_vals = []
        self.y_vals = []

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        dt = self.data.datetime.date(0)

        # Track start from the first August encountered in historical data
        if dt.month == 8 and not self.august_started:
            self.august_started = True

        current_x = len(self)
        if self.august_started:
            self.x_vals.append(current_x)
            self.y_vals.append(self.data.low[0])

        # Determine ascending trendline automatically via linear regression on Lows
        trendline_val = None
        is_ascending = False
        n = len(self.x_vals)
        if n >= 2:
            mean_x = sum(self.x_vals) / n
            mean_y = sum(self.y_vals) / n
            num = sum((x - mean_x) * (y - mean_y) for x, y in zip(self.x_vals, self.y_vals))
            den = sum((x - mean_x) ** 2 for x in self.x_vals)
            slope = num / den if den != 0 else 0
            intercept = mean_y - slope * mean_x
            trendline_val = slope * current_x + intercept
            is_ascending = slope > 0

        # Position Exit Logic
        if self.position:
            if self.order is None:
                # Take-profit: Price reaches $240 level
                if self.data.high[0] >= 240.0:
                    self.order = self.close()
                # Stop-loss: Price breaks lower boundary ($165) and closes below it
                elif self.data.close[0] < 165.0:
                    self.order = self.close()
            return

        # Position Entry Logic
        if not self.position and self.order is None and self.august_started and is_ascending and trendline_val is not None:
            # 1. Price touches or enters the accumulation zone ($165 - $170)
            in_zone = (self.data.low[0] <= 170.0) and (self.data.high[0] >= 165.0)

            # 2. Price is above or on the ascending trendline
            above_trendline = self.data.close[0] >= trendline_val

            # 3. Signs of buyer absorption (bullish candle or long lower wick)
            candle_range = self.data.high[0] - self.data.low[0]
            body_bottom = min(self.data.open[0], self.data.close[0])
            lower_wick = body_bottom - self.data.low[0]

            is_bullish = self.data.close[0] > self.data.open[0]
            has_long_lower_wick = (candle_range > 0) and (lower_wick >= 0.33 * candle_range)
            buyer_absorption = is_bullish or has_long_lower_wick

            if in_zone and above_trendline and buyer_absorption:
                # Risk management: Risk no more than 2% of total capital per trade
                close_price = self.data.close[0]
                risk_per_share = close_price - 165.0
                if risk_per_share > 0:
                    total_capital = self.broker.getvalue()
                    max_risk_amount = total_capital * 0.02
                    size = int(max_risk_amount / risk_per_share)

                    cash = self.broker.getcash()
                    max_affordable_size = int(cash / close_price)
                    size = min(size, max_affordable_size)

                    if size > 0:
                        self.order = self.buy(size=size)
