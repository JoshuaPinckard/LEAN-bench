import backtrader as bt


class SupertrendStrategy(bt.Strategy):
    params = (
        ('atr_period', 10),
        ('atr_mult', 3.0),
        ('vol_period', 20),
        ('equity_pct', 0.25),
    )

    def __init__(self):
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.vol_sma = bt.indicators.SMA(self.data.volume, period=self.p.vol_period)

        self.final_upper = None
        self.final_lower = None
        self.trend = None
        self.orders = set()

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.orders.discard(order)

    def next(self):
        if self.orders:
            return

        hl2 = (self.data.high[0] + self.data.low[0]) / 2.0
        atr_val = self.atr[0]
        basic_upper = hl2 + self.p.atr_mult * atr_val
        basic_lower = hl2 - self.p.atr_mult * atr_val

        # Initial bar setup when indicator minperiod is reached
        if self.final_upper is None:
            self.final_upper = basic_upper
            self.final_lower = basic_lower
            if self.data.close[0] > basic_upper:
                self.trend = 1
            elif self.data.close[0] < basic_lower:
                self.trend = -1
            else:
                self.trend = 1
            return

        prior_final_upper = self.final_upper
        prior_final_lower = self.final_lower
        prior_trend = self.trend

        close_prev = self.data.close[-1]
        close_curr = self.data.close[0]

        if basic_upper < prior_final_upper or close_prev > prior_final_upper:
            final_upper = basic_upper
        else:
            final_upper = prior_final_upper

        if basic_lower > prior_final_lower or close_prev < prior_final_lower:
            final_lower = basic_lower
        else:
            final_lower = prior_final_lower

        if close_curr > prior_final_upper:
            trend = 1
        elif close_curr < prior_final_lower:
            trend = -1
        else:
            trend = prior_trend

        self.final_upper = final_upper
        self.final_lower = final_lower
        self.trend = trend

        volume_ok = self.data.volume[0] > self.vol_sma[0]

        # Trade logic
        if self.position.size > 0:
            if trend == -1 and prior_trend == 1:
                o_close = self.close()
                if o_close:
                    self.orders.add(o_close)
                if volume_ok:
                    size = int((self.broker.getvalue() * self.p.equity_pct) / close_curr)
                    if size > 0:
                        o_sell = self.sell(size=size)
                        if o_sell:
                            self.orders.add(o_sell)
        elif self.position.size < 0:
            if trend == 1 and prior_trend == -1:
                o_close = self.close()
                if o_close:
                    self.orders.add(o_close)
                if volume_ok:
                    size = int((self.broker.getvalue() * self.p.equity_pct) / close_curr)
                    if size > 0:
                        o_buy = self.buy(size=size)
                        if o_buy:
                            self.orders.add(o_buy)
        else:
            if trend == 1 and prior_trend == -1:
                if volume_ok:
                    size = int((self.broker.getvalue() * self.p.equity_pct) / close_curr)
                    if size > 0:
                        o_buy = self.buy(size=size)
                        if o_buy:
                            self.orders.add(o_buy)
            elif trend == -1 and prior_trend == 1:
                if volume_ok:
                    size = int((self.broker.getvalue() * self.p.equity_pct) / close_curr)
                    if size > 0:
                        o_sell = self.sell(size=size)
                        if o_sell:
                            self.orders.add(o_sell)
