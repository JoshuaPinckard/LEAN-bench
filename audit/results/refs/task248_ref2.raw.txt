import backtrader as bt


class LiquidityShockStrategy(bt.Strategy):
    def __init__(self):
        # Proxy spread s = (high - low) / close and its SMA(20)
        self.s = (self.data.high - self.data.low) / self.data.close
        self.sma_s_20 = bt.ind.SMA(self.s, period=20)

        # Volume SMAs
        self.sma_vol_10 = bt.ind.SMA(self.data.volume, period=10)
        self.sma_vol_20 = bt.ind.SMA(self.data.volume, period=20)

        # Volatility indicators: ATR(10) and SMA(ATR(10), 50)
        self.atr_10 = bt.ind.ATR(self.data, period=10)
        self.sma_atr_50 = bt.ind.SMA(self.atr_10, period=50)

        # CrossOver indicator between s and SMA(s, 20)
        self.crossover = bt.ind.CrossOver(self.s, self.sma_s_20)

        self.order = None
        self.lots_count = 0

    def notify_order(self, order):
        if order.status in [order.Completed]:
            if not self.position:
                self.lots_count = 0
            else:
                if (order.isbuy() and self.position.size > 0) or (
                    order.issell() and self.position.size < 0
                ):
                    self.lots_count += 1

        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            self.order = None

    def next(self):
        if self.order:
            return

        # Common background conditions
        vol_cond = self.sma_vol_10[0] > 1.5 * self.sma_vol_20[0]
        volatility_cond = self.atr_10[0] < self.sma_atr_50[0]
        ret_cond = (
            abs(self.data.close[0] - self.data.close[-1]) / self.data.close[-1]
            <= 0.001
        )

        # Entry conditions
        long_cond = (
            self.s[0] < 0.8 * self.sma_s_20[0]
            and vol_cond
            and volatility_cond
            and ret_cond
        )
        short_cond = (
            self.s[0] > 1.2 * self.sma_s_20[0]
            and vol_cond
            and volatility_cond
            and ret_cond
        )

        if not self.position:
            if long_cond:
                size = (self.broker.getvalue() * 0.01) / self.data.close[0]
                self.order = self.buy(size=size)
            elif short_cond:
                size = (self.broker.getvalue() * 0.01) / self.data.close[0]
                self.order = self.sell(size=size)
        else:
            pos_price = self.position.price
            if self.position.size > 0:
                pnl_pct = (self.data.close[0] - pos_price) / pos_price
                cross_exit = self.crossover[0] != 0
                tp_exit = pnl_pct >= 0.005
                sl_exit = pnl_pct <= -0.003

                if cross_exit or tp_exit or sl_exit:
                    self.order = self.close()
                elif long_cond and self.lots_count < 3:
                    size = (self.broker.getvalue() * 0.01) / self.data.close[0]
                    self.order = self.buy(size=size)

            elif self.position.size < 0:
                pnl_pct = (pos_price - self.data.close[0]) / pos_price
                cross_exit = self.crossover[0] != 0
                tp_exit = pnl_pct >= 0.005
                sl_exit = pnl_pct <= -0.003

                if cross_exit or tp_exit or sl_exit:
                    self.order = self.close()
                elif short_cond and self.lots_count < 3:
                    size = (self.broker.getvalue() * 0.01) / self.data.close[0]
                    self.order = self.sell(size=size)
