import backtrader as bt


class MACDZeroCrossATRStrategy(bt.Strategy):
    params = dict(
        macd_fast=12,
        macd_slow=26,
        macd_signal=9,
        atr_period=14,
        risk_pct=0.01,      # see assumptions: fixed choice within stated 1-2% range
        sl_atr_mult=1.0,
        tp_atr_mult=3.0,
    )

    def __init__(self):
        self.macd = bt.indicators.MACD(
            self.data.close,
            period_me1=self.p.macd_fast,
            period_me2=self.p.macd_slow,
            period_signal=self.p.macd_signal,
        )
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.order = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return
        if order == self.order:
            if order.status in [order.Completed, order.Canceled,
                                 order.Margin, order.Rejected]:
                self.order = None

    def next(self):
        if self.order:
            return
        if self.position:
            return

        if len(self) < 2:
            return

        macd_now = self.macd.macd[0]
        macd_prev = self.macd.macd[-1]

        cross_up = macd_prev <= 0.0 and macd_now > 0.0
        cross_down = macd_prev >= 0.0 and macd_now < 0.0

        if not (cross_up or cross_down):
            return

        atr_val = self.atr[0]
        if atr_val <= 0:
            return

        capital = self.broker.getvalue()
        risk_amount = capital * self.p.risk_pct
        stop_distance = self.p.sl_atr_mult * atr_val
        size = int(risk_amount / stop_distance)
        if size <= 0:
            return

        entry_price = self.data.close[0]

        if cross_up:
            sl_price = entry_price - self.p.sl_atr_mult * atr_val
            tp_price = entry_price + self.p.tp_atr_mult * atr_val
            orders = self.buy_bracket(
                size=size,
                exectype=bt.Order.Market,
                stopprice=sl_price,
                limitprice=tp_price,
            )
            self.order = orders[0]
        elif cross_down:
            sl_price = entry_price + self.p.sl_atr_mult * atr_val
            tp_price = entry_price - self.p.tp_atr_mult * atr_val
            orders = self.sell_bracket(
                size=size,
                exectype=bt.Order.Market,
                stopprice=sl_price,
                limitprice=tp_price,
            )
            self.order = orders[0]
