import backtrader as bt


class NVDAMeanReversionStrategy(bt.Strategy):
    def __init__(self):
        self.rsi = bt.indicators.RSI(self.data.close, period=14)
        self.sma = bt.indicators.SMA(self.data.close, period=50)
        self.order = None
        self.entry_bar = None

    def next(self):
        if self.order:
            return

        if not self.position:
            if self.data.close[0] < self.sma[0] and self.rsi[0] < 31:
                self.order = self.buy()
        else:
            held_bars = len(self) - self.entry_bar
            if self.rsi[0] > 50 or held_bars >= 4:
                self.order = self.close()

    def notify_order(self, order):
        if order.status == order.Completed:
            if order.isbuy():
                self.entry_bar = len(self)
            elif order.issell():
                self.entry_bar = None
            self.order = None
        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            self.order = None
