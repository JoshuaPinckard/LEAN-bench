import backtrader as bt
import math
import statistics


class AutoregressiveGradientDescentStrategy(bt.Strategy):
    params = (
        ("learning_rate", 0.01),
    )

    def __init__(self):
        self.sma = bt.indicators.SMA(self.data.close, period=50)
        self.rsi = bt.indicators.RSI(self.data.close, period=14)
        self.macd = bt.indicators.MACDHisto(
            self.data.close,
            period_me1=12,
            period_me2=26,
            period_signal=9,
        )

        self.order = None
        self._entry_order_ref = None
        self._exit_order_ref = None
        self.entry_price = None
        self.entry_direction = 0

        self.ar_coefficient = 0.0
        self.ar_intercept = 0.0
        self._last_close = None
        self._previous_return = None
        self.predicted_return = None

        self.volatility_history = []

    def notify_order(self, order):
        if order.status in (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            if order.status == order.Completed:
                if order.ref == self._entry_order_ref:
                    self.entry_price = order.executed.price
                    self.entry_direction = 1 if order.isbuy() else -1
                    self._entry_order_ref = None
                elif order.ref == self._exit_order_ref:
                    self.entry_price = None
                    self.entry_direction = 0
                    self._exit_order_ref = None

            if order.ref == self._entry_order_ref and order.status != order.Completed:
                self._entry_order_ref = None
            if order.ref == self._exit_order_ref and order.status != order.Completed:
                self._exit_order_ref = None

            self.order = None

    def _update_autoregressive_model(self):
        close = self.data.close[0]

        if self._last_close is None:
            self._last_close = close
            return

        current_return = (close / self._last_close) - 1.0

        if self._previous_return is not None:
            model_output = (
                self.ar_coefficient * self._previous_return
                + self.ar_intercept
            )
            error = current_return - model_output
            self.ar_coefficient += (
                self.p.learning_rate * error * self._previous_return
            )
            self.ar_intercept += self.p.learning_rate * error

            self.predicted_return = (
                self.ar_coefficient * current_return
                + self.ar_intercept
            )

        self._previous_return = current_return
        self._last_close = close

    def _histogram_crossed_above_signal(self):
        return (
            self.macd.histo[-1] <= self.macd.signal[-1]
            and self.macd.histo[0] > self.macd.signal[0]
        )

    def _histogram_crossed_below_signal(self):
        return (
            self.macd.histo[-1] >= self.macd.signal[-1]
            and self.macd.histo[0] < self.macd.signal[0]
        )

    def next(self):
        self._update_autoregressive_model()

        current_volatility = statistics.pstdev(
            list(self.data.close.get(size=20))
        )
        historical_volatility = (
            sum(self.volatility_history) / len(self.volatility_history)
            if self.volatility_history
            else None
        )

        if self.order:
            self.volatility_history.append(current_volatility)
            return

        close = self.data.close[0]

        if self.position:
            if self.entry_direction > 0:
                if (
                    close >= self.entry_price * 1.02
                    or close <= self.entry_price * 0.99
                    or self.rsi[0] > 70
                    or self._histogram_crossed_below_signal()
                ):
                    self.order = self.close()
                    self._exit_order_ref = self.order.ref
            elif self.entry_direction < 0:
                if (
                    close <= self.entry_price * 0.98
                    or close >= self.entry_price * 1.01
                    or self.rsi[0] < 30
                    or self._histogram_crossed_above_signal()
                ):
                    self.order = self.close()
                    self._exit_order_ref = self.order.ref

            self.volatility_history.append(current_volatility)
            return

        high_volatility = (
            historical_volatility is None
            or current_volatility > historical_volatility * 1.5
        )

        if not high_volatility and self.predicted_return is not None:
            size = math.floor((self.broker.getvalue() * 0.01) / close)

            if size > 0 and 40 <= self.rsi[0] <= 60:
                if (
                    self.predicted_return > 0
                    and close > self.sma[0]
                    and self._histogram_crossed_above_signal()
                ):
                    self.order = self.buy(size=size)
                    self._entry_order_ref = self.order.ref
                elif (
                    self.predicted_return < 0
                    and close < self.sma[0]
                    and self._histogram_crossed_below_signal()
                ):
                    self.order = self.sell(size=size)
                    self._entry_order_ref = self.order.ref

        self.volatility_history.append(current_volatility)
