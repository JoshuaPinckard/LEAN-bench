import backtrader as bt


class Strategy(bt.Strategy):
    params = (
        ('stop_pct', 0.005),
    )

    def __init__(self):
        self.williams_r = bt.indicators.WilliamsR(self.data, period=14)
        self.sma9 = bt.indicators.SMA(self.data.close, period=9)
        self.sma26 = bt.indicators.SMA(self.data.close, period=26)
        self.sma52 = bt.indicators.SMA(self.data.close, period=52)

        self.williams_cross_down = bt.indicators.CrossDown(self.williams_r, -80.0)
        self.williams_cross_up = bt.indicators.CrossUp(self.williams_r, -20.0)
        self.sma_cross_up = bt.indicators.CrossUp(self.sma9, self.sma26)

        self.order = None
        self.stop_order = None
        self.stop_price = None

    def notify_order(self, order):
        if order.status in (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            if order == self.order:
                self.order = None
            if order == self.stop_order:
                self.stop_order = None

    def next(self):
        cloud_top = max(self.sma9[0], self.sma26[0], self.sma52[0])
        cloud_bot = min(self.sma9[0], self.sma26[0], self.sma52[0])

        if self.order:
            return

        if not self.position:
            self.stop_price = None

            if (
                self.williams_cross_down[0]
                and self.data.close[0] > cloud_top
                and self.sma_cross_up[0]
            ):
                self.order = self.buy()
            elif (
                self.williams_cross_up[0]
                and self.data.close[0] < cloud_bot
                and self.sma9[0] < self.sma26[0]
            ):
                self.order = self.sell()
            return

        if self.position.size > 0:
            if self.data.close[0] < cloud_bot:
                if self.stop_order:
                    self.cancel(self.stop_order)
                    self.stop_order = None

                if self.stop_order:

                self.stop_price = candidate_stop
                self.stop_order = self.sell(
                    exectype=bt.Order.Stop,
                    price=self.stop_price,
                )

        else:
            if self.data.close[0] > cloud_top:
                if self.stop_order:
                    self.cancel(self.stop_order)
                    self.stop_order = None
                self.order = self.close()
                return

            candidate_stop = self.sma26[0] * (1.0 + self.p.stop_pct)
            if self.stop_price is None or candidate_stop < self.stop_price:
