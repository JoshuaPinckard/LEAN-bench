class BollingerEMATrendStrategy(bt.Strategy):
    params = dict(
        bb_period=20,
        bb_devfactor=2,
        ema_period=50,
    )

    def __init__(self):
        self.boll = bt.indicators.BollingerBands(
            self.data.close, period=self.p.bb_period, devfactor=self.p.bb_devfactor
        )
        self.ema = bt.indicators.EMA(self.data.close, period=self.p.ema_period)

        self.cross_upper = bt.indicators.CrossOver(self.data.close, self.boll.lines.top)
        self.cross_lower = bt.indicators.CrossOver(self.data.close, self.boll.lines.bot)
        self.cross_ema = bt.indicators.CrossOver(self.data.close, self.ema)

        self.order = None

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        if self.order:
            return

        long_signal = self.cross_upper[0] > 0 and self.data.close[0] > self.ema[0]
        short_signal = self.cross_lower[0] < 0 and self.data.close[0] < self.ema[0]

        if self.position:
            if self.position.size > 0:
                if short_signal:
                    self.close()
                    self.order = self.sell()
                elif self.cross_ema[0] < 0:
                    self.order = self.close()
            else:
                if long_signal:
                    self.close()
                    self.order = self.buy()
                elif self.cross_ema[0] > 0:
                    self.order = self.close()
        else:
            if long_signal:
                self.order = self.buy()
            elif short_signal:
                self.order = self.sell()
