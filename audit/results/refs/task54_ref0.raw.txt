import backtrader as bt
import collections
import statistics


class MultiTimeframeRSIStrategy(bt.Strategy):
    params = dict(
        rsi_period=21,
        rsi_upper=70.0,
        rsi_lower=30.0,
        trade_size=100,
    )

    def __init__(self):
        self.data_m1 = self.datas[0]
        self.rsi_m1 = bt.indicators.RSI(self.data_m1.close, period=self.p.rsi_period)

        self.order = None

        self._m5_bar_count = 0
        self._m5_closes = collections.deque(maxlen=self.p.rsi_period + 1)
        self.rsi_m5 = None

        self._m15_bar_count = 0
        self._m15_closes = collections.deque(maxlen=self.p.rsi_period + 1)
        self.rsi_m15 = None

    def _manual_rsi(self, closes):
        prices = list(closes)
        if len(prices) < self.p.rsi_period + 1:
            return None
        gains = []
        losses = []
        for i in range(1, len(prices)):
            change = prices[i] - prices[i - 1]
            if change > 0:
                gains.append(change)
                losses.append(0.0)
            else:
                gains.append(0.0)
                losses.append(-change)
        avg_gain = statistics.mean(gains[-self.p.rsi_period:])
        avg_loss = statistics.mean(losses[-self.p.rsi_period:])
        if avg_loss == 0:
            return 100.0
        rs = avg_gain / avg_loss
        return 100.0 - (100.0 / (1.0 + rs))

    def _update_multi_timeframe(self):
        self._m5_bar_count += 1
        if self._m5_bar_count % 5 == 0:
            self._m5_closes.append(self.data_m1.close[0])
            self.rsi_m5 = self._manual_rsi(self._m5_closes)

        self._m15_bar_count += 1
        if self._m15_bar_count % 15 == 0:
            self._m15_closes.append(self.data_m1.close[0])
            self.rsi_m15 = self._manual_rsi(self._m15_closes)

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        self._update_multi_timeframe()

        if self.order:
            return

        rsi_now = self.rsi_m1[0]
        rsi_prev = self.rsi_m1[-1]

        crossed_above_upper = rsi_prev <= self.p.rsi_upper and rsi_now > self.p.rsi_upper
        crossed_below_lower = rsi_prev >= self.p.rsi_lower and rsi_now < self.p.rsi_lower

        if not self.position:
            if rsi_now < self.p.rsi_lower:
                self.order = self.buy(size=self.p.trade_size)
            elif rsi_now > self.p.rsi_upper:
                self.order = self.sell(size=self.p.trade_size)
        else:
            if self.position.size > 0 and crossed_above_upper:
                self.order = self.close()
            elif self.position.size < 0 and crossed_below_lower:
                self.order = self.close()
