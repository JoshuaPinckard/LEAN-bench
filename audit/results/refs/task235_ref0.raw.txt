import backtrader as bt


class MeanReversionFalseBreakout(bt.Strategy):
    params = dict(
        upper_level=450.00,
        lower_level=445.00,
        atr_period=14,
        atr_sma_period=50,
        rsi_period=14,
        bb_period=20,
        bb_devfactor=2.0,
        trade_size=2,
        short_sl_price=450.25,
        short_tp_zone=447.5,
        short_tp_close=446.0,
        long_sl_price=444.75,
        long_tp_high=449.5,
        long_tp_close=449.0,
        trail_trigger_mult=0.75,
        trail_step_mult=0.5,
    )

    def __init__(self):
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.atr_avg = bt.indicators.SMA(self.atr, period=self.p.atr_sma_period)
        self.rsi = bt.indicators.RSI(self.data, period=self.p.rsi_period)
        self.bb = bt.indicators.BollingerBands(
            self.data, period=self.p.bb_period, devfactor=self.p.bb_devfactor
        )

        self.entry_order = None
        self.stop_order = None
        self.stale_stop_refs = set()

        self.direction = None
        self.entry_price = None
        self.stop_price = None
        self.bars_in_trade = 0
        self.last_entry_date = None

    def _reset_trade_state(self):
        self.direction = None
        self.entry_price = None
        self.stop_price = None
        self.stop_order = None
        self.bars_in_trade = 0

    def _replace_stop(self, new_price, is_long):
        if self.stop_order is not None:
            self.stale_stop_refs.add(self.stop_order.ref)
            self.cancel(self.stop_order)
        self.stop_price = new_price
        if is_long:
            self.stop_order = self.sell(
                size=self.p.trade_size, exectype=bt.Order.Stop, price=new_price
            )
        else:
            self.stop_order = self.buy(
                size=self.p.trade_size, exectype=bt.Order.Stop, price=new_price
            )

    def _close_position(self):
        if self.stop_order is not None:
            self.stale_stop_refs.add(self.stop_order.ref)
            self.cancel(self.stop_order)
        self.close()
        self._reset_trade_state()

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return

        if order.ref in self.stale_stop_refs:
            self.stale_stop_refs.discard(order.ref)
            return

        if order is self.entry_order:
            if order.status == order.Completed:
                self.entry_price = order.executed.price
                is_long = self.direction == 'long'
                initial_stop = (
                    self.p.long_sl_price if is_long else self.p.short_sl_price
                )
                self.stop_price = initial_stop
                if is_long:
                    self.stop_order = self.sell(
                        size=self.p.trade_size,
                        exectype=bt.Order.Stop,
                        price=initial_stop,
                    )
                else:
                    self.stop_order = self.buy(
                        size=self.p.trade_size,
                        exectype=bt.Order.Stop,
                        price=initial_stop,
                    )
                self.bars_in_trade = 0
            else:
                self._reset_trade_state()
            self.entry_order = None
            return

        if order is self.stop_order:
            if order.status == order.Completed:
                self._reset_trade_state()
            elif order.status in (order.Canceled, order.Margin, order.Rejected):
                self.stop_order = None
            return

    def next(self):
        dt = self.data.datetime.date(0)
        high = self.data.high[0]
        low = self.data.low[0]
        close = self.data.close[0]

        if self.position:
            if self.direction == 'long':
                profit = close - self.entry_price
                if profit >= self.p.trail_trigger_mult * self.atr[0]:
                    candidate = close - self.p.trail_step_mult * self.atr[0]
                    if candidate > self.stop_price:
                        self._replace_stop(candidate, is_long=True)

                tp_hit = high >= self.p.long_tp_high or close >= self.p.long_tp_close
            else:
                profit = self.entry_price - close
                if profit >= self.p.trail_trigger_mult * self.atr[0]:
                    candidate = close + self.p.trail_step_mult * self.atr[0]
                    if candidate < self.stop_price:
                        self._replace_stop(candidate, is_long=False)

                tp_hit = (
                    low <= self.p.short_tp_zone
                    or close <= self.p.short_tp_close
                    or low <= self.bb.mid[0]
                )

            if tp_hit:
                self._close_position()
                return

            if self.bars_in_trade >= 1:
                self._close_position()
                return

            self.bars_in_trade += 1
            return

        if self.entry_order is not None:
            return

        if self.last_entry_date == dt:
            return

        if self.atr[0] <= self.atr_avg[0]:
            return

        if high > self.p.upper_level and close < self.p.upper_level:
            if self.rsi[0] > 60 and high >= self.bb.top[0]:
                self.direction = 'short'
                self.last_entry_date = dt
                self.entry_order = self.sell(size=self.p.trade_size)
                return

        if low < self.p.lower_level and close > self.p.lower_level:
            if self.rsi[0] < 40 and low <= self.bb.bot[0]:
                self.direction = 'long'
                self.last_entry_date = dt
                self.entry_order = self.buy(size=self.p.trade_size)
                return
