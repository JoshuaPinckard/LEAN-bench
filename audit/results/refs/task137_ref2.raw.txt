import math
import backtrader as bt


class SmaCrossoverStrategy(bt.Strategy):
    params = (
        ('fast_period', 10),
        ('slow_period', 50),
    )

    def __init__(self):
        self.fast_sma = bt.indicators.SMA(self.data.close, period=self.params.fast_period)
        self.slow_sma = bt.indicators.SMA(self.data.close, period=self.params.slow_period)
        self.signal = self.fast_sma > self.slow_sma
        self.order = None

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        if self.order:
            return

        if math.isnan(self.signal[0]) or math.isnan(self.signal[-1]):
            return

        diff = self.signal[0] - self.signal[-1]

        if diff == 1.0:
            if not self.position:
                self.order = self.buy()
        elif diff == -1.0:
            if self.position:
                self.order = self.close()
