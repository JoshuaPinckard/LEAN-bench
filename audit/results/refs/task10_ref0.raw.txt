import backtrader as bt


class OpeningRangeBreakoutStrategy(bt.Strategy):
    SESSION_STARTS = {
        'Tokyo': (9, 0),
        'London': (8, 0),
        'New York': (9, 30),
    }

    params = (
        ('session', 'New York'),
        ('range_minutes', 120),
        ('tp_multiple', 1.5),
        ('risk_pct', 0.01),
        ('vol_filter_pct', 0.002),
    )

    def __init__(self):
        start_hour, start_minute = self.SESSION_STARTS[self.p.session]
        self.session_start_min = start_hour * 60 + start_minute
        self.range_end_min = self.session_start_min + self.p.range_minutes

        self.current_day = None
        self.range_high = None
        self.range_low = None
        self.zone_ready = False
        self.zone_valid = False
        self.zone_upper = None
        self.zone_lower = None

        self.order = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            if order == self.order:
                self.order = None

    def next(self):
        dt = self.data.datetime.datetime(0)
        bar_date = dt.date()
        bar_min = dt.hour * 60 + dt.minute

        if bar_date != self.current_day:
            self.current_day = bar_date
            self.range_high = None
            self.range_low = None
            self.zone_ready = False
            self.zone_valid = False
            self.zone_upper = None
            self.zone_lower = None

        high = self.data.high[0]
        low = self.data.low[0]
        close = self.data.close[0]

        if self.session_start_min <= bar_min < self.range_end_min:
            self.range_high = high if self.range_high is None else max(self.range_high, high)
            self.range_low = low if self.range_low is None else min(self.range_low, low)
            return

        if bar_min < self.session_start_min:
            return

        if not self.zone_ready:
            if self.range_high is None or self.range_low is None:
                return
            self.zone_upper = self.range_high
            self.zone_lower = self.range_low
            zone_range = self.zone_upper - self.zone_lower
            self.zone_valid = zone_range >= (close * self.p.vol_filter_pct)
            self.zone_ready = True

        if not self.zone_valid:
            return

        if self.order or self.position:
            return

        zone_range = self.zone_upper - self.zone_lower

        if close > self.zone_upper:
            stop_price = self.zone_lower
            take_profit = self.zone_upper + self.p.tp_multiple * zone_range
            risk_per_share = close - stop_price
            if risk_per_share <= 0:
                return
            size = int((self.broker.getvalue() * self.p.risk_pct) / risk_per_share)
            if size <= 0:
                return
            orders = self.buy_bracket(
                size=size,
                price=close,
                exectype=bt.Order.Market,
                stopprice=stop_price,
                limitprice=take_profit,
            )
            self.order = orders[0]

        elif close < self.zone_lower:
            stop_price = self.zone_upper
            take_profit = self.zone_lower - self.p.tp_multiple * zone_range
            risk_per_share = stop_price - close
            if risk_per_share <= 0:
                return
            size = int((self.broker.getvalue() * self.p.risk_pct) / risk_per_share)
            if size <= 0:
                return
            orders = self.sell_bracket(
                size=size,
                price=close,
                exectype=bt.Order.Market,
                stopprice=stop_price,
                limitprice=take_profit,
            )
            self.order = orders[0]
