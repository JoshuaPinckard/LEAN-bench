import backtrader as bt
import math


class RSITradingSessionsStrategy(bt.Strategy):
    def __init__(self):
        self.rsi = bt.indicators.RSI(self.data.close, period=14)
        self.order = None

    def _active_session(self):
        dt = self.data.datetime.datetime(0)
        minutes = dt.hour * 60 + dt.minute
        return dt.weekday() < 5 and 570 <= minutes < 960

    def _size(self):
        commission = self.broker.getcommissioninfo(self.data).p.commission
        return math.floor(self.broker.getcash() / (self.data.close[0] * (1.0 + commission)))

    def next(self):
        if not self._active_session() or self.order:
            return

        if self.position.size > 0:
            if self.rsi[0] > 70:
                self.order = self.close()
            return

        if self.position.size < 0:
            if self.rsi[0] < 30:
                self.order = self.close()
            return

        size = self._size()
        if size <= 0:
            return

        if self.rsi[-1] < 30 and self.rsi[0] > 30:
            self.order = self.buy(size=size)
        elif self.rsi[-1] > 70 and self.rsi[0] < 70:
            self.order = self.sell(size=size)

    def notify_order(self, order):
        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            self.order = None
