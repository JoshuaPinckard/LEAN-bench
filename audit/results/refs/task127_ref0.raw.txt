import backtrader as bt
import statistics


class MSTRSignalConflictStrategy(bt.Strategy):
    def __init__(self):
        self.sma365 = bt.indicators.SMA(self.data.close, period=365)
        self.sma200 = bt.indicators.SMA(self.data.close, period=200)
        self.sma10 = bt.indicators.SMA(self.data.close, period=10)
        self.cross365 = bt.indicators.CrossOver(self.data.close, self.sma365)
        self.sma365_history = []
        self.order = None

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        self.sma365_history.append(self.sma365[0])

        if self.order:
            return

        if not self.position:
            if len(self.sma365_history) < 2:
                return
            threshold = statistics.quantiles(self.sma365_history, n=100, method='inclusive')[43]
            if self.cross365[0] > 0 and self.sma365[0] > threshold:
                self.order = self.buy()
        else:
            if self.cross365[0] < 0 or self.data.close[0] < self.sma200[0]:
                self.order = self.close()
