import math
import backtrader as bt


class SyntheticHedgeStrategy(bt.Strategy):
    params = dict(
        sma_period=50,
        atr_period=14,
        bb_long_period=50,
        bb_short_period=20,
        bb_devfactor=2.0,
        macd_me1=12,
        macd_me2=26,
        macd_signal=9,
        atr_vol_threshold=0.005,
        hedge_min=0.0,
        hedge_max=1.5,
        notional=100000.0,
        sigma_eps=1e-9,
    )

    def __init__(self):
        self.sma = bt.indicators.SMA(self.data.close, period=self.p.sma_period)
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.bb_long = bt.indicators.BollingerBands(
            self.data.close, period=self.p.bb_long_period, devfactor=self.p.bb_devfactor
        )
        self.bb_short = bt.indicators.BollingerBands(
            self.data.close, period=self.p.bb_short_period, devfactor=self.p.bb_devfactor
        )
        self.macd = bt.indicators.MACD(
            self.data.close,
            period_me1=self.p.macd_me1,
            period_me2=self.p.macd_me2,
            period_signal=self.p.macd_signal,
        )
        self.crossover = bt.indicators.CrossOver(self.data.close, self.sma)

        self.order = None

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return
        if order.status in (order.Completed, order.Canceled, order.Margin, order.Rejected):
            self.order = None

    def _hedge_coefficient(self):
        sigma_long = (self.bb_long.top[0] - self.bb_long.mid[0]) / 2.0
        sigma_short = (self.bb_short.top[0] - self.bb_short.mid[0]) / 2.0
        if abs(sigma_short) < self.p.sigma_eps:
            return 0.0
        ratio = sigma_long / sigma_short
        return max(self.p.hedge_min, min(self.p.hedge_max, ratio))

    def _target_units_net(self):
        close = self.data.close[0]
        units_main = math.floor(self.p.notional / close)
        h = self._hedge_coefficient()
        units_hedge = math.floor(units_main * h)
        return units_main - units_hedge

    def next(self):
        if self.order:
            return

        close = self.data.close[0]
        atr_ratio = self.atr[0] / close
        macd_val = self.macd.macd[0]
        signal_val = self.macd.signal[0]

        vol_ok = atr_ratio >= self.p.atr_vol_threshold
        long_trend_ok = macd_val > signal_val > 0
        short_trend_ok = macd_val < signal_val < 0

        pos_size = self.position.size

        if pos_size > 0:
            if self.crossover[0] < 0:
                self.order = self.close()
                return
            if long_trend_ok and vol_ok:
                target = self._target_units_net()
                self.order = self.order_target_size(target=target)
        elif pos_size < 0:
            if self.crossover[0] > 0:
                self.order = self.close()
                return
            if short_trend_ok and vol_ok:
                target = -self._target_units_net()
                self.order = self.order_target_size(target=target)
        else:
            if self.crossover[0] > 0 and long_trend_ok and vol_ok:
                target = self._target_units_net()
                self.order = self.order_target_size(target=target)
            elif self.crossover[0] < 0 and short_trend_ok and vol_ok:
                target = -self._target_units_net()
                self.order = self.order_target_size(target=target)
