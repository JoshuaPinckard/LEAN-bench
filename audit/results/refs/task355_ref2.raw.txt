import backtrader as bt
import math


class IntradayOutsideBarReversal(bt.Strategy):
    params = (
        ("n_decline", 6),
        ("tick_size", 0.01),
        ("risk_pct", 0.01),
    )

    def __init__(self):
        self.order = None
        self.stop_order = None
        self.entry_bar_low = None
        self.entry_bar_high = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order == self.order:
            if order.status in [
                order.Completed,
                order.Canceled,
                order.Margin,
                order.Rejected,
            ]:
                self.order = None

        if order == self.stop_order:
            if order.status in [
                order.Completed,
                order.Canceled,
                order.Margin,
                order.Rejected,
            ]:
                self.stop_order = None

    def next(self):
        if len(self.data) <= self.p.n_decline:
            return

        # Check regular US market hours (09:30 to 16:00)
        t = self.data.datetime.time(0)
        in_market_hours = (t.hour > 9 or (t.hour == 9 and t.minute >= 30)) and (
            t.hour < 16
        )

        # Handle exits for open positions
        if self.position:
            if self.position.size > 0:
                if (
                    self.entry_bar_low is not None
                    and self.data.close[0] < self.entry_bar_low
                ):
                    if self.stop_order and self.stop_order.status in [
                        bt.Order.Submitted,
                        bt.Order.Accepted,
                    ]:
                        self.cancel(self.stop_order)
                        self.stop_order = None
                    self.order = self.close()
            elif self.position.size < 0:
                if (
                    self.entry_bar_high is not None
                    and self.data.close[0] > self.entry_bar_high
                ):
                    if self.stop_order and self.stop_order.status in [
                        bt.Order.Submitted,
                        bt.Order.Accepted,
                    ]:
                        self.cancel(self.stop_order)
                        self.stop_order = None
                    self.order = self.close()
            return

        # Do not enter if an order is currently pending or outside market hours
        if self.order is not None or not in_market_hours:
            return

        # Outside Bar definition
        is_outside = (self.data.high[0] > self.data.high[-1]) and (
            self.data.low[0] < self.data.low[-1]
        )
        is_bullish_outside = is_outside and (self.data.close[0] > self.data.open[0])
        is_bearish_outside = is_outside and (self.data.close[0] < self.data.open[0])

        # Trend context filters
        after_decline = self.data.close[0] < self.data.close[-self.p.n_decline]
        after_rally = self.data.close[0] > self.data.close[-self.p.n_decline]

        portfolio_value = self.broker.getvalue()

        # Long Entry
        if is_bullish_outside and after_decline:
            entry_price = self.data.close[0]
            stop_price = self.data.low[0] - self.p.tick_size
            risk_per_share = entry_price - stop_price
            if risk_per_share > 0:
                risk_amount = self.p.risk_pct * portfolio_value
                size = math.floor(risk_amount / risk_per_share)
                if size > 0:
                    orders = self.buy_bracket(size=size, stopprice=stop_price)
                    self.order = orders[0]
                    self.stop_order = orders[1]
                    self.entry_bar_low = self.data.low[0]
                    self.entry_bar_high = self.data.high[0]

        # Short Entry
        elif is_bearish_outside and after_rally:
            entry_price = self.data.close[0]
            stop_price = self.data.high[0] + self.p.tick_size
            risk_per_share = stop_price - entry_price
            if risk_per_share > 0:
                risk_amount = self.p.risk_pct * portfolio_value
                size = math.floor(risk_amount / risk_per_share)
                if size > 0:
                    orders = self.sell_bracket(size=size, stopprice=stop_price)
                    self.order = orders[0]
                    self.stop_order = orders[1]
                    self.entry_bar_low = self.data.low[0]
                    self.entry_bar_high = self.data.high[0]
