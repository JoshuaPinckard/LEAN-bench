import backtrader as bt


class IntradayMeanReversion(bt.Strategy):
    params = (
        ("bb_period", 20),
        ("bb_dev", 2.0),
        ("rsi_period", 14),
        ("atr_period", 14),
        ("rsi_upper", 70.0),
        ("rsi_lower", 30.0),
        ("atr_mult", 2.0),
    )

    def __init__(self):
        self.bb = bt.indicators.BollingerBands(
            self.data.close, period=self.p.bb_period, devfactor=self.p.bb_dev
        )
        self.rsi = bt.indicators.RSI(
            self.data.close, period=self.p.rsi_period
        )
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)

        self.order = None
        self.entry_price = None
        self.stop_price = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            if order.isbuy():
                if self.position.size > 0:
                    self.entry_price = order.executed.price
                    self.stop_price = self.entry_price - (
                        self.p.atr_mult * self.atr[0]
                    )
                else:
                    self.entry_price = None
                    self.stop_price = None
            elif order.issell():
                if self.position.size < 0:
                    self.entry_price = order.executed.price
                    self.stop_price = self.entry_price + (
                        self.p.atr_mult * self.atr[0]
                    )
                else:
                    self.entry_price = None
                    self.stop_price = None

        self.order = None

    def next(self):
        if self.order:
            return

        if not self.position:
            if self.data.close[0] < self.bb.bot[0] and self.rsi[0] < self.p.rsi_lower:
                self.order = self.buy()
            elif self.data.close[0] > self.bb.top[0] and self.rsi[0] > self.p.rsi_upper:
                self.order = self.sell()
        else:
            if self.position.size > 0:
                if self.stop_price is not None and self.data.low[0] <= self.stop_price:
                    self.order = self.close()
                elif self.bb.bot[0] <= self.data.close[0] <= self.bb.top[0]:
                    self.order = self.close()
                elif self.data.close[0] > self.bb.top[0] and self.rsi[0] > self.p.rsi_upper:
                    self.order = self.close()

            elif self.position.size < 0:
                if self.stop_price is not None and self.data.high[0] >= self.stop_price:
                    self.order = self.close()
                elif self.bb.bot[0] <= self.data.close[0] <= self.bb.top[0]:
                    self.order = self.close()
                elif self.data.close[0] < self.bb.bot[0] and self.rsi[0] < self.p.rsi_lower:
                    self.order = self.close()
