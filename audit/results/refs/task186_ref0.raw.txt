import backtrader as bt


class SentimentProxyWeakSignalAmplifier(bt.Strategy):
    params = dict(
        rsi_period=14,
        rsi_lookback=3,
        macd_fast=12,
        macd_slow=26,
        macd_signal=9,
        bb_period=20,
        bb_devfactor=2.0,
        atr_period=14,
        atr_sma_period=100,
        vol_sma_period=20,
        take_profit_pct=0.05,
        stop_loss_pct=0.03,
        rsi_reversal_pts=10.0,
        risk_pct=0.01,
        high_vol_multiplier=1.5,
        gap_atr_mult=0.5,
        gap_vol_mult=1.5,
        delta_rsi_norm=15.0,
        delta_rsi_high=10.0,
        high_vol_volume_mult=1.3,
    )

    def __init__(self):
        self.rsi = bt.indicators.RSI(self.data.close, period=self.p.rsi_period)
        self.macd = bt.indicators.MACD(
            self.data.close,
            period_me1=self.p.macd_fast,
            period_me2=self.p.macd_slow,
            period_signal=self.p.macd_signal,
        )
        self.crossover = bt.indicators.CrossOver(self.macd.macd, self.macd.signal)
        self.bb = bt.indicators.BollingerBands(
            self.data.close, period=self.p.bb_period, devfactor=self.p.bb_devfactor
        )
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.atr_sma = bt.indicators.SMA(self.atr, period=self.p.atr_sma_period)
        self.vol_sma = bt.indicators.SMA(self.data.volume, period=self.p.vol_sma_period)

        self.entry_order = None
        self.stop_order = None
        self.limit_order = None
        self.exit_order = None
        self.entry_price = None
        self.rsi_extreme = None

    def _cancel_bracket(self):
        if self.stop_order is not None:
            self.cancel(self.stop_order)
            self.stop_order = None
        if self.limit_order is not None:
            self.cancel(self.limit_order)
            self.limit_order = None

    def _calc_size(self, price):
        if price <= 0:
            return 0
        cash_alloc = self.broker.getvalue() * self.p.risk_pct
        return int(cash_alloc / price)

    def _manage_position(self):
        if self.exit_order is not None:
            return
        rsi0 = self.rsi[0]
        if self.position.size > 0:
            if self.rsi_extreme is None or rsi0 > self.rsi_extreme:
                self.rsi_extreme = rsi0
            if self.rsi_extreme - rsi0 >= self.p.rsi_reversal_pts:
                self._cancel_bracket()
                self.exit_order = self.close()
        else:
            if self.rsi_extreme is None or rsi0 < self.rsi_extreme:
                self.rsi_extreme = rsi0
            if rsi0 - self.rsi_extreme >= self.p.rsi_reversal_pts:
                self._cancel_bracket()
                self.exit_order = self.close()

    def next(self):
        if self.position:
            self._manage_position()
            return

        if self.entry_order is not None:
            return

        close0 = self.data.close[0]
        delta_rsi = self.rsi[0] - self.rsi[-self.p.rsi_lookback]
        high_vol = self.atr[0] > self.p.high_vol_multiplier * self.atr_sma[0]

        bb_top = self.bb.top[0]
        bb_bot = self.bb.bot[0]
        bb_mid = self.bb.mid[0]
        bb_width = bb_top - bb_bot

        no_upper_break = all(
            self.data.close[-i] <= self.bb.top[-i] for i in range(0, 3)
        )
        not_above_mid_half = close0 <= bb_mid + 0.5 * bb_width
        price_not_reacted_long = no_upper_break and not_above_mid_half

        no_lower_break = all(
            self.data.close[-i] >= self.bb.bot[-i] for i in range(0, 3)
        )
        not_below_mid_half = close0 >= bb_mid - 0.5 * bb_width
        price_not_reacted_short = no_lower_break and not_below_mid_half

        if high_vol:
            volume_confirm = self.data.volume[0] > self.p.high_vol_volume_mult * self.vol_sma[0]
            signal_general_long = (
                delta_rsi >= self.p.delta_rsi_high
                and self.crossover[0] == 1
                and volume_confirm
            )
            signal_general_short = (
                delta_rsi <= -self.p.delta_rsi_high
                and self.crossover[0] == -1
                and volume_confirm
            )
        else:
            signal_general_long = delta_rsi >= self.p.delta_rsi_norm
            signal_general_short = delta_rsi <= -self.p.delta_rsi_norm

        entry_a_long = signal_general_long and price_not_reacted_long
        entry_a_short = signal_general_short and price_not_reacted_short

        gap_long = (
            self.data.open[0] > self.data.high[-1] + self.p.gap_atr_mult * self.atr[0]
            and self.data.volume[0] > self.p.gap_vol_mult * self.vol_sma[0]
        )
        gap_short = (
            self.data.open[0] < self.data.low[-1] - self.p.gap_atr_mult * self.atr[0]
            and self.data.volume[0] > self.p.gap_vol_mult * self.vol_sma[0]
        )

        entry_long = entry_a_long or gap_long
        entry_short = entry_a_short or gap_short

        if entry_long and not entry_short:
            size = self._calc_size(close0)
            if size > 0:
                self.entry_order = self.buy(size=size)
        elif entry_short and not entry_long:
            size = self._calc_size(close0)
            if size > 0:
                self.entry_order = self.sell(size=size)

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return

        if order.status == order.Completed:
            if order is self.entry_order:
                self.entry_order = None
                self.entry_price = order.executed.price
                self.rsi_extreme = self.rsi[0]
                size = abs(order.executed.size)
                if order.isbuy():
                    self.stop_order = self.sell(
                        exectype=bt.Order.Stop,
                        price=self.entry_price * (1.0 - self.p.stop_loss_pct),
                        size=size,
                    )
                    self.limit_order = self.sell(
                        exectype=bt.Order.Limit,
                        price=self.entry_price * (1.0 + self.p.take_profit_pct),
                        size=size,
                    )
                else:
                    self.stop_order = self.buy(
                        exectype=bt.Order.Stop,
                        price=self.entry_price * (1.0 + self.p.stop_loss_pct),
                        size=size,
                    )
                    self.limit_order = self.buy(
                        exectype=bt.Order.Limit,
                        price=self.entry_price * (1.0 - self.p.take_profit_pct),
                        size=size,
                    )
            elif order is self.stop_order:
                self.stop_order = None
                if self.limit_order is not None:
                    self.cancel(self.limit_order)
                    self.limit_order = None
                self.entry_price = None
                self.rsi_extreme = None
            elif order is self.limit_order:
                self.limit_order = None
                if self.stop_order is not None:
                    self.cancel(self.stop_order)
                    self.stop_order = None
                self.entry_price = None
                self.rsi_extreme = None
            elif order is self.exit_order:
                self.exit_order = None
                self.entry_price = None
                self.rsi_extreme = None

        elif order.status in (order.Canceled, order.Margin, order.Rejected):
            if order is self.entry_order:
                self.entry_order = None
            elif order is self.stop_order:
                self.stop_order = None
            elif order is self.limit_order:
                self.limit_order = None
            elif order is self.exit_order:
                self.exit_order = None
