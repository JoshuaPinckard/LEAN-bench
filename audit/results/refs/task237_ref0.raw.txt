import backtrader as bt


class MeanReversionStrategy(bt.Strategy):
    def __init__(self):
        self.bb = bt.indicators.BollingerBands(self.data.close, period=20, devfactor=2)
        self.vol_sma20 = bt.indicators.SMA(self.data.volume, period=20)
        self.vol_sma50 = bt.indicators.SMA(self.data.volume, period=50)
        self.rsi = bt.indicators.RSI(self.data.close, period=14)
        self.crossover = bt.indicators.CrossOver(self.data.close, self.bb.lines.mid)

        self.order = None
        self.bars_in_trade = 0
        self.day_start_value = None

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        if self.day_start_value is None:
            self.day_start_value = self.broker.getvalue()

        portfolio_value = self.broker.getvalue()
        daily_loss_hit = (portfolio_value - self.day_start_value) <= -0.005 * self.day_start_value

        if self.position:
            self.bars_in_trade += 1
            if self.crossover[0] > 0 or self.bars_in_trade >= 3:
                self.order = self.close()
        elif not self.order and not daily_loss_hit:
            if (self.data.close[0] < self.bb.lines.bot[0]
                    and self.data.volume[0] > 1.5 * self.vol_sma20[0]
                    and self.rsi[0] < 30
                    and self.vol_sma50[0] > 1000000):
                self.order = self.buy(size=100)
                self.bars_in_trade = 0

        self.day_start_value = portfolio_value
