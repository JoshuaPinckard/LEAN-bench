import backtrader as bt


class ConsolidationBreakout(bt.Strategy):
    def __init__(self):
        self.boll = bt.indicators.BollingerBands(
            self.data.close, period=20, devfactor=2.0
        )
        self.sma_vol = bt.indicators.SimpleMovingAverage(
            self.data.volume, period=20
        )
        self.rsi = bt.indicators.RSI(self.data.close, period=14)

        self.order = None
        self.entry_price = None
        self.entry_bar = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            if self.position:
                self.entry_price = order.executed.price
                self.entry_bar = len(self)
            else:
                self.entry_price = None
                self.entry_bar = None

        self.order = None

    def next(self):
        if self.order:
            return

        if not self.position:
            # Check 10-bar consolidation prior to current breakout bar
            is_consolidating = all(
                self.data.high[i] <= self.boll.top[i]
                and self.data.low[i] >= self.boll.bot[i]
                for i in range(-10, 0)
            )

            # Volume requirement: current volume 50% higher than prior 20-bar average
            vol_breakout = self.data.volume[0] > 1.5 * self.sma_vol[-1]

            if is_consolidating and vol_breakout:
                size = (self.broker.getvalue() * 0.01) / self.data.close[0]
                if self.data.close[0] > self.boll.top[0] and self.rsi[0] > 50:
                    self.order = self.buy(size=size)
                elif self.data.close[0] < self.boll.bot[0] and self.rsi[0] < 50:
                    self.order = self.sell(size=size)
        else:
            if self.entry_price is None:
                return

            bars_since_entry = len(self) - self.entry_bar

            if self.position.size > 0:
                sl_hit = self.data.low[0] <= self.entry_price * 0.985
                tp_hit = self.data.high[0] >= self.entry_price * 1.02
                mid_hit = self.data.close[0] < self.boll.mid[0]
                quick_return = (bars_since_entry <= 3) and (
                    self.data.close[0] <= self.boll.top[0]
                )

                if sl_hit or tp_hit or mid_hit or quick_return:
                    self.order = self.close()

            elif self.position.size < 0:
                sl_hit = self.data.high[0] >= self.entry_price * 1.015
                tp_hit = self.data.low[0] <= self.entry_price * 0.98
                mid_hit = self.data.close[0] > self.boll.mid[0]
                quick_return = (bars_since_entry <= 3) and (
                    self.data.close[0] >= self.boll.bot[0]
                )

                if sl_hit or tp_hit or mid_hit or quick_return:
                    self.order = self.close()
