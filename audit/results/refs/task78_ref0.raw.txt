import backtrader as bt


class ORBFibRetracementStrategy(bt.Strategy):
    params = (
        ('fib_50', 0.5),
        ('fib_618', 0.618),
        ('sl_buffer_pct', 0.001),
        ('rr_ratio', 1.5),
        ('risk_pct', 0.01),
    )

    def __init__(self):
        self.order = None

        self.current_day = None
        self.orb_high = None
        self.orb_low = None
        self.orb_set = False

        self.fib50 = None
        self.fib618 = None

        self.broke_up = False
        self.broke_down = False
        self.long_used = False
        self.short_used = False

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return
        if self.order is not None and order.ref == self.order.ref:
            if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
                self.order = None

    def next(self):
        dt = self.data.datetime.date(0)
        if dt != self.current_day:
            self.current_day = dt
            self.orb_high = None
            self.orb_low = None
            self.orb_set = False
            self.fib50 = None
            self.fib618 = None
            self.broke_up = False
            self.broke_down = False
            self.long_used = False
            self.short_used = False

        if not self.orb_set:
            self.orb_high = self.data.high[0]
            self.orb_low = self.data.low[0]
            rng = self.orb_high - self.orb_low
            self.fib50 = self.orb_low + self.p.fib_50 * rng
            self.fib618 = self.orb_low + self.p.fib_618 * rng
            self.orb_set = True
            return

        if self.order or self.position:
            return

        close = self.data.close[0]
        open_ = self.data.open[0]

        if close > self.orb_high:
            self.broke_up = True
        if close < self.orb_low:
            self.broke_down = True

        zone_lo = self.fib50
        zone_hi = self.fib618

        if self.broke_up and not self.long_used and zone_lo <= close <= zone_hi and close > open_:
            stop_price = self.fib618 * (1.0 - self.p.sl_buffer_pct)
            risk_per_share = close - stop_price
            if risk_per_share > 0:
                risk_amount = self.broker.getvalue() * self.p.risk_pct
                size = int(risk_amount / risk_per_share)
                if size >= 1:
                    take_profit = close + self.p.rr_ratio * risk_per_share
                    orders = self.buy_bracket(
                        size=size,
                        exectype=bt.Order.Market,
                        stopprice=stop_price,
                        limitprice=take_profit,
                    )
                    self.order = orders[0]
                    self.long_used = True
            return

        if self.broke_down and not self.short_used and zone_lo <= close <= zone_hi and close < open_:
            stop_price = self.fib618 * (1.0 + self.p.sl_buffer_pct)
            risk_per_share = stop_price - close
            if risk_per_share > 0:
                risk_amount = self.broker.getvalue() * self.p.risk_pct
                size = int(risk_amount / risk_per_share)
                if size >= 1:
                    take_profit = close - self.p.rr_ratio * risk_per_share
                    orders = self.sell_bracket(
                        size=size,
                        exectype=bt.Order.Market,
                        stopprice=stop_price,
                        limitprice=take_profit,
                    )
                    self.order = orders[0]
                    self.short_used = True
            return
