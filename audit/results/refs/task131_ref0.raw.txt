import backtrader as bt
import collections
import statistics


class NVDAPreEarningsMomentumBreakout(bt.Strategy):
    params = dict(
        support_low=6455.0,      # lower boundary of nearest support zone
        support_high=6465.0,     # upper boundary of nearest support zone
        key_support=6430.75,     # 50% retracement key support
        resistance=6496.25,      # previous high / breakout level
        target=6508.75,          # take-profit target (new ATH)
        stop_price=6455.0,       # stop-loss at lower boundary of support zone
        risk_pct=0.02,           # max loss per trade as fraction of capital (1-2% -> 2% cap)
        trend_period=200,        # 200-period SMA on the daily timeframe
    )

    def __init__(self):
        self.order = None
        self.pullback_occurred = False
        self.current_day = None
        self.daily_closes = collections.deque(maxlen=self.p.trend_period)

    def notify_order(self, order):
        if order.status in (order.Completed, order.Canceled, order.Margin, order.Rejected):
            self.order = None

    def next(self):
        d = self.data

        # --- build the daily-timeframe series for the 200-day SMA (feed is 1h) ---
        dt = d.datetime.date(0)
        if self.current_day is None:
            self.current_day = dt
        elif dt != self.current_day:
            self.daily_closes.append(d.close[-1])  # prior day's closing bar
            self.current_day = dt

        # --- pullback to the key support (50% retracement) has occurred ---
        if d.low[0] <= self.p.key_support:
            self.pullback_occurred = True

        if len(d) < 2:
            return
        if self.order:
            return

        if not self.position:
            # Entry condition 1: price above or within nearest support zone (>= 6455)
            in_support = d.close[0] >= self.p.support_low
            # Entry condition 2: a pullback from 6430.75 occurred and price is recovering
            recovering = self.pullback_occurred and d.close[0] > d.close[-1]
            # Entry condition 3: break above resistance 6496.25 with increased volume
            breakout = d.close[0] > self.p.resistance and d.close[-1] <= self.p.resistance
            rising_vol = d.volume[0] > d.volume[-1]
            # Optional confirmation: global uptrend (price above 200-day SMA)
            trend_ok = (len(self.daily_closes) >= self.p.trend_period and
                        d.close[0] > statistics.mean(self.daily_closes))

            if in_support and recovering and breakout and rising_vol and trend_ok:
                entry_price = d.close[0]
                stop_dist = entry_price - self.p.stop_price
                risk_capital = self.broker.getvalue() * self.p.risk_pct
                size = int(risk_capital / stop_dist) if stop_dist > 0 else 0
                max_affordable = int(self.broker.getcash() / entry_price)
                size = min(size, max_affordable)
                if size >= 1:
                    self.order = self.buy(size=size)
                    self.pullback_occurred = False
        else:
            # Exit 1: take-profit on reaching the target (new ATH)
            if d.high[0] >= self.p.target:
                self.order = self.close()
            # Exit 2: stop-loss on dropping below the support zone (stop at 6455)
            elif d.low[0] <= self.p.stop_price:
                self.order = self.close()
            else:
                # Exit 3 (alternative): bearish reversal candle on increased volume
                bearish_engulf = (d.close[0] < d.open[0] and
                                  d.close[-1] > d.open[-1] and
                                  d.close[0] < d.open[-1] and
                                  d.open[0] > d.close[-1])
                if bearish_engulf and d.volume[0] > d.volume[-1]:
                    self.order = self.close()
