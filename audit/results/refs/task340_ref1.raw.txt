import backtrader as bt


class IntradayMeanReversionStrategy(bt.Strategy):
    def __init__(self):
        self.williams_r = bt.indicators.WilliamsR(self.data, period=14)
        self.vwap = None
        self._session_date = None
        self._cumulative_tpv = 0.0
        self._cumulative_volume = 0.0
        self.order = None
        self.stop_order = None
        self.entry_vwap = None

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            if order == self.order:
                if order.status == order.Completed and order.isbuy():
                    self.stop_order = self.sell(
                        exectype=bt.Order.Stop,
                        price=max(order.executed.price - 2.0, self.entry_vwap - 2.0),
                    )
                elif order.status == order.Completed and order.issell():
                    self.stop_order = self.buy(
                        exectype=bt.Order.Stop,
                        price=min(order.executed.price + 2.0, self.entry_vwap + 2.0),
                    )
                self.order = None

            if order == self.stop_order:
                self.stop_order = None

    def next(self):
        current_date = self.data.datetime.date(0)
        if current_date != self._session_date:
            self._session_date = current_date
            self._cumulative_tpv = 0.0
            self._cumulative_volume = 0.0

        typical_price = (self.data.high[0] + self.data.low[0] + self.data.close[0]) / 3.0
        self._cumulative_tpv += typical_price * self.data.volume[0]
        self._cumulative_volume += self.data.volume[0]
        self.vwap = self._cumulative_tpv / self._cumulative_volume

        if len(self) < 2 or self.order:
            return

        previous_typical_price = (
            self.data.high[-1] + self.data.low[-1] + self.data.close[-1]
        ) / 3.0
        previous_date = self.data.datetime.date(-1)
        if previous_date != current_date:
            previous_vwap = previous_typical_price
        else:
            previous_vwap = (
                self._cumulative_tpv - typical_price * self.data.volume[0]
            ) / (self._cumulative_volume - self.data.volume[0])

        if self.position.size > 0:
            if self.data.close[-1] <= previous_vwap and self.data.close[0] > self.vwap:
                if self.stop_order:
                    self.cancel(self.stop_order)
                    self.stop_order = None
                self.order = self.close()

        elif self.position.size < 0:
            if self.data.close[-1] >= previous_vwap and self.data.close[0] < self.vwap:
                if self.stop_order:
                    self.cancel(self.stop_order)
                    self.stop_order = None
                self.order = self.close()

        elif not self.stop_order:
            if (
                self.data.close[0] < self.vwap
                and self.williams_r[-1] >= -80
                and self.williams_r[0] < -80
            ):
                self.entry_vwap = self.vwap
                self.order = self.buy()

            elif (
                self.data.close[0] > self.vwap
                and self.williams_r[-1] <= -20
                and self.williams_r[0] > -20
            ):
                self.entry_vwap = self.vwap
                self.order = self.sell()
