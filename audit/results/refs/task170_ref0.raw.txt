import backtrader as bt
import collections


class SwingBreakoutStrategy(bt.Strategy):

    def __init__(self):
        self.sma200 = bt.indicators.SMA(self.data.close, period=200)
        self.bb = bt.indicators.BollingerBands(self.data.close, period=20, devfactor=2)
        self.atr = bt.indicators.ATR(self.data, period=14)
        self.vol_sma = bt.indicators.SMA(self.data.volume, period=20)
        self.hhv = bt.indicators.Highest(self.data.high, period=20)
        self.llv = bt.indicators.Lowest(self.data.low, period=20)

        self.cons_flags = collections.deque(maxlen=10)

        self.entry_order = None
        self.entry_price = None
        self.stop_price = None

        self.leg1_limit = None
        self.leg1_stop = None
        self.leg2_limit = None
        self.leg2_stop = None
        self.leg3_stop = None
        self.leg3_size = 0
        self.leg3_stop_price = None
        self.highest_close_since_entry = None

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return

        if order is self.entry_order:
            self.entry_order = None
            if order.status == order.Completed:
                gap_violation = self.data.open[0] > self.entry_price
                vol_violation = self.data.volume[0] <= self.vol_sma[0] * 1.3
                if gap_violation or vol_violation:
                    self.close()
                else:
                    self._setup_exit_legs(order.executed.size)
            return

        if order is self.leg1_limit or order is self.leg1_stop:
            self.leg1_limit = None
            self.leg1_stop = None
            return

        if order is self.leg2_limit or order is self.leg2_stop:
            self.leg2_limit = None
            self.leg2_stop = None
            return

        if order is self.leg3_stop:
            self.leg3_stop = None
            self.leg3_size = 0
            return

    def _setup_exit_legs(self, filled_size):
        s1 = filled_size // 3
        s2 = filled_size // 3
        s3 = filled_size - s1 - s2

        t1_price = self.entry_price * 1.02
        t2_price = self.entry_price * 1.07
        stop_price = self.stop_price

        if s1 > 0:
            self.leg1_limit = self.sell(size=s1, exectype=bt.Order.Limit, price=t1_price)
            self.leg1_stop = self.sell(size=s1, exectype=bt.Order.Stop, price=stop_price,
                                        oco=self.leg1_limit)
        if s2 > 0:
            self.leg2_limit = self.sell(size=s2, exectype=bt.Order.Limit, price=t2_price)
            self.leg2_stop = self.sell(size=s2, exectype=bt.Order.Stop, price=stop_price,
                                        oco=self.leg2_limit)
        if s3 > 0:
            self.leg3_size = s3
            self.leg3_stop_price = stop_price
            self.leg3_stop = self.sell(size=s3, exectype=bt.Order.Stop, price=stop_price)
            self.highest_close_since_entry = self.data.close[0]

        self.entry_price = None
        self.stop_price = None

    def next(self):
        cond_atr = (self.atr[0] / self.data.close[0]) < 0.02
        cond_bb = ((self.bb.top[0] - self.bb.bot[0]) / self.bb.mid[0]) < 0.04
        self.cons_flags.append(1 if (cond_atr and cond_bb) else 0)

        if self.position:
            if self.leg3_stop is not None and self.leg3_size > 0:
                if self.data.close[0] > self.highest_close_since_entry:
                    self.highest_close_since_entry = self.data.close[0]
                new_trail = self.highest_close_since_entry - 3 * self.atr[0]
                if new_trail > self.leg3_stop_price:
                    self.cancel(self.leg3_stop)
                    self.leg3_stop_price = new_trail
                    self.leg3_stop = self.sell(size=self.leg3_size, exectype=bt.Order.Stop,
                                                price=self.leg3_stop_price)
            return

        if self.entry_order is not None:
            atr_close = self.atr[0] / self.data.close[0]
            bb_width = (self.bb.top[0] - self.bb.bot[0]) / self.bb.mid[0]

            cancel = False
            if self.data.open[0] > self.entry_price:
                cancel = True
            if atr_close >= 0.025 or bb_width >= 0.06:
                cancel = True
            if self.data.low[0] <= self.stop_price:
                cancel = True

            if cancel:
                self.cancel(self.entry_order)
                self.entry_order = None
                self.entry_price = None
                self.stop_price = None
            return

        if not (self.data.close[0] > self.sma200[0]):
            return

        if len(self.cons_flags) < 10 or sum(self.cons_flags) < 8:
            return

        resistance = self.hhv[0]
        support = self.llv[0]
        entry_price = resistance + max(0.001 * self.data.close[0], 0.25 * self.atr[0])
        stop_price = support - 0.25 * self.atr[0]

        self.entry_price = entry_price
        self.stop_price = stop_price
        self.entry_order = self.buy(exectype=bt.Order.Stop, price=entry_price)
