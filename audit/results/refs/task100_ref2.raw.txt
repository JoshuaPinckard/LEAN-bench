import backtrader as bt


class DcaRsiStrategy(bt.Strategy):
    params = (
        ('rsi_period', 14),
        ('rsi_oversold', 30.0),
        ('rsi_overbought', 70.0),
        ('rsi_neutral', 50.0),
        ('dca_pct', 0.017),
        ('position_pct', 0.01),
    )

    def __init__(self):
        self.rsi = bt.indicators.RSI(self.data.close, period=self.p.rsi_period)
        self.order = None
        self.primary_price = None
        self.dca_done = False
        self.initial_cash = None

    def start(self):
        self.initial_cash = self.broker.getvalue()

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status == order.Completed:
            if self.position.size == 0:
                self.primary_price = None
                self.dca_done = False
            else:
                if self.primary_price is None:
                    self.primary_price = order.executed.price
        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            if self.position.size == 0:
                self.primary_price = None
                self.dca_done = False
            else:
                self.dca_done = False

        self.order = None

    def next(self):
        if self.order is not None:
            return

        if self.initial_cash is None:
            self.initial_cash = self.broker.getvalue()

        size = (self.initial_cash * self.p.position_pct) / self.data.close[0]

        if not self.position:
            if self.rsi[0] < self.p.rsi_oversold:
                self.order = self.buy(size=size)
            elif self.rsi[0] > self.p.rsi_overbought:
                self.order = self.sell(size=size)
        else:
            if self.position.size > 0:
                if self.rsi[0] > self.p.rsi_neutral:
                    self.order = self.close()
                elif not self.dca_done and self.primary_price is not None:
                    if self.data.close[0] <= self.primary_price * (1.0 - self.p.dca_pct):
                        self.order = self.buy(size=size)
                        self.dca_done = True
            elif self.position.size < 0:
                if self.rsi[0] < self.p.rsi_neutral:
                    self.order = self.close()
                elif not self.dca_done and self.primary_price is not None:
                    if self.data.close[0] >= self.primary_price * (1.0 + self.p.dca_pct):
                        self.order = self.sell(size=size)
                        self.dca_done = True
