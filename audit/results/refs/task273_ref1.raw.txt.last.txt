import backtrader as bt
import math


class FairValueArbitrage(bt.Strategy):
    def __init__(self):
        self.ema50 = bt.indicators.EMA(self.data.close, period=50)
        self.rsi14 = bt.indicators.RSI(self.data.close, period=14)
        self.atr14 = bt.indicators.ATR(self.data, period=14)
        self.bbands = bt.indicators.BollingerBands(
            self.data.close, period=20, devfactor=2
        )
        self.crossover = bt.indicators.CrossOver(self.data.close, self.ema50)

        self.order = None
        self.entry_price = None

    def notify_order(self, order):
        if order.status == order.Completed:
            if order.isbuy() and self.position.size > 0:
                self.entry_price = order.executed.price
            elif order.issell() and self.position.size < 0:
                self.entry_price = order.executed.price
            elif self.position.size == 0:
                self.entry_price = None

        if order.status in (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            self.order = None

    def next(self):
        if self.order:
            return

        close = self.data.close[0]
        ema = self.ema50[0]
        spread = (close - ema) / ema
        annual_volatility = (self.atr14[0] / close) * math.sqrt(252)
        bandwidth = (self.bbands.top[0] - self.bbands.bot[0]) / self.bbands.mid[0]

        if self.position:
            if self.position.size > 0:
                if (
                    self.crossover[0] != 0
                    or close <= self.entry_price * 0.95
                    or close >= self.entry_price * 1.02
                ):
                    self.order = self.close()
            else:
                if (
                    self.crossover[0] != 0
                    or close >= self.entry_price * 1.05
                    or close <= self.entry_price * 0.98
                ):
                    self.order = self.close()
            return

        if annual_volatility > 0.80 or bandwidth >= 0.30:
            return

        multiplier = min(1.5, 1 + abs(spread) / 0.02)
        size = (self.broker.getvalue() * 0.02 * multiplier) / close

        if spread <= -0.01 and self.rsi14[0] < 45:
            self.order = self.buy(size=size)
        elif spread >= 0.01 and self.rsi14[0] > 55:
            self.order = self.sell(size=size)
