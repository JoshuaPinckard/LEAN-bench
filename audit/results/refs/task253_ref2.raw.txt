import backtrader as bt


class PanicEuphoria(bt.Strategy):
    params = (
        ("sma_long_period", 200),
        ("bb_period", 20),
        ("bb_dev", 2.0),
        ("rsi_period", 14),
        ("atr_period", 14),
        ("vol_sma_period", 50),
        ("cash_alloc", 0.99),
    )

    def __init__(self):
        self.order = None

        self.sma200 = bt.indicators.SimpleMovingAverage(
            self.data.close, period=self.p.sma_long_period
        )

        self.bb = bt.indicators.BollingerBands(
            self.data.close, period=self.p.bb_period, devfactor=self.p.bb_dev
        )
        self.bb_width = self.bb.top - self.bb.bot

        self.rsi = bt.indicators.RSI(self.data.close, period=self.p.rsi_period)

        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)

        self.sma_atr = bt.indicators.SimpleMovingAverage(
            self.atr, period=self.p.vol_sma_period
        )
        self.sma_width = bt.indicators.SimpleMovingAverage(
            self.bb_width, period=self.p.vol_sma_period
        )

    def notify_order(self, order):
        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            self.order = None

    def next(self):
        if self.order:
            return

        close = self.data.close[0]

        if not self.position:
            uptrend = close > self.sma200[0]
            panic_close = close < self.bb.bot[0]
            panic_rsi = self.rsi[0] <= 30
            elevated_vol = (self.atr[0] > self.sma_atr[0]) or (
                self.bb_width[0] > self.sma_width[0]
            )

            if uptrend and panic_close and panic_rsi and elevated_vol:
                cash = self.broker.getcash()
                size = int((cash * self.p.cash_alloc) / close)
                if size > 0:
                    self.order = self.buy(size=size)
        else:
            euphoria_close = close > self.bb.top[0]
            euphoria_rsi = self.rsi[0] >= 65
            decreasing_vol = (self.atr[0] < self.sma_atr[0]) or (
                self.bb_width[0] < self.sma_width[0]
            )

            if euphoria_close and euphoria_rsi and decreasing_vol:
                self.order = self.close()
