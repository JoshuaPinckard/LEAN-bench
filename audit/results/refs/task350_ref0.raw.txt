import backtrader as bt


class WilliamsCloudStrategy(bt.Strategy):
    params = (
        ('wr_period', 14),
        ('sma_fast', 9),
        ('sma_mid', 26),
        ('sma_slow', 52),
        ('stop_pct', 0.005),
    )

    def __init__(self):
        self.wr = bt.indicators.WilliamsR(self.data, period=self.p.wr_period)
        self.sma9 = bt.indicators.SMA(self.data.close, period=self.p.sma_fast)
        self.sma26 = bt.indicators.SMA(self.data.close, period=self.p.sma_mid)
        self.sma52 = bt.indicators.SMA(self.data.close, period=self.p.sma_slow)
        self.order = None
        self.stop_price = None

    def notify_order(self, order):
        if order.status in (order.Completed, order.Canceled, order.Margin, order.Rejected):
            self.order = None
            if order.status == order.Completed and not self.position:
                self.stop_price = None

    def next(self):
        if self.order:
            return

        cloud_top = max(self.sma9[0], self.sma26[0], self.sma52[0])
        cloud_bot = min(self.sma9[0], self.sma26[0], self.sma52[0])

        wr_cross_below_80 = self.wr[-1] >= -80 and self.wr[0] < -80
        wr_cross_above_20 = self.wr[-1] <= -20 and self.wr[0] > -20
        sma_cross_up = self.sma9[-1] <= self.sma26[-1] and self.sma9[0] > self.sma26[0]
        sma_cross_down = self.sma9[-1] >= self.sma26[-1] and self.sma9[0] < self.sma26[0]

        if self.position:
            if self.position.size > 0:
                new_stop = self.sma26[0] * (1 - self.p.stop_pct)
                if self.stop_price is None or new_stop > self.stop_price:
                    self.stop_price = new_stop
                if self.data.low[0] <= self.stop_price:
                    self.order = self.close()
                    return
                if self.data.close[0] < cloud_bot:
                    self.order = self.close()
                    return
            elif self.position.size < 0:
                new_stop = self.sma26[0] * (1 + self.p.stop_pct)
                if self.stop_price is None or new_stop < self.stop_price:
                    self.stop_price = new_stop
                if self.data.high[0] >= self.stop_price:
                    self.order = self.close()
                    return
                if self.data.close[0] > cloud_top:
                    self.order = self.close()
                    return
        else:
            if wr_cross_below_80 and self.data.close[0] > cloud_top and sma_cross_up:
                self.stop_price = self.sma26[0] * (1 - self.p.stop_pct)
                self.order = self.buy()
            elif wr_cross_above_20 and self.data.close[0] < cloud_bot and sma_cross_down:
                self.stop_price = self.sma26[0] * (1 + self.p.stop_pct)
                self.order = self.sell()
