import backtrader as bt


class TightScalpingStrategy(bt.Strategy):
    params = (
        ("size", 10),
        ("movement_pct", 0.001),
        ("volatility_pct", 0.002),
        ("take_profit_pct", 0.002),
        ("stop_loss_pct", 0.002),
    )

    def __init__(self):
        self.order = None
        self.stop_order = None
        self.take_profit_order = None

    def next(self):
        if self.order or self.stop_order or self.take_profit_order:
            return

        if self.position:
            return

        previous_close = self.data.close[-1]
        current_close = self.data.close[0]
        movement = (current_close - previous_close) / previous_close
        volatility = (self.data.high[0] - self.data.low[0]) / current_close

        if volatility < self.p.volatility_pct:
            return

        if movement >= self.p.movement_pct:
            self.order = self.buy(size=self.p.size)
        elif movement <= -self.p.movement_pct:
            self.order = self.sell(size=self.p.size)

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return

        if order.status == order.Completed:
            if order == self.order:
                entry_price = order.executed.price

                if order.isbuy():
                    self.stop_order = self.sell(
                        size=self.p.size,
                        exectype=bt.Order.Stop,
                        price=entry_price * (1.0 - self.p.stop_loss_pct),
                    )
                    self.take_profit_order = self.sell(
                        size=self.p.size,
                        exectype=bt.Order.Limit,
                        price=entry_price * (1.0 + self.p.take_profit_pct),
                        oco=self.stop_order,
                    )
                else:
                    self.stop_order = self.buy(
                        size=self.p.size,
                        exectype=bt.Order.Stop,
                        price=entry_price * (1.0 + self.p.stop_loss_pct),
                    )
                    self.take_profit_order = self.buy(
                        size=self.p.size,
                        exectype=bt.Order.Limit,
                        price=entry_price * (1.0 - self.p.take_profit_pct),
                        oco=self.stop_order,
                    )

                self.order = None
                return

            if order == self.stop_order or order == self.take_profit_order:
                self.stop_order = None
                self.take_profit_order = None
                return

        if order.status in (order.Canceled, order.Margin, order.Rejected):
            if order == self.order:
                self.order = None
            elif order == self.stop_order:
                self.stop_order = None
            elif order == self.take_profit_order:
                self.take_profit_order = None
