import backtrader as bt
import math


class APlusReversalStrategy(bt.Strategy):
    def __init__(self):
        self.addminperiod(2)

        self.entry_order = None
        self.stop_order = None
        self.target_orders = []

        self.previous_close_target = None
        self.previous_high_target = None
        self.ict_midnight_open_target = None
        self.stop_price = None
        self.entry_size = None
        self.replace_stop = False

    def next(self):
        if self.position or self.entry_order or self.stop_order or self.target_orders:
            return

        previous_high = self.data.high[-1]
        previous_close = self.data.close[-1]
        session_open = self.data.open[0]

        # The only available daily open is used as the ICT Midnight Open proxy.
        ict_midnight_open = session_open

        outside_day = session_open > previous_high
        gap_up = session_open > previous_close
        above_ict_open = session_open > ict_midnight_open

        if outside_day and gap_up and above_ict_open:
            self.previous_close_target = previous_close
            self.previous_high_target = previous_high
            self.ict_midnight_open_target = ict_midnight_open

            # Aggressive entry: midpoint of the stated $70-$80 expected spike,
            # while also covering the available bar's local high.
            spike_high = max(self.data.high[0], session_open + 75.0)
            self.stop_price = math.nextafter(spike_high, math.inf)

            self.entry_order = self.sell()

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return

        if order is self.entry_order:
            self.entry_order = None

            if order.status == order.Completed:
                self.entry_size = abs(order.executed.size)
                self._place_stop()
                self._place_targets()
            return

        if order is self.stop_order:
            self.stop_order = None

            if order.status == order.Completed:
                for target_order in self.target_orders:
                    self.cancel(target_order)
                self.target_orders = []
                self.replace_stop = False
            elif self.replace_stop and self.position:
                self.replace_stop = False
                self._place_stop()
            return

        if order in self.target_orders:
            self.target_orders.remove(order)

            if order.status == order.Completed:
                if self.stop_order:
                    self.replace_stop = bool(self.position)
                    self.cancel(self.stop_order)

                if not self.position:
                    for target_order in self.target_orders:
                        self.cancel(target_order)
                    self.target_orders = []

    def _place_stop(self):
        if self.position:
            self.stop_order = self.buy(
                size=abs(self.position.size),
                exectype=bt.Order.Stop,
                price=self.stop_price,
            )

    def _place_targets(self):
        partial_size = self.entry_size / 3.0

        self.target_orders = [
            self.buy(
                size=partial_size,
                exectype=bt.Order.Limit,
                price=self.previous_close_target,
            ),
            self.buy(
                size=partial_size,
                exectype=bt.Order.Limit,
                price=self.previous_high_target,
            ),
            self.buy(
                size=partial_size,
                exectype=bt.Order.Limit,
                price=self.ict_midnight_open_target,
            ),
        ]
