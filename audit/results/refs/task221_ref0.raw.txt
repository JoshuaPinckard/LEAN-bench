import backtrader as bt


class ShortVolatilityProxyStrategy(bt.Strategy):
    params = dict(
        bb_period=20,
        bb_devfactor=2.0,
        bbw_ema_period=100,
        atr_period=14,
        atr_ema_period=100,
        rsi_period=14,
        vol_sma_period=20,
        vol_spike_bbw_mult=1.5,
        vol_spike_atr_mult=1.2,
        vol_spike_vol_mult=1.2,
        entry_window_min=5,
        entry_window_max=21,
        rsi_short_thresh=60,
        rsi_long_thresh=40,
        event_gap_atr_mult=1.5,
        event_vol_mult=1.5,
        event_atr_mult=1.3,
        max_hold_bars=21,
        stop_atr_mult=1.5,
        risk_pct=0.015,
        liquidity_threshold=5_000_000,
    )

    def __init__(self):
        self.bb = bt.indicators.BollingerBands(
            self.data.close, period=self.p.bb_period, devfactor=self.p.bb_devfactor)
        self.bbw = (self.bb.top - self.bb.bot) / self.bb.mid
        self.ema_bbw = bt.indicators.EMA(self.bbw, period=self.p.bbw_ema_period)
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.ema_atr = bt.indicators.EMA(self.atr, period=self.p.atr_ema_period)
        self.rsi = bt.indicators.RSI(self.data.close, period=self.p.rsi_period)
        self.vol_sma = bt.indicators.SMA(self.data.volume, period=self.p.vol_sma_period)

        self.bars_since_vol_spike = None
        self.vol_spike_prev = False

        self.entry_order = None
        self.exit_order = None
        self.stop_order = None
        self.entry_bar = None
        self.direction = None
        self.pending_atr = None

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return

        if order.status == order.Completed:
            if order == self.entry_order:
                self.entry_bar = len(self)
                entry_price = order.executed.price
                size = abs(order.executed.size)
                stop_dist = self.p.stop_atr_mult * self.pending_atr
                if self.direction == 'long':
                    stop_price = entry_price - stop_dist
                    self.stop_order = self.sell(
                        exectype=bt.Order.Stop, price=stop_price, size=size)
                else:
                    stop_price = entry_price + stop_dist
                    self.stop_order = self.buy(
                        exectype=bt.Order.Stop, price=stop_price, size=size)
                self.entry_order = None
            elif order == self.stop_order:
                self.stop_order = None
                self.exit_order = None
                self.entry_order = None
                self.entry_bar = None
                self.direction = None
            elif order == self.exit_order:
                self.exit_order = None
                self.stop_order = None
                self.entry_order = None
                self.entry_bar = None
                self.direction = None
        elif order.status in (order.Canceled, order.Margin, order.Rejected):
            if order == self.entry_order:
                self.entry_order = None
                self.direction = None
            elif order == self.stop_order:
                self.stop_order = None
            elif order == self.exit_order:
                self.exit_order = None

    def next(self):
        vol_spike_now = (
            self.bbw[0] > self.p.vol_spike_bbw_mult * self.ema_bbw[0]
            and self.atr[0] > self.p.vol_spike_atr_mult * self.ema_atr[0]
            and self.data.volume[0] > self.p.vol_spike_vol_mult * self.vol_sma[0]
        )
        if vol_spike_now and not self.vol_spike_prev:
            self.bars_since_vol_spike = 0
        elif self.bars_since_vol_spike is not None:
            self.bars_since_vol_spike += 1
        self.vol_spike_prev = vol_spike_now

        if self.entry_order or self.exit_order:
            return

        if self.position:
            event = (
                abs(self.data.open[0] - self.data.close[-1]) > self.p.event_gap_atr_mult * self.atr[0]
                and self.data.volume[0] > self.p.event_vol_mult * self.vol_sma[0]
                and self.atr[0] > self.p.event_atr_mult * self.ema_atr[0]
            )
            if self.direction == 'long':
                mid_reversion = self.data.close[0] >= self.bb.mid[0]
            else:
                mid_reversion = self.data.close[0] <= self.bb.mid[0]
            bbw_crash = self.bbw[0] < self.ema_bbw[0]
            bars_held = len(self) - self.entry_bar if self.entry_bar is not None else 0
            timeout = bars_held >= self.p.max_hold_bars

            # self.close() fills at next bar's open by backtrader's default
            # market-order timing, giving the "event -> close at next open" rule for free.
            if event or mid_reversion or bbw_crash or timeout:
                if self.stop_order:
                    self.cancel(self.stop_order)
                self.exit_order = self.close()
            return

        if self.bars_since_vol_spike is None:
            return
        if not (self.p.entry_window_min <= self.bars_since_vol_spike <= self.p.entry_window_max):
            return
        if self.vol_sma[0] <= self.p.liquidity_threshold:
            return

        if self.data.close[0] > self.bb.top[0] and self.rsi[0] > self.p.rsi_short_thresh:
            direction = 'short'
        elif self.data.close[0] < self.bb.bot[0] and self.rsi[0] < self.p.rsi_long_thresh:
            direction = 'long'
        else:
            return

        atr_val = self.atr[0]
        if atr_val <= 0:
            return
        equity = self.broker.getvalue()
        risk_dollars = equity * self.p.risk_pct
        stop_dist = self.p.stop_atr_mult * atr_val
        size = int(risk_dollars / stop_dist)
        if size < 1:
            return

        self.direction = direction
        self.pending_atr = atr_val
        if direction == 'short':
            self.entry_order = self.sell(size=size)
        else:
            self.entry_order = self.buy(size=size)
