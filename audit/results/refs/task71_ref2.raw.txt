```python
import datetime
import backtrader as bt


class FuturesORBEMAStrategy(bt.Strategy):
    def __init__(self):
        self.ema9 = bt.ind.EMA(self.data.close, period=9)
        self.ema21 = bt.ind.EMA(self.data.close, period=21)
        self.crossover = bt.ind.CrossOver(self.ema9, self.ema21)

        self.order = None

        # ORB state
        self.orb_high = -float("inf")
        self.orb_low = float("inf")
        self.orb_ready = False
        self.current_date = None

        # Daily TR/ATR tracking
        self.daily_high = None
        self.daily_low = None
        self.daily_close = None
        self.prev_daily_close = None
        self.daily_trs = []
        self.high_volatility = False

    def notify_order(self, order):
        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            self.order = None

    def next(self):
        dt = self.data.datetime.datetime(0)
        curr_date = dt.date()
        curr_time = dt.time()

        # Day rollover and daily ATR calculation
        if self.current_date != curr_date:
            if self.current_date is not None and self.daily_high is not None:
                tr = self.daily_high - self.daily_low
                if self.prev_daily_close is not None:
                    tr = max(
                        tr,
                        abs(self.daily_high - self.prev_daily_close),
                        abs(self.daily_low - self.prev_daily_close),
                    )
                self.daily_trs.append(tr)
                self.prev_daily_close = self.daily_close

                if len(self.daily_trs) >= 20:
                    atr_prev = sum(self.daily_trs[-14:]) / 14.0
                    atr_20 = sum(self.daily_trs[-20:]) / 20.0
                    self.high_volatility = atr_prev > 1.20 * atr_20
                else:
                    self.high_volatility = False

            self.current_date = curr_date
            self.daily_high = self.data.high[0]
            self.daily_low = self.data.low[0]
            self.daily_close = self.data.close[0]

            self.orb_high = -float("inf")
            self.orb_low = float("inf")
            self.orb_ready = False
        else:
            self.daily_high = max(self.daily_high, self.data.high[0])
            self.daily_low = min(self.daily_low, self.data.low[0])
            self.daily_close = self.data.close[0]

        # ORB window: 09:30 to 10:00 (bars before 10:00)
        if datetime.time(9, 30) <= curr_time < datetime.time(10, 0):
            self.orb_high = max(self.orb_high, self.data.high[0])
            self.orb_low = min(self.orb_low, self.data.low[0])
        elif curr_time >= datetime.time(10, 0):
            if (
                not self.orb_ready
                and self.orb_high != -float("inf")
                and self.orb_low != float("inf")
            ):
                self.orb_ready = True

        # Handle Exits for Open Positions
        if self.position:
            entry_price = self.position.price

            # Time-based Exit: 15 minutes before market close (15:45)
            if curr_time >= datetime.time(15, 45):
                if not self.order:
                    self.order = self.close()
                return

            if self.position.size > 0:
                tp_price = self.orb_high + 0.005 * entry_price
                sl_price = self.orb_low
                if (
                    self.data.high[0] >= tp_price
                    or self.data.low[0] <= sl_price
                ):
                    if not self.order:
                        self.order = self.close()
                    return
            elif self.position.size < 0:
                tp_price = self.orb_low - 0.005 * entry_price
                sl_price = self.orb_high
                if (
                    self.data.low[0] <= tp_price
                    or self.data.high[0] >= sl_price
                ):
                    if not self.order:
                        self.order = self.close()
                    return

        # Entry Logic
        if not self.position and self.order is None and self.orb_ready:
            if (
                datetime.time(10, 0) <= curr_time < datetime.time(15, 45)
                and not self.high_volatility
            ):
                close = self.data.close[0]
                prev_close = self.data.close[-1]

                # Long Entry Condition
                if self.ema9[0] > self.ema21[0] and close > self.orb_high:
                    if (prev_close <= self.orb_high) or (
                        self.crossover[0] > 0
                    ):
                        risk_per_unit = close - self.orb_low
                        if risk_per_unit > 0:
                            equity = self.broker.getvalue()
                            risk_amount = 0.01 * equity
                            size = int(risk_amount / risk_per_unit)
                            if size > 0:
                                self.order = self.buy(size=size)
                                return

                # Short Entry Condition
                if self.ema9[0] < self.ema21[0] and close < self.orb_low:
                    if (prev_close >= self.orb_low) or (self.crossover[0] < 0):
                        risk_per_unit = self.orb_high - close
                        if risk_per_unit > 0:
                            equity = self.broker.getvalue()
                            risk_amount = 0.01 * equity
                            size = int(risk_amount / risk_per_unit)
                            if size > 0:
                                self.order = self.sell(size=size)
                                return
```
