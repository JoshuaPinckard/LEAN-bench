import backtrader as bt


class MonthlyMomentumStopStrategy(bt.Strategy):
    params = dict(
        sma_period=20,
        entry_day=1,
        exit_day=5,
        stop_loss_pct=0.02,
    )

    def __init__(self):
        self.sma = bt.indicators.SimpleMovingAverage(
            self.data.close, period=self.p.sma_period
        )
        self.order = None
        self.stop_order = None
        self.entry_price = None
        self.current_month = None
        self.trading_day_in_month = 0

    def notify_order(self, order):
        if order.status not in (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            return

        if order is self.order:
            self.order = None

        if order is self.stop_order:
            self.stop_order = None
            if order.status == order.Completed:
                self.entry_price = None
            return

        if order.status == order.Completed:
            if order.isbuy():
                self.entry_price = order.executed.price
                stop_price = self.entry_price * (1 - self.p.stop_loss_pct)
                self.stop_order = self.sell(
                    size=self.position.size,
                    exectype=bt.Order.Stop,
                    price=stop_price,
                )
            else:
                self.entry_price = None

    def next(self):
        dt = self.data.datetime.date(0)
        if self.current_month != dt.month:
            self.current_month = dt.month
            self.trading_day_in_month = 1
        else:
            self.trading_day_in_month += 1

        if self.order:
            return

        if not self.position:
            if (
                self.trading_day_in_month == self.p.entry_day
                and self.data.close[0] > self.sma[0]
            ):
                cash = self.broker.getcash()
                commission = self.broker.getcommissioninfo(self.data).params.commission
                price = self.data.close[0]
                size = int(cash / (price * (1 + commission)))
                if size > 0:
                    self.order = self.buy(size=size)
        else:
            if self.trading_day_in_month == self.p.exit_day:
                if self.stop_order:
                    self.cancel(self.stop_order)
                self.order = self.close()
