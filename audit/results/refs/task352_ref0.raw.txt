import backtrader as bt


class IntradayTrendATRBands(bt.Strategy):
    params = dict(
        sma_period=20,
        atr_period=14,
        atr_mult=2.0,
    )

    def __init__(self):
        self.sma = bt.indicators.SimpleMovingAverage(self.data.close, period=self.p.sma_period)
        self.atr = bt.indicators.AverageTrueRange(self.data, period=self.p.atr_period)
        self.upper_band = self.sma + self.p.atr_mult * self.atr
        self.lower_band = self.sma - self.p.atr_mult * self.atr
        self.order = None

    def notify_order(self, order):
        if order.status in (order.Completed, order.Canceled, order.Margin, order.Rejected):
            self.order = None

    def next(self):
        if self.order:
            return

        close = self.data.close[0]

        if not self.position:
            if close > self.upper_band[0]:
                self.order = self.buy()
            elif close < self.lower_band[0]:
                self.order = self.sell()
        elif self.position.size > 0:
            if close < self.sma[0] or close <= self.lower_band[0]:
                self.order = self.close()
        elif self.position.size < 0:
            if close > self.sma[0] or close >= self.upper_band[0]:
                self.order = self.close()
