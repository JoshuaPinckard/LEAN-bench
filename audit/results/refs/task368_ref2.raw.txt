import backtrader as bt


class WilliamsRMeanReversion(bt.Strategy):
    params = (
        ("wr_period", 14),
        ("sma_period", 20),
        ("std_period", 20),
        ("oversold", -80.0),
        ("overbought", -20.0),
        ("stop_loss_pct", 0.02),
    )

    def __init__(self):
        self.wr = bt.indicators.WilliamsR(self.data, period=self.p.wr_period)
        self.sma_wr = bt.indicators.SMA(self.wr, period=self.p.sma_period)
        self.dev = self.wr - self.sma_wr
        self.std_dev = bt.indicators.StdDev(self.dev, period=self.p.std_period)

        self.cross_up_oversold = bt.indicators.CrossUp(
            self.wr, self.p.oversold
        )
        self.cross_down_overbought = bt.indicators.CrossDown(
            self.wr, self.p.overbought
        )

        self.cross_down_sma = bt.indicators.CrossDown(self.wr, self.sma_wr)
        self.cross_up_sma = bt.indicators.CrossUp(self.wr, self.sma_wr)

        self.order = None
        self.stop_order = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            if order == self.order:
                self.order = None
                if self.position:
                    if self.position.size > 0:
                        stop_price = order.executed.price * (
                            1.0 - self.p.stop_loss_pct
                        )
                        self.stop_order = self.sell(
                            exectype=bt.Order.Stop, price=stop_price
                        )
                    elif self.position.size < 0:
                        stop_price = order.executed.price * (
                            1.0 + self.p.stop_loss_pct
                        )
                        self.stop_order = self.buy(
                            exectype=bt.Order.Stop, price=stop_price
                        )
            elif order == self.stop_order:
                self.stop_order = None

        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            if order == self.order:
                self.order = None
            if order == self.stop_order:
                self.stop_order = None

    def next(self):
        if self.order:
            return

        entry_filter = abs(self.dev[0]) > 2.0 * self.std_dev[0]

        if not self.position:
            if self.cross_up_oversold[0] and entry_filter:
                self.order = self.buy()
            elif self.cross_down_overbought[0] and entry_filter:
                self.order = self.sell()
        else:
            if self.position.size > 0:
                if self.cross_down_sma[0]:
                    if self.stop_order:
                        self.cancel(self.stop_order)
                    self.order = self.close()
            elif self.position.size < 0:
                if self.cross_up_sma[0]:
                    if self.stop_order:
                        self.cancel(self.stop_order)
                    self.order = self.close()
