import backtrader as bt


class Strategy(bt.Strategy):
    params = (
        ('vwap_threshold_pct', 0.003),
        ('rsi_oversold', 30),
        ('rsi_overbought', 70),
        ('stop_loss_pct', 0.02),
    )

    def __init__(self):
        self.rsi = bt.indicators.RSI(self.data.close, period=14)
        self.order = None
        self.order_role = None
        self.entry_price = None
        self.session_date = None
        self.cumulative_pv = 0.0
        self.cumulative_volume = 0.0
        self.vwap = None

    def notify_order(self, order):
        if order.status == order.Completed:
            if self.order_role == 'entry':
                self.entry_price = order.executed.price
            elif self.order_role == 'exit':
                self.entry_price = None

        if order.status in (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            self.order = None
            self.order_role = None

    def next(self):
        current_date = self.data.datetime.date(0)

        if current_date != self.session_date:
            self.session_date = current_date
            self.cumulative_pv = 0.0
            self.cumulative_volume = 0.0

        typical_price = (
            self.data.high[0] + self.data.low[0] + self.data.close[0]
        ) / 3.0
        self.cumulative_pv += typical_price * self.data.volume[0]
        self.cumulative_volume += self.data.volume[0]

        if self.cumulative_volume == 0:
            return

        self.vwap = self.cumulative_pv / self.cumulative_volume

        if self.order:
            return

        long_signal = (
            self.data.close[0] < self.vwap * (1 - self.p.vwap_threshold_pct)
            and self.rsi[0] < self.p.rsi_oversold
        )
        short_signal = (
            self.data.close[0] > self.vwap * (1 + self.p.vwap_threshold_pct)
            and self.rsi[0] > self.p.rsi_overbought
        )

        if self.position.size > 0:
            if (
                self.entry_price is not None
                and self.data.close[0]
                <= self.entry_price * (1 - self.p.stop_loss_pct)
            ) or short_signal:
                self.order = self.close()
                self.order_role = 'exit'

        elif self.position.size < 0:
            if (
                self.entry_price is not None
                and self.data.close[0]
                >= self.entry_price * (1 + self.p.stop_loss_pct)
            ) or long_signal:
                self.order = self.close()
                self.order_role = 'exit'

        elif long_signal:
            self.order = self.buy()
            self.order_role = 'entry'

        elif short_signal:
            self.order = self.sell()
            self.order_role = 'entry'
