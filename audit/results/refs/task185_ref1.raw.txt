import backtrader as bt


class IntradayMomentumStrategy(bt.Strategy):
    def __init__(self):
        self.atr = bt.indicators.ATR(self.data, period=14)
        self.atr_sma = bt.indicators.SMA(self.atr, period=20)
        self.sma5 = bt.indicators.SMA(self.data.close, period=5)
        self.ema9 = bt.indicators.EMA(self.data.close, period=9)
        self.macd = bt.indicators.MACDHisto(self.data.close)
        self.stoch = bt.indicators.Stochastic(self.data)
        self.price_ema_cross = bt.indicators.CrossOver(self.data.close, self.ema9)

        self.session_date = None
        self.day_open = None
        self.pending_entry = None
        self.stop_order = None
        self.target_order = None
        self.eod_order = None
        self.entry_price = None

    def notify_order(self, order):
        terminal = (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        )

        if order.status in terminal:
            if order is self.pending_entry:
                self.pending_entry = None
            if order is self.stop_order:
                self.stop_order = None
            if order is self.target_order:
                self.target_order = None
            if order is self.eod_order:
                self.eod_order = None

        if order.status == order.Completed and order.isbuy():
            self.entry_price = order.executed.price
            entry_atr = self.atr[0]

            self.stop_order = self.close(
                exectype=bt.Order.Stop,
                price=self.entry_price - entry_atr,
            )
            self.target_order = self.close(
                exectype=bt.Order.Limit,
                price=self.entry_price + (2.5 * entry_atr),
                oco=self.stop_order,
            )

        if order.status == order.Completed and order.issell():
            self.entry_price = None

    def next(self):
        current_dt = self.data.datetime.datetime(0)
        current_date = current_dt.date()
        minutes = current_dt.hour * 60 + current_dt.minute

        first_candle_of_day = current_date != self.session_date
        if first_candle_of_day:
            self.session_date = current_date
            self.day_open = self.data.open[0]

        if minutes == (15 * 60 + 55) and self.position and not self.eod_order:
            if self.stop_order:
                self.cancel(self.stop_order)
            if self.target_order:
                self.cancel(self.target_order)
            self.eod_order = self.close()
            return

        if self.position or self.pending_entry or self.eod_order:
            return

        volatility_ok = self.atr[0] > self.atr_sma[0]
        direction_ok = (
            self.data.close[0] > self.sma5[0]
            or self.macd.histo[0] > 0
        )

        if first_candle_of_day and volatility_ok and direction_ok:
            self.pending_entry = self.buy(size=1)
            return

        in_first_hour = (9 * 60 + 30) <= minutes < (10 * 60 + 30)
        dropped_from_open = self.data.close[0] <= self.day_open * 0.995
        stochastic_reversal = (
            self.stoch.percK[-1] < 20
            and self.stoch.percD[-1] < 20
            and self.stoch.percK[0] > self.stoch.percD[0]
            and self.stoch.percK[-1] <= self.stoch.percD[-1]
        )
        reversal_confirmed = stochastic_reversal or self.price_ema_cross[0] > 0

        if (
            in_first_hour
            and dropped_from_open
            and reversal_confirmed
            and volatility_ok
            and direction_ok
        ):
            self.pending_entry = self.buy(size=1)
