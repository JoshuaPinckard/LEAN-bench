import backtrader as bt


class BollingerBandsStrategy(bt.Strategy):
    def __init__(self):
        self.bb = bt.indicators.BollingerBands(
            self.data.close,
            period=20,
            devfactor=2,
            movav=bt.indicators.SimpleMovingAverage,
        )
        self.middle_bb = self.bb.mid
        self.upper_bb = self.bb.top
        self.lower_bb = self.bb.bot
        self.order = None

    def notify_order(self, order):
        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            self.order = None

    def next(self):
        if self.order:
            return

        previous_close = self.data.close[-1]
        current_close = self.data.close[0]

        if not self.position:
            if previous_close < self.lower_bb[-1] and current_close > self.lower_bb[0]:
                self.order = self.buy()
            elif previous_close > self.upper_bb[-1] and current_close < self.upper_bb[0]:
                self.order = self.sell()
        elif self.position.size > 0:
            if previous_close > self.middle_bb[-1] and current_close < self.middle_bb[0]:
                self.order = self.close()
        else:
            if previous_close < self.middle_bb[-1] and current_close > self.middle_bb[0]:
                self.order = self.close()
