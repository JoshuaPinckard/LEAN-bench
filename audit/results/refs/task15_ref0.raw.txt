import statistics
from collections import deque

import backtrader as bt


class ScalpingPriceActionStrategy(bt.Strategy):
    params = (
        ('entry_pct', 0.0015),
        ('tp_pct', 0.003),
        ('sl_pct', 0.0015),
        ('size', 100),
        ('vol_short_window', 5),
        ('vol_long_window', 60),
        ('vol_multiplier', 1.5),
    )

    def __init__(self):
        self.order = None
        self.entry_price = None
        self.returns = deque(maxlen=self.p.vol_long_window)

    def notify_order(self, order):
        if order.status == order.Completed:
            self.entry_price = order.executed.price if self.position else None
            self.order = None
        elif order.status in (order.Canceled, order.Margin, order.Rejected):
            self.order = None

    def next(self):
        if len(self.data) < 2:
            return

        pct_change = (self.data.close[0] - self.data.close[-1]) / self.data.close[-1]
        self.returns.append(abs(pct_change))

        if self.order:
            return

        if self.position:
            if self.position.size > 0:
                sl_level = self.entry_price * (1 - self.p.sl_pct)
                tp_level = self.entry_price * (1 + self.p.tp_pct)
                if self.data.low[0] <= sl_level:
                    self.order = self.close()
                elif self.data.high[0] >= tp_level:
                    self.order = self.close()
            else:
                sl_level = self.entry_price * (1 + self.p.sl_pct)
                tp_level = self.entry_price * (1 - self.p.tp_pct)
                if self.data.high[0] >= sl_level:
                    self.order = self.close()
                elif self.data.low[0] <= tp_level:
                    self.order = self.close()
            return

        if len(self.returns) < self.p.vol_long_window:
            return

        baseline_vol = statistics.mean(self.returns)
        recent_vol = statistics.mean(list(self.returns)[-self.p.vol_short_window:])
        high_volatility = baseline_vol > 0 and recent_vol > baseline_vol * self.p.vol_multiplier

        if not high_volatility:
            return

        if pct_change >= self.p.entry_pct:
            self.order = self.buy(size=self.p.size)
        elif pct_change <= -self.p.entry_pct:
            self.order = self.sell(size=self.p.size)
