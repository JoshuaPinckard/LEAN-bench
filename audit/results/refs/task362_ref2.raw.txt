import math
import backtrader as bt


class ChandelierADXStrategy(bt.Strategy):
    params = (
        ('lookback_n', 22),
        ('atr_mult', 3.0),
        ('adx_threshold', 25.0),
    )

    lines = ('long_stop_raw', 'short_stop_raw', 'long_stop', 'short_stop')

    def __init__(self):
        self.order = None
        self.atr = bt.indicators.ATR(self.data, period=14)
        self.adx = bt.indicators.ADX(self.data, period=14)
        self.highest = bt.indicators.Highest(self.data.high, period=self.p.lookback_n)
        self.lowest = bt.indicators.Lowest(self.data.low, period=self.p.lookback_n)

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            if self.order == order:
                self.order = None

    def next(self):
        atr_val = self.atr[0]
        highest_val = self.highest[0]
        lowest_val = self.lowest[0]

        long_stop_raw = highest_val - self.p.atr_mult * atr_val
        short_stop_raw = lowest_val + self.p.atr_mult * atr_val

        self.lines.long_stop_raw[0] = long_stop_raw
        self.lines.short_stop_raw[0] = short_stop_raw

        try:
            prev_long = self.lines.long_stop[-1]
        except (IndexError, TypeError):
            prev_long = float('nan')

        try:
            prev_short = self.lines.short_stop[-1]
        except (IndexError, TypeError):
            prev_short = float('nan')

        if self.position.size > 0:
            if not math.isnan(prev_long):
                long_stop = max(prev_long, long_stop_raw)
            else:
                long_stop = long_stop_raw
            short_stop = short_stop_raw
        elif self.position.size < 0:
            long_stop = long_stop_raw
            if not math.isnan(prev_short):
                short_stop = min(prev_short, short_stop_raw)
            else:
                short_stop = short_stop_raw
        else:
            long_stop = long_stop_raw
            short_stop = short_stop_raw

        self.lines.long_stop[0] = long_stop
        self.lines.short_stop[0] = short_stop

        if math.isnan(prev_long) or math.isnan(prev_short):
            return

        if self.order:
            return

        close_curr = self.data.close[0]
        close_prev = self.data.close[-1]

        long_stop_curr = self.lines.long_stop[0]
        long_stop_prev = self.lines.long_stop[-1]

        short_stop_curr = self.lines.short_stop[0]
        short_stop_prev = self.lines.short_stop[-1]

        close_cross_above_long = (close_prev <= long_stop_prev) and (close_curr > long_stop_curr)
        close_cross_below_long = (close_prev >= long_stop_prev) and (close_curr < long_stop_curr)

        close_cross_below_short = (close_prev >= short_stop_prev) and (close_curr < short_stop_curr)
        close_cross_above_short = (close_prev <= short_stop_prev) and (close_curr > short_stop_curr)

        adx_valid = self.adx[0] > self.p.adx_threshold

        if self.position.size == 0:
            if adx_valid and close_cross_above_long:
                self.order = self.buy()
            elif adx_valid and close_cross_below_short:
                self.order = self.sell()

        elif self.position.size > 0:
            if adx_valid and close_cross_below_short:
                self.close()
                self.order = self.sell()
            elif close_cross_below_long:
                self.order = self.close()

        elif self.position.size < 0:
            if adx_valid and close_cross_above_long:
                self.close()
                self.order = self.buy()
            elif close_cross_above_short:
                self.order = self.close()
