import backtrader as bt


class MultifunctionalAIStrategy(bt.Strategy):
    params = (
        ('sequence_length', 20),      # LSTM input-sequence window (proxy)
        ('signal_threshold', 0.2),    # BUY/SELL/HOLD imbalance threshold
        ('atr_period', 14),           # volatility measure for Stop Loss
        ('atr_mult', 1.5),            # SL distance = atr_mult * ATR
        ('risk_reward', 2.0),         # Take Profit = R/R * risk (1:2)
        ('risk_per_trade', 0.01),     # fixed % of equity risked per trade
        ('max_daily_loss', 0.02),     # daily-loss halt threshold (2%)
        ('max_drawdown', 0.05),       # drawdown suspend threshold (5%)
        ('close_before_news', False), # flatten before high-impact news
    )

    def __init__(self):
        # volatility indicator used for Stop Loss / position sizing
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)

        # pending-order guard / intent
        self.order = None
        self.order_intent = None
        self.pending_stop_distance = None

        # open-position state
        self.trade_direction = 0      # 1 long, -1 short, 0 flat
        self.entry_price = None
        self.stop_loss = None
        self.take_profit = None

        # daily risk / drawdown tracking (auto-reset each day)
        self.current_day = None
        self.day_start_equity = None
        self.day_peak_equity = None
        self.trading_halted = False

        # logs
        self.decision_log = []
        self.trade_log = []
        self.risk_log = []

    # ---- LSTM signal generation (deterministic sequence-imbalance proxy) ----
    def _predict_signal(self):
        n = self.p.sequence_length
        up = down = flat = 0
        for k in range(n):
            c0 = self.data.close[-k]
            c1 = self.data.close[-k - 1]
            if c0 > c1:
                up += 1
            elif c0 < c1:
                down += 1
            else:
                flat += 1
        # imbalance between BUY (up) / SELL (down) / HOLD (flat) tendencies
        imbalance = (up - down) / float(n)
        if imbalance >= self.p.signal_threshold:
            return 'BUY'
        if imbalance <= -self.p.signal_threshold:
            return 'SELL'
        return 'HOLD'

    def _log_trade(self, decision, action, size=None, price=None,
                   stop=None, tp=None, expected_risk=None):
        self.trade_log.append({
            'datetime': self.data.datetime.datetime(0),
            'decision': decision,
            'action': action,
            'size': size,
            'price': price,
            'stop_loss': stop,
            'take_profit': tp,
            'expected_risk': expected_risk,
            'equity': self.broker.getvalue(),
        })

    def _close_position(self, decision, reason):
        self.order = self.close()
        self.order_intent = 'exit'
        self._log_trade(decision, reason)

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return
        if order.status == order.Completed:
            if self.order_intent == 'entry':
                # set SL from volatility and TP from Risk/Reward off the fill
                self.entry_price = order.executed.price
                sd = self.pending_stop_distance
                if self.trade_direction == 1:
                    self.stop_loss = self.entry_price - sd
                    self.take_profit = self.entry_price + self.p.risk_reward * sd
                else:
                    self.stop_loss = self.entry_price + sd
                    self.take_profit = self.entry_price - self.p.risk_reward * sd
            elif self.order_intent == 'exit':
                self.trade_direction = 0
                self.entry_price = None
                self.stop_loss = None
                self.take_profit = None
        elif order.status in (order.Canceled, order.Margin, order.Rejected):
            if self.order_intent == 'entry':
                self.trade_direction = 0
                self.stop_loss = None
                self.take_profit = None
        if order.status in (order.Completed, order.Canceled,
                            order.Margin, order.Rejected):
            self.order = None
            self.order_intent = None

    def next(self):
        dt = self.data.datetime.datetime(0)
        day = dt.date()

        # ---- auto-reset of daily-risk / drawdown parameters every day ----
        if self.current_day != day:
            self.current_day = day
            self.day_start_equity = self.broker.getvalue()
            self.day_peak_equity = self.broker.getvalue()
            self.trading_halted = False

        equity = self.broker.getvalue()
        if equity > self.day_peak_equity:
            self.day_peak_equity = equity

        # ---- maximum daily risk ----
        daily_loss = self.day_start_equity - equity
        if daily_loss >= self.p.max_daily_loss * self.day_start_equity:
            self.trading_halted = True
        # ---- maximum drawdown ----
        if self.day_peak_equity > 0:
            drawdown = (self.day_peak_equity - equity) / self.day_peak_equity
            if drawdown >= self.p.max_drawdown:
                self.trading_halted = True

        # guard pending orders
        if self.order:
            return

        # warm-up: need sequence_length + 1 bars for the sequence access
        if len(self.data) <= self.p.sequence_length:
            return

        signal = self._predict_signal()
        self.decision_log.append({'datetime': dt, 'decision': signal,
                                  'equity': equity})

        # ---- close before major news (option; no news feed available) ----
        if self.p.close_before_news and self.position:
            pass  # infeasible: OHLCV feed carries no economic calendar

        # ---- manage an open position ----
        if self.position:
            if self.trade_direction == 1:  # long
                if self.data.low[0] <= self.stop_loss:
                    self._close_position(signal, 'STOP_LOSS')
                    return
                if self.data.high[0] >= self.take_profit:
                    self._close_position(signal, 'TAKE_PROFIT')
                    return
                if signal == 'SELL':  # opposite signal -> close immediately
                    self._close_position(signal, 'OPPOSITE_SIGNAL')
                    return
                if signal == 'BUY':   # same-direction signal -> trail stop
                    new_sl = self.data.close[0] - self.p.atr_mult * self.atr[0]
                    if new_sl > self.stop_loss:
                        self.stop_loss = new_sl
            elif self.trade_direction == -1:  # short
                if self.data.high[0] >= self.stop_loss:
                    self._close_position(signal, 'STOP_LOSS')
                    return
                if self.data.low[0] <= self.take_profit:
                    self._close_position(signal, 'TAKE_PROFIT')
                    return
                if signal == 'BUY':   # opposite signal -> close immediately
                    self._close_position(signal, 'OPPOSITE_SIGNAL')
                    return
                if signal == 'SELL':  # same-direction signal -> trail stop
                    new_sl = self.data.close[0] + self.p.atr_mult * self.atr[0]
                    if new_sl < self.stop_loss:
                        self.stop_loss = new_sl
            return

        # ---- entries: blocked while trading is halted ----
        if self.trading_halted:
            return

        if signal == 'HOLD':
            return

        atr_val = self.atr[0]
        stop_distance = self.p.atr_mult * atr_val
        if stop_distance <= 0:
            return

        price = self.data.close[0]

        # position size so that risk = fixed % of equity
        risk_amount = equity * self.p.risk_per_trade
        size = risk_amount / stop_distance

        # cap to what the cash account can actually fund (environment limit)
        cash = self.broker.getcash()
        if price > 0:
            affordable = 0.99 * cash / price
            if size > affordable:
                size = affordable
        if size <= 0:
            return

        # expected risk = size in points * lot size
        expected_risk = stop_distance * size
        self.pending_stop_distance = stop_distance

        if signal == 'BUY':
            self.trade_direction = 1
            self.order = self.buy(size=size)
            self.order_intent = 'entry'
        else:  # SELL -> open short
            self.trade_direction = -1
            self.order = self.sell(size=size)
            self.order_intent = 'entry'

        self.risk_log.append({
            'datetime': dt,
            'signal': signal,
            'stop_distance': stop_distance,
            'size': size,
            'expected_risk': expected_risk,
        })
        self._log_trade(signal, 'ENTRY', size=size, price=price,
                        expected_risk=expected_risk)
