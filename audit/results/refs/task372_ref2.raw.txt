import backtrader as bt


class KeltnerMeanReversion(bt.Strategy):
    params = (
        ("ema_period", 20),
        ("atr_period", 14),
        ("atr_mult", 2.0),
        ("sl_atr_mult", 2.0),
    )

    def __init__(self):
        self.order = None
        self.entry_price = None
        self.stop_price = None

        self.middle = bt.indicators.EMA(self.data.close, period=self.p.ema_period)
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.upper = self.middle + self.p.atr_mult * self.atr
        self.lower = self.middle - self.p.atr_mult * self.atr

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            if self.position.size > 0:
                self.entry_price = order.executed.price
                self.stop_price = self.entry_price - (self.p.sl_atr_mult * self.atr[0])
            elif self.position.size < 0:
                self.entry_price = order.executed.price
                self.stop_price = self.entry_price + (self.p.sl_atr_mult * self.atr[0])
            else:
                self.entry_price = None
                self.stop_price = None

        self.order = None

    def next(self):
        if self.order:
            return

        if not self.position:
            bullish = self.data.close[0] > self.data.open[0]
            bearish = self.data.close[0] < self.data.open[0]

            if bullish and self.data.close[0] < self.lower[0]:
                self.order = self.buy()
            elif bearish and self.data.close[0] > self.upper[0]:
                self.order = self.sell()
        else:
            if self.position.size > 0:
                crossed_middle = self.data.close[0] >= self.middle[0]
                hit_sl = (
                    self.stop_price is not None
                    and self.data.low[0] <= self.stop_price
                )
                if crossed_middle or hit_sl:
                    self.order = self.close()
            elif self.position.size < 0:
                crossed_middle = self.data.close[0] <= self.middle[0]
                hit_sl = (
                    self.stop_price is not None
                    and self.data.high[0] >= self.stop_price
                )
                if crossed_middle or hit_sl:
                    self.order = self.close()
