import backtrader as bt


class MeanReversionBollingerATRStrategy(bt.Strategy):
    params = dict(
        bb_period=20,
        bb_devfactor=2.0,
        atr_period=14,
        atr_mult=2.0,
    )

    def __init__(self):
        self.bband = bt.indicators.BollingerBands(
            self.data.close, period=self.p.bb_period, devfactor=self.p.bb_devfactor
        )
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.order = None
        self.stop_price = None

    def notify_order(self, order):
        if order.status == order.Completed:
            if self.position.size > 0 and order.isbuy():
                self.stop_price = order.executed.price - self.p.atr_mult * self.atr[0]
            elif self.position.size < 0 and order.issell():
                self.stop_price = order.executed.price + self.p.atr_mult * self.atr[0]
            elif self.position.size == 0:
                self.stop_price = None
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def _calc_size(self):
        cash = self.broker.getcash()
        price = self.data.close[0]
        return int(cash / price)

    def next(self):
        if self.order:
            return

        if not self.position:
            if self.data.close[0] < self.bband.bot[0]:
                size = self._calc_size()
                if size > 0:
                    self.order = self.buy(size=size)
            elif self.data.close[0] > self.bband.top[0]:
                size = self._calc_size()
                if size > 0:
                    self.order = self.sell(size=size)
        else:
            if self.position.size > 0:
                if self.data.close[0] <= self.stop_price:
                    self.order = self.close()
                elif self.data.close[0] >= self.bband.mid[0]:
                    self.order = self.close()
            elif self.position.size < 0:
                if self.data.close[0] >= self.stop_price:
                    self.order = self.close()
                elif self.data.close[0] <= self.bband.mid[0]:
                    self.order = self.close()
