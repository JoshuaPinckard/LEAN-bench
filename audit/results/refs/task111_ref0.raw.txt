import backtrader as bt
from collections import deque


class CupAndHandleQuarterlyStrategy(bt.Strategy):
    params = dict(
        quarter_sma_period=200,
        weekly_sma_period=20,
        cup_min_depth=0.10,
        rim_tolerance=0.07,
        handle_max_retrace=0.50,
        risk_pct=0.02,
    )

    def __init__(self):
        self.order = None

        self.q_key = None
        self.q_high = None
        self.q_low = None
        self.q_close = None
        self.q_bars = []

        self.w_key = None
        self.w_high = None
        self.w_low = None
        self.w_close = None
        self.w_closes = deque(maxlen=self.p.weekly_sma_period)

        self.cup_resistance = None
        self.cup_height = None
        self.handle_low = None
        self.breakout_pending = False
        self.entry_ref_price = None
        self.target_price = None

    def notify_order(self, order):
        if order.status in (order.Completed, order.Canceled, order.Margin, order.Rejected):
            self.order = None

    def _finalize_week(self):
        self.w_closes.append(self.w_close)
        if self.position and self.handle_low is not None and self.w_close < self.handle_low:
            if not self.order:
                self.order = self.close()

    def _weekly_sma20(self):
        if len(self.w_closes) < self.p.weekly_sma_period:
            return None
        return sum(self.w_closes) / self.p.weekly_sma_period

    def _finalize_quarter(self):
        self.q_bars.append((self.q_high, self.q_low, self.q_close))
        self._detect_pattern_and_breakout()

    def _detect_pattern_and_breakout(self):
        n = len(self.q_bars)
        if n < 5:
            return

        highs = [b[0] for b in self.q_bars]
        lows = [b[1] for b in self.q_bars]
        closes = [b[2] for b in self.q_bars]

        pivot_highs = []
        pivot_lows = []
        for i in range(1, n - 1):
            if highs[i] > highs[i - 1] and highs[i] > highs[i + 1]:
                pivot_highs.append(i)
            if lows[i] < lows[i - 1] and lows[i] < lows[i + 1]:
                pivot_lows.append(i)

        if len(pivot_highs) < 2 or not pivot_lows:
            return

        p2 = pivot_highs[-1]
        troughs_before_p2 = [t for t in pivot_lows if t < p2]
        if not troughs_before_p2:
            return
        t_cup = troughs_before_p2[-1]
        peaks_before_t = [p for p in pivot_highs if p < t_cup]
        if not peaks_before_t:
            return
        p1 = peaks_before_t[-1]

        rim = min(highs[p1], highs[p2])
        if rim <= 0:
            return
        depth = (rim - lows[t_cup]) / rim
        symmetry = abs(highs[p1] - highs[p2]) / rim
        if depth < self.p.cup_min_depth or symmetry > self.p.rim_tolerance:
            return

        cup_resistance = max(highs[p1], highs[p2])
        cup_height = cup_resistance - lows[t_cup]

        handle_troughs = [t for t in pivot_lows if t > p2]
        if not handle_troughs:
            return
        t_handle = handle_troughs[-1]
        handle_low = lows[t_handle]

        if handle_low <= lows[t_cup]:
            return
        if (cup_resistance - handle_low) > self.p.handle_max_retrace * cup_height:
            return

        self.cup_resistance = cup_resistance
        self.cup_height = cup_height
        self.handle_low = handle_low

        last_close = closes[-1]
        all_time_high_close = max(closes[:-1])
        quarter_sma200 = None
        if n >= self.p.quarter_sma_period:
            quarter_sma200 = sum(closes[-self.p.quarter_sma_period:]) / self.p.quarter_sma_period

        if (last_close > self.cup_resistance
                and last_close > all_time_high_close
                and quarter_sma200 is not None
                and last_close > quarter_sma200):
            self.breakout_pending = True
            self.entry_ref_price = last_close

    def next(self):
        dt = self.data.datetime.date(0)
        q_key = (dt.year, (dt.month - 1) // 3 + 1)
        w_key = dt.isocalendar()[:2]

        if self.w_key is None:
            self.w_key = w_key
            self.w_high = self.data.high[0]
            self.w_low = self.data.low[0]
            self.w_close = self.data.close[0]
        elif w_key != self.w_key:
            self._finalize_week()
            self.w_key = w_key
            self.w_high = self.data.high[0]
            self.w_low = self.data.low[0]
            self.w_close = self.data.close[0]
        else:
            self.w_high = max(self.w_high, self.data.high[0])
            self.w_low = min(self.w_low, self.data.low[0])
            self.w_close = self.data.close[0]

        if self.q_key is None:
            self.q_key = q_key
            self.q_high = self.data.high[0]
            self.q_low = self.data.low[0]
            self.q_close = self.data.close[0]
        elif q_key != self.q_key:
            self._finalize_quarter()
            self.q_key = q_key
            self.q_high = self.data.high[0]
            self.q_low = self.data.low[0]
            self.q_close = self.data.close[0]
        else:
            self.q_high = max(self.q_high, self.data.high[0])
            self.q_low = min(self.q_low, self.data.low[0])
            self.q_close = self.data.close[0]

        if self.order:
            return

        if self.position:
            if self.target_price is not None and self.data.high[0] >= self.target_price:
                self.order = self.close()
            return

        if self.breakout_pending:
            self.breakout_pending = False
            weekly_sma20 = self._weekly_sma20()
            if weekly_sma20 is not None and self.data.close[0] > weekly_sma20:
                risk_per_share = self.entry_ref_price - self.handle_low
                if risk_per_share > 0:
                    risk_amount = self.p.risk_pct * self.broker.getvalue()
                    size = risk_amount / risk_per_share
                    if size > 0:
                        self.target_price = self.entry_ref_price + self.cup_height
                        self.order = self.buy(size=size)
