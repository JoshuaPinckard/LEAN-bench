import math
import backtrader as bt


class ManagedIncomeCapitalProtection(bt.Strategy):
    params = dict(
        sma_fast_period=50,
        sma_slow_period=200,
        atr_period=14,
        atr_close_max=0.015,
        size_pct=0.05,
        tp_pct=0.05,
        tp_atr_mult=2.5,
        dd_limit=0.10,
    )

    def __init__(self):
        self.sma_fast = bt.indicators.SMA(self.data.close, period=self.p.sma_fast_period)
        self.sma_slow = bt.indicators.SMA(self.data.close, period=self.p.sma_slow_period)
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.order = None
        self.tp_price = None
        self.starting_equity = self.broker.getvalue()

    def notify_order(self, order):
        if order.status in (order.Completed, order.Canceled, order.Margin, order.Rejected):
            if order.status == order.Completed:
                if order.isbuy():
                    entry_price = order.executed.price
                    self.tp_price = min(
                        entry_price * (1.0 + self.p.tp_pct),
                        entry_price + self.p.tp_atr_mult * self.atr[0],
                    )
                elif order.issell():
                    self.tp_price = None
            self.order = None

    def next(self):
        if self.order:
            return

        equity = self.broker.getvalue()
        drawdown = (self.starting_equity - equity) / self.starting_equity

        if drawdown >= self.p.dd_limit:
            if self.position:
                self.order = self.close()
            return

        close = self.data.close[0]

        if self.position:
            if self.tp_price is not None and close >= self.tp_price:
                self.order = self.close()
            elif close < self.sma_fast[0]:
                self.order = self.close()
            return

        if (
            close > self.sma_fast[0]
            and close > self.sma_slow[0]
            and self.atr[0] / close <= self.p.atr_close_max
        ):
            size = math.floor(self.p.size_pct * equity / close)
            if size > 0:
                self.order = self.buy(size=size)
