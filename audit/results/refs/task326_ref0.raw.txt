import backtrader as bt


class MacdBollingerAtrStrategy(bt.Strategy):
    params = dict(
        macd_fast=12,
        macd_slow=26,
        macd_signal=9,
        boll_period=20,
        boll_devfactor=2,
        atr_period=14,
        atr_mult=2,
    )

    def __init__(self):
        self.macd = bt.indicators.MACD(
            self.data,
            period_me1=self.p.macd_fast,
            period_me2=self.p.macd_slow,
            period_signal=self.p.macd_signal,
        )
        self.crossover = bt.indicators.CrossOver(self.macd.macd, self.macd.signal)
        self.boll = bt.indicators.BollingerBands(
            self.data, period=self.p.boll_period, devfactor=self.p.boll_devfactor
        )
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)

        self.entry_order = None
        self.exit_order = None
        self.stop_order = None

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return

        if order.status == order.Completed:
            if order == self.entry_order:
                self.entry_order = None
                stop_dist = self.atr[0] * self.p.atr_mult
                if order.isbuy():
                    stop_price = order.executed.price - stop_dist
                    self.stop_order = self.sell(exectype=bt.Order.Stop, price=stop_price)
                else:
                    stop_price = order.executed.price + stop_dist
                    self.stop_order = self.buy(exectype=bt.Order.Stop, price=stop_price)
            elif order == self.exit_order:
                self.exit_order = None
                self.stop_order = None
            elif order == self.stop_order:
                self.stop_order = None
                self.exit_order = None
        elif order.status in (order.Canceled, order.Margin, order.Rejected):
            if order == self.entry_order:
                self.entry_order = None
            elif order == self.exit_order:
                self.exit_order = None
            elif order == self.stop_order:
                self.stop_order = None

    def next(self):
        if self.entry_order or self.exit_order:
            return

        if not self.position:
            if self.crossover[0] > 0 and self.data.close[0] <= self.boll.bot[0]:
                self.entry_order = self.buy()
            elif self.crossover[0] < 0 and self.data.close[0] >= self.boll.top[0]:
                self.entry_order = self.sell()
        else:
            if self.position.size > 0 and self.data.close[0] >= self.boll.mid[0]:
                if self.stop_order:
                    self.cancel(self.stop_order)
                    self.stop_order = None
                self.exit_order = self.close()
            elif self.position.size < 0 and self.data.close[0] <= self.boll.mid[0]:
                if self.stop_order:
                    self.cancel(self.stop_order)
                    self.stop_order = None
                self.exit_order = self.close()
