import backtrader as bt


class SmallCapMomentum(bt.Strategy):
    params = (
        ("sma_period", 50),
        ("macd_fast", 12),
        ("macd_slow", 26),
        ("macd_signal", 9),
        ("vol_sma_period", 50),
        ("vol_spike_period", 30),
        ("vol_spike_mult", 2.0),
        ("accumulation_days", 5),
        ("rsi_period", 14),
        ("capital_pct", 0.05),
        ("take_profit_pct", 1.0),
        ("stop_loss_pct", 0.15),
    )

    def __init__(self):
        self.sma50 = bt.indicators.SMA(
            self.data.close, period=self.p.sma_period
        )
        self.macd = bt.indicators.MACDHisto(
            self.data.close,
            period_me1=self.p.macd_fast,
            period_me2=self.p.macd_slow,
            period_signal=self.p.macd_signal,
        )
        self.vol_sma50 = bt.indicators.SMA(
            self.data.volume, period=self.p.vol_sma_period
        )
        self.vol_sma30 = bt.indicators.SMA(
            self.data.volume, period=self.p.vol_spike_period
        )
        self.rsi = bt.indicators.RSI(self.data.close, period=self.p.rsi_period)

        self.order = None
        self.entry_price = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            if order.isbuy():
                self.entry_price = order.executed.price
            elif order.issell():
                self.entry_price = None
            self.order = None
        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        if self.order:
            return

        if not self.position:
            cond_trend = self.data.close[0] > self.sma50[0]
            cond_macd = self.macd.histo[0] > self.macd.histo[-1]
            cond_accum = all(
                self.data.volume[-i] > self.vol_sma50[-i]
                for i in range(self.p.accumulation_days)
            )
            cond_spike = (
                self.data.volume[0]
                >= self.p.vol_spike_mult * self.vol_sma30[0]
            )

            if cond_trend and cond_macd and cond_accum and cond_spike:
                capital = self.broker.getvalue() * self.p.capital_pct
                size = int(capital / self.data.close[0])
                if size > 0:
                    self.order = self.buy(size=size)
        else:
            entry = (
                self.entry_price
                if self.entry_price is not None
                else self.position.price
            )

            cond_tp = self.data.close[0] >= entry * (1.0 + self.p.take_profit_pct)
            cond_sl = self.data.close[0] <= entry * (1.0 - self.p.stop_loss_pct)
            cond_exit_macd = self.macd.histo[0] < 0

            if cond_tp or cond_sl or cond_exit_macd:
                self.order = self.close()
