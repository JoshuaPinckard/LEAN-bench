import backtrader as bt


class MACDVolumeStrategy(bt.Strategy):
    params = (
        ('macd_fast', 12),
        ('macd_slow', 26),
        ('macd_signal', 9),
        ('vol_period', 20),
        ('atr_period', 14),
        ('atr_mult', 2.0),
    )

    def __init__(self):
        self.macd = bt.indicators.MACD(
            self.data.close,
            period_me1=self.p.macd_fast,
            period_me2=self.p.macd_slow,
            period_signal=self.p.macd_signal
        )
        self.crossover = bt.indicators.CrossOver(self.macd.macd, self.macd.signal)
        self.vol_sma = bt.indicators.SMA(self.data.volume, period=self.p.vol_period)
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)

        self.order = None
        self.stop_order = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            if order == self.order:
                self.order = None
                if self.position.size > 0:
                    stop_price = order.executed.price - self.p.atr_mult * self.atr[0]
                    self.stop_order = self.sell(exectype=bt.Order.Stop, price=stop_price)
                elif self.position.size < 0:
                    stop_price = order.executed.price + self.p.atr_mult * self.atr[0]
                    self.stop_order = self.buy(exectype=bt.Order.Stop, price=stop_price)
            elif order == self.stop_order:
                self.stop_order = None

        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            if order == self.order:
                self.order = None
            elif order == self.stop_order:
                self.stop_order = None

    def next(self):
        if self.order:
            return

        volume_spike = self.data.volume[0] >= 2.0 * self.vol_sma[0]
        bullish_signal = (self.crossover[0] > 0) and volume_spike
        bearish_signal = (self.crossover[0] < 0) and volume_spike

        if not self.position:
            if bullish_signal:
                self.order = self.buy()
            elif bearish_signal:
                self.order = self.sell()
        else:
            if self.position.size > 0 and bearish_signal:
                if self.stop_order:
                    self.cancel(self.stop_order)
                    self.stop_order = None
                self.close()
                self.order = self.sell()
            elif self.position.size < 0 and bullish_signal:
                if self.stop_order:
                    self.cancel(self.stop_order)
                    self.stop_order = None
                self.close()
                self.order = self.buy()
