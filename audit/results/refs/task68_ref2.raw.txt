```python
import collections
import math
import statistics
import backtrader as bt


class AITradingStrategy(bt.Strategy):
    params = (
        ("risk_per_trade", 0.01),  # 1% risk per trade
        ("rr_ratio", 2.0),  # Risk/Reward ratio 1:2
        ("atr_period", 14),
        ("atr_mult", 2.0),
        ("max_daily_risk", 0.02),  # 2% max daily risk
        ("max_drawdown", 0.05),  # 5% max drawdown limit
        ("close_before_news", False),  # Option for news event closing
    )

    def __init__(self):
        self.order = None
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.sma20 = bt.indicators.SMA(self.data, period=20)
        self.sma50 = bt.indicators.SMA(self.data, period=50)
        self.rsi = bt.indicators.RSI(self.data, period=14)

        self.signal_history = collections.deque(maxlen=100)
        self.trade_log = []

        # Risk management tracking
        self.current_date = None
        self.day_start_equity = 10000.0
        self.daily_pnl = 0.0
        self.daily_halted = False

        self.peak_equity = 10000.0
        self.drawdown_halted = False

        # Open trade metadata
        self.trade_dir = None  # 'LONG' or 'SHORT'
        self.sl_price = None
        self.tp_price = None
        self.entry_price = None
        self.trade_size = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def notify_trade(self, trade):
        if trade.isclosed:
            pnl = trade.pnlcomm
            self.daily_pnl += pnl
            if self.daily_pnl < -(self.day_start_equity * self.p.max_daily_risk):
                self.daily_halted = True

    def _get_62_features(self):
        features = []
        close = self.data.close[0]

        # 1-5: Primary OHLCV
        features.extend([
            self.data.open[0],
            self.data.high[0],
            self.data.low[0],
            close,
            self.data.volume[0]
        ])

        # 6-10: Returns over multiple windows
        for p in [1, 2, 3, 5, 10]:
            ret = (close - self.data.close[-p]) / self.data.close[-p] if len(self.data) > p else 0.0
            features.append(ret)

        # 11-15: High-Low range relative to close
        for p in range(5):
            hl = (self.data.high[-p] - self.data.low[-p]) / self.data.close[-p] if len(self.data) > p else 0.0
            features.append(hl)

        # 16-25: Moving Average differences
        for p in [5, 10, 15, 20, 30, 40, 50, 60, 100, 200]:
            if len(self.data) >= p:
                vals = self.data.close.get(size=p)
                sma = sum(vals) / p
                features.append((close - sma) / sma if sma != 0 else 0.0)
            else:
                features.append(0.0)

        # 26-35: Exponential / Price offset proxies
        for p in [5, 10, 15, 20, 30, 40, 50, 60, 100, 200]:
            if len(self.data) >= p:
                vals = self.data.close.get(size=p)
                features.append(close - (sum(vals) / p))
            else:
                features.append(0.0)

        # 36-40: Volume features
        for p in [1, 2, 3, 5, 10]:
            vol = self.data.volume[-p] if len(self.data) > p else 0.0
            features.append(vol)

        # 41-45: Price momentum
        for p in [1, 3, 5, 10, 20]:
            mom = close - self.data.close[-p] if len(self.data) > p else 0.0
            features.append(mom)

        # 46-50: Price standard deviation
        for p in [5, 10, 15, 20, 30]:
            if len(self.data) >= p:
                vals = self.data.close.get(size=p)
                mean_val = sum(vals) / p
                var_val = sum((x - mean_val) ** 2 for x in vals) / p
                features.append(math.sqrt(var_val))
            else:
                features.append(0.0)

        # 51-55: High minus Close offset
        for p in range(5):
            features.append(self.data.high[-p] - close if len(self.data) > p else 0.0)

        # 56-59: Close minus Low offset
        for p in range(4):
            features.append(close - self.data.low[-p] if len(self.data) > p else 0.0)

        # 60-62: Signal imbalances (counts of past BUY, SELL, HOLD decisions)
        buy_cnt = sum(1 for s in self.signal_history if s == 1)
        sell_cnt = sum(1 for s in self.signal_history if s == -1)
        hold_cnt = sum(1 for s in self.signal_history if s == 0)
        features.extend([buy_cnt, sell_cnt, hold_cnt])

        return features

    def _predict_action(self, features):
        if self.sma20[0] > self.sma50[0] and self.rsi[0] > 55:
            return 1  # BUY
        elif self.sma20[0] < self.sma50[0] and self.rsi[0] < 45:
            return -1  # SELL
        else:
            return 0  # HOLD

    def next(self):
        dt = self.data.datetime.date(0)

        # 24-hour Auto-reset of risk parameters at day start
        if self.current_date != dt:
            self.current_date = dt
            self.day_start_equity = self.broker.getvalue()
            self.daily_pnl = 0.0
            self.daily_halted = False
            self.peak_equity = self.broker.getvalue()
            self.drawdown_halted = False

        # Maximum Drawdown check
        current_equity = self.broker.getvalue()
        if current_equity > self.peak_equity:
            self.peak_equity = current_equity
        drawdown = (self.peak_equity - current_equity) / self.peak_equity if self.peak_equity > 0 else 0.0
        if drawdown >= self.p.max_drawdown:
            self.drawdown_halted = True

        features = self.get_62_features = self._get_62_features()
        signal = self._predict_action(features)
        self.signal_history.append(signal)

        action_name = "BUY" if signal == 1 else ("SELL" if signal == -1 else "HOLD")
        self.trade_log.append({
            "datetime": self.data.datetime.datetime(0),
            "decision": action_name,
            "close": self.data.close[0]
        })

        if self.order:
            return

        # Position Exits & Trailing Stops
        if self.position:
            if self.trade_dir == 'LONG':
                # SL / TP Trigger Check
                if self.data.low[0] <= self.sl_price or self.data.high[0] >= self.tp_price:
                    self.close()
                    self.trade_dir = None
                    return

                # Opposite Signal Exit
                if signal == -1:
                    self.close()
                    self.trade_dir = None
                    return

                # Trailing Stop Update on Same Direction Signal
                if signal == 1:
                    atr_val = self.atr[0]
                    new_sl = self.data.close[0] - (atr_val * self.p.atr_mult)
                    if new_sl > self.sl_price:
                        self.sl_price = new_sl

            elif self.trade_dir == 'SHORT':
                # SL / TP Trigger Check
                if self.data.high[0] >= self.sl_price or self.data.low[0] <= self.tp_price:
                    self.close()
                    self.trade_dir = None
                    return

                # Opposite Signal Exit
                if signal == 1:
                    self.close()
                    self.trade_dir = None
                    return

                # Trailing Stop Update on Same Direction Signal
                if signal == -1:
                    atr_val = self.atr[0]
                    new_sl = self.data.close[0] + (atr_val * self.p.atr_mult)
                    if new_sl < self.sl_price:
                        self.sl_price = new_sl

            return

        # Entry logic
        if self.daily_halted or self.drawdown_halted:
            return

        atr_val = self.atr[0]
        if atr_val <= 0:
            return

        if signal == 1:  # BUY
            sl_dist = atr_val * self.p.atr_mult
            tp_dist = sl_dist * self.p.rr_ratio
            entry_price = self.data.close[0]

            self.sl_price = entry_price - sl_dist
            self.tp_price = entry_price + tp_dist

            risk_budget = current_equity * self.p.risk_per_trade
            size = risk_budget / sl_dist
            expected_risk = sl_dist * size

            self.trade_dir = 'LONG'
            self.entry_price = entry_price
            self.trade_size = size

            self.order = self.buy(size=size)

        elif signal == -1:  # SELL
            sl_dist = atr_val * self.p.atr_mult
            tp_dist = sl_dist * self.p.rr_ratio
            entry_price = self.data.close[0]

            self.sl_price = entry_price + sl_dist
            self.tp_price = entry_price - tp_dist

            risk_budget = current_equity * self.p.risk_per_trade
            size = risk_budget / sl_dist
            expected_risk = sl_dist * size

            self.trade_dir = 'SHORT'
            self.entry_price = entry_price
            self.trade_size = size

            self.order = self.sell(size=size)
```
