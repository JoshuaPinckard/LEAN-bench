```python
import backtrader as bt
import math
import collections


class AITradingStrategy(bt.Strategy):

    params = (
        ('atr_period', 14),
        ('atr_multiplier', 1.5),
        ('risk_per_trade', 0.01),
        ('rr_ratio', 2.0),
        ('max_daily_risk_pct', 0.02),
        ('max_drawdown_pct', 0.05),
        ('signal_history_len', 20),
        ('fast_ema', 9),
        ('slow_ema', 21),
        ('mid_ema', 50),
        ('rsi_period', 14),
        ('macd_fast', 12),
        ('macd_slow', 26),
        ('macd_signal', 9),
        ('bb_period', 20),
        ('bb_dev', 2.0),
        ('stoch_period', 14),
        ('stoch_period_dfast', 3),
        ('mom_period', 10),
        ('wr_period', 14),
        ('close_before_news', False),
    )

    def __init__(self):
        c = self.data.close

        self.ema_fast = bt.indicators.EMA(c, period=self.p.fast_ema)
        self.ema_slow = bt.indicators.EMA(c, period=self.p.slow_ema)
        self.ema_mid = bt.indicators.EMA(c, period=self.p.mid_ema)
        self.rsi = bt.indicators.RSI(c, period=self.p.rsi_period)
        self.macd_ind = bt.indicators.MACD(
            c,
            period1=self.p.macd_fast,
            period2=self.p.macd_slow,
            period3=self.p.macd_signal,
        )
        self.bb = bt.indicators.BollingerBands(
            c, period=self.p.bb_period, devfactor=self.p.bb_dev
        )
        self.stoch = bt.indicators.Stochastic(
            self.data,
            period=self.p.stoch_period,
            period_dfast=self.p.stoch_period_dfast,
        )
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.mom = bt.indicators.Momentum(c, period=self.p.mom_period)
        self.wpr = bt.indicators.WilliamsR(self.data, period=self.p.wr_period)

        self.order = None
        self.sl_price = None
        self.tp_price = None

        # Rolling history of generated signals for BUY/SELL/HOLD imbalance features
        self.signal_history = collections.deque(maxlen=self.p.signal_history_len)

        self.current_date = None
        self.daily_start_equity = None
        self.peak_equity = None
        self.halted_daily = False
        self.halted_dd = False

        self.trade_log = []

    # ------------------------------------------------------------------
    # Risk-management helpers
    # ------------------------------------------------------------------

    def _reset_if_new_day(self):
        bar_date = self.data.datetime.date(0)
        if bar_date != self.current_date:
            self.current_date = bar_date
            equity = self.broker.getvalue()
            self.daily_start_equity = equity
            self.peak_equity = equity
            self.halted_daily = False
            self.halted_dd = False

    def _update_halts(self):
        equity = self.broker.getvalue()
        if equity > self.peak_equity:
            self.peak_equity = equity
        if self.daily_start_equity and self.daily_start_equity > 0:
            daily_loss = (self.daily_start_equity - equity) / self.daily_start_equity
            if daily_loss >= self.p.max_daily_risk_pct:
                self.halted_daily = True
        if self.peak_equity and self.peak_equity > 0:
            dd = (self.peak_equity - equity) / self.peak_equity
            if dd >= self.p.max_drawdown_pct:
                self.halted_dd = True

    def _trading_allowed(self):
        return not self.halted_daily and not self.halted_dd

    # ------------------------------------------------------------------
    # Signal generation (LSTM proxy — see assumptions)
    # ------------------------------------------------------------------

    def _get_signal(self):
        n = len(self.signal_history)
        buy_frac = sum(1 for s in self.signal_history if s == 'BUY') / n if n else 0.0
        sell_frac = sum(1 for s in self.signal_history if s == 'SELL') / n if n else 0.0

        c = self.data.close[0]
        ema_bull = self.ema_fast[0] > self.ema_slow[0]
        macd_bull = self.macd_ind.macd[0] > self.macd_ind.signal[0]
        rsi_v = self.rsi[0]
        stoch_k = self.stoch.percK[0]
        mom_v = self.mom[0]
        wpr_v = self.wpr[0]
        bb_mid = self.bb.mid[0]
        mid_bull = c > self.ema_mid[0]

        if (ema_bull and macd_bull and mom_v > 0
                and 40 < rsi_v < 70
                and stoch_k < 80
                and wpr_v > -80
                and c > bb_mid
                and mid_bull
                and buy_frac < 0.6):
            return 'BUY'

        if (not ema_bull and not macd_bull and mom_v < 0
                and 30 < rsi_v < 60
                and stoch_k > 20
                and wpr_v < -20
                and c < bb_mid
                and not mid_bull
                and sell_frac < 0.6):
            return 'SELL'

        return 'HOLD'

    # ------------------------------------------------------------------
    # Position sizing
    # ------------------------------------------------------------------

    def _calc_size(self, sl_dist):
        equity = self.broker.getvalue()
        risk_amt = equity * self.p.risk_per_trade
        return max(1, int(risk_amt / sl_dist))

    # ------------------------------------------------------------------
    # Trade log
    # ------------------------------------------------------------------

    def _log(self, decision, size=None, sl=None, tp=None, risk=None):
        dt = self.data.datetime.datetime(0)
        price = self.data.close[0]
        self.trade_log.append({
            'datetime': dt,
            'decision': decision,
            'price': price,
            'size': size,
            'sl': sl,
            'tp': tp,
            'risk': risk,
        })
        if risk is not None:
            print(
                f"[{dt}] {decision} price={price:.2f} size={size} "
                f"SL={sl:.2f} TP={tp:.2f} risk={risk:.2f}"
            )

    # ------------------------------------------------------------------
    # Order lifecycle
    # ------------------------------------------------------------------

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return
        if order.status == order.Completed:
            self.order = None
        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            self.order = None
            self.sl_price = None
            self.tp_price = None

    # ------------------------------------------------------------------
    # Main bar logic
    # ------------------------------------------------------------------

    def next(self):
        self._reset_if_new_day()
        self._update_halts()

        if self.order:
            return

        signal = self._get_signal()
        self.signal_history.append(signal)

        high = self.data.high[0]
        low = self.data.low[0]
        close = self.data.close[0]
        sl_dist = self.atr[0] * self.p.atr_multiplier
        if sl_dist <= 0:
            return

        if self.position:
            is_long = self.position.size > 0

            if is_long:
                # SL breached (bar low touched or crossed SL level)
                if self.sl_price is not None and low <= self.sl_price:
                    self._log('CLOSE_SL')
                    self.order = self.close()
                    self.sl_price = self.tp_price = None
                    return
                # TP breached
                if self.tp_price is not None and high >= self.tp_price:
                    self._log('CLOSE_TP')
                    self.order = self.close()
                    self.sl_price = self.tp_price = None
                    return
                # Opposite signal — close immediately
                if signal == 'SELL':
                    self._log('CLOSE_OPPOSITE')
                    self.order = self.close()
                    self.sl_price = self.tp_price = None
                    return
                # Same-direction signal — trail stop upward
                if signal == 'BUY' and self.sl_price is not None:
                    new_sl = close - sl_dist
                    if new_sl > self.sl_price:
                        self.sl_price = new_sl

            else:  # short position
                # SL breached (bar high touched or crossed SL level)
                if self.sl_price is not None and high >= self.sl_price:
                    self._log('CLOSE_SL')
                    self.order = self.close()
                    self.sl_price = self.tp_price = None
                    return
                # TP breached
                if self.tp_price is not None and low <= self.tp_price:
                    self._log('CLOSE_TP')
                    self.order = self.close()
                    self.sl_price = self.tp_price = None
                    return
                # Opposite signal — close immediately
                if signal == 'BUY':
                    self._log('CLOSE_OPPOSITE')
                    self.order = self.close()
                    self.sl_price = self.tp_price = None
                    return
                # Same-direction signal — trail stop downward
                if signal == 'SELL' and self.sl_price is not None:
                    new_sl = close + sl_dist
                    if new_sl < self.sl_price:
                        self.sl_price = new_sl

        else:
            if not self._trading_allowed():
                self._log('HOLD')
                return

            if signal == 'BUY':
                sl = close - sl_dist
                tp = close + sl_dist * self.p.rr_ratio
                size = self._calc_size(sl_dist)
                risk = sl_dist * size
                self._log('BUY', size=size, sl=sl, tp=tp, risk=risk)
                self.order = self.buy(size=size)
                self.sl_price = sl
                self.tp_price = tp

            elif signal == 'SELL':
                sl = close + sl_dist
                tp = close - sl_dist * self.p.rr_ratio
                size = self._calc_size(sl_dist)
                risk = sl_dist * size
                self._log('SELL', size=size, sl=sl, tp=tp, risk=risk)
                self.order = self.sell(size=size)
                self.sl_price = sl
                self.tp_price = tp

            else:
                self._log('HOLD')
```
