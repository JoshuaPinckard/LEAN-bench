import backtrader as bt
import collections


class QCBStrategy(bt.Strategy):

    def __init__(self):
        self.order = None
        self.entry_price = None

        # 4H bar aggregation (4 consecutive 1H bars)
        self._h1_count = 0
        self._h4_o = self._h4_h = self._h4_l = self._h4_c = self._h4_v = None

        # Completed 4H bar history; each entry: {open,high,low,close,volume,rsi}
        self._bars = collections.deque(maxlen=200)

        # Wilder RSI(14) state
        self._rsi_prev = None
        self._rsi_ag = 0.0
        self._rsi_al = 0.0
        self._rsi_init_g = []
        self._rsi_init_l = []
        self._rsi_ready = False

    # ------------------------------------------------------------------ #
    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled,
                            order.Margin, order.Rejected]:
            if order.status == order.Completed and order.isbuy():
                self.entry_price = order.executed.price
            self.order = None

    # ------------------------------------------------------------------ #
    def _rsi(self, close):
        """Wilder RSI(14) on 4H closes; returns value or None during warm-up."""
        if self._rsi_prev is None:
            self._rsi_prev = close
            return None
        chg = close - self._rsi_prev
        self._rsi_prev = close
        g = chg if chg > 0 else 0.0
        l = -chg if chg < 0 else 0.0
        if not self._rsi_ready:
            self._rsi_init_g.append(g)
            self._rsi_init_l.append(l)
            if len(self._rsi_init_g) == 14:
                self._rsi_ag = sum(self._rsi_init_g) / 14.0
                self._rsi_al = sum(self._rsi_init_l) / 14.0
                self._rsi_ready = True
            else:
                return None
        else:
            self._rsi_ag = (self._rsi_ag * 13.0 + g) / 14.0
            self._rsi_al = (self._rsi_al * 13.0 + l) / 14.0
        if self._rsi_al == 0.0:
            return 100.0
        return 100.0 - 100.0 / (1.0 + self._rsi_ag / self._rsi_al)

    # ------------------------------------------------------------------ #
    def _swing_lows(self, w=2):
        bars = list(self._bars)
        n = len(bars)
        idx = []
        for i in range(w, n - w):
            v = bars[i]['low']
            if (all(v < bars[i - j]['low'] for j in range(1, w + 1)) and
                    all(v < bars[i + j]['low'] for j in range(1, w + 1))):
                idx.append(i)
        return idx

    def _swing_highs(self, w=2):
        bars = list(self._bars)
        n = len(bars)
        idx = []
        for i in range(w, n - w):
            v = bars[i]['high']
            if (all(v > bars[i - j]['high'] for j in range(1, w + 1)) and
                    all(v > bars[i + j]['high'] for j in range(1, w + 1))):
                idx.append(i)
        return idx

    def _bull_div(self):
        sw = self._swing_lows()
        if len(sw) < 2:
            return False
        bars = list(self._bars)
        i1, i2 = sw[-2], sw[-1]
        r1, r2 = bars[i1]['rsi'], bars[i2]['rsi']
        if r1 is None or r2 is None:
            return False
        return bars[i2]['low'] < bars[i1]['low'] and r2 > r1

    def _bear_div(self):
        sw = self._swing_highs()
        if len(sw) < 2:
            return False
        bars = list(self._bars)
        i1, i2 = sw[-2], sw[-1]
        r1, r2 = bars[i1]['rsi'], bars[i2]['rsi']
        if r1 is None or r2 is None:
            return False
        return bars[i2]['high'] > bars[i1]['high'] and r2 < r1

    # ------------------------------------------------------------------ #
    def _on_h4_bar(self):
        bars = list(self._bars)
        cur = bars[-1]
        n = len(bars)

        if not self.position and self.order is None:
            dt = self.data.datetime.datetime(0)
            if not (dt.year == 2025 and dt.month in (11, 12)):
                return
            if cur['rsi'] is None or n < 21:
                return
            # (1) Bullish divergence
            if not self._bull_div():
                return
            # (2) Close above high of last 20 bars (prior 20, not current)
            resistance = max(b['high'] for b in bars[-21:-1])
            if cur['close'] <= resistance:
                return
            # (3) Volume > SMA(volume, 20) of prior 20 bars
            vol_sma = sum(b['volume'] for b in bars[-21:-1]) / 20.0
            if cur['volume'] <= vol_sma:
                return
            # Position size: 1% of capital / (2% × entry_price)
            capital = self.broker.getvalue()
            size = int((0.01 * capital) / (0.02 * cur['close']))
            if size > 0:
                self.order = self.buy(size=size)

        elif self.position and self.order is None:
            # Alternative exit: RSI bearish divergence
            if self._bear_div():
                self.order = self.close()
                return
            # Alternative exit: close below low of last 10 bars (prior 10)
            if n >= 11:
                low_10 = min(b['low'] for b in bars[-11:-1])
                if cur['close'] < low_10:
                    self.order = self.close()

    # ------------------------------------------------------------------ #
    def next(self):
        o = self.data.open[0]
        h = self.data.high[0]
        l = self.data.low[0]
        c = self.data.close[0]
        v = self.data.volume[0]

        # Aggregate 1H bars into 4H bars
        if self._h1_count == 0:
            self._h4_o, self._h4_h, self._h4_l, self._h4_v = o, h, l, v
        else:
            if h > self._h4_h:
                self._h4_h = h
            if l < self._h4_l:
                self._h4_l = l
            self._h4_v += v
        self._h4_c = c
        self._h1_count += 1

        if self._h1_count == 4:
            self._bars.append({
                'open':   self._h4_o,
                'high':   self._h4_h,
                'low':    self._h4_l,
                'close':  self._h4_c,
                'volume': self._h4_v,
                'rsi':    self._rsi(self._h4_c),
            })
            self._on_h4_bar()
            self._h1_count = 0

        # Stop-loss and take-profit checked on every 1H bar using intrabar high/low
        if self.position and self.order is None and self.entry_price is not None:
            sl = self.entry_price * 0.98
            tp = self.entry_price * 1.06
            if self.data.low[0] <= sl or self.data.high[0] >= tp:
                self.order = self.close()
