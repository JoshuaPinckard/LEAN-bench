import backtrader as bt


class RsiDivergenceBreakout(bt.Strategy):
    def __init__(self):
        self.rsi = bt.indicators.RSI(self.data.close, period=14)
        self.vol_sma = bt.indicators.SMA(self.data.volume, period=20)

        self.order = None
        self.sl_order = None
        self.tp_order = None

        self.entry_price = None
        self.bullish_div_active = False

        self.last_local_low = None  # (price, rsi)
        self.last_local_high = None  # (price, rsi)

    def notify_order(self, order):
        if order.status in [order.Completed]:
            if order.isbuy():
                self.entry_price = order.executed.price
                sl_price = self.entry_price * 0.98
                tp_price = self.entry_price * 1.06
                self.sl_order = self.sell(
                    exectype=bt.Order.Stop,
                    price=sl_price,
                    size=order.executed.size,
                )
                self.tp_order = self.sell(
                    exectype=bt.Order.Limit,
                    price=tp_price,
                    size=order.executed.size,
                )
            else:
                if self.sl_order and order.ref == self.sl_order.ref:
                    if self.tp_order:
                        self.cancel(self.tp_order)
                elif self.tp_order and order.ref == self.tp_order.ref:
                    if self.sl_order:
                        self.cancel(self.sl_order)
                self.sl_order = None
                self.tp_order = None
                self.entry_price = None

            self.order = None

        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            if self.order and order.ref == self.order.ref:
                self.order = None
            if self.sl_order and order.ref == self.sl_order.ref:
                self.sl_order = None
            if self.tp_order and order.ref == self.tp_order.ref:
                self.tp_order = None

    def next(self):
        if len(self) < 25:
            return

        # Local low detection at bar -2 (5-bar pivot window)
        low_m2 = self.data.low[-2]
        if (
            low_m2 < self.data.low[-4]
            and low_m2 < self.data.low[-3]
            and low_m2 < self.data.low[-1]
            and low_m2 < self.data.low[0]
        ):
            curr_low_price = low_m2
            curr_low_rsi = self.rsi[-2]
            if self.last_local_low is not None:
                prev_low_price, prev_low_rsi = self.last_local_low
                if (
                    curr_low_price < prev_low_price
                    and curr_low_rsi > prev_low_rsi
                ):
                    self.bullish_div_active = True
                else:
                    self.bullish_div_active = False
            self.last_local_low = (curr_low_price, curr_low_rsi)

        # Local high detection at bar -2 (5-bar pivot window)
        bearish_div = False
        high_m2 = self.data.high[-2]
        if (
            high_m2 > self.data.high[-4]
            and high_m2 > self.data.high[-3]
            and high_m2 > self.data.high[-1]
            and high_m2 > self.data.high[0]
        ):
            curr_high_price = high_m2
            curr_high_rsi = self.rsi[-2]
            if self.last_local_high is not None:
                prev_high_price, prev_high_rsi = self.last_local_high
                if (
                    curr_high_price > prev_high_price
                    and curr_high_rsi < prev_high_rsi
                ):
                    bearish_div = True
            self.last_local_high = (curr_high_price, curr_high_rsi)

        # Time filter: November - December 2025
        dt = self.data.datetime.datetime(0)
        in_time_window = dt.year == 2025 and dt.month in (11, 12)

        # High of the last 20 bars prior to current bar
        high_20 = max([self.data.high[-i] for i in range(1, 21)])

        # Low of the last 10 bars prior to current bar
        low_10 = min([self.data.low[-i] for i in range(1, 11)])

        # Long entry conditions
        if not self.position and not self.order:
            if in_time_window and self.bullish_div_active:
                breakout = self.data.close[0] > high_20
                vol_confirm = self.data.volume[0] > self.vol_sma[0]

                if breakout and vol_confirm:
                    close_price = self.data.close[0]
                    capital = self.broker.getvalue()
                    risk_amount = 0.01 * capital
                    risk_per_share = 0.02 * close_price

                    if risk_per_share > 0:
                        size = int(risk_amount / risk_per_share)
                        if size > 0:
                            self.order = self.buy(size=size)
                            self.bullish_div_active = False

        # Alternative exit conditions while in position
        elif self.position and not self.order:
            close_below_10_low = self.data.close[0] < low_10

            if bearish_div or close_below_10_low:
                if self.sl_order:
                    self.cancel(self.sl_order)
                    self.sl_order = None
                if self.tp_order:
                    self.cancel(self.tp_order)
                    self.tp_order = None
                self.order = self.close()
