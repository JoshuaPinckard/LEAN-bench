import backtrader as bt


class FibonacciStructureStrategy(bt.Strategy):
    def __init__(self):
        self.ha = bt.indicators.HeikinAshi(self.data)
        self.addminperiod(3)

        self._entry_order = None
        self._stop_order = None
        self._target_order = None
        self._entry_setup = None

        self._last_pivot_type = None
        self._last_pivot_price = None
        self._direction = None
        self._fib_zero = None
        self._fib_618 = None
        self._zone_low = None
        self._zone_high = None

    def next(self):
        high_pivot = (
            self.data.high[-1] > self.data.high[-2]
            and self.data.high[-1] > self.data.high[0]
        )
        low_pivot = (
            self.data.low[-1] < self.data.low[-2]
            and self.data.low[-1] < self.data.low[0]
        )

        if high_pivot and not low_pivot:
            pivot_price = self.data.high[-1]
            if self._last_pivot_type == "low":
                low = self._last_pivot_price
                high = pivot_price
                span = high - low
                self._direction = "up"
                self._fib_zero = low
                self._fib_618 = low + 0.618 * span
                self._zone_low = low + 0.236 * span
                self._zone_high = low + 0.786 * span
            self._last_pivot_type = "high"
            self._last_pivot_price = pivot_price

        elif low_pivot and not high_pivot:
            pivot_price = self.data.low[-1]
            if self._last_pivot_type == "high":
                high = self._last_pivot_price
                low = pivot_price
                span = high - low
                self._direction = "down"
                self._fib_zero = high
                self._fib_618 = high - 0.618 * span
                self._zone_low = high - 0.786 * span
                self._zone_high = high - 0.236 * span
            self._last_pivot_type = "low"
            self._last_pivot_price = pivot_price

        if (
            self.position
            or self._entry_order
            or self._stop_order
            or self._target_order
            or self._direction is None
        ):
            return

        in_zone = self._zone_low <= self.data.close[0] <= self._zone_high

        if (
            self._direction == "down"
            and in_zone
            and self.ha.ha_low[0] == self.ha.ha_open[0]
        ):
            self._entry_setup = ("long", self._fib_618, self._fib_zero)
            self._entry_order = self.buy()

        elif (
            self._direction == "up"
            and in_zone
            and self.ha.ha_high[0] == self.ha.ha_open[0]
        ):
            self._entry_setup = ("short", self._fib_618, self._fib_zero)
            self._entry_order = self.sell()

    def notify_order(self, order):
        if order.status == order.Completed:
            if order is self._entry_order:
                direction, stop_price, target_price = self._entry_setup
                self._entry_order = None
                self._entry_setup = None

                if direction == "long":
                    self._stop_order = self.sell(
                        exectype=bt.Order.Stop, price=stop_price
                    )
                    self._target_order = self.sell(
                        exectype=bt.Order.Limit,
                        price=target_price,
                        oco=self._stop_order,
                    )
                else:
                    self._stop_order = self.buy(
                        exectype=bt.Order.Stop, price=stop_price
                    )
                    self._target_order = self.buy(
                        exectype=bt.Order.Limit,
                        price=target_price,
                        oco=self._stop_order,
                    )

            elif order is self._stop_order:
                self._stop_order = None

            elif order is self._target_order:
                self._target_order = None

        elif order.status in (
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            if order is self._entry_order:
                self._entry_order = None
                self._entry_setup = None
            elif order is self._stop_order:
                self._stop_order = None
            elif order is self._target_order:
                self._target_order = None
