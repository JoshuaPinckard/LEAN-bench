import backtrader as bt


class MomentumBollingerRSI(bt.Strategy):
    def __init__(self):
        self.boll = bt.indicators.BollingerBands(
            self.data.close, period=20, devfactor=2
        )
        self.rsi = bt.indicators.RSI(self.data.close, period=14)

        self.order = None
        self.entry_price = None
        self.highest_price = None
        self.trailing_stop_loss = None
        self.exit_signal = 0
        self.is_entry_day = False

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            if order.isbuy():
                self.entry_price = order.executed.price
                self.trailing_stop_loss = self.entry_price * 0.99
                self.highest_price = self.entry_price
                self.is_entry_day = True
            elif order.issell():
                self.trailing_stop_loss = None
                self.highest_price = None
                self.entry_price = None

        self.order = None

    def next(self):
        self.exit_signal = 0

        if self.position:
            if self.is_entry_day:
                self.highest_price = max(self.highest_price, self.data.high[0])
                self.trailing_stop_loss = max(
                    self.trailing_stop_loss, self.highest_price * 0.99
                )
                self.is_entry_day = False
            else:
                self.highest_price = max(self.highest_price, self.data.high[0])
                self.trailing_stop_loss = max(
                    self.trailing_stop_loss, self.highest_price * 0.99
                )

            if self.data.close[0] <= self.trailing_stop_loss:
                self.exit_signal = 1
                if self.order is None:
                    self.order = self.close()
        else:
            self.trailing_stop_loss = None
            self.highest_price = None

            if self.order is None:
                cross_above = (self.data.close[0] > self.boll.top[0]) and (
                    self.data.close[-1] <= self.boll.top[-1]
                )
                rsi_valid = self.rsi[0] > 55

                if cross_above and rsi_valid:
                    self.order = self.buy(size=1)
