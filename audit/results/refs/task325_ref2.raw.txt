import backtrader as bt


class IntradayORBStrategy(bt.Strategy):
    def __init__(self):
        self.sma_vol = bt.indicators.SMA(self.data.volume, period=20)
        self.atr = bt.indicators.ATR(self.data, period=14)
        self.rsi = bt.indicators.RSI(self.data.close, period=14)

        self.order = None
        self.stop_order = None

        self.current_date = None
        self.day_bars = []
        self.or_high = None
        self.or_low = None
        self.or_ready = False

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            if order == self.order:
                self.order = None
                if self.position.size > 0:
                    trail_amount = 1.5 * self.atr[0]
                    self.stop_order = self.sell(
                        exectype=bt.Order.StopTrail, trailamount=trail_amount
                    )
                elif self.position.size < 0:
                    trail_amount = 1.5 * self.atr[0]
                    self.stop_order = self.buy(
                        exectype=bt.Order.StopTrail, trailamount=trail_amount
                    )
            elif order == self.stop_order:
                self.stop_order = None
        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            if order == self.order:
                self.order = None
            if order == self.stop_order:
                self.stop_order = None

    def next(self):
        dt = self.data.datetime.date(0)

        if dt != self.current_date:
            self.current_date = dt
            self.day_bars = []
            self.or_high = None
            self.or_low = None
            self.or_ready = False

        if len(self.day_bars) < 4:
            self.day_bars.append((self.data.high[0], self.data.low[0]))
            if len(self.day_bars) == 4:
                self.or_high = max(b[0] for b in self.day_bars)
                self.or_low = min(b[1] for b in self.day_bars)
            return

        self.or_ready = True

        t = self.data.datetime.time(0)
        is_rth = (t.hour > 9 or (t.hour == 9 and t.minute >= 30)) and (
            t.hour < 16 or (t.hour == 16 and t.minute == 0)
        )
        if not is_rth:
            return

        if self.sma_vol[0] == 0 or self.sma_vol[0] != self.sma_vol[0]:
            return
        rvol = self.data.volume[0] / self.sma_vol[0]
        rvol_high = rvol > 1.5

        long_breakout = self.data.close[0] > self.or_high and rvol_high
        short_breakout = self.data.close[0] < self.or_low and rvol_high

        rsi_cross_below_50 = (self.rsi[-1] >= 50) and (self.rsi[0] < 50)
        rsi_cross_above_50 = (self.rsi[-1] <= 50) and (self.rsi[0] > 50)

        if self.order:
            return

        if self.position.size > 0:
            if rsi_cross_below_50 or short_breakout:
                if self.stop_order:
                    self.cancel(self.stop_order)
                    self.stop_order = None
                self.order = self.close()
        elif self.position.size < 0:
            if rsi_cross_above_50 or long_breakout:
                if self.stop_order:
                    self.cancel(self.stop_order)
                    self.stop_order = None
                self.order = self.close()
        else:
            if long_breakout:
                self.order = self.buy()
            elif short_breakout:
                self.order = self.sell()
