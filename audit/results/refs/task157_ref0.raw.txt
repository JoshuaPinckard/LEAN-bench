import backtrader as bt
import collections


class CheapVolatilityStrategy(bt.Strategy):
    params = (
        ('bb1_period', 20),
        ('bb2_period', 100),
        ('ema_period', 20),
        ('sma_period', 50),
        ('rsi_period', 14),
        ('atr_period', 14),
        ('vol_sma_short', 20),
        ('vol_sma_long', 100),
        ('ratio_lookback', 252),
        ('ratio_percentile', 0.20),
        ('width_lookback', 21),
        ('vol_breakout_mult', 1.5),
        ('stop_atr_mult', 1.5),
        ('target_atr_mult', 3.0),
        ('trail_atr_mult', 1.0),
    )

    def __init__(self):
        self.bb20 = bt.indicators.BollingerBands(self.data, period=self.p.bb1_period)
        self.bb100 = bt.indicators.BollingerBands(self.data, period=self.p.bb2_period)
        self.ema20 = bt.indicators.EMA(self.data, period=self.p.ema_period)
        self.sma50 = bt.indicators.SMA(self.data, period=self.p.sma_period)
        self.rsi = bt.indicators.RSI(self.data, period=self.p.rsi_period)
        self.macd = bt.indicators.MACD(self.data)
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.vol_sma20 = bt.indicators.SMA(self.data.volume, period=self.p.vol_sma_short)
        self.vol_sma100 = bt.indicators.SMA(self.data.volume, period=self.p.vol_sma_long)

        self.bb20_width = self.bb20.top - self.bb20.bot
        self.bb100_width = self.bb100.top - self.bb100.bot

        self.ratio_hist = collections.deque(maxlen=self.p.ratio_lookback)
        self.width_hist = collections.deque(maxlen=self.p.width_lookback)

        self.order = None
        self.entry_price = None
        self.entry_atr = None
        self.stop_price = None
        self.target_price = None
        self.trail_extreme = None

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return
        if order.status in (order.Completed, order.Canceled, order.Margin, order.Rejected):
            self.order = None

    def next(self):
        bb100w = self.bb100_width[0]
        ratio = self.bb20_width[0] / bb100w if bb100w != 0 else None
        if ratio is not None:
            self.ratio_hist.append(ratio)
        self.width_hist.append(self.bb20_width[0])

        if self.order:
            return

        if self.position:
            self._manage_exit()
            return

        liquidity_ok = self.vol_sma20[0] > self.vol_sma100[0]
        if not liquidity_ok:
            return

        bullish = (self.ema20[0] > self.sma50[0]) and (self.rsi[0] > 55) and (self.macd.macd[0] > 0)
        bearish = (self.ema20[0] < self.sma50[0]) and (self.rsi[0] < 45) and (self.macd.macd[0] < 0)

        cheap_vol = False
        if ratio is not None and len(self.ratio_hist) == self.p.ratio_lookback:
            sorted_vals = sorted(self.ratio_hist)
            idx = int(self.p.ratio_percentile * (len(sorted_vals) - 1))
            threshold = sorted_vals[idx]
            cheap_vol = ratio <= threshold

        if cheap_vol and bullish:
            self.order = self.buy()
            return
        if cheap_vol and bearish:
            self.order = self.sell()
            return

        width_is_min = (len(self.width_hist) == self.p.width_lookback
                         and self.bb20_width[0] == min(self.width_hist))
        vol_breakout = self.data.volume[0] > self.p.vol_breakout_mult * self.vol_sma20[0]

        if width_is_min and vol_breakout:
            if self.data.close[0] > self.bb20.top[0]:
                self.order = self.buy()
            elif self.data.close[0] < self.bb20.bot[0]:
                self.order = self.sell()

    def _manage_exit(self):
        is_long = self.position.size > 0

        if self.entry_price is None:
            self.entry_price = self.position.price
            self.entry_atr = self.atr[0]
            if is_long:
                self.stop_price = self.entry_price - self.p.stop_atr_mult * self.entry_atr
                self.target_price = self.entry_price + self.p.target_atr_mult * self.entry_atr
            else:
                self.stop_price = self.entry_price + self.p.stop_atr_mult * self.entry_atr
                self.target_price = self.entry_price - self.p.target_atr_mult * self.entry_atr
            self.trail_extreme = self.entry_price

        close = self.data.close[0]

        if is_long:
            self.trail_extreme = max(self.trail_extreme, close)
            trail_stop = self.trail_extreme - self.p.trail_atr_mult * self.atr[0]
            reverse = self.ema20[0] < self.sma50[0]
            exit_now = (close <= self.stop_price or close >= self.target_price
                        or close <= trail_stop or reverse)
        else:
            self.trail_extreme = min(self.trail_extreme, close)
            trail_stop = self.trail_extreme + self.p.trail_atr_mult * self.atr[0]
            reverse = self.ema20[0] > self.sma50[0]
            exit_now = (close >= self.stop_price or close <= self.target_price
                        or close >= trail_stop or reverse)

        if exit_now:
            self.order = self.close()
            self.entry_price = None
            self.entry_atr = None
            self.stop_price = None
            self.target_price = None
            self.trail_extreme = None
