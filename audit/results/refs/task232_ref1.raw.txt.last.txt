import backtrader as bt
import math


class MeanReversionBollingerStrategy(bt.Strategy):
    def __init__(self):
        self.bollinger = bt.indicators.BollingerBands(
            self.data.close, period=20, devfactor=1.5
        )
        self.sma200 = bt.indicators.SimpleMovingAverage(self.data.close, period=200)
        self.atr = bt.indicators.ATR(self.data, period=14)

        self.lower_cross = bt.indicators.CrossOver(
            self.data.close, self.bollinger.lines.bot
        )
        self.upper_cross = bt.indicators.CrossOver(
            self.data.close, self.bollinger.lines.top
        )
        self.middle_cross = bt.indicators.CrossOver(
            self.data.close, self.bollinger.lines.mid
        )

        self.order = None
        self.entry_bar = None
        self.initial_capital = self.broker.getcash()

    def notify_order(self, order):
        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            if order.status == order.Completed:
                if self.position.size:
                    self.entry_bar = len(self)
                else:
                    self.entry_bar = None

            if order is self.order:
                self.order = None

    def next(self):
        if self.order:
            return

        if self.position.size:
            if self.position.size > 0:
                floating_pnl = (
                    self.data.close[0] - self.position.price
                ) * self.position.size
            else:
                floating_pnl = (
                    self.position.price - self.data.close[0]
                ) * abs(self.position.size)

            if floating_pnl < -(0.02 * self.initial_capital):
                self.order = self.close()
                return

            if (
                self.middle_cross[0] != 0
                or (
                    self.entry_bar is not None
                    and len(self) >= self.entry_bar + 5
                )
            ):
                self.order = self.close()
            return

        if len(self) < 265:
            return

        atr_values = sorted(float(self.atr[-i]) for i in range(252))
        percentile_position = (len(atr_values) - 1) * 0.20
        lower_index = int(math.floor(percentile_position))
        upper_index = int(math.ceil(percentile_position))
        fraction = percentile_position - lower_index
        atr_percentile = (
            atr_values[lower_index]
            + (atr_values[upper_index] - atr_values[lower_index]) * fraction
        )

        if self.atr[0] < atr_percentile:
            return

        size = 1000.0 / self.data.close[0]

        if (
            self.lower_cross[0] == -1
            and self.data.close[0] <= self.sma200[0]
        ):
            self.order = self.buy(size=size)
        elif (
            self.upper_cross[0] == 1
            and self.data.close[0] >= self.sma200[0]
        ):
            self.order = self.sell(size=size)
