import backtrader as bt


class SwingMomentumStrategy(bt.Strategy):
    params = dict(
        sma_trend_period=200,
        ema_fast_period=20,
        ema_slow_period=50,
        rsi_period=14,
        rsi_level=50,
        atr_period=14,
        volume_sma_period=20,
        require_volume_confirmation=False,
        take_profit_pct=0.18,
        initial_stop_pct=0.08,
        initial_stop_atr_mult=2.0,
        trail_atr_mult=3.0,
        reserve_pct=0.20,
        initial_entry_pct=0.60,
        addon_pct=0.20,
        max_position_pct=0.80,
        pullback_min_pct=0.10,
        pullback_max_pct=0.15,
    )

    def __init__(self):
        self.sma200 = bt.indicators.SMA(self.data.close, period=self.p.sma_trend_period)
        self.ema20 = bt.indicators.EMA(self.data.close, period=self.p.ema_fast_period)
        self.ema50 = bt.indicators.EMA(self.data.close, period=self.p.ema_slow_period)
        self.rsi = bt.indicators.RSI(self.data.close, period=self.p.rsi_period)
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.vol_sma = bt.indicators.SMA(self.data.volume, period=self.p.volume_sma_period)
        self.ema_cross = bt.indicators.CrossOver(self.ema20, self.ema50)

        self.order = None
        self.entry_price = None
        self.initial_stop = None
        self.highest_close = None
        self.added = False

    def _rsi_cross_up(self):
        return self.rsi[-1] < self.p.rsi_level and self.rsi[0] > self.p.rsi_level

    def _rsi_cross_down(self):
        return self.rsi[-1] > self.p.rsi_level and self.rsi[0] < self.p.rsi_level

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return
        if order.status == order.Completed:
            if order.isbuy():
                if self.entry_price is None:
                    self.entry_price = order.executed.price
                    self.initial_stop = self.entry_price - max(
                        self.p.initial_stop_pct * self.entry_price,
                        self.p.initial_stop_atr_mult * self.atr[0],
                    )
            elif order.issell():
                self.entry_price = None
                self.initial_stop = None
                self.highest_close = None
                self.added = False
        self.order = None

    def next(self):
        if self.order:
            return

        close = self.data.close[0]

        if not self.position:
            if close <= self.sma200[0]:
                return
            ema_up = self.ema_cross[0] > 0
            rsi_up = self._rsi_cross_up()
            if not (ema_up or rsi_up):
                return
            if self.p.require_volume_confirmation and not (self.data.volume[0] > self.vol_sma[0]):
                return
            value = self.broker.getvalue() * self.p.initial_entry_pct
            size = int(value / close)
            if size > 0:
                self.order = self.buy(size=size)
            return

        if self.highest_close is None:
            self.highest_close = close
        else:
            self.highest_close = max(self.highest_close, close)

        trailing_stop = max(self.initial_stop, self.highest_close - self.p.trail_atr_mult * self.atr[0])
        take_profit_price = self.position.price * (1.0 + self.p.take_profit_pct)
        trend_reversal = (self.ema_cross[0] < 0) or self._rsi_cross_down()

        if close <= trailing_stop or close >= take_profit_price or trend_reversal:
            self.order = self.close()
            return

        if not self.added:
            pullback = (self.highest_close - close) / self.highest_close
            trend_ok = close > self.sma200[0] and self.ema20[0] > self.ema50[0]
            if self.p.pullback_min_pct <= pullback <= self.p.pullback_max_pct and trend_ok:
                portfolio_value = self.broker.getvalue()
                current_position_value = self.position.size * close
                max_value = portfolio_value * self.p.max_position_pct
                addon_budget = min(portfolio_value * self.p.addon_pct, max_value - current_position_value)
                if addon_budget > 0:
                    size = int(addon_budget / close)
                    if size > 0:
                        self.order = self.buy(size=size)
                        self.added = True
