import backtrader as bt


class MacdHistoAtrStrategy(bt.Strategy):
    params = dict(
        macd1=12,
        macd2=26,
        macdsig=9,
        atr_period=14,
        histo_threshold=0.5,
        atr_mult=2.0,
        stake=100,
    )

    def __init__(self):
        self.macd = bt.indicators.MACDHisto(
            self.data,
            period_me1=self.p.macd1,
            period_me2=self.p.macd2,
            period_signal=self.p.macdsig,
        )
        self.histo = self.macd.histo
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)

        self.order = None
        self.entry_order = False
        self.stop_price = None

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return

        if order.status == order.Completed:
            if self.entry_order:
                if order.isbuy():
                    self.stop_price = order.executed.price - self.p.atr_mult * self.atr[0]
                else:
                    self.stop_price = order.executed.price + self.p.atr_mult * self.atr[0]
            else:
                self.stop_price = None

        self.order = None
        self.entry_order = False

    def next(self):
        if self.order:
            return

        histo0 = self.histo[0]
        histo1 = self.histo[-1]

        cross_up = histo1 <= self.p.histo_threshold and histo0 > self.p.histo_threshold
        cross_down = histo1 >= -self.p.histo_threshold and histo0 < -self.p.histo_threshold

        if self.position:
            if self.position.size > 0:
                stop_hit = self.stop_price is not None and self.data.low[0] <= self.stop_price
                if stop_hit or cross_down:
                    self.entry_order = False
                    self.order = self.close()
            elif self.position.size < 0:
                stop_hit = self.stop_price is not None and self.data.high[0] >= self.stop_price
                if stop_hit or cross_up:
                    self.entry_order = False
                    self.order = self.close()
        else:
            if cross_up:
                self.entry_order = True
                self.order = self.buy(size=self.p.stake)
            elif cross_down:
                self.entry_order = True
                self.order = self.sell(size=self.p.stake)
