import backtrader as bt


class EmaStochScalpingStrategy(bt.Strategy):
    params = dict(
        ema_period=20,
        stoch_period=14,
        stoch_pfast=3,
        stoch_pslow=3,
        oversold=20,
        overbought=80,
        take_profit=0.005,
        stop_loss=0.003,
        risk_pct=0.01,
    )

    def __init__(self):
        self.ema = bt.indicators.ExponentialMovingAverage(
            self.data.close, period=self.p.ema_period
        )
        self.stoch = bt.indicators.Stochastic(
            self.data,
            period=self.p.stoch_period,
            period_dfast=self.p.stoch_pfast,
            period_dslow=self.p.stoch_pslow,
        )
        self.crossover = bt.indicators.CrossOver(self.stoch.percK, self.stoch.percD)
        self.order = None

    def notify_order(self, order):
        if order.status in (order.Completed, order.Canceled, order.Margin, order.Rejected):
            self.order = None

    def next(self):
        if self.order is not None:
            return

        if self.position:
            entry_price = self.position.price
            if self.position.size > 0:
                tp_price = entry_price * (1 + self.p.take_profit)
                sl_price = entry_price * (1 - self.p.stop_loss)
                if (
                    self.crossover[0] < 0
                    or self.data.high[0] >= tp_price
                    or self.data.low[0] <= sl_price
                ):
                    self.order = self.close()
            else:
                tp_price = entry_price * (1 - self.p.take_profit)
                sl_price = entry_price * (1 + self.p.stop_loss)
                if (
                    self.crossover[0] > 0
                    or self.data.low[0] <= tp_price
                    or self.data.high[0] >= sl_price
                ):
                    self.order = self.close()
            return

        k = self.stoch.percK[0]
        d = self.stoch.percD[0]

        if (
            self.data.close[0] > self.ema[0]
            and self.crossover[0] > 0
            and k < self.p.oversold
            and d < self.p.oversold
        ):
            size = (self.p.risk_pct * self.broker.getvalue()) / self.data.close[0]
            self.order = self.buy(size=size)
        elif (
            self.data.close[0] < self.ema[0]
            and self.crossover[0] < 0
            and k > self.p.overbought
            and d > self.p.overbought
        ):
            size = (self.p.risk_pct * self.broker.getvalue()) / self.data.close[0]
            self.order = self.sell(size=size)
