import math
import backtrader as bt


class MarketOverheatingIndicator(bt.Indicator):
    lines = ('state', 'value')
    params = (
        ('period', 200),
        ('heavy_threshold', 1.5),
        ('overheat_threshold', 2.4),
    )

    def __init__(self):
        sma = bt.ind.SMA(self.data.close, period=self.p.period)
        self.lines.value = self.data.close / sma

    def next(self):
        val = self.lines.value[0]
        if val <= self.p.heavy_threshold:
            self.lines.state[0] = 1.0  # Stay heavy
        elif val <= self.p.overheat_threshold:
            self.lines.state[0] = 3.0  # Initial phase of overheating
        else:
            self.lines.state[0] = 2.0  # Stay light


class MarketOverheatingStrategy(bt.Strategy):
    params = (
        ('period', 200),
        ('heavy_threshold', 1.5),
        ('overheat_threshold', 2.4),
        ('qld_weight', 0.5),
        ('tqqq_weight', 0.5),
    )

    def __init__(self):
        self.indicator = MarketOverheatingIndicator(
            self.data,
            period=self.p.period,
            heavy_threshold=self.p.heavy_threshold,
            overheat_threshold=self.p.overheat_threshold,
        )
        self.order = None

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        if self.order:
            return

        current_state = self.indicator.state[0]
        try:
            prev_state = self.indicator.state[-1]
        except IndexError:
            prev_state = None

        if prev_state is not None and math.isnan(prev_state):
            prev_state = None

        # State 1: Stay Heavy -> Maximum position size (100% of capital)
        if current_state == 1.0:
            if prev_state != 1.0 or not self.position:
                self.order = self.order_target_percent(target=1.0)

        # State 2: Stay Light -> Minimum position size (0% / close position)
        elif current_state == 2.0:
            if prev_state != 2.0 or self.position:
                self.order = self.order_target_percent(target=0.0)
