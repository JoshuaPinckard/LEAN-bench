import backtrader as bt


class JanuaryEffectStrategy(bt.Strategy):
    def __init__(self):
        self.sma20 = bt.indicators.SMA(self.data.close, period=20)

        self.buy_order = None
        self.stop_order = None
        self.exit_order = None
        self.entry_price = None
        self.prev_month = None

    def notify_order(self, order):
        if order.status == order.Completed:
            if order == self.buy_order:
                self.entry_price = order.executed.price
                stop_price = self.entry_price * 0.98
                self.stop_order = self.sell(
                    exectype=bt.Order.Stop,
                    price=stop_price,
                    size=order.executed.size,
                )
                self.buy_order = None
            elif order == self.stop_order:
                self.stop_order = None
                self.entry_price = None
            elif order == self.exit_order:
                if self.stop_order is not None:
                    self.cancel(self.stop_order)
                    self.stop_order = None
                self.exit_order = None
                self.entry_price = None
        elif order.status in (order.Canceled, order.Margin, order.Rejected):
            if order == self.buy_order:
                self.buy_order = None
            elif order == self.stop_order:
                self.stop_order = None
            elif order == self.exit_order:
                self.exit_order = None

    def next(self):
        current_date = self.data.datetime.date(0)
        current_month = current_date.month
        is_month_start = self.prev_month is None or current_month != self.prev_month
        self.prev_month = current_month

        if not is_month_start:
            return

        if current_month == 1 and not self.position and self.buy_order is None:
            size = int(self.broker.getcash() / self.data.close[0])
            if size > 0:
                self.buy_order = self.buy(size=size)
        elif current_month == 2 and self.position and self.exit_order is None:
            if self.stop_order is not None:
                self.cancel(self.stop_order)
            self.exit_order = self.close()
