import backtrader as bt


class MeanReversionATRStrategy(bt.Strategy):
    def __init__(self):
        self.sma = bt.indicators.SimpleMovingAverage(self.data.close, period=20)
        self.atr = bt.indicators.AverageTrueRange(self.data, period=14)
        self.order = None
        self.stop_order = None
        self.entry_price = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status == order.Completed:
            if order == self.stop_order:
                self.stop_order = None
                self.entry_price = None
                self.order = None
            elif not self.position:
                self.stop_order = None
                self.entry_price = None
                self.order = None
            else:
                self.entry_price = order.executed.price
                atr_val = self.atr[0]
                if self.position.size > 0:
                    stop_price = self.entry_price - 2 * atr_val
                    self.stop_order = self.sell(
                        exectype=bt.Order.Stop, price=stop_price, size=self.position.size
                    )
                else:
                    stop_price = self.entry_price + 2 * atr_val
                    self.stop_order = self.buy(
                        exectype=bt.Order.Stop, price=stop_price, size=abs(self.position.size)
                    )
                self.order = None
        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            if order == self.stop_order:
                self.stop_order = None
            else:
                self.order = None

    def next(self):
        if self.order:
            return

        close = self.data.close[0]
        sma = self.sma[0]
        atr = self.atr[0]

        if self.position:
            if self.position.size > 0 and close >= sma:
                if self.stop_order:
                    self.cancel(self.stop_order)
                self.order = self.close()
            elif self.position.size < 0 and close <= sma:
                if self.stop_order:
                    self.cancel(self.stop_order)
                self.order = self.close()
        else:
            if close < sma - 2 * atr:
                self.order = self.buy()
            elif close > sma + 2 * atr:
                self.order = self.sell()
