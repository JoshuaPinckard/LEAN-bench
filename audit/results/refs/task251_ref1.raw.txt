import backtrader as bt
import math


class BreakoutSupportResistanceStrategy(bt.Strategy):
    def __init__(self):
        self.resistance = bt.indicators.Highest(self.data.high(-1), period=60)
        self.support = bt.indicators.Lowest(self.data.low(-1), period=60)
        self.volume_sma = bt.indicators.SMA(self.data.volume, period=20)

        self.addminperiod(101)

        self.order = None
        self.pending_entry_context = None
        self.stop_order = None
        self.target_order = None
        self.entry_bar = None
        self.broken_level = None
        self.direction = None
        self.force_exit = False

    def notify_order(self, order):
        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
            order.Expired,
        ]:
            if order is self.order:
                self.order = None

                if order.status == order.Completed and self.pending_entry_context:
                    self.direction, self.broken_level = self.pending_entry_context
                    self.entry_bar = len(self)
                    entry_price = order.executed.price
                    size = abs(self.position.size)

                    if self.direction == "long":
                        stop_price = self.broken_level * 0.99
                        target_price = entry_price * 1.02
                        self.stop_order = self.sell(
                            size=size,
                            exectype=bt.Order.Stop,
                            price=stop_price,
                        )
                        self.target_order = self.sell(
                            size=size,
                            exectype=bt.Order.Limit,
                            price=target_price,
                            oco=self.stop_order,
                        )
                    else:
                        stop_price = self.broken_level * 1.01
                        target_price = entry_price * 0.98
                        self.stop_order = self.buy(
                            size=size,
                            exectype=bt.Order.Stop,
                            price=stop_price,
                        )
                        self.target_order = self.buy(
                            size=size,
                            exectype=bt.Order.Limit,
                            price=target_price,
                            oco=self.stop_order,
                        )

                    self.pending_entry_context = None
                elif order.status != order.Completed:
                    self.pending_entry_context = None

    def next(self):
        if self.order:
            return

        if self.position:
            if self.force_exit:
                if (self.stop_order and self.stop_order.alive()) or (
                    self.target_order and self.target_order.alive()
                ):
                    return
                self.order = self.close()
                self.force_exit = False
                return

            candles_after_entry = len(self) - self.entry_bar
            if 1 <= candles_after_entry <= 5:
                if self.direction == "long" and self.data.close[0] < self.broken_level:
                    if self.stop_order and self.stop_order.alive():
                        self.cancel(self.stop_order)
                    if self.target_order and self.target_order.alive():
                        self.cancel(self.target_order)
                    self.force_exit = True
                elif self.direction == "short" and self.data.close[0] > self.broken_level:
                    if self.stop_order and self.stop_order.alive():
                        self.cancel(self.stop_order)
                    if self.target_order and self.target_order.alive():
                        self.cancel(self.target_order)
                    self.force_exit = True
            return

        if (self.stop_order and self.stop_order.alive()) or (
            self.target_order and self.target_order.alive()
        ):
            return

        self.stop_order = None
        self.target_order = None
        self.entry_bar = None
        self.broken_level = None
        self.direction = None

        resistance = self.resistance[0]
        support = self.support[0]
        volume_confirmed = self.data.volume[0] >= self.volume_sma[0] * 1.30

        resistance_touches = sum(
            abs(self.data.high[-i] - resistance) / resistance <= 0.002
            for i in range(1, 101)
        )
        support_touches = sum(
            abs(self.data.low[-i] - support) / support <= 0.002
            for i in range(1, 101)
        )

        equity = self.broker.getvalue()

        if (
            self.data.close[0] >= resistance * 1.005
            and volume_confirmed
            and resistance_touches >= 3
        ):
            stop_price = resistance * 0.99
            size = math.floor(equity * 0.01 / abs(self.data.close[0] - stop_price))
            if size > 0:
                self.pending_entry_context = ("long", resistance)
                self.order = self.buy(size=size)

        elif (
            self.data.close[0] <= support * 0.995
            and volume_confirmed
            and support_touches >= 3
        ):
            stop_price = support * 1.01
            size = math.floor(equity * 0.01 / abs(self.data.close[0] - stop_price))
            if size > 0:
                self.pending_entry_context = ("short", support)
                self.order = self.sell(size=size)
