import backtrader as bt


class VolatilityStrategy(bt.Strategy):
    params = (
        ('N', 5),
        ('pct_up', 0.03),
        ('tp_pct', 0.06),
        ('sl_pct', 0.03),
        ('vol_thresh', 0.01),
        ('position_usdt', 100.0),
        ('use_atr_filter', True),
        ('mode', 'test'),
    )

    def __init__(self):
        self.atr = bt.indicators.ATR(self.data, period=14)
        self.order = None

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            if self.order and order.ref == self.order.ref:
                self.order = None

    def next(self):
        if len(self.data) <= self.p.N:
            return

        pct_change = (self.data.close[0] / self.data.close[-self.p.N]) - 1.0
        atr_pct = self.atr[0] / self.data.close[0]

        atr_cond = (atr_pct >= self.p.vol_thresh) if self.p.use_atr_filter else True
        entry_cond = (pct_change >= self.p.pct_up) and atr_cond

        if not self.position and not self.order:
            if entry_cond:
                price = self.data.close[0]
                size = self.p.position_usdt / price
                tp_price = price * (1.0 + self.p.tp_pct)
                sl_price = price * (1.0 - self.p.sl_pct)

                orders = self.buy_bracket(
                    size=size,
                    limitprice=tp_price,
                    stopprice=sl_price
                )
                self.order = orders[0]
