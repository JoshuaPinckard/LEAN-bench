import backtrader as bt


class PeriodicRebalance(bt.Strategy):
    # Single-asset periodic rebalancing at a constant 100% target weight.
    # Enter 100% on the start date, force a full close + immediate re-open
    # every 12 weeks (to book transaction costs), and exit at 52 weeks.

    params = (
        ('rebalance_weeks', 12),
        ('horizon_weeks', 52),
        ('target', 1.0),
    )

    def __init__(self):
        self.order = None
        self.entered = False
        self.done = False
        self.pending_reentry = False
        self.weeks_elapsed = 0
        self.last_week_key = None

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return
        if order.status in (order.Completed, order.Canceled,
                             order.Margin, order.Rejected):
            self.order = None

    def next(self):
        # Weekly cadence derived from the daily feed via ISO-week boundaries.
        week_key = self.data.datetime.date(0).isocalendar()[:2]
        new_week = week_key != self.last_week_key
        self.last_week_key = week_key

        # Entry: on the start date (first available bar) buy 100% of portfolio.
        if not self.entered:
            self.order = self.order_target_percent(target=self.p.target)
            self.entered = True
            self.weeks_elapsed = 0
            return

        if new_week:
            self.weeks_elapsed += 1

        # Rebalance second leg: re-open 100% the bar after the forced close,
        # once the close has settled and cash is freed.
        if self.pending_reentry and not self.position and self.order is None:
            self.order = self.order_target_percent(target=self.p.target)
            self.pending_reentry = False
            return

        if self.done:
            return

        if new_week and self.order is None:
            # Exit: at the end of the 52-week horizon, close and lock in.
            if self.weeks_elapsed >= self.p.horizon_weeks:
                self.order = self.close()
                self.done = True
                return
            # Rebalance every 12 weeks: full close, then immediate re-open.
            if (self.weeks_elapsed > 0 and
                    self.weeks_elapsed % self.p.rebalance_weeks == 0):
                self.order = self.close()
                self.pending_reentry = True
                return
