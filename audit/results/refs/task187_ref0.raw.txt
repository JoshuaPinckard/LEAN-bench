```python
import backtrader as bt


class SingleAssetPeriodicRebalance(bt.Strategy):
    params = dict(
        rebalance_weeks=12,
        horizon_weeks=52,
        commission=0.001,
    )

    def __init__(self):
        self.order = None
        self.action = None
        self.last_week_key = None
        self.week_count = 0
        self.cycle_entry_week = None
        self.last_rebalance_elapsed = 0
        self.need_open = True
        self.is_new_cycle = True

    def _target_size(self):
        cash = self.broker.getcash()
        price = self.data.close[0]
        if price <= 0:
            return 0
        return int(cash / (price * (1 + self.p.commission)))

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return
        if order.status == order.Completed:
            if self.action == 'open_new_cycle':
                self.cycle_entry_week = self.week_count
                self.last_rebalance_elapsed = 0
            elif self.action == 'close_rebalance':
                self.need_open = True
                self.is_new_cycle = False
            elif self.action == 'exit':
                self.need_open = True
                self.is_new_cycle = True
                self.cycle_entry_week = None
        elif order.status in (order.Canceled, order.Margin, order.Rejected):
            if self.action in ('open_new_cycle', 'open_rebalance'):
                self.need_open = True
        self.order = None
        self.action = None

    def next(self):
        key = self.data.datetime.date(0).isocalendar()[:2]
        if key != self.last_week_key:
            self.last_week_key = key
            self.week_count += 1

        if self.order is not None:
            return

        if self.need_open:
            size = self._target_size()
            if size > 0:
                self.order = self.buy(size=size)
                self.action = 'open_new_cycle' if self.is_new_cycle else 'open_rebalance'
            self.need_open = False
            return

        if self.position and self.cycle_entry_week is not None:
            elapsed = self.week_count - self.cycle_entry_week
            if elapsed >= self.p.horizon_weeks:
                self.order = self.close()
                self.action = 'exit'
                return
            if elapsed > 0 and elapsed % self.p.rebalance_weeks == 0 and elapsed != self.last_rebalance_elapsed:
                self.last_rebalance_elapsed = elapsed
                self.order = self.close()
                self.action = 'close_rebalance'
                return
```
