import backtrader as bt


class WilliamsRBandStrategy(bt.Strategy):
    params = (
        ("period_willr", 14),
        ("period_band", 20),
        ("devfactor", 2.0),
        ("stop_loss", 1.0),
    )

    def __init__(self):
        self.order = None

        self.willr = bt.indicators.WilliamsR(self.data, period=self.p.period_willr)
        self.wr_mean = bt.indicators.SMA(self.willr, period=self.p.period_band)
        self.wr_std = bt.indicators.StdDev(self.willr, period=self.p.period_band)

        self.upper_band = self.wr_mean + self.p.devfactor * self.wr_std
        self.lower_band = self.wr_mean - self.p.devfactor * self.wr_std

    def notify_order(self, order):
        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            self.order = None

    def next(self):
        if self.order:
            return

        cross_up_upper = (
            self.willr[0] > self.upper_band[0]
            and self.willr[-1] <= self.upper_band[-1]
        )
        cross_down_mean = (
            self.willr[0] < self.wr_mean[0]
            and self.willr[-1] >= self.wr_mean[-1]
        )
        cross_down_lower = (
            self.willr[0] < self.lower_band[0]
            and self.willr[-1] >= self.lower_band[-1]
        )
        cross_up_mean = (
            self.willr[0] > self.wr_mean[0]
            and self.willr[-1] <= self.wr_mean[-1]
        )

        if not self.position:
            if cross_up_upper:
                self.order = self.buy()
            elif cross_down_lower:
                self.order = self.sell()
        else:
            if self.position.size > 0:
                stop_price = self.position.price * (
                    1.0 - self.p.stop_loss / 100.0
                )
                if self.data.low[0] <= stop_price or cross_down_mean:
                    self.order = self.close()
            elif self.position.size < 0:
                stop_price = self.position.price * (
                    1.0 + self.p.stop_loss / 100.0
                )
                if self.data.high[0] >= stop_price or cross_up_mean:
                    self.order = self.close()
