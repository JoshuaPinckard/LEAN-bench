import backtrader as bt


class MartingaleWithRSIAndBollingerBandsFilter(bt.Strategy):
    params = (
        ("initial_lot_size", 0.01),
        ("lotexp", 1.2345),
        ("max_martingale_level", 6),
        ("max_drawdown", 350.0),
    )

    def __init__(self):
        self.rsi = bt.indicators.RSI(self.data.close, period=14)
        self.bollinger = bt.indicators.BollingerBands(
            self.data.close, period=20, devfactor=2
        )

        self.order = None
        self.order_action = None
        self.martingale_level = 0
        self.peak_value = None

    def start(self):
        self.peak_value = self.broker.getvalue()

    def notify_order(self, order):
        if order.status == order.Completed:
            if order is self.order:
                if self.order_action == "long_entry" or self.order_action == "short_entry":
                    self.martingale_level += 1
                elif self.order_action == "close" and not self.position:
                    self.martingale_level = 0

                self.order = None
                self.order_action = None

        elif order.status in (order.Canceled, order.Margin, order.Rejected):
            if order is self.order:
                self.order = None
                self.order_action = None

    def next(self):
        if self.order:
            return

        account_value = self.broker.getvalue()
        if account_value > self.peak_value:
            self.peak_value = account_value

        if self.peak_value - account_value >= self.p.max_drawdown:
            if self.position:
                self.order = self.close()
                self.order_action = "close"
            return

        close = self.data.close[0]
        rsi = self.rsi[0]
        lower = self.bollinger.bot[0]
        upper = self.bollinger.top[0]

        if self.position.size > 0:
            if close > upper or rsi > 50:
                self.order = self.close()
                self.order_action = "close"
            elif (
                close < lower
                and rsi < 30
                and self.martingale_level < self.p.max_martingale_level
            ):
                size = self.p.initial_lot_size * (
                    self.p.lotexp ** self.martingale_level
                )
                self.order = self.buy(size=size)
                self.order_action = "long_entry"

        elif self.position.size < 0:
            if close < lower or rsi < 50:
                self.order = self.close()
                self.order_action = "close"
            elif (
                close > upper
                and rsi > 70
                and self.martingale_level < self.p.max_martingale_level
            ):
                size = self.p.initial_lot_size * (
                    self.p.lotexp ** self.martingale_level
                )
                self.order = self.sell(size=size)
                self.order_action = "short_entry"

        elif close < lower and rsi < 30:
            size = self.p.initial_lot_size
            self.order = self.buy(size=size)
            self.order_action = "long_entry"

        elif close > upper and rsi > 70:
            size = self.p.initial_lot_size
            self.order = self.sell(size=size)
            self.order_action = "short_entry"
