import backtrader as bt


class RegimeVolatilityStrategy(bt.Strategy):
    params = (
        ("atr_period", 14),
        ("bb_period", 20),
        ("bb_dev", 2.0),
        ("ema_smooth_period", 50),
        ("ema_fast", 10),
        ("ema_slow", 30),
        ("sma_filter", 50),
        ("rsi_period", 14),
        ("macd_fast", 12),
        ("macd_slow", 26),
        ("macd_signal", 9),
        ("risk_pct", 0.02),
    )

    def __init__(self):
        self.order = None
        self.entry_regime = None
        self.stop_price = None

        # Base Indicators
        self.atr = bt.ind.ATR(self.data, period=self.p.atr_period)
        self.bb = bt.ind.BollingerBands(
            self.data, period=self.p.bb_period, devfactor=self.p.bb_dev
        )
        self.bb_width = self.bb.top - self.bb.bot

        # Composite Volatility Index (CVI) components
        self.ema_atr = bt.ind.EMA(self.atr, period=self.p.ema_smooth_period)
        self.ema_bbw = bt.ind.EMA(self.bb_width, period=self.p.ema_smooth_period)

        # Proxy PC1: Composite Volatility Index (CVI)
        self.cvi = 0.5 * (self.atr / self.ema_atr) + 0.5 * (
            self.bb_width / self.ema_bbw
        )

        # Proxy PC3: Observed Noise Metric J
        self.J = self.atr / self.bb_width

        # Trend & Momentum Indicators
        self.ema10 = bt.ind.EMA(self.data.close, period=self.p.ema_fast)
        self.ema30 = bt.ind.EMA(self.data.close, period=self.p.ema_slow)
        self.sma50 = bt.ind.SMA(self.data.close, period=self.p.sma_filter)
        self.rsi = bt.ind.RSI(self.data.close, period=self.p.rsi_period)

        self.macd = bt.ind.MACDHistogram(
            self.data.close,
            period1=self.p.macd_fast,
            period2=self.p.macd_slow,
            period_df=self.p.macd_signal,
        )

        # Crossovers
        self.crossover_ema = bt.ind.CrossOver(self.ema10, self.ema30)
        self.crossover_macd = bt.ind.CrossOver(self.macd.histo, 0.0)
        self.crossover_cvi = bt.ind.CrossOver(self.cvi, 1.0)

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            if not self.position:
                self.entry_regime = None
                self.stop_price = None
            self.order = None
        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def calculate_position_size(self, cvi, atr):
        value = self.broker.getvalue()
        if cvi < 0.9:
            mult = 1.3
            stop_mult = 1.5
        elif cvi > 1.1:
            mult = 0.7
            stop_mult = 2.5
        else:
            mult = 1.0
            stop_mult = 2.0

        risk_amount = value * self.p.risk_pct
        risk_per_share = stop_mult * atr
        if risk_per_share <= 0:
            return 1

        base_shares = risk_amount / risk_per_share
        size = int(base_shares * mult)

        cash = self.broker.getcash()
        max_shares = int(cash / self.data.close[0])
        size = min(size, max_shares)
        return max(size, 1) if max_shares >= 1 else 0

    def next(self):
        if self.order:
            return

        cvi_val = self.cvi[0]
        atr_val = self.atr[0]
        close_val = self.data.close[0]

        if not self.position:
            # 1. Low Volatility Regime (CVI < 0.9) - Trend Logic
            if cvi_val < 0.9:
                if (
                    self.crossover_ema[0] == 1
                    and close_val > self.sma50[0]
                    and self.rsi[0] > 50
                ):
                    size = self.calculate_position_size(cvi_val, atr_val)
                    if size > 0:
                        self.order = self.buy(size=size)
                        self.entry_regime = "low_vol"
                        self.stop_price = close_val - 1.5 * atr_val
                elif (
                    self.crossover_ema[0] == -1
                    and close_val < self.sma50[0]
                    and self.rsi[0] < 50
                ):
                    size = self.calculate_position_size(cvi_val, atr_val)
                    if size > 0:
                        self.order = self.sell(size=size)
                        self.entry_regime = "low_vol"
                        self.stop_price = close_val + 1.5 * atr_val

            # 2. High Volatility Regime (CVI > 1.1) - Mean Reversion Logic
            elif cvi_val > 1.1:
                if (
                    self.data.low[0] <= self.bb.bot[0]
                    or close_val <= self.bb.bot[0]
                ) and self.rsi[0] < 35:
                    size = self.calculate_position_size(cvi_val, atr_val)
                    if size > 0:
                        self.order = self.buy(size=size)
                        self.entry_regime = "high_vol"
                        self.stop_price = close_val - 2.5 * atr_val
                elif (
                    self.data.high[0] >= self.bb.top[0]
                    or close_val >= self.bb.top[0]
                ) and self.rsi[0] > 65:
                    size = self.calculate_position_size(cvi_val, atr_val)
                    if size > 0:
                        self.order = self.sell(size=size)
                        self.entry_regime = "high_vol"
                        self.stop_price = close_val + 2.5 * atr_val

            # 3. Entry on Regime Shift (PC2-like trigger)
            if not self.order:
                if self.crossover_cvi[0] == 1 and self.crossover_macd[0] == 1:
                    if self.rsi[0] > 50 and close_val > self.sma50[0]:
                        size = self.calculate_position_size(cvi_val, atr_val)
                        if size > 0:
                            self.order = self.buy(size=size)
                            self.entry_regime = "regime_shift"
                            self.stop_price = close_val - 1.5 * atr_val
                elif self.crossover_cvi[0] == -1 and self.crossover_macd[0] == -1:
                    if self.rsi[0] < 50 and close_val < self.sma50[0]:
                        size = self.calculate_position_size(cvi_val, atr_val)
                        if size > 0:
                            self.order = self.sell(size=size)
                            self.entry_regime = "regime_shift"
                            self.stop_price = close_val + 1.5 * atr_val

        else:
            is_long = self.position.size > 0

            # Exit Condition 1: Regime change
            regime_exit = False
            if self.entry_regime == "high_vol" and cvi_val < 1.0:
                regime_exit = True
            elif self.entry_regime == "low_vol" and cvi_val > 1.0:
                regime_exit = True

            # Exit Condition 2: Technical stop (ATR stop)
            stop_exit = False
            if self.stop_price is not None:
                if is_long and self.data.low[0] <= self.stop_price:
                    stop_exit = True
                elif not is_long and self.data.high[0] >= self.stop_price:
                    stop_exit = True

            # Exit Condition 3: Mean reversion targets (high volatility trades)
            target_exit = False
            if self.entry_regime == "high_vol":
                if is_long:
                    if close_val >= self.bb.mid[0] or self.rsi[0] > 55:
                        target_exit = True
                else:
                    if close_val <= self.bb.mid[0] or self.rsi[0] < 45:
                        target_exit = True

            if regime_exit or stop_exit or target_exit:
                self.order = self.close()
