import backtrader as bt
import math


class TimeSeriesMomentumStrategy(bt.Strategy):
    params = (
        ("momentum_window", 90),
        ("rebalancing_frequency", 21),
        ("target_allocation", 0.20),
    )

    def __init__(self):
        self.momentum = self.data.close / self.data.close(-self.p.momentum_window) - 1.0
        self.average_dollar_volume = bt.indicators.SMA(
            self.data.close * self.data.volume, period=21
        )
        self.crossover = bt.indicators.CrossOver(
            self.data.close, self.data.close(-self.p.momentum_window)
        )

        self.order = None
        self._order_role = None
        self._rebalance_target = 0
        self._waiting_for_rebalance = False

    def _target_size(self, direction):
        allocation = min(self.p.target_allocation, 0.10)
        size = int((self.broker.getvalue() * allocation) / self.data.close[0])
        return direction * size

    def _submit_entry(self, target):
        if target > 0:
            self.order = self.buy(size=target)
            self._order_role = "entry"
        elif target < 0:
            self.order = self.sell(size=abs(target))
            self._order_role = "entry"

    def notify_order(self, order):
        if order.status not in (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            return

        if order is not self.order:
            return

        role = self._order_role
        self.order = None
        self._order_role = None

        if role == "rebalance_close":
            target = self._rebalance_target
            self._rebalance_target = 0
            if order.status == order.Completed:
                self._submit_entry(target)

    def next(self):
        if self.order:
            return

        is_rebalance = len(self) % self.p.rebalancing_frequency == 0
        liquid = self.average_dollar_volume[0] >= 1000000

        if is_rebalance:
            self._waiting_for_rebalance = False
            target = 0

            if liquid:
                if self.momentum[0] > 0 and not math.isclose(
                    self.momentum[0], 0.0, abs_tol=1e-12
                ):
                    target = self._target_size(1)
                elif self.momentum[0] < 0 and not math.isclose(
                    self.momentum[0], 0.0, abs_tol=1e-12
                ):
                    target = self._target_size(-1)

            if self.position.size:
                self._rebalance_target = target
                self.order = self.close()
                self._order_role = "rebalance_close"
            else:
                self._submit_entry(target)
            return

        if (
            self.position.size
            and not self._waiting_for_rebalance
            and self.crossover[0] != 0
        ):
            self._waiting_for_rebalance = True
            self.order = self.close()
            self._order_role = "exit"
