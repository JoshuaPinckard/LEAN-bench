import backtrader as bt


class MomentumTrailingStrategy(bt.Strategy):
    params = (
        ('bb_period', 20),
        ('bb_devfactor', 2),
        ('rsi_period', 14),
    )

    def __init__(self):
        self.bb = bt.indicators.BollingerBands(
            self.data.close,
            period=self.p.bb_period,
            devfactor=self.p.bb_devfactor,
        )
        self.rsi = bt.indicators.RSI(self.data.close, period=self.p.rsi_period)

        self.order = None
        self.highest_price = None
        self.trailing_stop_loss = None
        self.entry_pending = False

    def notify_order(self, order):
        if order.status == order.Completed:
            if order.isbuy():
                self.highest_price = order.executed.price
                self.trailing_stop_loss = order.executed.price * (1 - 1 / 100)
                self.entry_pending = True
            elif order.issell():
                self.highest_price = None
                self.trailing_stop_loss = None

        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            self.order = None

    def next(self):
        if self.order:
            return

        if self.position:
            if self.entry_pending:
                self.entry_pending = False
                return

            self.highest_price = max(self.highest_price, self.data.high[0])
            self.trailing_stop_loss = max(
                self.trailing_stop_loss,
                self.highest_price * (1 - 1 / 100),
            )

            if self.data.close[0] <= self.trailing_stop_loss:
                self.order = self.close()
            return

        crossed_above_upper_band = (
            self.data.close[-1] <= self.bb.top[-1]
            and self.data.close[0] > self.bb.top[0]
        )

        if crossed_above_upper_band and self.rsi[0] > 55:
            self.order = self.buy(size=1)
