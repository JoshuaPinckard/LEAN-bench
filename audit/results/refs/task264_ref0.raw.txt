import backtrader as bt


class BreakoutTrendVolatilityStrategy(bt.Strategy):
    def __init__(self):
        self.sma200 = bt.indicators.SMA(self.data.close, period=200)
        self.bb = bt.indicators.BollingerBands(self.data, period=20, devfactor=2)
        self.atr = bt.indicators.ATR(self.data, period=14)
        self.width = self.bb.top - self.bb.bot
        self.sma_width = bt.indicators.SMA(self.width, period=100)
        self.atr_sma = bt.indicators.SMA(self.atr, period=100)
        self.cross_up = bt.indicators.CrossOver(self.data.close, self.bb.top)
        self.cross_down = bt.indicators.CrossOver(self.data.close, self.bb.bot)

        self.order = None
        self.pending_action = None
        self.pending_stop_distance = 0.0

        self.entry_price = None
        self.stop_price = None
        self.take_price = None
        self.r_distance = None
        self.trail_extreme = None
        self.breakeven_done = False
        self.exit_bar = None

    def notify_order(self, order):
        if order.status == order.Completed:
            if self.pending_action == 'enter':
                self.entry_price = order.executed.price
                stop_distance = self.pending_stop_distance
                self.r_distance = stop_distance
                if order.isbuy():
                    self.stop_price = self.entry_price - stop_distance
                    self.take_price = self.entry_price + 3 * stop_distance
                else:
                    self.stop_price = self.entry_price + stop_distance
                    self.take_price = self.entry_price - 3 * stop_distance
                self.trail_extreme = self.entry_price
                self.breakeven_done = False
            elif self.pending_action == 'exit':
                self.exit_bar = len(self)
            self.pending_action = None
            self.order = None
        elif order.status in (order.Canceled, order.Margin, order.Rejected):
            self.pending_action = None
            self.order = None

    def next(self):
        if self.order:
            return

        if self.position:
            self._manage_open_position()
        else:
            self._check_entry()

    def _check_entry(self):
        if self.exit_bar is not None and (len(self) - self.exit_bar) < 20:
            return

        close = self.data.close[0]
        consolidated = all(
            self.width[-i] < self.sma_width[-i] * 0.8 for i in range(10)
        )
        vol_ok = 0.7 <= (self.atr[0] / self.atr_sma[0]) <= 1.3

        long_signal = (
            close > self.sma200[0]
            and consolidated
            and self.cross_up[0] > 0
            and vol_ok
        )
        short_signal = (
            close < self.sma200[0]
            and consolidated
            and self.cross_down[0] < 0
            and vol_ok
        )

        if long_signal:
            self._enter(is_long=True)
        elif short_signal:
            self._enter(is_long=False)

    def _enter(self, is_long):
        stop_distance = 3 * self.atr[0]
        if stop_distance <= 0:
            return
        size = int((0.01 * self.broker.getvalue()) / stop_distance)
        if size <= 0:
            return
        self.pending_stop_distance = stop_distance
        self.pending_action = 'enter'
        if is_long:
            self.order = self.buy(size=size)
        else:
            self.order = self.sell(size=size)

    def _manage_open_position(self):
        close = self.data.close[0]

        if self.position.size > 0:
            self.trail_extreme = max(self.trail_extreme, close)

            if not self.breakeven_done and (close - self.entry_price) >= 1.5 * self.r_distance:
                self.stop_price = max(self.stop_price, self.entry_price)
                self.breakeven_done = True

            chandelier_stop = self.trail_extreme - 3 * self.atr[0]
            if chandelier_stop > self.stop_price:
                self.stop_price = chandelier_stop

            if close >= self.take_price or close <= self.stop_price:
                self.pending_action = 'exit'
                self.order = self.close()
        else:
            self.trail_extreme = min(self.trail_extreme, close)

            if not self.breakeven_done and (self.entry_price - close) >= 1.5 * self.r_distance:
                self.stop_price = min(self.stop_price, self.entry_price)
                self.breakeven_done = True

            chandelier_stop = self.trail_extreme + 3 * self.atr[0]
            if chandelier_stop < self.stop_price:
                self.stop_price = chandelier_stop

            if close <= self.take_price or close >= self.stop_price:
                self.pending_action = 'exit'
                self.order = self.close()
