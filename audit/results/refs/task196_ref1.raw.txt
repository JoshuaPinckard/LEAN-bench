import backtrader as bt


class VolatilityTrendStrategy(bt.Strategy):
    def __init__(self):
        self.atr = bt.indicators.ATR(self.data, period=14)
        self.atr_ema = bt.indicators.EMA(self.atr, period=14)
        self.sma = bt.indicators.SMA(self.data.close, period=20)
        self.rsi = bt.indicators.RSI(self.data.close, period=14)
        self.volume_sma = bt.indicators.SMA(self.data.volume, period=20)
        self.rsi_cross_down = bt.indicators.CrossDown(self.rsi, 50.0)
        self.rsi_cross_up = bt.indicators.CrossUp(self.rsi, 50.0)

        self.order = None
        self.pending_role = None
        self.entry_atr = None
        self.stop_loss = None
        self.take_profit = None

    def next(self):
        if self.order:
            return

        close = self.data.close[0]

        if self.position:
            if self.position.size > 0:
                if (close < self.sma[0] or self.rsi_cross_down[0] or
                        close <= self.stop_loss or close >= self.take_profit):
                    self.order = self.close()
                    self.pending_role = 'exit'
            else:
                if (close > self.sma[0] or self.rsi_cross_up[0] or
                        close >= self.stop_loss or close <= self.take_profit):
                    self.order = self.close()
                    self.pending_role = 'exit'
            return

        volatility_expanding = self.atr[0] > self.atr_ema[0]
        volume_confirmed = self.data.volume[0] > self.volume_sma[0]

        if (close > self.sma[0] and self.rsi[0] > 55 and
                volatility_expanding and volume_confirmed):
            self.entry_atr = self.atr[0]
            self.order = self.buy()
            self.pending_role = 'long_entry'

        elif (close < self.sma[0] and self.rsi[0] < 45 and
                volatility_expanding and volume_confirmed):
            self.entry_atr = self.atr[0]
            self.order = self.sell()
            self.pending_role = 'short_entry'

    def notify_order(self, order):
        if order != self.order:
            return

        if order.status == order.Completed:
            if self.pending_role == 'long_entry':
                self.stop_loss = order.executed.price - 1.5 * self.entry_atr
                self.take_profit = order.executed.price + 2.0 * self.entry_atr
            elif self.pending_role == 'short_entry':
                self.stop_loss = order.executed.price + 1.5 * self.entry_atr
                self.take_profit = order.executed.price - 2.0 * self.entry_atr
            elif self.pending_role == 'exit':
                self.entry_atr = None
                self.stop_loss = None
                self.take_profit = None

        if order.status in [order.Completed, order.Canceled,
                            order.Margin, order.Rejected]:
            if order.status != order.Completed and self.pending_role in [
                    'long_entry', 'short_entry']:
                self.entry_atr = None
            self.order = None
            self.pending_role = None
