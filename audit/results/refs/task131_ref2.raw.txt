import backtrader as bt


class NVDAPreEarningsBreakout(bt.Strategy):
    params = (
        ("key_support", 6430.75),
        ("support_low", 6455.0),
        ("support_high", 6465.0),
        ("resistance", 6496.25),
        ("target", 6508.75),
        ("volume_period", 20),
        ("lookback_pullback", 50),
        ("risk_pct", 0.01),
    )

    def __init__(self):
        self.order = None
        self.vol_sma = bt.indicators.SimpleMovingAverage(
            self.data.volume, period=self.p.volume_period
        )

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return
        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            self.order = None

    def next(self):
        if self.order:
            return

        if self.position:
            # Exit 1: Take-profit upon reaching target level 6508.75
            if (
                self.data.high[0] >= self.p.target
                or self.data.close[0] >= self.p.target
            ):
                self.order = self.close()
                return

            # Exit 2: Stop-loss if price drops below lower boundary of nearest support zone (6455)
            if (
                self.data.low[0] <= self.p.support_low
                or self.data.close[0] < self.p.support_low
            ):
                self.order = self.close()
                return
        else:
            # Condition 1: ES price above or within nearest support zone (6455-6465)
            c_above_support = self.data.close[0] >= self.p.support_low

            # Condition 2: Pullback from key support (6430.75) has occurred and price recovered
            lookback = min(len(self), self.p.lookback_pullback)
            recent_lows = [self.data.low[-i] for i in range(lookback)]
            pullback_occurred = (min(recent_lows) <= self.p.key_support) and (
                self.data.close[0] > self.p.key_support
            )

            # Condition 3: ES price breaks above resistance level 6496.25 with increased volume
            breakout = (self.data.close[0] > self.p.resistance) and (
                self.data.close[-1] <= self.p.resistance
            )
            increased_volume = self.data.volume[0] > self.vol_sma[0]

            if (
                c_above_support
                and pullback_occurred
                and breakout
                and increased_volume
            ):
                entry_price = self.data.close[0]
                risk_per_unit = entry_price - self.p.support_low
                if risk_per_unit > 0:
                    portfolio_value = self.broker.getvalue()
                    max_risk_amount = portfolio_value * self.p.risk_pct
                    size = max(1, int(max_risk_amount / risk_per_unit))
                    self.order = self.buy(size=size)
