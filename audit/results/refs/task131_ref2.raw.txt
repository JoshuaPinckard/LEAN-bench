import backtrader as bt


class NVDAPreEarningsMomentumBreakout(bt.Strategy):

    params = (
        ('support_low', 6455.0),
        ('support_high', 6465.0),
        ('key_support', 6430.75),
        ('resistance', 6496.25),
        ('target', 6508.75),
        ('stop_level', 6455.0),
        ('risk_pct', 0.01),
        ('vol_ma_period', 20),
    )

    def __init__(self):
        self.order = None
        self.had_pullback = False
        self.vol_ma = bt.indicators.SMA(self.data.volume, period=self.p.vol_ma_period)

    def notify_order(self, order):
        if order.status in [
            order.Completed, order.Canceled, order.Margin, order.Rejected
        ]:
            if order.status == order.Completed and order.issell():
                # Position closed; require a fresh pullback for the next setup
                self.had_pullback = False
            self.order = None

    def next(self):
        # Track whether price has visited the key support (50% retracement) level
        if self.data.low[0] <= self.p.key_support:
            self.had_pullback = True

        if self.order:
            return

        if not self.position:
            # Condition 1: price is above or within the nearest support zone (>= 6455)
            above_support = self.data.close[0] >= self.p.support_low

            # Condition 2: pullback from key support has occurred and price has started to recover
            recovering = self.had_pullback and self.data.close[0] > self.p.key_support

            # Condition 3: bar closes above resistance 6496.25 (breakout) with increased volume
            breakout = (
                self.data.close[-1] <= self.p.resistance
                and self.data.close[0] > self.p.resistance
                and self.data.volume[0] > self.vol_ma[0]
            )

            if above_support and recovering and breakout:
                stop_distance = self.data.close[0] - self.p.stop_level
                if stop_distance > 0:
                    total_value = self.broker.getvalue()
                    size = int((total_value * self.p.risk_pct) / stop_distance)
                    if size > 0:
                        self.order = self.buy(size=size)

        else:
            # Exit 1: take-profit — price reaches target 6508.75 (new ATH)
            if self.data.close[0] >= self.p.target:
                self.order = self.close()
                return

            # Exit 2: stop-loss — price drops below lower boundary of support zone (6455)
            if self.data.close[0] < self.p.stop_level:
                self.order = self.close()
                return

            # Exit 3 (alternative): bearish engulfing candlestick pattern with increased volume
            prev_bullish = self.data.close[-1] > self.data.open[-1]
            curr_bearish = self.data.close[0] < self.data.open[0]
            bearish_engulfing = (
                curr_bearish
                and prev_bullish
                and self.data.open[0] >= self.data.close[-1]
                and self.data.close[0] <= self.data.open[-1]
            )
            if bearish_engulfing and self.data.volume[0] > self.vol_ma[0]:
                self.order = self.close()
