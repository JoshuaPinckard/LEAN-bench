import backtrader as bt


class ORBTwoAttemptRetestStrategy(bt.Strategy):
    def __init__(self):
        self.ema20 = bt.indicators.EMA(self.data.close, period=20)
        self.ema50 = bt.indicators.EMA(self.data.close, period=50)
        self.bb = bt.indicators.BollingerBands(self.data.close, period=20, devfactor=2)
        self.atr = bt.indicators.ATR(self.data, period=14)

        self.order = None
        self.pending = None

        self.episode_state = 'idle'
        self.trend_dir = 0

        self.entry_price = None
        self.tp_price = None
        self.sl_price = None
        self.exit_reason = None

        self.retest_days_checked = 0
        self.second_initial_sl = None
        self.trail_stop = None
        self.extreme_since_entry = None

    def _reset_episode(self):
        self.trend_dir = 0
        self._reset_trade_levels()

    def _reset_trade_levels(self):
        self.entry_price = None
        self.tp_price = None
        self.sl_price = None
        self.exit_reason = None
        self.second_initial_sl = None
        self.trail_stop = None
        self.extreme_since_entry = None

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return

        if order.status == order.Completed:
            if self.pending == 'first_entry':
                self.entry_price = order.executed.price
                if self.trend_dir == 1:
                    self.tp_price = self.entry_price * 1.005
                    self.sl_price = self.entry_price * 0.997
                else:
                    self.tp_price = self.entry_price * 0.995
                    self.sl_price = self.entry_price * 1.003
                self.episode_state = 'first_pos'

            elif self.pending == 'first_exit':
                if self.exit_reason == 'sl':
                    self.episode_state = 'retest_wait'
                    self.retest_days_checked = 0
                    self._reset_trade_levels()
                else:
                    self.episode_state = 'idle'
                    self._reset_episode()

            elif self.pending == 'second_entry':
                self.entry_price = order.executed.price
                self.trail_stop = self.second_initial_sl
                if self.trend_dir == 1:
                    self.tp_price = self.entry_price * 1.006
                    self.extreme_since_entry = self.data.high[0]
                else:
                    self.tp_price = self.entry_price * 0.994
                    self.extreme_since_entry = self.data.low[0]
                self.episode_state = 'second_pos'

            elif self.pending == 'second_exit':
                self.episode_state = 'idle'
                self._reset_episode()

        self.order = None
        self.pending = None

    def _try_start_episode(self):
        if self.ema20[0] > self.ema50[0]:
            self.trend_dir = 1
            self.order = self.buy(size=1)
            self.pending = 'first_entry'
        elif self.ema20[0] < self.ema50[0]:
            self.trend_dir = -1
            self.order = self.sell(size=1)
            self.pending = 'first_entry'

    def _check_retest(self):
        mid = self.bb.mid[0]
        if self.trend_dir == 1:
            condition = (self.data.low[0] <= mid) and (self.data.close[0] > mid)
        else:
            condition = (self.data.high[0] >= mid) and (self.data.close[0] < mid)

        if condition:
            self.second_initial_sl = self.data.low[0] if self.trend_dir == 1 else self.data.high[0]
            self.episode_state = 'second_pending'
            return

        self.retest_days_checked += 1
        if self.retest_days_checked >= 3:
            self.episode_state = 'idle'
            self._reset_episode()

    def _enter_second_trade(self):
        if self.trend_dir == 1:
            self.order = self.buy(size=1)
        else:
            self.order = self.sell(size=1)
        self.pending = 'second_entry'

    def _manage_first_pos(self):
        high = self.data.high[0]
        low = self.data.low[0]
        if self.trend_dir == 1:
            hit_tp = high >= self.tp_price
            hit_sl = low <= self.sl_price
        else:
            hit_tp = low <= self.tp_price
            hit_sl = high >= self.sl_price

        if hit_sl:
            self.exit_reason = 'sl'
            self.order = self.close()
            self.pending = 'first_exit'
        elif hit_tp:
            self.exit_reason = 'tp'
            self.order = self.close()
            self.pending = 'first_exit'

    def _manage_second_pos(self):
        high = self.data.high[0]
        low = self.data.low[0]
        if self.trend_dir == 1:
            hit_tp = high >= self.tp_price
            hit_trail = low <= self.trail_stop
        else:
            hit_tp = low <= self.tp_price
            hit_trail = high >= self.trail_stop

        if hit_trail:
            self.order = self.close()
            self.pending = 'second_exit'
            return
        elif hit_tp:
            self.order = self.close()
            self.pending = 'second_exit'
            return

        if self.trend_dir == 1:
            self.extreme_since_entry = max(self.extreme_since_entry, high)
            candidate = self.extreme_since_entry - self.atr[0] * 1.0
            self.trail_stop = max(self.trail_stop, candidate)
        else:
            self.extreme_since_entry = min(self.extreme_since_entry, low)
            candidate = self.extreme_since_entry + self.atr[0] * 1.0
            self.trail_stop = min(self.trail_stop, candidate)

    def next(self):
        if self.order:
            return

        if self.position:
            if self.episode_state == 'first_pos':
                self._manage_first_pos()
            elif self.episode_state == 'second_pos':
                self._manage_second_pos()
            return

        if self.episode_state == 'idle':
            self._try_start_episode()
        elif self.episode_state == 'retest_wait':
            self._check_retest()
        elif self.episode_state == 'second_pending':
            self._enter_second_trade()
