import backtrader as bt


class SmallCapMomentumStrategy(bt.Strategy):
    def __init__(self):
        self.sma50 = bt.indicators.SMA(self.data.close, period=50)
        self.volume_sma50 = bt.indicators.SMA(self.data.volume, period=50)
        self.volume_sma30 = bt.indicators.SMA(self.data.volume, period=30)

        self.macd_line = (
            bt.indicators.EMA(self.data.close, period=12)
            - bt.indicators.EMA(self.data.close, period=26)
        )
        self.macd_signal = bt.indicators.SMA(self.macd_line, period=9)
        self.macd_histogram = self.macd_line - self.macd_signal

        self.order = None
        self.entry_price = None

    def notify_order(self, order):
        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            if order.status == order.Completed:
                if order.isbuy():
                    self.entry_price = order.executed.price
                elif order.issell():
                    self.entry_price = None
            self.order = None

    def next(self):
        if self.order:
            return

        if self.position:
            take_profit = self.entry_price * 2.0
            stop_loss = self.entry_price * 0.85
            bearish_histogram_cross = (
                self.macd_histogram[0] < 0
                and self.macd_histogram[-1] >= 0
            )

            if (
                self.data.close[0] >= take_profit
                or self.data.close[0] <= stop_loss
                or bearish_histogram_cross
            ):
                self.order = self.close()
            return

        trend = self.data.close[0] > self.sma50[0]
        histogram_curling_up = (
            self.macd_histogram[0] > self.macd_histogram[-1]
            and self.macd_histogram[-1] < self.macd_histogram[-2]
        )
        sustained_volume = all(
            self.data.volume[-offset] > self.volume_sma50[-offset]
            for offset in range(5)
        )
        abnormal_volume = self.data.volume[0] >= self.volume_sma30[0] * 2.0

        if trend and histogram_curling_up and sustained_volume and abnormal_volume:
            self.order = self.buy()
