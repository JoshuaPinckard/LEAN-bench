import backtrader as bt


class CCISlopeZScoreStrategy(bt.Strategy):
    params = (
        ("cci_period", 20),
        ("slope_period", 20),
        ("stats_period", 20),
        ("z_threshold", 2.0),
        ("stop_loss_pct", 0.02),
    )

    def __init__(self):
        self.cci = bt.indicators.CommodityChannelIndex(
            self.data, period=self.p.cci_period
        )
        self.slope = bt.indicators.LinearRegressionSlope(
            self.cci, period=self.p.slope_period
        )
        self.mean_slope = bt.indicators.SMA(
            self.slope, period=self.p.stats_period
        )
        self.std_slope = bt.indicators.StdDev(
            self.slope, period=self.p.stats_period
        )

        self.order = None
        self.stop_order = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            if order == self.order:
                self.order = None
                if self.position:
                    exec_price = order.executed.price
                    if self.position.size > 0:
                        stop_price = exec_price * (1.0 - self.p.stop_loss_pct)
                        self.stop_order = self.sell(
                            exectype=bt.Order.Stop, price=stop_price
                        )
                    elif self.position.size < 0:
                        stop_price = exec_price * (1.0 + self.p.stop_loss_pct)
                        self.stop_order = self.buy(
                            exectype=bt.Order.Stop, price=stop_price
                        )
            elif order == self.stop_order:
                self.stop_order = None

        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            if order == self.order:
                self.order = None
            if order == self.stop_order:
                self.stop_order = None

    def next(self):
        if self.order:
            return

        if not self.position:
            std_val = self.std_slope[0]
            if std_val == 0:
                return

            slope_val = self.slope[0]
            mean_val = self.mean_slope[0]
            z = (slope_val - mean_val) / std_val

            if z > self.p.z_threshold:
                self.order = self.buy()
            elif z < -self.p.z_threshold:
                self.order = self.sell()
        else:
            slope_val = self.slope[0]
            mean_val = self.mean_slope[0]

            if self.position.size > 0:
                if slope_val <= mean_val:
                    if self.stop_order:
                        self.cancel(self.stop_order)
                        self.stop_order = None
                    self.order = self.close()
            elif self.position.size < 0:
                if slope_val >= mean_val:
                    if self.stop_order:
                        self.cancel(self.stop_order)
                        self.stop_order = None
                    self.order = self.close()
