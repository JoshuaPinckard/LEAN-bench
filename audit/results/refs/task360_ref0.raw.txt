import backtrader as bt


class BollingerAtrTrailStrategy(bt.Strategy):
    params = (
        ('boll_period', 20),
        ('boll_devfactor', 2.0),
        ('atr_period', 14),
        ('atr_avg_period', 20),
        ('atr_stop_mult', 2.0),
    )

    def __init__(self):
        self.boll = bt.indicators.BollingerBands(
            self.data, period=self.p.boll_period, devfactor=self.p.boll_devfactor)
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.atr_sma = bt.indicators.SMA(self.atr, period=self.p.atr_avg_period)

        self.cross_top = bt.indicators.CrossOver(self.data.close, self.boll.lines.top)
        self.cross_bot = bt.indicators.CrossOver(self.data.close, self.boll.lines.bot)

        self.order = None
        self.entry_price = None
        self.stop_price = None
        self.highest_close = None
        self.lowest_close = None

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return
        if order.status == order.Completed:
            if self.position.size != 0:
                self.entry_price = order.executed.price
                self.highest_close = self.data.close[0]
                self.lowest_close = self.data.close[0]
                if self.position.size > 0:
                    self.stop_price = self.entry_price - self.p.atr_stop_mult * self.atr[0]
                else:
                    self.stop_price = self.entry_price + self.p.atr_stop_mult * self.atr[0]
            else:
                self.entry_price = None
                self.stop_price = None
                self.highest_close = None
                self.lowest_close = None
        self.order = None

    def next(self):
        if self.order:
            return

        vol_filter = self.atr[0] > self.atr_sma[0]
        long_signal = self.cross_top[0] == 1 and vol_filter
        short_signal = self.cross_bot[0] == -1 and vol_filter

        if not self.position:
            if long_signal:
                self.order = self.buy()
            elif short_signal:
                self.order = self.sell()
            return

        if self.position.size > 0:
            self.highest_close = max(self.highest_close, self.data.close[0])
            self.stop_price = max(
                self.stop_price, self.highest_close - self.p.atr_stop_mult * self.atr[0])
            if self.data.close[0] <= self.stop_price or short_signal:
                self.order = self.close()
        elif self.position.size < 0:
            self.lowest_close = min(self.lowest_close, self.data.close[0])
            self.stop_price = min(
                self.stop_price, self.lowest_close + self.p.atr_stop_mult * self.atr[0])
            if self.data.close[0] >= self.stop_price or long_signal:
                self.order = self.close()
