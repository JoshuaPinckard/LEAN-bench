import backtrader as bt
import collections
import statistics


class AlphatrendSqueezeMomentumStrategy(bt.Strategy):

    def __init__(self):
        self.ema50 = bt.indicators.EMA(self.data.close, period=50)
        self.atr = bt.indicators.ATR(self.data, period=14)
        self.bb = bt.indicators.BollingerBands(self.data.close, period=20, devfactor=2)
        self.macd = bt.indicators.MACD(self.data.close, period_me1=12, period_me2=26, period_signal=9)

        self.bbwidth = self.bb.top - self.bb.bot
        self.bbwidth_sma = bt.indicators.SMA(self.bbwidth, period=100)
        self.macd_hist = self.macd.macd - self.macd.signal

        self.order = None
        self.pending_entry = False
        self.pending_entry_atr = None
        self.entry_price = None
        self.take_profit_price = None
        self.trade_pnls = collections.deque(maxlen=20)

    def _squeeze_recent(self):
        for i in range(5):
            if self.bbwidth[-i] <= self.bbwidth_sma[-i] * 0.85:
                return True
        return False

    def _risk_fraction(self):
        if len(self.trade_pnls) < 10:
            return 0.01
        wins = [p for p in self.trade_pnls if p > 0]
        losses = [abs(p) for p in self.trade_pnls if p < 0]
        if not wins or not losses:
            return 0.01
        w = len(wins) / len(self.trade_pnls)
        avg_win = statistics.mean(wins)
        avg_loss = statistics.mean(losses)
        if avg_loss <= 0:
            return 0.01
        r = avg_win / avg_loss
        if r <= 0:
            return 0.01
        kelly = w - (1 - w) / r
        return max(0.005, min(0.02, kelly))

    def _compute_size(self, price, stop_price, risk_fraction):
        risk_per_share = abs(price - stop_price)
        if risk_per_share <= 0:
            return 0
        cash_at_risk = risk_fraction * self.broker.getvalue()
        return int(cash_at_risk / risk_per_share)

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return
        if order.status == order.Completed:
            if self.pending_entry:
                self.entry_price = order.executed.price
                if order.isbuy():
                    self.take_profit_price = self.entry_price + 1.5 * self.pending_entry_atr
                else:
                    self.take_profit_price = self.entry_price - 1.5 * self.pending_entry_atr
                self.pending_entry = False
            self.order = None
        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            self.pending_entry = False
            self.order = None

    def notify_trade(self, trade):
        if trade.isclosed:
            self.trade_pnls.append(trade.pnlcomm)
            self.entry_price = None
            self.take_profit_price = None

    def next(self):
        if self.order:
            return

        dt = self.data.datetime.time(0)
        session_end = (dt.hour, dt.minute) >= (15, 30)

        close = self.data.close[0]
        ema = self.ema50[0]
        atr = self.atr[0]

        if self.position:
            is_long = self.position.size > 0
            if session_end:
                self.order = self.close()
                return
            if is_long:
                stop_level = ema - 0.5 * atr
                if close < stop_level:
                    self.order = self.close()
                    return
                if self.take_profit_price is not None and close >= self.take_profit_price:
                    self.order = self.close()
                    return
            else:
                stop_level = ema + 0.5 * atr
                if close > stop_level:
                    self.order = self.close()
                    return
                if self.take_profit_price is not None and close <= self.take_profit_price:
                    self.order = self.close()
                    return
            return

        uptrend = close > ema and (close - ema) > 0.5 * atr
        downtrend = close < ema and (ema - close) > 0.5 * atr

        squeeze_recent = self._squeeze_recent()
        expansion = self.bbwidth[0] > self.bbwidth[-1]
        hist0 = self.macd_hist[0]
        hist_prev = self.macd_hist[-1]

        vol_long = squeeze_recent and expansion and hist0 > 0 and hist0 > hist_prev
        vol_short = squeeze_recent and expansion and hist0 < 0 and hist0 < hist_prev

        risk_fraction = self._risk_fraction()

        if uptrend:
            stop_price = ema - 0.5 * atr
            full_size = self._compute_size(close, stop_price, risk_fraction)
            size = full_size if vol_long else full_size // 2
            if size > 0:
                self.pending_entry = True
                self.pending_entry_atr = atr
                self.order = self.buy(size=size)
        elif downtrend:
            stop_price = ema + 0.5 * atr
            full_size = self._compute_size(close, stop_price, risk_fraction)
            size = full_size if vol_short else full_size // 2
            if size > 0:
                self.pending_entry = True
                self.pending_entry_atr = atr
                self.order = self.sell(size=size)
