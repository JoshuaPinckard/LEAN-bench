import backtrader as bt


class ADXRocBreakoutStrategy(bt.Strategy):
    params = (
        ("adx_period", 14),
        ("roc_period", 20),
        ("band_period", 20),
        ("stop_loss_pct", 0.02),
    )

    def __init__(self):
        self.adx = bt.indicators.ADX(self.data, period=self.p.adx_period)
        self.adx_roc = bt.indicators.ROC(self.adx, period=self.p.roc_period)
        self.roc_sma = bt.indicators.SimpleMovingAverage(
            self.adx_roc, period=self.p.band_period
        )
        self.roc_std = bt.indicators.StandardDeviation(
            self.adx_roc, period=self.p.band_period
        )

        self.order = None
        self.stop_order = None
        self.exit_requested = False

    def next(self):
        if self.order:
            return

        if not self.position:
            if self.adx_roc[0] > self.roc_sma[0] + 2 * self.roc_std[0]:
                self.order = self.buy()
            elif self.adx_roc[0] < self.roc_sma[0] - 2 * self.roc_std[0]:
                self.order = self.sell()
            return

        crossed_below = self.adx_roc[-1] >= self.roc_sma[-1] and self.adx_roc[0] < self.roc_sma[0]
        crossed_above = self.adx_roc[-1] <= self.roc_sma[-1] and self.adx_roc[0] > self.roc_sma[0]

        if (
            (self.position.size > 0 and crossed_below)
            or (self.position.size < 0 and crossed_above)
        ):
            self.exit_requested = True
            if self.stop_order:
                self.cancel(self.stop_order)
            else:
                self.order = self.close()

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order == self.stop_order:
            if order.status in [
                order.Completed,
                order.Canceled,
                order.Margin,
                order.Rejected,
            ]:
                self.stop_order = None
                if self.exit_requested and self.position and not self.order:
                    self.exit_requested = False
                    self.order = self.close()
            return

        if order.status == order.Completed:
            if order.isbuy() and self.position.size > 0:
                self.stop_order = self.sell(
                    exectype=bt.Order.Stop,
                    price=order.executed.price * (1 - self.p.stop_loss_pct),
                )
            elif order.issell() and self.position.size < 0:
                self.stop_order = self.buy(
                    exectype=bt.Order.Stop,
                    price=order.executed.price * (1 + self.p.stop_loss_pct),
                )

        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            if order == self.order:
                self.order = None
