import backtrader as bt


class RangeBreakoutVolumeStrategy(bt.Strategy):
    params = (
        ("period", 20),
        ("volume_factor", 2.15),
        ("target_price", 28.00),
    )

    def __init__(self):
        self.order = None
        self.highest_high = bt.indicators.Highest(
            self.data.high(-1), period=self.p.period
        )
        self.sma_volume = bt.indicators.SMA(
            self.data.volume(-1), period=self.p.period
        )

    def notify_order(self, order):
        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            self.order = None

    def next(self):
        if self.order:
            return

        if not self.position:
            breakout = self.data.open[0] > self.highest_high[0]
            vol_spike = (
                self.data.volume[0]
                > self.p.volume_factor * self.sma_volume[0]
            )

            if breakout and vol_spike:
                cash = self.broker.getcash()
                exec_price = self.data.close[0]
                size = int(cash / (exec_price * 1.001))
                if size > 0:
                    self.order = self.buy(size=size)
        else:
            if self.data.high[0] >= self.p.target_price:
                self.order = self.close()
