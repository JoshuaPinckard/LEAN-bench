```python
        self.stop_order = None
        self.limit_order = None
        self.trade_dir = 0
        self.entry_price = None
        self.entry_atr = None
        self.extreme = None
        self.trail_active = False

    @staticmethod
    def _acc(form, o, h, l, c, v):
        if h > form[1]:
            form[1] = h
        if l < form[2]:
            form[2] = l
        form[3] = c
        form[4] += v

    def _order_blocks(self):
        tf = self.tf60
        n = self.p.ob_period
        o = tf.o; h = tf.h; l = tf.l; c = tf.c
        N = len(c)
        bull = None
        bear = None
        # bullish OB: most recent down candle before a 20-bar breakout above Donchian high
        i = N - 1
        while i >= n:
            dch = max(h[i - n:i])
            if c[i] > dch:
                j = i - 1
                while j >= 0:
                    if c[j] < o[j]:
                        bull = (l[j], h[j])
                        break
                    j -= 1
                break
            i -= 1
        # bearish OB: most recent up candle before a 20-bar breakdown below Donchian low
        i = N - 1
        while i >= n:
            dcl = min(l[i - n:i])
            if c[i] < dcl:
                j = i - 1
                while j >= 0:
                    if c[j] > o[j]:
                        bear = (l[j], h[j])
                        break
                    j -= 1
                break
            i -= 1
        return bull, bear

    def next(self):
        d = self.data
        dt = d.datetime.datetime(0)
        o = d.open[0]; h = d.high[0]; l = d.low[0]; c = d.close[0]; v = d.volume[0]

        # --- base 60m timeframe (every bar is complete) ---
        self.tf60.update(o, h, l, c, v)
        m60 = self.tf60.score()
        atr60 = self.tf60.atr

        # --- daily bucket (by calendar date); score updates on completion ---
        dk = dt.date()
        if dk != self.cur_day:
            if self.form_d is not None:
                f = self.form_d
                self.tfd.update(f[0], f[1], f[2], f[3], f[4])
                self.daily_score = self.tfd.score()
            self.form_d = [o, h, l, c, v]
            self.cur_day = dk
        else:
            self._acc(self.form_d, o, h, l, c, v)

        # --- 4H bucket (by (date, hour//4)); score updates on completion ---
        hk = (dk, dt.hour // 4)
        if hk != self.cur_h4:
            if self.form_h is not None:
                f = self.form_h
                self.tfh4.update(f[0], f[1], f[2], f[3], f[4])
                self.h4_score = self.tfh4.score()
            self.form_h = [o, h, l, c, v]
            self.cur_h4 = hk
        else:
            self._acc(self.form_h, o, h, l, c, v)

        # --- pending-order guard ---
        if self.entry_order is not None or self.exit_order is not None:
            return

        # --- in-position management (trailing + early exit) ---
        if self.position:
            self._manage(m60, atr60)
            return

        if atr60 is None:
            return

        # --- entry evaluation with order-block filter ---
        bull_ob, bear_ob = self._order_blocks()
        price = c
        in_bull = bull_ob is not None and bull_ob[0] <= price <= bull_ob[1]
        in_bear = bear_ob is not None and bear_ob[0] <= price <= bear_ob[1]

        long_ok = (self.daily_score >= self.p.daily_long_th and
                   self.h4_score >= self.p.h4_long_th and
                   m60 >= self.p.m60_long_th and not in_bear)
        short_ok = (self.daily_score <= -self.p.daily_short_th and
                    self.h4_score <= -self.p.h4_short_th and
                    m60 <= -self.p.m60_short_th and not in_bull)

        if long_ok:
            stop = price - self.p.stop_atr_mult * atr60
            limit = price + self.p.tp_atr_mult * atr60
            br = self.buy_bracket(stopprice=stop, limitprice=limit)
            self.entry_order = br[0]; self.stop_order = br[1]; self.limit_order = br[2]
            self.trade_dir = 1
            self.entry_atr = atr60
        elif short_ok:
            stop = price + self.p.stop_atr_mult * atr60
            limit = price - self.p.tp_atr_mult * atr60
            br = self.sell_bracket(stopprice=stop, limitprice=limit)
            self.entry_order = br[0]; self.stop_order = br[1]; self.limit_order = br[2]
            self.trade_dir = -1
            self.entry_atr = atr60

    def _manage(self, m60, atr60):
        p = self.p
        # early exit: 60m score flip or daily filter no longer supports position
        if self.trade_dir == 1:
            if m60 <= -p.m60_short_th or self.daily_score < p.daily_long_th:
                self._exit_now()
                return
        else:
            if m60 >= p.m60_long_th or self.daily_score > -p.daily_short_th:
                self._exit_now()
                return

        # trailing stop
        if not p.use_trailing or atr60 is None or self.entry_price is None:
            return
        if self.trade_dir == 1:
            if self.extreme is None or self.data.high[0] > self.extreme:
                self.extreme = self.data.high[0]
            if (self.extreme - self.entry_price) >= p.trail_trigger_atr * atr60:
                self.trail_active = True
            if self.trail_active:
                ns = self.extreme - p.trail_atr_mult * atr60
                cur = self.stop_order.created.price if (self.stop_order is not None and self.stop_order.alive()) else None
                if cur is None or ns > cur:
                    self._replace_stop(ns)
        else:
            if self.extreme is None or self.data.low[0] < self.extreme:
                self.extreme = self.data.low[0]
            if (self.entry_price - self.extreme) >= p.trail_trigger_atr * atr60:
                self.trail_active = True
            if self.trail_active:
                ns = self.extreme + p.trail_atr_mult * atr60
                cur = self.stop_order.created.price if (self.stop_order is not None and self.stop_order.alive()) else None
                if cur is None or ns < cur:
                    self._replace_stop(ns)

    def _replace_stop(self, ns):
        if self.stop_order is not None and self.stop_order.alive():
            self.cancel(self.stop_order)
        sz = abs(self.position.size)
        if sz <= 0:
            self.stop_order = None
            return
        if self.trade_dir == 1:
            self.stop_order = self.sell(exectype=bt.Order.Stop, price=ns, size=sz)
        else:
            self.stop_order = self.buy(exectype=bt.Order.Stop, price=ns, size=sz)

    def _cancel_children(self):
        for od in (self.stop_order, self.limit_order):
            if od is not None and od.alive():
                self.cancel(od)

    def _exit_now(self):
        self._cancel_children()
        self.exit_order = self.close()

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return
        # order is done: Completed / Canceled / Margin / Rejected / Expired
        if order is self.entry_order:
            if order.status == order.Completed:
                self.entry_price = order.executed.price
                self.extreme = self.data.high[0] if self.trade_dir == 1 else self.data.low[0]
                self.trail_active = False
            else:
                # entry failed -> discard its (auto-cancelled) bracket children
                self.stop_order = None
                self.limit_order = None
                self.trade_dir = 0
            self.entry_order = None
        if order is self.exit_order:
            self.exit_order = None

    def notify_trade(self, trade):
        if trade.isclosed:
            self._cancel_children()
            self.stop_order = None
            self.limit_order = None
            self.trade_dir = 0
            self.entry_price = None
            self.entry_atr = None
            self.extreme = None
            self.trail_active = False
```
