import backtrader as bt
import math


class SyntheticHedgeStrategy(bt.Strategy):
    def __init__(self):
        self.sma = bt.indicators.SMA(self.data.close, period=50)
        self.atr = bt.indicators.ATR(self.data, period=14)

        self.bb_long = bt.indicators.BollingerBands(
            self.data.close, period=50, devfactor=2
        )
        self.bb_short = bt.indicators.BollingerBands(
            self.data.close, period=20, devfactor=2
        )

        self.macd = bt.indicators.MACD(
            self.data.close, period_me1=12, period_me2=26, period_signal=9
        )
        self.crossover = bt.indicators.CrossOver(self.data.close, self.sma)

        self.order = None

    def notify_order(self, order):
        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            self.order = None

    def _target_units(self, direction):
        close = self.data.close[0]
        sigma_long = (self.bb_long.top[0] - self.bb_long.mid[0]) / 2.0
        sigma_short = (self.bb_short.top[0] - self.bb_short.mid[0]) / 2.0

        if abs(sigma_short) <= 1e-12:
            hedge_coefficient = 0.0
        else:
            hedge_coefficient = max(0.0, min(1.5, sigma_long / sigma_short))

        units_main = math.floor(100000.0 / close)
        units_hedge = math.floor(units_main * hedge_coefficient)
        units_net = units_main - units_hedge

        return direction * units_net

    def next(self):
        if self.order is not None:
            return

        close = self.data.close[0]
        volatility_valid = (self.atr[0] / close) >= 0.005
        long_valid = (
            volatility_valid
            and self.macd.macd[0] > self.macd.signal[0]
            and self.macd.macd[0] > 0
        )
        short_valid = (
            volatility_valid
            and self.macd.macd[0] < self.macd.signal[0]
            and self.macd.macd[0] < 0
        )

        if self.position.size > 0:
            if self.crossover[0] < 0:
                self.order = self.close()
            elif long_valid:
                self.order = self.order_target_size(
                    target=self._target_units(1)
                )
            return

        if self.position.size < 0:
            if self.crossover[0] > 0:
                self.order = self.close()
            elif short_valid:
                self.order = self.order_target_size(
                    target=self._target_units(-1)
                )
            return

        if self.crossover[0] > 0 and long_valid:
            self.order = self.order_target_size(target=self._target_units(1))
        elif self.crossover[0] < 0 and short_valid:
            self.order = self.order_target_size(target=self._target_units(-1))
