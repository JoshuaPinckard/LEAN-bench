import backtrader as bt


class SyntheticCoveredCallStrategy(bt.Strategy):
    params = (
        ('cycle_length', 21),
        ('regular_sale_fraction', 0.40),
        ('roll_sale_fraction', 0.50),
    )

    def __init__(self):
        self.sma20 = bt.indicators.SimpleMovingAverage(self.data.close, period=20)
        self.boll = bt.indicators.BollingerBands(
            self.data.close, period=20, devfactor=2.0
        )
        self.atr14 = bt.indicators.ATR(self.data, period=14)

        self.order = None
        self.cycle_bar = 0
        self.strike = None
        self.repurchase_size = 0
        self.assignment_active = False

    def notify_order(self, order):
        if order.status in (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            self.order = None

    def _start_cycle(self):
        self.cycle_bar = 1
        self.strike = max(
            self.data.close[0] * 1.15,
            self.boll.top[0],
        )
        self.assignment_active = False

    def next(self):
        if self.order:
            return

        if not self.position:
            self.order = self.buy()
            return

        if self.strike is None:
            self._start_cycle()
            return

        remaining_bars = self.p.cycle_length - self.cycle_bar
        position_size = self.position.size

        if (
            remaining_bars <= 5
            and self.data.close[0] > self.strike + self.atr14[0]
            and position_size > 0
        ):
            sale_size = position_size * self.p.roll_sale_fraction
            if sale_size > 0:
                self.order = self.sell(size=sale_size)
                self.repurchase_size = sale_size
                self.assignment_active = True
            self._start_cycle()
            return

        if (
            self.assignment_active
            and self.repurchase_size > 0
            and (
                self.data.close[0] <= self.sma20[0]
                or self.data.low[0] <= self.boll.bot[0]
            )
        ):
            self.order = self.buy(size=self.repurchase_size)
            self.repurchase_size = 0
            self.assignment_active = False
            self.cycle_bar += 1
            return

        if (
            not self.assignment_active
            and remaining_bars > 2
            and self.data.high[0] >= self.strike
            and position_size > 0
        ):
            sale_size = position_size * self.p.regular_sale_fraction
            if sale_size > 0:
                self.order = self.sell(size=sale_size)
                self.repurchase_size = sale_size
                self.assignment_active = True

        self.cycle_bar += 1

        if self.cycle_bar > self.p.cycle_length:
            self._start_cycle()
