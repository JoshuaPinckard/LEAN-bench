import backtrader as bt


class TSIBullishDivergenceStrategy(bt.Strategy):
    params = dict(
        tsi_fast=25,
        tsi_slow=13,
        tsi_smooth=2,
    )

    def __init__(self):
        momentum = self.data.close - self.data.close(-1)
        abs_momentum = abs(momentum)

        ema1 = bt.indicators.EMA(momentum, period=self.p.tsi_fast)
        ema2 = bt.indicators.EMA(ema1, period=self.p.tsi_slow)
        ema3 = bt.indicators.EMA(ema2, period=self.p.tsi_smooth)

        abs_ema1 = bt.indicators.EMA(abs_momentum, period=self.p.tsi_fast)
        abs_ema2 = bt.indicators.EMA(abs_ema1, period=self.p.tsi_slow)
        abs_ema3 = bt.indicators.EMA(abs_ema2, period=self.p.tsi_smooth)

        self.tsi = 100.0 * (ema3 / abs_ema3)

        self.order = None
        self.stop_order = None
        self.close_order = None
        self.entry_stop_price = None

        self.last_pivot_low = None
        self.last_pivot_tsi = None

    def next(self):
        low_m2 = self.data.low[-2]
        low_m1 = self.data.low[-1]
        low_0 = self.data.low[0]

        is_pivot = low_m1 < low_m2 and low_m1 < low_0

        if is_pivot:
            pivot_price = low_m1
            pivot_tsi = self.tsi[-1]

            if (not self.position) and self.order is None and self.last_pivot_low is not None:
                if pivot_price < self.last_pivot_low and pivot_tsi > self.last_pivot_tsi:
                    self.entry_stop_price = self.data.low[0]
                    self.order = self.buy()

            self.last_pivot_low = pivot_price
            self.last_pivot_tsi = pivot_tsi

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.isbuy():
            if order.status == order.Completed:
                self.stop_order = self.close(exectype=bt.Order.Stop, price=self.entry_stop_price)
                self.close_order = self.close(exectype=bt.Order.Close, oco=self.stop_order)
            self.order = None
        else:
            if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
                self.stop_order = None
                self.close_order = None
                self.entry_stop_price = None
