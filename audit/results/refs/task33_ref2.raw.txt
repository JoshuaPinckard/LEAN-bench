import math
import backtrader as bt


class BullishReversalSellingClimax(bt.Strategy):
    params = (
        ('sma_period', 200),
        ('vol_sma_period', 50),
        ('atr_period', 14),
        ('climax_lookback', 10),
        ('vol_mult', 1.5),
        ('atr_stop_mult', 2.0),
        ('rr_ratio', 2.0),
        ('risk_pct', 0.01),
    )

    def __init__(self):
        self.sma200 = bt.indicators.SimpleMovingAverage(
            self.data.close, period=self.p.sma_period
        )
        self.vol_sma50 = bt.indicators.SimpleMovingAverage(
            self.data.volume, period=self.p.vol_sma_period
        )
        self.atr14 = bt.indicators.ATR(self.data, period=self.p.atr_period)

        self.order = None
        self.stop_price = None
        self.tp_price = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        if self.order:
            return

        if not self.position:
            # Trend Filter: Price must be above 200-day SMA
            if self.data.close[0] > self.sma200[0]:
                # Identify Selling Climax in the last 5-10 trading days
                climax_found = False
                climax_high = None

                for i in range(1, self.p.climax_lookback + 1):
                    is_down_day = self.data.close[-i] < self.data.open[-i]
                    is_vol_spike = (
                        self.data.volume[-i] > self.p.vol_mult * self.vol_sma50[-i]
                    )
                    if is_down_day and is_vol_spike:
                        climax_found = True
                        climax_high = self.data.high[-i]
                        break

                # Entry Signal: Close above high of selling climax day
                if climax_found and self.data.close[0] > climax_high:
                    entry_price = self.data.close[0]
                    stop_dist = self.p.atr_stop_mult * self.atr14[0]
                    if stop_dist <= 0:
                        return

                    stop_price = entry_price - stop_dist
                    tp_price = entry_price + (stop_dist * self.p.rr_ratio)

                    portfolio_value = self.broker.getvalue()
                    risk_amount = portfolio_value * self.p.risk_pct
                    size = math.floor(risk_amount / stop_dist)

                    if size > 0:
                        self.stop_price = stop_price
                        self.tp_price = tp_price
                        self.order = self.buy(size=size)
        else:
            # Exit Condition 3: Price closes below 200-day SMA
            if self.data.close[0] < self.sma200[0]:
                self.order = self.close()
                return

            # Exit Condition 1: Stop-loss
            if self.stop_price is not None and self.data.low[0] <= self.stop_price:
                self.order = self.close()
                return

            # Exit Condition 2: Take-profit
            if self.tp_price is not None and self.data.high[0] >= self.tp_price:
                self.order = self.close()
                return
