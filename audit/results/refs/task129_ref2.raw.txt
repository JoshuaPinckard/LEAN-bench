import backtrader as bt


class FallingChannelBreakout(bt.Strategy):

    def __init__(self):
        self.bb = bt.indicators.BollingerBands(self.data, period=20, devfactor=2)
        self.vol_ma = bt.indicators.SimpleMovingAverage(self.data.volume, period=10)

        self.order = None
        self.stop_order = None
        self.tp_order = None
        self.pending_channel_height = 0.0
        self.pending_stop_price = 0.0
        self.peak_equity = 0.0

    def start(self):
        self.peak_equity = self.broker.getvalue()

    def notify_order(self, order):
        if order.status in [order.Canceled, order.Margin, order.Rejected]:
            if order is self.order:
                self.order = None
            elif order is self.stop_order:
                self.stop_order = None
                tp = self.tp_order
                self.tp_order = None
                if tp is not None:
                    self.cancel(tp)
            elif order is self.tp_order:
                self.tp_order = None
                sl = self.stop_order
                self.stop_order = None
                if sl is not None:
                    self.cancel(sl)
        elif order.status == order.Completed:
            if order is self.order:
                entry_price = order.executed.price
                tp_price = entry_price + self.pending_channel_height
                exec_size = order.executed.size
                self.stop_order = self.sell(
                    exectype=bt.Order.Stop,
                    price=self.pending_stop_price,
                    size=exec_size,
                )
                self.tp_order = self.sell(
                    exectype=bt.Order.Limit,
                    price=tp_price,
                    size=exec_size,
                )
                self.order = None
            elif order is self.stop_order:
                self.stop_order = None
                tp = self.tp_order
                self.tp_order = None
                if tp is not None:
                    self.cancel(tp)
            elif order is self.tp_order:
                self.tp_order = None
                sl = self.stop_order
                self.stop_order = None
                if sl is not None:
                    self.cancel(sl)

    def _check_falling_channel(self):
        window = 20
        # bars [-window] to [-1]: chronological order, current bar excluded
        highs = [self.data.high[-(window - i)] for i in range(window)]
        lows = [self.data.low[-(window - i)] for i in range(window)]

        swing_highs = []
        for i in range(1, window - 1):
            if highs[i] > highs[i - 1] and highs[i] > highs[i + 1]:
                swing_highs.append(highs[i])

        swing_lows = []
        for i in range(1, window - 1):
            if lows[i] < lows[i - 1] and lows[i] < lows[i + 1]:
                swing_lows.append(lows[i])

        lower_highs = any(
            swing_highs[i] > swing_highs[i + 1] > swing_highs[i + 2]
            for i in range(len(swing_highs) - 2)
        )
        lower_lows = any(
            swing_lows[i] > swing_lows[i + 1] > swing_lows[i + 2]
            for i in range(len(swing_lows) - 2)
        )

        return lower_highs and lower_lows

    def next(self):
        current_equity = self.broker.getvalue()
        if current_equity > self.peak_equity:
            self.peak_equity = current_equity

        drawdown = (self.peak_equity - current_equity) / self.peak_equity
        if drawdown > 0.10:
            return

        # Close position if both exit orders are unexpectedly gone
        if self.position and self.order is None and self.stop_order is None and self.tp_order is None:
            self.close()
            return

        if self.order is not None or self.position:
            return

        upper_band = self.bb.lines.top[0]
        lower_band = self.bb.lines.bot[0]
        close = self.data.close[0]

        if close <= upper_band:
            return

        if not self._check_falling_channel():
            return

        if self.data.volume[0] <= self.vol_ma[0]:
            return

        channel_height = upper_band - lower_band
        stop_price = lower_band

        risk_per_share = close - stop_price
        if risk_per_share <= 0:
            return

        equity = self.broker.getvalue()
        size = int((equity * 0.01) / risk_per_share)
        if size <= 0:
            return

        self.pending_channel_height = channel_height
        self.pending_stop_price = stop_price
        self.order = self.buy(size=size)
