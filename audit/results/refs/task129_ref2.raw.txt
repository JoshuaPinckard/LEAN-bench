import backtrader as bt


class FallingChannelBreakout(bt.Strategy):
    params = (
        ("bb_period", 20),
        ("bb_dev", 2.0),
        ("vol_period", 10),
        ("risk_pct", 0.01),
        ("max_drawdown", 0.10),
    )

    def __init__(self):
        self.bb = bt.indicators.BollingerBands(
            self.data.close, period=self.p.bb_period, devfactor=self.p.bb_dev
        )
        self.vol_sma = bt.indicators.SimpleMovingAverage(
            self.data.volume, period=self.p.vol_period
        )

        self.order = None
        self.peak_equity = 0.0

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return
        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            if self.order and order.ref == self.order.ref:
                self.order = None

    def next(self):
        equity = self.broker.getvalue()
        if equity > self.peak_equity:
            self.peak_equity = equity

        drawdown = (
            (self.peak_equity - equity) / self.peak_equity
            if self.peak_equity > 0
            else 0.0
        )

        if self.position or self.order:
            return

        if drawdown > self.p.max_drawdown:
            return

        falling_structure = (
            self.data.high[-1]
            < self.data.high[-2]
            < self.data.high[-3]
            < self.data.high[-4]
        ) and (
            self.data.low[-1]
            < self.data.low[-2]
            < self.data.low[-3]
            < self.data.low[-4]
        )

        vol_spike = self.data.volume[0] > self.vol_sma[0]
        breakout = self.data.close[0] > self.bb.top[0]

        if falling_structure and vol_spike and breakout:
            channel_height = self.bb.top[0] - self.bb.bot[0]
            entry_price = self.data.close[0]
            sl_price = self.bb.bot[0]
            tp_price = entry_price + channel_height

            risk_per_share = entry_price - sl_price
            if risk_per_share <= 0:
                return

            risk_amount = equity * self.p.risk_pct
            size = int(risk_amount / risk_per_share)

            cash = self.broker.getcash()
            max_size = int(cash / entry_price)
            size = min(size, max_size)

            if size > 0:
                orders = self.buy_bracket(
                    size=size,
                    exectype=bt.Order.Market,
                    limitprice=tp_price,
                    stopprice=sl_price,
                )
                self.order = orders[0]
