import backtrader as bt


class VolBalancedMonoAssetStrategy(bt.Strategy):
    params = dict(
        atr_period=20,
        sma_period=20,
        boll_period=20,
        boll_devfactor=2,
        risk_pct=0.01,
        stop_atr_mult=2.0,
        tp_atr_mult=4.0,
        min_size=0.01,
    )

    def __init__(self):
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.sma = bt.indicators.SMA(self.data.close, period=self.p.sma_period)
        self.boll = bt.indicators.BollingerBands(
            self.data.close, period=self.p.boll_period, devfactor=self.p.boll_devfactor
        )
        self.crossover = bt.indicators.CrossOver(self.data.close, self.sma)

        self.order = None
        self.stop_price = None
        self.tp_price = None

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return
        if order.status in (order.Completed, order.Canceled, order.Margin, order.Rejected):
            self.order = None
        if not self.position:
            self.stop_price = None
            self.tp_price = None

    def next(self):
        if self.order:
            return

        atr = self.atr[0]
        close = self.data.close[0]
        capital = self.broker.getvalue()

        raw_size = min((capital * self.p.risk_pct) / (self.p.stop_atr_mult * atr), capital / close)
        size = max(raw_size, self.p.min_size)

        if self.position:
            if self.position.size > 0:
                if close <= self.stop_price or close >= self.tp_price:
                    self.order = self.close()
                    return
            else:
                if close >= self.stop_price or close <= self.tp_price:
                    self.order = self.close()
                    return

            if close > self.boll.top[0] or close < self.boll.bot[0]:
                self.order = self.close()
                return

            target = size if self.position.size > 0 else -size
            if abs(target - self.position.size) > 1e-9:
                self.order = self.order_target_size(target=target)
            return

        if self.crossover[0] > 0:
            self.stop_price = close - self.p.stop_atr_mult * atr
            self.tp_price = close + self.p.tp_atr_mult * atr
            self.order = self.buy(size=size)
        elif self.crossover[0] < 0:
            self.stop_price = close + self.p.stop_atr_mult * atr
            self.tp_price = close - self.p.tp_atr_mult * atr
            self.order = self.sell(size=size)
