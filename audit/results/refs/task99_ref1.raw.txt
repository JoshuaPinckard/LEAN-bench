import backtrader as bt
import math


class AccumulationBreakoutStrategy(bt.Strategy):
    params = (
        ('zone_low', 165.0),
        ('zone_high', 170.0),
        ('target', 240.0),
        ('risk_fraction', 0.02),
    )

    def __init__(self):
        self.order = None
        self.zone_entry_used = False
        self.trendline_lows = []

    def notify_order(self, order):
        if order.status in (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            self.order = None

    def _trendline_value(self):
        count = len(self.trendline_lows)
        if count < 2:
            return None

        x_mean = (count - 1) / 2.0
        y_mean = sum(self.trendline_lows) / count
        numerator = sum(
            (index - x_mean) * (value - y_mean)
            for index, value in enumerate(self.trendline_lows)
        )
        denominator = sum(
            (index - x_mean) ** 2
            for index in range(count)
        )

        if denominator == 0:
            return None

        slope = numerator / denominator
        if slope < 0:
            return None

        intercept = y_mean - slope * x_mean
        return intercept + slope * (count - 1)

    def next(self):
        current_date = self.data.datetime.date(0)

        if current_date.month >= 8:
            self.trendline_lows.append(float(self.data.low[0]))

        trendline = self._trendline_value()
        low = float(self.data.low[0])
        high = float(self.data.high[0])
        close = float(self.data.close[0])
        open_price = float(self.data.open[0])

        in_zone = low <= self.p.zone_high and high >= self.p.zone_low
        if not in_zone:
            self.zone_entry_used = False

        if self.order:
            return

        if self.position:
            if high >= self.p.target or close < self.p.zone_low:
                self.order = self.close()
            return

        lower_wick = min(open_price, close) - low
        real_body = abs(close - open_price)
        long_lower_wick = lower_wick >= real_body and lower_wick > 0
        bullish_support_candle = close > open_price and in_zone
        buyer_absorption = long_lower_wick or bullish_support_candle

        if (
            not self.zone_entry_used
            and in_zone
            and trendline is not None
            and close >= trendline
            and buyer_absorption
            and close > self.p.zone_low
        ):
            risk_per_share = close - self.p.zone_low
            risk_capital = self.broker.getvalue() * self.p.risk_fraction
            risk_size = math.floor(risk_capital / risk_per_share)

            commission_info = self.broker.getcommissioninfo(self.data)
            commission = commission_info.p.commission
            cash_size = math.floor(
                self.broker.getcash() / (close * (1.0 + commission))
            )
            size = min(risk_size, cash_size)

            if size > 0:
                self.zone_entry_used = True
                self.order = self.buy(size=size)
