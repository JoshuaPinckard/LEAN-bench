import backtrader as bt
from collections import deque


class EMABreakoutRetestStrategy(bt.Strategy):
    def __init__(self):
        self.order = None
        self.entry_price = None

        self._h4_buffer = []
        self._h4_count = 0

        self._ema = None
        self._previous_ema = None
        self._previous_close = None
        self._ema_seed = deque(maxlen=200)
        self._sma_closes = deque(maxlen=60)
        self._sma60 = None

        self._broken_below_ema = False
        self._break_h4_count = None

    def notify_order(self, order):
        if order.status == order.Completed:
            if order.isbuy():
                self.entry_price = order.executed.price
            elif order.issell():
                self.entry_price = None
            self.order = None
        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        self._h4_buffer.append((
            float(self.data.open[0]),
            float(self.data.high[0]),
            float(self.data.low[0]),
            float(self.data.close[0]),
        ))

        if len(self._h4_buffer) < 4:
            return

        bars = self._h4_buffer
        self._h4_buffer = []

        h4_open = bars[0][0]
        h4_high = max(bar[1] for bar in bars)
        h4_low = min(bar[2] for bar in bars)
        h4_close = bars[-1][3]

        self._on_h4_bar(h4_open, h4_high, h4_low, h4_close)

    def _on_h4_bar(self, h4_open, h4_high, h4_low, h4_close):
        self._h4_count += 1

        previous_ema = self._ema
        previous_close = self._previous_close

        self._ema_seed.append(h4_close)
        if self._ema is None:
            if len(self._ema_seed) == 200:
                self._ema = sum(self._ema_seed) / 200.0
        else:
            self._ema = self._ema + (2.0 / 201.0) * (h4_close - self._ema)

        self._sma_closes.append(h4_close)
        if len(self._sma_closes) == 60:
            self._sma60 = sum(self._sma_closes) / 60.0
        else:
            self._sma60 = None

        if self.position and self.order is None and self.entry_price is not None:
            stop_price = self.entry_price * 0.70
            target_price = (
                self._sma60
                if self._sma60 is not None
                else self.entry_price * 1.05
            )

            if h4_low <= stop_price:
                self.order = self.close()
            elif h4_high >= target_price:
                self.order = self.close()

        if (
            not self.position
            and self.order is None
            and self._ema is not None
            and previous_ema is not None
            and previous_close is not None
        ):
            if previous_close > previous_ema and h4_close < self._ema:
                self._broken_below_ema = True
                self._break_h4_count = self._h4_count

            if (
                self._broken_below_ema
                and self._h4_count > self._break_h4_count
                and h4_low <= self._ema <= h4_high
                and h4_close > h4_open
                and h4_close > self._ema
            ):
                size = (self.broker.getvalue() * 0.01) / h4_close
                self.order = self.buy(size=size)
                self._broken_below_ema = False
                self._break_h4_count = None

        self._previous_ema = self._ema
        self._previous_close = h4_close
