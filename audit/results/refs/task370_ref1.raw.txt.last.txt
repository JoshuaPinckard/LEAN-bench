import backtrader as bt


class IntradayVWAPTrendStrategy(bt.Strategy):
    class IntradayVWAP(bt.Indicator):
        lines = ("vwap",)

        def __init__(self):
            self._last_date = None
            self._cum_typical_volume = 0.0
            self._cum_volume = 0.0

        def next(self):
            current_date = self.data.datetime.date(0)

            if current_date != self._last_date:
                self._last_date = current_date
                self._cum_typical_volume = 0.0
                self._cum_volume = 0.0

            typical_price = (
                self.data.high[0] + self.data.low[0] + self.data.close[0]
            ) / 3.0
            volume = self.data.volume[0]

            self._cum_typical_volume += typical_price * volume
            self._cum_volume += volume

            self.lines.vwap[0] = (
                self._cum_typical_volume / self._cum_volume
                if self._cum_volume else typical_price
            )

    params = (
        ("atr_period", 14),
        ("atr_mult", 2.0),
        ("stop_pct", 0.01),
        ("use_atr_stop", True),
        ("use_pct_stop", True),
        ("stake", 1),
    )

    def __init__(self):
        self.vwap = self.IntradayVWAP(self.data)
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.crossover = bt.indicators.CrossOver(self.data.close, self.vwap)

        self.order = None
        self.order_role = None
        self.entry_price = None

    def notify_order(self, order):
        if order.status == order.Completed:
            if self.order_role == "entry":
                self.entry_price = order.executed.price
            elif self.order_role == "exit":
                self.entry_price = None

        if order.status in (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            self.order = None
            self.order_role = None

    def _close_position(self):
        self.order = self.close()
        self.order_role = "exit"

    def next(self):
        if self.order:
            return

        if not self.position:
            if self.crossover[0] > 0:
                self.order = self.buy(size=self.p.stake)
                self.order_role = "entry"
            elif self.crossover[0] < 0:
                self.order = self.sell(size=self.p.stake)
                self.order_role = "entry"
            return

        stop_levels = []

        if self.p.use_atr_stop:
            if self.position.size > 0:
                stop_levels.append(
                    self.entry_price - self.p.atr_mult * self.atr[0]
                )
            else:
                stop_levels.append(
                    self.entry_price + self.p.atr_mult * self.atr[0]
                )

        if self.p.use_pct_stop:
            if self.position.size > 0:
                stop_levels.append(self.entry_price * (1.0 - self.p.stop_pct))
            else:
                stop_levels.append(self.entry_price * (1.0 + self.p.stop_pct))

        if self.position.size > 0:
            if stop_levels:
                stop_level = max(stop_levels)
                if self.data.low[0] <= stop_level:
                    self._close_position()
                    return

            if self.crossover[0] < 0:
                self._close_position()
        else:
            if stop_levels:
                stop_level = min(stop_levels)
                if self.data.high[0] >= stop_level:
                    self._close_position()
                    return

            if self.crossover[0] > 0:
                self._close_position()
