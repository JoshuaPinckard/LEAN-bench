import backtrader as bt


class ConvertibleBondConversionStrategy(bt.Strategy):
    params = (
        ("sma_period", 50),
        ("rsi_period", 14),
        ("tp_price", 160.0),
        ("min_prev_close", 200.0),
        ("rsi_threshold", 50.0),
        ("stop_loss_pct", 0.05),
        ("position_pct", 0.02),
        ("max_hold_days", 5),
    )

    def __init__(self):
        self.sma = bt.indicators.SMA(self.data.close, period=self.p.sma_period)
        self.rsi = bt.indicators.RSI(self.data.close, period=self.p.rsi_period)

        self.order = None
        self.stop_order = None
        self.entry_bar = None
        self.entry_price = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            if order.issell():
                # Short entry order completed
                self.entry_bar = len(self)
                self.entry_price = order.executed.price
                stop_price = self.entry_price * (1.0 + self.p.stop_loss_pct)
                self.stop_order = self.buy(
                    exectype=bt.Order.Stop,
                    price=stop_price,
                    size=abs(order.executed.size),
                )

        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            if order == self.order:
                self.order = None
            if self.stop_order and order == self.stop_order:
                self.stop_order = None

    def next(self):
        # Do not submit new orders if an entry/exit order is pending
        if self.order:
            return

        if self.position:
            # Check exit conditions for open short position
            close_price = self.data.close[0]
            bars_held = (
                len(self) - self.entry_bar if self.entry_bar is not None else 0
            )

            # Exit Condition 1: Closing price reaches target level ($160.00)
            tp_signal = close_price <= self.p.tp_price

            # Exit Condition 2: Closing price rises above SMA(50)
            sma_signal = close_price > self.sma[0]

            # Exit Condition 3: After 5 trading days from entry
            time_signal = bars_held >= self.p.max_hold_days

            if tp_signal or sma_signal or time_signal:
                if self.stop_order:
                    self.cancel(self.stop_order)
                    self.stop_order = None
                self.order = self.close()
        else:
            dt = self.data.datetime.date(0)

            # Entry Filter 1 & Additional Rule: Strategy inactive before October 1
            if dt.month < 10:
                return

            # Additional Rule: Skip signal if asset price at entry time is below $160.00
            if self.data.open[0] < self.p.tp_price:
                return

            # Entry Condition 2: Previous day closing price > $200.00
            cond2 = self.data.close[-1] > self.p.min_prev_close

            # Entry Condition 3: Previous day RSI > 50
            cond3 = self.rsi[-1] > self.p.rsi_threshold

            # Entry Condition 4: Current day open < previous day close
            cond4 = self.data.open[0] < self.data.close[-1]

            if cond2 and cond3 and cond4:
                # Position sizing: 2% of total trading capital per trade
                capital = self.broker.getvalue()
                target_value = capital * self.p.position_pct
                size = int(target_value / self.data.open[0])
                if size < 1:
                    size = 1

                self.order = self.sell(size=size)
