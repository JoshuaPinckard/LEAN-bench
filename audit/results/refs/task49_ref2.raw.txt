import math
import statistics
import backtrader as bt


class LogisticRegressionStrategy(bt.Strategy):
    params = (
        ('train_ratio', 0.8),
        ('buy_threshold', 0.6),
        ('size', 10),
        ('commission', 10.0),
        ('learning_rate', 0.05),
        ('epochs', 1000),
    )

    def __init__(self):
        self.order = None
        self.broker.setcommission(commission=self.p.commission, commtype=bt.CommInfoBase.COMM_FIXED)

        # Technical indicators
        self.rsi = bt.indicators.RSI(self.data.close, period=14)
        self.mom = bt.indicators.Momentum(self.data.close, period=12)

        # Data collection for model training
        self.X_train = []
        self.y_train = []

        # Model state
        self.is_trained = False
        self.w0 = 0.0
        self.w1 = 0.0
        self.w2 = 0.0
        self.mean_x1 = 0.0
        self.std_x1 = 1.0
        self.mean_x2 = 0.0
        self.std_x2 = 1.0

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def _sigmoid(self, z):
        if z < -40.0:
            return 0.0
        elif z > 40.0:
            return 1.0
        return 1.0 / (1.0 + math.exp(-z))

    def _train_model(self):
        if not self.X_train:
            return

        n = len(self.X_train)
        x1_vals = [x[0] for x in self.X_train]
        x2_vals = [x[1] for x in self.X_train]

        self.mean_x1 = statistics.mean(x1_vals)
        self.std_x1 = statistics.stdev(x1_vals) if len(x1_vals) > 1 else 1.0
        if self.std_x1 == 0:
            self.std_x1 = 1.0

        self.mean_x2 = statistics.mean(x2_vals)
        self.std_x2 = statistics.stdev(x2_vals) if len(x2_vals) > 1 else 1.0
        if self.std_x2 == 0:
            self.std_x2 = 1.0

        # Normalize features
        norm_X = []
        for x1, x2 in self.X_train:
            z1 = (x1 - self.mean_x1) / self.std_x1
            z2 = (x2 - self.mean_x2) / self.std_x2
            norm_X.append((z1, z2))

        # Train logistic regression via Gradient Descent
        w0, w1, w2 = 0.0, 0.0, 0.0
        lr = self.p.learning_rate

        for _ in range(self.p.epochs):
            g0, g1, g2 = 0.0, 0.0, 0.0
            for i in range(n):
                z1, z2 = norm_X[i]
                y = self.y_train[i]
                pred = self._sigmoid(w0 + w1 * z1 + w2 * z2)
                err = pred - y
                g0 += err
                g1 += err * z1
                g2 += err * z2

            w0 -= lr * (g0 / n)
            w1 -= lr * (g1 / n)
            w2 -= lr * (g2 / n)

        self.w0 = w0
        self.w1 = w1
        self.w2 = w2
        self.is_trained = True

    def _predict_proba(self, x1, x2):
        z1 = (x1 - self.mean_x1) / self.std_x1
        z2 = (x2 - self.mean_x2) / self.std_x2
        return self._sigmoid(self.w0 + self.w1 * z1 + self.w2 * z2)

    def next(self):
        total_bars = self.data.buflen()
        train_cutoff = int(self.p.train_ratio * total_bars)

        # Training phase (first 80% of data)
        if len(self) <= train_cutoff:
            x1 = self.rsi[-1]
            x2 = self.mom[-1]
            y = 1.0 if self.data.close[0] >= self.data.open[0] else 0.0
            self.X_train.append((x1, x2))
            self.y_train.append(y)
            if len(self) == train_cutoff:
                self._train_model()
            return

        # Ensure model is trained
        if not self.is_trained:
            self._train_model()

        # Testing / Trading phase (remaining 20% of data)
        x1 = self.rsi[-1]
        x2 = self.mom[-1]
        prob_up = self._predict_proba(x1, x2)

        if self.order:
            return

        # Entry logic: Predict upward probability > 60%
        if not self.position:
            if prob_up > self.p.buy_threshold:
                self.order = self.buy(size=self.p.size)
        # Exit logic: Predict decline (class 0, i.e., prob_up <= 0.5)
        else:
            if prob_up <= 0.5:
                self.order = self.close()
