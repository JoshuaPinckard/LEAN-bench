import backtrader as bt


class VolatilityExtremeFadeStrategy(bt.Strategy):
    params = (
        ("stop_loss_pct", 0.02),
    )

    def __init__(self):
        self.atr = bt.indicators.ATR(self.data, period=14)
        self.atr_sma = bt.indicators.SMA(self.atr, period=20)
        self.atr_stddev = bt.indicators.StandardDeviation(self.atr, period=20)
        self.close_sma = bt.indicators.SMA(self.data.close, period=20)

        self.atr_cross_up = bt.indicators.CrossUp(self.atr, self.atr_sma)
        self.atr_cross_down = bt.indicators.CrossDown(self.atr, self.atr_sma)

        self.order = None
        self.stop_order = None
        self.exit_requested = False

    def next(self):
        if self.order is not None:
            return

        if self.position.size:
            if self.exit_requested:
                if self.stop_order is not None:
                    self.cancel(self.stop_order)
                    return

                self.order = self.close()
                self.exit_requested = False
                return

            if ((self.position.size > 0 and self.atr_cross_up[0]) or
                    (self.position.size < 0 and self.atr_cross_down[0])):
                self.exit_requested = True
                if self.stop_order is not None:
                    self.cancel(self.stop_order)
                return

            return

        self.exit_requested = False

        if (self.atr[0] < self.atr_sma[0] - 2 * self.atr_stddev[0] and
                self.data.close[0] < self.close_sma[0]):
            self.order = self.buy()

        elif (self.atr[0] > self.atr_sma[0] + 2 * self.atr_stddev[0] and
                self.data.close[0] > self.close_sma[0]):
            self.order = self.sell()

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return

        if order is self.stop_order:
            if order.status in (order.Completed, order.Canceled,
                                order.Margin, order.Rejected):
                self.stop_order = None
            return

        if order.status == order.Completed:
            if order is self.order and self.position.size:
                if self.position.size > 0:
                    self.stop_order = self.sell(
                        exectype=bt.Order.Stop,
                        price=order.executed.price * (1 - self.p.stop_loss_pct),
                    )
                else:
                    self.stop_order = self.buy(
                        exectype=bt.Order.Stop,
                        price=order.executed.price * (1 + self.p.stop_loss_pct),
                    )

        if order.status in (order.Completed, order.Canceled,
                            order.Margin, order.Rejected):
            if order is self.order:
                self.order = None
