import math
import backtrader as bt


class LogReturn(bt.Indicator):
    lines = ('log_ret',)

    def __init__(self):
        self.addminperiod(2)

    def next(self):
        self.lines.log_ret[0] = math.log(self.data[0] / self.data[-1])


class MeanVarianceStrategy(bt.Strategy):
    params = (
        ('lambda_param', 2.5),
        ('sma_period', 252),
        ('atr_period', 63),
        ('vol_threshold', 0.005),
        ('wmin', 0.0),
        ('wmax', 1.0),
    )

    def __init__(self):
        self.order = None
        log_ret = LogReturn(self.data.close)
        self.mu_hat = bt.ind.SMA(log_ret, period=self.p.sma_period)
        self.atr = bt.ind.ATR(self.data, period=self.p.atr_period)
        self.sigma_hat = self.atr / self.data.close

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        if self.order:
            return

        if self.data.datetime.date(0).month != self.data.datetime.date(-1).month:
            sigma = self.sigma_hat[0]
            if sigma < self.p.vol_threshold:
                w = 0.0
            else:
                mu = self.mu_hat[0]
                raw_w = mu / (self.p.lambda_param * (sigma ** 2))
                w = max(self.p.wmin, min(self.p.wmax, raw_w))

            self.order = self.order_target_percent(target=w)
