import backtrader as bt
import math


class MSTRSignalConflict(bt.Strategy):
    def __init__(self):
        self.sma365 = bt.ind.SMA(self.data.close, period=365)
        self.sma200 = bt.ind.SMA(self.data.close, period=200)
        self.sma10 = bt.ind.SMA(self.data.close, period=10)
        self.crossover365 = bt.ind.CrossOver(self.data.close, self.sma365)

        self.order = None
        self.sma365_history = []

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def _get_44th_percentile(self):
        data = sorted(self.sma365_history)
        n = len(data)
        if n == 0:
            return 0.0
        if n == 1:
            return data[0]
        k = (n - 1) * 0.44
        i = int(k)
        f = k - i
        if i + 1 < n:
            return data[i] + f * (data[i + 1] - data[i])
        return data[i]

    def next(self):
        self.sma365_history.append(self.sma365[0])

        if self.order:
            return

        p44 = self._get_44th_percentile()

        if not self.position:
            if self.crossover365[0] > 0 and self.sma365[0] > p44:
                self.order = self.buy()
        else:
            if self.crossover365[0] < 0 or self.data.close[0] < self.sma200[0]:
                self.order = self.close()
