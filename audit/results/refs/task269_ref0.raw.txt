    def _manage_exit(self, position, close):
        theta = self.theta_fixed
        sigma = self.sigma_fixed
        prev_close = self.data.close[-1]

        if position.size > 0:
            crossup = prev_close < theta and close >= theta
            sigma_stop = close < (theta - self.p.m * sigma)
            atr_stop = close < (self.entry_price - self.p.atr_stop_mult * self.entry_atr)
            if crossup or sigma_stop or atr_stop:
                self.order = self.close()
        elif position.size < 0:
            crossdown = prev_close > theta and close <= theta
            sigma_stop = close > (theta + self.p.m * sigma)
            atr_stop = close > (self.entry_price + self.p.atr_stop_mult * self.entry_atr)
            if crossdown or sigma_stop or atr_stop:
                self.order = self.close()

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return
        if order.status == order.Completed:
            if self.position.size != 0:
                self.entry_price = order.executed.price
        if order.status in (order.Completed, order.Canceled, order.Margin, order.Rejected):
            self.order = None

    def next(self):
        dt = self.data.datetime.date(0)

        if self.t0 is None:
            self.t0 = self.data.datetime.date(1 - len(self))
            self._set_window(0)

        while dt >= self.oos_end:
            self._set_window(self.window_idx + 1)

        if not self.calibrated and dt >= self.in_sample_end:
            self.theta_fixed = self.sma252[0]
            self.sigma_fixed = (self.bb.top[0] - self.bb.bot[0]) / (2 * self.bb.p.devfactor)
            self.calibrated = True

        if self.order:
            return

        position = self.position
        close = self.data.close[0]

        if position:
            self._manage_exit(position, close)
        else:
            in_trading = self.oos_start <= dt < self.oos_end
            if in_trading and self.theta_fixed is not None:
                self._evaluate_entry(close)
