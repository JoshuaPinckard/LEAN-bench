import backtrader as bt


class BollingerMeanReversionStrategy(bt.Strategy):
    params = (
        ('bb_period', 20),
        ('bb_devfactor', 2.0),
        ('atr_period', 14),
        ('atr_multiplier', 2.0),
    )

    def __init__(self):
        self.bbands = bt.indicators.BollingerBands(
            self.data.close,
            period=self.p.bb_period,
            devfactor=self.p.bb_devfactor,
        )
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)

        self.entry_order = None
        self.exit_order = None
        self.stop_order = None
        self.exit_after_stop_cancel = False
        self.entry_side = None

    def next(self):
        if self.entry_order or self.exit_order or self.exit_after_stop_cancel:
            return

        if not self.position:
            size = self.broker.getcash() / self.data.close[0]

            if self.data.close[0] < self.bbands.bot[0]:
                self.entry_side = 1
                self.entry_order = self.buy(size=size)
            elif self.data.close[0] > self.bbands.top[0]:
                self.entry_side = -1
                self.entry_order = self.sell(size=size)
            return

        returned_to_middle = (
            (self.position.size > 0 and self.data.close[0] >= self.bbands.mid[0])
            or
            (self.position.size < 0 and self.data.close[0] <= self.bbands.mid[0])
        )

        if returned_to_middle:
            if self.stop_order:
                self.exit_after_stop_cancel = True
                self.cancel(self.stop_order)
            else:
                self.exit_order = self.close()

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return

        if order is self.entry_order:
            self.entry_order = None

            if order.status == order.Completed:
                stop_distance = self.atr[0] * self.p.atr_multiplier
                size = abs(order.executed.size)

                if self.entry_side == 1:
                    self.stop_order = self.sell(
                        size=size,
                        exectype=bt.Order.Stop,
                        price=order.executed.price - stop_distance,
                    )
                else:
                    self.stop_order = self.buy(
                        size=size,
                        exectype=bt.Order.Stop,
                        price=order.executed.price + stop_distance,
                    )

        elif order is self.stop_order:
            self.stop_order = None

            if (
                order.status in (
                    order.Canceled,
                    order.Margin,
                    order.Rejected,
                )
                and self.exit_after_stop_cancel
            ):
                self.exit_after_stop_cancel = False
                self.exit_order = self.close()

        elif order is self.exit_order:
            self.exit_order = None
            self.entry_side = None
