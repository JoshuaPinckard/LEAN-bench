import datetime
import math
import backtrader as bt


class MomentumStrategy(bt.Strategy):
    def __init__(self):
        self.sma20 = bt.ind.SMA(self.data.close, period=20)
        self.sma50 = bt.ind.SMA(self.data.close, period=50)
        self.crossover = bt.ind.CrossOver(self.sma20, self.sma50)
        self.rsi = bt.ind.RSI(self.data.close, period=14)
        self.vol_sma20 = bt.ind.SMA(self.data.volume, period=20)
        self.atr20 = bt.ind.ATR(self.data, period=20)
        self.atr_sma1950 = bt.ind.SMA(self.atr20, period=1950)

        self.order = None
        self.entry_price = None
        self.initial_stop = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            if order.isbuy():
                self.entry_price = order.executed.price
            elif order.issell():
                self.entry_price = None
                self.initial_stop = None
        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            pass

        self.order = None

    def next(self):
        if self.order:
            return

        if self.position.size > 0:
            entry_p = (
                self.entry_price
                if self.entry_price is not None
                else self.position.price
            )

            stop_hit = (
                self.initial_stop is not None
                and self.data.close[0] < self.initial_stop
            )
            tp_hit = (
                entry_p is not None and self.data.close[0] >= entry_p * 1.02
            )
            cross_down = self.crossover[0] == -1

            if stop_hit or tp_hit or cross_down:
                self.order = self.close()
            return

        current_time = self.data.datetime.time(0)
        in_time_window = (
            datetime.time(9, 30) <= current_time <= datetime.time(16, 0)
        )

        price_valid = 10.0 <= self.data.close[0] <= 200.0
        volume_valid = self.data.volume[0] > self.vol_sma20[0] * 1.2
        crossover_valid = any(self.crossover[-i] == 1 for i in range(6))
        rsi_valid = 40.0 <= self.rsi[0] <= 60.0
        volatility_valid = self.atr20[0] > self.atr_sma1950[0]

        if (
            in_time_window
            and price_valid
            and volume_valid
            and crossover_valid
            and rsi_valid
            and volatility_valid
        ):
            entry_price_est = self.data.close[0]
            last_10_lows = [self.data.low[-i] for i in range(10)]
            stop_price = min(last_10_lows)

            dist = entry_price_est - stop_price
            if dist <= 0:
                return

            if dist > 0.02 * entry_price_est:
                return

            equity = self.broker.getvalue()
            risk_amount = equity * 0.01
            qty = math.floor(risk_amount / dist)

            if qty > 0:
                self.initial_stop = stop_price
                self.order = self.buy(size=qty)
