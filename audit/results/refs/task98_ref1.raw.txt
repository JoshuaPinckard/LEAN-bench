import backtrader as bt
import math


class MeanReversionStrategy(bt.Strategy):
    params = (
        ("risk_percent", 0.01),
        ("volume_period", 20),
        ("volume_threshold", 0),
        ("minimum_history", 200),
    )

    def __init__(self):
        self.order = None
        self.entry_order = None
        self.stop_order = None
        self.take_profit_order = None
        self.entry_std = None
        self.entry_target = None

        self.rsi = bt.indicators.RSI(self.data.close, period=14)
        self.fast_sma = bt.indicators.SMA(self.data.close, period=20)
        self.slow_sma = bt.indicators.SMA(self.data.close, period=50)
        self.bbands = bt.indicators.BollingerBands(
            self.data.close, period=20, devfactor=2.0
        )
        self.volume_sum = bt.indicators.SumN(
            self.data.volume, period=self.p.volume_period
        )
        self.middle_cross = bt.indicators.CrossOver(
            self.data.close, self.bbands.mid
        )

    def _size(self):
        capital = self.broker.getvalue()
        return int(math.floor(
            (capital * self.p.risk_percent) / self.data.close[0]
        ))

    def _cancel_protection(self):
        if self.stop_order is not None:
            self.cancel(self.stop_order)
        if self.take_profit_order is not None:
            self.cancel(self.take_profit_order)

    def next(self):
        if len(self.data) < self.p.minimum_history:
            return

        if self.order is not None:
            return

        if self.position:
            if self.position.size > 0:
                if self.middle_cross[0] > 0 or self.rsi[0] > 70:
                    self._cancel_protection()
                    self.order = self.close()
            else:
                if self.middle_cross[0] < 0 or self.rsi[0] < 30:
                    self._cancel_protection()
                    self.order = self.close()
            return

        if self.volume_sum[0] <= self.p.volume_threshold:
            return

        size = self._size()
        if size <= 0:
            return

        if (
            self.data.close[0] < self.bbands.bot[0]
            and self.rsi[0] < 30
            and self.fast_sma[0] > self.slow_sma[0]
        ):
            self.entry_std = self.bbands.stddev[0]
            self.entry_target = self.bbands.mid[0]
            self.entry_order = self.buy(size=size)
            self.order = self.entry_order

        elif (
            self.data.close[0] > self.bbands.top[0]
            and self.rsi[0] > 70
            and self.fast_sma[0] < self.slow_sma[0]
        ):
            self.entry_std = self.bbands.stddev[0]
            self.entry_target = self.bbands.mid[0]
            self.entry_order = self.sell(size=size)
            self.order = self.entry_order

    def notify_order(self, order):
        if order.status in (
            order.Submitted,
            order.Accepted,
        ):
            return

        if order.status == order.Completed:
            if order is self.entry_order:
                self.order = None
                self.entry_order = None
                size = abs(order.executed.size)

                if order.isbuy():
                    self.stop_order = self.sell(
                        size=size,
                        exectype=bt.Order.Stop,
                        price=order.executed.price - (1.5 * self.entry_std),
                    )
                    self.take_profit_order = self.sell(
                        size=size,
                        exectype=bt.Order.Limit,
                        price=self.entry_target,
                    )
                else:
                    self.stop_order = self.buy(
                        size=size,
                        exectype=bt.Order.Stop,
                        price=order.executed.price + (1.5 * self.entry_std),
                    )
                    self.take_profit_order = self.buy(
                        size=size,
                        exectype=bt.Order.Limit,
                        price=self.entry_target,
                    )

            elif order is self.stop_order:
                self.stop_order = None
                if self.take_profit_order is not None:
                    self.cancel(self.take_profit_order)

            elif order is self.take_profit_order:
                self.take_profit_order = None
                if self.stop_order is not None:
                    self.cancel(self.stop_order)

            elif order is self.order:
                self.order = None
                self._cancel_protection()

        elif order.status in (
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            if order is self.order:
                self.order = None
            if order is self.entry_order:
                self.entry_order = None
            if order is self.stop_order:
                self.stop_order = None
            if order is self.take_profit_order:
                self.take_profit_order = None
