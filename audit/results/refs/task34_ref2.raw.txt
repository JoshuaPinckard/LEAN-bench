import backtrader as bt


class WyckoffRangeBreakout(bt.Strategy):
    params = (
        ("lookback", 20),
        ("touch_tolerance", 0.005),
        ("min_range_pct", 0.02),
        ("vol_factor", 1.2),
        ("risk_pct", 0.01),
        ("sl_pct", 0.005),
        ("cooldown_bars", 5),
    )

    def __init__(self):
        self.sma20 = bt.indicators.SMA(self.data.close, period=20)
        self.vol_sma20 = bt.indicators.SMA(self.data.volume, period=20)
        self.order = None
        self.last_entry_bar = -999
        self.sl_price = None
        self.tp_price = None
        self.entry_range_sup = None
        self.entry_range_res = None

    def notify_order(self, order):
        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            self.order = None

    def next(self):
        if self.order:
            return

        if self.position:
            pos_size = self.position.size
            close_price = self.data.close[0]
            high_price = self.data.high[0]
            low_price = self.data.low[0]

            if pos_size > 0:
                if low_price <= self.sl_price:
                    self.close()
                    return
                if high_price >= self.tp_price:
                    self.close()
                    return
                if self.entry_range_sup <= close_price <= self.entry_range_res:
                    self.close()
                    return

            elif pos_size < 0:
                if high_price >= self.sl_price:
                    self.close()
                    return
                if low_price <= self.tp_price:
                    self.close()
                    return
                if self.entry_range_sup <= close_price <= self.entry_range_res:
                    self.close()
                    return
            return

        if len(self) - self.last_entry_bar < self.p.cooldown_bars:
            return

        if len(self) < self.p.lookback + 1:
            return

        if self.data.close[0] <= self.sma20[0]:
            return

        highs = [self.data.high[-i] for i in range(1, self.p.lookback + 1)]
        lows = [self.data.low[-i] for i in range(1, self.p.lookback + 1)]
        res = max(highs)
        sup = min(lows)

        high_touches = sum(
            1 for h in highs if h >= res * (1.0 - self.p.touch_tolerance)
        )
        low_touches = sum(
            1 for l in lows if l <= sup * (1.0 + self.p.touch_tolerance)
        )

        if high_touches < 2 or low_touches < 2:
            return

        range_height = res - sup
        if range_height / self.data.close[0] < self.p.min_range_pct:
            return

        if self.data.volume[0] < self.p.vol_factor * self.vol_sma20[0]:
            return

        capital = self.broker.getvalue()

        if self.data.close[0] > res:
            sl = sup * (1.0 - self.p.sl_pct)
            tp = res + range_height
            risk_per_share = self.data.close[0] - sl
            if risk_per_share > 0:
                risk_amount = capital * self.p.risk_pct
                size = int(risk_amount / risk_per_share)
                if size > 0:
                    self.sl_price = sl
                    self.tp_price = tp
                    self.entry_range_sup = sup
                    self.entry_range_res = res
                    self.last_entry_bar = len(self)
                    self.order = self.buy(size=size)

        elif self.data.close[0] < sup:
            sl = res * (1.0 + self.p.sl_pct)
            tp = sup - range_height
            risk_per_share = sl - self.data.close[0]
            if risk_per_share > 0:
                risk_amount = capital * self.p.risk_pct
                size = int(risk_amount / risk_per_share)
                if size > 0:
                    self.sl_price = sl
                    self.tp_price = tp
                    self.entry_range_sup = sup
                    self.entry_range_res = res
                    self.last_entry_bar = len(self)
                    self.order = self.sell(size=size)
