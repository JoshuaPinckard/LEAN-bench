import backtrader as bt


class VolatilitySpikeStrategy(bt.Strategy):
    def __init__(self):
        self.sma20 = bt.ind.SMA(self.data.close, period=20)
        self.sma50 = bt.ind.SMA(self.data.close, period=50)

        self.bb = bt.ind.BollingerBands(self.data.close, period=20, devfactor=2)
        self.bb_width = self.bb.lines.top - self.bb.lines.bot
        self.sma_bb_width = bt.ind.SMA(self.bb_width, period=20)

        self.atr = bt.ind.ATR(self.data, period=14)
        self.sma_vol = bt.ind.SMA(self.data.volume, period=20)

        self.co_long = bt.ind.CrossOver(self.data.close, self.bb.lines.top)
        self.co_short = bt.ind.CrossOver(self.data.close, self.bb.lines.bot)

        self.entry_order = None
        self.stop_order = None
        self.limit_order = None
        self.close_order = None

        self.entry_bar = 0
        self.atr_entry = 0.0

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            if order == self.entry_order:
                self.entry_order = None
                self.entry_bar = len(self)
                entry_price = order.executed.price
                atr_val = self.atr_entry

                if self.position.size > 0:
                    stop_price = entry_price - 1.0 * atr_val
                    limit_price = entry_price + 2.0 * atr_val
                    self.stop_order = self.sell(
                        exectype=bt.Order.Stop, price=stop_price
                    )
                    self.limit_order = self.sell(
                        exectype=bt.Order.Limit,
                        price=limit_price,
                        oco=self.stop_order,
                    )
                elif self.position.size < 0:
                    stop_price = entry_price + 1.0 * atr_val
                    limit_price = entry_price - 2.0 * atr_val
                    self.stop_order = self.buy(
                        exectype=bt.Order.Stop, price=stop_price
                    )
                    self.limit_order = self.buy(
                        exectype=bt.Order.Limit,
                        price=limit_price,
                        oco=self.stop_order,
                    )

            elif order in [self.stop_order, self.limit_order]:
                self.stop_order = None
                self.limit_order = None

            elif order == self.close_order:
                self.close_order = None

        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            if order == self.entry_order:
                self.entry_order = None
            if order == self.stop_order:
                self.stop_order = None
            if order == self.limit_order:
                self.limit_order = None
            if order == self.close_order:
                self.close_order = None

    def next(self):
        if self.position:
            if self.entry_order is None and self.close_order is None:
                bars_held = len(self) - self.entry_bar
                if bars_held >= 3:
                    if self.stop_order:
                        self.cancel(self.stop_order)
                    if self.limit_order:
                        self.cancel(self.limit_order)
                    self.close_order = self.close()
            return

        if self.entry_order is not None:
            return

        long_trend = self.sma20[0] > self.sma50[0]
        short_trend = self.sma20[0] < self.sma50[0]

        volatility_expansion = self.bb_width[0] > (self.sma_bb_width[0] * 1.3)
        volume_filter = self.data.volume[0] > (1.5 * self.sma_vol[0])

        long_signal = self.co_long[0] > 0
        short_signal = self.co_short[0] < 0

        if long_trend and volatility_expansion and volume_filter and long_signal:
            self.atr_entry = self.atr[0]
            self.entry_order = self.buy()
        elif short_trend and volatility_expansion and volume_filter and short_signal:
            self.atr_entry = self.atr[0]
            self.entry_order = self.sell()
