import backtrader as bt
import math


class BuyAndHoldStrategy(bt.Strategy):
    class PerShareCommission(bt.CommInfoBase):
        params = (
            ('stocklike', True),
            ('commtype', bt.CommInfoBase.COMM_FIXED),
        )

        def _getcommission(self, size, price, pseudoexec):
            return max(abs(size) * 0.01, 1.0)

    def __init__(self):
        self.broker.addcommissioninfo(self.PerShareCommission())
        self.order = None
        self.entered = False
        self.exited = False
        self.peak_value = self.broker.getvalue()

    def next(self):
        value = self.broker.getvalue()
        if value > self.peak_value:
            self.peak_value = value

        if not self.entered and not self.exited and self.order is None:
            cash = self.broker.getcash()
            size = math.floor((cash - 1.0) / self.data.close[0])
            if size > 0:
                self.order = self.buy(size=size)
                self.entered = True
            return

        if (self.position.size > 0 and not self.exited and
                self.order is None and
                value <= self.peak_value * 0.70):
            self.order = self.close()
            self.exited = True

    def notify_order(self, order):
        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            self.order = None
