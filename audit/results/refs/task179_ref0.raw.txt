import backtrader as bt


class BuyingSpikeMomentum(bt.Strategy):
    def __init__(self):
        self.sma_price = bt.indicators.SMA(self.data.close, period=20)
        self.sma_vol5 = bt.indicators.SMA(self.data.volume, period=5)
        self.sma_vol20 = bt.indicators.SMA(self.data.volume, period=20)
        self.stoch = bt.indicators.Stochastic(
            self.data, period=14, period_dfast=3, period_dslow=3
        )
        self.crossover = bt.indicators.CrossOver(self.stoch.percK, self.stoch.percD)
        self.atr = bt.indicators.ATR(self.data, period=14)

        self.order = None
        self.stop_order = None
        self.entry_price = None
        self.entry_bar = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order == self.stop_order:
            if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
                self.stop_order = None
            return

        if order.status == order.Completed:
            if order.isbuy():
                self.entry_price = order.executed.price
                self.entry_bar = len(self)
                stop_price = self.entry_price * 0.98
                self.stop_order = self.sell(
                    exectype=bt.Order.Stop, price=stop_price, size=order.executed.size
                )
        self.order = None

    def next(self):
        if self.order:
            return

        if not self.position:
            price_trend = self.data.close[0] > self.sma_price[0]
            volume_spike = self.sma_vol5[0] >= 3 * self.sma_vol20[0]
            stoch_confirm = (self.stoch.percK[0] > 80) and (
                (self.crossover[0] > 0) or (self.data.close[0] > self.data.close[-1])
            )
            atr_filter = (self.atr[0] / self.data.close[0]) > 0.015
            liquidity_filter = (self.sma_vol20[0] * self.sma_price[0]) >= 10_000_000

            if price_trend and volume_spike and stoch_confirm and atr_filter and liquidity_filter:
                size = int((self.broker.getvalue() * 0.02) / self.data.close[0])
                if size > 0:
                    self.order = self.buy(size=size)
        else:
            bars_held = len(self) - self.entry_bar
            exit_signal = (self.data.close[0] < self.sma_price[0]) or (bars_held >= 10)
            if exit_signal:
                if self.stop_order is not None:
                    self.cancel(self.stop_order)
                    self.stop_order = None
                self.order = self.close()
