import math
import statistics

import backtrader as bt


class LogisticRegressionRsiMomentumStrategy(bt.Strategy):
    params = (
        ('rsi_period', 14),
        ('mom_period', 12),
        ('train_frac', 0.8),
        ('entry_threshold', 0.6),
        ('lr', 0.1),
        ('epochs', 500),
    )

    def __init__(self):
        self.rsi = bt.indicators.RSI(self.data.close, period=self.p.rsi_period)
        self.mom = bt.indicators.Momentum(self.data.close, period=self.p.mom_period)
        self.order = None
        self.train_x = []
        self.train_y = []
        self.feat_stats = None
        self.weights = None
        self.split_idx = None
        self.trained = False

    def notify_order(self, order):
        if order.status in (order.Completed, order.Canceled, order.Margin, order.Rejected):
            self.order = None

    def _sigmoid(self, z):
        if z < -60:
            return 0.0
        if z > 60:
            return 1.0
        return 1.0 / (1.0 + math.exp(-z))

    def _train(self):
        n = len(self.train_y)
        if n == 0:
            return
        rsis = [x[0] for x in self.train_x]
        moms = [x[1] for x in self.train_x]
        rsi_mean = statistics.mean(rsis)
        rsi_std = statistics.pstdev(rsis) or 1.0
        mom_mean = statistics.mean(moms)
        mom_std = statistics.pstdev(moms) or 1.0
        self.feat_stats = (rsi_mean, rsi_std, mom_mean, mom_std)

        w0 = w1 = w2 = 0.0
        for _ in range(self.p.epochs):
            g0 = g1 = g2 = 0.0
            for (r, m), y in zip(self.train_x, self.train_y):
                rs = (r - rsi_mean) / rsi_std
                ms = (m - mom_mean) / mom_std
                p = self._sigmoid(w0 + w1 * rs + w2 * ms)
                err = p - y
                g0 += err
                g1 += err * rs
                g2 += err * ms
            w0 -= self.p.lr * g0 / n
            w1 -= self.p.lr * g1 / n
            w2 -= self.p.lr * g2 / n
        self.weights = (w0, w1, w2)
        self.trained = True

    def _predict_proba(self, r, m):
        rsi_mean, rsi_std, mom_mean, mom_std = self.feat_stats
        rs = (r - rsi_mean) / rsi_std
        ms = (m - mom_mean) / mom_std
        w0, w1, w2 = self.weights
        return self._sigmoid(w0 + w1 * rs + w2 * ms)

    def next(self):
        if self.split_idx is None:
            self.split_idx = int(self.data.buflen() * self.p.train_frac)

        # RSI minperiod (15) plus one extra bar so the t-1 lagged value is also valid
        min_len = self.p.rsi_period + 2
        if len(self) < min_len:
            return

        if len(self) <= self.split_idx:
            self.train_x.append((self.rsi[-1], self.mom[-1]))
            self.train_y.append(1 if self.data.close[0] >= self.data.open[0] else 0)
            return

        if not self.trained:
            self._train()
            if not self.trained:
                return

        if self.order:
            return

        proba_up = self._predict_proba(self.rsi[-1], self.mom[-1])
        prediction = 1 if proba_up >= 0.5 else 0

        if not self.position:
            if proba_up > self.p.entry_threshold:
                self.order = self.buy(size=10)
        else:
            if prediction == 0:
                self.order = self.close()
