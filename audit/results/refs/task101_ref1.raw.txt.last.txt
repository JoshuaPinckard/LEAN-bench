import backtrader as bt


class TrailingStopRunners(bt.Strategy):
    params = (
        ('atr_period', 14),
        ('sma_period', 9),
        ('macd_fast', 12),
        ('macd_slow', 9),
        ('macd_signal', 26),
    )

    def __init__(self):
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.sma = bt.indicators.SMA(self.data.close, period=self.p.sma_period)
        self.macd = bt.indicators.MACD(
            self.data.close,
            period_me1=self.p.macd_fast,
            period_me2=self.p.macd_slow,
            period_signal=self.p.macd_signal,
        )
        self.histogram = self.macd.macd - self.macd.signal

        self.order = None
        self.trailing_stop = None

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            if order.status == order.Completed and order.isbuy():
                self.trailing_stop = order.executed.price - (2 * self.atr[0])

            if order.status == order.Completed and order.issell():
                self.trailing_stop = None

            self.order = None

    def next(self):
        if self.order:
            return

        histogram_cross_up = self.histogram[-1] < 0 and self.histogram[0] > 0
        macd_cross_up = (
            self.macd.macd[-1] < self.macd.signal[-1]
            and self.macd.macd[0] > self.macd.signal[0]
        )

        histogram_cross_down = self.histogram[-1] > 0 and self.histogram[0] < 0
        macd_cross_down = (
            self.macd.macd[-1] > self.macd.signal[-1]
            and self.macd.macd[0] < self.macd.signal[0]
        )

        if not self.position:
            if self.data.close[0] > self.sma[0] and (histogram_cross_up or macd_cross_up):
                self.order = self.buy()
            return

        if self.data.low[0] <= self.trailing_stop or histogram_cross_down or macd_cross_down:
            self.order = self.close()
            return

        self.trailing_stop = max(
            self.trailing_stop,
            self.data.high[0] - (2 * self.atr[0]),
        )
