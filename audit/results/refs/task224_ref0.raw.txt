            # next month's first bar arrives; the forced sale is placed here

            # rule 3's "next bar" without look-ahead.
            if self.lots_sold_month == 0 and self.lots_month > 0:
                if self.data.close[-1] >= self.p.strike and not self._bullish_impulse(-1):
                    size = self.lots_month * self.p.lot_size
                    if size > 0 and self.position.size >= size:
                        self.order = self.sell(size=size)
                        self.remaining_limit_usd -= self.lots_month * self.p.strike * self.p.lot_size
            self.current_month = dt.month
            self.lots_month = self._new_lots_month()
            self.lots_sold_month = 0
            if self.order:
                return
