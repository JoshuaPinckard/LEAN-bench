import math
import backtrader as bt


class PseudoProtectivePut(bt.Strategy):
    params = (
        ('base_price', 50.0),
        ('main_shares', 1000),
    )

    def __init__(self):
        self.atr = bt.indicators.ATR(self.data, period=14)
        self.order = None
        self.activated = False
        self.max_k_breached = 0
        self.current_term = 90
        self.entry_price = None
        self.entry_atr = None
        self.price_stop = None
        self.entry_bar = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            if order.issell():
                self.entry_price = order.executed.price
                self.entry_atr = self.atr[0]
                self.price_stop = self.entry_price + 1.0 * self.entry_atr
                self.entry_bar = len(self)
            self.order = None
        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        if self.order:
            return

        close = self.data.close[0]
        atr = self.atr[0]

        if close <= 0 or atr <= 0:
            return

        vol_filter = (atr / close) <= 0.04

        ratio = close / self.p.base_price
        if ratio >= 1.10:
            current_k = int(math.floor(round((ratio - 1.0) / 0.10, 6)))
        else:
            current_k = 0

        new_threshold_breached = current_k > self.max_k_breached

        if not self.activated:
            if close >= 3.0 * self.p.base_price and vol_filter:
                qty = int(math.floor(0.02 * (self.p.main_shares * close) / atr))
                if qty > 0:
                    self.activated = True
                    self.max_k_breached = max(self.max_k_breached, current_k)
                    self.current_term = 90
                    self.order = self.sell(size=qty)
                    return
        else:
            if new_threshold_breached and vol_filter:
                qty = int(math.floor(0.02 * (self.p.main_shares * close) / atr))
                if qty > 0:
                    self.max_k_breached = current_k
                    self.current_term = 180
                    if self.position:
                        self.close()
                    self.order = self.sell(size=qty)
                    return

        if self.position:
            if close >= self.price_stop:
                self.order = self.close()
                return
            if close <= self.entry_price * 0.85:
                self.order = self.close()
                return
            if self.entry_bar is not None and (len(self) - self.entry_bar) >= self.current_term:
                self.order = self.close()
                return
