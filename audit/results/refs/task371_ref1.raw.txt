import backtrader as bt


class IntradayTrendFollowingStrategy(bt.Strategy):
    params = (
        ("P_smooth", 20),
        ("P_atr", 14),
        ("atr_mult", 2.0),
    )

    def __init__(self):
        self.ema = bt.indicators.EMA(self.data.close, period=self.p.P_smooth)
        self.bbands = bt.indicators.BollingerBands(
            self.data.close, period=20, devfactor=2
        )
        self.atr = bt.indicators.ATR(self.data, period=self.p.P_atr)

        self.order = None
        self.stop_order = None
        self.pending_side = None

    def next(self):
        if self.order:
            return

        long_signal = (
            self.data.close[-1] < self.bbands.bot[-1]
            and self.data.close[0] > self.bbands.bot[0]
            and self.data.close[0] > self.ema[0]
        )
        short_signal = (
            self.data.close[-1] > self.bbands.top[-1]
            and self.data.close[0] < self.bbands.top[0]
            and self.data.close[0] < self.ema[0]
        )

        if not self.position:
            if long_signal:
                self.pending_side = "long"
                self.order = self.buy()
            elif short_signal:
                self.pending_side = "short"
                self.order = self.sell()
        elif self.position.size > 0 and short_signal:
            if self.stop_order:
                self.cancel(self.stop_order)
            self.order = self.close()
        elif self.position.size < 0 and long_signal:
            if self.stop_order:
                self.cancel(self.stop_order)
            self.order = self.close()

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return

        if order is self.stop_order:
            if order.status in (
                order.Completed,
                order.Canceled,
                order.Margin,
                order.Rejected,
            ):
                self.stop_order = None
            return

        if order is self.order:
            if order.status == order.Completed:
                if self.pending_side == "long":
                    stop_price = order.executed.price - (
                        self.atr[0] * self.p.atr_mult
                    )
                    self.stop_order = self.sell(
                        exectype=bt.Order.Stop, price=stop_price
                    )
                elif self.pending_side == "short":
                    stop_price = order.executed.price + (
                        self.atr[0] * self.p.atr_mult
                    )
                    self.stop_order = self.buy(
                        exectype=bt.Order.Stop, price=stop_price
                    )
                self.pending_side = None

            if order.status in (
                order.Completed,
                order.Canceled,
                order.Margin,
                order.Rejected,
            ):
                self.order = None
                if order.status != order.Completed:
                    self.pending_side = None
