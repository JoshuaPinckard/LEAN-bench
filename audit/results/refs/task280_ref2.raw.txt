import backtrader as bt


class BearishWeeklyShortStrategy(bt.Strategy):
    params = (
        ('risk_per_trade', 0.01),
        ('k', 2.0),
        ('atr_period', 14),
        ('hold_days', 20),
        ('interval', 7),
    )

    def __init__(self):
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.entry_order = None
        self.tp_order = None
        self.sl_order = None
        self.close_order = None
        self.entry_bar = None
        self.entry_price = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status == order.Completed:
            if order == self.entry_order:
                self.entry_price = order.executed.price
                self.entry_bar = len(self)
                size = abs(order.executed.size)
                atr_val = self.atr[0]

                tp_price = self.entry_price - (0.5 * self.p.k * atr_val)
                sl_price = self.entry_price + (self.p.k * atr_val)

                self.tp_order = self.buy(
                    size=size,
                    price=tp_price,
                    exectype=bt.Order.Limit
                )
                self.sl_order = self.buy(
                    size=size,
                    price=sl_price,
                    exectype=bt.Order.Stop,
                    oco=self.tp_order
                )
                self.entry_order = None

            elif order in [self.tp_order, self.sl_order]:
                self.tp_order = None
                self.sl_order = None

            elif order == self.close_order:
                self.close_order = None

        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            if order == self.entry_order:
                self.entry_order = None
            if order == self.tp_order:
                self.tp_order = None
            if order == self.sl_order:
                self.sl_order = None
            if order == self.close_order:
                self.close_order = None

    def next(self):
        if self.position:
            if self.entry_bar is not None and (len(self) - self.entry_bar >= self.p.hold_days):
                if self.tp_order:
                    self.cancel(self.tp_order)
                if self.sl_order:
                    self.cancel(self.sl_order)
                self.close_order = self.close()
            return

        has_pending_order = (
            self.entry_order is not None or
            self.close_order is not None
        )

        if not has_pending_order and (len(self) % self.p.interval == 0):
            atr_val = self.atr[0]
            if atr_val > 0:
                capital = self.broker.getvalue()
                risk_amt = capital * self.p.risk_per_trade
                risk_per_share = self.p.k * atr_val
                if risk_per_share > 0:
                    size = int(risk_amt / risk_per_share)
                    if size > 0:
                        self.entry_order = self.sell(size=size)
