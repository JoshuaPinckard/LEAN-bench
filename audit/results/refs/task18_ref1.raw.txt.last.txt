import backtrader as bt
import math


class ZScoreMeanReversionStrategy(bt.Strategy):
    params = (
        ("period", 120),
        ("stop_loss_pct", 0.05),
        ("trailing_stop_pct", 0.01),
    )

    def __init__(self):
        self.addminperiod(self.p.period)
        self.order = None
        self.fixed_stop = None
        self.trailing_stop = None
        self.entry_price = None

    def _zscore(self):
        values = list(self.data.close.get(size=self.p.period))
        n = self.p.period
        x_mean = (n - 1) / 2.0
        y_mean = sum(values) / n
        denominator = sum((x - x_mean) ** 2 for x in range(n))
        slope = sum(
            (x - x_mean) * (value - y_mean)
            for x, value in enumerate(values)
        ) / denominator
        moving_average = y_mean + slope * ((n - 1) - x_mean)
        standard_deviation = math.sqrt(
            sum((value - y_mean) ** 2 for value in values) / n
        )

        if standard_deviation == 0:
            return 0.0

        return (self.data.close[0] - moving_average) / standard_deviation

    def _cancel_protective_orders(self):
        if self.fixed_stop is not None:
            self.cancel(self.fixed_stop)
        if self.trailing_stop is not None:
            self.cancel(self.trailing_stop)

    def _clear_trade_state(self):
        self.fixed_stop = None
        self.trailing_stop = None
        self.entry_price = None

    def next(self):
        if self.order is not None:
            return

        zscore = self._zscore()

        if not self.position:
            if zscore > 0:
                self.order = self.buy()
            elif zscore < 0:
                self.order = self.sell()
            return

        if self.position.size > 0 and zscore <= -1.5:
            self._cancel_protective_orders()
            self.order = self.close()
        elif self.position.size < 0 and zscore >= 1.5:
            self._cancel_protective_orders()
            self.order = self.close()

    def notify_order(self, order):
        if order.status in (
            order.Submitted,
            order.Accepted,
        ):
            return

        if order.status == order.Completed:
            if order is self.order:
                self.order = None

                if order.isbuy() and self.position.size > 0:
                    self.entry_price = order.executed.price
                    self.fixed_stop = self.sell(
                        exectype=bt.Order.Stop,
                        price=self.entry_price * (1.0 - self.p.stop_loss_pct),
                    )
                    self.trailing_stop = self.sell(
                        exectype=bt.Order.StopTrail,
                        trailpercent=self.p.trailing_stop_pct,
                        oco=self.fixed_stop,
                    )
                elif order.issell() and self.position.size < 0:
                    self.entry_price = order.executed.price
                    self.fixed_stop = self.buy(
                        exectype=bt.Order.Stop,
                        price=self.entry_price * (1.0 + self.p.stop_loss_pct),
                    )
                    self.trailing_stop = self.buy(
                        exectype=bt.Order.StopTrail,
                        trailpercent=self.p.trailing_stop_pct,
                        oco=self.fixed_stop,
                    )
                elif not self.position:
                    self._clear_trade_state()

            elif order is self.fixed_stop or order is self.trailing_stop:
                self._clear_trade_state()

        elif order.status in (
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            if order is self.order:
                self.order = None

            if order is self.fixed_stop:
                self.fixed_stop = None
            elif order is self.trailing_stop:
                self.trailing_stop = None
