import backtrader as bt
import math


class QuasiCreditSpreadsForSmallAccounts(bt.Strategy):
    params = (("risk_pct", 0.01),)

    def __init__(self):
        self.sma50 = bt.indicators.SMA(self.data.close, period=50)
        self.bb = bt.indicators.BollingerBands(self.data.close, period=20, devfactor=2)
        self.sma20 = bt.indicators.SMA(self.data.close, period=20)
        self.atr = bt.indicators.ATR(self.data, period=14)
        self.rsi = bt.indicators.RSI(self.data.close, period=14)

        self.bb_width = (self.bb.top - self.bb.bot) / self.bb.mid
        self.bb_width_sma = bt.indicators.SMA(self.bb_width, period=100)
        self.volume_sma = bt.indicators.SMA(self.data.volume, period=20)

        self.order = None
        self.entry_context = None
        self.entry_price = None
        self.target_price = None
        self.stop_price = None
        self.entry_bar = None
        self.entry_sign = None

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            if order is self.order:
                self.order = None

            if order.status == order.Completed and self.entry_context is not None:
                self.entry_sign, sma20_entry = self.entry_context
                self.entry_price = order.executed.price
                entry_atr = float(self.atr[0])
                self.target_price = self.entry_price - self.entry_sign * (
                    0.7 * (self.entry_price - sma20_entry)
                )
                self.stop_price = self.entry_price - self.entry_sign * (1.2 * entry_atr)
                self.entry_bar = len(self)
                self.entry_context = None
            elif order.status != order.Completed and self.entry_context is not None:
                self.entry_context = None

    def next(self):
        if self.order:
            return

        if self.position:
            sign = 1 if self.position.size > 0 else -1
            close = float(self.data.close[0])
            atr = float(self.atr[0])

            target_hit = (
                (sign > 0 and close >= self.target_price)
                or (sign < 0 and close <= self.target_price)
            )
            sma_crossed = (
                (sign > 0 and close >= float(self.sma20[0]))
                or (sign < 0 and close <= float(self.sma20[0]))
            )
            stop_hit = (
                (sign > 0 and close <= self.stop_price)
                or (sign < 0 and close >= self.stop_price)
            )
            beyond_band = (
                (sign > 0 and close <= float(self.bb.bot[0]) - 0.5 * atr)
                or (sign < 0 and close >= float(self.bb.top[0]) + 0.5 * atr)
            )
            time_exit = len(self) - self.entry_bar >= 6

            if target_hit or sma_crossed or stop_hit or beyond_band or time_exit:
                self.order = self.close()
            return

        filters_met = (
            float(self.bb_width[0]) < float(self.bb_width_sma[0])
            and float(self.data.volume[0]) <= 1.5 * float(self.volume_sma[0])
            and abs(float(self.data.open[0]) - float(self.data.close[-1]))
            < 1.5 * float(self.atr[0])
        )

        atr = float(self.atr[0])
        size = math.floor(
            (self.broker.getvalue() * self.p.risk_pct) / (1.2 * atr)
        )

        if size <= 0 or not filters_met:
            return

        if (
            float(self.data.close[0]) <= float(self.bb.bot[0])
            and float(self.data.close[0]) > float(self.sma50[0])
            and float(self.rsi[0]) < 45
        ):
            self.entry_context = (1, float(self.sma20[0]))
            self.order = self.buy(size=size)

        elif (
            float(self.data.close[0]) >= float(self.bb.top[0])
            and float(self.data.close[0]) < float(self.sma50[0])
            and float(self.rsi[0]) > 55
        ):
            self.entry_context = (-1, float(self.sma20[0]))
            self.order = self.sell(size=size)
