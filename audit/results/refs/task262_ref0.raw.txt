import math
import backtrader as bt


class MarketMakingMeanReversion(bt.Strategy):
    def __init__(self):
        self.bb = bt.indicators.BollingerBands(self.data.close, period=20, devfactor=2)
        self.atr = bt.indicators.ATR(self.data, period=14)
        self.vol_sma = bt.indicators.SMA(self.data.volume, period=20)

        self.order_buy_entry = None
        self.order_sell_entry = None
        self.order_tp = None
        self.order_close = None

        self.tp_placed_bar = None
        self.timeout_pending = False

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return

        if order.status == order.Completed:
            if order is self.order_buy_entry:
                entry_price = order.executed.price
                entry_size = order.executed.size
                self.order_buy_entry = None
                if self.order_sell_entry is not None and self.order_sell_entry.alive():
                    self.cancel(self.order_sell_entry)
                tp_price = entry_price + 0.30 * self.atr[0]
                self.order_tp = self.sell(exectype=bt.Order.Limit, price=tp_price, size=entry_size)
                self.tp_placed_bar = len(self.data)
            elif order is self.order_sell_entry:
                entry_price = order.executed.price
                entry_size = abs(order.executed.size)
                self.order_sell_entry = None
                if self.order_buy_entry is not None and self.order_buy_entry.alive():
                    self.cancel(self.order_buy_entry)
                tp_price = entry_price - 0.30 * self.atr[0]
                self.order_tp = self.buy(exectype=bt.Order.Limit, price=tp_price, size=entry_size)
                self.tp_placed_bar = len(self.data)
            elif order is self.order_tp:
                self.order_tp = None
                self.tp_placed_bar = None
            elif order is self.order_close:
                self.order_close = None
                self.order_tp = None
                self.tp_placed_bar = None
        elif order.status in (order.Canceled, order.Margin, order.Rejected):
            if order is self.order_buy_entry:
                self.order_buy_entry = None
            elif order is self.order_sell_entry:
                self.order_sell_entry = None
            elif order is self.order_tp:
                self.order_tp = None
                self.tp_placed_bar = None
            elif order is self.order_close:
                self.order_close = None

    def next(self):
        if self.timeout_pending:
            self.timeout_pending = False
            self.order_close = self.close()
            return

        if self.position:
            if self.position.size > 0 and self.data.close[0] < self.bb.bot[0]:
                if self.order_tp is not None and self.order_tp.alive():
                    self.cancel(self.order_tp)
                self.order_close = self.close()
                return
            if self.position.size < 0 and self.data.close[0] > self.bb.top[0]:
                if self.order_tp is not None and self.order_tp.alive():
                    self.cancel(self.order_tp)
                self.order_close = self.close()
                return

        if self.order_tp is not None and self.order_tp.alive():
            if len(self.data) - self.tp_placed_bar >= 1:
                self.cancel(self.order_tp)
                self.timeout_pending = True
            return

        if self.position:
            return

        if self.order_buy_entry is not None or self.order_sell_entry is not None:
            return

        bar_time = self.data.datetime.time(0)
        minutes = bar_time.hour * 60 + bar_time.minute
        if minutes < 10 * 60 or minutes > 15 * 60 + 30:
            return

        bb_mid = self.bb.mid[0]
        bb_top = self.bb.top[0]
        bb_bot = self.bb.bot[0]
        bb_width = (bb_top - bb_bot) / bb_mid
        atr_val = self.atr[0]
        close = self.data.close[0]
        rel_vol = atr_val / close

        if bb_width > 0.003 or rel_vol > 0.004 or self.data.volume[0] < self.vol_sma[0]:
            return

        capital = self.broker.getvalue()
        open_value = abs(self.position.size) * close
        for o in (self.order_buy_entry, self.order_sell_entry, self.order_tp):
            if o is not None and o.alive():
                open_value += abs(o.created.size) * o.created.price
        if open_value >= 0.05 * capital:
            return

        size = math.floor(0.01 * capital / close)
        if size <= 0:
            return

        buy_price = bb_mid - 0.15 * atr_val
        sell_price = bb_mid + 0.15 * atr_val

        self.order_buy_entry = self.buy(exectype=bt.Order.Limit, price=buy_price, size=size)
        self.order_sell_entry = self.sell(exectype=bt.Order.Limit, price=sell_price, size=size)
