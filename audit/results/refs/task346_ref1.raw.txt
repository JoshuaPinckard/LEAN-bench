import backtrader as bt


class IchimokuSmaStrategy(bt.Strategy):
    params = (("stake", 10),)

    def __init__(self):
        self.tenkan = bt.indicators.SMA(self.data.close, period=9)
        self.kijun = bt.indicators.SMA(self.data.close, period=26)
        self.span_b = bt.indicators.SMA(self.data.close, period=52)
        self.cross = bt.indicators.CrossOver(self.tenkan, self.kijun)
        self.order = None

    def notify_order(self, order):
        if order.status in (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            self.order = None

    def next(self):
        if self.order:
            return

        span_a = (self.tenkan[0] + self.kijun[0]) / 2.0
        cloud_top = max(span_a, self.span_b[0])
        cloud_bottom = min(span_a, self.span_b[0])

        if self.position.size > 0:
            if self.cross[0] < 0 or self.data.low[0] <= self.kijun[0]:
                self.order = self.close()

        elif self.position.size < 0:
            if self.cross[0] > 0 or self.data.high[0] >= self.kijun[0]:
                self.order = self.close()

        elif self.cross[0] > 0 and self.data.close[0] > cloud_top:
            self.order = self.buy(size=self.p.stake)

        elif self.cross[0] < 0 and self.data.close[0] < cloud_bottom:
            self.order = self.sell(size=self.p.stake)
