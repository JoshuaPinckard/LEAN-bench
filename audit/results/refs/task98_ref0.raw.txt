import backtrader as bt


class MeanReversionStrategy(bt.Strategy):
    params = (
        ('rsi_period', 14),
        ('sma_fast_period', 20),
        ('sma_slow_period', 50),
        ('bb_period', 20),
        ('bb_devfactor', 2.0),
        ('rsi_oversold', 30),
        ('rsi_overbought', 70),
        ('position_pct', 0.01),
        ('stop_devfactor', 1.5),
        ('vol_lookback', 20),
        ('vol_threshold', 1000000),
        ('min_bars', 200),
    )

    def __init__(self):
        self.rsi = bt.indicators.RSI(self.data.close, period=self.p.rsi_period)
        self.sma_fast = bt.indicators.SMA(self.data.close, period=self.p.sma_fast_period)
        self.sma_slow = bt.indicators.SMA(self.data.close, period=self.p.sma_slow_period)
        self.bb = bt.indicators.BollingerBands(
            self.data.close, period=self.p.bb_period, devfactor=self.p.bb_devfactor)
        self.stddev = bt.indicators.StandardDeviation(self.data.close, period=self.p.bb_period)
        self.vol_sma = bt.indicators.SMA(self.data.volume, period=self.p.vol_lookback)
        self.cross_mid = bt.indicators.CrossOver(self.data.close, self.bb.lines.mid)

        self.order = None
        self.stop_order = None
        self.limit_order = None

    def _cancel_bracket(self):
        if self.stop_order is not None and self.stop_order.alive():
            self.cancel(self.stop_order)
        if self.limit_order is not None and self.limit_order.alive():
            self.cancel(self.limit_order)
        self.stop_order = None
        self.limit_order = None

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted, order.Partial):
            return

        if order.status == order.Completed:
            if order == self.order:
                self.order = None
            elif order == self.stop_order:
                if self.limit_order is not None and self.limit_order.alive():
                    self.cancel(self.limit_order)
                self.stop_order = None
                self.limit_order = None
            elif order == self.limit_order:
                if self.stop_order is not None and self.stop_order.alive():
                    self.cancel(self.stop_order)
                self.stop_order = None
                self.limit_order = None
        elif order.status in (order.Canceled, order.Margin, order.Rejected):
            if order == self.order:
                self.order = None
            elif order == self.stop_order:
                self.stop_order = None
            elif order == self.limit_order:
                self.limit_order = None

    def next(self):
        if len(self) < self.p.min_bars:
            return

        if self.order:
            return

        price = self.data.close[0]

        if not self.position:
            volume_ok = self.vol_sma[0] > self.p.vol_threshold
            if not volume_ok:
                return

            long_signal = (
                price < self.bb.lines.bot[0]
                and self.rsi[0] < self.p.rsi_oversold
                and self.sma_fast[0] > self.sma_slow[0]
            )
            short_signal = (
                price > self.bb.lines.top[0]
                and self.rsi[0] > self.p.rsi_overbought
                and self.sma_fast[0] < self.sma_slow[0]
            )

            size = (self.broker.getvalue() * self.p.position_pct) / price

            if long_signal:
                stop_price = price - self.p.stop_devfactor * self.stddev[0]
                limit_price = self.bb.lines.mid[0]
                main, stop, limit = self.buy_bracket(
                    size=size,
                    exectype=bt.Order.Market,
                    stopprice=stop_price,
                    limitprice=limit_price,
                )
                self.order, self.stop_order, self.limit_order = main, stop, limit
            elif short_signal:
                stop_price = price + self.p.stop_devfactor * self.stddev[0]
                limit_price = self.bb.lines.mid[0]
                main, stop, limit = self.sell_bracket(
                    size=size,
                    exectype=bt.Order.Market,
                    stopprice=stop_price,
                    limitprice=limit_price,
                )
                self.order, self.stop_order, self.limit_order = main, stop, limit
        else:
            cross_exit = self.cross_mid[0] != 0

            if self.position.size > 0:
                rsi_exit = self.rsi[0] > self.p.rsi_overbought
                if cross_exit or rsi_exit:
                    self._cancel_bracket()
                    self.order = self.close()
            elif self.position.size < 0:
                rsi_exit = self.rsi[0] < self.p.rsi_oversold
                if cross_exit or rsi_exit:
                    self._cancel_bracket()
                    self.order = self.close()
