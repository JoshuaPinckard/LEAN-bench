import backtrader as bt


class DistributionReturnStrategy(bt.Strategy):
    """
    Reference implementation of a trading strategy that analyzes the distribution of future
    returns under aggregated 30-minute market conditions using 5-minute AAPL bar data.
    """

    def __init__(self):
        # Indicators on 5-minute data
        self.sma_fast = bt.ind.SMA(self.data.close, period=13)
        self.sma_slow = bt.ind.SMA(self.data.close, period=40)
        self.crossover = bt.ind.CrossOver(self.sma_fast, self.sma_slow)
        self.rsi = bt.ind.RSI(self.data.close, period=14)
        self.macd = bt.ind.MACD(
            self.data.close, period_me1=12, period_me2=26, period_signal=9
        )

        # Order tracking & position state
        self.order = None
        self.entry_price = None
        self.periods_held = 0

        # 30-minute aggregation state
        self.current_bucket = None
        self.window_crosses = 0
        self.window_rsi_sum = 0.0
        self.window_bar_count = 0

    def notify_order(self, order):
        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            if order.status == order.Completed:
                if order.isbuy():
                    self.entry_price = order.executed.price
                    self.periods_held = 0
                elif order.issell():
                    self.entry_price = None
                    self.periods_held = 0
            self.order = None

    def next(self):
        dt = self.data.datetime.datetime(0)
        bucket = (dt.date(), dt.hour, dt.minute // 30)

        # Start a new 30-minute aggregation window if bucket changes
        if bucket != self.current_bucket:
            self.current_bucket = bucket
            self.window_crosses = 0
            self.window_rsi_sum = 0.0
            self.window_bar_count = 0

        # Accumulate 5-minute bar metrics
        is_cross = 1 if self.crossover[0] > 0 else 0
        self.window_crosses += is_cross
        self.window_rsi_sum += self.rsi[0]
        self.window_bar_count += 1

        # A 30-minute candle completes at minute 25/55 or on the 6th 5m bar
        is_30m_end = (dt.minute in (25, 55)) or (self.window_bar_count == 6)

        # Manage open position risk and fixed holding period
        if self.position and self.order is None:
            # Stop-loss: price drops 1.5% below entry price
            if self.data.low[0] <= self.entry_price * 0.985:
                self.order = self.close()
            # Take-profit: price rises 3.0% above entry price
            elif self.data.high[0] >= self.entry_price * 1.03:
                self.order = self.close()
            elif is_30m_end:
                self.periods_held += 1
                # Close after 6 30-minute periods following opening
                if self.periods_held >= 6:
                    self.order = self.close()

        # Signal generation evaluated at the end of each 30-minute candle
        if is_30m_end:
            avg_rsi = self.window_rsi_sum / self.window_bar_count
            ma_crosses = self.window_crosses

            if not self.position and self.order is None:
                if ma_crosses == 1 and avg_rsi > 70.0:
                    cash = self.broker.getcash()
                    # Allocation: 100% of available capital (with 0.1% buffer for commission)
                    size = int((cash * 0.999) / self.data.close[0])
                    if size > 0:
                        self.order = self.buy(size=size)
