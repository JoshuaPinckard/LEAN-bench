import backtrader as bt


class ReturnDistributionStrategy(bt.Strategy):

    def __init__(self):
        self.sma_fast = bt.indicators.SMA(self.data.close, period=13)
        self.sma_slow = bt.indicators.SMA(self.data.close, period=40)
        self.rsi = bt.indicators.RSI(self.data.close, period=14)
        self.macd = bt.indicators.MACD(
            self.data.close, period1=12, period2=26, period9=9
        )
        self.order = None
        self.stop_price = None
        self.take_price = None
        self.periods_held = 0

    def notify_order(self, order):
        if order.status == order.Completed and order.isbuy():
            ep = order.executed.price
            self.stop_price = ep * (1.0 - 0.015)
            self.take_price = ep * (1.0 + 0.030)
            self.periods_held = 0
        if order.status in (
            order.Completed, order.Canceled, order.Margin, order.Rejected
        ):
            self.order = None

    def next(self):
        dt = self.data.datetime.datetime(0)
        # Last 5-min bar of each 30-min window has open-minute ≡ 25 (mod 30):
        # 09:55→55%30=25, 10:25→25%30=25, 10:55→55%30=25, …
        is_period_end = (dt.minute % 30 == 25)

        if self.position:
            close = self.data.close[0]
            if close <= self.stop_price:
                if self.order is None:
                    self.order = self.close()
                return
            if close >= self.take_price:
                if self.order is None:
                    self.order = self.close()
                return
            if is_period_end:
                self.periods_held += 1
                if self.periods_held >= 6 and self.order is None:
                    self.order = self.close()
            return

        if not is_period_end or self.order is not None:
            return

        # Count upward SMA(13)/SMA(40) crossovers within the current 30-min
        # window (bars at indices 0…-5).  Detecting a cross at bar -j requires
        # the preceding bar at -(j+1); for j=5 that is bar -6, the final bar of
        # the previous window, which is always available once next() is reached.
        ma_crosses = 0
        for j in range(6):
            if (
                self.sma_fast[-j] > self.sma_slow[-j]
                and self.sma_fast[-(j + 1)] <= self.sma_slow[-(j + 1)]
            ):
                ma_crosses += 1

        avg_rsi = sum(self.rsi[-j] for j in range(6)) / 6.0

        if ma_crosses == 1 and avg_rsi > 70:
            size = int(self.broker.getcash() / self.data.close[0])
            if size > 0:
                self.order = self.buy(size=size)
