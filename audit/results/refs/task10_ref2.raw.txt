```python
import datetime
import backtrader as bt


class SessionBreakoutStrategy(bt.Strategy):
    params = (
        ('session_start_hour', 9),
        ('session_start_minute', 30),
        ('range_minutes', 120),
        ('risk_perc', 0.01),
        ('tp_factor', 1.5),
        ('vol_filter_pct', 0.002),
    )

    def __init__(self):
        self.order = None
        self.current_date = None
        self.session_started = False
        self.session_start_dt = None
        self.zone_ready = False
        self.range_high = None
        self.range_low = None
        self.upper_bound = None
        self.lower_bound = None
        self.traded_this_session = False

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            if self.order and order.ref == self.order.ref:
                self.order = None

    def next(self):
        dt = self.data.datetime.datetime(0)
        current_date = dt.date()
        current_time = dt.time()

        if current_date != self.current_date:
            self.current_date = current_date
            self.session_started = False
            self.session_start_dt = None
            self.zone_ready = False
            self.range_high = None
            self.range_low = None
            self.upper_bound = None
            self.lower_bound = None
            self.traded_this_session = False

        session_start_time = datetime.time(
            self.p.session_start_hour, self.p.session_start_minute
        )

        if not self.session_started:
            if current_time >= session_start_time:
                self.session_started = True
                self.session_start_dt = dt
                self.range_high = self.data.high[0]
                self.range_low = self.data.low[0]
            else:
                return

        if self.session_started and not self.zone_ready:
            elapsed_minutes = (dt - self.session_start_dt).total_seconds() / 60.0
            if elapsed_minutes < self.p.range_minutes:
                self.range_high = max(self.range_high, self.data.high[0])
                self.range_low = min(self.range_low, self.data.low[0])
            else:
                self.zone_ready = True
                self.upper_bound = self.range_high
                self.lower_bound = self.range_low

        if self.zone_ready:
            if self.position.size != 0 or self.order is not None or self.traded_this_session:
                return

            range_width = self.upper_bound - self.lower_bound
            current_price = self.data.close[0]

            if range_width < (current_price * self.p.vol_filter_pct):
                return

            capital = self.broker.getvalue()
            risk_amount = capital * self.p.risk_perc

            if current_price > self.upper_bound:
                sl = self.lower_bound
                tp = self.upper_bound + (self.p.tp_factor * range_width)
                risk_per_share = current_price - sl
                if risk_per_share > 0:
                    size = int(risk_amount / risk_per_share)
                    if size > 0:
                        orders = self.buy_bracket(size=size, stopprice=sl, limitprice=tp)
                        self.order = orders[0]
                        self.traded_this_session = True

            elif current_price < self.lower_bound:
                sl = self.upper_bound
                tp = self.lower_bound - (self.p.tp_factor * range_width)
                risk_per_share = sl - current_price
                if risk_per_share > 0:
                    size = int(risk_amount / risk_per_share)
                    if size > 0:
                        orders = self.sell_bracket(size=size, stopprice=sl, limitprice=tp)
                        self.order = orders[0]
                        self.traded_this_session = True
```
