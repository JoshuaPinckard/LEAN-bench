import backtrader as bt


class VolatilityContractionHMA(bt.Strategy):
    params = (
        ("hma_period", 9),
        ("bb_period", 20),
        ("bb_dev", 2.0),
        ("bb_width_sma_period", 50),
        ("contraction_factor", 0.8),
        ("atr_period", 14),
        ("atr_mult", 2.0),
    )

    def __init__(self):
        self.hma = bt.indicators.HMA(self.data.close, period=self.p.hma_period)
        self.bb = bt.indicators.BollingerBands(
            self.data.close, period=self.p.bb_period, devfactor=self.p.bb_dev
        )
        self.bb_width = (self.bb.lines.top - self.bb.lines.bot) / self.bb.lines.mid
        self.bb_width_sma = bt.indicators.SMA(
            self.bb_width, period=self.p.bb_width_sma_period
        )
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.crossover = bt.indicators.CrossOver(self.data.close, self.hma)

        self.order = None
        self.stop_order = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            if order == self.order:
                self.order = None
                if self.position.size > 0:
                    stop_price = order.executed.price - self.atr[0] * self.p.atr_mult
                    self.stop_order = self.sell(
                        exectype=bt.Order.Stop, price=stop_price
                    )
                elif self.position.size < 0:
                    stop_price = order.executed.price + self.atr[0] * self.p.atr_mult
                    self.stop_order = self.buy(
                        exectype=bt.Order.Stop, price=stop_price
                    )
                else:
                    if self.stop_order:
                        self.cancel(self.stop_order)
                        self.stop_order = None
            elif order == self.stop_order:
                self.stop_order = None

        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            if order == self.order:
                self.order = None
            elif order == self.stop_order:
                self.stop_order = None

    def next(self):
        if self.order:
            return

        long_signal = (
            self.crossover[0] > 0
            and self.hma[0] > self.hma[-1]
            and self.bb_width[0] < self.bb_width_sma[0] * self.p.contraction_factor
        )

        short_signal = (
            self.crossover[0] < 0
            and self.hma[0] < self.hma[-1]
            and self.bb_width[0] < self.bb_width_sma[0] * self.p.contraction_factor
        )

        if not self.position:
            if long_signal:
                self.order = self.buy()
            elif short_signal:
                self.order = self.sell()
        else:
            if self.position.size > 0 and short_signal:
                if self.stop_order:
                    self.cancel(self.stop_order)
                    self.stop_order = None
                self.order = self.close()
            elif self.position.size < 0 and long_signal:
                if self.stop_order:
                    self.cancel(self.stop_order)
                    self.stop_order = None
                self.order = self.close()
