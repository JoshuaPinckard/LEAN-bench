import backtrader as bt


class ParabolicSARTrendStrategy(bt.Strategy):
    params = (
        ('af', 0.02),
        ('afmax', 0.2),
    )

    def __init__(self):
        self.sar = bt.indicators.ParabolicSAR(
            self.data,
            af=self.p.af,
            afmax=self.p.afmax,
        )
        self.cross = bt.indicators.CrossOver(self.data.close, self.sar)
        self.orders = []

    def notify_order(self, order):
        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            if order in self.orders:
                self.orders.remove(order)

    def _submit(self, order):
        if order is not None:
            self.orders.append(order)

    def next(self):
        if self.orders:
            return

        if self.cross > 0:
            if self.position.size < 0:
                self._submit(self.close())
            self._submit(self.buy())

        elif self.cross < 0:
            if self.position.size > 0:
                self._submit(self.close())
            self._submit(self.sell())
