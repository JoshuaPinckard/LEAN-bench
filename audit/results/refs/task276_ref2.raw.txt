import backtrader as bt


class SwingMomentumStrategy(bt.Strategy):
    params = dict(
        sma200_period=200,
        ema_fast_period=20,
        ema_slow_period=50,
        rsi_period=14,
        vol_sma_period=20,
        atr_period=14,
        tp_pct=0.18,
        init_stop_pct=0.08,
        atr_init_mult=2.0,
        atr_trail_mult=3.0,
        initial_capital_pct=0.60,
        addon_capital_pct=0.20,
        pullback_min=0.10,
        pullback_max=0.15,
    )

    def __init__(self):
        self.sma200 = bt.ind.SMA(self.data.close, period=self.p.sma200_period)
        self.ema20 = bt.ind.EMA(self.data.close, period=self.p.ema_fast_period)
        self.ema50 = bt.ind.EMA(self.data.close, period=self.p.ema_slow_period)
        self.rsi = bt.ind.RSI(self.data.close, period=self.p.rsi_period)
        self.vol_sma = bt.ind.SMA(
            self.data.volume, period=self.p.vol_sma_period
        )
        self.atr = bt.ind.ATR(self.data, period=self.p.atr_period)

        self.ema_cross_up = bt.ind.CrossUp(self.ema20, self.ema50)
        self.rsi_cross_up = bt.ind.CrossUp(self.rsi, 50.0)
        self.ema_cross_down = bt.ind.CrossDown(self.ema20, self.ema50)
        self.rsi_cross_down = bt.ind.CrossDown(self.rsi, 50.0)

        self.order = None
        self.initial_stop = None
        self.highest_close = None
        self.added_on = False

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            if self.position.size > 0:
                if self.initial_stop is None:
                    entry_price = order.executed.price
                    atr_val = self.atr[0]
                    stop_dist = max(
                        self.p.init_stop_pct * entry_price,
                        self.p.atr_init_mult * atr_val,
                    )
                    self.initial_stop = entry_price - stop_dist
                    self.highest_close = self.data.close[0]
                    self.added_on = False
            else:
                self.initial_stop = None
                self.highest_close = None
                self.added_on = False

        self.order = None

    def next(self):
        if self.order:
            return

        if not self.position:
            trend_ok = self.data.close[0] > self.sma200[0]
            entry_signal = (
                self.ema_cross_up[0] > 0 or self.rsi_cross_up[0] > 0
            )
            vol_ok = self.data.volume[0] > self.vol_sma[0]

            if trend_ok and entry_signal and vol_ok:
                capital = self.broker.getvalue()
                size = int(
                    (capital * self.p.initial_capital_pct) / self.data.close[0]
                )
                if size > 0:
                    self.order = self.buy(size=size)
        else:
            if (
                self.highest_close is None
                or self.data.close[0] > self.highest_close
            ):
                self.highest_close = self.data.close[0]

            trailing_stop = max(
                self.initial_stop,
                self.highest_close - self.p.atr_trail_mult * self.atr[0],
            )
            take_profit = self.position.price * (1.0 + self.p.tp_pct)

            tp_hit = self.data.high[0] >= take_profit
            stop_hit = self.data.low[0] <= trailing_stop
            reversal_hit = (
                self.ema_cross_down[0] > 0 or self.rsi_cross_down[0] > 0
            )

            if tp_hit or stop_hit or reversal_hit:
                self.order = self.close()
                return

            if not self.added_on:
                pullback = (
                    self.highest_close - self.data.close[0]
                ) / self.highest_close
                pullback_ok = (
                    self.p.pullback_min <= pullback <= self.p.pullback_max
                )
                trend_ok = (
                    self.data.close[0] > self.sma200[0]
                    and self.ema20[0] > self.ema50[0]
                )

                if pullback_ok and trend_ok:
                    capital = self.broker.getvalue()
                    addon_size = int(
                        (capital * self.p.addon_capital_pct)
                        / self.data.close[0]
                    )
                    if addon_size > 0:
                        self.order = self.buy(size=addon_size)
                        self.added_on = True
