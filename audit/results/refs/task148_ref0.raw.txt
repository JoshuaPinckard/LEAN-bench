import backtrader as bt


class ConvertibleBondDropStrategy(bt.Strategy):
    params = dict(
        sma_period=50,
        rsi_period=14,
        active_month=10,
        active_day=1,
        prev_close_threshold=200.0,
        rsi_threshold=50.0,
        take_profit=160.0,
        stop_loss_pct=0.05,
        max_hold_days=5,
        capital_pct=0.02,
        min_entry_price=160.0,
    )

    def __init__(self):
        self.sma = bt.indicators.SMA(self.data.close, period=self.p.sma_period)
        self.rsi = bt.indicators.RSI(self.data.close, period=self.p.rsi_period)
        self.order = None
        self.entry_bar = None
        self.entry_price = None
        self.stop_price = None

    def notify_order(self, order):
        if order.status == order.Completed:
            if order.issell():
                self.entry_price = order.executed.price
                self.stop_price = self.entry_price * (1 + self.p.stop_loss_pct)
                self.entry_bar = len(self)
            elif order.isbuy():
                self.entry_price = None
                self.stop_price = None
                self.entry_bar = None

        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        dt = self.data.datetime.date(0)
        is_active = (dt.month, dt.day) >= (self.p.active_month, self.p.active_day)

        if self.position:
            if self.data.high[0] >= self.stop_price:
                self.close()
                return
            if self.data.close[0] <= self.p.take_profit:
                self.close()
                return
            if self.data.close[0] > self.sma[0]:
                self.close()
                return
            if (len(self) - self.entry_bar) >= self.p.max_hold_days:
                self.close()
                return
            return

        if self.order:
            return

        if not is_active:
            return

        if len(self.data) < 2:
            return

        prev_close = self.data.close[-1]
        prev_rsi = self.rsi[-1]
        cur_open = self.data.open[0]

        if (prev_close > self.p.prev_close_threshold
                and prev_rsi > self.p.rsi_threshold
                and cur_open < prev_close):

            if cur_open < self.p.min_entry_price:
                return

            capital = self.broker.getvalue() * self.p.capital_pct
            size = capital / self.data.close[0]
            if size <= 0:
                return

            self.order = self.sell(size=size)
