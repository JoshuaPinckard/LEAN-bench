import backtrader as bt


class IntradayMeanReversion(bt.Strategy):
    params = (
        ("atr_period", 14),
        ("ma_period", 20),
        ("ma_type", "sma"),  # 'sma' or 'ema'
        ("stop_loss_pct", 0.02),
    )

    def __init__(self):
        if str(self.params.ma_type).lower() == "ema":
            self.ma = bt.indicators.EMA(self.data.close, period=self.params.ma_period)
        else:
            self.ma = bt.indicators.SMA(self.data.close, period=self.params.ma_period)

        self.atr = bt.indicators.ATR(self.data, period=self.params.atr_period)
        self.crossover = bt.indicators.CrossOver(self.data.close, self.ma)

        self.order = None
        self.day_open = None
        self.last_date = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        current_date = self.data.datetime.date(0)
        if self.last_date != current_date:
            self.last_date = current_date
            self.day_open = self.data.open[0]

        if self.order:
            return

        if not self.position:
            if self.day_open is not None:
                upper_bound = self.day_open + 2.0 * self.atr[0]
                lower_bound = self.day_open - 2.0 * self.atr[0]

                if self.data.close[0] > upper_bound:
                    self.order = self.sell()
                elif self.data.close[0] < lower_bound:
                    self.order = self.buy()
        else:
            if self.position.size > 0:
                stop_price = self.position.price * (1.0 - self.params.stop_loss_pct)
                if self.data.close[0] <= stop_price or self.crossover[0] > 0:
                    self.order = self.close()
            elif self.position.size < 0:
                stop_price = self.position.price * (1.0 + self.params.stop_loss_pct)
                if self.data.close[0] >= stop_price or self.crossover[0] < 0:
                    self.order = self.close()
