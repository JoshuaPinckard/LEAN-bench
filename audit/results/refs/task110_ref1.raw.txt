import backtrader as bt


class RSIReversalStrategy(bt.Strategy):
    def __init__(self):
        self.order = None
        self.rsi = bt.indicators.RSI(self.data.close, period=14)
        self.rsi_cross_up_30 = bt.indicators.CrossUp(self.rsi, 30.0)
        self.rsi_cross_down_70 = bt.indicators.CrossDown(self.rsi, 70.0)

    def next(self):
        if self.order:
            return

        if not self.position and self.rsi_cross_up_30[0]:
            self.order = self.buy()
        elif self.position and self.rsi_cross_down_70[0]:
            self.order = self.close()

    def notify_order(self, order):
        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            self.order = None
