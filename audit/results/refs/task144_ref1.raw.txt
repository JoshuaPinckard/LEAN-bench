import backtrader as bt


class RangeControlStrategy(bt.Strategy):
    params = (
        ('consolidation_period', 10),
        ('volume_sma_period', 20),
        ('range_percent', 0.02),
        ('volume_multiple', 3.0),
        ('take_profit', 0.05),
        ('stop_loss', 0.02),
        ('capital_fraction', 0.01),
    )

    def __init__(self):
        self.volume_sma = bt.indicators.SMA(
            self.data.volume, period=self.p.volume_sma_period
        )
        self.order = None
        self.order_role = None
        self.pending_upper = None
        self.pending_lower = None
        self.entry_upper = None
        self.entry_lower = None
        self.entry_price = None

    def _consolidation_range(self):
        highs = [self.data.high[-i] for i in range(1, self.p.consolidation_period + 1)]
        lows = [self.data.low[-i] for i in range(1, self.p.consolidation_period + 1)]
        upper = max(highs)
        lower = min(lows)

        if upper - lower <= self.p.range_percent * self.data.close[0]:
            return upper, lower
        return None

    def _position_size(self):
        return (self.broker.getvalue() * self.p.capital_fraction) / self.data.close[0]

    def next(self):
        if self.order:
            return

        if self.position:
            if self.entry_price is None:
                return

            if self.position.size > 0:
                if (self.data.close[0] >= self.entry_price * (1.0 + self.p.take_profit) or
                        self.data.close[0] <= self.entry_price * (1.0 - self.p.stop_loss) or
                        self.data.close[0] < self.entry_lower):
                    self.order = self.close()
                    self.order_role = 'exit'
            else:
                if (self.data.close[0] <= self.entry_price * (1.0 - self.p.take_profit) or
                        self.data.close[0] >= self.entry_price * (1.0 + self.p.stop_loss) or
                        self.data.close[0] > self.entry_upper):
                    self.order = self.close()
                    self.order_role = 'exit'
            return

        consolidation = self._consolidation_range()
        if consolidation is None:
            return

        upper, lower = consolidation
        if self.data.volume[0] < self.volume_sma[0] * self.p.volume_multiple:
            return

        size = self._position_size()
        if self.data.close[0] > upper:
            self.pending_upper = upper
            self.pending_lower = lower
            self.order = self.buy(size=size)
            self.order_role = 'entry'
        elif self.data.close[0] < lower:
            self.pending_upper = upper
            self.pending_lower = lower
            self.order = self.sell(size=size)
            self.order_role = 'entry'

    def notify_order(self, order):
        if order.status == order.Completed:
            if self.order_role == 'entry':
                self.entry_price = order.executed.price
                self.entry_upper = self.pending_upper
                self.entry_lower = self.pending_lower
                self.pending_upper = None
                self.pending_lower = None
            elif self.order_role == 'exit':
                self.entry_price = None
                self.entry_upper = None
                self.entry_lower = None

        if order.status in (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            if self.order_role == 'entry' and order.status != order.Completed:
                self.pending_upper = None
                self.pending_lower = None
            self.order = None
            self.order_role = None
