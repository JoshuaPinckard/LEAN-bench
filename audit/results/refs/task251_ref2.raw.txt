import math
import backtrader as bt


class BreakoutSRStrategy(bt.Strategy):
    def __init__(self):
        self.sma_vol = bt.indicators.SMA(self.data.volume, period=20)
        self.entry_order = None
        self.sl_order = None
        self.tp_order = None
        self.broken_level = None
        self.entry_bar = None
        self.entry_price = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            if order == self.entry_order:
                self.entry_order = None
                self.entry_price = order.executed.price
                self.entry_bar = len(self)

                if self.position.size > 0:
                    sl_price = self.broken_level * 0.99
                    tp_price = self.entry_price * 1.02
                    self.sl_order = self.sell(
                        size=abs(self.position.size),
                        price=sl_price,
                        exectype=bt.Order.Stop,
                    )
                    self.tp_order = self.sell(
                        size=abs(self.position.size),
                        price=tp_price,
                        exectype=bt.Order.Limit,
                    )
                elif self.position.size < 0:
                    sl_price = self.broken_level * 1.01
                    tp_price = self.entry_price * 0.98
                    self.sl_order = self.buy(
                        size=abs(self.position.size),
                        price=sl_price,
                        exectype=bt.Order.Stop,
                    )
                    self.tp_order = self.buy(
                        size=abs(self.position.size),
                        price=tp_price,
                        exectype=bt.Order.Limit,
                    )

            elif order in (self.sl_order, self.tp_order):
                if order == self.sl_order and self.tp_order:
                    self.cancel(self.tp_order)
                    self.tp_order = None
                elif order == self.tp_order and self.sl_order:
                    self.cancel(self.sl_order)
                    self.sl_order = None
                self.sl_order = None
                self.tp_order = None

        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            if order == self.entry_order:
                self.entry_order = None
            elif order == self.sl_order:
                self.sl_order = None
            elif order == self.tp_order:
                self.tp_order = None

    def next(self):
        if len(self) < 101:
            return

        if self.position:
            bars_held = len(self) - self.entry_bar
            if 1 <= bars_held <= 5:
                if self.position.size > 0 and self.data.close[0] < self.broken_level:
                    if self.sl_order:
                        self.cancel(self.sl_order)
                        self.sl_order = None
                    if self.tp_order:
                        self.cancel(self.tp_order)
                        self.tp_order = None
                    self.close()
                elif self.position.size < 0 and self.data.close[0] > self.broken_level:
                    if self.sl_order:
                        self.cancel(self.sl_order)
                        self.sl_order = None
                    if self.tp_order:
                        self.cancel(self.tp_order)
                        self.tp_order = None
                    self.close()
            return

        if self.entry_order is not None:
            return

        highs_60 = [self.data.high[-i] for i in range(1, 61)]
        lows_60 = [self.data.low[-i] for i in range(1, 61)]
        resistance = max(highs_60)
        support = min(lows_60)

        highs_100 = [self.data.high[-i] for i in range(1, 101)]
        lows_100 = [self.data.low[-i] for i in range(1, 101)]

        res_touches = sum(1 for h in highs_100 if abs(h - resistance) / resistance <= 0.002)
        sup_touches = sum(1 for l in lows_100 if abs(l - support) / support <= 0.002)

        vol_ok = self.data.volume[0] > self.sma_vol[0] * 1.30

        if self.data.close[0] >= resistance * 1.005 and vol_ok and res_touches >= 3:
            entry_est = self.data.close[0]
            stop_price = resistance * 0.99
            risk_per_share = abs(entry_est - stop_price)
            if risk_per_share > 0:
                equity = self.broker.getvalue()
                size = math.floor((equity * 0.01) / risk_per_share)
                if size > 0:
                    self.broken_level = resistance
                    self.entry_order = self.buy(size=size)

        elif self.data.close[0] <= support * 0.995 and vol_ok and sup_touches >= 3:
            entry_est = self.data.close[0]
            stop_price = support * 1.01
            risk_per_share = abs(entry_est - stop_price)
            if risk_per_share > 0:
                equity = self.broker.getvalue()
                size = math.floor((equity * 0.01) / risk_per_share)
                if size > 0:
                    self.broken_level = support
                    self.entry_order = self.sell(size=size)
