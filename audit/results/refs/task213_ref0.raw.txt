import backtrader as bt


class CVaRProxyMomentumStrategy(bt.Strategy):
    params = (
        ('ema_fast', 20),
        ('ema_slow', 100),
        ('macd_fast', 12),
        ('macd_slow', 26),
        ('macd_signal', 9),
        ('atr_period', 14),
        ('c', 2.0),
        ('r', 0.02),
        ('vol_sma_period', 20),
        ('vol_min', 1000000),
        ('rebalance_period', 'daily'),
    )

    def __init__(self):
        self.ema_fast_ind = bt.indicators.EMA(self.data.close, period=self.p.ema_fast)
        self.ema_slow_ind = bt.indicators.EMA(self.data.close, period=self.p.ema_slow)
        self.macd = bt.indicators.MACD(
            self.data.close,
            period_me1=self.p.macd_fast,
            period_me2=self.p.macd_slow,
            period_signal=self.p.macd_signal,
        )
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.vol_sma = bt.indicators.SMA(self.data.volume, period=self.p.vol_sma_period)
        self.crossover = bt.indicators.CrossOver(self.ema_fast_ind, self.ema_slow_ind)

        self.order = None
        self.last_rebalance_date = None

    def _is_rebalance_day(self):
        if self.p.rebalance_period == 'daily':
            return True
        dt = self.data.datetime.date(0)
        if self.last_rebalance_date is None:
            return True
        if self.p.rebalance_period == 'weekly':
            return dt.isocalendar()[:2] != self.last_rebalance_date.isocalendar()[:2]
        if self.p.rebalance_period == 'monthly':
            return (dt.year, dt.month) != (self.last_rebalance_date.year, self.last_rebalance_date.month)
        return True

    def next(self):
        if self.order:
            return

        if self.position and self.crossover[0] < 0:
            self.order = self.close()
            return

        if not self._is_rebalance_day():
            return

        self.last_rebalance_date = self.data.datetime.date(0)

        bullish = self.ema_fast_ind[0] > self.ema_slow_ind[0] and self.macd.macd[0] > 0
        liquidity_ok = self.vol_sma[0] > self.p.vol_min

        if bullish and liquidity_ok:
            es95 = self.p.c * self.atr[0] / self.data.close[0]
            w = min(1.0, self.p.r / es95) if es95 > 0 else 0.0
        else:
            w = 0.0

        self.order = self.order_target_percent(target=w)

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None
