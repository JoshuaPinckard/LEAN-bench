import backtrader as bt


class MultiTimeframePatternStrategy(bt.Strategy):
    """Reference implementation of Multi-Timeframe Candlestick Pattern Aggregation Strategy."""

    params = (
        ("risk_pct", 0.01),
        ("atr_period", 14),
        ("stop_atr_mult", 0.5),
        ("target_atr_mult", 1.5),
    )

    def __init__(self):
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.order = None
        self.entry_price = None
        self.stop_price = None
        self.take_profit_price = None
        self.entry_atr = None
        self.latest_counts = {}

    def notify_order(self, order):
        if order.status in [order.Completed]:
            if order.isbuy():
                self.entry_price = order.executed.price
                if self.entry_atr is not None:
                    self.stop_price = (
                        self.entry_price - self.p.stop_atr_mult * self.entry_atr
                    )
                    self.take_profit_price = (
                        self.entry_price + self.p.target_atr_mult * self.entry_atr
                    )
            elif order.issell():
                self.entry_price = None
                self.stop_price = None
                self.take_profit_price = None
                self.entry_atr = None
            self.order = None
        elif order.status in [
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            self.order = None

    def _is_bullish_engulfing(self, c1, c2):
        c1_open, c1_high, c1_low, c1_close = c1
        c2_open, c2_high, c2_low, c2_close = c2
        is_c1_bearish = c1_close < c1_open
        is_c2_bullish = c2_close > c2_open
        engulfs = (c2_open <= c1_close) and (c2_close >= c1_open)
        return is_c1_bearish and is_c2_bullish and engulfs

    def _is_bearish_engulfing(self, c1, c2):
        c1_open, c1_high, c1_low, c1_close = c1
        c2_open, c2_high, c2_low, c2_close = c2
        is_c1_bullish = c1_close > c1_open
        is_c2_bearish = c2_close < c2_open
        engulfs = (c2_open >= c1_close) and (c2_close <= c1_open)
        return is_c1_bullish and is_c2_bearish and engulfs

    def _aggregate_candles(self, bars, n_bars_per_candle):
        candles = []
        num_candles = len(bars) // n_bars_per_candle
        for k in range(num_candles):
            group = bars[k * n_bars_per_candle : (k + 1) * n_bars_per_candle]
            c_open = group[0][0]
            c_high = max(b[1] for b in group)
            c_low = min(b[2] for b in group)
            c_close = group[-1][3]
            candles.append((c_open, c_high, c_low, c_close))
        return candles

    def _count_patterns(self, candles, pattern_fn):
        count = 0
        for i in range(len(candles) - 1):
            if pattern_fn(candles[i], candles[i + 1]):
                count += 1
        return count

    def next(self):
        if len(self.data) < 16:
            return

        bars = [
            (
                self.data.open[i],
                self.data.high[i],
                self.data.low[i],
                self.data.close[i],
            )
            for i in range(-15, 1)
        ]

        if self.position:
            if self.order is not None:
                return

            if self.stop_price is not None and self.data.low[0] <= self.stop_price:
                self.order = self.close()
                return

            if (
                self.take_profit_price is not None
                and self.data.high[0] >= self.take_profit_price
            ):
                self.order = self.close()
                return

            c15_candles = self._aggregate_candles(bars, 1)
            if self._count_patterns(c15_candles[-2:], self._is_bearish_engulfing) > 0:
                self.order = self.close()
                return

            return

        if self.order is not None:
            return

        candles_120m = self._aggregate_candles(bars, 8)
        candles_60m = self._aggregate_candles(bars, 4)
        candles_30m = self._aggregate_candles(bars, 2)
        candles_15m = self._aggregate_candles(bars, 1)

        c120 = self._count_patterns(candles_120m, self._is_bullish_engulfing)
        c60 = self._count_patterns(candles_60m, self._is_bullish_engulfing)
        c30 = self._count_patterns(candles_30m, self._is_bullish_engulfing)
        c15 = self._count_patterns(candles_15m, self._is_bullish_engulfing)

        self.latest_counts = {
            "120m": c120,
            "60m": c60,
            "30m": c30,
            "15m": c15,
        }

        if c120 >= 1 and c60 > c120 and c30 > c60 and c15 > c30:
            atr_val = self.atr[0]
            if atr_val <= 0:
                return

            stop_dist = self.p.stop_atr_mult * atr_val
            capital = self.broker.getvalue()
            risk_amt = capital * self.p.risk_pct

            size = int(risk_amt / stop_dist)
            if size > 0:
                self.entry_atr = atr_val
                self.order = self.buy(size=size)
