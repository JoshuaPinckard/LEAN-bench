import backtrader as bt
import math


class SentimentDonchianStrategy(bt.Strategy):
    def __init__(self):
        self.order = None
        self.stop_order = None
        self.stop_loss = None
        self.entry_atr = None

        self.donchian_upper = bt.indicators.Highest(self.data.high, period=20)
        self.donchian_lower = bt.indicators.Lowest(self.data.low, period=20)
        self.atr = bt.indicators.ATR(self.data, period=14)

        self.sentiment = getattr(self.data, 'sentiment', None)
        self.sentiment_sma20 = (
            bt.indicators.SMA(self.sentiment, period=20)
            if self.sentiment is not None else None
        )

    def next(self):
        if self.order:
            return

        if self.sentiment is None:
            return

        sentiment = self.sentiment[0]
        sentiment_sma20 = self.sentiment_sma20[0]

        long_signal = (
            self.data.close[-1] <= self.donchian_upper[-1]
            and self.data.close[0] > self.donchian_upper[-1]
            and sentiment > sentiment_sma20
        )
        short_signal = (
            self.data.close[-1] >= self.donchian_lower[-1]
            and self.data.close[0] < self.donchian_lower[-1]
            and sentiment < sentiment_sma20
        )

        if self.position.size > 0:
            if short_signal:
                if self.stop_order:
                    self.cancel(self.stop_order)
                self.order = self.close()
            return

        if self.position.size < 0:
            if long_signal:
                if self.stop_order:
                    self.cancel(self.stop_order)
                self.order = self.close()
            return

        sentiment_strength = abs(
            (sentiment - sentiment_sma20) / max(abs(sentiment_sma20), 1e-6)
        )
        position_size = 1.0 * (1.0 + 2.0 * sentiment_strength)

        if long_signal:
            self.entry_atr = self.atr[0]
            self.order = self.buy(size=position_size)
        elif short_signal:
            self.entry_atr = self.atr[0]
            self.order = self.sell(size=position_size)

    def notify_order(self, order):
        if order.status in [order.Completed]:
            if order.isbuy() and self.position.size > 0:
                self.stop_loss = order.executed.price - (2.0 * self.entry_atr)
                self.stop_order = self.sell(
                    exectype=bt.Order.Stop,
                    price=self.stop_loss,
                    size=self.position.size,
                )
            elif order.issell() and self.position.size < 0:
                self.stop_loss = order.executed.price + (2.0 * self.entry_atr)
                self.stop_order = self.buy(
                    exectype=bt.Order.Stop,
                    price=self.stop_loss,
                    size=abs(self.position.size),
                )

        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            if order is self.order:
                self.order = None
            if order is self.stop_order:
                self.stop_order = None
                self.stop_loss = None
