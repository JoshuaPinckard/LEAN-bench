import backtrader as bt


class IntradayMomentumODTE(bt.Strategy):
    def __init__(self):
        # Volatility filter indicators
        self.atr = bt.ind.ATR(self.data, period=14)
        self.atr_sma = bt.ind.SMA(self.atr, period=20)

        # Direction filter indicators
        self.sma5 = bt.ind.SMA(self.data.close, period=5)
        self.macd = bt.ind.MACDHisto(
            self.data.close, period_me1=12, period_me2=26, period_signal=9
        )

        # Reversal confirmation indicators
        self.stoch = bt.ind.Stochastic(
            self.data, period=14, period_dfast=3, period_dslow=3
        )
        self.ema9 = bt.ind.EMA(self.data.close, period=9)

        # Order and state tracking
        self.order = None
        self.current_date = None
        self.day_open = None
        self.dip_occurred = False
        self.entered_today = False
        self.entry_price = None
        self.entry_atr = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            if order.isbuy():
                self.entry_price = order.executed.price
                self.entry_atr = self.atr[0]
            self.order = None
        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        date_now = self.data.datetime.date(0)

        # New day reset
        if date_now != self.current_date:
            self.current_date = date_now
            self.day_open = self.data.open[0]
            self.dip_occurred = False
            self.entered_today = False

        # Track dip condition (drop >= 0.5% from open price)
        if self.day_open is not None and self.day_open > 0:
            if (self.day_open - self.data.low[0]) / self.day_open >= 0.005:
                self.dip_occurred = True

        # Position exit management
        if self.position:
            t = self.data.datetime.time(0)
            # Forced close at end of day (15:55 or later)
            if (t.hour == 15 and t.minute >= 55) or t.hour > 15:
                if not self.order:
                    self.order = self.close()
                return

            entry_p = (
                self.entry_price
                if self.entry_price is not None
                else self.position.price
            )
            entry_a = (
                self.entry_atr if self.entry_atr is not None else self.atr[0]
            )

            take_profit = entry_p + 2.5 * entry_a
            stop_loss = entry_p - 1.0 * entry_a

            if self.data.high[0] >= take_profit or self.data.low[0] <= stop_loss:
                if not self.order:
                    self.order = self.close()
                return

        # Position entry evaluation
        if not self.position and not self.order and not self.entered_today:
            vol_filter = self.atr[0] > self.atr_sma[0]
            dir_filter = (self.data.close[0] > self.sma5[0]) or (
                self.macd.histo[0] > 0
            )

            t = self.data.datetime.time(0)
            is_open_bar = t.hour == 9 and t.minute == 30
            in_dip_window = (t.hour == 9 and t.minute > 30) or (
                t.hour == 10 and t.minute <= 30
            )

            # Entry 1 (open): buy on the first candle of the day if filters are met
            if is_open_bar:
                if vol_filter and dir_filter:
                    self.order = self.buy(size=1)
                    self.entered_today = True

            # Entry 2 (dip): within first 30-60 minutes if dip occurred and reversal confirmed
            elif in_dip_window:
                stoch_cross = (
                    (self.stoch.percK[-1] <= self.stoch.percD[-1])
                    and (self.stoch.percK[0] > self.stoch.percD[0])
                    and (self.stoch.percK[-1] < 20 or self.stoch.percK[0] < 20)
                )
                ema_cross = (self.data.close[-1] <= self.ema9[-1]) and (
                    self.data.close[0] > self.ema9[0]
                )
                reversal_confirm = stoch_cross or ema_cross

                if (
                    self.dip_occurred
                    and reversal_confirm
                    and vol_filter
                    and dir_filter
                ):
                    self.order = self.buy(size=1)
                    self.entered_today = True
