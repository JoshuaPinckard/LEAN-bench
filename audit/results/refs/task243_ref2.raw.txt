import backtrader as bt


class BreakoutBollingerStrategy(bt.Strategy):
    def __init__(self):
        self.boll = bt.indicators.BollingerBands(
            self.data.close, period=20, devfactor=2.0
        )
        self.order = None
        self.tp_order = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            if order.isbuy():
                entry_price = order.executed.price
                tp_price = entry_price * 1.02
                self.tp_order = self.sell(
                    exectype=bt.Order.Limit,
                    price=tp_price,
                    size=order.executed.size,
                )
            elif order.issell():
                if self.tp_order and order.ref == self.tp_order.ref:
                    self.tp_order = None

        if self.order and order.ref == self.order.ref:
            if order.status in [
                order.Completed,
                order.Canceled,
                order.Margin,
                order.Rejected,
            ]:
                self.order = None

        if self.tp_order and order.ref == self.tp_order.ref:
            if order.status in [order.Canceled, order.Margin, order.Rejected]:
                self.tp_order = None

    def next(self):
        if self.order:
            return

        if not self.position:
            breakout = self.data.close[0] > self.boll.lines.top[0]
            last_3_below = (
                self.data.close[-1] <= self.boll.lines.top[-1]
                and self.data.close[-2] <= self.boll.lines.top[-2]
                and self.data.close[-3] <= self.boll.lines.top[-3]
            )
            sma_rising = self.boll.lines.mid[0] > self.boll.lines.mid[-1]

            if breakout and last_3_below and sma_rising:
                entry_price = self.data.close[0]
                upper_band = self.boll.lines.top[0]
                risk_per_share = entry_price - upper_band

                if risk_per_share > 0:
                    capital = self.broker.getvalue()
                    max_risk = capital * 0.01
                    size = int(max_risk / risk_per_share)

                    cash = self.broker.getcash()
                    max_cash_size = int(cash / entry_price)
                    size = min(size, max_cash_size)

                    if size > 0:
                        self.order = self.buy(size=size)
        else:
            if self.data.close[0] < self.boll.lines.top[0]:
                if self.tp_order:
                    self.cancel(self.tp_order)
                    self.tp_order = None
                self.order = self.close()
