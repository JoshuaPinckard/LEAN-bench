import backtrader as bt


class DcaRsiStrategy(bt.Strategy):
    params = (("rsi_period", 14),)

    def __init__(self):
        self.rsi = bt.indicators.RSI(self.data.close, period=self.p.rsi_period)
        self.order = None
        self.primary_entry_price = None
        self.primary_direction = None
        self.dca_added = False
        self.initial_capital = 10000.0

    def _size(self):
        return (self.initial_capital * 0.01) / self.data.close[0]

    def notify_order(self, order):
        if order.status in (order.Completed, order.Canceled,
                            order.Margin, order.Rejected):
            if order.status == order.Completed:
                if self.primary_entry_price is None and order.isbuy():
                    self.primary_entry_price = order.executed.price
                    self.primary_direction = 1
                elif self.primary_entry_price is None and order.issell():
                    self.primary_entry_price = order.executed.price
                    self.primary_direction = -1

            self.order = None

    def next(self):
        if self.order:
            return

        if not self.position:
            self.primary_entry_price = None
            self.primary_direction = None
            self.dca_added = False

            if self.rsi[0] < 30:
                self.order = self.buy(size=self._size())
            elif self.rsi[0] > 70:
                self.order = self.sell(size=self._size())
            return

        if self.position.size > 0:
            if self.rsi[0] > 50:
                self.order = self.close()
            elif (not self.dca_added and
                  self.data.close[0] <= self.primary_entry_price * (1 - 0.017)):
                self.dca_added = True
                self.order = self.buy(size=self._size())

        elif self.position.size < 0:
            if self.rsi[0] < 50:
                self.order = self.close()
            elif (not self.dca_added and
                  self.data.close[0] >= self.primary_entry_price * (1 + 0.017)):
                self.dca_added = True
                self.order = self.sell(size=self._size())
