import backtrader as bt


class NXXTMultiMATrendStrategy(bt.Strategy):
    params = dict(
        sma_fast=30,
        sma_mid=60,
        sma_slow=120,
        consolidation_pct=0.05,
        consolidation_bars=5,
        take_profit_pct=0.10,
        risk_pct=0.02,
        atr_period=14,
        atr_avg_period=30,
        vol_mult=1.5,
    )

    def __init__(self):
        self.sma30 = bt.indicators.SMA(self.data.close, period=self.p.sma_fast)
        self.sma60 = bt.indicators.SMA(self.data.close, period=self.p.sma_mid)
        self.sma120 = bt.indicators.SMA(self.data.close, period=self.p.sma_slow)
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.atr_avg = bt.indicators.SMA(self.atr, period=self.p.atr_avg_period)

        self.order = None
        self.entry_price = None

    def notify_order(self, order):
        if order.status == order.Completed:
            if order.isbuy():
                self.entry_price = order.executed.price
            self.order = None
        elif order.status in (order.Canceled, order.Margin, order.Rejected):
            self.order = None

    def _consolidation_breakout(self):
        n = self.p.consolidation_bars
        window = [self.data.close[-i] for i in range(1, n + 1)]
        hi = max(window)
        lo = min(window)
        if lo <= 0:
            return False, None
        in_range = (hi - lo) / lo <= self.p.consolidation_pct
        if not in_range:
            return False, None
        return self.data.close[0] > hi, hi

    def next(self):
        if self.order:
            return

        if self.position:
            if self.entry_price is not None and \
                    self.data.close[0] >= self.entry_price * (1 + self.p.take_profit_pct):
                self.order = self.close()
                return
            if self.data.close[0] < self.sma30[0]:
                self.order = self.close()
                return
            return

        above_all = (self.data.close[0] > self.sma30[0] and
                     self.data.close[0] > self.sma60[0] and
                     self.data.close[0] > self.sma120[0])
        rising = (self.sma30[0] > self.sma30[-1] and
                  self.sma60[0] > self.sma60[-1] and
                  self.sma120[0] > self.sma120[-1])
        if not (above_all and rising):
            return

        breakout, range_hi = self._consolidation_breakout()
        if not breakout:
            return

        if self.atr_avg[0] and self.atr[0] > self.p.vol_mult * self.atr_avg[0]:
            return

        stop_price = self.sma30[0]
        signal_price = self.data.close[0]
        risk_per_share = signal_price - stop_price
        if risk_per_share <= 0:
            return

        risk_amount = self.p.risk_pct * self.broker.getvalue()
        size = risk_amount / risk_per_share
        if size <= 0:
            return

        self.order = self.buy(size=size)
