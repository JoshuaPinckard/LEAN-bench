import math

import backtrader as bt


class ScalpingPullbackStrategy(bt.Strategy):

    def __init__(self):
        self.sma50 = bt.indicators.SMA(self.data.close, period=50)
        self.rsi = bt.indicators.RSI(self.data.close, period=14)
        self.atr = bt.indicators.ATR(self.data, period=14)
        self.atr_sma = bt.indicators.SMA(self.atr, period=50)
        self.boll = bt.indicators.BollingerBands(self.data.close, period=20, devfactor=2)
        self.vol_sma = bt.indicators.SMA(self.data.volume, period=20)

        self.order = None
        self.entry_price = None
        self.stop_price = None
        self.target_price = None
        self.signal_atr = None
        self.entry_bar_index = None

    def notify_order(self, order):
        if order.status in (order.Completed, order.Canceled, order.Margin, order.Rejected):
            if order.status == order.Completed:
                if order.isbuy():
                    self.entry_price = order.executed.price
                    self.stop_price = self.entry_price - 1.5 * self.signal_atr
                    self.target_price = self.entry_price + 1.5 * self.signal_atr
                    self.entry_bar_index = len(self)
                else:
                    self.entry_price = None
                    self.stop_price = None
                    self.target_price = None
                    self.signal_atr = None
                    self.entry_bar_index = None
            self.order = None

    def next(self):
        if self.order:
            return

        if not self.position:
            trend_ok = self.data.close[0] > self.sma50[0]

            pullback_ok = False
            for i in range(1, 6):
                if self.rsi[-i] < 50 and self.rsi[-i - 1] >= 50:
                    pullback_ok = True
                    break

            trigger_ok = self.rsi[0] > 50 and self.rsi[-1] <= 50

            volume_ok = self.data.volume[0] > self.vol_sma[0]

            atr_val = self.atr[0]
            volatility_ok = atr_val <= self.atr_sma[0] * 1.3
            gap = abs(self.data.open[0] - self.data.close[-1])
            gap_ok = gap <= atr_val

            if trend_ok and pullback_ok and trigger_ok and volume_ok and volatility_ok and gap_ok:
                risk_per_share = 1.5 * atr_val
                if risk_per_share > 0:
                    equity = self.broker.getvalue()
                    size = math.floor(0.02 * equity / risk_per_share)
                    if size > 0:
                        self.signal_atr = atr_val
                        self.order = self.buy(size=size)
        else:
            bars_since_entry = len(self) - self.entry_bar_index

            if self.data.close[0] <= self.stop_price:
                self.order = self.close()
            elif bars_since_entry < 3 and (self.data.close[0] - self.entry_price) >= self.signal_atr:
                self.order = self.close()
            elif self.data.close[0] >= self.target_price:
                self.order = self.close()
            elif bars_since_entry < 5 and self.data.high[0] >= self.boll.top[0]:
                self.order = self.close()
            elif self.rsi[0] < 40:
                self.order = self.close()
