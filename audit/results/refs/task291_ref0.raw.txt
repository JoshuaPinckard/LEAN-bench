import backtrader as bt


class BreakoutRetestStrategy(bt.Strategy):
    params = (
        ('ema_period', 10),
        ('breakout_period', 50),
        ('volume_period', 50),
        ('atr_period', 14),
        ('retest_window', 5),
        ('take_profit_pct', 0.10),
        ('risk_pct', 0.02),
    )

    def __init__(self):
        # 10-day EMA: support level used for the retest / stop
        self.ema = bt.indicators.ExponentialMovingAverage(
            self.data.close, period=self.p.ema_period)

        # Breakout level: highest high over the last 50 days PRIOR to the current day
        self.breakout_level = bt.indicators.Highest(
            self.data.high(-1), period=self.p.breakout_period)

        # Average daily volume over the last 50 days
        self.volavg = bt.indicators.SimpleMovingAverage(
            self.data.volume, period=self.p.volume_period)

        # ATR(14) volatility measure (comparison to market average is infeasible here)
        self.atr = bt.indicators.AverageTrueRange(
            self.data, period=self.p.atr_period)

        self.order = None
        self.retest_days_left = 0
        self.entry_ema = None
        self.take_profit_price = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return
        if order.status == order.Completed:
            if order.isbuy():
                # Stop-loss level fixed at the 10-day EMA at the time of entry
                self.entry_ema = self.ema[0]
                # Take-profit at 10% above the entry (fill) price
                self.take_profit_price = order.executed.price * (1.0 + self.p.take_profit_pct)
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        if self.order:
            return

        if self.position:
            # Stop-loss: price closes below the 10-day EMA (level fixed at entry)
            if self.data.close[0] < self.entry_ema:
                self.order = self.close()
            # Take-profit: reaching 10% from the entry price
            elif self.data.close[0] >= self.take_profit_price:
                self.order = self.close()
            return

        # Flat: manage the breakout -> retest state machine
        if self.data.close[0] > self.breakout_level[0]:
            # Breakout confirmed (close exceeds the 50-day high); open a 5-day retest window
            self.retest_days_left = self.p.retest_window
        elif self.retest_days_left > 0:
            # Within 5 days after the breakout, look for a retest of the 10-day EMA
            touched_ema = self.data.low[0] <= self.ema[0]          # pulls back to the EMA
            not_below = self.data.close[0] >= self.ema[0]          # but does not close below it
            volume_ok = self.data.volume[0] > self.volavg[0]       # demand confirmation
            if touched_ema and not_below and volume_ok:
                # Position size = 2% of total trading capital
                cash_alloc = self.broker.getvalue() * self.p.risk_pct
                size = cash_alloc / self.data.close[0]
                self.order = self.buy(size=size)
                self.retest_days_left = 0
            else:
                self.retest_days_left -= 1
