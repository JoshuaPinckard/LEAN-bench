import backtrader as bt


class BreakoutRetestStrategy(bt.Strategy):
    params = dict(
        ema_period=10,
        breakout_period=50,
        volume_period=50,
        retest_window=5,
        take_profit_pct=0.10,
        position_pct=0.02,
    )

    def __init__(self):
        self.ema10 = bt.indicators.EMA(self.data.close, period=self.p.ema_period)
        self.highest50 = bt.indicators.Highest(self.data.high, period=self.p.breakout_period)
        # "highest price over the last 50 days prior to the current day" -> shift back one bar
        self.breakout_level = self.highest50(-1)
        self.avgvol50 = bt.indicators.SMA(self.data.volume, period=self.p.volume_period)

        self.order = None
        self.awaiting_retest = False
        self.breakout_bar = None
        self.entry_price = None

    def notify_order(self, order):
        if order.status in (order.Completed, order.Canceled, order.Margin, order.Rejected):
            if order.status == order.Completed and order.isbuy():
                self.entry_price = order.executed.price
            self.order = None

    def next(self):
        if self.order:
            return

        if self.position:
            if self.data.close[0] < self.ema10[0]:
                self.order = self.close()
                return
            if self.entry_price is not None and self.data.close[0] >= self.entry_price * (1 + self.p.take_profit_pct):
                self.order = self.close()
                return
            return

        if self.awaiting_retest:
            days_elapsed = len(self) - self.breakout_bar
            if self.data.close[0] < self.ema10[0]:
                self.awaiting_retest = False
            elif self.data.low[0] <= self.ema10[0] and self.data.volume[0] > self.avgvol50[0]:
                size = (self.broker.getvalue() * self.p.position_pct) / self.data.close[0]
                self.order = self.buy(size=size)
                self.awaiting_retest = False
                return
            elif days_elapsed >= self.p.retest_window:
                self.awaiting_retest = False

        if not self.awaiting_retest and not self.position:
            if self.data.close[0] > self.breakout_level[0]:
                self.awaiting_retest = True
                self.breakout_bar = len(self)
