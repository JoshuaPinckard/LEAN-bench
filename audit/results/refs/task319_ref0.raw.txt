import backtrader as bt


class RsiDonchianBreakoutStrategy(bt.Strategy):
    params = (
        ('rsi_period', 14),
        ('donchian_period', 20),
        ('rsi_oversold', 30),
        ('rsi_overbought', 70),
        ('stop_pct', 0.02),
    )

    def __init__(self):
        self.rsi = bt.indicators.RSI(self.data.close, period=self.p.rsi_period)
        self.donch_high = bt.indicators.Highest(self.data.high, period=self.p.donchian_period)
        self.donch_low = bt.indicators.Lowest(self.data.low, period=self.p.donchian_period)
        self.donch_mid = (self.donch_high + self.donch_low) / 2.0

        # Reference the channel one bar back so today's close is compared
        # against a band built strictly from prior bars (see assumptions).
        self.donch_high_prev = self.donch_high(-1)
        self.donch_low_prev = self.donch_low(-1)
        self.donch_mid_prev = self.donch_mid(-1)

        self.order = None
        self.stop_order = None
        self.entry_price = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status == order.Completed:
            if order == self.stop_order:
                self.stop_order = None
                self.entry_price = None
            elif order == self.order:
                if self.position:
                    self.entry_price = order.executed.price
                    if order.isbuy():
                        self.stop_order = self.sell(
                            exectype=bt.Order.Stop,
                            price=self.entry_price * (1.0 - self.p.stop_pct),
                        )
                    else:
                        self.stop_order = self.buy(
                            exectype=bt.Order.Stop,
                            price=self.entry_price * (1.0 + self.p.stop_pct),
                        )
                else:
                    if self.stop_order:
                        self.cancel(self.stop_order)
                    self.stop_order = None
                    self.entry_price = None
                self.order = None

        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            if order == self.order:
                self.order = None
            if order == self.stop_order:
                self.stop_order = None

    def next(self):
        if self.order:
            return

        if not self.position:
            if (self.rsi[0] < self.p.rsi_oversold
                    and self.data.close[0] > self.donch_high_prev[0]):
                self.order = self.buy()
            elif (self.rsi[0] > self.p.rsi_overbought
                    and self.data.close[0] < self.donch_low_prev[0]):
                self.order = self.sell()
        else:
            if self.position.size > 0 and self.data.close[0] < self.donch_mid_prev[0]:
                if self.stop_order:
                    self.cancel(self.stop_order)
                self.order = self.close()
            elif self.position.size < 0 and self.data.close[0] > self.donch_mid_prev[0]:
                if self.stop_order:
                    self.cancel(self.stop_order)
                self.order = self.close()
