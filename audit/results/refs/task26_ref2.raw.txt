import backtrader as bt


class IntradayDevFromOpen(bt.Strategy):
    params = (("dev_bps", 50.0),)

    def __init__(self):
        self.order = None
        self.current_date = None
        self.day_open = None
        self.traded_today = False

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return
        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            self.order = None

    def next(self):
        dt = self.data.datetime.date(0)

        # Detect new trading day and record morning open
        if dt != self.current_date:
            self.current_date = dt
            self.day_open = self.data.open[0]
            self.traded_today = False

        # Detect if current minute bar is the last bar of the trading day
        is_last_bar_of_day = False
        if len(self.data) == self.data.buflen():
            is_last_bar_of_day = True
        elif self.data.datetime.date(1) != dt:
            is_last_bar_of_day = True

        # End of day forced exit
        if is_last_bar_of_day:
            if self.position and self.order is None:
                self.order = self.close()
            return

        # Entry logic: maximum one trade per day
        if self.order is None and not self.position and not self.traded_today:
            dev_ratio = self.p.dev_bps / 10000.0
            long_threshold = self.day_open * (1.0 - dev_ratio)
            short_threshold = self.day_open * (1.0 + dev_ratio)

            close_price = self.data.close[0]

            if close_price < long_threshold:
                self.order = self.buy()
                self.traded_today = True
            elif close_price > short_threshold:
                self.order = self.sell()
                self.traded_today = True
