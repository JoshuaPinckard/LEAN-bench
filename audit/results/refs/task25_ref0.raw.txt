import backtrader as bt


class SessionTimeStrategy(bt.Strategy):
    params = dict(
        sl_pct=0.005,
        tp_pct=0.01,
        size_pct=0.01,
        atr_period=5,
        atr_threshold_pct=0.0005,
        ny_open_min=570,          # 09:30
        ny_first2h_end_min=690,   # 11:30
        power_hour_start_min=900, # 15:00
        ny_close_min=960,         # 16:00
        pre_ny_ban_start_min=540, # 09:00
        london_start_min=180,     # 03:00
        london_end_min=480,       # 08:00
        asian_start_min=1140,     # 19:00
        asian_end_min=180,        # 03:00 (wraps midnight)
    )

    def __init__(self):
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)

        self.order = None
        self.entering = False
        self.entry_price = None
        self.entry_date = None
        self.stop_price = None
        self.take_price = None
        self.window_end_min = None

        self.asian_high = None
        self.asian_low = None
        self._asian_high_today = None
        self._asian_low_today = None
        self._in_asian_prev = False
        self._in_london_prev = False
        self.swept_above = False
        self.swept_below = False

    def _minutes(self):
        t = self.data.datetime.time(0)
        return t.hour * 60 + t.minute

    def _in_range(self, m, start, end):
        if start <= end:
            return start <= m < end
        return m >= start or m < end

    def notify_order(self, order):
        if order.status == order.Completed:
            if self.entering:
                self.entry_price = order.executed.price
                self.entry_date = self.data.datetime.date(0)
                if order.isbuy():
                    self.stop_price = self.entry_price * (1 - self.p.sl_pct)
                    self.take_price = self.entry_price * (1 + self.p.tp_pct)
                else:
                    self.stop_price = self.entry_price * (1 + self.p.sl_pct)
                    self.take_price = self.entry_price * (1 - self.p.tp_pct)
            else:
                self.entry_price = None
                self.entry_date = None
                self.stop_price = None
                self.take_price = None
                self.window_end_min = None
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        m = self._minutes()

        in_asian = self._in_range(m, self.p.asian_start_min, self.p.asian_end_min)
        if in_asian:
            if self._asian_high_today is None:
                self._asian_high_today = self.data.high[0]
                self._asian_low_today = self.data.low[0]
            else:
                self._asian_high_today = max(self._asian_high_today, self.data.high[0])
                self._asian_low_today = min(self._asian_low_today, self.data.low[0])
        elif self._in_asian_prev and self._asian_high_today is not None:
            self.asian_high = self._asian_high_today
            self.asian_low = self._asian_low_today
            self._asian_high_today = None
            self._asian_low_today = None
        self._in_asian_prev = in_asian

        in_london = self._in_range(m, self.p.london_start_min, self.p.london_end_min)
        if in_london and not self._in_london_prev:
            self.swept_above = False
            self.swept_below = False
        if in_london:
            if self.asian_high is not None and self.data.high[0] > self.asian_high:
                self.swept_above = True
            if self.asian_low is not None and self.data.low[0] < self.asian_low:
                self.swept_below = True
        self._in_london_prev = in_london

        if self.position:
            if self.order:
                return
            cur_date = self.data.datetime.date(0)
            exit_now = False
            if self.entry_date is not None and cur_date != self.entry_date:
                exit_now = True  # window definitions are time-of-day only; a new day means it elapsed
            elif self._in_range(m, self.p.pre_ny_ban_start_min, self.p.ny_open_min):
                exit_now = True
            elif self.window_end_min is not None and m >= self.window_end_min:
                exit_now = True
            elif self.position.size > 0:
                if self.data.low[0] <= self.stop_price or self.data.high[0] >= self.take_price:
                    exit_now = True
            elif self.position.size < 0:
                if self.data.high[0] >= self.stop_price or self.data.low[0] <= self.take_price:
                    exit_now = True
            if exit_now:
                self.entering = False
                self.order = self.close()
            return

        if self.order:
            return

        if self._in_range(m, self.p.pre_ny_ban_start_min, self.p.ny_open_min):
            return

        if self.atr[0] < self.p.atr_threshold_pct * self.data.close[0]:
            return

        price = self.data.close[0]
        size = int((self.p.size_pct * self.broker.getvalue()) / price)
        if size <= 0:
            return

        in_ny_first2h = self._in_range(m, self.p.ny_open_min, self.p.ny_first2h_end_min)
        in_power_hour = self._in_range(m, self.p.power_hour_start_min, self.p.ny_close_min)

        if in_ny_first2h or in_power_hour:
            window_end = self.p.ny_first2h_end_min if in_ny_first2h else self.p.ny_close_min
            if self.data.close[0] > self.data.open[0]:
                self.entering = True
                self.window_end_min = window_end
                self.order = self.buy(size=size)
            elif self.data.close[0] < self.data.open[0]:
                self.entering = True
                self.window_end_min = window_end
                self.order = self.sell(size=size)
        elif in_london:
            if self.swept_below and self.asian_low is not None and self.data.close[0] > self.asian_low:
                self.entering = True
                self.window_end_min = self.p.london_end_min
                self.swept_below = False
                self.order = self.buy(size=size)
            elif self.swept_above and self.asian_high is not None and self.data.close[0] < self.asian_high:
                self.entering = True
                self.window_end_min = self.p.london_end_min
                self.swept_above = False
                self.order = self.sell(size=size)
