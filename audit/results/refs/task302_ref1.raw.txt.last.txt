import backtrader as bt


class SupertrendStrategy(bt.Strategy):
    class Supertrend(bt.Indicator):
        lines = ("supertrend", "final_upper", "final_lower")
        params = (("period", 10), ("multiplier", 3.0))

        def __init__(self):
            self.atr = bt.indicators.ATR(self.data, period=self.p.period)

        def next(self):
            hl2 = (self.data.high[0] + self.data.low[0]) / 2.0
            basic_upper = hl2 + self.p.multiplier * self.atr[0]
            basic_lower = hl2 - self.p.multiplier * self.atr[0]

            if len(self) == 1:
                self.lines.final_upper[0] = basic_upper
                self.lines.final_lower[0] = basic_lower
                self.lines.supertrend[0] = basic_upper
                return

            if basic_upper < self.lines.final_upper[-1] or self.data.close[-1] > self.lines.final_upper[-1]:
                self.lines.final_upper[0] = basic_upper
            else:
                self.lines.final_upper[0] = self.lines.final_upper[-1]

            if basic_lower > self.lines.final_lower[-1] or self.data.close[-1] < self.lines.final_lower[-1]:
                self.lines.final_lower[0] = basic_lower
            else:
                self.lines.final_lower[0] = self.lines.final_lower[-1]

            if self.lines.supertrend[-1] == self.lines.final_upper[-1]:
                if self.data.close[0] <= self.lines.final_upper[0]:
                    self.lines.supertrend[0] = self.lines.final_upper[0]
                else:
                    self.lines.supertrend[0] = self.lines.final_lower[0]
            else:
                if self.data.close[0] >= self.lines.final_lower[0]:
                    self.lines.supertrend[0] = self.lines.final_lower[0]
                else:
                    self.lines.supertrend[0] = self.lines.final_upper[0]

    def __init__(self):
        self.supertrend = self.Supertrend(self.data, period=10, multiplier=3.0)
        self.pending_orders = set()

    def notify_order(self, order):
        if order.status in (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            self.pending_orders.discard(order.ref)

    def next(self):
        if self.pending_orders or len(self.supertrend) < 2:
            return

        was_bearish = self.supertrend[-1] > self.data.close[-1]
        is_bullish = self.supertrend[0] < self.data.close[0]
        was_bullish = self.supertrend[-1] < self.data.close[-1]
        is_bearish = self.supertrend[0] > self.data.close[0]

        orders = []

        if was_bearish and is_bullish:
            if self.position.size < 0:
                orders.append(self.close())
            if self.position.size <= 0:
                orders.append(self.buy())

        elif was_bullish and is_bearish:
            if self.position.size > 0:
                orders.append(self.close())
            if self.position.size >= 0:
                orders.append(self.sell())

        for order in orders:
            if order is not None:
                self.pending_orders.add(order.ref)
