Let me finalize the design decisions I'd converged on and write the complete strategy now, without further deliberation.

**Locked-in decisions**: freeze ATR(14) at the OR-bar close and reuse it for all that day's levels (stop orders need fixed prices, not intrabar-jittering ones); check the direction filter once, right when the OR window closes, and place a single conditional StopLimit order (true OCO linkage is moot since `SMA20>SMA50` and `SMA20<SMA50` can never both hold); use resting StopLimit bracket orders for stop-loss (resized on every tranche change) and per-tranche take-profit; evaluate trailing-stop and two-consecutive-inside-OR-closes bar-by-bar since they need running state, firing a reactive StopLimit full-exit; add-ons are also reactive StopLimit orders once their compound trigger is confirmed; EOD uses `self.close()` per the literal "close all positions" canonical default, everything else routes through stop-limit per rule 7. R sized at the midpoint of the stated 0.5–1.0% range, against current portfolio value. No `datetime` import needed — session-end and day-change checks use the `.date()`/`.time()` objects backtrader already returns, compared via their own attributes.

```python
import backtrader as bt


class ORBIntradayMomentum(bt.Strategy):
    params = dict(
        risk_pct=0.0075,       # midpoint of stated 0.5-1.0% risk range
        session_end_hour=15,
        session_end_minute=45,  # last regular 15m bar of the NYSE session (bar timestamped by interval start)
    )

    def __init__(self):
        self.atr = bt.indicators.ATR(self.data, period=14)
        self.sma20 = bt.indicators.SMA(self.data, period=20)
        self.sma50 = bt.indicators.SMA(self.data, period=50)
        self.macd = bt.indicators.MACD(self.data, period_me1=12, period_me2=26, period_signal=9)
        self.rsi = bt.indicators.RSI(self.data, period=14)
        self.bb = bt.indicators.BollingerBands(self.data, period=20, devfactor=2)

        self.current_date = None
        self.today_high = None
        self.today_low = None
        self.yesterday_high = None
        self.yesterday_low = None

        self.order_tag = {}
        self._reset_day_state()

    # ---------- day bookkeeping ----------

    def _reset_day_state(self):
        self.or_high = None
        self.or_low = None
        self.or_w = None
        self.day_atr = None
        self.day_skip = True
        self.initial_order_checked = False
        self.pending_entry_order = None
        self.pending_sl = None

        self.direction = None
        self.trades_today = 0
        self.entry_bar_len = None
        self.stop_price = None
        self.stop_order = None
        self.tranches = []
        self.next_tranche_id = 0
        self.extreme_price = None
        self.addon_order = None
        self.exit_order = None

    def _is_new_day(self):
        return self.current_date is None or self.data.datetime.date(0) != self.current_date

    def _is_eod_bar(self):
        t = self.data.datetime.time(0)
        return (t.hour, t.minute) >= (self.p.session_end_hour, self.p.session_end_minute)

    # ---------- order helpers ----------

    def _cancel_all_pending(self):
        for o in (self.pending_entry_order, self.stop_order, self.addon_order, self.exit_order):
            if o is not None and o.alive():
                self.cancel(o)
        for tr in self.tranches:
            o = tr.get('order')
            if o is not None and o.alive():
                self.cancel(o)

    def _momentum_ok(self, direction):
        if direction == 'long':
            return self.macd.macd[0] > 0 or self.rsi[0] > 55
        return self.macd.macd[0] < 0 or self.rsi[0] < 45

    def _resubmit_stop(self):
        if self.stop_order is not None and self.stop_order.alive():
            self.cancel(self.stop_order)
        size = abs(self.position.size)
        if size <= 0:
            self.stop_order = None
            return
        limoff = 0.05 * self.day_atr
        if self.direction == 'long':
            o = self.sell(exectype=bt.Order.StopLimit, price=self.stop_price,
                           plimit=self.stop_price - limoff, size=size)
        else:
            o = self.buy(exectype=bt.Order.StopLimit, price=self.stop_price,
                          plimit=self.stop_price + limoff, size=size)
        self.order_tag[o.ref] = 'stop_loss'
        self.stop_order = o

    def _submit_tp(self, tranche_id, size, target):
        limoff = 0.05 * self.day_atr
        if self.direction == 'long':
            o = self.sell(exectype=bt.Order.StopLimit, price=target,
                           plimit=target - limoff, size=size)
        else:
            o = self.buy(exectype=bt.Order.StopLimit, price=target,
                          plimit=target + limoff, size=size)
        self.order_tag[o.ref] = 'tp_%d' % tranche_id
        return o

    def _submit_addon(self, size):
        limoff = 0.05 * self.day_atr
        px = self.data.close[0]
        if self.direction == 'long':
            o = self.buy(exectype=bt.Order.StopLimit, price=px, plimit=px + limoff, size=size)
        else:
            o = self.sell(exectype=bt.Order.StopLimit, price=px, plimit=px - limoff, size=size)
        self.order_tag[o.ref] = 'addon'
        self.addon_order = o

    def _full_exit(self, tag):
        self._cancel_all_pending()
        size = abs(self.position.size)
        if size <= 0:
            return
        limoff = 0.05 * self.day_atr
        px = self.data.close[0]
        if self.direction == 'long':
            o = self.sell(exectype=bt.Order.StopLimit, price=px, plimit=px - limoff, size=size)
        else:
            o = self.buy(exectype=bt.Order.StopLimit, price=px, plimit=px + limoff, size=size)
        self.order_tag[o.ref] = tag
        self.exit_order = o

    # ---------- main loop ----------

    def next(self):
        if self._is_new_day():
            if self.current_date is not None:
                self.yesterday_high = self.today_high
                self.yesterday_low = self.today_low
            self.current_date = self.data.datetime.date(0)
            self.today_high = self.data.high[0]
            self.today_low = self.data.low[0]

            self._reset_day_state()
            self.or_high = self.data.high[0]
            self.or_low = self.data.low[0]
            self.or_w = self.or_high - self.or_low
            self.day_atr = self.atr[0]
            close0 = self.data.close[0]
            self.day_skip = (self.day_atr / close0 < 0.008) or (self.or_w < 0.3 * self.day_atr)
            return

        self.today_high = max(self.today_high, self.data.high[0])
        self.today_low = min(self.today_low, self.data.low[0])

        if self.day_skip:
            return

        if self._is_eod_bar():
            self._cancel_all_pending()
            if self.position:
                self.close()
            return

        if not self.initial_order_checked:
            self.initial_order_checked = True
            long_ok = self.sma20[0] > self.sma50[0] and (self.macd.macd[0] > 0 or self.rsi[0] > 55)
            short_ok = self.sma20[0] < self.sma50[0] and (self.macd.macd[0] < 0 or self.rsi[0] < 45)
            limoff = 0.05 * self.day_atr
            cash = self.broker.getvalue()
            risk_dollars = self.p.risk_pct * cash

            if long_ok:
                stop = self.or_high + 0.1 * self.day_atr
                sl = self.or_low - 0.1 * self.day_atr
                size = int(risk_dollars / abs(stop - sl))
                if size > 0:
                    o = self.buy(exectype=bt.Order.StopLimit, price=stop,
                                  plimit=stop + limoff, size=size)
                    self.order_tag[o.ref] = 'initial_long'
                    self.pending_entry_order = o
                    self.pending_sl = sl
            elif short_ok:
                stop = self.or_low - 0.1 * self.day_atr
                sl = self.or_high + 0.1 * self.day_atr
                size = int(risk_dollars / abs(stop - sl))
                if size > 0:
                    o = self.sell(exectype=bt.Order.StopLimit, price=stop,
                                   plimit=stop - limoff, size=size)
                    self.order_tag[o.ref] = 'initial_short'
                    self.pending_entry_order = o
                    self.pending_sl = sl
            return

        if not self.position:
            return

        if self.exit_order is not None and self.exit_order.alive():
            return

        if self.direction == 'long':
            self.extreme_price = max(self.extreme_price, self.data.high[0])
        else:
            self.extreme_price = min(self.extreme_price, self.data.low[0])

        inside = self.or_low <= self.data.close[0] <= self.or_high
        prev_inside = self.or_low <= self.data.close[-1] <= self.or_high
        if inside and prev_inside:
            self._full_exit('consec_inside_exit')
            return

        trail_dist = 1.5 * self.day_atr
        if self.direction == 'long':
            if self.data.close[0] <= self.extreme_price - trail_dist:
                self._full_exit('trail_exit')
                return
        else:
            if self.data.close[0] >= self.extreme_price + trail_dist:
                self._full_exit('trail_exit')
                return

        if (self.trades_today < 4 and self.addon_order is None
                and self.entry_bar_len is not None and len(self) > self.entry_bar_len):
            zone = 0.1 * self.day_atr
            mom_ok = self._momentum_ok(self.direction)
            if self.direction == 'long':
                retest = (self.data.low[0] <= self.or_high + zone
                          and self.data.high[0] >= self.or_high - zone)
                breakout_y = self.yesterday_high is not None and self.data.close[0] > self.yesterday_high
                triggered = mom_ok and (retest or breakout_y)
            else:
                retest = (self.data.low[0] <= self.or_low + zone
                          and self.data.high[0] >= self.or_low - zone)
                breakout_y = self.yesterday_low is not None and self.data.close[0] < self.yesterday_low
                triggered = mom_ok and (retest or breakout_y)

            if triggered and self.tranches:
                addon_size = int(self.tranches[0]['size'] * 0.5)
                if addon_size > 0:
                    self._submit_addon(addon_size)

    # ---------- fill routing ----------

    def _on_initial_fill(self, order, tag):
        self.direction = 'long' if tag == 'initial_long' else 'short'
        entry_price = order.executed.price
        size = abs(order.executed.size)
        self.trades_today = 1
        self.stop_price = self.pending_sl
        self.extreme_price = entry_price
        self.entry_bar_len = len(self)

        if self.direction == 'long':
            target = entry_price + 0.85 * self.or_w
        else:
            target = entry_price - 0.85 * self.or_w

        tr_id = self.next_tranche_id
        self.next_tranche_id += 1
        tp_order = self._submit_tp(tr_id, size, target)
        self.tranches.append(dict(id=tr_id, size=size, entry=entry_price,
                                   target=target, order=tp_order))
        self._resubmit_stop()

    def _on_addon_fill(self, order):
        entry_price = order.executed.price
        size = abs(order.executed.size)
        self.trades_today += 1
        if self.direction == 'long':
            target = entry_price + 0.7 * (self.bb.top[0] - entry_price)
        else:
            target = entry_price - 0.7 * (entry_price - self.bb.bot[0])
        tr_id = self.next_tranche_id
        self.next_tranche_id += 1
        tp_order = self._submit_tp(tr_id, size, target)
        self.tranches.append(dict(id=tr_id, size=size, entry=entry_price,
                                   target=target, order=tp_order))
        self._resubmit_stop()

    def _on_tp_fill(self, tag):
        tr_id = int(tag.split('_')[1])
        self.tranches = [t for t in self.tranches if t['id'] != tr_id]
        if self.position.size == 0:
            if self.stop_order is not None and self.stop_order.alive():
                self.cancel(self.stop_order)
            self.stop_order = None
        else:
            self._resubmit_stop()

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return

        tag = self.order_tag.get(order.ref)

        if order.status == order.Completed:
            if tag in ('initial_long', 'initial_short'):
                self._on_initial_fill(order, tag)
            elif tag == 'addon':
                self._on_addon_fill(order)
            elif tag is not None and tag.startswith('tp_'):
                self._on_tp_fill(tag)
            elif tag == 'stop_loss':
                self.tranches = []
            elif tag in ('trail_exit', 'consec_inside_exit'):
                self.tranches = []

        if order.status in (order.Completed, order.Canceled, order.Margin, order.Rejected):
            self.order_tag.pop(order.ref, None)
            if order is self.pending_entry_order:
                self.pending_entry_order = None
            if order is self.addon_order:
                self.addon_order = None
            if order is self.exit_order:
                self.exit_order = None
            if order is self.stop_order:
                self.stop_order = None
```
