import backtrader as bt


class Strategy(bt.Strategy):
    params = (("longcondition", None),)

    def __init__(self):
        self.longcondition = self.p.longcondition
        self.previous_close = self.data.close(-1)
        self.previousExitPrice = None
        self.order = None

    def next(self):
        if self.order:
            return

        if not self.position:
            if (self.longcondition[0] and
                    (self.previousExitPrice is None or
                     self.previous_close[0] < self.previousExitPrice)):
                self.order = self.order_target_percent(target=1.0)
        elif not self.longcondition[0]:
            self.order = self.close()

    def notify_order(self, order):
        if order.status == order.Completed:
            if order.issell():
                self.previousExitPrice = order.executed.price
            self.order = None
        elif order.status in (order.Canceled, order.Margin, order.Rejected):
            self.order = None
