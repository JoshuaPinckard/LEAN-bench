class HmaAtrReversalStrategy(bt.Strategy):
    params = dict(
        hma_period=9,
        atr_period=14,
        atr_mult=2.0,
    )

    def __init__(self):
        self.hma = bt.indicators.HullMovingAverage(self.data, period=self.p.hma_period)
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.order = None
        self.stop_order = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status == order.Completed:
            if order is self.order:
                self.order = None
                if order.isbuy():
                    stop_price = self.data.low[0] - self.p.atr_mult * self.atr[0]
                else:
                    stop_price = self.data.high[0] + self.p.atr_mult * self.atr[0]
                self.stop_order = self.close(exectype=bt.Order.Stop, price=stop_price)
            elif order is self.stop_order:
                self.stop_order = None
        else:
            if order is self.order:
                self.order = None
            elif order is self.stop_order:
                self.stop_order = None

    def next(self):
        if self.order or self.stop_order or self.position:
            return

        prev_slope = self.hma[-1] - self.hma[-2]
        curr_slope = self.hma[0] - self.hma[-1]

        if prev_slope < 0 and curr_slope > 0:
            self.order = self.buy()
        elif prev_slope > 0 and curr_slope < 0:
            self.order = self.sell()
