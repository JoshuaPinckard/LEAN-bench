import backtrader as bt


class MeanReversionStrategy(bt.Strategy):
    params = (
        ("base_threshold", 2.0),
        ("risk_fraction", 0.01),
    )

    def __init__(self):
        self.bbands = bt.indicators.BollingerBands(
            self.data.close, period=30, devfactor=2
        )
        self.atr = bt.indicators.ATR(self.data, period=14)
        self.atr_ema = bt.indicators.EMA(self.atr, period=50)
        self.rsi = bt.indicators.RSI(self.data.close, period=14)
        self.volume_sma = bt.indicators.SMA(self.data.volume, period=20)

        self.order = None
        self.entry_price = None
        self.entry_bar = None

    def notify_order(self, order):
        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            self.order = None

            if order.status == order.Completed:
                if order.isbuy() and self.position.size == order.executed.size:
                    self.entry_price = order.executed.price
                    self.entry_bar = len(self)
                elif order.issell() and self.position.size == order.executed.size:
                    self.entry_price = order.executed.price
                    self.entry_bar = len(self)

            if not self.position:
                self.entry_price = None
                self.entry_bar = None

    def next(self):
        if self.order:
            return

        mid = self.bbands.mid[0]
        std = (self.bbands.top[0] - mid) / 2.0
        atr = self.atr[0]
        atr_ema = self.atr_ema[0]

        if std == 0 or atr <= 0 or atr_ema <= 0:
            return

        z = (self.data.close[0] - mid) / std
        previous_std = (self.bbands.top[-1] - self.bbands.mid[-1]) / 2.0
        previous_z = (
            (self.data.close[-1] - self.bbands.mid[-1]) / previous_std
            if previous_std != 0
            else 0.0
        )

        vr = atr / atr_ema
        clipped_vr = max(0.7, min(1.6, vr))
        threshold = self.p.base_threshold * clipped_vr

        if self.position:
            time_stop = (
                self.entry_bar is not None and len(self) - self.entry_bar >= 7
            )

            if self.position.size > 0:
                stop_loss = self.data.close[0] <= self.entry_price - (1.5 * atr)
                zero_cross = previous_z < 0 and z >= 0
            else:
                stop_loss = self.data.close[0] >= self.entry_price + (1.5 * atr)
                zero_cross = previous_z > 0 and z <= 0

            if zero_cross or stop_loss or time_stop:
                self.order = self.close()
            return

        risk_amount = self.broker.getvalue() * self.p.risk_fraction
        quantity = risk_amount / (1.5 * atr)

        if (
            z < -threshold
            and self.rsi[0] < 45
            and self.data.volume[0] > self.volume_sma[0]
        ):
            self.order = self.buy(size=quantity)
        elif (
            z > threshold
            and self.rsi[0] > 55
            and self.data.volume[0] > self.volume_sma[0]
        ):
            self.order = self.sell(size=quantity)
