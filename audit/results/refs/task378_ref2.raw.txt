import backtrader as bt


class IntradayVWAPBreakout(bt.Strategy):
    params = (
        ('period', 20),
        ('stop_pct', 0.005),
    )

    def __init__(self):
        self.order = None
        self.stop_order = None

        pv = self.data.close * self.data.volume
        sum_pv = bt.indicators.SumN(pv, period=self.p.period)
        sum_vol = bt.indicators.SumN(self.data.volume, period=self.p.period)
        self.vwap20 = sum_pv / sum_vol
        self.vwap_slope = self.vwap20 - self.vwap20(-1)
        self.sma20 = bt.indicators.SMA(self.data.close, period=self.p.period)

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            if order == self.order:
                self.order = None
            elif order == self.stop_order:
                self.stop_order = None
        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            if order == self.order:
                self.order = None
            elif order == self.stop_order:
                self.stop_order = None

    def next(self):
        if self.order:
            return

        if not self.position:
            if self.stop_order:
                self.cancel(self.stop_order)
                self.stop_order = None

            cross_above_vwap = (self.data.close[0] > self.vwap20[0]) and (self.data.close[-1] <= self.vwap20[-1])
            cross_below_vwap = (self.data.close[0] < self.vwap20[0]) and (self.data.close[-1] >= self.vwap20[-1])

            if cross_above_vwap and (self.vwap_slope[0] > 0) and (self.data.close[0] > self.sma20[0]):
                stop_price = self.data.close[0] * (1.0 - self.p.stop_pct)
                orders = self.buy_bracket(stopprice=stop_price)
                self.order = orders[0]
                self.stop_order = orders[-1]

            elif cross_below_vwap and (self.vwap_slope[0] < 0) and (self.data.close[0] < self.sma20[0]):
                stop_price = self.data.close[0] * (1.0 + self.p.stop_pct)
                orders = self.sell_bracket(stopprice=stop_price)
                self.order = orders[0]
                self.stop_order = orders[-1]

        else:
            if self.position.size > 0:
                cross_below_sma = (self.data.close[0] < self.sma20[0]) and (self.data.close[-1] >= self.sma20[-1])
                if cross_below_sma:
                    if self.stop_order:
                        self.cancel(self.stop_order)
                        self.stop_order = None
                    self.order = self.close()

            elif self.position.size < 0:
                cross_above_sma = (self.data.close[0] > self.sma20[0]) and (self.data.close[-1] <= self.sma20[-1])
                if cross_above_sma:
                    if self.stop_order:
                        self.cancel(self.stop_order)
                        self.stop_order = None
                    self.order = self.close()
