import backtrader as bt


class StochDonchianBreakout(bt.Strategy):
    params = (
        ("period", 260),
        ("upper_threshold", 85.0),
        ("lower_threshold", 1.0),
    )

    def __init__(self):
        self.order = None
        stoch = bt.indicators.StochFast(self.data, period=self.p.period)
        self.percK = stoch.percK

    def notify_order(self, order):
        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            self.order = None

    def next(self):
        if self.order:
            return

        if not self.position:
            if (
                self.percK[0] > self.p.upper_threshold
                and self.percK[-1] <= self.p.upper_threshold
            ):
                cash = self.broker.getcash()
                size = int(cash / self.data.close[0])
                if size > 0:
                    self.order = self.buy(size=size)
        else:
            if self.percK[0] < self.p.lower_threshold:
                self.order = self.close()
