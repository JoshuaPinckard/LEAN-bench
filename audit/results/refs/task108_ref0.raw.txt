import backtrader as bt


class CorrectionOrReversalStrategy(bt.Strategy):
    params = dict(
        sma_period=50,
        range_period=20,
        range_lag=5,
        vol_avg_period=20,
        rebound_pct=0.06,
        signal_window=15,
        key_level_short=175.0,
        target_long=250.0,
        target_short_1=3350.0,
        target_short_2=2700.0,
        stop_buffer=0.01,
        risk_pct=0.02,
    )

    def __init__(self):
        self.sma = bt.indicators.SMA(self.data.close, period=self.p.sma_period)
        self.range_high = bt.indicators.Highest(self.data.high, period=self.p.range_period)
        self.range_low = bt.indicators.Lowest(self.data.low, period=self.p.range_period)
        self.vol_avg = bt.indicators.SMA(self.data.volume, period=self.p.vol_avg_period)

        self.order = None
        self.stop_price = None

        self.breakdown_price = None
        self.breakdown_bar = None
        self.breakdown_volume = None
        self.anchor_confirmed = False
        self.support_level = None

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        # indicator[-lag] needs lag extra bars beyond the indicator's own minperiod;
        # backtrader's automatic minperiod tracking does not see this dynamic offset.
        if len(self) <= self.p.range_period + self.p.range_lag:
            return

        est_high = self.range_high[-self.p.range_lag]
        est_low = self.range_low[-self.p.range_lag]

        if self.data.low[0] < est_low and self.data.volume[0] > self.vol_avg[0]:
            self.breakdown_price = self.data.low[0]
            self.breakdown_bar = len(self)
            self.breakdown_volume = self.data.volume[0]
            self.anchor_confirmed = False

        if (self.breakdown_price is not None and not self.anchor_confirmed
                and self.data.close[0] >= self.breakdown_price * (1 + self.p.rebound_pct)):
            self.anchor_confirmed = True
            self.support_level = self.breakdown_price

        if self.order:
            return

        if not self.position:
            if self._long_signal(est_high, est_low):
                value = self.broker.getvalue()
                size = (value * self.p.risk_pct) / self.data.close[0]
                self.stop_price = min(self.data.low[-1], self.support_level) * (1 - self.p.stop_buffer)
                self.order = self.buy(size=size)
            elif self._short_signal(est_high, est_low):
                value = self.broker.getvalue()
                size = (value * self.p.risk_pct) / self.data.close[0]
                self.stop_price = max(self.p.key_level_short, est_high) * (1 + self.p.stop_buffer)
                self.order = self.sell(size=size)

        elif self.position.size > 0:
            if (self.data.low[0] <= self.stop_price
                    or self.data.close[0] >= self.p.target_long
                    or self.data.close[0] < self.sma[0]
                    or self.data.close[0] < self.data.low[-1]):
                self.order = self.close()

        elif self.position.size < 0:
            if (self.data.high[0] >= self.stop_price
                    or self.data.close[0] >= self.p.target_short_2
                    or self.data.close[0] >= self.p.target_short_1
                    or self.data.close[0] > est_high):
                self.order = self.close()

    def _long_signal(self, est_high, est_low):
        if self.support_level is None or not self.anchor_confirmed:
            return False
        if len(self) - self.breakdown_bar > self.p.signal_window:
            return False
        in_range = est_low <= self.data.close[0] <= est_high
        trend_intact = self.data.close[0] > self.sma[0]
        bounced = self.data.close[0] > self.support_level
        buying_vol_rising = self.data.volume[0] > self.data.volume[-1]
        sell_vol_declining = self.data.volume[0] < self.breakdown_volume
        return in_range and trend_intact and bounced and buying_vol_rising and sell_vol_declining

    def _short_signal(self, est_high, est_low):
        returned_to_range = est_low <= self.data.close[0] <= est_high
        selling_resumed = (self.data.volume[0] > self.vol_avg[0]
                            and self.data.close[0] < self.data.open[0])
        support_broken = self.data.close[0] < self.data.low[-1]
        below_key_level = (self.data.close[0] < self.p.key_level_short
                            and self.data.close[-1] < self.p.key_level_short)
        return returned_to_range and selling_resumed and support_broken and below_key_level
