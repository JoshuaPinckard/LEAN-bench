import backtrader as bt


class MeanReversionStrategy(bt.Strategy):
    params = (
        ('rsi_period', 14),
        ('band_period', 20),
    )

    def __init__(self):
        self.rsi = bt.indicators.RSI(self.data.close, period=self.p.rsi_period)
        self.rsi_sma = bt.indicators.SMA(self.rsi, period=self.p.band_period)
        self.rsi_std = bt.indicators.StandardDeviation(
            self.rsi, period=self.p.band_period
        )
        self.upper_band = self.rsi_sma + 2 * self.rsi_std
        self.lower_band = self.rsi_sma - 2 * self.rsi_std

        self.order = None
        self.stop_order = None
        self.entry_direction = None
        self.was_below_lower = False
        self.was_above_upper = False

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled,
                            order.Margin, order.Rejected]:
            if order == self.order:
                self.order = None

            if order == self.stop_order:
                self.stop_order = None

            if order.status == order.Completed:
                if order.isbuy() and self.entry_direction == 'long':
                    self.stop_order = self.sell(
                        exectype=bt.Order.Stop,
                        price=order.executed.price * 0.98,
                    )
                elif order.issell() and self.entry_direction == 'short':
                    self.stop_order = self.buy(
                        exectype=bt.Order.Stop,
                        price=order.executed.price * 1.02,
                    )

            if not self.position and order == self.stop_order:
                self.entry_direction = None

    def next(self):
        if self.order:
            return

        rsi = self.rsi[0]
        rsi_sma = self.rsi_sma[0]

        crosses_up_sma = self.rsi[-1] <= self.rsi_sma[-1] and rsi > rsi_sma
        crosses_down_sma = self.rsi[-1] >= self.rsi_sma[-1] and rsi < rsi_sma

        if self.position:
            if ((self.position.size > 0 and crosses_down_sma) or
                    (self.position.size < 0 and crosses_up_sma)):
                if self.stop_order:
                    self.cancel(self.stop_order)
                self.order = self.close()
                self.entry_direction = None
            return

        if rsi < self.lower_band[0]:
            self.was_below_lower = True

        if rsi > self.upper_band[0]:
            self.was_above_upper = True

        if self.was_below_lower and crosses_up_sma:
            self.order = self.buy()
            self.entry_direction = 'long'
            self.was_below_lower = False
            self.was_above_upper = False
        elif self.was_above_upper and crosses_down_sma:
            self.order = self.sell()
            self.entry_direction = 'short'
            self.was_below_lower = False
            self.was_above_upper = False
