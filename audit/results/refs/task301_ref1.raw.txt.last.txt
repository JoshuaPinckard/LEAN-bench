import backtrader as bt


class IntradayMeanReversionStrategy(bt.Strategy):
    params = (
        ('ma_type', 'sma'),
    )

    def __init__(self):
        self.atr = bt.indicators.ATR(self.data, period=14)
        if self.p.ma_type.lower() == 'ema':
            self.ma = bt.indicators.EMA(self.data.close, period=20)
        else:
            self.ma = bt.indicators.SMA(self.data.close, period=20)

        self.cross = bt.indicators.CrossOver(self.data.close, self.ma)
        self.order = None
        self.reference = None
        self.reference_date = None
        self.entry_price = None

    def prenext(self):
        self._update_reference()

    def next(self):
        self._update_reference()

        if self.order:
            return

        if self.position:
            if self.position.size > 0:
                if self.cross[0] > 0 or self.data.close[0] <= self.entry_price * 0.98:
                    self.order = self.close()
            else:
                if self.cross[0] < 0 or self.data.close[0] >= self.entry_price * 1.02:
                    self.order = self.close()
            return

        if self.data.close[0] > self.reference + 2 * self.atr[0]:
            self.order = self.sell()
        elif self.data.close[0] < self.reference - 2 * self.atr[0]:
            self.order = self.buy()

    def _update_reference(self):
        current_date = self.data.datetime.date(0)
        if current_date != self.reference_date:
            self.reference_date = current_date
            self.reference = self.data.open[0]

    def notify_order(self, order):
        if order.status == order.Completed:
            if order.isbuy() and self.position.size > 0:
                self.entry_price = order.executed.price
            elif order.issell() and self.position.size < 0:
                self.entry_price = order.executed.price
            elif not self.position:
                self.entry_price = None
            self.order = None
        elif order.status in (order.Canceled, order.Margin, order.Rejected):
            self.order = None
