import backtrader as bt


class SARAtrStopReverse(bt.Strategy):
    params = (("atr_period", 14), ("multiplier", 2.5))

    def __init__(self):
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.order = None
        self.stop_order = None
        self.pending_reverse = None

        self.prev_up_stop = None
        self.prev_down_stop = None
        self.prev_sar_atr = None
        self.sar_atr = None
        self.prev_close = None

    def _cancel_stop(self):
        if self.stop_order is not None:
            self.cancel(self.stop_order)
            self.stop_order = None

    def _submit_entry(self, direction):
        self.pending_reverse = direction
        if direction > 0:
            self.order = self.buy()
        else:
            self.order = self.sell()

    def next(self):
        close = self.data.close[0]
        atr = self.atr[0]

        if self.prev_sar_atr is None:
            self.prev_up_stop = close - self.p.multiplier * atr
            self.prev_down_stop = close + self.p.multiplier * atr
            self.sar_atr = self.prev_up_stop
            self.prev_sar_atr = self.sar_atr
            self.prev_close = close
            return

        up_stop = max(
            self.prev_up_stop,
            close - self.p.multiplier * atr,
        )
        down_stop = min(
            self.prev_down_stop,
            close + self.p.multiplier * atr,
        )
        self.sar_atr = up_stop if close > self.prev_sar_atr else down_stop

        crossover = 0
        if self.prev_close <= self.prev_sar_atr and close > self.sar_atr:
            crossover = 1
        elif self.prev_close >= self.prev_sar_atr and close < self.sar_atr:
            crossover = -1

        if self.order is None:
            if not self.position:
                if crossover > 0:
                    self._submit_entry(1)
                elif crossover < 0:
                    self._submit_entry(-1)
            elif self.position.size > 0 and crossover < 0:
                self._cancel_stop()
                self.close()
                self._submit_entry(-1)
            elif self.position.size < 0 and crossover > 0:
                self._cancel_stop()
                self.close()
                self._submit_entry(1)

        self.prev_up_stop = up_stop
        self.prev_down_stop = down_stop
        self.prev_sar_atr = self.sar_atr
        self.prev_close = close

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order == self.stop_order:
            if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
                self.stop_order = None
            return

        if order.status == order.Completed:
            if order.isbuy() and self.position.size > 0:
                self.stop_order = self.sell(
                    exectype=bt.Order.Stop,
                    price=self.sar_atr,
                )
            elif order.issell() and self.position.size < 0:
                self.stop_order = self.buy(
                    exectype=bt.Order.Stop,
                    price=self.sar_atr,
                )

        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            if order == self.order:
                self.order = None
