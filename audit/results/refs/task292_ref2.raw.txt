import math
import backtrader as bt


class SmaCrossoverIntraday(bt.Strategy):
    def __init__(self):
        self.sma10 = bt.indicators.SMA(self.data.close, period=10)
        self.sma20 = bt.indicators.SMA(self.data.close, period=20)
        self.atr = bt.indicators.ATR(self.data, period=14)
        self.crossover = bt.indicators.CrossOver(self.sma10, self.sma20)

        self.order = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return
        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            self.order = None

    def next(self):
        if self.order:
            return

        t = self.data.datetime.time(0)
        t_min = t.hour * 60 + t.minute

        if self.position:
            if t_min >= 955:
                self.close()
                return

            entry_price = self.position.price
            if self.position.size > 0:
                if (
                    self.crossover[0] < 0
                    or self.data.low[0] <= entry_price * 0.99
                ):
                    self.close()
            elif self.position.size < 0:
                if (
                    self.crossover[0] > 0
                    or self.data.high[0] >= entry_price * 1.01
                ):
                    self.close()
            return

        if 575 <= t_min < 930:
            dist = abs(self.data.close[0] - self.sma20[0])
            if dist <= 0.5 * self.atr[0]:
                equity = self.broker.getvalue()
                close_price = self.data.close[0]
                qty = math.floor((equity * 0.01) / (close_price * 0.01))
                if qty > 0:
                    if self.crossover[0] > 0:
                        self.order = self.buy(size=qty)
                    elif self.crossover[0] < 0:
                        self.order = self.sell(size=qty)
