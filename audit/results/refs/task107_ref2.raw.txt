import backtrader as bt


class CombinedMomentumStrategy(bt.Strategy):
    params = (
        ("rsi_period", 14),
        ("rsi_lower_entry", 40.0),
        ("rsi_upper_entry", 60.0),
        ("rsi_long_exit", 70.0),
        ("rsi_short_exit", 30.0),
        ("macd_fast", 12),
        ("macd_slow", 26),
        ("macd_signal", 9),
        ("sma_period", 20),
        ("atr_period", 14),
        ("atr_threshold", 0.02),
        ("stop_loss", 0.01),
        ("take_profit", 0.02),
        ("capital_risk_pct", 0.01),
    )

    def __init__(self):
        self.rsi = bt.indicators.RSI(
            self.data.close, period=self.p.rsi_period
        )
        self.macd = bt.indicators.MACD(
            self.data.close,
            period_me1=self.p.macd_fast,
            period_me2=self.p.macd_slow,
            period_signal=self.p.macd_signal,
        )
        self.macd_cross = bt.indicators.CrossOver(
            self.macd.macd, self.macd.signal
        )
        self.sma = bt.indicators.SMA(
            self.data.close, period=self.p.sma_period
        )
        self.momentum = self.data.close - self.sma
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)

        self.order = None
        self.sl_order = None
        self.tp_order = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            if order == self.order:
                self.order = None
                entry_price = order.executed.price
                size = abs(order.executed.size)
                if self.position.size > 0:
                    sl_price = entry_price * (1.0 - self.p.stop_loss)
                    tp_price = entry_price * (1.0 + self.p.take_profit)
                    self.sl_order = self.sell(
                        size=size, exectype=bt.Order.Stop, price=sl_price
                    )
                    self.tp_order = self.sell(
                        size=size, exectype=bt.Order.Limit, price=tp_price
                    )
                elif self.position.size < 0:
                    sl_price = entry_price * (1.0 + self.p.stop_loss)
                    tp_price = entry_price * (1.0 - self.p.take_profit)
                    self.sl_order = self.buy(
                        size=size, exectype=bt.Order.Stop, price=sl_price
                    )
                    self.tp_order = self.buy(
                        size=size, exectype=bt.Order.Limit, price=tp_price
                    )
            elif order in [self.sl_order, self.tp_order]:
                if order == self.sl_order and self.tp_order:
                    self.cancel(self.tp_order)
                elif order == self.tp_order and self.sl_order:
                    self.cancel(self.sl_order)
                self.sl_order = None
                self.tp_order = None

        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            if order == self.order:
                self.order = None
            if order == self.sl_order:
                self.sl_order = None
            if order == self.tp_order:
                self.tp_order = None

    def next(self):
        if self.order:
            return

        if not self.position:
            if self.atr[0] > self.p.atr_threshold * self.data.close[0]:
                return

            rsi_val = self.rsi[0]
            in_neutral_rsi = (
                self.p.rsi_lower_entry <= rsi_val <= self.p.rsi_upper_entry
            )

            if (
                in_neutral_rsi
                and self.macd_cross[0] > 0
                and self.momentum[0] > 0
            ):
                size = (
                    self.broker.getvalue() * self.p.capital_risk_pct
                ) / self.data.close[0]
                if size > 0:
                    self.order = self.buy(size=size)

            elif (
                in_neutral_rsi
                and self.macd_cross[0] < 0
                and self.momentum[0] < 0
            ):
                size = (
                    self.broker.getvalue() * self.p.capital_risk_pct
                ) / self.data.close[0]
                if size > 0:
                    self.order = self.sell(size=size)

        else:
            if self.position.size > 0:
                if (
                    self.rsi[0] > self.p.rsi_long_exit
                    or self.macd_cross[0] < 0
                ):
                    if self.sl_order:
                        self.cancel(self.sl_order)
                        self.sl_order = None
                    if self.tp_order:
                        self.cancel(self.tp_order)
                        self.tp_order = None
                    self.order = self.close()
            elif self.position.size < 0:
                if (
                    self.rsi[0] < self.p.rsi_short_exit
                    or self.macd_cross[0] > 0
                ):
                    if self.sl_order:
                        self.cancel(self.sl_order)
                        self.sl_order = None
                    if self.tp_order:
                        self.cancel(self.tp_order)
                        self.tp_order = None
                    self.order = self.close()
