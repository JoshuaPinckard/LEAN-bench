import backtrader as bt
import math


class MondayFridaySMAStrategy(bt.Strategy):
    params = (("sma_period", 20), ("stop_loss_pct", 0.02),)

    def __init__(self):
        self.sma = bt.indicators.SimpleMovingAverage(
            self.data.close, period=self.p.sma_period
        )
        self.entry_order = None
        self.exit_order = None
        self.stop_order = None

    def _size(self):
        return math.floor(
            self.broker.getcash() / (self.data.close[0] * (1.0 + 0.001))
        )

    def next(self):
        weekday = self.data.datetime.date(0).weekday()

        if self.position:
            should_exit = (
                (self.position.size > 0 and weekday == 4)
                or (self.position.size < 0 and weekday == 0)
            )

            if should_exit and self.exit_order is None:
                if self.stop_order is not None:
                    self.cancel(self.stop_order)
                self.exit_order = self.close()
            return

        if self.entry_order is not None or self.exit_order is not None:
            return

        size = self._size()
        if size <= 0:
            return

        if weekday == 0 and self.data.close[0] > self.sma[0]:
            self.entry_order = self.buy(size=size)
        elif weekday == 4 and self.data.close[0] < self.sma[0]:
            self.entry_order = self.sell(size=size)

    def notify_order(self, order):
        if order.status in (
            order.Submitted,
            order.Accepted,
        ):
            return

        if order is self.entry_order:
            if order.status == order.Completed:
                if order.isbuy():
                    self.stop_order = self.sell(
                        size=order.executed.size,
                        exectype=bt.Order.Stop,
                        price=order.executed.price * (1.0 - self.p.stop_loss_pct),
                    )
                else:
                    self.stop_order = self.buy(
                        size=abs(order.executed.size),
                        exectype=bt.Order.Stop,
                        price=order.executed.price * (1.0 + self.p.stop_loss_pct),
                    )
            if order.status in (
                order.Completed,
                order.Canceled,
                order.Margin,
                order.Rejected,
            ):
                self.entry_order = None

        if order is self.exit_order and order.status in (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            self.exit_order = None

        if order is self.stop_order and order.status in (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            self.stop_order = None
