import backtrader as bt


class IntradayMomentumODTEStrategy(bt.Strategy):
    params = (
        ('atr_period', 14),
        ('atr_sma_period', 20),
        ('sma_period', 5),
        ('ema_period', 9),
        ('dip_pct', 0.005),
        ('stoch_oversold', 20.0),
        ('entry_window_minutes', 60),
        ('tp_atr_mult', 2.5),
        ('sl_atr_mult', 1.0),
        ('session_end_hour', 15),
        ('session_end_minute', 55),
    )

    def __init__(self):
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.atr_sma = bt.indicators.SMA(self.atr, period=self.p.atr_sma_period)
        self.sma5 = bt.indicators.SMA(self.data.close, period=self.p.sma_period)
        self.macd = bt.indicators.MACDHisto(self.data)
        self.ema9 = bt.indicators.EMA(self.data.close, period=self.p.ema_period)
        self.stoch = bt.indicators.Stochastic(self.data)
        self.stoch_cross = bt.indicators.CrossOver(self.stoch.percK, self.stoch.percD)
        self.price_cross_ema9 = bt.indicators.CrossOver(self.data.close, self.ema9)

        self.order = None
        self.entry_price = None
        self.entry_atr = None
        self.day_open = None
        self.day_open_dt = None

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return
        if order.status == order.Completed and order.isbuy():
            self.entry_price = order.executed.price
            self.entry_atr = self.atr[0]
        if order.status in (order.Completed, order.Canceled, order.Margin, order.Rejected):
            self.order = None

    def next(self):
        cur_date = self.data.datetime.date(0)
        if self.day_open_dt is None or cur_date != self.data.datetime.date(-1):
            self.day_open = self.data.open[0]
            self.day_open_dt = self.data.datetime.datetime(0)

        if self.order:
            return

        cur_time = self.data.datetime.time(0)
        session_end = (cur_time.hour == self.p.session_end_hour and
                        cur_time.minute == self.p.session_end_minute)

        if self.position:
            if session_end:
                self.order = self.close()
                return
            tp_price = self.entry_price + self.p.tp_atr_mult * self.entry_atr
            sl_price = self.entry_price - self.p.sl_atr_mult * self.entry_atr
            if self.data.high[0] >= tp_price or self.data.low[0] <= sl_price:
                self.order = self.close()
            return

        vol_filter = self.atr[0] > self.atr_sma[0]
        dir_filter = (self.data.close[0] > self.sma5[0]) or (self.macd.histo[0] > 0)
        filters_ok = vol_filter and dir_filter

        elapsed_minutes = (self.data.datetime.datetime(0) - self.day_open_dt).total_seconds() / 60.0

        if elapsed_minutes == 0:
            if filters_ok:
                self.order = self.buy(size=1)
            return

        if elapsed_minutes <= self.p.entry_window_minutes:
            dip = (self.day_open - self.data.close[0]) / self.day_open >= self.p.dip_pct
            reversal = ((self.stoch_cross[0] > 0 and
                         self.stoch.percK[-1] < self.p.stoch_oversold and
                         self.stoch.percD[-1] < self.p.stoch_oversold) or
                        (self.price_cross_ema9[0] > 0))
            if dip and reversal and filters_ok:
                self.order = self.buy(size=1)
