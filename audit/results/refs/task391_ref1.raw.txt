import backtrader as bt


class MACDIntradayTrendStrategy(bt.Strategy):
    params = (
        ('fast', 12),
        ('slow', 26),
        ('signal', 9),
    )

    def __init__(self):
        macd = bt.indicators.MACD(
            self.data.close,
            period_me1=self.p.fast,
            period_me2=self.p.slow,
            period_signal=self.p.signal,
        )
        self.crossover = bt.indicators.CrossOver(macd.macd, macd.signal)
        self.order = None
        self.stop_order = None

    def next(self):
        if self.order:
            return

        if not self.position:
            if self.crossover > 0:
                self.order = self.buy()
            elif self.crossover < 0:
                self.order = self.sell()
        elif ((self.position.size > 0 and self.crossover < 0) or
              (self.position.size < 0 and self.crossover > 0)):
            if self.stop_order:
                self.cancel(self.stop_order)
                self.stop_order = None
            self.order = self.close()

    def notify_order(self, order):
        if order.status in [order.Completed]:
            if order is self.order:
                if order.isbuy():
                    self.stop_order = self.sell(
                        exectype=bt.Order.Stop,
                        price=order.executed.price * 0.98,
                    )
                elif order.issell() and self.position.size < 0:
                    self.stop_order = self.buy(
                        exectype=bt.Order.Stop,
                        price=order.executed.price * 1.02,
                    )
                self.order = None
            elif order is self.stop_order:
                self.stop_order = None

        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            if order is self.order:
                self.order = None
            elif order is self.stop_order:
                self.stop_order = None
