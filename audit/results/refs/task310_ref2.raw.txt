import math
import statistics
import backtrader as bt


class HurstExponent(bt.Indicator):
    lines = ('hurst',)
    params = (('period', 100),)

    def __init__(self):
        self.addminperiod(self.p.period)

    def next(self):
        prices = self.data.get(size=self.p.period)
        if len(prices) < self.p.period:
            self.lines.hurst[0] = 0.5
            return

        returns = [math.log(prices[i] / prices[i - 1]) for i in range(1, len(prices))]
        n_total = len(returns)
        sub_lens = [8, 16, 32, 64]
        log_n = []
        log_rs = []

        for n in sub_lens:
            num_chunks = n_total // n
            if num_chunks < 1:
                continue
            rs_list = []
            for i in range(num_chunks):
                chunk = returns[i * n : (i + 1) * n]
                mean_val = statistics.mean(chunk)
                cum_dev = []
                curr = 0.0
                for x in chunk:
                    curr += x - mean_val
                    cum_dev.append(curr)
                r = max(cum_dev) - min(cum_dev)
                s = statistics.stdev(chunk) if len(chunk) > 1 else 0.0
                if s > 1e-12:
                    rs_list.append(r / s)
            if rs_list:
                avg_rs = statistics.mean(rs_list)
                if avg_rs > 0:
                    log_n.append(math.log(n))
                    log_rs.append(math.log(avg_rs))

        if len(log_n) >= 2:
            mean_x = statistics.mean(log_n)
            mean_y = statistics.mean(log_rs)
            var_x = sum((x - mean_x) ** 2 for x in log_n)
            cov_xy = sum((x - mean_x) * (y - mean_y) for x, y in zip(log_n, log_rs))
            self.lines.hurst[0] = cov_xy / var_x if var_x > 0 else 0.5
        else:
            self.lines.hurst[0] = 0.5


class IntradayHurstPSARStrategy(bt.Strategy):
    params = (
        ('hurst_period', 100),
        ('hurst_threshold', 0.55),
        ('sar_af', 0.02),
        ('sar_maxaf', 0.2),
        ('atr_period', 14),
        ('atr_mult', 2.0),
        ('risk_pct', 0.01),
    )

    def __init__(self):
        self.hurst = HurstExponent(self.data, period=self.p.hurst_period)
        self.psar = bt.indicators.ParabolicSAR(
            self.data, af=self.p.sar_af, maxaf=self.p.sar_maxaf
        )
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.order = None

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def _calc_size(self, price, stop_price):
        equity = self.broker.getvalue()
        risk_amt = equity * self.p.risk_pct
        dist = abs(price - stop_price)
        if dist < 1e-5:
            dist = self.p.atr_mult * self.atr[0]
        if dist < 1e-5:
            dist = price * 0.01
        size = int(risk_amt / dist)
        max_size = int(equity / price) if price > 0 else 0
        return max(0, min(size, max_size))

    def next(self):
        if self.order:
            return

        cross_above = (self.data.close[-1] < self.psar[-1]) and (self.data.close[0] > self.psar[0])
        cross_below = (self.data.close[-1] > self.psar[-1]) and (self.data.close[0] < self.psar[0])
        hurst_ok = self.hurst[0] > self.p.hurst_threshold

        if not self.position:
            if hurst_ok:
                if cross_above:
                    price = self.data.close[0]
                    stop_price = max(self.psar[0], price - self.p.atr_mult * self.atr[0])
                    size = self._calc_size(price, stop_price)
                    if size > 0:
                        self.order = self.buy(size=size)
                elif cross_below:
                    price = self.data.close[0]
                    stop_price = min(self.psar[0], price + self.p.atr_mult * self.atr[0])
                    size = self._calc_size(price, stop_price)
                    if size > 0:
                        self.order = self.sell(size=size)

        elif self.position.size > 0:
            entry_price = self.position.price
            stop_price = max(self.psar[0], entry_price - self.p.atr_mult * self.atr[0])
            if self.data.low[0] <= stop_price:
                self.order = self.close()
            elif cross_below:
                if hurst_ok:
                    price = self.data.close[0]
                    short_stop = min(self.psar[0], price + self.p.atr_mult * self.atr[0])
                    short_size = self._calc_size(price, short_stop)
                    total_size = self.position.size + short_size
                    self.order = self.sell(size=total_size)
                else:
                    self.order = self.close()

        elif self.position.size < 0:
            entry_price = self.position.price
            stop_price = min(self.psar[0], entry_price + self.p.atr_mult * self.atr[0])
            if self.data.high[0] >= stop_price:
                self.order = self.close()
            elif cross_above:
                if hurst_ok:
                    price = self.data.close[0]
                    long_stop = max(self.psar[0], price - self.p.atr_mult * self.atr[0])
                    long_size = self._calc_size(price, long_stop)
                    total_size = abs(self.position.size) + long_size
                    self.order = self.buy(size=total_size)
                else:
                    self.order = self.close()
