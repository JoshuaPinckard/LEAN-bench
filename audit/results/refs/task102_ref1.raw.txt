import backtrader as bt


class TradingFairValueGaps(bt.Strategy):
    params = (
        ("stop_offset", 1.0),
        ("volume_lookback", 20),
        ("volume_multiplier", 1.5),
    )

    def __init__(self):
        self.entry_order = None
        self.stop_order = None
        self.target_order = None
        self.entry_fvg = None
        self.active_fvgs = []
        self.volumes = []
        self.session_date = None
        self.session_high = None
        self.session_low = None
        self.previous_session_high = None
        self.previous_session_low = None

    def _in_ny_session(self):
        now = self.data.datetime.datetime(0)
        current = now.hour * 60 + now.minute
        return 9 * 60 + 30 <= current <= 16 * 60

    def _update_session_levels(self):
        now = self.data.datetime.datetime(0)
        date = now.date()

        if self.session_date != date:
            if self.session_date is not None:
                self.previous_session_high = self.session_high
                self.previous_session_low = self.session_low
            self.session_date = date
            self.session_high = None
            self.session_low = None
            self.active_fvgs = []

        high = self.data.high[0]
        low = self.data.low[0]
        self.session_high = high if self.session_high is None else max(self.session_high, high)
        self.session_low = low if self.session_low is None else min(self.session_low, low)

    def _reversal_signal(self):
        open_ = self.data.open[0]
        high = self.data.high[0]
        low = self.data.low[0]
        close = self.data.close[0]

        body = abs(close - open_)
        candle_range = high - low
        if candle_range <= 0:
            return False

        effective_body = max(body, 1e-12)
        upper_wick = high - max(open_, close)
        lower_wick = min(open_, close) - low

        hammer = lower_wick >= 2 * effective_body and upper_wick <= effective_body
        shooting_star = upper_wick >= 2 * effective_body and lower_wick <= effective_body
        long_wick = upper_wick >= 2 * effective_body or lower_wick >= 2 * effective_body

        volume_spike = False
        if len(self.volumes) >= self.p.volume_lookback:
            average_volume = sum(self.volumes[-self.p.volume_lookback:]) / self.p.volume_lookback
            volume_spike = average_volume > 0 and self.data.volume[0] >= (
                average_volume * self.p.volume_multiplier
            )

        return hammer or shooting_star or long_wick or volume_spike

    def _create_fvg(self):
        if len(self) < 3 or not self._in_ny_session():
            return

        first_dt = self.data.datetime.datetime(-2)
        second_dt = self.data.datetime.datetime(-1)
        third_dt = self.data.datetime.datetime(0)

        if (
            first_dt.date() != third_dt.date()
            or second_dt.date() != third_dt.date()
            or not (9 * 60 + 30 <= first_dt.hour * 60 + first_dt.minute <= 16 * 60)
            or not (9 * 60 + 30 <= second_dt.hour * 60 + second_dt.minute <= 16 * 60)
        ):
            return

        first_high = self.data.high[-2]
        first_low = self.data.low[-2]
        third_high = self.data.high[0]
        third_low = self.data.low[0]

        if third_low > first_high:
            self.active_fvgs.append(
                {
                    "side": "bullish",
                    "lower": first_high,
                    "upper": third_low,
                    "date": third_dt.date(),
                }
            )
        elif third_high < first_low:
            self.active_fvgs.append(
                {
                    "side": "bearish",
                    "lower": third_high,
                    "upper": first_low,
                    "date": third_dt.date(),
                }
            )

    def _mitigate_fvgs(self):
        close = self.data.close[0]
        remaining = []

        for fvg in self.active_fvgs:
            if fvg["side"] == "bullish":
                mitigated = close <= fvg["upper"]
            else:
                mitigated = close >= fvg["lower"]

            if not mitigated:
                remaining.append(fvg)

        self.active_fvgs = remaining

    def _enter_from_fvg(self):
        if (
            not self._in_ny_session()
            or self.position
            or self.entry_order is not None
            or self.stop_order is not None
            or self.target_order is not None
            or not self._reversal_signal()
        ):
            return

        high = self.data.high[0]
        low = self.data.low[0]
        current_date = self.data.datetime.datetime(0).date()

        for fvg in reversed(self.active_fvgs):
            if fvg["date"] != current_date:
                continue

            retraced_into_gap = low <= fvg["upper"] and high >= fvg["lower"]
            if not retraced_into_gap:
                continue

            if fvg["side"] == "bullish" and self.previous_session_high is not None:
                self.entry_fvg = fvg
                self.entry_order = self.buy()
                return

            if fvg["side"] == "bearish" and self.previous_session_low is not None:
                self.entry_fvg = fvg
                self.entry_order = self.sell()
                return

    def next(self):
        if not self._in_ny_session():
            return

        self._update_session_levels()
        self._mitigate_fvgs()
        self._enter_from_fvg()
        self._create_fvg()

        self.volumes.append(self.data.volume[0])

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return

        if order is self.entry_order:
            self.entry_order = None

            if order.status == order.Completed and self.entry_fvg is not None:
                size = abs(order.executed.size)

                if self.entry_fvg["side"] == "bullish":
                    self.stop_order = self.sell(
                        size=size,
                        exectype=bt.Order.Stop,
                        price=self.entry_fvg["lower"] - self.p.stop_offset,
                    )
                    self.target_order = self.sell(
                        size=size,
                        exectype=bt.Order.Limit,
                        price=self.previous_session_high,
                    )
                else:
                    self.stop_order = self.buy(
                        size=size,
                        exectype=bt.Order.Stop,
                        price=self.entry_fvg["upper"] + self.p.stop_offset,
                    )
                    self.target_order = self.buy(
                        size=size,
                        exectype=bt.Order.Limit,
                        price=self.previous_session_low,
                    )
            return

        if order is self.stop_order:
            if order.status in (order.Completed, order.Canceled, order.Margin, order.Rejected):
                self.stop_order = None
                if order.status == order.Completed and self.target_order is not None:
                    self.cancel(self.target_order)
            return

        if order is self.target_order:
            if order.status in (order.Completed, order.Canceled, order.Margin, order.Rejected):
                self.target_order = None
                if order.status == order.Completed and self.stop_order is not None:
                    self.cancel(self.stop_order)
