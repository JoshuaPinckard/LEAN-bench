import backtrader as bt


class IntraRangeMomentumTripleBarriers(bt.Strategy):
    params = (
        ("sma_period", 20),
        ("stoch_period", 14),
        ("stoch_dfast", 3),
        ("stoch_dslow", 3),
        ("bb_period", 20),
        ("bb_devfactor", 2),
        ("atr_period", 14),
        ("risk_fraction", 0.01),
        ("range_threshold", 0.002),
        ("atr_threshold", 0.0015),
        ("time_stop_bars", 20),
    )

    def __init__(self):
        self.broker.setcommission(commission=0.0002)
        self.broker.set_slippage_perc(perc=0.001)

        self.sma = bt.indicators.SMA(self.data.close, period=self.p.sma_period)
        self.stoch = bt.indicators.Stochastic(
            self.data,
            period=self.p.stoch_period,
            period_dfast=self.p.stoch_dfast,
            period_dslow=self.p.stoch_dslow,
        )
        self.bbands = bt.indicators.BollingerBands(
            self.data.close,
            period=self.p.bb_period,
            devfactor=self.p.bb_devfactor,
        )
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.stoch_cross = bt.indicators.CrossOver(
            self.stoch.percK, self.stoch.percD
        )
        self.price_sma_cross = bt.indicators.CrossOver(self.data.close, self.sma)

        self.order = None
        self.entry_order = None
        self.entry_bar = None

    def notify_order(self, order):
        if order.status == order.Completed:
            if order is self.entry_order:
                self.entry_bar = len(self.data)

        if order.status in (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            if order is self.entry_order and order.status != order.Completed:
                self.entry_order = None
            if order is self.order:
                self.order = None

    def next(self):
        if self.order:
            return

        close = self.data.close[0]
        daily_range_ratio = (self.data.high[0] - self.data.low[0]) / close
        atr_ratio = self.atr[0] / close
        range_and_volatility_ok = (
            daily_range_ratio >= self.p.range_threshold
            and atr_ratio > self.p.atr_threshold
        )

        if not self.position:
            size = (self.broker.getvalue() * self.p.risk_fraction) / close

            if (
                close > self.sma[0]
                and self.stoch.percK[0] > 66
                and self.stoch_cross[0] > 0
                and range_and_volatility_ok
            ):
                self.order = self.buy(size=size)
                self.entry_order = self.order

            elif (
                close < self.sma[0]
                and self.stoch.percK[0] < 33
                and self.stoch_cross[0] < 0
                and range_and_volatility_ok
            ):
                self.order = self.sell(size=size)
                self.entry_order = self.order

            return

        time_stop = (
            self.entry_bar is not None
            and len(self.data) - self.entry_bar >= self.p.time_stop_bars
        )

        if self.position.size > 0:
            if (
                self.data.high[0] >= self.bbands.top[0]
                or self.data.low[0] <= self.bbands.bot[0]
                or self.price_sma_cross[0] < 0
                or time_stop
            ):
                self.order = self.close()

        else:
            if (
                self.data.low[0] <= self.bbands.bot[0]
                or self.data.high[0] >= self.bbands.top[0]
                or self.price_sma_cross[0] > 0
                or time_stop
            ):
                self.order = self.close()
