import backtrader as bt


class IntradayVWAP(bt.Indicator):
    lines = ('vwap',)

    def __init__(self):
        self.addminperiod(1)
        self.cum_tp_vol = 0.0
        self.cum_vol = 0.0

    def next(self):
        if len(self) == 1 or self.data.datetime.date(0) != self.data.datetime.date(-1):
            self.cum_tp_vol = 0.0
            self.cum_vol = 0.0

        tp = (self.data.high[0] + self.data.low[0] + self.data.close[0]) / 3.0
        vol = self.data.volume[0]

        self.cum_tp_vol += tp * vol
        self.cum_vol += vol

        if self.cum_vol > 0:
            self.vwap[0] = self.cum_tp_vol / self.cum_vol
        else:
            self.vwap[0] = tp


class IntradayVWAPStrategy(bt.Strategy):
    params = (
        ('atr_period', 14),
        ('atr_mult', 2.0),
        ('stop_pct', 0.01),
        ('use_atr_stop', True),
        ('use_pct_stop', True),
        ('stake', 100),
    )

    def __init__(self):
        self.vwap = IntradayVWAP(self.data)
        self.crossover = bt.ind.CrossOver(self.data.close, self.vwap)
        self.atr = bt.ind.ATR(self.data, period=self.p.atr_period)
        self.order = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        if self.order:
            return

        pos_size = self.position.size

        if pos_size > 0:  # LONG position
            # Exit LONG signal: close crosses below VWAP
            if self.crossover[0] < 0:
                self.order = self.close()
                return

            # Risk Management Stop Loss check
            entry_price = self.position.price
            atr_stop = entry_price - self.p.atr_mult * self.atr[0]
            pct_stop = entry_price * (1.0 - self.p.stop_pct)

            if self.p.use_atr_stop and self.p.use_pct_stop:
                stop_level = max(atr_stop, pct_stop)
            elif self.p.use_atr_stop:
                stop_level = atr_stop
            elif self.p.use_pct_stop:
                stop_level = pct_stop
            else:
                stop_level = None

            if stop_level is not None and self.data.low[0] <= stop_level:
                self.order = self.close()
                return

        elif pos_size < 0:  # SHORT position
            # Exit SHORT signal: close crosses above VWAP
            if self.crossover[0] > 0:
                self.order = self.close()
                return

            # Risk Management Stop Loss check
            entry_price = self.position.price
            atr_stop = entry_price + self.p.atr_mult * self.atr[0]
            pct_stop = entry_price * (1.0 + self.p.stop_pct)

            if self.p.use_atr_stop and self.p.use_pct_stop:
                stop_level = min(atr_stop, pct_stop)
            elif self.p.use_atr_stop:
                stop_level = atr_stop
            elif self.p.use_pct_stop:
                stop_level = pct_stop
            else:
                stop_level = None

            if stop_level is not None and self.data.high[0] >= stop_level:
                self.order = self.close()
                return

        else:  # No open position
            if self.crossover[0] > 0:
                self.order = self.buy(size=self.p.stake)
            elif self.crossover[0] < 0:
                self.order = self.sell(size=self.p.stake)
