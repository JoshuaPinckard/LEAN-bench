import backtrader as bt


class CorrectionOrReversalDaily(bt.Strategy):
    params = (
        ("consolidation_period", 20),
        ("rebound_fraction", 0.06),
        ("capital_fraction", 0.02),
        ("stop_buffer", 3.0),
    )

    def __init__(self):
        self.sma50 = bt.indicators.SMA(self.data.close, period=50)

        # The consolidation range excludes the current bar.
        self.range_low = bt.indicators.Lowest(
            self.data.low(-1), period=self.p.consolidation_period
        )
        self.range_high = bt.indicators.Highest(
            self.data.high(-1), period=self.p.consolidation_period
        )

        self.entry_order = None
        self.exit_order = None
        self.stop_order = None
        self.entry_side = None
        self.anchor_low = None
        self.anchor_candidates = []
        self.after_breakout = False

    def _entry_size(self):
        return (self.broker.getvalue() * self.p.capital_fraction) / self.data.close[0]

    def _cancel_stop(self):
        if self.stop_order is not None:
            self.cancel(self.stop_order)

    def _place_stop(self):
        if self.entry_side == "long":
            self.stop_order = self.sell(
                exectype=bt.Order.Stop,
                price=self.data.low[-1] - self.p.stop_buffer,
            )
        elif self.entry_side == "short":
            self.stop_order = self.buy(
                exectype=bt.Order.Stop,
                price=175.0 + self.p.stop_buffer,
            )

    def notify_order(self, order):
        if order.status in (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            if order is self.entry_order:
                if order.status == order.Completed:
                    self._place_stop()
                self.entry_order = None

            if order is self.exit_order:
                self.exit_order = None

            if order is self.stop_order:
                self.stop_order = None

    def next(self):
        current_low = self.data.low[0]
        current_high = self.data.high[0]
        current_close = self.data.close[0]
        current_volume = self.data.volume[0]

        # An anchor becomes valid once its Low is followed by an approximately 6% rebound.
        for candidate_low in self.anchor_candidates:
            if current_high >= candidate_low * (1.0 + self.p.rebound_fraction):
                self.anchor_low = candidate_low
        self.anchor_candidates.append(current_low)

        consolidation_low = self.range_low[0]
        consolidation_high = self.range_high[0]
        in_consolidation = (
            consolidation_low <= current_close <= consolidation_high
        )

        if current_close > consolidation_high or current_close < consolidation_low:
            self.after_breakout = True
        strategy_active = in_consolidation or self.after_breakout

        previous_low = self.data.low[-1]
        previous_close = self.data.close[-1]
        previous_volume = self.data.volume[-1]

        if self.position:
            if self.exit_order is not None:
                return

            if self.position.size > 0:
                long_target_reached = current_high >= 250.0
                sma_broken_downward = current_close < self.sma50[0]
                dynamic_support_broken = current_close < previous_low

                if (
                    long_target_reached
                    or sma_broken_downward
                    or dynamic_support_broken
                ):
                    self._cancel_stop()
                    self.exit_order = self.close()

            else:
                first_short_target_reached = current_low <= 3350.0
                second_short_target_reached = current_low <= 2700.0
                upper_range_broken = current_close > consolidation_high

                if (
                    first_short_target_reached
                    or second_short_target_reached
                    or upper_range_broken
                ):
                    self._cancel_stop()
                    self.exit_order = self.close()
            return

        if self.entry_order is not None or not strategy_active:
            return

        false_breakdown_on_volume = (
            current_low < consolidation_low
            and current_close >= consolidation_low
            and current_volume > previous_volume
        )
        impulse_trendline_not_broken = current_close >= self.sma50[0]
        bounced_from_anchor_support = (
            self.anchor_low is not None
            and current_low <= self.anchor_low
            and current_close > self.anchor_low
        )
        buying_volume_rising = current_volume > previous_volume
        weak_selling_pressure = (
            previous_close < self.data.close[-2]
            and previous_volume < self.data.volume[-2]
        )

        if (
            in_consolidation
            and false_breakdown_on_volume
            and impulse_trendline_not_broken
            and bounced_from_anchor_support
            and buying_volume_rising
            and weak_selling_pressure
        ):
            self.entry_side = "long"
            self.entry_order = self.buy(size=self._entry_size())
            return

        previous_decline = previous_close < self.data.close[-3]
        returned_to_consolidation = in_consolidation
        strong_selling_resumption = (
            current_close < previous_close
            and current_volume > previous_volume
        )
        dynamic_support_broken = current_close < previous_low
        consolidating_below_175 = (
            current_close < 175.0 and previous_close < 175.0
        )

        if (
            previous_decline
            and returned_to_consolidation
            and strong_selling_resumption
            and dynamic_support_broken
            and consolidating_below_175
        ):
            self.entry_side = "short"
            self.entry_order = self.sell(size=self._entry_size())
