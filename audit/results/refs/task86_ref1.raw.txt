import backtrader as bt
import math


class FibonacciTrendlineStrategy(bt.Strategy):
    params = (
        ('risk_fraction', 0.01),
        ('stop_offset', 0.01),
    )

    def __init__(self):
        self.sma50 = bt.indicators.SimpleMovingAverage(self.data.close, period=50)

        self.fib_618 = 124000.0 - 0.618 * (124000.0 - 74500.0)

        self.local_lows = []
        self.breakout_bar = None

        self.entry_order = None
        self.stop_order = None
        self.target_order = None
        self.exit_order = None

    def _support_at(self, bar_number):
        if len(self.local_lows) < 2:
            return None

        first_bar, first_low = self.local_lows[-2]
        second_bar, second_low = self.local_lows[-1]

        if second_low <= first_low:
            return None

        slope = (second_low - first_low) / float(second_bar - first_bar)
        return second_low + slope * (bar_number - second_bar)

    def next(self):
        if len(self) >= 3:
            if self.data.low[-1] < self.data.low[-2] and self.data.low[-1] <= self.data.low[0]:
                local_low = (len(self) - 1, float(self.data.low[-1]))
                if not self.local_lows or self.local_lows[-1][0] != local_low[0]:
                    self.local_lows.append(local_low)

        if self.exit_order:
            return

        support_now = self._support_at(len(self))
        support_previous = self._support_at(len(self) - 1)

        if self.position:
            if support_now is not None and support_previous is not None:
                broke_below = (
                    self.data.close[-1] > support_previous and
                    self.data.close[0] <= support_now
                )
                if broke_below:
                    if self.stop_order:
                        self.cancel(self.stop_order)
                    if self.target_order:
                        self.cancel(self.target_order)
                    self.exit_order = self.close()
            return

        if self.entry_order:
            return

        if support_now is not None and support_previous is not None:
            broke_above = (
                self.data.close[-1] < support_previous and
                self.data.close[0] >= support_now
            )
            if broke_above:
                self.breakout_bar = len(self)

        if (
            self.breakout_bar is None or
            len(self) <= self.breakout_bar or
            self.data.low[0] > self.fib_618 or
            self.sma50[0] <= self.sma50[-1] or
            support_now is None
        ):
            return

        nearest_local_low = self.local_lows[-1][1] if self.local_lows else support_now
        stop_reference = min(support_now, nearest_local_low)
        stop_price = stop_reference * (1.0 - self.p.stop_offset)
        entry_reference = float(self.data.close[0])

        if stop_price >= entry_reference:
            return

        risk_per_share = (
            (entry_reference - stop_price) +
            (2.0 * entry_reference * self.broker.getcommissioninfo(self.data).p.commission)
        )
        risk_capital = self.broker.getvalue() * self.p.risk_fraction
        cash_limited_size = self.broker.getcash() / (
            entry_reference * (1.0 + self.broker.getcommissioninfo(self.data).p.commission)
        )
        size = int(min(risk_capital / risk_per_share, cash_limited_size))

        if size <= 0:
            return

        target_price = entry_reference + 2.0 * (entry_reference - stop_price)
        orders = self.buy_bracket(
            size=size,
            exectype=bt.Order.Market,
            stopprice=stop_price,
            limitprice=target_price,
        )
        self.entry_order, self.stop_order, self.target_order = orders
        self.breakout_bar = None

    def notify_order(self, order):
        terminal = (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        )

        if order.status in terminal:
            if order == self.entry_order:
                self.entry_order = None
            elif order == self.stop_order:
                self.stop_order = None
            elif order == self.target_order:
                self.target_order = None
            elif order == self.exit_order:
                self.exit_order = None
