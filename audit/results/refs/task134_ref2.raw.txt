import backtrader as bt


class FibRsiStrategy(bt.Strategy):
    params = (
        ("rsi_period", 14),
        ("swing_period", 50),
        ("htf_sma_period", 200),
        ("risk_pct", 0.015),
    )

    def __init__(self):
        self.rsi = bt.indicators.RSI(self.data.close, period=self.p.rsi_period)
        self.htf_sma = bt.indicators.SMA(
            self.data.close, period=self.p.htf_sma_period
        )
        self.order = None
        self.sl_price = None
        self.tp_price = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return
        self.order = None

    def next(self):
        if not self.position:
            self.sl_price = None
            self.tp_price = None

        if self.order:
            return

        # Manage existing position exits
        if self.position:
            if self.position.size > 0:  # Long position
                if self.sl_price and self.data.low[0] <= self.sl_price:
                    self.order = self.close()
                    return
                if self.tp_price and self.data.high[0] >= self.tp_price:
                    self.order = self.close()
                    return
                # Alternative exit: RSI crosses 50 downwards
                if self.rsi[-1] >= 50 and self.rsi[0] < 50:
                    self.order = self.close()
                    return

            elif self.position.size < 0:  # Short position
                if self.sl_price and self.data.high[0] >= self.sl_price:
                    self.order = self.close()
                    return
                if self.tp_price and self.data.low[0] <= self.tp_price:
                    self.order = self.close()
                    return
                # Alternative exit: RSI crosses 50 upwards
                if self.rsi[-1] <= 50 and self.rsi[0] > 50:
                    self.order = self.close()
                    return

            return  # Hold existing position

        # Check entry conditions (only if no position)
        if len(self.data) < max(self.p.swing_period, self.p.htf_sma_period):
            return

        window_highs = [self.data.high[-i] for i in range(self.p.swing_period)]
        window_lows = [self.data.low[-i] for i in range(self.p.swing_period)]
        H = max(window_highs)
        L = min(window_lows)

        if H == L:
            return

        # Top-down Fibonacci retracement levels
        p0 = H
        p236 = H - 0.236 * (H - L)
        p382 = H - 0.382 * (H - L)
        p500 = H - 0.500 * (H - L)
        p618 = H - 0.618 * (H - L)
        p786 = H - 0.786 * (H - L)
        p100 = L

        fib_levels = [p0, p236, p382, p500, p618, p786, p100]

        htf_uptrend = self.data.close[0] > self.htf_sma[0]
        htf_downtrend = self.data.close[0] < self.htf_sma[0]

        # Long Entry Signals
        rsi_long_signal = (self.rsi[-1] < 30) and (self.rsi[0] > self.rsi[-1])
        bounce_382_long = (self.data.low[0] <= p382) and (
            self.data.close[0] > p382
        )
        bounce_618_long = (self.data.low[0] <= p618) and (
            self.data.close[0] > p618
        )

        if htf_uptrend and rsi_long_signal and (bounce_382_long or bounce_618_long):
            entry_price = self.data.close[0]
            lows_below = [
                self.data.low[-i]
                for i in range(self.p.swing_period)
                if self.data.low[-i] < entry_price
            ]
            nearest_low = max(lows_below) if lows_below else p786

            sl_candidates = [s for s in [p786, nearest_low] if s < entry_price]
            sl_price = max(sl_candidates) if sl_candidates else entry_price * 0.98

            levels_above = sorted([f for f in fib_levels if f > entry_price])
            tp_price = levels_above[0] if levels_above else entry_price * 1.04

            risk_per_share = entry_price - sl_price
            if risk_per_share > 0:
                capital = self.broker.getvalue()
                risk_amount = capital * self.p.risk_pct
                size = int(risk_amount / risk_per_share)
                size = max(1, size)
                self.sl_price = sl_price
                self.tp_price = tp_price
                self.order = self.buy(size=size)
                return

        # Short Entry Signals
        rsi_short_signal = (self.rsi[-1] > 70) and (self.rsi[0] < self.rsi[-1])
        bounce_382_short = (self.data.high[0] >= p382) and (
            self.data.close[0] < p382
        )
        bounce_618_short = (self.data.high[0] >= p618) and (
            self.data.close[0] < p618
        )

        if htf_downtrend and rsi_short_signal and (bounce_382_short or bounce_618_short):
            entry_price = self.data.close[0]
            highs_above = [
                self.data.high[-i]
                for i in range(self.p.swing_period)
                if self.data.high[-i] > entry_price
            ]
            nearest_high = min(highs_above) if highs_above else p236

            sl_candidates = [s for s in [p236, nearest_high] if s > entry_price]
            sl_price = min(sl_candidates) if sl_candidates else entry_price * 1.02

            levels_below = sorted([f for f in fib_levels if f < entry_price], reverse=True)
            tp_price = levels_below[0] if levels_below else entry_price * 0.96

            risk_per_share = sl_price - entry_price
            if risk_per_share > 0:
                capital = self.broker.getvalue()
                risk_amount = capital * self.p.risk_pct
                size = int(risk_amount / risk_per_share)
                size = max(1, size)
                self.sl_price = sl_price
                self.tp_price = tp_price
                self.order = self.sell(size=size)
                return
