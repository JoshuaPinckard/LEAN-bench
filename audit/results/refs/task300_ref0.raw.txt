import backtrader as bt


class ScalpingStrategy(bt.Strategy):
    params = (
        ('symbol', 'SNAP'),
        ('stoch_period', 5),
        ('stoch_pfast', 3),
        ('stoch_pslow', 3),
        ('rsi_period', 3),
        ('vol_sma_period', 20),
        ('vol_mult', 1.2),
        ('atr_period', 14),
        ('atr_threshold', 0.03),
        ('take_profit', 0.01),
        ('stop_loss', 0.02),
        ('timeout_bars', 3),
        ('size', 100),
        ('use_pdt_rule', False),
        ('pdt_balance_threshold', 25000),
        ('session_start_hour', 9),
        ('session_start_minute', 30),
        ('session_end_hour', 16),
        ('session_end_minute', 0),
        ('warmup_minutes', 3),
        ('liquidity_threshold', 0),
    )

    def __init__(self):
        self.stoch = bt.indicators.Stochastic(
            self.data,
            period=self.p.stoch_period,
            period_dfast=self.p.stoch_pfast,
            period_dslow=self.p.stoch_pslow,
        )
        self.percK = self.stoch.percK
        self.percD = self.stoch.percD
        self.crossover = bt.indicators.CrossOver(self.percK, self.percD)
        self.rsi = bt.indicators.RSI(self.data, period=self.p.rsi_period)
        self.vol_sma = bt.indicators.SMA(self.data.volume, period=self.p.vol_sma_period)
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)

        self.order = None
        self.entry_price = None
        self.entry_bar = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return
        if order.status == order.Completed:
            if not self.position:
                self.entry_price = None
                self.entry_bar = None
            else:
                self.entry_price = order.executed.price
                self.entry_bar = len(self)
        self.order = None

    def next(self):
        if self.order:
            return

        t = self.data.datetime.time(0)
        minutes_of_day = t.hour * 60 + t.minute
        session_start = self.p.session_start_hour * 60 + self.p.session_start_minute
        session_end = self.p.session_end_hour * 60 + self.p.session_end_minute
        trading_start = session_start + self.p.warmup_minutes
        in_main_session = session_start <= minutes_of_day < session_end
        past_warmup = minutes_of_day >= trading_start

        if not self.position:
            if not (in_main_session and past_warmup):
                return
            if self.p.use_pdt_rule and self.broker.getvalue() <= self.p.pdt_balance_threshold:
                return
            if self.vol_sma[0] <= self.p.liquidity_threshold:
                return

            volume_ok = self.data.volume[0] > self.p.vol_mult * self.vol_sma[0]
            atr_ok = self.atr[0] >= self.p.atr_threshold

            long_signal = (
                self.crossover[0] > 0
                and self.percK[0] > 80
                and self.rsi[0] > 55
                and volume_ok
                and atr_ok
            )
            short_signal = (
                self.crossover[0] < 0
                and self.percK[0] < 20
                and self.rsi[0] < 45
                and volume_ok
                and atr_ok
            )

            if long_signal:
                self.order = self.buy(size=self.p.size)
            elif short_signal:
                self.order = self.sell(size=self.p.size)
            return

        price = self.data.close[0]
        bars_held = len(self) - self.entry_bar if self.entry_bar is not None else 0
        timeout = bars_held >= self.p.timeout_bars

        if self.position.size > 0:
            tp_hit = price >= self.entry_price + self.p.take_profit
            sl_hit = price <= self.entry_price - self.p.stop_loss
            reversal = (self.crossover[0] < 0) or (self.percK[0] < 50)
            if tp_hit or sl_hit or reversal or timeout:
                self.order = self.close()
        elif self.position.size < 0:
            tp_hit = price <= self.entry_price - self.p.take_profit
            sl_hit = price >= self.entry_price + self.p.stop_loss
            reversal = (self.crossover[0] > 0) or (self.percK[0] > 50)
            if tp_hit or sl_hit or reversal or timeout:
                self.order = self.close()
