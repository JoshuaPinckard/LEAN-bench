import backtrader as bt


class VolatilityTrendStrategy(bt.Strategy):
    params = (
        ('atr_period', 14),
        ('sma_period', 20),
        ('rsi_period', 14),
        ('ema_period', 14),
        ('vol_sma_period', 20),
        ('rsi_long_threshold', 55),
        ('rsi_short_threshold', 45),
        ('rsi_cross_level', 50),
        ('sl_atr_mult', 1.5),
        ('tp_atr_mult', 2.0),
    )

    def __init__(self):
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.sma = bt.indicators.SMA(self.data.close, period=self.p.sma_period)
        self.rsi = bt.indicators.RSI(self.data.close, period=self.p.rsi_period)
        self.atr_ema = bt.indicators.EMA(self.atr, period=self.p.ema_period)
        self.vol_sma = bt.indicators.SMA(self.data.volume, period=self.p.vol_sma_period)

        self.rsi_cross_below = bt.indicators.CrossDown(self.rsi, self.p.rsi_cross_level)
        self.rsi_cross_above = bt.indicators.CrossUp(self.rsi, self.p.rsi_cross_level)

        self.order = None
        self.sl_order = None
        self.tp_order = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            if order == self.order:
                if self.position:
                    if self.position.size > 0:
                        sl_price = order.executed.price - self.p.sl_atr_mult * self.atr[0]
                        tp_price = order.executed.price + self.p.tp_atr_mult * self.atr[0]
                        self.sl_order = self.sell(exectype=bt.Order.Stop, price=sl_price)
                        self.tp_order = self.sell(exectype=bt.Order.Limit, price=tp_price, oco=self.sl_order)
                    elif self.position.size < 0:
                        sl_price = order.executed.price + self.p.sl_atr_mult * self.atr[0]
                        tp_price = order.executed.price - self.p.tp_atr_mult * self.atr[0]
                        self.sl_order = self.buy(exectype=bt.Order.Stop, price=sl_price)
                        self.tp_order = self.buy(exectype=bt.Order.Limit, price=tp_price, oco=self.sl_order)
                self.order = None
            elif order in [self.sl_order, self.tp_order]:
                self.sl_order = None
                self.tp_order = None

        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            if order == self.order:
                self.order = None
            if order == self.sl_order:
                self.sl_order = None
            if order == self.tp_order:
                self.tp_order = None

    def next(self):
        if self.order:
            return

        if not self.position:
            long_cond = (
                self.data.close[0] > self.sma[0] and
                self.rsi[0] > self.p.rsi_long_threshold and
                self.atr[0] > self.atr_ema[0] and
                self.data.volume[0] > self.vol_sma[0]
            )
            short_cond = (
                self.data.close[0] < self.sma[0] and
                self.rsi[0] < self.p.rsi_short_threshold and
                self.atr[0] > self.atr_ema[0] and
                self.data.volume[0] > self.vol_sma[0]
            )

            if long_cond:
                self.order = self.buy()
            elif short_cond:
                self.order = self.sell()

        else:
            if self.position.size > 0:
                long_exit = (
                    self.data.close[0] < self.sma[0] or
                    self.rsi_cross_below[0] > 0
                )
                if long_exit:
                    if self.sl_order:
                        self.cancel(self.sl_order)
                        self.sl_order = None
                    if self.tp_order:
                        self.cancel(self.tp_order)
                        self.tp_order = None
                    self.order = self.close()

            elif self.position.size < 0:
                short_exit = (
                    self.data.close[0] > self.sma[0] or
                    self.rsi_cross_above[0] > 0
                )
                if short_exit:
                    if self.sl_order:
                        self.cancel(self.sl_order)
                        self.sl_order = None
                    if self.tp_order:
                        self.cancel(self.tp_order)
                        self.tp_order = None
                    self.order = self.close()
