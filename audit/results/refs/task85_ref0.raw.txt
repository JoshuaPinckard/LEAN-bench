import backtrader as bt


class BreakRetestStrategy(bt.Strategy):
    params = dict(
        take_profit_pct=0.015,
        stop_loss_pct=0.005,
        risk_pct=0.01,
        retest_window=5,
    )

    def __init__(self):
        self.order = None
        self.stop_order = None
        self.limit_order = None
        self.entry_price = None

        self.current_date = None
        self.bar_in_day = -1
        self.day_high = None
        self.day_low = None
        self.prev_day_high = None
        self.prev_day_low = None
        self.first5_high = None
        self.first5_low = None
        self.resistance = None

        self.breakout_active = False
        self.retest_confirmed = False
        self.bars_since_breakout = 0
        self.retest_close = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status == order.Completed:
            if order.isbuy():
                self.entry_price = order.executed.price
                size = order.executed.size
                stop_price = self.entry_price * (1 - self.p.stop_loss_pct)
                target_price = self.entry_price * (1 + self.p.take_profit_pct)
                self.stop_order = self.sell(exectype=bt.Order.Stop, price=stop_price, size=size)
                self.limit_order = self.sell(exectype=bt.Order.Limit, price=target_price, size=size,
                                              oco=self.stop_order)
                self.order = None
            else:
                self.entry_price = None
                self.stop_order = None
                self.limit_order = None
                self.order = None
                self.breakout_active = False
                self.retest_confirmed = False
                self.bars_since_breakout = 0
                self.retest_close = None
        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            if order.isbuy():
                self.order = None
            else:
                if order is self.stop_order:
                    self.stop_order = None
                if order is self.limit_order:
                    self.limit_order = None

    def next(self):
        dt = self.data.datetime.date(0)
        if dt != self.current_date:
            if self.current_date is not None:
                self.prev_day_high = self.day_high
                self.prev_day_low = self.day_low
            self.current_date = dt
            self.bar_in_day = 0
            self.day_high = self.data.high[0]
            self.day_low = self.data.low[0]
            self.first5_high = self.data.high[0]
            self.first5_low = self.data.low[0]
        else:
            self.bar_in_day += 1
            self.day_high = max(self.day_high, self.data.high[0])
            self.day_low = min(self.day_low, self.data.low[0])

        if self.bar_in_day == 0:
            return

        if self.bar_in_day == 1:
            components = [self.first5_high]
            if self.prev_day_high is not None:
                components.append(self.prev_day_high)
            self.resistance = max(components)
            self.breakout_active = False
            self.retest_confirmed = False
            self.bars_since_breakout = 0
            self.retest_close = None

        if self.resistance is None or self.position or self.order is not None:
            return

        close = self.data.close[0]

        if self.retest_confirmed:
            if close > self.retest_close:
                size = self._calc_size(close)
                if size > 0:
                    self.order = self.buy(size=size)
                self.breakout_active = False
                self.retest_confirmed = False
                self.bars_since_breakout = 0
                self.retest_close = None
        elif self.breakout_active:
            self.bars_since_breakout += 1
            if close <= self.resistance:
                self.retest_confirmed = True
                self.retest_close = close
            elif self.bars_since_breakout >= self.p.retest_window:
                self.breakout_active = False
                self.bars_since_breakout = 0
        else:
            if close > self.resistance:
                self.breakout_active = True
                self.bars_since_breakout = 0

    def _calc_size(self, entry_price):
        value = self.broker.getvalue()
        risk_amount = value * self.p.risk_pct
        risk_per_share = entry_price * self.p.stop_loss_pct
        if risk_per_share <= 0:
            return 0
        return int(risk_amount / risk_per_share)
