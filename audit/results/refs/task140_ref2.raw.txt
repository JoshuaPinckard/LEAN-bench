import backtrader as bt


class MonthlyStrategy(bt.Strategy):
    def __init__(self):
        self.order = None
        self.initial_portfolio_value = None
        self.final_portfolio_value = None

    def start(self):
        self.initial_portfolio_value = self.broker.getvalue()

    def stop(self):
        self.final_portfolio_value = self.broker.getvalue()

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        if self.order:
            return

        # Check if today is the first trading day of the month
        if len(self) == 1:
            is_first_day = True
        else:
            is_first_day = self.data.datetime.date(0).month != self.data.datetime.date(-1).month

        # Check if today is the last trading day of the month
        if len(self) == self.data.buflen():
            is_last_day = True
        else:
            is_last_day = self.data.datetime.date(1).month != self.data.datetime.date(0).month

        if is_last_day and self.position:
            self.order = self.close()
        elif is_first_day and not self.position:
            cash = self.broker.getcash()
            price = self.data.close[0]
            if price > 0:
                size = int(cash / price)
                if size > 0:
                    self.order = self.buy(size=size)
