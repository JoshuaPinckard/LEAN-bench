import backtrader as bt


class MovingAverageExitStrategy(bt.Strategy):
    params = (("sma_period", 200),)

    def __init__(self):
        self.order = None
        self.sma = bt.indicators.SimpleMovingAverage(
            self.data.close, period=self.p.sma_period
        )
        self.cross_up = bt.indicators.CrossUp(self.data.close, self.sma)
        self.cross_down = bt.indicators.CrossDown(self.data.close, self.sma)

    def next(self):
        if self.order:
            return

        if not self.position:
            if self.cross_up[0] > 0:
                size = (self.broker.getvalue() * 0.01) / self.data.close[0]
                self.order = self.buy(size=size)
        elif self.cross_down[0] > 0:
            self.order = self.close()

    def notify_order(self, order):
        if order.status in (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            self.order = None
