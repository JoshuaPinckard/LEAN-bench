import backtrader as bt


class VolTargetATRMonthly(bt.Strategy):
    def __init__(self):
        self.atr21 = bt.indicators.ATR(self.data, period=21)
        self.atr5 = bt.indicators.ATR(self.data, period=5)
        self.order = None
        self.current_ym = None

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        dt = self.data.datetime.date(0)
        ym = (dt.year, dt.month)
        is_new_month = ym != self.current_ym
        self.current_ym = ym

        if not is_new_month or self.order is not None:
            return

        close = self.data.close[0]
        atr5_ratio = self.atr5[0] / close

        if atr5_ratio > 0.02:
            self.order = self.close()
        else:
            atr21_ratio = self.atr21[0] / close
            target_weight = min(0.4, 0.01 / atr21_ratio)
            self.order = self.order_target_percent(target=target_weight)
