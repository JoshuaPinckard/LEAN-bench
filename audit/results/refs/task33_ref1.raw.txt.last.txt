import backtrader as bt
import math


class BullishReversalOffSellingClimax(bt.Strategy):
    def __init__(self):
        self.sma200 = bt.indicators.SMA(self.data.close, period=200)
        self.atr14 = bt.indicators.ATR(self.data, period=14)
        self.avg_volume50 = bt.indicators.SMA(self.data.volume, period=50)

        self.entry_order = None
        self.stop_order = None
        self.target_order = None
        self.close_order = None

    def next(self):
        if self.close_order or self.stop_order or self.target_order:
            return

        if self.position:
            if self.data.close[0] < self.sma200[0]:
                self.close_order = self.close()
            return

        if self.entry_order:
            return

        if self.data.close[0] <= self.sma200[0]:
            return

        recent_high = max(self.data.high[-i] for i in range(20))
        recent_low = min(self.data.low[-i] for i in range(20))
        sideways = (recent_high - recent_low) / recent_low <= 0.10
        prolonged_downtrend = self.data.close[-1] <= self.data.close[-10]
        strong_sustained_uptrend = (
            self.data.close[0] > self.sma200[0] * 1.10
            and self.data.close[0] > self.data.close[-20] * 1.10
        )

        if (not sideways and not prolonged_downtrend) or strong_sustained_uptrend:
            return

        bullish_engulfing = (
            self.data.close[0] > self.data.open[0]
            and self.data.open[0] <= self.data.close[-1]
            and self.data.close[0] >= self.data.open[-1]
        )
        if not bullish_engulfing:
            return

        selling_climax_high = None
        for bars_ago in range(1, 11):
            sharp_drop = (
                (self.data.close[-bars_ago] - self.data.close[-bars_ago - 1])
                / self.data.close[-bars_ago - 1]
                <= -0.05
            )
            abnormal_volume = (
                self.data.volume[-bars_ago]
                >= self.avg_volume50[-bars_ago] * 1.50
            )

            if sharp_drop and abnormal_volume:
                selling_climax_high = self.data.high[-bars_ago]
                break

        if selling_climax_high is None:
            return

        if self.data.close[0] <= selling_climax_high:
            return

        risk_per_share = self.atr14[0] * 2.0
        if risk_per_share <= 0:
            return

        size = math.floor((self.broker.getvalue() * 0.01) / risk_per_share)
        if size > 0:
            self.entry_order = self.buy(size=size)

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order is self.entry_order:
            self.entry_order = None

            if order.status == order.Completed:
                risk_per_share = self.atr14[0] * 2.0
                stop_price = order.executed.price - risk_per_share
                target_price = order.executed.price + (risk_per_share * 2.0)

                self.stop_order = self.sell(
                    size=self.position.size,
                    exectype=bt.Order.Stop,
                    price=stop_price,
                )
                self.target_order = self.sell(
                    size=self.position.size,
                    exectype=bt.Order.Limit,
                    price=target_price,
                )
            return

        if order is self.stop_order:
            self.stop_order = None
            if order.status == order.Completed and self.target_order:
                self.cancel(self.target_order)
            return

        if order is self.target_order:
            self.target_order = None
            if order.status == order.Completed and self.stop_order:
                self.cancel(self.stop_order)
            return

        if order is self.close_order:
            self.close_order = None
