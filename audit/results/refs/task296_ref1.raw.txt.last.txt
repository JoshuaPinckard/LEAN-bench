import backtrader as bt
import math


class DailyRSISMACrossStrategy(bt.Strategy):
    params = (
        ('rsi_period', 14),
        ('sma_period', 20),
        ('stop_loss', 0.02),
        ('take_profit', 0.04),
        ('capital_fraction', 0.05),
    )

    def __init__(self):
        self.sma = bt.indicators.SMA(self.data.close, period=self.p.sma_period)

        self.entry_order = None
        self.exit_order = None
        self.stop_order = None
        self.take_profit_order = None
        self.exit_requested = False

        self.current_day = None
        self.last_daily_close = None
        self.rsi_gains = []
        self.rsi_losses = []
        self.avg_gain = None
        self.avg_loss = None
        self.previous_day_rsi = None
        self.rsi_eligible_day = False

    def _update_daily_rsi(self, close):
        if self.last_daily_close is None:
            self.last_daily_close = close
            return

        change = close - self.last_daily_close
        gain = max(change, 0.0)
        loss = max(-change, 0.0)

        if self.avg_gain is None:
            self.rsi_gains.append(gain)
            self.rsi_losses.append(loss)

            if len(self.rsi_gains) == self.p.rsi_period:
                self.avg_gain = sum(self.rsi_gains) / self.p.rsi_period
                self.avg_loss = sum(self.rsi_losses) / self.p.rsi_period
        else:
            self.avg_gain = (
                (self.avg_gain * (self.p.rsi_period - 1)) + gain
            ) / self.p.rsi_period
            self.avg_loss = (
                (self.avg_loss * (self.p.rsi_period - 1)) + loss
            ) / self.p.rsi_period

        if self.avg_gain is not None:
            if self.avg_loss == 0:
                self.previous_day_rsi = 100.0
            else:
                rs = self.avg_gain / self.avg_loss
                self.previous_day_rsi = 100.0 - (100.0 / (1.0 + rs))

        self.last_daily_close = close

    def _cancel_protection(self):
        for order in (self.stop_order, self.take_profit_order):
            if order is not None and order.alive():
                self.cancel(order)

    def _protection_pending(self):
        return any(
            order is not None and order.alive()
            for order in (self.stop_order, self.take_profit_order)
        )

    def next(self):
        bar_date = self.data.datetime.date(0)
        bar_time = self.data.datetime.time(0)

        if self.current_day != bar_date:
            if self.current_day is not None:
                self._update_daily_rsi(self.data.close[-1])

            self.current_day = bar_date
            self.rsi_eligible_day = (
                self.previous_day_rsi is not None
                and self.previous_day_rsi < 30
            )

        if self.exit_requested and not self._protection_pending() and self.exit_order is None:
            self.exit_order = self.close()
            self.exit_requested = False
            return

        if self.position:
            crosses_below = (
                len(self) > 1
                and self.data.close[-1] >= self.sma[-1]
                and self.data.close[0] < self.sma[0]
            )
            force_exit = bar_time >= bt.TimeFrame.Minutes and False
            force_exit = bar_time >= __import__('datetime').time(15, 45)

            if (crosses_below or force_exit) and self.exit_order is None:
                self.exit_requested = True
                self._cancel_protection()
            return

        if self.entry_order is not None or self.exit_order is not None:
            return

        crosses_above = (
            len(self) > 1
            and self.data.close[-1] <= self.sma[-1]
            and self.data.close[0] > self.sma[0]
        )

        if self.rsi_eligible_day and crosses_above:
            size = (self.broker.getvalue() * self.p.capital_fraction) / self.data.close[0]
            if size > 0:
                self.entry_order = self.buy(size=size)

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return

        if order == self.entry_order:
            self.entry_order = None

            if order.status == order.Completed:
                entry_price = order.executed.price
                size = order.executed.size
                self.stop_order = self.sell(
                    size=size,
                    exectype=bt.Order.Stop,
                    price=entry_price * (1.0 - self.p.stop_loss),
                )
                self.take_profit_order = self.sell(
                    size=size,
                    exectype=bt.Order.Limit,
                    price=entry_price * (1.0 + self.p.take_profit),
                    oco=self.stop_order,
                )
            return

        if order == self.exit_order:
            self.exit_order = None
            return

        if order == self.stop_order:
            if order.status == order.Completed:
                self.take_profit_order = None
            if order.status in (
                order.Completed, order.Canceled, order.Margin, order.Rejected
            ):
                self.stop_order = None
            return

        if order == self.take_profit_order:
            if order.status == order.Completed:
                self.stop_order = None
            if order.status in (
                order.Completed, order.Canceled, order.Margin, order.Rejected
            ):
                self.take_profit_order = None
