import math
import backtrader as bt


class SlippageAssessmentStrategy(bt.Strategy):
    """
    Reference implementation of Slippage Assessment on Daily Data for Position Management.

    Calculates price_change_per_volume = (log(high) - log(low)) / volume.
    Sizes positions inversely proportional to price_change_per_volume, capped at 5% of daily volume
    and available cash. Executes all entry and exit orders as Market on Close (MOC) orders and
    dynamically accounts for estimated slippage on closing orders.
    """

    params = (
        ("target_impact", 1e-6),  # Target log-price impact constant for inverse sizing
        ("max_vol_pct", 0.05),    # Maximum trade volume as a fraction of daily volume (5%)
    )

    def __init__(self):
        self.order = None

    def next(self):
        if self.order:
            return

        high = self.data.high[0]
        low = self.data.low[0]
        volume = self.data.volume[0]
        close = self.data.close[0]

        # Guard against zero or non-positive volume / prices
        if volume <= 0 or high <= 0 or low <= 0 or high <= low:
            return

        # Calculate price_change_per_volume metric: (log(high) - log(low)) / volume
        price_change_per_volume = (math.log(high) - math.log(low)) / volume

        if price_change_per_volume <= 0:
            return

        if not self.position:
            # Calculate position size inversely proportional to price_change_per_volume
            raw_size = int(self.p.target_impact / price_change_per_volume)

            # Volume filter: trade volume must not exceed 5% of total daily volume
            max_vol_size = int(self.p.max_vol_pct * volume)

            # Available cash constraint
            available_cash = self.broker.getcash() * 0.95
            max_cash_size = int(available_cash / close) if close > 0 else 0

            size = min(raw_size, max_vol_size, max_cash_size)

            if size > 0:
                # Set estimated slippage on entry
                est_slippage_pct = min(max(price_change_per_volume * size, 0.0), 0.05)
                self.broker.set_slippage_perc(est_slippage_pct)

                # Order execution at Market on Close (MOC)
                self.order = self.buy(size=size, exectype=bt.Order.Close)
        else:
            # Close existing position at Market on Close (MOC)
            size = self.position.size

            # Estimate and account for slippage when closing positions at the end of the trading day
            est_slippage_pct = min(max(price_change_per_volume * size, 0.0), 0.05)
            self.broker.set_slippage_perc(est_slippage_pct)

            self.order = self.close(exectype=bt.Order.Close)

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return
        # Reset order tracking on Completion, Cancellation, Margin call, or Rejection
        self.order = None
