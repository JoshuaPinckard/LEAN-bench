import backtrader as bt


class MicroMomentumStrategy(bt.Strategy):
    def __init__(self):
        self.ema5 = bt.indicators.EMA(self.data.close, period=5)
        self.ema10 = bt.indicators.EMA(self.data.close, period=10)
        self.rsi7 = bt.indicators.RSI(self.data.close, period=7)
        self.volume_sma20 = bt.indicators.SMA(self.data.volume, period=20)
        self.atr14 = bt.indicators.ATR(self.data, period=14)
        self.crossover = bt.indicators.CrossOver(self.ema5, self.ema10)

        self.pending_order = None
        self.entry_price = None
        self.entry_atr = None
        self.entry_bar = None

        self.broker.set_slippage_perc(perc=0.0005)

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return

        if order.status == order.Completed:
            role = getattr(order.info, "role", None)

            if role == "entry":
                self.entry_price = order.executed.price
                self.entry_atr = self.atr14[0]
                self.entry_bar = len(self)

            elif role == "exit":
                self.entry_price = None
                self.entry_atr = None
                self.entry_bar = None

        if order.status in (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            if order == self.pending_order:
                self.pending_order = None

    def _close_position(self):
        self.pending_order = self.close()
        if self.pending_order is not None:
            self.pending_order.addinfo(role="exit")

    def next(self):
        if self.pending_order is not None:
            return

        current_dt = self.data.datetime.datetime(0)
        session_cutoff = (
            current_dt.hour > 15
            or (current_dt.hour == 15 and current_dt.minute >= 55)
        )

        if self.position:
            if session_cutoff:
                self._close_position()
                return

            take_profit = self.entry_price + (0.15 * self.entry_atr)
            stop_loss = self.entry_price - (0.10 * self.entry_atr)

            if self.position.size > 0:
                if self.data.low[0] <= stop_loss:
                    self._close_position()
                    return
                if self.data.high[0] >= take_profit:
                    self._close_position()
                    return
            else:
                short_take_profit = self.entry_price - (0.15 * self.entry_atr)
                short_stop_loss = self.entry_price + (0.10 * self.entry_atr)

                if self.data.high[0] >= short_stop_loss:
                    self._close_position()
                    return
                if self.data.low[0] <= short_take_profit:
                    self._close_position()
                    return

            if len(self) - self.entry_bar >= 3:
                self._close_position()
            return

        if session_cutoff:
            return

        capital = self.broker.getvalue()
        size = (capital * 0.001) / self.data.close[0]

        if (
            self.crossover[0] > 0
            and self.rsi7[0] > 55
            and self.data.volume[0] > self.volume_sma20[0]
            and self.atr14[0] / self.data.close[0] > 0.001
            and self.data.volume[0] > self.volume_sma20[0]
        ):
            self.pending_order = self.buy(size=size)
            self.pending_order.addinfo(role="entry")

        elif (
            self.crossover[0] < 0
            and self.rsi7[0] < 45
            and self.data.volume[0] > self.volume_sma20[0]
            and self.atr14[0] / self.data.close[0] > 0.001
            and self.data.volume[0] > self.volume_sma20[0]
        ):
            self.pending_order = self.sell(size=size)
            self.pending_order.addinfo(role="entry")
