import backtrader as bt
import math


class PostDisasterCapitulationBounce(bt.Strategy):
    def __init__(self):
        self.sma20 = bt.indicators.SMA(self.data.close, period=20)
        self.sma50 = bt.indicators.SMA(self.data.close, period=50)
        self.rsi = bt.indicators.RSI(self.data.close, period=14)
        self.atr = bt.indicators.ATR(self.data, period=14)
        self.volsma20 = bt.indicators.SMA(self.data.volume, period=20)
        self.rsi_cross_30 = bt.indicators.CrossOver(self.rsi, 30.0)

        self.order = None
        self.cap_low = None
        self.cap_high = None
        self.cap_bar = None
        self.stop_price = None
        self._pending_stop = None

    def notify_order(self, order):
        if order.status in (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            if order.status == order.Completed:
                if order.isbuy():
                    self.stop_price = self._pending_stop
                elif order.issell():
                    self.stop_price = None

            if order.status in (order.Canceled, order.Margin, order.Rejected):
                self._pending_stop = None

            self.order = None

    def next(self):
        is_cap_day = (
            self.data.close[0] < self.data.open[0]
            and self.data.volume[0] > 7.0 * self.volsma20[0]
        )

        if is_cap_day:
            self.cap_low = self.data.low[0]
            self.cap_high = max(self.data.high.get(size=20))
            self.cap_bar = len(self)

        if self.order:
            return

        if self.position:
            if self.stop_price is not None:
                tp1 = self.cap_low + 0.382 * (self.cap_high - self.cap_low)
                tp2 = self.cap_low + 0.500 * (self.cap_high - self.cap_low)
                tp3 = self.cap_low + 0.618 * (self.cap_high - self.cap_low)
                tp4 = self.cap_high

                if (
                    self.data.low[0] <= self.stop_price
                    or self.data.high[0] >= tp2
                    or self.data.high[0] >= self.sma50[0]
                ):
                    self.order = self.close()
            return

        if self.cap_bar is None or len(self) - self.cap_bar > 10:
            return

        capitulation_drop = self.data.close[0] <= self.data.close[-5] * 0.85
        reversal_signal = (
            self.data.close[0] > self.data.open[0]
            and self.rsi_cross_30[0] > 0
            and self.data.volume[0] >= 0.8 * self.volsma20[0]
            and (self.data.high[0] - self.data.low[0]) / self.data.open[0] >= 0.03
        )

        if (
            capitulation_drop
            and reversal_signal
            and self.data.close[0] > self.sma20[0]
        ):
            stop = self.cap_low - self.atr[0]
            risk_per_share = self.data.close[0] - stop

            if risk_per_share > 0:
                size = math.floor(
                    (0.02 * self.broker.getvalue()) / risk_per_share
                )

                if size >= 1:
                    self._pending_stop = stop
                    self.order = self.buy(size=size)
