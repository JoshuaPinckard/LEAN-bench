import backtrader as bt


class MidTermBreakoutSAR(bt.Strategy):
    params = (
        ("af", 0.02),
        ("afmax", 0.2),
        ("vol_period", 20),
    )

    def __init__(self):
        self.order = None
        self.sar = bt.indicators.ParabolicSAR(
            self.data, af=self.p.af, afmax=self.p.afmax
        )
        self.avg_volume = bt.indicators.SMA(
            self.data.volume, period=self.p.vol_period
        )

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return
        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            self.order = None

    def next(self):
        if self.order:
            return

        if self.position:
            if self.position.size > 0:
                if self.data.close[0] < self.sar[0]:
                    self.order = self.close()
            elif self.position.size < 0:
                if self.data.close[0] > self.sar[0]:
                    self.order = self.close()
        else:
            if (
                self.data.close[0] > self.sar[0]
                and self.data.volume[0] > self.avg_volume[0]
            ):
                self.order = self.buy()
            elif (
                self.data.close[0] < self.sar[0]
                and self.data.volume[0] > self.avg_volume[0]
            ):
                self.order = self.sell()
