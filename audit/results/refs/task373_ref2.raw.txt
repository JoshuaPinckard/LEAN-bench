import datetime
import backtrader as bt


class LunchBreakFade(bt.Strategy):
    def __init__(self):
        self.rsi = bt.indicators.RSI(self.data.close, period=14)
        self.vol_sma = bt.indicators.SMA(self.data.volume, period=20)

        self.order = None
        self.day_open = None
        self.bias = None
        self.current_date = None
        self.bias_evaluated = False

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return
        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            self.order = None

    def next(self):
        cur_dt = self.data.datetime.datetime(0)
        cur_date = cur_dt.date()
        cur_time = cur_dt.time()

        if self.current_date != cur_date:
            self.current_date = cur_date
            self.day_open = self.data.open[0]
            self.bias = None
            self.bias_evaluated = False

        if cur_time == datetime.time(9, 30):
            self.day_open = self.data.open[0]

        if not self.bias_evaluated and cur_time >= datetime.time(11, 30):
            if self.day_open is not None and self.day_open > 0:
                pct_change = (
                    self.data.close[0] - self.day_open
                ) / self.day_open
                if pct_change >= 0.005:
                    self.bias = "short"
                elif pct_change <= -0.005:
                    self.bias = "long"
                else:
                    self.bias = "no_trade"
            self.bias_evaluated = True

        if self.order:
            return

        if self.position:
            if cur_time >= datetime.time(14, 30):
                self.order = self.close()
                return

            if self.position.size > 0:
                stop_price = self.position.price * 0.98
                if self.data.low[0] <= stop_price:
                    self.order = self.close()
                    return
                if self.rsi[-1] >= 40 and self.rsi[0] < 40:
                    self.order = self.close()
                    return
            elif self.position.size < 0:
                stop_price = self.position.price * 1.02
                if self.data.high[0] >= stop_price:
                    self.order = self.close()
                    return
                if self.rsi[-1] <= 60 and self.rsi[0] > 60:
                    self.order = self.close()
                    return
        else:
            if datetime.time(11, 30) <= cur_time < datetime.time(14, 30):
                if self.data.volume[0] < self.vol_sma[0]:
                    if self.bias == "short":
                        if self.rsi[-1] >= 60 and self.rsi[0] < 60:
                            self.order = self.sell()
                    elif self.bias == "long":
                        if self.rsi[-1] <= 40 and self.rsi[0] > 40:
                            self.order = self.buy()
