import backtrader as bt
import math


class GeneticHeuristicStrategy(bt.Strategy):
    params = (
        ("ma_type", "SMA"),
        ("fast_period", 20),
        ("slow_period", 50),
        ("rsi_period", 14),
        ("macd_fast_period", 12),
        ("macd_slow_period", 26),
        ("macd_signal_period", 9),
        ("take_profit", 0.02),
        ("stop_loss", 0.01),
        ("capital_fraction", 0.02),
        ("enable_trend_filter", False),
        ("trend_period", 200),
        ("enable_volatility_filter", False),
        ("atr_period", 14),
        ("volatility_min", 0.0),
        ("volatility_max", math.inf),
    )

    def __init__(self):
        ma_class = bt.ind.EMA if self.p.ma_type.upper() == "EMA" else bt.ind.SMA

        self.fast_ma = ma_class(self.data.close, period=self.p.fast_period)
        self.slow_ma = ma_class(self.data.close, period=self.p.slow_period)
        self.rsi = bt.ind.RSI(self.data.close, period=self.p.rsi_period)

        self.macd_fast = bt.ind.EMA(
            self.data.close, period=self.p.macd_fast_period
        )
        self.macd_slow = bt.ind.EMA(
            self.data.close, period=self.p.macd_slow_period
        )
        self.macd = self.macd_fast - self.macd_slow
        self.macd_signal = bt.ind.SMA(
            self.macd, period=self.p.macd_signal_period
        )

        self.crossover = bt.ind.CrossOver(self.fast_ma, self.slow_ma)
        self.trend_ma = bt.ind.SMA(self.data.close, period=self.p.trend_period)
        self.atr = bt.ind.ATR(self.data, period=self.p.atr_period)
        self.order = None

    def notify_order(self, order):
        if order.status in (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            self.order = None

    def _volatility_allowed(self):
        if not self.p.enable_volatility_filter:
            return True
        return self.p.volatility_min <= self.atr[0] <= self.p.volatility_max

    def _size(self):
        return (self.broker.getcash() * self.p.capital_fraction) / self.data.close[0]

    def next(self):
        if self.order:
            return

        price = self.data.close[0]

        if self.position:
            if self.position.size > 0:
                exit_signal = (
                    self.crossover[0] < 0
                    or price >= self.position.price * (1.0 + self.p.take_profit)
                    or price <= self.position.price * (1.0 - self.p.stop_loss)
                )
            else:
                exit_signal = (
                    self.crossover[0] > 0
                    or price <= self.position.price * (1.0 - self.p.take_profit)
                    or price >= self.position.price * (1.0 + self.p.stop_loss)
                )

            if exit_signal:
                self.order = self.close()
            return

        if not self._volatility_allowed():
            return

        size = self._size()
        if size <= 0:
            return

        if self.crossover[0] > 0:
            if not self.p.enable_trend_filter or price > self.trend_ma[0]:
                self.order = self.buy(size=size)
        elif self.crossover[0] < 0:
            if not self.p.enable_trend_filter or price < self.trend_ma[0]:
                self.order = self.sell(size=size)
