import math
import backtrader as bt


class SyntheticHedgeStrategy(bt.Strategy):
    def __init__(self):
        self.order = None

        self.sma = bt.indicators.SMA(self.data.close, period=50)
        self.atr = bt.indicators.ATR(self.data, period=14)
        self.bb50 = bt.indicators.BollingerBands(
            self.data.close, period=50, devfactor=2.0
        )
        self.bb20 = bt.indicators.BollingerBands(
            self.data.close, period=20, devfactor=2.0
        )
        self.macd = bt.indicators.MACD(
            self.data.close, period1=12, period2=26, period3=9
        )
        self.crossover = bt.indicators.CrossOver(self.data.close, self.sma)

    def notify_order(self, order):
        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            self.order = None

    def _get_units_net(self):
        close_price = self.data.close[0]
        if close_price <= 0:
            return 0

        sigma_long = (self.bb50.top[0] - self.bb50.mid[0]) / 2.0
        sigma_short = (self.bb20.top[0] - self.bb20.mid[0]) / 2.0

        if abs(sigma_short) < 1e-9:
            h = 0.0
        else:
            h = sigma_long / sigma_short
            h = max(0.0, min(1.5, h))

        units_main = math.floor(100000.0 / close_price)
        units_hedge = math.floor(units_main * h)
        units_net = units_main - units_hedge
        return units_net

    def next(self):
        if self.order:
            return

        close_price = self.data.close[0]
        if close_price <= 0:
            return

        atr_ratio = self.atr[0] / close_price

        long_filter = (
            (self.macd.macd[0] > self.macd.signal[0])
            and (self.macd.macd[0] > 0)
            and (atr_ratio >= 0.005)
        )
        short_filter = (
            (self.macd.macd[0] < self.macd.signal[0])
            and (self.macd.macd[0] < 0)
            and (atr_ratio >= 0.005)
        )

        pos_size = self.position.size

        if pos_size == 0:
            if self.crossover[0] > 0 and long_filter:
                units_net = self._get_units_net()
                self.order = self.order_target_size(target=units_net)
            elif self.crossover[0] < 0 and short_filter:
                units_net = self._get_units_net()
                self.order = self.order_target_size(target=-units_net)

        elif pos_size > 0:
            if self.crossover[0] < 0:
                self.order = self.close()
            else:
                if long_filter:
                    units_net = self._get_units_net()
                    self.order = self.order_target_size(target=units_net)

        elif pos_size < 0:
            if self.crossover[0] > 0:
                self.order = self.close()
            else:
                if short_filter:
                    units_net = self._get_units_net()
                    self.order = self.order_target_size(target=-units_net)
