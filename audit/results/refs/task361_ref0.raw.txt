import backtrader as bt


class MomentumTrendReversalStrategy(bt.Strategy):
    params = (
        ('momentum_period', 10),
        ('momentum_threshold', 5.0),
        ('sma_period', 50),
        ('atr_period', 14),
        ('stop_pct', 0.02),
        ('stop_method', 'max'),
        ('reverse_on_signal', False),
    )

    def __init__(self):
        self.mom = bt.indicators.RateOfChange100(
            self.data.close, period=self.p.momentum_period
        )
        self.sma = bt.indicators.SimpleMovingAverage(
            self.data.close, period=self.p.sma_period
        )
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.order = None
        self.entry_price = None

    def _long_stop(self):
        pct_stop = self.entry_price * (1 - self.p.stop_pct)
        atr_stop = self.entry_price - self.atr[0]
        if self.p.stop_method == 'percent':
            return pct_stop
        if self.p.stop_method == 'atr':
            return atr_stop
        return max(pct_stop, atr_stop)

    def _short_stop(self):
        pct_stop = self.entry_price * (1 + self.p.stop_pct)
        atr_stop = self.entry_price + self.atr[0]
        if self.p.stop_method == 'percent':
            return pct_stop
        if self.p.stop_method == 'atr':
            return atr_stop
        return min(pct_stop, atr_stop)

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return
        if order.status == order.Completed:
            self.entry_price = order.executed.price if self.position.size != 0 else None
        if order.status in (order.Completed, order.Canceled, order.Margin, order.Rejected):
            self.order = None

    def next(self):
        if self.order:
            return

        mom = self.mom[0]

        if self.position:
            if self.position.size > 0:
                opposite_signal = mom < -self.p.momentum_threshold
                stop_hit = self.data.close[0] <= self._long_stop()
                if opposite_signal or stop_hit:
                    self.order = self.close()
                    if opposite_signal and self.p.reverse_on_signal:
                        self.order = self.sell()
            elif self.position.size < 0:
                opposite_signal = mom > self.p.momentum_threshold
                stop_hit = self.data.close[0] >= self._short_stop()
                if opposite_signal or stop_hit:
                    self.order = self.close()
                    if opposite_signal and self.p.reverse_on_signal:
                        self.order = self.buy()
        else:
            if mom > self.p.momentum_threshold and self.data.close[0] > self.sma[0]:
                self.order = self.buy()
            elif mom < -self.p.momentum_threshold and self.data.close[0] < self.sma[0]:
                self.order = self.sell()
