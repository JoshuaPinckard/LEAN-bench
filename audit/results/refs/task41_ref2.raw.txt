import datetime
import backtrader as bt


class LeveragedETFLongTermStrategy(bt.Strategy):
    params = (
        ("sma_period", 200),
        ("leverage", 2.0),
        ("stop_loss_pct", 0.20),
        ("holding_years", 5),
        ("rebalance_days", 365),
    )

    def __init__(self):
        self.sma = bt.indicators.SimpleMovingAverage(
            self.data.close, period=self.p.sma_period
        )
        self.order = None
        self.entry_price = None
        self.entry_date = None
        self.last_rebalance_date = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            if self.position and self.entry_price is None:
                self.entry_price = order.executed.price
                self.entry_date = self.data.datetime.date(0)
                self.last_rebalance_date = self.data.datetime.date(0)

        self.order = None

    def next(self):
        if self.order:
            return

        current_date = self.data.datetime.date(0)
        current_price = self.data.close[0]

        if not self.position:
            self.entry_price = None
            self.entry_date = None
            self.last_rebalance_date = None

            if current_price > self.sma[0]:
                target_value = self.broker.getvalue() * self.p.leverage
                target_size = int(target_value / current_price)
                if target_size > 0:
                    self.order = self.buy(size=target_size)
        else:
            # Stop-loss check (20% drop from entry price / initial position value)
            if (
                self.entry_price is not None
                and current_price < (1.0 - self.p.stop_loss_pct) * self.entry_price
            ):
                self.order = self.close()
                return

            # Exit condition 1: Price falls below 200-day SMA
            if current_price < self.sma[0]:
                self.order = self.close()
                return

            # Exit condition 2: Fixed holding period of 5 years
            if self.entry_date is not None:
                days_held = (current_date - self.entry_date).days
                if days_held >= self.p.holding_years * 365:
                    self.order = self.close()
                    return

            # Annual rebalancing to maintain 2:1 leverage ratio
            if self.last_rebalance_date is not None:
                days_since_rebalance = (current_date - self.last_rebalance_date).days
                if days_since_rebalance >= self.p.rebalance_days:
                    target_value = self.broker.getvalue() * self.p.leverage
                    target_size = int(target_value / current_price)
                    if target_size != self.position.size:
                        self.order = self.order_target_size(target=target_size)
                        self.last_rebalance_date = current_date
