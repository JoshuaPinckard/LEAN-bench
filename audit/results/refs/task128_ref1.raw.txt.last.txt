import backtrader as bt


class StatisticalCapitulationBounce(bt.Strategy):
    def __init__(self):
        self.sma100 = bt.indicators.SimpleMovingAverage(self.data.close, period=100)
        self.rsi = bt.indicators.RSI(self.data.close, period=14)
        self.stochastic = bt.indicators.StochasticSlow(
            self.data, period=14, period_dfast=3, period_dslow=3
        )
        self.mfi = bt.indicators.MoneyFlowIndex(self.data, period=14)
        self.williams_r = bt.indicators.WilliamsR(self.data, period=14)
        self.cci = bt.indicators.CommodityChannelIndex(self.data, period=20)
        self.roc = bt.indicators.ROC(self.data.close, period=10)

        self.order = None
        self.deviation_history = []

    def next(self):
        if self.order:
            return

        deviation = self.data.close[0] - self.sma100[0]
        self.deviation_history.append(deviation)
        deviation_percentile = (
            sum(value <= deviation for value in self.deviation_history)
            / len(self.deviation_history)
        )

        if self.position:
            if self.data.close[-1] > self.sma100[-1] and self.data.close[0] < self.sma100[0]:
                self.order = self.close()
            return

        price_deviation_condition = deviation_percentile <= 0.02
        conditions = [
            price_deviation_condition,
            self.rsi[0] < 20,
            self.stochastic.percK[0] < 20,
            self.mfi[0] < 20,
            self.williams_r[0] < -80,
            self.cci[0] < -100,
            self.roc[0] < -10,
            self.data.close[0] < self.data.open[0],
            self.data.close[0] < self.data.close[-1],
        ]

        if price_deviation_condition and sum(conditions) >= 9:
            self.order = self.buy()

    def notify_order(self, order):
        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            self.order = None
