import backtrader as bt


class WilliamsRMomentumATRStrategy(bt.Strategy):
    params = (
        ('williams_period', 14),   # Williams %R period (spec)
        ('momentum_period', 14),   # Momentum period (spec)
        ('atr_period', 14),        # ATR period (spec)
        ('atr_mult', 2.0),         # adjustable ATR stop multiple (spec: adjustable; default chosen)
        ('trailing', False),       # False = static stop from entry (default); True = update each bar
        ('wr_long_level', -80.0),  # Williams %R cross-up level for LONG entry (spec)
        ('wr_short_level', -20.0), # Williams %R cross-down level for SHORT entry (spec)
    )

    def __init__(self):
        # Built-in Backtrader indicators, exactly the parameters named in the spec
        self.wr = bt.indicators.WilliamsR(self.data, period=self.p.williams_period)
        self.mom = bt.indicators.Momentum(self.data, period=self.p.momentum_period)
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)

        self.order = None        # pending-order guard
        self.entry_price = None  # fill price of the current position
        self.stop_price = None   # ATR protective stop level

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return

        if order.status == order.Completed:
            if self.position:
                # Completion established a position -> set the ATR protective stop
                self.entry_price = order.executed.price
                if self.position.size > 0:
                    # LONG: stop = entry_price - atr_mult * ATR
                    self.stop_price = self.entry_price - self.p.atr_mult * self.atr[0]
                else:
                    # SHORT: stop = entry_price + atr_mult * ATR
                    self.stop_price = self.entry_price + self.p.atr_mult * self.atr[0]
            else:
                # Completion flattened the position
                self.entry_price = None
                self.stop_price = None

        if order.status in (order.Completed, order.Canceled, order.Margin, order.Rejected):
            self.order = None

    def next(self):
        if self.order:
            return

        wr_prev = self.wr[-1]
        wr_now = self.wr[0]

        # (1) LONG: Williams %R crosses up through -80 AND Momentum > 0
        long_signal = (wr_prev <= self.p.wr_long_level and
                       wr_now > self.p.wr_long_level and
                       self.mom[0] > 0)

        # (2) SHORT: Williams %R crosses down through -20 AND Momentum < 0
        short_signal = (wr_prev >= self.p.wr_short_level and
                        wr_now < self.p.wr_short_level and
                        self.mom[0] < 0)

        if self.position:
            # Optional: update the ATR stop each bar as a trailing/volatility stop
            if self.p.trailing and self.entry_price is not None:
                if self.position.size > 0:
                    self.stop_price = self.entry_price - self.p.atr_mult * self.atr[0]
                else:
                    self.stop_price = self.entry_price + self.p.atr_mult * self.atr[0]

            # ATR-based protective stop
            if self.position.size > 0:
                if self.stop_price is not None and self.data.low[0] <= self.stop_price:
                    self.order = self.close()
                    return
            else:
                if self.stop_price is not None and self.data.high[0] >= self.stop_price:
                    self.order = self.close()
                    return

            # Exit: close open position when an opposite entry signal occurs
            if self.position.size > 0 and short_signal:
                self.order = self.close()
                return
            if self.position.size < 0 and long_signal:
                self.order = self.close()
                return
        else:
            # Flat: take the entry signal
            if long_signal:
                self.order = self.buy()
            elif short_signal:
                self.order = self.sell()
