import backtrader as bt


class IntradayBreakoutStrategy(bt.Strategy):
    def __init__(self):
        self.bb = bt.indicators.BollingerBands(
            self.data.close, period=20, devfactor=2.0
        )
        self.atr = bt.indicators.ATR(self.data, period=14)
        self.band_width = self.bb.top - self.bb.bot
        self.squeeze_threshold = bt.indicators.SMA(self.band_width, period=20)

        self.entry_order = None
        self.stop_order = None
        self.exit_order = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            if order == self.entry_order:
                self.entry_order = None
                entry_price = order.executed.price
                atr_val = self.atr[0]
                if self.position.size > 0:
                    stop_price = entry_price - 2.0 * atr_val
                    self.stop_order = self.sell(
                        exectype=bt.Order.Stop, price=stop_price
                    )
                elif self.position.size < 0:
                    stop_price = entry_price + 2.0 * atr_val
                    self.stop_order = self.buy(
                        exectype=bt.Order.Stop, price=stop_price
                    )
            elif order == self.stop_order:
                self.stop_order = None
            elif order == self.exit_order:
                self.exit_order = None

        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            if order == self.entry_order:
                self.entry_order = None
            elif order == self.stop_order:
                self.stop_order = None
            elif order == self.exit_order:
                self.exit_order = None

    def next(self):
        if not self.position:
            if self.entry_order is not None:
                return

            if self.band_width[0] < self.squeeze_threshold[0]:
                if (
                    self.data.close[0] > self.bb.top[0]
                    and self.data.close[-1] <= self.bb.top[-1]
                ):
                    self.entry_order = self.buy()
                elif (
                    self.data.close[0] < self.bb.bot[0]
                    and self.data.close[-1] >= self.bb.bot[-1]
                ):
                    self.entry_order = self.sell()

        else:
            if self.exit_order is not None:
                return

            if self.position.size > 0:
                if (
                    self.data.close[0] < self.bb.top[0]
                    and self.data.close[-1] >= self.bb.top[-1]
                ):
                    if self.stop_order:
                        self.cancel(self.stop_order)
                        self.stop_order = None
                    self.exit_order = self.close()
            elif self.position.size < 0:
                if (
                    self.data.close[0] > self.bb.bot[0]
                    and self.data.close[-1] <= self.bb.bot[-1]
                ):
                    if self.stop_order:
                        self.cancel(self.stop_order)
                        self.stop_order = None
                    self.exit_order = self.close()
