import backtrader as bt


class AccumulationBreakoutStrategy(bt.Strategy):
    params = (
        ('zone_low', 165.0),
        ('zone_high', 170.0),
        ('resistance', 240.0),
        ('swing_lag', 3),
        ('absorption_lookback', 3),
        ('risk_pct', 0.02),
    )

    def __init__(self):
        self.order = None
        self.bar_index = -1
        self.prev_month = None
        self.swing_lows = []  # (bar_index, low) points since last August anchor

    def _update_trendline_state(self):
        self.bar_index += 1
        dt = self.data.datetime.date(0)
        if self.prev_month is not None and dt.month == 8 and self.prev_month != 8:
            self.swing_lows = []
        self.prev_month = dt.month

        lag = self.p.swing_lag
        if self.bar_index >= 2 * lag:
            candidate_x = self.bar_index - lag
            window = [self.data.low[-i] for i in range(0, 2 * lag + 1)]
            candidate_low = self.data.low[-lag]
            if candidate_low == min(window):
                self.swing_lows.append((candidate_x, candidate_low))

    def _trendline_value(self):
        pts = self.swing_lows
        if len(pts) < 2:
            return None
        n = len(pts)
        mean_x = sum(p[0] for p in pts) / n
        mean_y = sum(p[1] for p in pts) / n
        num = sum((p[0] - mean_x) * (p[1] - mean_y) for p in pts)
        den = sum((p[0] - mean_x) ** 2 for p in pts)
        if den == 0:
            return None
        slope = num / den
        if slope <= 0:
            return None
        intercept = mean_y - slope * mean_x
        return slope * self.bar_index + intercept

    def _absorption_signal(self):
        n = min(self.p.absorption_lookback, len(self))
        for i in range(n):
            o = self.data.open[-i]
            c = self.data.close[-i]
            l = self.data.low[-i]
            body = abs(c - o)
            lower_wick = min(o, c) - l
            if c > o or lower_wick > body:
                return True
        return False

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        self._update_trendline_state()

        if self.order:
            return

        if self.position:
            if self.data.high[0] >= self.p.resistance:
                self.order = self.close()
            elif self.data.close[0] < self.p.zone_low:
                self.order = self.close()
            return

        zone_touch = (self.data.low[0] <= self.p.zone_high) and (self.data.high[0] >= self.p.zone_low)
        if not zone_touch:
            return

        trendline_value = self._trendline_value()
        if trendline_value is None or self.data.close[0] < trendline_value:
            return

        if not self._absorption_signal():
            return

        entry_price = self.data.close[0]
        risk_per_share = entry_price - self.p.zone_low
        if risk_per_share <= 0:
            return

        risk_amount = self.p.risk_pct * self.broker.getvalue()
        size = int(risk_amount / risk_per_share)
        if size <= 0:
            return

        self.order = self.buy(size=size)
