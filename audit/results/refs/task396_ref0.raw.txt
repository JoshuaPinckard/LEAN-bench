import backtrader as bt


class BollingerSqueezeBreakout(bt.Strategy):
    def __init__(self):
        self.boll = bt.indicators.BollingerBands(self.data, period=20, devfactor=2.0)
        self.atr = bt.indicators.ATR(self.data, period=14)

        self.band_width = self.boll.lines.top - self.boll.lines.bot
        self.squeeze_threshold = bt.indicators.SMA(self.band_width, period=20)

        self.cross_top = bt.indicators.CrossOver(self.data.close, self.boll.lines.top)
        self.cross_bot = bt.indicators.CrossOver(self.data.close, self.boll.lines.bot)

        self.order = None
        self.stop_order = None

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return

        if order.status == order.Completed:
            if order == self.order:
                self.order = None
                if order.isbuy():
                    stop_price = order.executed.price - 2 * self.atr[0]
                    self.stop_order = self.sell(exectype=bt.Order.Stop, price=stop_price)
                elif order.issell():
                    stop_price = order.executed.price + 2 * self.atr[0]
                    self.stop_order = self.buy(exectype=bt.Order.Stop, price=stop_price)
            elif order == self.stop_order:
                self.stop_order = None
        elif order.status in (order.Canceled, order.Margin, order.Rejected):
            if order == self.order:
                self.order = None
            elif order == self.stop_order:
                self.stop_order = None

    def next(self):
        if self.order:
            return

        squeeze = self.band_width[0] < self.squeeze_threshold[0]

        if not self.position:
            if squeeze and self.cross_top[0] > 0:
                self.order = self.buy()
            elif squeeze and self.cross_bot[0] < 0:
                self.order = self.sell()
        else:
            if self.position.size > 0 and self.cross_top[0] < 0:
                if self.stop_order:
                    self.cancel(self.stop_order)
                self.order = self.close()
            elif self.position.size < 0 and self.cross_bot[0] > 0:
                if self.stop_order:
                    self.cancel(self.stop_order)
                self.order = self.close()
