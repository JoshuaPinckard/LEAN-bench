import backtrader as bt


class SupertrendStrategy(bt.Strategy):
    params = (
        ('atr_period', 10),
        ('mult', 3.0),
    )

    def __init__(self):
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.order = None
        self.final_up = None
        self.final_dn = None
        self.trend = 1
        self.supertrend = None

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        hl2 = (self.data.high[0] + self.data.low[0]) / 2.0
        basic_up = hl2 - self.p.mult * self.atr[0]
        basic_dn = hl2 + self.p.mult * self.atr[0]

        old_up = self.final_up
        old_dn = self.final_dn
        old_trend = self.trend
        old_supertrend = self.supertrend

        if old_up is None:
            new_up = basic_up
        else:
            new_up = max(basic_up, old_up) if self.data.close[-1] > old_up else basic_up

        if old_dn is None:
            new_dn = basic_dn
        else:
            new_dn = min(basic_dn, old_dn) if self.data.close[-1] < old_dn else basic_dn

        new_trend = old_trend
        if old_trend == -1 and old_dn is not None and self.data.close[0] > old_dn:
            new_trend = 1
        elif old_trend == 1 and old_up is not None and self.data.close[0] < old_up:
            new_trend = -1

        pos_size = self.position.size if self.position else 0

        if not self.order and old_supertrend is not None:
            if self.data.close[0] > old_supertrend and pos_size <= 0:
                self.order = self.order_target_percent(target=1.0)
            elif self.data.close[0] < old_supertrend and pos_size >= 0:
                self.order = self.order_target_percent(target=-1.0)

        self.final_up = new_up
        self.final_dn = new_dn
        self.trend = new_trend
        self.supertrend = new_up if new_trend == 1 else new_dn
