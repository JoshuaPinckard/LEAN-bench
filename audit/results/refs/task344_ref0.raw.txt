import backtrader as bt


class DonchianSentimentBreakoutStrategy(bt.Strategy):
    params = dict(
        donchian_period=20,
        atr_period=14,
    )

    def __init__(self):
        self.donchian_upper = bt.indicators.Highest(
            self.data.high, period=self.p.donchian_period)
        self.donchian_lower = bt.indicators.Lowest(
            self.data.low, period=self.p.donchian_period)
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)

        self.cross_upper = bt.indicators.CrossOver(self.data.close, self.donchian_upper)
        self.cross_lower = bt.indicators.CrossOver(self.data.close, self.donchian_lower)

        self.order = None
        self.pending_side = None
        self.stop_price = None

    def notify_order(self, order):
        if order.status == order.Completed:
            if self.pending_side == 'long':
                self.stop_price = order.executed.price - 2 * self.atr[0]
            elif self.pending_side == 'short':
                self.stop_price = order.executed.price + 2 * self.atr[0]

        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        if self.order:
            return

        if not self.position:
            if self.cross_upper[0] > 0:
                self.pending_side = 'long'
                self.order = self.buy()
            elif self.cross_lower[0] < 0:
                self.pending_side = 'short'
                self.order = self.sell()
            return

        if self.position.size > 0:
            if self.data.low[0] <= self.stop_price or self.cross_lower[0] < 0:
                self.pending_side = None
                self.order = self.close()
        elif self.position.size < 0:
            if self.data.high[0] >= self.stop_price or self.cross_upper[0] > 0:
                self.pending_side = None
                self.order = self.close()
