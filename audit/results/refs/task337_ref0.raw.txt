import backtrader as bt


class AccumDistSmaStrategy(bt.Strategy):
    params = dict(
        sma_period=20,
        stop_pct=0.0,
    )

    def __init__(self):
        self.sma = bt.indicators.SMA(self.data.close, period=self.p.sma_period)
        self.ad = bt.indicators.AccumulationDistribution(self.data)
        self.order = None
        self.stop_order = None

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return

        if order.status == order.Completed:
            if order is self.stop_order:
                self.stop_order = None
            elif order is self.order:
                self.order = None
                if self.position and self.p.stop_pct > 0:
                    entry_price = order.executed.price
                    if order.isbuy():
                        stop_price = entry_price * (1 - self.p.stop_pct)
                        self.stop_order = self.sell(exectype=bt.Order.Stop, price=stop_price)
                    else:
                        stop_price = entry_price * (1 + self.p.stop_pct)
                        self.stop_order = self.buy(exectype=bt.Order.Stop, price=stop_price)
        elif order.status in (order.Canceled, order.Margin, order.Rejected):
            if order is self.stop_order:
                self.stop_order = None
            elif order is self.order:
                self.order = None

    def next(self):
        if self.order:
            return

        if not self.position:
            if self.data.close[0] > self.sma[0] and self.ad[0] > self.ad[-1]:
                self.order = self.buy()
            elif self.data.close[0] < self.sma[0] and self.ad[0] < self.ad[-1]:
                self.order = self.sell()
        else:
            exit_signal = False
            if self.position.size > 0:
                if self.ad[0] <= self.ad[-1]:
                    exit_signal = True
            elif self.position.size < 0:
                if self.ad[0] >= self.ad[-1]:
                    exit_signal = True

            if exit_signal:
                if self.stop_order is not None:
                    self.cancel(self.stop_order)
                    self.stop_order = None
                self.order = self.close()
