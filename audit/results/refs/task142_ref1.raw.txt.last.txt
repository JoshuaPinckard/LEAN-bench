import backtrader as bt


class NXXTConsolidationTrading(bt.Strategy):
    params = (
        ('support_level', 170.00),
        ('trend_lookback', 2),
        ('take_profit_pct', 0.02),
        ('risk_pct', 0.02),
    )

    def __init__(self):
        self.sma60 = bt.indicators.SimpleMovingAverage(self.data.close, period=60)
        self.sma120 = bt.indicators.SimpleMovingAverage(self.data.close, period=120)
        self.order = None
        self.entry_price = None

    def notify_order(self, order):
        if order.status in (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            if order.status == order.Completed and order.isbuy():
                self.entry_price = order.executed.price
            self.order = None

    def next(self):
        if self.order:
            return

        sma_up = (
            self.sma60[0] > self.sma60[-self.p.trend_lookback]
            and self.sma120[0] > self.sma120[-self.p.trend_lookback]
        )
        sma_down = (
            self.sma60[0] < self.sma60[-self.p.trend_lookback]
            and self.sma120[0] < self.sma120[-self.p.trend_lookback]
        )

        if not self.position:
            if (
                self.data.close[0] <= self.p.support_level
                and sma_up
                and self.data.close[0] > self.sma60[0]
                and self.data.close[0] > self.sma120[0]
            ):
                risk_per_share = self.data.close[0] - self.p.support_level
                if risk_per_share > 0:
                    size = int(
                        (self.broker.getvalue() * self.p.risk_pct) / risk_per_share
                    )
                    if size > 0:
                        self.order = self.buy(size=size)
            return

        take_profit = (
            self.entry_price is not None
            and self.data.close[0]
            >= self.entry_price * (1.0 + self.p.take_profit_pct)
        )

        if (
            self.data.close[0] < self.p.support_level
            or take_profit
            or sma_down
        ):
            self.order = self.close()
