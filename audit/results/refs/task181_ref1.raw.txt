import backtrader as bt


class VolatilityTradingStrategy(bt.Strategy):
    params = (
        ("N", 5),
        ("pct_up", 0.03),
        ("tp_pct", 0.06),
        ("sl_pct", 0.03),
        ("vol_thresh", 0.01),
        ("position_usdt", 100.0),
        ("enable_atr_filter", True),
        ("mode", "test"),
    )

    def __init__(self):
        self.pct_change = bt.indicators.PercentChange(
            self.data.close, period=self.p.N
        )
        self.atr = bt.indicators.ATR(self.data, period=14)

        self.entry_order = None
        self.exit_orders = []

    def next(self):
        if self.position or self.entry_order or self.exit_orders:
            return

        atr_filter_passes = (
            not self.p.enable_atr_filter
            or (self.atr[0] / self.data.close[0] >= self.p.vol_thresh)
        )

        if self.pct_change[0] >= self.p.pct_up and atr_filter_passes:
            size = self.p.position_usdt / self.data.close[0]
            self.entry_order = self.buy(size=size)

    def notify_order(self, order):
        if order.status not in (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            return

        role = order.info.get("role")

        if role == "entry":
            self.entry_order = None

            if order.status == order.Completed:
                size = order.executed.size
                entry_price = order.executed.price

                stop_order = self.sell(
                    size=size,
                    exectype=bt.Order.Stop,
                    price=entry_price * (1.0 - self.p.sl_pct),
                )
                stop_order.addinfo(role="exit")

                limit_order = self.sell(
                    size=size,
                    exectype=bt.Order.Limit,
                    price=entry_price * (1.0 + self.p.tp_pct),
                    oco=stop_order,
                )
                limit_order.addinfo(role="exit")

                self.exit_orders = [stop_order, limit_order]

        elif role == "exit":
            if order.status == order.Completed:
                self.exit_orders = []
            else:
                self.exit_orders = [
                    pending for pending in self.exit_orders if pending is not order
                ]

    def start(self):
        pass

    def prenext(self):
        self.next()

    def notify_trade(self, trade):
        pass

    def buy(self, *args, **kwargs):
        order = super().buy(*args, **kwargs)
        order.addinfo(role="entry")
        return order
