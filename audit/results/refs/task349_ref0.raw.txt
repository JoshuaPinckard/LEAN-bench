```python
import backtrader as bt


class DonchianVolumeBreakoutStrategy(bt.Strategy):
    params = (
        ('period', 20),
        ('vol_sma_period', 20),
        ('stop_percent', 0.02),
    )

    def __init__(self):
        self.donchian_high = bt.indicators.Highest(self.data.high(-1), period=self.p.period)
        self.donchian_low = bt.indicators.Lowest(self.data.low(-1), period=self.p.period)
        self.vol_sma = bt.indicators.SMA(self.data.volume, period=self.p.vol_sma_period)

        self.cross_up = bt.indicators.CrossOver(self.data.close, self.donchian_high)
        self.cross_down = bt.indicators.CrossOver(self.data.close, self.donchian_low)

        self.order = None
        self.stop_order = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order == self.stop_order:
            self.stop_order = None

        if order == self.order:
            self.order = None
            if order.status != order.Completed and self.stop_order is not None:
                self.cancel(self.stop_order)
                self.stop_order = None

    def next(self):
        if self.order:
            return

        vol_confirmed = (self.data.volume[0] > self.data.volume[-1]
                          and self.data.volume[0] > self.vol_sma[0])

        if not self.position:
            if self.cross_up[0] > 0 and vol_confirmed:
                entry = self.buy()
                self.order = entry
                stop_price = self.donchian_high[0] * (1 - self.p.stop_percent)
                self.stop_order = self.sell(exectype=bt.Order.Stop, price=stop_price, size=abs(entry.size))
            elif self.cross_down[0] < 0 and vol_confirmed:
                entry = self.sell()
                self.order = entry
                stop_price = self.donchian_low[0] * (1 + self.p.stop_percent)
                self.stop_order = self.buy(exectype=bt.Order.Stop, price=stop_price, size=abs(entry.size))
        elif self.position.size > 0:
            if self.data.close[0] < self.donchian_high[0] or self.data.volume[0] < self.vol_sma[0]:
                if self.stop_order:
                    self.cancel(self.stop_order)
                    self.stop_order = None
                self.order = self.close()
        elif self.position.size < 0:
            if self.data.close[0] > self.donchian_low[0] or self.data.volume[0] < self.vol_sma[0]:
                if self.stop_order:
                    self.cancel(self.stop_order)
                    self.stop_order = None
                self.order = self.close()
```
