import math
import backtrader as bt


class TDP(bt.Indicator):
    lines = ('tdp',)
    params = (
        ('N', 100),
        ('decay', 0.05),
    )

    def __init__(self):
        self.addminperiod(self.p.N)

    def next(self):
        N = self.p.N
        decay = self.p.decay
        current_close = self.data.close[0]

        weighted_count = 0.0
        total_weight = 0.0

        for k in range(N):
            w = math.exp(-decay * k)
            total_weight += w
            if self.data.close[-k] <= current_close:
                weighted_count += w

        self.lines.tdp[0] = (weighted_count / total_weight) * 100.0 if total_weight > 0 else 50.0


class TDPStrategy(bt.Strategy):
    params = (
        ('N', 100),
        ('decay', 0.05),
        ('ema_period', 22),
        ('long_oversold', 30.0),
        ('long_extreme', 15.0),
        ('short_overbought', 70.0),
        ('short_extreme', 85.0),
        ('exit_long_mid', 50.0),
        ('exit_long_high', 70.0),
        ('exit_short_mid', 50.0),
        ('exit_short_low', 30.0),
        ('D', 20),
        ('use_divergence', False),
        ('stop_type', 'percent'),  # 'percent' or 'atr'
        ('stop_pct', 0.05),
        ('atr_period', 14),
        ('atr_mult', 2.0),
    )

    def __init__(self):
        self.tdp = TDP(self.data.close, N=self.p.N, decay=self.p.decay)
        self.tdp_ema = bt.indicators.EMA(self.tdp, period=self.p.ema_period)
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)

        self.order = None
        self.entry_price = None
        self.stop_price = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            self.entry_price = order.executed.price
            if self.position.size > 0:
                if self.p.stop_type == 'atr':
                    self.stop_price = self.entry_price - self.p.atr_mult * self.atr[0]
                else:
                    self.stop_price = self.entry_price * (1.0 - self.p.stop_pct)
            elif self.position.size < 0:
                if self.p.stop_type == 'atr':
                    self.stop_price = self.entry_price + self.p.atr_mult * self.atr[0]
                else:
                    self.stop_price = self.entry_price * (1.0 + self.p.stop_pct)
            else:
                self.entry_price = None
                self.stop_price = None
            self.order = None
        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        if self.order:
            return

        tdp_curr = self.tdp_ema[0]
        tdp_prev = self.tdp_ema[-1]

        if self.position.size > 0:
            if self.stop_price is not None and self.data.low[0] <= self.stop_price:
                self.order = self.close()
                return

            cross_down_50 = (tdp_prev > self.p.exit_long_mid and tdp_curr <= self.p.exit_long_mid)
            cross_up_70 = (tdp_prev < self.p.exit_long_high and tdp_curr >= self.p.exit_long_high)
            if cross_down_50 or cross_up_70:
                self.order = self.close()
                return

        elif self.position.size < 0:
            if self.stop_price is not None and self.data.high[0] >= self.stop_price:
                self.order = self.close()
                return

            cross_up_50 = (tdp_prev < self.p.exit_short_mid and tdp_curr >= self.p.exit_short_mid)
            cross_down_30 = (tdp_prev > self.p.exit_short_low and tdp_curr <= self.p.exit_short_low)
            if cross_up_50 or cross_down_30:
                self.order = self.close()
                return

        else:
            cross_up_30 = (tdp_prev < self.p.long_oversold and tdp_curr >= self.p.long_oversold)
            cross_up_15 = (tdp_prev < self.p.long_extreme and tdp_curr >= self.p.long_extreme)
            long_signal = cross_up_30 or cross_up_15

            cross_down_70 = (tdp_prev > self.p.short_overbought and tdp_curr <= self.p.short_overbought)
            cross_down_85 = (tdp_prev > self.p.short_extreme and tdp_curr <= self.p.short_extreme)
            short_signal = cross_down_70 or cross_down_85

            if self.p.use_divergence:
                D = self.p.D
                half_D = D // 2

                min_close_older = min(self.data.close[i] for i in range(-D + 1, -half_D + 1))
                min_close_recent = min(self.data.close[i] for i in range(-half_D + 1, 1))
                min_tdp_older = min(self.tdp_ema[i] for i in range(-D + 1, -half_D + 1))
                min_tdp_recent = min(self.tdp_ema[i] for i in range(-half_D + 1, 1))
                bullish_div = (min_close_recent < min_close_older) and (min_tdp_recent > min_tdp_older)

                max_close_older = max(self.data.close[i] for i in range(-D + 1, -half_D + 1))
                max_close_recent = max(self.data.close[i] for i in range(-half_D + 1, 1))
                max_tdp_older = max(self.tdp_ema[i] for i in range(-D + 1, -half_D + 1))
                max_tdp_recent = max(self.tdp_ema[i] for i in range(-half_D + 1, 1))
                bearish_div = (max_close_recent > max_close_older) and (max_tdp_recent < max_tdp_older)

                long_signal = long_signal and bullish_div
                short_signal = short_signal and bearish_div

            if long_signal:
                self.order = self.buy()
            elif short_signal:
                self.order = self.sell()
