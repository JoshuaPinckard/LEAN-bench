import backtrader as bt
import collections


class FVGSwingFailureBreakoutStrategy(bt.Strategy):
    # Parameters (values chosen only where the spec is silent; see assumptions).
    params = dict(
        swing_lookback=1,     # fractal pivot half-width for 4h swing highs/lows
        flag_window=288,      # 24h (in 5m bars): validity of the 4h setup conditions
        ob_body_mult=1.5,     # "strong candle" body vs recent average body
        ob_avg_period=20,     # lookback for the average body used above
        ob_window=48,         # order block must be within 4h of the breakout
        rr=1000.0 / 300.0,    # 1:3.33 reward-to-risk (from the $300 -> $1000 example)
        risk_pct=0.01,        # 1% of capital risked per trade
        max_hold_bars=48,     # 4 hours maximum holding time
    )

    def __init__(self):
        # ---- 4h aggregation state (single 5m feed resampled internally) ----
        self.cur_bucket = None
        self.h4_open = self.h4_high = self.h4_low = self.h4_close = None
        self.h4o, self.h4h, self.h4l, self.h4c = [], [], [], []

        # ---- 4h swing state ----
        self.last_swing_high = None
        self.prev_swing_high = None
        self.last_swing_low = None
        self.prev_swing_low = None

        # ---- setup condition timestamps (in 5m bar index) ----
        self.fvg_zone = None          # (low, high) of most recent bullish FVG
        self.fvg_enter_bar = None     # price entered the FVG zone
        self.swing_fail_bar = None    # bullish swing-failure formed
        self.mss_bar = None           # bullish market structure shift formed

        # ---- order block state (5m) ----
        self.ob_top = None
        self.ob_bottom = None
        self.ob_bar = None
        self.bodies = collections.deque(maxlen=self.p.ob_avg_period)
        self.prev_close = None

        # ---- trade state ----
        self.order = None
        self.entry_price = None
        self.stop_price = None
        self.tp1_price = None
        self.tp2_price = None
        self.half_size = 0.0
        self.tp1_done = False
        self.entry_bar = None
        self.exiting = False

        # ---- bar counter ----
        self.bar_i = -1

    # ------------------------------------------------------------------
    def _reset_trade(self):
        self.entry_price = None
        self.stop_price = None
        self.tp1_price = None
        self.tp2_price = None
        self.half_size = 0.0
        self.tp1_done = False
        self.entry_bar = None
        self.exiting = False

    def _valid(self, stamp):
        return stamp is not None and (self.bar_i - stamp) <= self.p.flag_window

    # ------------------------------------------------------------------
    def _process_4h_bar(self):
        n = len(self.h4c)
        lb = self.p.swing_lookback

        # --- 4h swing detection (fractal pivot) ---
        if n >= 2 * lb + 1:
            i = n - 1 - lb
            is_high = (all(self.h4h[i] > self.h4h[i - k] for k in range(1, lb + 1)) and
                       all(self.h4h[i] > self.h4h[i + k] for k in range(1, lb + 1)))
            is_low = (all(self.h4l[i] < self.h4l[i - k] for k in range(1, lb + 1)) and
                      all(self.h4l[i] < self.h4l[i + k] for k in range(1, lb + 1)))
            if is_high:
                self.prev_swing_high = self.last_swing_high
                self.last_swing_high = self.h4h[i]
                # bullish swing failure: higher high with an immediate lower-high pullback bar
                if self.prev_swing_high is not None and self.last_swing_high > self.prev_swing_high:
                    self.swing_fail_bar = self.bar_i
            if is_low:
                self.prev_swing_low = self.last_swing_low
                self.last_swing_low = self.h4l[i]

        # --- FVG detection (bullish: bull candle then bear candle, non-overlapping bodies) ---
        if n >= 2:
            o2, c2, h2 = self.h4o[-2], self.h4c[-2], self.h4h[-2]
            o1, c1, l1 = self.h4o[-1], self.h4c[-1], self.h4l[-1]
            bull_first = c2 > o2
            bear_second = c1 < o1
            if bull_first and bear_second and c2 < c1:  # bodies do not overlap (upward gap)
                low = min(h2, l1)
                high = max(h2, l1)
                if high > low:
                    self.fvg_zone = (low, high)

        # --- MSS (bullish): a 4h close breaks below the last confirmed swing low ---
        if self.last_swing_low is not None and self.h4c[-1] < self.last_swing_low:
            self.mss_bar = self.bar_i

    # ------------------------------------------------------------------
    def next(self):
        self.bar_i += 1
        dt = self.data.datetime.datetime(0)
        o = self.data.open[0]
        h = self.data.high[0]
        l = self.data.low[0]
        c = self.data.close[0]

        # ---------- 1) 4h aggregation ----------
        bucket = (dt.year, dt.month, dt.day, dt.hour // 4)
        if self.cur_bucket is None:
            self.cur_bucket = bucket
            self.h4_open, self.h4_high, self.h4_low, self.h4_close = o, h, l, c
        elif bucket != self.cur_bucket:
            self.h4o.append(self.h4_open)
            self.h4h.append(self.h4_high)
            self.h4l.append(self.h4_low)
            self.h4c.append(self.h4_close)
            self._process_4h_bar()
            self.cur_bucket = bucket
            self.h4_open, self.h4_high, self.h4_low, self.h4_close = o, h, l, c
        else:
            self.h4_high = max(self.h4_high, h)
            self.h4_low = min(self.h4_low, l)
            self.h4_close = c

        # ---------- 2) price enters the FVG zone ----------
        if self.fvg_zone is not None:
            zlow, zhigh = self.fvg_zone
            if l <= zhigh and h >= zlow:
                self.fvg_enter_bar = self.bar_i

        # ---------- 3) manage an open position ----------
        if self.position and not self.exiting:
            bars_open = self.bar_i - self.entry_bar if self.entry_bar is not None else 0
            if self.stop_price is not None and l <= self.stop_price:          # stop-loss
                self.close()
                self.exiting = True
            elif bars_open > self.p.max_hold_bars:                             # max holding time
                self.close()
                self.exiting = True
            elif self.tp2_price is not None and h >= self.tp2_price:           # 100% take-profit
                self.close()
                self.exiting = True
            elif (not self.tp1_done) and self.tp1_price is not None and h >= self.tp1_price:
                self.sell(size=self.half_size)                                # 50% partial take-profit
                self.tp1_done = True

        # ---------- 4) entry ----------
        can_enter = ((not self.position) and self.order is None and
                     (not self.exiting) and dt.weekday() < 5)
        if (can_enter and self._valid(self.fvg_enter_bar) and
                self._valid(self.swing_fail_bar) and self._valid(self.mss_bar)):
            if (self.ob_top is not None and self.ob_bar is not None and
                    (self.bar_i - self.ob_bar) <= self.p.ob_window and
                    self.prev_close is not None and
                    self.prev_close <= self.ob_top and c > self.ob_top):       # upward OB breakout
                stop = self.ob_bottom
                per_unit = c - stop
                if per_unit > 0:
                    value = self.broker.getvalue()
                    cash = self.broker.getcash()
                    size = (self.p.risk_pct * value) / per_unit
                    max_aff = 0.995 * cash / c if c > 0 else 0.0
                    if size > max_aff:
                        size = max_aff
                    if size > 0:
                        self.entry_price = c
                        self.stop_price = stop
                        self.tp1_price = c + 0.5 * self.p.rr * per_unit
                        self.tp2_price = c + self.p.rr * per_unit
                        self.half_size = size / 2.0
                        self.tp1_done = False
                        self.entry_bar = self.bar_i
                        self.order = self.buy(size=size)
                        # one trade per confirmed setup
                        self.fvg_enter_bar = None
                        self.swing_fail_bar = None
                        self.mss_bar = None

        # ---------- 5) update order block candidate from this bar ----------
        body = abs(c - o)
        if len(self.bodies) >= self.p.ob_avg_period:
            avg_body = sum(self.bodies) / len(self.bodies)
            if c < o and body >= self.p.ob_body_mult * avg_body:  # last strong bearish candle
                self.ob_top = o
                self.ob_bottom = c
                self.ob_bar = self.bar_i
        self.bodies.append(body)
        self.prev_close = c

    # ------------------------------------------------------------------
    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return
        if order.status == order.Completed:
            if order is self.order:
                self.order = None
        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            if order is self.order:
                self.order = None
                if not self.position:
                    self._reset_trade()

    def notify_trade(self, trade):
        if trade.isclosed:
            self._reset_trade()
