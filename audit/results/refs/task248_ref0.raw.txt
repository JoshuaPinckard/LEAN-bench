import backtrader as bt


class LiquidityShockProxyStrategy(bt.Strategy):
    params = dict(
        s_period=20,
        vol_fast_period=10,
        vol_slow_period=20,
        atr_period=10,
        atr_sma_period=50,
        contraction_mult=0.8,
        expansion_mult=1.2,
        vol_ratio_mult=1.5,
        price_change_thresh=0.001,
        take_profit_pct=0.005,
        stop_loss_pct=0.003,
        risk_pct=0.01,
        max_lots=3,
    )

    def __init__(self):
        self.s = (self.data.high - self.data.low) / self.data.close
        self.sma_s = bt.indicators.SMA(self.s, period=self.p.s_period)
        self.sma_vol_fast = bt.indicators.SMA(self.data.volume, period=self.p.vol_fast_period)
        self.sma_vol_slow = bt.indicators.SMA(self.data.volume, period=self.p.vol_slow_period)
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.sma_atr = bt.indicators.SMA(self.atr, period=self.p.atr_sma_period)
        self.cross_s = bt.indicators.CrossOver(self.s, self.sma_s)

        self.lots = []
        self._lot_seq = 0
        self.exit_refs = {}

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return

        if order.status == order.Completed:
            if order.ref in self.exit_refs:
                lot_id = self.exit_refs.pop(order.ref)
                self.lots = [lot for lot in self.lots if lot['id'] != lot_id]
            else:
                direction = 1 if order.isbuy() else -1
                self._lot_seq += 1
                self.lots.append({
                    'id': self._lot_seq,
                    'entry_price': order.executed.price,
                    'size': abs(order.executed.size),
                    'direction': direction,
                    'order_ref': None,
                })
        else:
            lot_id = self.exit_refs.pop(order.ref, None)
            if lot_id is not None:
                for lot in self.lots:
                    if lot['id'] == lot_id:
                        lot['order_ref'] = None
                        break

    def next(self):
        price = self.data.close[0]

        if self.lots:
            direction = self.lots[0]['direction']

            crossed_exit = (direction == 1 and self.cross_s[0] > 0) or \
                           (direction == -1 and self.cross_s[0] < 0)
            if crossed_exit:
                for lot in self.lots:
                    if lot['order_ref'] is not None:
                        continue
                    order = self.close(size=lot['size'])
                    lot['order_ref'] = order.ref
                    self.exit_refs[order.ref] = lot['id']
                return

            for lot in self.lots:
                if lot['order_ref'] is not None:
                    continue
                entry = lot['entry_price']
                if direction == 1:
                    hit_tp = price >= entry * (1 + self.p.take_profit_pct)
                    hit_sl = price <= entry * (1 - self.p.stop_loss_pct)
                else:
                    hit_tp = price <= entry * (1 - self.p.take_profit_pct)
                    hit_sl = price >= entry * (1 + self.p.stop_loss_pct)
                if hit_tp or hit_sl:
                    order = self.close(size=lot['size'])
                    lot['order_ref'] = order.ref
                    self.exit_refs[order.ref] = lot['id']

        if len(self.lots) >= self.p.max_lots:
            return

        s0 = self.s[0]
        sma_s0 = self.sma_s[0]
        price_change = abs(self.data.close[0] - self.data.close[-1]) / self.data.close[-1]

        vol_cond = self.sma_vol_fast[0] > self.p.vol_ratio_mult * self.sma_vol_slow[0]
        atr_cond = self.atr[0] < self.sma_atr[0]
        price_cond = price_change <= self.p.price_change_thresh

        long_cond = s0 < self.p.contraction_mult * sma_s0 and vol_cond and atr_cond and price_cond
        short_cond = s0 > self.p.expansion_mult * sma_s0 and vol_cond and atr_cond and price_cond

        if not (long_cond or short_cond):
            return

        if self.lots:
            existing_direction = self.lots[0]['direction']
            if (existing_direction == 1 and not long_cond) or \
               (existing_direction == -1 and not short_cond):
                return

        size = (self.p.risk_pct * self.broker.getvalue()) / price
        if size <= 0:
            return

        if long_cond:
            self.buy(size=size)
        else:
            self.sell(size=size)
