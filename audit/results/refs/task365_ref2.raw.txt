import backtrader as bt


class FibonacciGoldenPocketStrategy(bt.Strategy):
    params = (
        ('swing_period', 200),
        ('exit_period', 20),
    )

    def __init__(self):
        self.swing_high = bt.indicators.Highest(self.data.high, period=self.p.swing_period)
        self.swing_low = bt.indicators.Lowest(self.data.low, period=self.p.swing_period)
        self.exit_high = bt.indicators.Highest(self.data.high, period=self.p.exit_period)

        self.order = None
        self.state = 0
        self.last_swing = None
        self.fired_in_swing = False

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        highs = self.data.high.get(size=self.p.swing_period)
        lows = self.data.low.get(size=self.p.swing_period)

        if len(highs) < self.p.swing_period or len(lows) < self.p.swing_period:
            return

        sh = self.swing_high[0]
        sl = self.swing_low[0]

        idx_high = max(i for i, v in enumerate(highs) if v == sh)
        idx_low = max(i for i, v in enumerate(lows) if v == sl)

        if idx_high < idx_low:
            level_618 = sh - 0.618 * (sh - sl)
            level_786 = sh - 0.786 * (sh - sl)
        else:
            level_618 = sl + 0.618 * (sh - sl)
            level_786 = sl + 0.786 * (sh - sl)

        current_swing = (sh, sl)
        if current_swing != self.last_swing:
            self.last_swing = current_swing
            self.state = 0
            self.fired_in_swing = False

        if self.order is not None:
            return

        if self.position:
            if self.data.close[0] >= self.exit_high[0]:
                self.order = self.close()
        else:
            if not self.fired_in_swing:
                if self.state == 0:
                    if self.data.close[0] < level_618 or self.data.low[0] <= level_618:
                        self.state = 1

                if self.state == 1:
                    if self.data.low[0] <= level_786:
                        self.state = 2

                if self.state == 2:
                    if self.data.close[0] > level_618:
                        self.state = 3
                        self.fired_in_swing = True
                        self.order = self.buy()
