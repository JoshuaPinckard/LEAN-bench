import backtrader as bt


class MACDAdaptiveBandsStrategy(bt.Strategy):
    params = (
        ("macd_me1", 12),
        ("macd_me2", 26),
        ("macd_signal", 9),
        ("band_period", 20),
        ("band_dev", 2.0),
        ("atr_period", 14),
    )

    def __init__(self):
        self.macd = bt.indicators.MACDHisto(
            self.data,
            period_me1=self.p.macd_me1,
            period_me2=self.p.macd_me2,
            period_signal=self.p.macd_signal,
        )

        self.sma_hist = bt.indicators.SMA(
            self.macd.histo, period=self.p.band_period
        )
        self.std_hist = bt.indicators.StandardDeviation(
            self.macd.histo, period=self.p.band_period
        )

        self.upper = self.sma_hist + self.p.band_dev * self.std_hist
        self.lower = self.sma_hist - self.p.band_dev * self.std_hist

        self.cross_macd = bt.indicators.CrossOver(
            self.macd.macd, self.macd.signal
        )
        self.cross_histo_upper = bt.indicators.CrossOver(
            self.macd.histo, self.upper
        )
        self.cross_histo_lower = bt.indicators.CrossOver(
            self.macd.histo, self.lower
        )

        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)

        self.order = None
        self.stop_order = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            if order == self.order:
                self.order = None
                if self.position.size > 0:
                    entry_price = order.executed.price
                    atr_val = self.atr[0]
                    stop_price = max(
                        entry_price * 0.98, entry_price - 2.0 * atr_val
                    )
                    self.stop_order = self.sell(
                        exectype=bt.Order.Stop, price=stop_price
                    )
                elif self.position.size < 0:
                    entry_price = order.executed.price
                    atr_val = self.atr[0]
                    stop_price = min(
                        entry_price * 1.02, entry_price + 2.0 * atr_val
                    )
                    self.stop_order = self.buy(
                        exectype=bt.Order.Stop, price=stop_price
                    )
                else:
                    if self.stop_order:
                        self.cancel(self.stop_order)
                        self.stop_order = None

            elif order == self.stop_order:
                self.stop_order = None

        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            if order == self.order:
                self.order = None
            elif order == self.stop_order:
                self.stop_order = None

    def next(self):
        if self.order:
            return

        long_setup = (self.cross_macd[0] > 0) and (
            self.cross_histo_upper[0] > 0
        )
        short_setup = (self.cross_macd[0] < 0) and (
            self.cross_histo_lower[0] < 0
        )

        if self.position.size > 0:
            if short_setup:
                if self.stop_order:
                    self.cancel(self.stop_order)
                    self.stop_order = None
                self.order = self.close()
        elif self.position.size < 0:
            if long_setup:
                if self.stop_order:
                    self.cancel(self.stop_order)
                    self.stop_order = None
                self.order = self.close()
        else:
            if long_setup:
                self.order = self.buy()
            elif short_setup:
                self.order = self.sell()
