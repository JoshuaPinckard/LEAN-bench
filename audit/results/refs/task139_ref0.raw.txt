import backtrader as bt


class InitialPositionStrategy(bt.Strategy):
    def __init__(self):
        self.sma = bt.indicators.SimpleMovingAverage(self.data.close, period=20)
        self.order = None
        self.entry_count = 0

    def start(self):
        # No public broker.set_position() API exists; the documented workaround
        # is to inject a Position object directly, bypassing cash accounting.
        self.broker.positions[self.data] = bt.Position(size=10, price=150.0)

    def notify_order(self, order):
        if order.status == order.Completed:
            if order.isbuy():
                self.entry_count += 1
            else:
                self.entry_count = 0
        if order.status in (order.Completed, order.Canceled, order.Margin, order.Rejected):
            self.order = None

    def next(self):
        if self.order:
            return

        close = self.data.close[0]
        sma = self.sma[0]

        if close < sma:
            self.order = self.close()
        elif close > sma:
            if self.entry_count < 3:
                size = int((self.broker.getvalue() * 0.05) / close)
                if size > 0:
                    self.order = self.buy(size=size)
