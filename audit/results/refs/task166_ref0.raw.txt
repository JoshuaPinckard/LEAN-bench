import math

import backtrader as bt


class IntraRangeMomentumTripleBarrier(bt.Strategy):
    params = dict(
        sma_period=20,
        stoch_period=14,
        stoch_pfast=3,
        stoch_pslow=3,
        bb_period=20,
        bb_devfactor=2.0,
        atr_period=14,
        stoch_upper=66,
        stoch_lower=33,
        range_pct=0.002,
        atr_pct=0.0015,
        time_stop=20,
        risk_pct=0.01,
    )

    def __init__(self):
        self.sma = bt.indicators.SMA(self.data, period=self.p.sma_period)
        self.stoch = bt.indicators.Stochastic(
            self.data,
            period=self.p.stoch_period,
            period_dfast=self.p.stoch_pfast,
            period_dslow=self.p.stoch_pslow,
        )
        self.bb = bt.indicators.BollingerBands(
            self.data, period=self.p.bb_period, devfactor=self.p.bb_devfactor
        )
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.stoch_cross = bt.indicators.CrossOver(self.stoch.percK, self.stoch.percD)
        self.sma_cross = bt.indicators.CrossOver(self.data.close, self.sma)

        self.order = None
        self.bar_executed = None

    def notify_order(self, order):
        if order.status in (order.Completed, order.Canceled, order.Margin, order.Rejected):
            if order.status == order.Completed:
                self.bar_executed = len(self)
            self.order = None

    def next(self):
        if self.order:
            return

        close = self.data.close[0]
        high = self.data.high[0]
        low = self.data.low[0]

        daily_range_pct = (high - low) / close
        atr_pct = self.atr[0] / close

        if not self.position:
            if daily_range_pct < self.p.range_pct or atr_pct <= self.p.atr_pct:
                return

            long_signal = (
                close > self.sma[0]
                and self.stoch.percK[0] > self.p.stoch_upper
                and self.stoch_cross[0] > 0
            )
            short_signal = (
                close < self.sma[0]
                and self.stoch.percK[0] < self.p.stoch_lower
                and self.stoch_cross[0] < 0
            )

            if long_signal:
                size = math.floor((self.broker.getvalue() * self.p.risk_pct) / close)
                if size > 0:
                    self.order = self.buy(size=size)
            elif short_signal:
                size = math.floor((self.broker.getvalue() * self.p.risk_pct) / close)
                if size > 0:
                    self.order = self.sell(size=size)
        else:
            bars_held = len(self) - self.bar_executed if self.bar_executed is not None else 0

            if self.position.size > 0:
                take_profit = high >= self.bb.top[0]
                stop = low <= self.bb.bot[0]
                sma_cross_exit = self.sma_cross[0] < 0
                time_stop = bars_held >= self.p.time_stop

                if take_profit or stop or sma_cross_exit or time_stop:
                    self.order = self.close()

            elif self.position.size < 0:
                take_profit = low <= self.bb.bot[0]
                stop = high >= self.bb.top[0]
                sma_cross_exit = self.sma_cross[0] > 0
                time_stop = bars_held >= self.p.time_stop

                if take_profit or stop or sma_cross_exit or time_stop:
                    self.order = self.close()
