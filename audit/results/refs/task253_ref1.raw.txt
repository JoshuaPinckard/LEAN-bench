import backtrader as bt
import math


class PanicEuphoria(bt.Strategy):
    params = (
        ("position_fraction", 1.0),
    )

    def __init__(self):
        self.sma200 = bt.indicators.SMA(self.data.close, period=200)
        self.boll = bt.indicators.BollingerBands(
            self.data.close, period=20, devfactor=2
        )
        self.rsi = bt.indicators.RSI(self.data.close, period=14)
        self.atr = bt.indicators.ATR(self.data, period=14)

        self.atr_sma50 = bt.indicators.SMA(self.atr, period=50)
        self.width = self.boll.top - self.boll.bot
        self.width_sma50 = bt.indicators.SMA(self.width, period=50)

        self.order = None

    def notify_order(self, order):
        if order.status in (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            self.order = None

    def next(self):
        if self.order:
            return

        elevated_volatility = (
            self.atr[0] > self.atr_sma50[0]
            or self.width[0] > self.width_sma50[0]
        )
        decreasing_volatility = (
            self.atr[0] < self.atr_sma50[0]
            or self.width[0] < self.width_sma50[0]
        )

        entry_signal = (
            self.data.close[0] > self.sma200[0]
            and self.data.close[0] < self.boll.bot[0]
            and self.rsi[0] <= 30
            and elevated_volatility
        )
        exit_signal = (
            self.data.close[0] > self.boll.top[0]
            and self.rsi[0] >= 65
            and decreasing_volatility
        )

        if not self.position:
            if entry_signal:
                cash = self.broker.getcash()
                size = math.floor(
                    (cash * self.p.position_fraction)
                    / (self.data.close[0] * (1.0 + self.broker.getcommissioninfo(self.data).p.commission))
                )
                if size > 0:
                    self.order = self.buy(size=size)
        elif exit_signal:
            self.order = self.close()
