import backtrader as bt


class MacdIntradayStrategy(bt.Strategy):
    params = (
        ("fast_ema", 12),
        ("slow_ema", 26),
        ("signal_period", 9),
        ("stop_loss_pct", 0.02),
    )

    def __init__(self):
        self.macd = bt.indicators.MACD(
            self.data,
            period_me1=self.p.fast_ema,
            period_me2=self.p.slow_ema,
            period_signal=self.p.signal_period,
        )
        self.crossover = bt.indicators.CrossOver(
            self.macd.macd, self.macd.signal
        )
        self.order = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return
        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            if self.order == order or self.order is None:
                self.order = None

    def next(self):
        if self.order:
            return

        if self.position.size > 0:
            stop_price = self.position.price * (1.0 - self.p.stop_loss_pct)
            if self.data.low[0] <= stop_price:
                self.order = self.close()
            elif self.crossover[0] < 0:
                self.close()
                self.order = self.sell()

        elif self.position.size < 0:
            stop_price = self.position.price * (1.0 + self.p.stop_loss_pct)
            if self.data.high[0] >= stop_price:
                self.order = self.close()
            elif self.crossover[0] > 0:
                self.close()
                self.order = self.buy()

        else:
            if self.crossover[0] > 0:
                self.order = self.buy()
            elif self.crossover[0] < 0:
                self.order = self.sell()
