import backtrader as bt


class DirectionalMovementFilteringByEMA200(bt.Strategy):
    def __init__(self):
        self.ema200 = bt.indicators.ExponentialMovingAverage(
            self.data.close, period=200
        )

        self.Totaldays = 0
        self.Totalup = 0
        self.Totaldown = 0
        self.Totaldojis = 0
        self.Totalabove = 0
        self.Totalbelow = 0
        self.Upabove = 0
        self.Downabove = 0
        self.Dojisabove = 0
        self.Upbelow = 0
        self.Downbelow = 0
        self.Dojisbelow = 0
        self.report = {}

    def next(self):
        self.Totaldays += 1

        if self.data.close[0] > self.data.open[0]:
            direction = "up"
            self.Totalup += 1
        elif self.data.close[0] < self.data.open[0]:
            direction = "down"
            self.Totaldown += 1
        else:
            direction = "doji"
            self.Totaldojis += 1

        if self.data.close[0] > self.ema200[0]:
            self.Totalabove += 1
            if direction == "up":
                self.Upabove += 1
            elif direction == "down":
                self.Downabove += 1
            else:
                self.Dojisabove += 1

        elif self.data.close[0] < self.ema200[0]:
            self.Totalbelow += 1
            if direction == "up":
                self.Upbelow += 1
            elif direction == "down":
                self.Downbelow += 1
            else:
                self.Dojisbelow += 1

    def stop(self):
        def item(value, total):
            return {
                "absolute": value,
                "percentage": (value / total * 100.0) if total else 0.0,
            }

        self.report = {
            "Totaldays": item(self.Totaldays, self.Totaldays),
            "Totalup": item(self.Totalup, self.Totaldays),
            "Totaldown": item(self.Totaldown, self.Totaldays),
            "Totaldojis": item(self.Totaldojis, self.Totaldays),
            "Totalabove": item(self.Totalabove, self.Totaldays),
            "Totalbelow": item(self.Totalbelow, self.Totaldays),
            "Upabove": item(self.Upabove, self.Totalabove),
            "Downabove": item(self.Downabove, self.Totalabove),
            "Dojisabove": item(self.Dojisabove, self.Totalabove),
            "Upbelow": item(self.Upbelow, self.Totalbelow),
            "Downbelow": item(self.Downbelow, self.Totalbelow),
            "Dojisbelow": item(self.Dojisbelow, self.Totalbelow),
        }
