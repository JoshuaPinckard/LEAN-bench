import backtrader as bt


class MonthlySmaStopStrategy(bt.Strategy):
    params = dict(
        sma_period=20,
        entry_day=25,
        exit_day=5,
        stop_pct=0.02,
    )

    def __init__(self):
        self.sma = bt.indicators.SMA(self.data.close, period=self.p.sma_period)
        self.order = None
        self.stop_order = None
        self.entry_trigger_key = None
        self.exit_trigger_key = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order is self.stop_order:
            if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
                self.stop_order = None
            return

        if order is self.order:
            if order.status == order.Completed and order.isbuy():
                stop_price = order.executed.price * (1 - self.p.stop_pct)
                self.stop_order = self.sell(
                    exectype=bt.Order.Stop,
                    price=stop_price,
                    size=order.executed.size,
                )
            if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
                self.order = None

    def next(self):
        dt = self.data.datetime.date(0)
        month_key = (dt.year, dt.month)

        if dt.day >= self.p.exit_day and self.exit_trigger_key != month_key:
            self.exit_trigger_key = month_key
            if self.position and self.order is None:
                if self.stop_order is not None:
                    self.cancel(self.stop_order)
                    self.stop_order = None
                self.order = self.close()

        if dt.day >= self.p.entry_day and self.entry_trigger_key != month_key:
            self.entry_trigger_key = month_key
            if not self.position and self.order is None:
                if self.data.close[0] > self.sma[0]:
                    cash = self.broker.get_cash()
                    price = self.data.close[0]
                    commission = self.broker.getcommissioninfo(self.data).p.commission
                    size = int(cash / (price * (1 + commission)))
                    if size > 0:
                        self.order = self.buy(size=size)
