import backtrader as bt
import math


class ArcherChannelBreakout(bt.Strategy):
    params = (
        ('sma_period', 50),
        ('volume_period', 20),
        ('resistance', 9.99),
        ('support', 8.78),
        ('second_support', 8.23),
        ('take_profit', 13.92),
        ('risk_fraction', 0.01),
    )

    def __init__(self):
        self.order = None

        self.weekly_closes = []
        self.weekly_volumes = []
        self.weekly_lows = []
        self.local_lows = []
        self.support_trendline = None

        self.current_week = None
        self.week_close = None
        self.week_volume = 0.0
        self.week_low = None

        self.previous_week_close = None
        self.entries_enabled = False
        self.was_below_sma = False

    def notify_order(self, order):
        if order.status in (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            self.order = None

    def _process_completed_week(self):
        close = self.week_close
        volume = self.week_volume
        low = self.week_low

        self.weekly_closes.append(close)
        self.weekly_volumes.append(volume)
        self.weekly_lows.append(low)

        if len(self.weekly_lows) >= 3:
            middle_index = len(self.weekly_lows) - 2
            if (
                self.weekly_lows[middle_index] <= self.weekly_lows[middle_index - 1]
                and self.weekly_lows[middle_index] <= self.weekly_lows[middle_index + 1]
            ):
                self.local_lows.append(
                    (middle_index, self.weekly_lows[middle_index])
                )

        if len(self.local_lows) >= 2:
            x1, y1 = self.local_lows[-2]
            x2, y2 = self.local_lows[-1]
            slope = (y2 - y1) / float(x2 - x1)
            self.support_trendline = y2 + slope * (
                (len(self.weekly_lows) - 1) - x2
            )

        if len(self.weekly_closes) < self.p.sma_period:
            self.previous_week_close = close
            return

        sma50 = sum(self.weekly_closes[-self.p.sma_period:]) / self.p.sma_period
        above_sma = close > sma50

        if not above_sma:
            self.was_below_sma = True
            self.entries_enabled = False
            if self.position and self.order is None:
                self.order = self.close()
        elif self.was_below_sma and self.previous_week_close is not None:
            self.entries_enabled = True
            self.was_below_sma = False
        elif self.previous_week_close is None:
            self.entries_enabled = True

        if (
            self.position
            and self.order is None
            and close < self.p.support
        ):
            self.order = self.close()

        if (
            not self.position
            and self.order is None
            and self.entries_enabled
            and above_sma
            and close > self.p.resistance
            and len(self.weekly_volumes) >= self.p.volume_period
        ):
            volume_ma = (
                sum(self.weekly_volumes[-self.p.volume_period:])
                / self.p.volume_period
            )

            if volume >= volume_ma and volume >= volume_ma * 0.70:
                risk_per_share = close - self.p.support
                if risk_per_share > 0:
                    risk_budget = self.broker.getvalue() * self.p.risk_fraction
                    size = math.floor(risk_budget / risk_per_share)
                    if size > 0:
                        self.order = self.buy(size=size)

        self.previous_week_close = close

    def next(self):
        date = self.data.datetime.date(0)
        iso_week = (date.isocalendar()[0], date.isocalendar()[1])

        if self.current_week is None:
            self.current_week = iso_week
            self.week_close = self.data.close[0]
            self.week_volume = self.data.volume[0]
            self.week_low = self.data.low[0]
        elif iso_week != self.current_week:
            self._process_completed_week()
            self.current_week = iso_week
            self.week_close = self.data.close[0]
            self.week_volume = self.data.volume[0]
            self.week_low = self.data.low[0]
        else:
            self.week_close = self.data.close[0]
            self.week_volume += self.data.volume[0]
            self.week_low = min(self.week_low, self.data.low[0])

        if self.position and self.order is None:
            if self.data.high[0] >= self.p.take_profit:
                self.order = self.close()
            elif self.data.low[0] < self.p.second_support:
                self.order = self.close()
