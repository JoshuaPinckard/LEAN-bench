import backtrader as bt


class CapitulationShortSqueezeStrategy(bt.Strategy):
    def __init__(self):
        self.bb = bt.indicators.BollingerBands(self.data.close, period=126, devfactor=2)
        self.rsi = bt.indicators.RSI(self.data.close, period=14)
        self.ema200 = bt.indicators.EMA(self.data.close, period=200)
        self.atr = bt.indicators.ATR(self.data, period=14)
        self.volsma = bt.indicators.SMA(self.data.volume, period=20)
        self.cross_lower = bt.indicators.CrossOver(self.data.close, self.bb.bot)
        self.cross_upper = bt.indicators.CrossOver(self.data.close, self.bb.top)

        self.order = None
        self.pending_action = None
        self.position_type = None
        self.entry_price = None
        self.entry_size = None
        self.stop_price = None
        self.stage = 0
        self.trail_stop = None
        self.max_line_at_entry = None

        self.cap_signal_bar = None
        self.cap_sig_low = None
        self.ob_signal_bar = None
        self.ob_sig_high = None

    def _reset_state(self):
        self.position_type = None
        self.entry_price = None
        self.entry_size = None
        self.stop_price = None
        self.stage = 0
        self.trail_stop = None
        self.max_line_at_entry = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return
        if order.status == order.Completed:
            if self.pending_action == 'enter_long':
                self.entry_price = order.executed.price
                self.entry_size = abs(order.executed.size)
                self.stage = 0
                if self.position_type == 'breakout_long':
                    cand1 = self.max_line_at_entry
                    cand2 = self.entry_price - 1.0 * self.atr[0]
                    self.stop_price = max(cand1, cand2)
            elif self.pending_action == 'enter_short':
                self.entry_price = order.executed.price
                self.entry_size = abs(order.executed.size)
                self.stage = 0
            elif self.pending_action == 'tp1':
                self.stage = 1
                self.stop_price = self.entry_price
            elif self.pending_action == 'tp2':
                self.stage = 2
                self.trail_stop = self.stop_price
            elif self.pending_action in ('tp3', 'stop_exit', 'full_exit'):
                self._reset_state()
        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            if self.pending_action in ('enter_long', 'enter_short'):
                self._reset_state()
        self.order = None
        self.pending_action = None

    def _manage_long(self):
        close = self.data.close[0]
        low = self.data.low[0]
        if close >= self.ema200[0] + 1.5 * self.atr[0]:
            self.pending_action = 'full_exit'
            self.order = self.close()
            return
        if self.stage == 0:
            if low <= self.stop_price:
                self.pending_action = 'stop_exit'
                self.order = self.close()
            elif close >= self.bb.mid[0]:
                self.pending_action = 'tp1'
                self.order = self.sell(size=self.entry_size * 0.5)
        elif self.stage == 1:
            if low <= self.stop_price:
                self.pending_action = 'stop_exit'
                self.order = self.close()
            elif close >= self.bb.top[0]:
                self.pending_action = 'tp2'
                self.order = self.sell(size=self.entry_size * 0.3)
        elif self.stage == 2:
            self.trail_stop = max(self.trail_stop, close - 2 * self.atr[0])
            target = self.bb.top[0] + 0.5 * self.atr[0]
            if close >= target or low <= self.trail_stop:
                self.pending_action = 'tp3'
                self.order = self.close()

    def _manage_short(self):
        close = self.data.close[0]
        high = self.data.high[0]
        if self.stage == 0:
            if high >= self.stop_price:
                self.pending_action = 'stop_exit'
                self.order = self.close()
            elif close <= self.bb.mid[0]:
                self.pending_action = 'tp1'
                self.order = self.buy(size=self.entry_size * 0.5)
        elif self.stage == 1:
            if high >= self.stop_price:
                self.pending_action = 'stop_exit'
                self.order = self.close()
            elif close <= self.bb.bot[0]:
                self.pending_action = 'tp2'
                self.order = self.buy(size=self.entry_size * 0.3)
        elif self.stage == 2:
            self.trail_stop = min(self.trail_stop, close + 2 * self.atr[0])
            if high >= self.trail_stop:
                self.pending_action = 'tp3'
                self.order = self.close()

    def next(self):
        if self.order:
            return

        if self.position:
            if self.position_type in ('cap_long', 'breakout_long'):
                self._manage_long()
            elif self.position_type == 'short':
                self._manage_short()
            return

        order_submitted = False

        if self.cap_signal_bar is not None and len(self) - self.cap_signal_bar == 1:
            confirmed = (self.cross_lower[0] == 1) or (
                (self.data.high[0] - self.data.low[0]) >= 1.2 * self.atr[0]
                and self.data.close[0] > self.data.open[0]
            )
            if confirmed:
                self.stop_price = self.cap_sig_low - 1.0 * self.atr[0]
                self.position_type = 'cap_long'
                self.pending_action = 'enter_long'
                self.order = self.buy()
                order_submitted = True
            self.cap_signal_bar = None
            self.cap_sig_low = None

        if not order_submitted and self.ob_signal_bar is not None and len(self) - self.ob_signal_bar == 1:
            confirmed = (self.cross_upper[0] == -1) or (
                (self.data.high[0] - self.data.low[0]) >= 1.2 * self.atr[0]
                and self.data.close[0] < self.data.open[0]
            )
            if confirmed:
                self.stop_price = self.ob_sig_high + 1.0 * self.atr[0]
                self.position_type = 'short'
                self.pending_action = 'enter_short'
                self.order = self.sell()
                order_submitted = True
            self.ob_signal_bar = None
            self.ob_sig_high = None

        if not order_submitted:
            max_prev = max(self.bb.mid[-1], self.ema200[-1])
            max_curr = max(self.bb.mid[0], self.ema200[0])
            breakout_up = self.data.close[-1] <= max_prev and self.data.close[0] > max_curr
            if breakout_up and self.data.volume[0] >= 1.5 * self.volsma[0] and self.rsi[0] > 50:
                self.max_line_at_entry = max_curr
                self.position_type = 'breakout_long'
                self.pending_action = 'enter_long'
                self.order = self.buy()
                order_submitted = True

        if not order_submitted and self.cap_signal_bar is None:
            if (self.data.close[0] < self.bb.bot[0] and self.rsi[0] < 30
                    and self.data.volume[0] >= 1.8 * self.volsma[0]):
                self.cap_signal_bar = len(self)
                lows = self.data.low.get(size=5)
                self.cap_sig_low = min(lows) if lows else self.data.low[0]

        if not order_submitted and self.ob_signal_bar is None:
            if (self.data.close[0] > self.bb.top[0] and self.rsi[0] > 70
                    and self.data.volume[0] >= 1.8 * self.volsma[0]):
                self.ob_signal_bar = len(self)
                self.ob_sig_high = self.data.high[0]
