import backtrader as bt


class TrendFollowingETFStrategy(bt.Strategy):
    params = (
        ("atr_period", 14),
        ("ema_period", 50),
        ("pullback_atr_mult", 1.5),
        ("tp_atr_mult", 2.5),
        ("sl_atr_mult", 1.5),
        ("max_entries", 4),
        ("pos_pct", 0.01),
    )

    def __init__(self):
        self.atr = bt.indicators.ATR(self.data, period=self.params.atr_period)
        self.ema = bt.indicators.EMA(self.data, period=self.params.ema_period)

        self.order = None
        self.entries_count = 0
        self.recent_high = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            if order.isbuy():
                self.entries_count += 1
                self.recent_high = self.data.high[0]
            elif order.issell():
                if self.position.size == 0:
                    self.entries_count = 0
                    self.recent_high = None

        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            pass

        self.order = None

    def next(self):
        if self.order:
            return

        atr_val = self.atr[0]

        if self.position.size > 0:
            avg_price = self.position.price

            # Exit Condition 2: Price closes below EMA(50)
            if self.data.close[0] < self.ema[0]:
                self.order = self.close()
                return

            # Exit Condition 1: Take Profit at 2.5 ATR above entry
            if self.data.high[0] >= avg_price + self.params.tp_atr_mult * atr_val:
                self.order = self.close()
                return

            # Risk Management: Stop Loss at 1.5 ATR below entry
            if self.data.low[0] <= avg_price - self.params.sl_atr_mult * atr_val:
                self.order = self.close()
                return

            # Update recent high for scaling pullbacks
            if self.recent_high is None or self.data.high[0] > self.recent_high:
                self.recent_high = self.data.high[0]

            # Scaling in condition (up to 4 entries total)
            if (
                self.data.close[0] > self.ema[0]
                and self.entries_count < self.params.max_entries
            ):
                if (
                    self.recent_high - self.data.close[0]
                    >= self.params.pullback_atr_mult * atr_val
                ):
                    size = int(
                        (self.broker.getvalue() * self.params.pos_pct)
                        / self.data.close[0]
                    )
                    if size > 0:
                        self.order = self.buy(size=size)

        else:
            # Initial Entry Condition
            if self.data.close[0] > self.ema[0]:
                if self.recent_high is None or self.data.high[0] > self.recent_high:
                    self.recent_high = self.data.high[0]

                if (
                    self.recent_high - self.data.close[0]
                    >= self.params.pullback_atr_mult * atr_val
                ):
                    size = int(
                        (self.broker.getvalue() * self.params.pos_pct)
                        / self.data.close[0]
                    )
                    if size > 0:
                        self.order = self.buy(size=size)
            else:
                self.recent_high = None
