import backtrader as bt


class TrendFollowingPullback(bt.Strategy):

    params = (
        ('atr_period', 14),
        ('ema_period', 50),
        ('pullback_mult', 1.5),
        ('tp_mult', 2.5),
        ('sl_mult', 1.5),
        ('risk_pct', 0.01),
        ('max_entries', 4),
        ('recent_high_lookback', 20),
    )

    def __init__(self):
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.ema = bt.indicators.EMA(self.data.close, period=self.p.ema_period)
        self.order = None
        self._reset_state()

    def _reset_state(self):
        self.entry_prices = []
        self.entry_atrs = []
        self.entry_stops = []
        self.entry_count = 0
        self.blocked = False
        self.ref_high = None

    def notify_order(self, order):
        if order.status == order.Completed:
            if order.isbuy():
                ep = order.executed.price
                atr_val = self.atr[0]
                self.entry_prices.append(ep)
                self.entry_atrs.append(atr_val)
                self.entry_stops.append(ep - self.p.sl_mult * atr_val)
                self.entry_count += 1
                if self.entry_count >= self.p.max_entries:
                    self.blocked = True
                self.ref_high = self.data.high[0]
            self.order = None
        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def notify_trade(self, trade):
        if trade.isclosed:
            self._reset_state()

    def next(self):
        if self.order:
            return

        close = self.data.close[0]
        high = self.data.high[0]
        low = self.data.low[0]
        atr = self.atr[0]
        ema = self.ema[0]

        if self.position:
            if self.ref_high is None or high > self.ref_high:
                self.ref_high = high

            # Stop-loss: any individual entry's stop breached
            for sl in self.entry_stops:
                if low <= sl:
                    self.order = self.close()
                    return

            # Exit: price closes below EMA(50)
            if close < ema:
                self.order = self.close()
                return

            # Take-profit: 2.5 ATR from first entry point
            if self.entry_prices:
                tp = self.entry_prices[0] + self.p.tp_mult * self.entry_atrs[0]
                if high >= tp:
                    self.order = self.close()
                    return

            # Scale-in: additional entry on next pullback
            if not self.blocked and self.ref_high is not None:
                pullback_level = self.ref_high - self.p.pullback_mult * atr
                if close <= pullback_level and close > ema:
                    size = int((self.broker.getvalue() * self.p.risk_pct) / close)
                    if size > 0:
                        self.order = self.buy(size=size)
        else:
            # Initial entry: price above EMA, pullback from rolling recent high
            if close > ema:
                lookback = min(self.p.recent_high_lookback, len(self.data))
                recent_high = max(self.data.high.get(size=lookback))
                pullback_level = recent_high - self.p.pullback_mult * atr
                if close <= pullback_level:
                    size = int((self.broker.getvalue() * self.p.risk_pct) / close)
                    if size > 0:
                        self.order = self.buy(size=size)
