import backtrader as bt


class TrendPullbackScaleIn(bt.Strategy):
    params = (
        ('atr_period', 14),
        ('ema_period', 50),
        ('high_lookback', 20),      # window used for the initial "recent high"
        ('pullback_mult', 1.5),     # pullback / stop distance in ATR
        ('tp_mult', 2.5),           # take-profit distance in ATR
        ('max_entries', 4),         # max consecutive entries in one direction
        ('risk_per_entry', 0.01),   # 1% of capital per individual position
    )

    def __init__(self):
        # --- Indicators (exactly as specified) ---
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.ema = bt.indicators.ExponentialMovingAverage(
            self.data.close, period=self.p.ema_period)
        # rolling "recent high" used for the initial pullback trigger
        self.recent_high = bt.indicators.Highest(
            self.data.high, period=self.p.high_lookback)

        # --- State ---
        self.order = None            # pending-order guard
        self._action = None          # what the pending order is doing
        self.entries = []            # open scaled entries: {size, price, atr, stop}
        self.entry_count = 0         # consecutive entries made this sequence
        self.pivot = None            # highest high since last entry ("next high")
        self.tp_level = None         # take-profit from the initial entry point

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return

        if order.status == order.Completed:
            action = self._action
            if action is not None and action[0] == 'buy':
                atr = action[1]
                price = order.executed.price
                size = order.executed.size
                # per-position stop-loss: 1.5 ATR below this entry point
                stop = price - self.p.pullback_mult * atr
                self.entries.append(
                    {'size': size, 'price': price, 'atr': atr, 'stop': stop})
                self.entry_count += 1
                if self.entry_count == 1:
                    # take-profit measured from the initial entry point
                    self.tp_level = price + self.p.tp_mult * atr
                # start tracking a fresh "next high" for the following scale-in
                self.pivot = None
            elif action is not None and action[0] == 'stop':
                e = action[1]
                if e in self.entries:
                    self.entries.remove(e)
            elif action is not None and action[0] == 'exit':
                self.entries = []

        if order.status in (order.Completed, order.Canceled,
                            order.Margin, order.Rejected):
            self.order = None
            self._action = None

    def next(self):
        if self.order:
            return

        price = self.data.close[0]
        atr = self.atr[0]
        ema = self.ema[0]

        if self.position:
            # track the highest high since the last entry (the "next high")
            if self.pivot is None:
                self.pivot = self.data.high[0]
            else:
                self.pivot = max(self.pivot, self.data.high[0])

            # Exit 1: take-profit of 2.5 ATR from the entry point -> close all
            if self.tp_level is not None and price >= self.tp_level:
                self.order = self.close()
                self._action = ('exit',)
                return

            # Exit 2: price closes below EMA(50) -> close the entire position
            if price < ema:
                self.order = self.close()
                self._action = ('exit',)
                return

            # Risk mgmt: per-position stop-loss 1.5 ATR below its entry point
            for e in self.entries:
                if price <= e['stop']:
                    self.order = self.sell(size=e['size'])
                    self._action = ('stop', e)
                    return

            # Entry 3: scale in on a further 1.5 ATR pullback from the next high
            # Entry 4: cap consecutive entries at 4 (also caps risk at 4%)
            if (self.entry_count < self.p.max_entries
                    and price > ema
                    and price <= self.pivot - self.p.pullback_mult * atr):
                size = self.p.risk_per_entry * self.broker.getvalue() / price
                self.order = self.buy(size=size)
                self._action = ('buy', atr)
            return

        # Flat -> start of a new sequence; reset all sequence state
        self.entry_count = 0
        self.entries = []
        self.pivot = None
        self.tp_level = None

        # Entry 1 & 2: bullish trend + 1.5 ATR pullback from a recent high
        if price > ema and price <= self.recent_high[0] - self.p.pullback_mult * atr:
            size = self.p.risk_per_entry * self.broker.getvalue() / price
            self.order = self.buy(size=size)
            self._action = ('buy', atr)
