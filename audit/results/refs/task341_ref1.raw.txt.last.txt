import backtrader as bt


class SMATrendStrategy(bt.Strategy):
    def __init__(self):
        self.sma5 = bt.indicators.SimpleMovingAverage(self.data.close, period=5)
        self.sma20 = bt.indicators.SimpleMovingAverage(self.data.close, period=20)
        self.sma50 = bt.indicators.SimpleMovingAverage(self.data.close, period=50)
        self.cross_down = bt.indicators.CrossDown(self.sma5, self.sma20)
        self.cross_up = bt.indicators.CrossUp(self.sma5, self.sma20)
        self.order = None

    def notify_order(self, order):
        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            self.order = None

    def next(self):
        if self.order:
            return

        price = self.data.close[0]

        if self.position.size > 0:
            if self.cross_down[0] or price <= self.position.price * 0.98:
                self.order = self.close()
            return

        if self.position.size < 0:
            if self.cross_up[0] or price >= self.position.price * 1.02:
                self.order = self.close()
            return

        size = self.broker.getcash() / (
            price * (1.0 + self.broker.getcommissioninfo(self.data).p.commission)
        )

        if self.sma5[0] > self.sma20[0] and self.sma20[0] > self.sma50[0]:
            self.order = self.buy(size=size)
        elif self.sma5[0] < self.sma20[0] and self.sma20[0] < self.sma50[0]:
            self.order = self.sell(size=size)
