import backtrader as bt


class EMARibbonStrategy(bt.Strategy):
    params = dict(
        stop_loss_pct=0.02,
        take_profit_pct=0.04,
        position_pct=0.02,
    )

    def __init__(self):
        self.ema8 = bt.indicators.EMA(self.data.close, period=8)
        self.ema13 = bt.indicators.EMA(self.data.close, period=13)
        self.ema21 = bt.indicators.EMA(self.data.close, period=21)
        self.ema34 = bt.indicators.EMA(self.data.close, period=34)
        self.ema55 = bt.indicators.EMA(self.data.close, period=55)

        self.entry_order = None
        self.stop_order = None
        self.limit_order = None
        self.ribbon_ordered_up = False

    def next(self):
        fast_min = min(self.ema8[0], self.ema13[0], self.ema21[0])
        slow_max = max(self.ema34[0], self.ema55[0])
        ordered_up = fast_min > slow_max

        if not self.position:
            if self.entry_order is None and ordered_up and not self.ribbon_ordered_up:
                cash_to_risk = self.broker.getvalue() * self.p.position_pct
                size = cash_to_risk / self.data.close[0]
                self.entry_order = self.buy(size=size)
        else:
            exit_signal = (not ordered_up) and self.ribbon_ordered_up
            if exit_signal:
                if self.stop_order is not None:
                    self.cancel(self.stop_order)
                if self.limit_order is not None:
                    self.cancel(self.limit_order)
                self.close()

        self.ribbon_ordered_up = ordered_up

    def notify_order(self, order):
        if order.status not in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            return

        if order == self.entry_order:
            if order.status == order.Completed:
                entry_price = order.executed.price
                stop_price = entry_price * (1 - self.p.stop_loss_pct)
                take_profit_price = entry_price * (1 + self.p.take_profit_pct)
                self.stop_order = self.sell(
                    size=order.executed.size,
                    exectype=bt.Order.Stop,
                    price=stop_price,
                )
                self.limit_order = self.sell(
                    size=order.executed.size,
                    exectype=bt.Order.Limit,
                    price=take_profit_price,
                )
            self.entry_order = None
        elif order == self.stop_order:
            if order.status == order.Completed and self.limit_order is not None:
                self.cancel(self.limit_order)
            self.stop_order = None
        elif order == self.limit_order:
            if order.status == order.Completed and self.stop_order is not None:
                self.cancel(self.stop_order)
            self.limit_order = None
