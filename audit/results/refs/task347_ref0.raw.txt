import backtrader as bt


class KeltnerStochasticStrategy(bt.Strategy):
    params = (
        ('stoch_period', 14),
        ('stoch_pfast', 3),
        ('stoch_pslow', 3),
        ('stoch_oversold', 20),
        ('stoch_overbought', 80),
        ('kc_ema_period', 20),
        ('atr_period', 14),
        ('kc_mult', 2.0),
        ('stop_mult', 2.0),
    )

    def __init__(self):
        self.stoch = bt.indicators.Stochastic(
            self.data,
            period=self.p.stoch_period,
            period_dfast=self.p.stoch_pfast,
            period_dslow=self.p.stoch_pslow,
        )
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.mid = bt.indicators.EMA(self.data.close, period=self.p.kc_ema_period)
        self.upper = self.mid + self.p.kc_mult * self.atr
        self.lower = self.mid - self.p.kc_mult * self.atr

        self.order = None
        self.stop_order = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status == order.Completed:
            if order == self.stop_order:
                self.stop_order = None
                self.order = None
            else:
                if self.position:
                    if order.isbuy():
                        stop_price = order.executed.price - self.p.stop_mult * self.atr[0]
                        self.stop_order = self.sell(exectype=bt.Order.Stop, price=stop_price)
                    else:
                        stop_price = order.executed.price + self.p.stop_mult * self.atr[0]
                        self.stop_order = self.buy(exectype=bt.Order.Stop, price=stop_price)
                else:
                    self.stop_order = None
                self.order = None
        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            if order == self.stop_order:
                self.stop_order = None
            else:
                self.order = None

    def next(self):
        if self.order:
            return

        if not self.position:
            if (self.stoch.percK[0] < self.p.stoch_oversold
                    and self.data.close[0] < self.lower[0]):
                self.order = self.buy()
            elif (self.stoch.percK[0] > self.p.stoch_overbought
                    and self.data.close[0] > self.upper[0]):
                self.order = self.sell()
        else:
            if self.position.size > 0 and self.data.close[0] >= self.mid[0]:
                if self.stop_order is not None:
                    self.cancel(self.stop_order)
                self.order = self.close()
            elif self.position.size < 0 and self.data.close[0] <= self.mid[0]:
                if self.stop_order is not None:
                    self.cancel(self.stop_order)
                self.order = self.close()
