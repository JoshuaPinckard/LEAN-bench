import backtrader as bt


class SmaAtrTrailStrategy(bt.Strategy):
    # NOTE: commission (0.001) and slippage are broker-level PLACEHOLDERS.
    # The actual broker/commission/slippage configuration is owned by the
    # harness at the cerebro level; the params below are inert placeholders
    # so the spec's "commission/slippage placeholders" are explicitly present.
    params = (
        ('sma_period', 20),
        ('atr_period', 14),
        ('atr_mult', 3),
        ('commission', 0.001),  # placeholder; real value set on broker by harness
        ('slippage', 0.0),      # placeholder; real value set on broker by harness
    )

    def __init__(self):
        # Indicators exactly as specified: SMA(20) on close and ATR(14).
        self.sma = bt.indicators.SimpleMovingAverage(
            self.data.close, period=self.p.sma_period)
        self.atr = bt.indicators.AverageTrueRange(
            self.data, period=self.p.atr_period)
        # Cross of close vs SMA(20): +1 crosses above, -1 crosses below.
        self.crossover = bt.indicators.CrossOver(self.data.close, self.sma)

        self.order = None         # pending entry order (guard)
        self.stop_order = None    # resting trailing-stop exit order
        self.trail_stop = None    # current ratcheting trailing-stop level
        self.highest_close = None  # highest close since entry (long)
        self.lowest_close = None   # lowest close since entry (short)

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return
        if order.status in (order.Completed, order.Canceled,
                             order.Margin, order.Rejected):
            if order is self.order:
                self.order = None
            if order is self.stop_order:
                self.stop_order = None

    def notify_trade(self, trade):
        # Reset per-trade trailing state once the position is fully closed.
        if trade.isclosed:
            self.trail_stop = None
            self.highest_close = None
            self.lowest_close = None

    def next(self):
        # ---- Entries: only when flat and no pending entry order ----
        if not self.position:
            if self.order is not None:
                return
            if self.crossover[0] > 0:      # prior-bar close crossed above SMA(20)
                self.order = self.buy()     # market -> fills at next bar open
            elif self.crossover[0] < 0:    # prior-bar close crossed below SMA(20)
                self.order = self.sell()    # market -> fills at next bar open
            return

        # ---- Manage open position: ratchet trailing stop in trade's favor ----
        if self.position.size > 0:          # LONG
            if self.highest_close is None:
                self.highest_close = self.data.close[0]
            else:
                self.highest_close = max(self.highest_close, self.data.close[0])
            candidate = self.highest_close - self.p.atr_mult * self.atr[0]
            if self.trail_stop is None:
                self.trail_stop = candidate
            else:
                # trail_stop = max(previous_trail_stop, highest_close - 3*ATR)
                self.trail_stop = max(self.trail_stop, candidate)

            if self.crossover[0] < 0:        # close crossed below SMA(20) -> exit
                if self.stop_order is not None:
                    self.cancel(self.stop_order)
                    self.stop_order = None
                self.close()
            else:
                # (Re)place the stop-loss exit order, updated each bar.
                if self.stop_order is not None:
                    self.cancel(self.stop_order)
                self.stop_order = self.sell(
                    size=abs(self.position.size),
                    exectype=bt.Order.Stop,
                    price=self.trail_stop)

        elif self.position.size < 0:        # SHORT
            if self.lowest_close is None:
                self.lowest_close = self.data.close[0]
            else:
                self.lowest_close = min(self.lowest_close, self.data.close[0])
            candidate = self.lowest_close + self.p.atr_mult * self.atr[0]
            if self.trail_stop is None:
                self.trail_stop = candidate
            else:
                # trail_stop = min(previous_trail_stop, lowest_close + 3*ATR)
                self.trail_stop = min(self.trail_stop, candidate)

            if self.crossover[0] > 0:        # close crossed above SMA(20) -> exit
                if self.stop_order is not None:
                    self.cancel(self.stop_order)
                    self.stop_order = None
                self.close()
            else:
                # (Re)place the stop-loss exit order, updated each bar.
                if self.stop_order is not None:
                    self.cancel(self.stop_order)
                self.stop_order = self.buy(
                    size=abs(self.position.size),
                    exectype=bt.Order.Stop,
                    price=self.trail_stop)
