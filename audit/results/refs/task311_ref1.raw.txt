import backtrader as bt


class ReferenceStrategy(bt.Strategy):
    params = (
        ('commission_placeholder', 0.001),
        ('slippage_placeholder', 0.0),
    )

    def __init__(self):
        self.sma = bt.indicators.SMA(self.data.close, period=20)
        self.atr = bt.indicators.ATR(self.data, period=14)

        self.entry_order = None
        self.exit_order = None
        self.stop_order = None

        self.highest_close_since_entry = None
        self.lowest_close_since_entry = None
        self.trail_stop = None

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled,
                            order.Margin, order.Rejected]:
            if order is self.entry_order:
                self.entry_order = None

            if order is self.exit_order:
                self.exit_order = None

            if order is self.stop_order:
                self.stop_order = None

            if order.status == order.Completed and order.isbuy():
                if self.position.size > 0:
                    self.highest_close_since_entry = self.data.close[0]
                    self.lowest_close_since_entry = None
                    self.trail_stop = (
                        self.highest_close_since_entry - 3 * self.atr[0]
                    )
                elif self.position.size == 0:
                    self.highest_close_since_entry = None
                    self.lowest_close_since_entry = None
                    self.trail_stop = None

            if order.status == order.Completed and order.issell():
                if self.position.size < 0:
                    self.lowest_close_since_entry = self.data.close[0]
                    self.highest_close_since_entry = None
                    self.trail_stop = (
                        self.lowest_close_since_entry + 3 * self.atr[0]
                    )
                elif self.position.size == 0:
                    self.highest_close_since_entry = None
                    self.lowest_close_since_entry = None
                    self.trail_stop = None

    def next(self):
        if self.entry_order or self.exit_order:
            return

        crossed_above = (
            self.data.close[-1] <= self.sma[-1]
            and self.data.close[0] > self.sma[0]
        )
        crossed_below = (
            self.data.close[-1] >= self.sma[-1]
            and self.data.close[0] < self.sma[0]
        )

        if not self.position:
            if self.stop_order:
                return

            if crossed_above:
                self.entry_order = self.buy()
            elif crossed_below:
                self.entry_order = self.sell()
            return

        if self.position.size > 0:
            if crossed_below:
                if self.stop_order:
                    self.cancel(self.stop_order)
                self.exit_order = self.close()
                return

            self.highest_close_since_entry = max(
                self.highest_close_since_entry, self.data.close[0]
            )
            new_trail_stop = (
                self.highest_close_since_entry - 3 * self.atr[0]
            )
            self.trail_stop = max(self.trail_stop, new_trail_stop)

            if self.stop_order:
                self.cancel(self.stop_order)
            else:
                self.stop_order = self.sell(
                    exectype=bt.Order.Stop,
                    price=self.trail_stop,
                )

        elif self.position.size < 0:
            if crossed_above:
                if self.stop_order:
                    self.cancel(self.stop_order)
                self.exit_order = self.close()
                return

            self.lowest_close_since_entry = min(
                self.lowest_close_since_entry, self.data.close[0]
            )
            new_trail_stop = (
                self.lowest_close_since_entry + 3 * self.atr[0]
            )
            self.trail_stop = min(self.trail_stop, new_trail_stop)

            if self.stop_order:
                self.cancel(self.stop_order)
            else:
                self.stop_order = self.buy(
                    exectype=bt.Order.Stop,
                    price=self.trail_stop,
                )
