import backtrader as bt


class MeanReversionStrategy(bt.Strategy):
    params = (
        ('atr_period', 14),
        ('ma_period', 20),
        ('ma_type', 'SMA'),
        ('atr_mult', 2.0),
        ('stop_pct', 0.02),
    )

    def __init__(self):
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        if self.p.ma_type == 'EMA':
            self.ma = bt.indicators.EMA(self.data, period=self.p.ma_period)
        else:
            self.ma = bt.indicators.SMA(self.data, period=self.p.ma_period)
        self.crossover = bt.indicators.CrossOver(self.data.close, self.ma)

        self.order = None
        self.entry_price = None
        self.current_day = None
        self.reference = None

    def notify_order(self, order):
        if order.status == order.Completed:
            if self.position.size != 0:
                self.entry_price = order.executed.price
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def _update_reference(self):
        bar_date = self.data.datetime.date(0)
        if bar_date != self.current_day:
            self.current_day = bar_date
            self.reference = self.data.open[0]

    def prenext(self):
        self._update_reference()

    def next(self):
        self._update_reference()

        if self.order:
            return

        close = self.data.close[0]

        if not self.position:
            if close > self.reference + self.p.atr_mult * self.atr[0]:
                self.order = self.sell()
            elif close < self.reference - self.p.atr_mult * self.atr[0]:
                self.order = self.buy()
        else:
            if self.position.size > 0:
                stop_price = self.entry_price * (1 - self.p.stop_pct)
                if self.crossover[0] > 0 or close <= stop_price:
                    self.order = self.close()
            elif self.position.size < 0:
                stop_price = self.entry_price * (1 + self.p.stop_pct)
                if self.crossover[0] < 0 or close >= stop_price:
                    self.order = self.close()
