import backtrader as bt
import collections
import statistics


class ORBFibPullbackStrategy(bt.Strategy):
    params = dict(
        or_minutes=60,
        fifteen_min=15,
        one_hour=60,
        trend_sma_period=50,
        momentum_mult=0.25,
        risk_pct=0.01,
        rr_ratio=2.0,
    )

    def __init__(self):
        self.order = None
        self.stop_price = None
        self.tp_price = None
        self.entry_dir = None

        self.session_date = None
        self.min_count = -1

        self.or_high = None
        self.or_low = None
        self.or_finalized = False

        self.breakout_dir = None
        self.impulse_extreme = None
        self.momentum_confirmed = False
        self.fib50 = None
        self.fib618 = None

        # kept across sessions intentionally: trend context should not reset daily
        self.hourly_closes = collections.deque(maxlen=self.p.trend_sma_period)
        self.trend_sma = None
        self.trend_dir = None

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def _reset_day_state(self):
        self.or_high = None
        self.or_low = None
        self.or_finalized = False
        self._reset_setup_state()

    def _reset_setup_state(self):
        self.breakout_dir = None
        self.impulse_extreme = None
        self.momentum_confirmed = False
        self.fib50 = None
        self.fib618 = None

    def next(self):
        dt = self.data.datetime.date(0)
        if dt != self.session_date:
            self.session_date = dt
            self.min_count = 0
            self._reset_day_state()
        else:
            self.min_count += 1

        if self.min_count < self.p.or_minutes:
            if self.or_high is None:
                self.or_high = self.data.high[0]
                self.or_low = self.data.low[0]
            else:
                self.or_high = max(self.or_high, self.data.high[0])
                self.or_low = min(self.or_low, self.data.low[0])
            if self.min_count == self.p.or_minutes - 1:
                self.or_finalized = True

        if (self.min_count + 1) % self.p.one_hour == 0:
            self.hourly_closes.append(self.data.close[0])
            if len(self.hourly_closes) == self.p.trend_sma_period:
                self.trend_sma = statistics.mean(self.hourly_closes)
        if self.trend_sma is not None:
            self.trend_dir = 'up' if self.hourly_closes[-1] > self.trend_sma else 'down'

        if not self.or_finalized:
            return
        if self.order:
            return
        if self.position:
            self._manage_position()
            return

        is_15m_close = (self.min_count + 1) % self.p.fifteen_min == 0
        if self.breakout_dir is None and is_15m_close:
            c = self.data.close[0]
            if c > self.or_high:
                self.breakout_dir = 'long'
                self.impulse_extreme = c
            elif c < self.or_low:
                self.breakout_dir = 'short'
                self.impulse_extreme = c
            return

        if self.breakout_dir is None:
            return

        if self.trend_dir is None:
            return
        if self.breakout_dir == 'long' and self.trend_dir != 'up':
            return
        if self.breakout_dir == 'short' and self.trend_dir != 'down':
            return

        if not self.momentum_confirmed:
            or_range = self.or_high - self.or_low
            if self.breakout_dir == 'long':
                self.impulse_extreme = max(self.impulse_extreme, self.data.high[0])
                if self.impulse_extreme >= self.or_high + self.p.momentum_mult * or_range:
                    self.momentum_confirmed = True
                    self.fib50 = self.impulse_extreme - 0.5 * (self.impulse_extreme - self.or_high)
                    self.fib618 = self.impulse_extreme - 0.618 * (self.impulse_extreme - self.or_high)
                elif self.data.close[0] <= self.or_high:
                    self._reset_setup_state()
            else:
                self.impulse_extreme = min(self.impulse_extreme, self.data.low[0])
                if self.impulse_extreme <= self.or_low - self.p.momentum_mult * or_range:
                    self.momentum_confirmed = True
                    self.fib50 = self.impulse_extreme + 0.5 * (self.or_low - self.impulse_extreme)
                    self.fib618 = self.impulse_extreme + 0.618 * (self.or_low - self.impulse_extreme)
                elif self.data.close[0] >= self.or_low:
                    self._reset_setup_state()
            return

        price = self.data.close[0]
        prev_close = self.data.close[-1]
        if self.breakout_dir == 'long':
            in_golden_zone = self.fib618 <= price <= self.fib50
            if in_golden_zone and price > prev_close:
                self._enter_long()
            elif price <= self.or_high:
                self._reset_setup_state()
        else:
            in_golden_zone = self.fib50 <= price <= self.fib618
            if in_golden_zone and price < prev_close:
                self._enter_short()
            elif price >= self.or_low:
                self._reset_setup_state()

    def _enter_long(self):
        stop = self.or_low
        entry = self.data.close[0]
        risk_per_share = entry - stop
        if risk_per_share <= 0:
            self._reset_setup_state()
            return
        size = int((self.broker.getvalue() * self.p.risk_pct) / risk_per_share)
        if size <= 0:
            self._reset_setup_state()
            return
        self.stop_price = stop
        self.tp_price = entry + self.p.rr_ratio * risk_per_share
        self.entry_dir = 'long'
        self.order = self.buy(size=size)

    def _enter_short(self):
        stop = self.or_high
        entry = self.data.close[0]
        risk_per_share = stop - entry
        if risk_per_share <= 0:
            self._reset_setup_state()
            return
        size = int((self.broker.getvalue() * self.p.risk_pct) / risk_per_share)
        if size <= 0:
            self._reset_setup_state()
            return
        self.stop_price = stop
        self.tp_price = entry - self.p.rr_ratio * risk_per_share
        self.entry_dir = 'short'
        self.order = self.sell(size=size)

    def _manage_position(self):
        price = self.data.close[0]
        if self.entry_dir == 'long':
            if price <= self.stop_price or price >= self.tp_price or price <= self.or_high:
                self.order = self.close()
                self._reset_setup_state()
        else:
            if price >= self.stop_price or price <= self.tp_price or price >= self.or_low:
                self.order = self.close()
                self._reset_setup_state()
