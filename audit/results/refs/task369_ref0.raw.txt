import backtrader as bt


class MeanReversionBandStrategy(bt.Strategy):
    def __init__(self):
        self.ema20 = bt.indicators.EMA(self.data.close, period=20)
        self.ema_mean = bt.indicators.SMA(self.ema20, period=20)
        self.ema_std = bt.indicators.StdDev(self.ema20, period=20)
        self.upper_band = self.ema_mean + 2.0 * self.ema_std
        self.lower_band = self.ema_mean - 2.0 * self.ema_std

        self.cross_lower = bt.indicators.CrossOver(self.ema20, self.lower_band)
        self.cross_upper = bt.indicators.CrossOver(self.ema20, self.upper_band)
        self.cross_mean = bt.indicators.CrossOver(self.ema20, self.ema_mean)

        self.order = None
        self.stop_order = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status == order.Completed:
            if order.exectype == bt.Order.Stop:
                self.stop_order = None
                self.order = None
            elif not self.position:
                self.order = None
            else:
                entry_price = order.executed.price
                if order.isbuy():
                    stop_price = entry_price * 0.98
                    self.order = None
                    self.stop_order = self.sell(exectype=bt.Order.Stop, price=stop_price)
                else:
                    stop_price = entry_price * 1.02
                    self.order = None
                    self.stop_order = self.buy(exectype=bt.Order.Stop, price=stop_price)

        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            if order.exectype == bt.Order.Stop:
                self.stop_order = None
            else:
                self.order = None

    def next(self):
        if self.order:
            return

        if not self.position:
            if self.cross_lower[0] > 0:
                self.order = self.buy()
            elif self.cross_upper[0] < 0:
                self.order = self.sell()
        else:
            if self.cross_mean[0] != 0:
                if self.stop_order:
                    self.cancel(self.stop_order)
                self.order = self.close()
