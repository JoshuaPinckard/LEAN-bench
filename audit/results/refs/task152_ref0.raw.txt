import statistics

import backtrader as bt


class LLYBullFlagStrategy(bt.Strategy):
    params = dict(
        bars_per_day=7,           # ~5078 bars / ~750 trading days in the feed
        sma_period_50d=350,       # 50 trading days * bars_per_day
        sma_period_100d=700,      # 100 trading days * bars_per_day
        sma_period_200d=1400,     # 200 trading days * bars_per_day
        weekly_support=774.48,    # weekly lower-high support level (given)
        fib_ratio=0.382,
        consolidation_lookback=10,
        impulse_lookback=30,
        stop_buffer=0.50,
        risk_pct=0.01,
        rr_ratio=2.0,
    )

    def __init__(self):
        self.sma50 = bt.indicators.SMA(self.data.close, period=self.p.sma_period_50d)
        self.sma100 = bt.indicators.SMA(self.data.close, period=self.p.sma_period_100d)
        self.sma200 = bt.indicators.SMA(self.data.close, period=self.p.sma_period_200d)

        self.order = None
        self.stop_price = None
        self.take_profit_price = None
        self.pending_stop = None
        self.pending_take_profit = None

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return

        if order.status == order.Completed:
            if order.isbuy():
                self.stop_price = self.pending_stop
                self.take_profit_price = self.pending_take_profit
            else:
                self.stop_price = None
                self.take_profit_price = None

        self.order = None

    def next(self):
        if self.order:
            return

        if not self.position:
            close = self.data.close[0]

            trend_ok = (
                close > self.sma50[0]
                and close > self.sma100[0]
                and close > self.sma200[0]
            )
            support_ok = close > self.p.weekly_support

            consol_highs = [self.data.high[-i] for i in range(1, self.p.consolidation_lookback + 1)]
            consol_lows = [self.data.low[-i] for i in range(1, self.p.consolidation_lookback + 1)]
            consol_vols = [self.data.volume[-i] for i in range(1, self.p.consolidation_lookback + 1)]

            impulse_start = self.p.consolidation_lookback + 1
            impulse_end = self.p.consolidation_lookback + self.p.impulse_lookback
            impulse_highs = [self.data.high[-i] for i in range(impulse_start, impulse_end + 1)]
            impulse_lows = [self.data.low[-i] for i in range(impulse_start, impulse_end + 1)]
            impulse_vols = [self.data.volume[-i] for i in range(impulse_start, impulse_end + 1)]

            flag_upper = max(consol_highs)
            flag_lower = min(consol_lows)
            impulse_high = max(impulse_highs)
            impulse_low = min(impulse_lows)

            fib_0382 = impulse_high - (impulse_high - impulse_low) * self.p.fib_ratio

            pattern_hold_ok = flag_lower >= fib_0382
            vol_ok = statistics.mean(consol_vols) < statistics.mean(impulse_vols)
            breakout_ok = close > flag_upper

            if trend_ok and support_ok and pattern_hold_ok and vol_ok and breakout_ok:
                stop_candidate_channel = flag_lower - self.p.stop_buffer
                stop_price = max(stop_candidate_channel, fib_0382)

                risk_per_share = close - stop_price
                if risk_per_share <= 0:
                    return

                risk_amount = self.broker.getvalue() * self.p.risk_pct
                size = int(risk_amount / risk_per_share)
                if size <= 0:
                    return

                resistances = [lvl for lvl in (self.sma50[0], self.sma100[0], self.sma200[0]) if lvl > close]
                rr_target = close + self.p.rr_ratio * risk_per_share
                take_profit_price = min(resistances + [rr_target])

                self.pending_stop = stop_price
                self.pending_take_profit = take_profit_price
                self.order = self.buy(size=size)

        else:
            close = self.data.close[0]
            if self.stop_price is not None and close <= self.stop_price:
                self.order = self.close()
            elif self.take_profit_price is not None and close >= self.take_profit_price:
                self.order = self.close()
