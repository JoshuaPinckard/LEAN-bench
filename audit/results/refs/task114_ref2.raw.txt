import math
import backtrader as bt


class EmaBreakoutRetest(bt.Strategy):
    """
    Swing strategy on breakout and retest of the 200-period EMA.
    Main idea: Detect top-to-bottom breakout of 200 EMA, then open a long position
    upon retest of the 200 EMA level confirmed by a bullish bounce.
    """

    def __init__(self):
        self.ema200 = bt.indicators.EMA(self.data.close, period=200)
        self.sma60 = bt.indicators.SMA(self.data.close, period=60)

        self.order = None
        self.entry_price = None
        self.breakout_occurred = False

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            if order.isbuy():
                self.entry_price = order.executed.price
            elif order.issell():
                self.entry_price = None

        # Reset order tracking attribute for Completed, Canceled, Margin, Rejected
        self.order = None

    def next(self):
        # Guard pending orders
        if self.order:
            return

        # Condition 1: Asset price breaks 200-period EMA from top to bottom
        if self.data.close[-1] > self.ema200[-1] and self.data.close[0] < self.ema200[0]:
            self.breakout_occurred = True

        if self.position:
            if self.entry_price is None:
                return

            # Exit Condition 2: Stop-loss at -30% from entry point
            if self.data.low[0] <= self.entry_price * 0.70:
                self.order = self.close()
                return

            # Exit Condition 1: Take-profit at 60-period SMA level, or 5% if SMA60 is unavailable
            if not math.isnan(self.sma60[0]):
                tp_target = self.sma60[0]
            else:
                tp_target = self.entry_price * 1.05

            if self.data.high[0] >= tp_target:
                self.order = self.close()
                return

        else:
            # Condition 2 & 3: Retest and confirmation of support
            if self.breakout_occurred:
                retest = self.data.high[0] >= self.ema200[0]
                confirmation = self.data.close[0] > self.data.open[0]

                if retest and confirmation:
                    # Position sizing: 2% of total trading capital
                    target_value = self.broker.getvalue() * 0.02
                    size = int(target_value / self.data.close[0])

                    max_cash_shares = int(self.broker.getcash() / self.data.close[0])
                    size = min(size, max_cash_shares)

                    if size < 1 and max_cash_shares >= 1:
                        size = 1

                    if size > 0:
                        self.order = self.buy(size=size)
                        self.breakout_occurred = False
