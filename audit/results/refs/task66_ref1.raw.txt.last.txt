import backtrader as bt


class DoubleBottomStrategy(bt.Strategy):
    def __init__(self):
        self.sma200 = bt.indicators.SMA(self.data.close, period=200)
        self.volume = self.data.volume
        self.volume_sma20 = bt.indicators.SMA(self.volume, period=20)

        self.order = None
        self.stop_order = None
        self.target_order = None
        self.awaiting_protection_cancel = False

        self.first_bottom = None
        self.first_bottom_bar = None
        self.neckline = None
        self.second_bottom = None

    def _reset_pattern(self):
        self.first_bottom = None
        self.first_bottom_bar = None
        self.neckline = None
        self.second_bottom = None

    def next(self):
        if self.awaiting_protection_cancel:
            if self.stop_order is None and self.target_order is None and self.order is None:
                self.order = self.close()
                self.awaiting_protection_cancel = False
            return

        if self.position:
            if self.data.close[0] < self.sma200[0]:
                self.awaiting_protection_cancel = True
                if self.stop_order is not None:
                    self.cancel(self.stop_order)
                if self.target_order is not None:
                    self.cancel(self.target_order)
            return

        if self.order is not None or self.stop_order is not None or self.target_order is not None:
            return

        if self.data.close[0] <= self.sma200[0]:
            self._reset_pattern()
            return

        is_local_low = self.data.low[-1] < self.data.low[-2] and self.data.low[-1] <= self.data.low[0]
        is_local_high = self.data.high[-1] > self.data.high[-2] and self.data.high[-1] >= self.data.high[0]
        pivot_bar = len(self) - 1

        if self.first_bottom is None:
            if is_local_low:
                self.first_bottom = self.data.low[-1]
                self.first_bottom_bar = pivot_bar
            return

        if self.neckline is None:
            if is_local_low and self.data.low[-1] < self.first_bottom:
                self.first_bottom = self.data.low[-1]
                self.first_bottom_bar = pivot_bar
            elif is_local_high:
                self.neckline = self.data.high[-1]
            return

        if self.second_bottom is None:
            if is_local_high and self.data.high[-1] > self.neckline:
                self.neckline = self.data.high[-1]

            if is_local_low:
                bottom = self.data.low[-1]
                spacing = pivot_bar - self.first_bottom_bar
                if spacing >= 10 and abs(bottom - self.first_bottom) / self.first_bottom <= 0.01:
                    self.second_bottom = bottom
                elif bottom < self.first_bottom:
                    self.first_bottom = bottom
                    self.first_bottom_bar = pivot_bar
                    self.neckline = None
            return

        if self.data.close[0] > self.neckline and self.volume[0] > self.volume_sma20[0]:
            self.order = self.buy()

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status == order.Completed:
            if order.isbuy():
                entry_price = order.executed.price
                stop_price = self.second_bottom * 0.995
                risk = entry_price - stop_price

                self.stop_order = self.sell(
                    exectype=bt.Order.Stop,
                    price=stop_price,
                )
                self.target_order = self.sell(
                    exectype=bt.Order.Limit,
                    price=entry_price + (2 * risk),
                )
                self._reset_pattern()

            elif order.ref == getattr(self.stop_order, "ref", None):
                self.stop_order = None
                if self.target_order is not None:
                    self.cancel(self.target_order)

            elif order.ref == getattr(self.target_order, "ref", None):
                self.target_order = None
                if self.stop_order is not None:
                    self.cancel(self.stop_order)

        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            if order.ref == getattr(self.order, "ref", None):
                self.order = None
            if order.status != order.Completed and order.ref == getattr(self.stop_order, "ref", None):
                self.stop_order = None
            if order.status != order.Completed and order.ref == getattr(self.target_order, "ref", None):
                self.target_order = None
