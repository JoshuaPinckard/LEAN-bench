import backtrader as bt
import collections


class NVDAPreEarningsMomentumBreakout(bt.Strategy):
    # Fixed price levels and parameters taken verbatim from the specification.
    params = dict(
        support_low=6455.0,     # lower boundary of nearest support zone
        support_high=6465.0,    # upper boundary of nearest support zone
        key_support=6430.75,    # key support = 50% retracement
        resistance=6496.25,     # resistance / previous high (breakout level)
        target=6508.75,         # take-profit target (new ATH)
        sma_period=200,         # 200-period SMA on the daily timeframe
        risk_pct=0.01,          # fixed risk per trade (within the 1-2% limit)
        cash_buffer=0.95,       # affordability headroom vs available cash
        commission=0.001,       # harness commission (for affordability sizing)
    )

    def __init__(self):
        self.order = None
        # Build a daily-close series from the 1h feed to approximate the
        # 200-day SMA (the harness feed is 1h, not daily).
        self.daily_closes = collections.deque(maxlen=self.p.sma_period)
        self.cur_day = None
        # Tracks that a pullback down to the key support has occurred.
        self.pullback_seen = False

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return
        if order.status in (order.Completed, order.Canceled, order.Margin,
                             order.Rejected):
            self.order = None

    def _update_daily_sma(self):
        d = self.data.datetime.date(0)
        if self.cur_day is None:
            self.cur_day = d
        elif d != self.cur_day:
            # A new calendar day started: the previous bar closed the prior day.
            self.daily_closes.append(self.data.close[-1])
            self.cur_day = d

    def next(self):
        self._update_daily_sma()

        c = self.data.close[0]
        o = self.data.open[0]
        h = self.data.high[0]
        l = self.data.low[0]

        # Track a pullback down to the key support (50% retracement level).
        if l <= self.p.key_support:
            self.pullback_seen = True

        if self.order is not None:
            return

        # -------- Exits (while in a position) --------
        if self.position:
            # 1) Take-profit: target level (new ATH) reached.
            if h >= self.p.target:
                self.order = self.close()
                return
            # 2) Stop-loss: price drops below the nearest support zone (6455).
            if l <= self.p.support_low:
                self.order = self.close()
                return
            # 3) Alternative exit: bearish candle on increased volume.
            if c < o and self.data.volume[0] > self.data.volume[-1]:
                self.order = self.close()
                return
            return

        # -------- Entry (long only) --------
        # 200-day SMA trend confirmation must be available and bullish.
        if len(self.daily_closes) < self.p.sma_period:
            return
        sma200 = sum(self.daily_closes) / len(self.daily_closes)
        trend_ok = c > sma200

        # Condition 1: price above or within the nearest support zone.
        cond1 = c >= self.p.support_low
        # Condition 2: a pullback to key support occurred and price recovers.
        cond2 = self.pullback_seen and c > self.data.close[-1]
        # Condition 3: breakout above resistance 6496.25 with increased volume.
        breakout = (self.data.close[-1] <= self.p.resistance and
                    c > self.p.resistance)
        cond3 = breakout and self.data.volume[0] > self.data.volume[-1]

        if trend_ok and cond1 and cond2 and cond3:
            stop_dist = c - self.p.support_low
            if stop_dist <= 0:
                return
            # Size so the potential loss at the stop stays within the risk limit,
            # capped by what the available cash can actually afford.
            risk_budget = self.broker.getvalue() * self.p.risk_pct
            size = risk_budget / stop_dist
            max_affordable = (self.p.cash_buffer * self.broker.getcash() /
                              (c * (1 + self.p.commission)))
            size = min(size, max_affordable)
            if size > 0:
                self.order = self.buy(size=size)
                # Require a fresh pullback before any subsequent entry.
                self.pullback_seen = False
