import backtrader as bt
import math


class NVDAPreEarningsMomentumBreakout(bt.Strategy):
    params = (
        ("nearest_support_lower", 6455.0),
        ("nearest_support_upper", 6465.0),
        ("key_support", 6430.75),
        ("resistance", 6496.25),
        ("target", 6508.75),
        ("daily_sma_period", 200),
        ("risk_fraction", 0.01),
    )

    def __init__(self):
        self.order = None
        self.pullback_seen = False
        self._daily_closes = []
        self._last_date = None
        self._last_close = None
        self.daily_sma_200 = None

    def notify_order(self, order):
        if order.status in (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            self.order = None

    def next(self):
        current_date = self.data.datetime.date(0)

        if self._last_date is None:
            self._last_date = current_date
        elif current_date != self._last_date:
            self._daily_closes.append(self._last_close)
            if len(self._daily_closes) > self.p.daily_sma_period:
                self._daily_closes.pop(0)
            if len(self._daily_closes) == self.p.daily_sma_period:
                self.daily_sma_200 = (
                    sum(self._daily_closes) / self.p.daily_sma_period
                )
            self._last_date = current_date

        self._last_close = self.data.close[0]

        if self.order:
            return

        if self.position:
            if self.data.low[0] < self.p.nearest_support_lower:
                self.order = self.close()
            elif self.data.high[0] >= self.p.target:
                self.order = self.close()
            elif (
                self.data.close[0] < self.data.open[0]
                and self.data.volume[0] > self.data.volume[-1]
            ):
                self.order = self.close()
            return

        if self.data.low[0] <= self.p.key_support:
            self.pullback_seen = True

        support_condition = self.data.close[0] >= self.p.nearest_support_lower
        recovery_condition = (
            self.pullback_seen
            and self.data.close[0] > self.data.close[-1]
            and self.data.close[0] > self.p.key_support
        )
        breakout_condition = (
            self.data.close[0] > self.p.resistance
            and self.data.volume[0] > self.data.volume[-1]
        )
        trend_condition = (
            self.daily_sma_200 is not None
            and self.data.close[0] > self.daily_sma_200
        )

        if (
            support_condition
            and recovery_condition
            and breakout_condition
            and trend_condition
        ):
            stop_distance = self.data.close[0] - self.p.nearest_support_lower
            risk_budget = self.broker.getvalue() * self.p.risk_fraction
            size = math.floor(risk_budget / stop_distance)

            if size > 0:
                self.order = self.buy(size=size)
                self.pullback_seen = False
