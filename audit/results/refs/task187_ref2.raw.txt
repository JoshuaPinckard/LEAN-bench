import backtrader as bt


class PeriodicRebalancingStrategy(bt.Strategy):
    """Reference implementation of a single-asset periodic rebalancing strategy.

    Maintains a 100% allocation in the asset, forcibly rebalancing every 12 weeks
    via a full close and immediate re-open, and closing the position at 52 weeks.
    """

    def __init__(self):
        self.order = None
        self.week_count = 0
        self.last_week_tuple = None
        self.last_rebalance_week = -1
        self.has_entered = False
        self.rebalancing = False
        self.need_rebuy = False

    def notify_order(self, order):
        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            self.order = None

        if order.status == order.Completed:
            if self.rebalancing:
                self.rebalancing = False
                self.need_rebuy = True
        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            if self.rebalancing:
                self.rebalancing = False

    def next(self):
        dt = self.data.datetime.date(0)
        current_week_tuple = dt.isocalendar()[:2]

        if self.last_week_tuple is None:
            self.last_week_tuple = current_week_tuple
        elif current_week_tuple != self.last_week_tuple:
            self.week_count += 1
            self.last_week_tuple = current_week_tuple

        # Handle immediate re-open after a rebalance close order completes
        if self.need_rebuy and self.order is None:
            self.need_rebuy = False
            cash = self.broker.getcash()
            price = self.data.close[0]
            if price > 0 and cash > 0:
                size = int((cash * 0.998) / price)
                if size > 0:
                    self.order = self.buy(size=size)

        # Guard pending orders
        if self.order is not None:
            return

        # Initial entry on the first available bar
        if not self.has_entered:
            cash = self.broker.getcash()
            price = self.data.close[0]
            if price > 0 and cash > 0:
                size = int((cash * 0.998) / price)
                if size > 0:
                    self.order = self.buy(size=size)
                    self.has_entered = True
            return

        # Exit at the end of 52 weeks
        if self.week_count >= 52:
            if self.position.size > 0:
                self.order = self.close()
            return

        # Rebalance every 12 weeks
        if (
            self.week_count > 0
            and self.week_count % 12 == 0
            and self.week_count != self.last_rebalance_week
        ):
            if self.position.size > 0:
                self.last_rebalance_week = self.week_count
                self.rebalancing = True
                self.order = self.close()
