```python
import backtrader as bt
import random
import math
import statistics


class PeriodicRebalancingStrategy(bt.Strategy):

    def __init__(self):
        self._closes = []
        self._dates = []

    def next(self):
        self._closes.append(float(self.data.close[0]))
        self._dates.append(self.data.datetime.date(0))

    def stop(self):
        # Resample daily bars to weekly: last daily close of each ISO calendar week
        week_map = {}
        week_order = []
        for date, close in zip(self._dates, self._closes):
            key = date.isocalendar()[:2]   # (ISO year, ISO week number)
            if key not in week_map:
                week_order.append(key)
            week_map[key] = close          # last close of week wins
        weekly_closes = [week_map[k] for k in week_order]
        n_weeks = len(weekly_closes)

        # Valid start indices: w0+52 must be a defined index
        valid_starts = [i for i in range(n_weeks) if i + 52 < n_weeks]

        random.seed(42)
        n_runs = min(50, len(valid_starts))
        start_indices = random.sample(valid_starts, n_runs)

        commission = 0.001          # 0.1% per trade side
        initial_cash = 10000.0
        rebalance_at = {12, 24, 36, 48}   # rebalance weeks within 52-week run

        total_returns = []
        max_drawdowns = []
        sharpe_ratios = []

        for w0 in start_indices:
            closes = [weekly_closes[w0 + k] for k in range(53)]   # weeks 0..52

            # Initial buy at close of start week
            shares = initial_cash / (closes[0] * (1.0 + commission))
            pv = [shares * closes[0]]      # portfolio value after initial buy

            for k in range(1, 53):
                price = closes[k]
                if k == 52:
                    # Final exit: sell, collect cash
                    cash = shares * price * (1.0 - commission)
                    pv.append(cash)
                elif k in rebalance_at:
                    # Full close then immediate re-open at same week close
                    cash = shares * price * (1.0 - commission)
                    shares = cash / (price * (1.0 + commission))
                    pv.append(shares * price)
                else:
                    pv.append(shares * price)

            # Total return relative to initial cash
            total_returns.append((pv[-1] - initial_cash) / initial_cash)

            # Maximum drawdown (negative fraction)
            peak = pv[0]
            mdd = 0.0
            for v in pv:
                if v > peak:
                    peak = v
                dd = (v - peak) / peak
                if dd < mdd:
                    mdd = dd
            max_drawdowns.append(mdd)

            # Annualised Sharpe ratio from 52 weekly returns, risk-free rate = 0
            weekly_rets = [(pv[i] - pv[i - 1]) / pv[i - 1] for i in range(1, 53)]
            mean_r = statistics.mean(weekly_rets)
            std_r = statistics.stdev(weekly_rets)   # sample stdev (N-1)
            sharpe = (mean_r / std_r * math.sqrt(52)) if std_r > 0.0 else 0.0
            sharpe_ratios.append(sharpe)

        avg_tr = statistics.mean(total_returns)
        avg_mdd = statistics.mean(max_drawdowns)
        avg_sharpe = statistics.mean(sharpe_ratios)

        print(f"Average Total Return: {avg_tr:.6f}")
        print(f"Average Max Drawdown: {avg_mdd:.6f}")
        print(f"Average Sharpe Ratio: {avg_sharpe:.6f}")

        self.avg_total_return = avg_tr
        self.avg_max_drawdown = avg_mdd
        self.avg_sharpe_ratio = avg_sharpe
```
