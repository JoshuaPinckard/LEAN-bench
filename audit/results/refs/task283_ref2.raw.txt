import math
import backtrader as bt


class DistributedSellStrategy(bt.Strategy):
    def __init__(self):
        self.order = None
        self.has_entered = False
        self.initial_investment = None
        self.cost_basis = None
        self.n = 1

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            if order.isbuy() and self.initial_investment is None:
                entry_price = self.data.close[0]
                self.initial_investment = 400.0 * entry_price
                self.cost_basis = self.initial_investment

        self.order = None

    def next(self):
        if self.order is not None:
            return

        if not self.has_entered:
            self.order = self.buy(size=400)
            self.has_entered = True
            return

        if self.position.size > 0 and self.initial_investment is not None:
            position_value = self.position.size * self.data.close[0]
            threshold_n = self.initial_investment * (2**self.n)

            if position_value >= threshold_n:
                sell_shares = math.floor(self.position.size * 0.5)
                if sell_shares > 0:
                    self.order = self.sell(size=sell_shares)
                    self.cost_basis *= 0.5
                    self.n += 1
