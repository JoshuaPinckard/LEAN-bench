import backtrader as bt


class Strategy(bt.Strategy):
    def __init__(self):
        self.hma = bt.indicators.HullMovingAverage(self.data.close, period=9)
        self.bbands = bt.indicators.BollingerBands(
            self.data.close, period=20, devfactor=2.0
        )
        self.bbwidth = (self.bbands.top - self.bbands.bot) / self.bbands.mid
        self.bbwidth_sma = bt.indicators.SimpleMovingAverage(
            self.bbwidth, period=50
        )
        self.atr = bt.indicators.ATR(self.data, period=14)

        self.entry_order = None
        self.exit_order = None
        self.stop_order = None
        self.stop_loss = None

    def notify_order(self, order):
        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            if order is self.entry_order:
                self.entry_order = None

                if order.status == order.Completed:
                    if self.position.size > 0:
                        self.stop_loss = order.executed.price - (self.atr[0] * 2.0)
                        self.stop_order = self.sell(
                            exectype=bt.Order.Stop,
                            price=self.stop_loss,
                        )
                    elif self.position.size < 0:
                        self.stop_loss = order.executed.price + (self.atr[0] * 2.0)
                        self.stop_order = self.buy(
                            exectype=bt.Order.Stop,
                            price=self.stop_loss,
                        )

            elif order is self.exit_order:
                self.exit_order = None
                if order.status == order.Completed:
                    self.stop_loss = None

            elif order is self.stop_order:
                self.stop_order = None
                if order.status == order.Completed:
                    self.stop_loss = None

    def next(self):
        long_signal = (
            self.data.close[0] > self.hma[0]
            and self.data.close[-1] <= self.hma[-1]
            and self.hma[0] > self.hma[-1]
            and self.bbwidth[0] < self.bbwidth_sma[0] * 0.8
        )
        short_signal = (
            self.data.close[0] < self.hma[0]
            and self.data.close[-1] >= self.hma[-1]
            and self.hma[0] < self.hma[-1]
            and self.bbwidth[0] < self.bbwidth_sma[0] * 0.8
        )

        if self.position.size == 0:
            if self.entry_order is None and self.exit_order is None:
                if long_signal:
                    self.entry_order = self.buy()
                elif short_signal:
                    self.entry_order = self.sell()

        elif self.exit_order is None:
            if (
                (self.position.size > 0 and short_signal)
                or (self.position.size < 0 and long_signal)
            ):
                if self.stop_order is not None:
                    self.cancel(self.stop_order)
                    self.stop_order = None
                self.exit_order = self.close()
