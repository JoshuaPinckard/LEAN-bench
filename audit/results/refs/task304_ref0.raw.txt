import backtrader as bt


class TweezerBottomStrategy(bt.Strategy):
    def __init__(self):
        self.sma20 = bt.indicators.SMA(self.data.close, period=20)
        self.order = None
        self.stop_order = None
        self.pending_stop_price = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status == order.Completed:
            if order == self.order:
                self.order = None
                if order.isbuy():
                    self.stop_order = self.sell(
                        exectype=bt.Order.Stop,
                        price=self.pending_stop_price,
                    )
            elif order == self.stop_order:
                self.stop_order = None
        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            if order == self.order:
                self.order = None
            elif order == self.stop_order:
                self.stop_order = None

    def next(self):
        if self.order:
            return

        if not self.position:
            decline = (self.data.close[0] < self.sma20[0]) and \
                      (self.data.close[0] < self.data.close[-5])

            tweezer = (
                abs(self.data.low[-2] - self.data.low[-1]) <= 0.001 * self.data.close[-1]
                and self.data.close[-2] < self.data.open[-2]
                and self.data.close[-1] > self.data.open[-1]
            )

            confirmation = self.data.close[0] > self.data.high[-1]

            if decline and tweezer and confirmation:
                self.pending_stop_price = min(self.data.low[-2], self.data.low[-1]) * 0.98
                self.order = self.buy()
        else:
            bearish_engulfing = (
                self.data.close[0] < self.data.open[0]
                and self.data.open[0] >= self.data.close[-1]
                and self.data.close[0] <= self.data.open[-1]
            )

            if bearish_engulfing:
                if self.stop_order:
                    self.cancel(self.stop_order)
                    self.stop_order = None
                self.order = self.close()
