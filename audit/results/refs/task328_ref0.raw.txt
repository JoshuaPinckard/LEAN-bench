import backtrader as bt


class SwingBreakoutStrategy(bt.Strategy):
    def __init__(self):
        self.sma = bt.indicators.SMA(self.data.close, period=20)

        self.swing_high = bt.indicators.Highest(self.data.high(-1), period=5)
        self.swing_low = bt.indicators.Lowest(self.data.low(-1), period=5)

        self.cross_high = bt.indicators.CrossOver(self.data.close, self.swing_high)
        self.cross_low = bt.indicators.CrossOver(self.data.close, self.swing_low)

        self.order = None

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        if self.order:
            return

        if not self.position:
            if self.cross_high[0] > 0 and self.data.close[0] > self.sma[0]:
                self.order = self.buy()
            elif self.cross_low[0] < 0 and self.data.close[0] < self.sma[0]:
                self.order = self.sell()
        else:
            if self.position.size > 0:
                if self.cross_low[0] < 0:
                    self.order = self.close()
            elif self.position.size < 0:
                if self.cross_high[0] > 0:
                    self.order = self.close()
