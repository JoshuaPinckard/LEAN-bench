import math
import backtrader as bt


class ShortTermEnergyETFStrategy(bt.Strategy):
    """
    Short-term speculative strategy on an energy sector ETF.
    Reflects all rules from the specification precisely.
    """

    def __init__(self):
        self.sma50 = bt.indicators.SimpleMovingAverage(self.data.close, period=50)
        self.rsi14 = bt.indicators.RelativeStrengthIndex(self.data.close, period=14)

        self.order = None
        self.entry_price = None
        self.entry_date = None

        # Min period set to 51 to safely access close[-50]
        self.addminperiod(51)

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            if order.isbuy():
                self.entry_price = order.executed.price
                self.entry_date = self.data.datetime.date(0)
            elif order.issell():
                self.entry_price = None
                self.entry_date = None

        self.order = None

    def next(self):
        if self.order:
            return

        if not self.position:
            # Condition 1: Closing price increases by 2% or more in one trading day
            price_increase = (self.data.close[0] - self.data.close[-1]) / self.data.close[-1] >= 0.02

            # Condition 2: 50-period SMA higher than closing price 50 days ago
            trend_filter = self.sma50[0] > self.data.close[-50]

            # Condition 3: RSI(14) between 40 and 65 inclusive
            momentum_filter = 40.0 <= self.rsi14[0] <= 65.0

            if price_increase and trend_filter and momentum_filter:
                portfolio_value = self.broker.getvalue()
                target_position_value = portfolio_value * 0.05
                size = int(target_position_value / self.data.close[0])

                if size > 0:
                    self.order = self.buy(size=size, exectype=bt.Order.Close)

        else:
            if self.entry_price is None or self.entry_date is None:
                return

            current_gain = (self.data.close[0] - self.entry_price) / self.entry_price
            days_open = (self.data.datetime.date(0) - self.entry_date).days

            # Exit Condition 1: Take-profit at 10% gain
            take_profit = current_gain >= 0.10

            # Exit Condition 2: Stop-loss at 5% loss
            stop_loss = current_gain <= -0.05

            # Exit Condition 3: Open for more than 90 calendar days
            time_exit = days_open > 90

            if take_profit or stop_loss or time_exit:
                self.order = self.close(exectype=bt.Order.Close)
