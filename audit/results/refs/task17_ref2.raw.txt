import backtrader as bt


class MeanReversionORB(bt.Strategy):
    params = (
        ("risk_pct", 0.01),
        ("ny_start_hour", 14),
        ("ny_end_trade_hour", 17),
        ("eod_exit_hour", 21),
    )

    def __init__(self):
        self.order = None
        self.orb_high = None
        self.orb_low = None
        self.orb_mid = None
        self.orb_ready = False
        self.was_below_lower = False
        self.was_above_upper = False
        self.current_day = None

    def notify_order(self, order):
        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            self.order = None

    def next(self):
        d = self.data.datetime.date(0)
        t = self.data.datetime.time(0)

        if self.current_day != d:
            self.current_day = d
            self.orb_high = None
            self.orb_low = None
            self.orb_mid = None
            self.orb_ready = False
            self.was_below_lower = False
            self.was_above_upper = False

        if t.hour == self.p.ny_start_hour:
            if self.orb_high is None:
                self.orb_high = self.data.high[0]
                self.orb_low = self.data.low[0]
            else:
                self.orb_high = max(self.orb_high, self.data.high[0])
                self.orb_low = min(self.orb_low, self.data.low[0])

        if (
            not self.orb_ready
            and self.orb_high is not None
            and t.hour >= self.p.ny_start_hour + 1
        ):
            self.orb_mid = (self.orb_high + self.orb_low) / 2.0
            self.orb_ready = True

        if self.position:
            if t.hour >= self.p.eod_exit_hour:
                if self.order is None:
                    self.order = self.close()
                return

            if self.position.size > 0:
                if (
                    self.data.low[0] <= self.orb_low
                    or self.data.high[0] >= self.orb_mid
                ):
                    if self.order is None:
                        self.order = self.close()
            elif self.position.size < 0:
                if (
                    self.data.high[0] >= self.orb_high
                    or self.data.low[0] <= self.orb_mid
                ):
                    if self.order is None:
                        self.order = self.close()
            return

        if (
            self.orb_ready
            and (
                self.p.ny_start_hour + 1 <= t.hour < self.p.ny_end_trade_hour
            )
            and self.order is None
        ):
            if self.data.low[0] < self.orb_low:
                self.was_below_lower = True
            if self.data.high[0] > self.orb_high:
                self.was_above_upper = True

            if self.was_below_lower and self.data.close[0] > self.orb_low:
                risk_per_unit = self.data.close[0] - self.orb_low
                if risk_per_unit > 0:
                    capital = self.broker.getvalue()
                    risk_amount = capital * self.p.risk_pct
                    size = int(risk_amount / risk_per_unit)
                    if size > 0:
                        self.order = self.buy(size=size)
                        self.was_below_lower = False

            elif self.was_above_upper and self.data.close[0] < self.orb_high:
                risk_per_unit = self.orb_high - self.data.close[0]
                if risk_per_unit > 0:
                    capital = self.broker.getvalue()
                    risk_amount = capital * self.p.risk_pct
                    size = int(risk_amount / risk_per_unit)
                    if size > 0:
                        self.order = self.sell(size=size)
                        self.was_above_upper = False
