import backtrader as bt


class MeanReversionStrategy(bt.Strategy):
    params = (
        ("bb_period", 20),
        ("bb_dev", 2.0),
        ("vol_sma_short", 20),
        ("vol_sma_long", 50),
        ("rsi_period", 14),
        ("rsi_thresh", 30),
        ("vol_multiplier", 1.5),
        ("liquidity_thresh", 1000000),
        ("trade_size", 100),
        ("max_bars_held", 3),
        ("daily_loss_limit", 0.005),
    )

    def __init__(self):
        self.bb = bt.indicators.BollingerBands(
            self.data.close,
            period=self.p.bb_period,
            devfactor=self.p.bb_dev,
        )
        self.vol_sma20 = bt.indicators.SMA(
            self.data.volume, period=self.p.vol_sma_short
        )
        self.vol_sma50 = bt.indicators.SMA(
            self.data.volume, period=self.p.vol_sma_long
        )
        self.rsi = bt.indicators.RSI(
            self.data.close, period=self.p.rsi_period
        )
        self.crossover = bt.indicators.CrossOver(
            self.data.close, self.bb.lines.mid
        )

        self.order = None
        self.entry_bar = None
        self.prev_val = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status == order.Completed:
            if order.isbuy():
                self.entry_bar = len(self)
            self.order = None
        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        current_val = self.broker.getvalue()

        if self.prev_val is None:
            self.prev_val = current_val

        daily_loss_reached = False
        if self.prev_val > 0:
            loss_pct = (self.prev_val - current_val) / self.prev_val
            if loss_pct >= self.p.daily_loss_limit:
                daily_loss_reached = True

        self.prev_val = current_val

        if self.order:
            return

        if not self.position:
            if daily_loss_reached:
                return

            cond1 = self.data.close[0] < self.bb.lines.bot[0]
            cond2 = (
                self.data.volume[0]
                > self.p.vol_multiplier * self.vol_sma20[0]
            )
            cond3 = self.rsi[0] < self.p.rsi_thresh
            cond4 = self.vol_sma50[0] > self.p.liquidity_thresh

            if cond1 and cond2 and cond3 and cond4:
                self.order = self.buy(size=self.p.trade_size)
        else:
            bars_held = (
                len(self) - self.entry_bar if self.entry_bar is not None else 0
            )
            if self.crossover[0] > 0 or bars_held >= self.p.max_bars_held:
                self.order = self.close()
