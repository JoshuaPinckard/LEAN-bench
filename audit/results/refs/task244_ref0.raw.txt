import backtrader as bt
import math


class LogReturn(bt.Indicator):
    lines = ('logret',)

    def __init__(self):
        self.addminperiod(2)

    def next(self):
        self.lines.logret[0] = math.log(self.data[0] / self.data[-1])


class MeanVarianceStrategy(bt.Strategy):
    params = (
        ('mu_period', 252),
        ('atr_period', 63),
        ('lam', 2.5),
        ('vol_threshold', 0.005),
        ('wmin', -1.0),
        ('wmax', 1.0),
    )

    def __init__(self):
        self.logret = LogReturn(self.data.close)
        self.mu_hat = bt.indicators.SMA(self.logret, period=self.p.mu_period)
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.order = None
        self.last_rebalance = None

    def notify_order(self, order):
        if order.status in (order.Completed, order.Canceled, order.Margin, order.Rejected):
            self.order = None

    def next(self):
        if self.order:
            return

        dt = self.data.datetime.date(0)
        key = (dt.year, dt.month)
        if key == self.last_rebalance:
            return
        self.last_rebalance = key

        sigma_hat = self.atr[0] / self.data.close[0]
        mu_hat = self.mu_hat[0]

        if sigma_hat < self.p.vol_threshold:
            w = 0.0
        else:
            w = mu_hat / (self.p.lam * sigma_hat ** 2)
            w = max(self.p.wmin, min(self.p.wmax, w))

        self.order = self.order_target_percent(target=w)
