import backtrader as bt


class EMAForecastingForPositionEntry(bt.Strategy):
    params = (
        ("ema_period", 21),
        ("threshold", 0.005),
        ("stop_loss", 0.02),
        ("take_profit", 0.03),
    )

    def __init__(self):
        self.ema = bt.indicators.ExponentialMovingAverage(
            self.data.close, period=self.p.ema_period
        )
        self.k = 2.0 / (self.p.ema_period + 1.0)

        self.previous_forecast = None
        self.entry_order = None
        self.exit_order = None
        self.stop_order = None
        self.take_profit_order = None

    def _has_pending_entry_or_exit(self):
        return self.entry_order is not None or self.exit_order is not None

    def _cancel_protective_orders(self):
        if self.stop_order is not None:
            self.cancel(self.stop_order)
        if self.take_profit_order is not None:
            self.cancel(self.take_profit_order)

    def _submit_protective_orders(self, entry_price):
        size = abs(self.position.size)

        if self.position.size > 0:
            self.stop_order = self.sell(
                exectype=bt.Order.Stop,
                price=entry_price * (1.0 - self.p.stop_loss),
                size=size,
            )
            self.take_profit_order = self.sell(
                exectype=bt.Order.Limit,
                price=entry_price * (1.0 + self.p.take_profit),
                size=size,
            )
        else:
            self.stop_order = self.buy(
                exectype=bt.Order.Stop,
                price=entry_price * (1.0 + self.p.stop_loss),
                size=size,
            )
            self.take_profit_order = self.buy(
                exectype=bt.Order.Limit,
                price=entry_price * (1.0 - self.p.take_profit),
                size=size,
            )

    def next(self):
        forecast = (
            self.data.close[0] * self.k
            + self.ema[0] * (1.0 - self.k)
        )

        if self._has_pending_entry_or_exit():
            self.previous_forecast = forecast
            return

        if self.position:
            if self.previous_forecast is not None:
                if (
                    self.position.size > 0
                    and self.data.close[-1] > self.previous_forecast
                    and self.data.close[0] < self.previous_forecast
                ):
                    self._cancel_protective_orders()
                    self.exit_order = self.close()

                elif (
                    self.position.size < 0
                    and self.data.close[-1] < self.previous_forecast
                    and self.data.close[0] > self.previous_forecast
                ):
                    self._cancel_protective_orders()
                    self.exit_order = self.close()
        else:
            size = self.broker.getcash() / self.data.close[0]

            if self.data.close[0] <= forecast * (1.0 - self.p.threshold):
                self.entry_order = self.buy(size=size)

            elif self.data.close[0] >= forecast * (1.0 + self.p.threshold):
                self.entry_order = self.sell(size=size)

        self.previous_forecast = forecast

    def notify_order(self, order):
        if order.status in (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            if order is self.entry_order:
                self.entry_order = None
                if order.status == order.Completed:
                    self._submit_protective_orders(order.executed.price)

            elif order is self.exit_order:
                self.exit_order = None

            elif order is self.stop_order:
                self.stop_order = None
                if order.status == order.Completed:
                    if self.take_profit_order is not None:
                        self.cancel(self.take_profit_order)

            elif order is self.take_profit_order:
                self.take_profit_order = None
                if order.status == order.Completed:
                    if self.stop_order is not None:
                        self.cancel(self.stop_order)
