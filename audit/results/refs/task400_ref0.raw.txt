```python
import backtrader as bt


class AdaptiveChannelBreakoutStrategy(bt.Strategy):
    params = dict(
        lookback=10,     # channel lookback period (adjustable)
        stop_mult=2,     # ATR stop multiplier (adjustable)
        atr_period=14,
        fast_period=2,
        slow_period=30,
    )

    def __init__(self):
        self.fast_ema = bt.indicators.EMA(self.data.close, period=self.p.fast_period)
        self.slow_ema = bt.indicators.EMA(self.data.close, period=self.p.slow_period)
        self.highest_high = bt.indicators.Highest(self.data.high, period=self.p.lookback)
        self.lowest_low = bt.indicators.Lowest(self.data.low, period=self.p.lookback)
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)

        self.order = None
        self.stop_price = None
        self.entry_atr = None

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return

        if order.status == order.Completed:
            if order.isbuy() and self.position.size > 0:
                self.stop_price = order.executed.price - self.p.stop_mult * self.entry_atr
            elif order.issell() and self.position.size < 0:
                self.stop_price = order.executed.price + self.p.stop_mult * self.entry_atr
            else:
                self.stop_price = None

        if order.status in (order.Canceled, order.Margin, order.Rejected):
            self.stop_price = None

        self.order = None

    def next(self):
        if self.order:
            return

        # channel measured on the prior `lookback` bars (excludes current bar,
        # since a window including the current bar's own high/low could never
        # be broken by that same bar's close)
        upper_band = self.highest_high[-1]
        lower_band = self.lowest_low[-1]

        long_signal = self.data.close[0] > upper_band and self.fast_ema[0] > self.slow_ema[0]
        short_signal = self.data.close[0] < lower_band and self.fast_ema[0] < self.slow_ema[0]

        if self.position:
            if self.position.size > 0:
                stop_hit = self.data.low[0] <= self.stop_price
                if stop_hit or short_signal:
                    self.order = self.close()
            elif self.position.size < 0:
                stop_hit = self.data.high[0] >= self.stop_price
                if stop_hit or long_signal:
                    self.order = self.close()
        else:
            if long_signal:
                self.entry_atr = self.atr[0]
                self.order = self.buy()
            elif short_signal:
                self.entry_atr = self.atr[0]
                self.order = self.sell()
```
