import backtrader as bt


class DonchianContractionBreakout(bt.Strategy):
    params = (
        ("donchian_period", 20),
        ("atr_period", 14),
        ("atr_mult", 2.0),
    )

    def __init__(self):
        # Donchian channel upper and lower bands over period 20, shifted by 1 bar
        # so that breakout comparisons are evaluated against prior bars' channel.
        self.donchian_high = bt.indicators.Highest(
            self.data.high, period=self.p.donchian_period
        )(-1)
        self.donchian_low = bt.indicators.Lowest(
            self.data.low, period=self.p.donchian_period
        )(-1)

        self.channel_width = self.donchian_high - self.donchian_low
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)

        # Volatility contraction: channel width < 2 * ATR(14)
        self.contraction = self.channel_width < (self.p.atr_mult * self.atr)

        # Crossover indicators
        self.cross_upper = bt.indicators.CrossOver(self.data.close, self.donchian_high)
        self.cross_lower = bt.indicators.CrossOver(self.data.close, self.donchian_low)

        self.order = None
        self.stop_price = None

    def notify_order(self, order):
        if order.status in [order.Completed]:
            if order.isbuy():
                if self.position.size > 0:
                    self.stop_price = order.executed.price - (self.p.atr_mult * self.atr[0])
            elif order.issell():
                if self.position.size < 0:
                    self.stop_price = order.executed.price + (self.p.atr_mult * self.atr[0])

            if self.position.size == 0:
                self.stop_price = None

        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        if self.order:
            return

        # Signals require crossover and active contraction (prior or current setup bar)
        long_signal = (self.cross_upper[0] == 1) and (self.contraction[-1] or self.contraction[0])
        short_signal = (self.cross_lower[0] == -1) and (self.contraction[-1] or self.contraction[0])

        if self.position.size > 0:
            # Long position management: Stop-loss or opposite breakout signal
            if self.stop_price is not None and self.data.low[0] <= self.stop_price:
                self.order = self.close()
            elif short_signal:
                self.order = self.close()
        elif self.position.size < 0:
            # Short position management: Stop-loss or opposite breakout signal
            if self.stop_price is not None and self.data.high[0] >= self.stop_price:
                self.order = self.close()
            elif long_signal:
                self.order = self.close()
        else:
            # Entry logic when flat
            if long_signal:
                self.order = self.buy()
            elif short_signal:
                self.order = self.sell()
