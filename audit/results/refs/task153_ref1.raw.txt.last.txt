import backtrader as bt


class MeanReversionZScoreStrategy(bt.Strategy):
    params = (("k", 3),)

    def __init__(self):
        self.sma = bt.indicators.SMA(self.data.close, period=100)
        self.bbands = bt.indicators.BollingerBands(
            self.data.close, period=100, devfactor=2
        )
        self.atr = bt.indicators.ATR(self.data, period=14)
        self.crossover = bt.indicators.CrossOver(self.data.close, self.sma)

        self.order = None
        self.entry_z = None
        self.entry_atr = None
        self.entry_price = None

    def notify_order(self, order):
        if order.status == order.Completed:
            if order.isbuy() or order.issell():
                if self.position.size != 0 and self.entry_price is None:
                    self.entry_price = order.executed.price
                    self.entry_z = self.zscore()
                    self.entry_atr = float(self.atr[0])

        if order.status in (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            self.order = None

    def zscore(self):
        mid = float(self.sma[0])
        top = float(self.bbands.top[0])
        denominator = (top - mid) / 2.0
        return (float(self.data.close[0]) - mid) / denominator

    def next(self):
        if self.order:
            return

        z = self.zscore()

        if not self.position:
            if z <= -2 or z >= 2:
                atr = float(self.atr[0])
                size = int((self.broker.getvalue() * 0.01) / atr)

                if size <= 0:
                    return

                if z <= -2:
                    self.order = self.buy(size=size)
                else:
                    self.order = self.sell(size=size)
            return

        half_life = abs(z) <= 0.5 * abs(self.entry_z)

        if self.position.size > 0:
            price_stop = self.entry_price - self.p.k * self.entry_atr
            if (
                z <= -3
                or float(self.data.close[0]) <= price_stop
                or self.crossover[0] > 0
                or half_life
            ):
                self.order = self.close()
        else:
            price_stop = self.entry_price + self.p.k * self.entry_atr
            if (
                z >= 3
                or float(self.data.close[0]) >= price_stop
                or self.crossover[0] < 0
                or half_life
            ):
                self.order = self.close()
