import backtrader as bt


class DonchianChannel(bt.Indicator):
    lines = ('hi', 'lo')
    params = (('period', 20),)

    def __init__(self):
        self.lines.hi = bt.ind.Highest(self.data.high, period=self.p.period)
        self.lines.lo = bt.ind.Lowest(self.data.low, period=self.p.period)


bt.ind.DonchianChannel = DonchianChannel


class DonchianWidthMeanReversion(bt.Strategy):
    params = (
        ('period', 20),
        ('deviation_multiplier', 2.0),
        ('stop_pct', 0.02),
    )

    def __init__(self):
        self.donchian = bt.ind.DonchianChannel(period=self.p.period)
        self.width = self.donchian.hi - self.donchian.lo
        self.width_sma = bt.ind.SMA(self.width, period=self.p.period)
        self.upper_extreme = self.width_sma * self.p.deviation_multiplier
        self.lower_extreme = self.width_sma / self.p.deviation_multiplier

        self.cross_upper = bt.ind.CrossOver(self.width, self.upper_extreme)
        self.cross_lower = bt.ind.CrossOver(self.width, self.lower_extreme)
        self.cross_sma = bt.ind.CrossOver(self.width, self.width_sma)

        self.upper_regime = False
        self.lower_regime = False

        self.order = None
        self.stop_order = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            if order == self.order:
                self.order = None
                if self.position:
                    if self.position.size > 0:
                        stop_price = order.executed.price * (
                            1.0 - self.p.stop_pct
                        )
                        self.stop_order = self.sell(
                            exectype=bt.Order.Stop, price=stop_price
                        )
                    elif self.position.size < 0:
                        stop_price = order.executed.price * (
                            1.0 + self.p.stop_pct
                        )
                        self.stop_order = self.buy(
                            exectype=bt.Order.Stop, price=stop_price
                        )
            elif order == self.stop_order:
                self.stop_order = None

        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            if order == self.order:
                self.order = None
            if order == self.stop_order:
                self.stop_order = None

    def next(self):
        if self.cross_upper[0] > 0:
            self.upper_regime = True
            self.lower_regime = False
        elif self.cross_lower[0] < 0:
            self.lower_regime = True
            self.upper_regime = False

        if self.order is not None:
            return

        if self.position:
            if self.position.size > 0:
                if self.cross_sma[0] > 0:
                    if self.stop_order:
                        self.cancel(self.stop_order)
                        self.stop_order = None
                    self.order = self.close()
            elif self.position.size < 0:
                if self.cross_sma[0] < 0:
                    if self.stop_order:
                        self.cancel(self.stop_order)
                        self.stop_order = None
                    self.order = self.close()
        else:
            if self.upper_regime and self.cross_upper[0] < 0:
                self.order = self.sell()
            elif self.lower_regime and self.cross_lower[0] > 0:
                self.order = self.buy()
