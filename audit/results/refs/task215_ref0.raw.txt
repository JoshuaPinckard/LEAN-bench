import backtrader as bt


class RRGProxyMomentum(bt.Strategy):
    params = (
        ('sma_fast_period', 50),
        ('sma_slow_period', 200),
        ('macd_fast', 12),
        ('macd_slow', 26),
        ('macd_signal', 9),
        ('rsi_period', 14),
        ('atr_period', 14),
        ('rsi_cross_level', 50),
        ('rsi_tp_level', 70),
        ('stop_pct', 0.02),
        ('atr_mult', 2.0),
        ('take_pct', 0.07),
        ('use_tp', True),
        ('risk_pct', 0.01),
        ('size_cap_pct', 0.10),
        ('use_volume_filter', False),
        ('volume_sma_period', 20),
    )

    def __init__(self):
        self.sma_fast = bt.indicators.SMA(self.data.close, period=self.p.sma_fast_period)
        self.sma_slow = bt.indicators.SMA(self.data.close, period=self.p.sma_slow_period)
        self.macd = bt.indicators.MACDHisto(
            self.data.close,
            period_me1=self.p.macd_fast,
            period_me2=self.p.macd_slow,
            period_signal=self.p.macd_signal,
        )
        self.rsi = bt.indicators.RSI(self.data.close, period=self.p.rsi_period)
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.macd_cross = bt.indicators.CrossOver(self.macd.macd, self.macd.signal)
        self.rsi_cross = bt.indicators.CrossOver(self.rsi, self.p.rsi_cross_level)
        self.vol_sma = bt.indicators.SMA(self.data.volume, period=self.p.volume_sma_period)

        self.order = None
        self.entry_price = None
        self.stop_amount = None

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return
        if order.status == order.Completed:
            if order.isbuy():
                self.entry_price = order.executed.price
                self.stop_amount = max(
                    self.p.stop_pct * self.entry_price,
                    self.p.atr_mult * self.atr[0],
                )
            else:
                self.entry_price = None
                self.stop_amount = None
        self.order = None

    def next(self):
        if self.order:
            return

        close = self.data.close[0]
        sma_f = self.sma_fast[0]
        sma_s = self.sma_slow[0]

        if self.position:
            hist = self.macd.histo[0]
            rsi = self.rsi[0]

            exit_momentum_weak = hist < 0 and close > sma_s
            exit_trend_reversal = close < sma_s or sma_f < sma_s
            exit_take_profit = self.p.use_tp and (
                close >= self.entry_price * (1 + self.p.take_pct) or rsi > self.p.rsi_tp_level
            )
            exit_stop_loss = (
                self.stop_amount is not None and close <= self.entry_price - self.stop_amount
            )

            if exit_momentum_weak or exit_trend_reversal or exit_take_profit or exit_stop_loss:
                self.order = self.close()
            return

        leadership = close > sma_s and sma_f > sma_s
        momentum = self.macd.histo[0] > 0
        accel = self.macd_cross[0] > 0 or self.rsi_cross[0] > 0
        volume_ok = (not self.p.use_volume_filter) or (self.data.volume[0] > self.vol_sma[0])

        if leadership and momentum and accel and volume_ok:
            stop_amount_est = max(self.p.stop_pct * close, self.p.atr_mult * self.atr[0])
            capital = self.broker.getvalue()
            shares_by_risk = (capital * self.p.risk_pct) / stop_amount_est
            # both terms of the sizing min() normalized to a share count
            shares_by_cap = (self.p.size_cap_pct * capital) / close
            size = int(min(shares_by_risk, shares_by_cap))
            if size > 0:
                self.order = self.buy(size=size)
