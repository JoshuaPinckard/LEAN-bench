import backtrader as bt


class RangeControlStrategy(bt.Strategy):
    params = dict(
        vol_sma_period=20,
        consolidation_lookback=10,
        consolidation_range_pct=0.02,
        vol_multiplier=3.0,
        take_profit_pct=0.05,
        stop_loss_pct=0.02,
        position_pct=0.01,
    )

    def __init__(self):
        self.vol_sma = bt.indicators.SMA(self.data.volume, period=self.p.vol_sma_period)
        self.order = None
        self.entry_price = None
        self.range_upper = None
        self.range_lower = None

    def notify_order(self, order):
        if order.status == order.Completed:
            if not self.position:
                self.entry_price = None
                self.range_upper = None
                self.range_lower = None
            elif self.entry_price is None:
                self.entry_price = order.executed.price
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        if self.order:
            return

        lb = self.p.consolidation_lookback
        if len(self) <= lb:
            return

        price = self.data.close[0]

        highs = [self.data.high[-i] for i in range(1, lb + 1)]
        lows = [self.data.low[-i] for i in range(1, lb + 1)]
        upper = max(highs)
        lower = min(lows)
        range_width = upper - lower
        consolidating = range_width <= (self.p.consolidation_range_pct * price)
        vol_spike = self.data.volume[0] >= (self.p.vol_multiplier * self.vol_sma[0])

        if not self.position:
            if consolidating and vol_spike:
                size = (self.p.position_pct * self.broker.getvalue()) / price
                if price > upper:
                    self.range_upper = upper
                    self.range_lower = lower
                    self.order = self.buy(size=size)
                elif price < lower:
                    self.range_upper = upper
                    self.range_lower = lower
                    self.order = self.sell(size=size)
        else:
            if self.entry_price is None:
                return
            if self.position.size > 0:
                if self.data.high[0] >= self.entry_price * (1 + self.p.take_profit_pct):
                    self.order = self.close()
                elif self.data.low[0] <= self.entry_price * (1 - self.p.stop_loss_pct):
                    self.order = self.close()
                elif price < self.range_lower:
                    self.order = self.close()
            elif self.position.size < 0:
                if self.data.low[0] <= self.entry_price * (1 - self.p.take_profit_pct):
                    self.order = self.close()
                elif self.data.high[0] >= self.entry_price * (1 + self.p.stop_loss_pct):
                    self.order = self.close()
                elif price > self.range_upper:
                    self.order = self.close()
