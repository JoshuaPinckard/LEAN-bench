import math

import backtrader as bt


class ATRFilteredTrendFollowingStrategy(bt.Strategy):
    def __init__(self):
        self.atr = bt.indicators.ATR(self.data, period=14)
        self.sma_atr50 = bt.indicators.SMA(self.atr, period=50)
        self.sma_atr200 = bt.indicators.SMA(self.atr, period=200)
        self.ema50 = bt.indicators.EMA(self.data.close, period=50)
        self.macd = bt.indicators.MACD(
            self.data.close, period_me1=12, period_me2=26, period_signal=9
        )

        self.order = None
        self.stop_price = None
        self.take_profit = None
        self.direction = None

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return
        if order.status in (order.Completed, order.Canceled, order.Margin, order.Rejected):
            self.order = None

    def next(self):
        if self.order:
            return

        if not self.position:
            not_flat = self.sma_atr50[0] >= 0.75 * self.sma_atr200[0]
            if not not_flat:
                return

            entry_close = self.data.close[0]
            atr_val = self.atr[0]

            long_signal = (
                entry_close > self.ema50[0]
                and self.ema50[0] > self.ema50[-1]
                and self.macd.macd[0] > self.macd.signal[0]
            )
            short_signal = (
                entry_close < self.ema50[0]
                and self.ema50[0] < self.ema50[-1]
                and self.macd.macd[0] < self.macd.signal[0]
            )

            if long_signal:
                stop = min(self.data.low[0], entry_close) - 1.5 * atr_val
                stop_distance = entry_close - stop
                if stop_distance <= 0:
                    return
                size = int(math.floor((self.broker.getvalue() * 0.01) / stop_distance))
                if size <= 0:
                    return
                self.stop_price = stop
                self.take_profit = entry_close + 2 * atr_val
                self.direction = "long"
                self.order = self.buy(size=size)

            elif short_signal:
                stop = max(self.data.high[0], entry_close) + 1.5 * atr_val
                stop_distance = stop - entry_close
                if stop_distance <= 0:
                    return
                size = int(math.floor((self.broker.getvalue() * 0.01) / stop_distance))
                if size <= 0:
                    return
                self.stop_price = stop
                self.take_profit = entry_close - 2 * atr_val
                self.direction = "short"
                self.order = self.sell(size=size)

        else:
            close = self.data.close[0]
            atr_val = self.atr[0]

            if self.direction == "long":
                self.stop_price = max(self.stop_price, close - 1.5 * atr_val)

                stop_hit = close <= self.stop_price
                tp_hit = close >= self.take_profit
                macd_cross_down = (
                    self.macd.macd[-1] > self.macd.signal[-1]
                    and self.macd.macd[0] < self.macd.signal[0]
                )
                ema_stops_rising = self.ema50[0] <= self.ema50[-1]

                if stop_hit or tp_hit or macd_cross_down or ema_stops_rising:
                    self.order = self.close()
                    self.stop_price = None
                    self.take_profit = None
                    self.direction = None

            elif self.direction == "short":
                self.stop_price = min(self.stop_price, close + 1.5 * atr_val)

                stop_hit = close >= self.stop_price
                tp_hit = close <= self.take_profit
                macd_cross_up = (
                    self.macd.macd[-1] < self.macd.signal[-1]
                    and self.macd.macd[0] > self.macd.signal[0]
                )
                ema_stops_falling = self.ema50[0] >= self.ema50[-1]

                if stop_hit or tp_hit or macd_cross_up or ema_stops_falling:
                    self.order = self.close()
                    self.stop_price = None
                    self.take_profit = None
                    self.direction = None
