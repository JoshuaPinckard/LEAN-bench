import backtrader as bt
import math


class HedgeOverlayStrategy(bt.Strategy):
    params = (('rsi_period', 14),)

    def __init__(self):
        self.rsi = bt.indicators.RSI(self.data.close, period=self.p.rsi_period)
        self.order = None

    def next(self):
        if self.order:
            return

        if not self.position:
            if self.rsi[0] > 70:
                size = math.floor(0.10 * self.broker.getvalue() / self.data.close[0])
                if size > 0:
                    self.order = self.sell(size=size)
        elif self.position.size < 0 and self.rsi[0] < 50:
            self.order = self.close()

    def notify_order(self, order):
        if order.status in (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            self.order = None
