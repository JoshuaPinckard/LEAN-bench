import backtrader as bt


class PullbackFibEmaStrategy(bt.Strategy):
    params = dict(
        ema_period=20,
        fractal_bars=2,          # bars on each side used to confirm a swing fractal
        fib_levels=(0.236, 0.382, 0.5, 0.618),
        risk_pct=0.01,           # fraction of capital risked per trade
        stop_buffer_pct=0.001,   # buffer placed beyond the stop reference price
        stop_lookback=5,         # bars scanned for the "nearest local low"
        rr_target=9.0,           # minimum reward:risk required to take the setup
    )

    def __init__(self):
        self.ema = bt.indicators.EMA(self.data.close, period=self.p.ema_period)

        self.swing_low = None
        self.swing_low_bar = None
        self.swing_high = None
        self.swing_high_bar = None

        self.order = None
        self.entry_price = None
        self.stop_price = None
        self.tp1_price = None
        self.tp2_price = None
        self.tp1_done = False

    def _update_swings(self):
        n = self.p.fractal_bars
        if len(self) < 2 * n + 1:
            return

        center_high = self.data.high[-n]
        if all(center_high > self.data.high[-n + off] for off in range(-n, n + 1) if off != 0):
            self.swing_high = center_high
            self.swing_high_bar = len(self) - n

        center_low = self.data.low[-n]
        if all(center_low < self.data.low[-n + off] for off in range(-n, n + 1) if off != 0):
            self.swing_low = center_low
            self.swing_low_bar = len(self) - n

    def _fib_prices(self):
        diff = self.swing_high - self.swing_low
        return {r: self.swing_high - r * diff for r in self.p.fib_levels}

    def notify_order(self, order):
        if order.status in (order.Completed, order.Canceled, order.Margin, order.Rejected):
            self.order = None

    def next(self):
        self._update_swings()

        if self.order:
            return

        if self.position:
            self._manage_position()
        else:
            self._check_entry()

    def _check_entry(self):
        if self.swing_high is None or self.swing_low is None:
            return
        if self.swing_low_bar is None or self.swing_high_bar is None:
            return
        if self.swing_low_bar >= self.swing_high_bar:
            return  # need a completed low-to-high impulse leg to retrace from

        if not (self.data.close[0] > self.ema[0]):
            return  # uptrend filter

        bullish_reversal = (
            self.data.close[0] > self.data.open[0]
            and self.data.close[0] > self.data.close[-1]
        )
        if not bullish_reversal:
            return

        levels = self._fib_prices()
        low = self.data.low[0]
        close = self.data.close[0]

        touched_price = None
        touched_ratio = None
        for ratio, price in levels.items():
            if low <= price < close:
                if touched_ratio is None or ratio > touched_ratio:
                    touched_ratio = ratio
                    touched_price = price

        if touched_price is None:
            return

        recent_low = min(self.data.low[-i] for i in range(self.p.stop_lookback))
        stop_ref = min(recent_low, touched_price)
        stop_price = stop_ref * (1 - self.p.stop_buffer_pct)

        entry_price = close
        risk = entry_price - stop_price
        if risk <= 0:
            return

        ratios_sorted = sorted(self.p.fib_levels)
        idx = ratios_sorted.index(touched_ratio)

        if idx == 0:
            tp1 = self.swing_high
            tp2 = self.swing_low + 1.272 * (self.swing_high - self.swing_low)
        elif idx == 1:
            tp1 = levels[ratios_sorted[0]]
            tp2 = self.swing_high
        else:
            tp1 = levels[ratios_sorted[idx - 1]]
            tp2 = levels[ratios_sorted[idx - 2]]

        reward2 = tp2 - entry_price
        if reward2 / risk < self.p.rr_target:
            return

        cash = self.broker.get_cash()
        size = (cash * self.p.risk_pct) / risk
        if size <= 0:
            return

        self.entry_price = entry_price
        self.stop_price = stop_price
        self.tp1_price = tp1
        self.tp2_price = tp2
        self.tp1_done = False

        self.order = self.buy(size=size)

    def _manage_position(self):
        low = self.data.low[0]
        high = self.data.high[0]

        if not self.tp1_done:
            if low <= self.stop_price:
                self.order = self.close()
                return
            if high >= self.tp1_price:
                half = self.position.size / 2
                self.order = self.sell(size=half)
                self.stop_price = self.entry_price
                self.tp1_done = True
        else:
            if low <= self.stop_price:
                self.order = self.close()
                return
            if high >= self.tp2_price:
                self.order = self.close()
