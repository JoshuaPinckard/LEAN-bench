import statistics
import backtrader as bt


class VaROverlappingIntervalsStrategy(bt.Strategy):
    params = (
        ('vol_period', 250),
        ('short_vol_period', 20),
        ('confidence_level', 0.95),
        ('high_vol_threshold', 2.0),
    )

    def __init__(self):
        self.order = None
        self.pct_change = bt.indicators.PercentChange(self.data.close, period=1)

        self.exceedances = []
        self.filtered_bars = 0
        self.valid_bars = 0
        self.total_bars = 0

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        self.total_bars += 1

        if len(self.pct_change) < self.p.vol_period + 1:
            return

        returns_250 = [self.pct_change[-i] for i in range(self.p.vol_period)]
        returns_250 = [r for r in returns_250 if r is not None]

        if len(returns_250) < self.p.vol_period:
            return

        vol_250 = statistics.stdev(returns_250)

        returns_20 = returns_250[:self.p.short_vol_period]
        vol_20 = statistics.stdev(returns_20) if len(returns_20) > 1 else vol_250

        if vol_250 > 0 and (vol_20 / vol_250) > self.p.high_vol_threshold:
            self.filtered_bars += 1
            if self.position and not self.order:
                self.order = self.close()
            return

        self.valid_bars += 1

        z_score = 1.644853
        var_pct = z_score * vol_250

        current_return = self.pct_change[0]
        if current_return is None:
            return

        actual_loss = -current_return
        is_exceedance = actual_loss > var_pct

        if is_exceedance:
            dt = self.data.datetime.date(0)
            self.exceedances.append((len(self), dt, actual_loss, var_pct))

            if not self.position and not self.order:
                self.order = self.buy()
        else:
            if self.position and not self.order:
                self.order = self.close()

    def stop(self):
        num_exceedances = len(self.exceedances)
        expected_rate = 1.0 - self.p.confidence_level
        actual_rate = (num_exceedances / self.valid_bars) if self.valid_bars > 0 else 0.0

        clustering_counts = 0
        if num_exceedances > 1:
            for i in range(1, num_exceedances):
                gap = self.exceedances[i][0] - self.exceedances[i - 1][0]
                if gap <= 5:
                    clustering_counts += 1

        print("=== VaR Model Adequacy & Overlapping Intervals Report ===")
        print(f"Total Bars Evaluated: {self.total_bars}")
        print(f"Filtered High Volatility/Black Swan Bars: {self.filtered_bars}")
        print(f"Valid Analysis Bars: {self.valid_bars}")
        print(f"Total VaR Exceedances: {num_exceedances}")
        print(f"Expected Exceedance Rate: {expected_rate:.2%}")
        print(f"Actual Exceedance Rate: {actual_rate:.2%}")
        print(f"Clustered Exceedances (gap <= 5 bars): {clustering_counts}")

        if actual_rate > expected_rate * 1.5:
            conclusion = "Model UNDERESTIMATES risk (too many exceedances, likely due to data dependence/overlap)."
        elif actual_rate < expected_rate * 0.5:
            conclusion = "Model OVERESTIMATES risk (too few exceedances)."
        else:
            conclusion = "Model VaR estimation is ADEQUATE for the specified confidence level."

        if clustering_counts > 0:
            conclusion += " Significant clustering observed, indicating dependence in overlapping intervals."
        else:
            conclusion += " No significant temporal clustering detected."

        print(f"Conclusion: {conclusion}")
