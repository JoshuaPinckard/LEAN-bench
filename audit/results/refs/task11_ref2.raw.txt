import backtrader as bt


class IntradayCrossoverStrategy(bt.Strategy):
    params = (
        ("sl_pct", 0.01),
        ("tp_pct", 0.02),
        ("capital_pct", 0.99),
    )

    def __init__(self):
        self.order = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return
        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            self.order = None

    def next(self):
        if self.order:
            return

        curr_close = self.data.close[0]
        prev_close = self.data.close[-1]

        if self.position:
            entry_price = self.position.price
            if self.position.size > 0:
                sl_price = entry_price * (1.0 - self.p.sl_pct)
                tp_price = entry_price * (1.0 + self.p.tp_pct)

                if self.data.low[0] <= sl_price:
                    self.order = self.close()
                elif self.data.high[0] >= tp_price:
                    self.order = self.close()
                elif curr_close < prev_close:
                    self.order = self.close()

            elif self.position.size < 0:
                sl_price = entry_price * (1.0 + self.p.sl_pct)
                tp_price = entry_price * (1.0 - self.p.tp_pct)

                if self.data.high[0] >= sl_price:
                    self.order = self.close()
                elif self.data.low[0] <= tp_price:
                    self.order = self.close()
                elif curr_close > prev_close:
                    self.order = self.close()
        else:
            t = self.data.datetime.time(0)
            if 10 <= t.hour < 16:
                size = int((self.broker.getcash() * self.p.capital_pct) / curr_close)
                if size > 0:
                    if curr_close > prev_close:
                        self.order = self.buy(size=size)
                    elif curr_close < prev_close:
                        self.order = self.sell(size=size)
