import backtrader as bt


class VolatilityTrendStrategy(bt.Strategy):
    params = (
        ("atr_period", 10),
        ("ema_period", 20),
        ("signal_period", 20),
        ("band_mult", 2.0),
        ("exp_mult", 1.0),
        ("atr_mult", 3.0),
    )

    def __init__(self):
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.ema = bt.indicators.EMA(self.data.close, period=self.p.ema_period)
        self.signal = self.data.close - self.ema
        self.signal_mean = bt.indicators.SMA(
            self.signal, period=self.p.signal_period
        )
        self.signal_std = bt.indicators.StdDev(
            self.signal, period=self.p.signal_period
        )

        self.upper_band = self.signal_mean + self.p.band_mult * self.signal_std
        self.lower_band = self.signal_mean - self.p.band_mult * self.signal_std

        self.crossover_upper = bt.indicators.CrossOver(
            self.signal, self.upper_band
        )
        self.crossover_lower = bt.indicators.CrossOver(
            self.signal, self.lower_band
        )
        self.crossover_mean = bt.indicators.CrossOver(
            self.signal, self.signal_mean
        )

        self.order = None
        self.stop_order = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            if order == self.order:
                self.order = None
                if self.position.size > 0:
                    stop_price = (
                        order.executed.price - self.p.atr_mult * self.atr[0]
                    )
                    self.stop_order = self.sell(
                        exectype=bt.Order.Stop, price=stop_price
                    )
                elif self.position.size < 0:
                    stop_price = (
                        order.executed.price + self.p.atr_mult * self.atr[0]
                    )
                    self.stop_order = self.buy(
                        exectype=bt.Order.Stop, price=stop_price
                    )
            elif order == self.stop_order:
                self.stop_order = None

        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            if order == self.order:
                self.order = None
            if order == self.stop_order:
                self.stop_order = None

    def next(self):
        if self.order:
            return

        expansion = (
            abs(self.signal[0] - self.signal[-1])
            > self.p.exp_mult * self.signal_std[0]
        )

        if not self.position:
            if self.crossover_upper[0] > 0 and expansion:
                self.order = self.buy()
            elif self.crossover_lower[0] < 0 and expansion:
                self.order = self.sell()
        else:
            if self.position.size > 0:
                if self.crossover_mean[0] < 0:
                    if self.stop_order:
                        self.cancel(self.stop_order)
                    self.order = self.close()
            elif self.position.size < 0:
                if self.crossover_mean[0] > 0:
                    if self.stop_order:
                        self.cancel(self.stop_order)
                    self.order = self.close()
