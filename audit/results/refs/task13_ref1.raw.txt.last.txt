import backtrader as bt


class PyramidingInTrend(bt.Strategy):
    params = (
        ('ema_period', 50),
        ('rsi_period', 14),
    )

    def __init__(self):
        self.ema = bt.indicators.ExponentialMovingAverage(
            self.data.close, period=self.p.ema_period
        )
        self.rsi = bt.indicators.RSI(self.data.close, period=self.p.rsi_period)

        self.order = None
        self.initial_capital = None
        self.fixed_volume = None

        self.entries_completed = 0
        self.rsi_below_50_seen = False
        self.rsi_70_seen = False
        self.rsi_above_60_seen = False
        self.break_even_taken = False
        self.take_profit_taken = False

    def start(self):
        self.initial_capital = self.broker.getcash()

    def notify_order(self, order):
        if order.status in (
            order.Completed, order.Canceled, order.Margin, order.Rejected
        ):
            if order.status == order.Completed and order.isbuy():
                self.entries_completed += 1

            if order.status == order.Completed and order.issell():
                if not self.position.size:
                    self.entries_completed = 0
                    self.rsi_below_50_seen = False
                    self.rsi_70_seen = False
                    self.rsi_above_60_seen = False
                    self.break_even_taken = False
                    self.take_profit_taken = False

            self.order = None

    def next(self):
        close = self.data.close[0]
        rsi = self.rsi[0]

        if not self.position.size:
            if rsi < 50:
                self.rsi_below_50_seen = True

            if self.order is None:
                if (
                    self.rsi_below_50_seen
                    and rsi > 50
                    and close > self.ema[0]
                ):
                    self.fixed_volume = (self.initial_capital * 0.01) / close
                    self.order = self.buy(
                        size=self.fixed_volume, exectype=bt.Order.Close
                    )
                    self.rsi_below_50_seen = False
            return

        if rsi > 60:
            self.rsi_above_60_seen = True

        if self.entries_completed >= 1 and rsi >= 70:
            self.rsi_70_seen = True

        if self.order is not None:
            return

        if close < self.ema[0]:
            self.order = self.close(exectype=bt.Order.Close)
            return

        if self.rsi_above_60_seen and rsi < 40:
            self.order = self.close(exectype=bt.Order.Close)
            return

        total_profit = (close - self.position.price) * self.position.size
        total_entry_value = self.position.price * self.position.size

        if not self.break_even_taken and total_profit >= 0:
            self.order = self.close(
                size=self.position.size * 0.25, exectype=bt.Order.Close
            )
            self.break_even_taken = True
            return

        if (
            not self.take_profit_taken
            and total_entry_value > 0
            and total_profit >= total_entry_value * 0.05
        ):
            self.order = self.close(
                size=self.position.size * 0.50, exectype=bt.Order.Close
            )
            self.take_profit_taken = True
            return

        if (
            self.entries_completed < 4
            and self.rsi_70_seen
            and rsi <= 60
            and close > self.ema[0]
        ):
            self.order = self.buy(
                size=self.fixed_volume, exectype=bt.Order.Close
            )
            self.rsi_70_seen = False
