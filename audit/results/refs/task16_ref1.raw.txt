import backtrader as bt


class PartialTakeProfitTrailingStop(bt.Strategy):
    params = (
        ("risk_fraction", 0.01),
        ("initial_stop_offset", 0.01),
        ("partial_fraction", 0.50),
        ("trailing_offset", 0.0075),
    )

    def __init__(self):
        self.order = None
        self.initial_stop_order = None
        self.partial_order = None
        self.trailing_stop_order = None
        self.entry_price = None
        self.risk_size = None
        self.partial_taken = False
        self.trailing_active = False
        self.trailing_extreme = None

    def notify_order(self, order):
        if order.status in (order.Canceled, order.Margin, order.Rejected):
            if order == self.order:
                self.order = None
            return

        if order.status != order.Completed:
            return

        if order == self.order:
            self.order = None
            if order.isbuy():
                self.entry_price = order.executed.price
                self.risk_size = self.entry_price * self.p.initial_stop_offset
                initial_stop_price = self.entry_price - self.risk_size
                self.initial_stop_order = self.sell(
                    exectype=bt.Order.Stop,
                    price=initial_stop_price,
                    size=abs(self.position.size),
                )
                partial_price = self.entry_price + 1.5 * self.risk_size
                partial_size = abs(self.position.size) * self.p.partial_fraction
                self.partial_order = self.sell(
                    exectype=bt.Order.Limit,
                    price=partial_price,
                    size=partial_size,
                )
            return

        if order == self.partial_order:
            self.partial_order = None
            self.partial_taken = True
            self.trailing_active = True
            self.trailing_extreme = order.executed.price

            if self.initial_stop_order is not None:
                self.cancel(self.initial_stop_order)
                self.initial_stop_order = None

            trailing_price = self.trailing_extreme * (1.0 - self.p.trailing_offset)
            self.trailing_stop_order = self.sell(
                exectype=bt.Order.Stop,
                price=trailing_price,
                size=abs(self.position.size),
            )
            return

        if order == self.initial_stop_order:
            self.initial_stop_order = None
            if self.partial_order is not None:
                self.cancel(self.partial_order)
                self.partial_order = None
            self._reset_trade_state()
            return

        if order == self.trailing_stop_order:
            self.trailing_stop_order = None
            self._reset_trade_state()

    def next(self):
        if not self.position and self.order is None:
            entry_price = self.data.open[0]
            risk_size = entry_price * self.p.initial_stop_offset
            risk_capital = self.broker.getvalue() * self.p.risk_fraction
            size = risk_capital / risk_size if risk_size else 0
            if size > 0:
                self.order = self.buy(size=size)
            return

        if self.position.size > 0 and self.trailing_active:
            if self.data.high[0] > self.trailing_extreme:
                self.trailing_extreme = self.data.high[0]
                new_stop = self.trailing_extreme * (1.0 - self.p.trailing_offset)
                if self.trailing_stop_order is not None:
                    self.cancel(self.trailing_stop_order)
                self.trailing_stop_order = self.sell(
                    exectype=bt.Order.Stop,
                    price=new_stop,
                    size=abs(self.position.size),
                )

    def _reset_trade_state(self):
        self.entry_price = None
        self.risk_size = None
        self.partial_taken = False
        self.trailing_active = False
        self.trailing_extreme = None
