import backtrader as bt


class ShortPutSpreadEmulator(bt.Strategy):
    def __init__(self):
        self.broker.set_coo(True)
        self.broker.set_coc(True)

        self.bb = bt.indicators.BollingerBands(
            self.data.close, period=20, devfactor=2.0
        )
        self.bw = (self.bb.top - self.bb.bot) / self.bb.mid
        self.bw_min = bt.indicators.Lowest(self.bw, period=100)
        self.bw_max = bt.indicators.Highest(self.bw, period=100)

        self.order = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            if order.isbuy():
                self.order = self.close(exectype=bt.Order.Close)
            else:
                self.order = None
        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        if self.order or self.position:
            return

        bw_prev = self.bw[-1]
        bw_min_prev = self.bw_min[-1]
        bw_max_prev = self.bw_max[-1]

        range_prev = bw_max_prev - bw_min_prev
        if range_prev > 0:
            iv_rank = (bw_prev - bw_min_prev) / range_prev
        else:
            iv_rank = 0.0

        cond_vol = iv_rank <= 0.20

        open_t = self.data.open[0]
        close_prev = self.data.close[-1]
        gap = (open_t - close_prev) / close_prev

        cond_gap = gap >= -0.004

        if cond_vol and cond_gap:
            self.order = self.buy()
