import math
import backtrader as bt


class MeanReversionZScore(bt.Strategy):
    params = (
        ("bb_period", 100),
        ("bb_devfactor", 2.0),
        ("sma_period", 100),
        ("atr_period", 14),
        ("k_atr", 3.0),
        ("risk_pct", 0.01),
    )

    def __init__(self):
        self.order = None
        self.bb = bt.indicators.BollingerBands(
            self.data.close,
            period=self.p.bb_period,
            devfactor=self.p.bb_devfactor,
        )
        self.sma = bt.indicators.SMA(self.data.close, period=self.p.sma_period)
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.crossover = bt.indicators.CrossOver(self.data.close, self.sma)

        self.entry_z = None
        self.entry_price = None
        self.entry_atr = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            if self.position:
                self.entry_price = order.executed.price
            else:
                self.entry_z = None
                self.entry_price = None
                self.entry_atr = None
        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            if not self.position:
                self.entry_z = None
                self.entry_price = None
                self.entry_atr = None

        self.order = None

    def next(self):
        if self.order:
            return

        close = self.data.close[0]
        mid = self.sma[0]
        top = self.bb.top[0]

        half_width = (top - mid) / 2.0
        if half_width == 0:
            return

        z = (close - mid) / half_width

        if not self.position:
            atr_val = self.atr[0]
            if atr_val <= 0:
                return

            size = int((self.broker.getvalue() * self.p.risk_pct) / atr_val)
            if size <= 0:
                return

            if z <= -2.0:
                self.entry_z = z
                self.entry_price = close
                self.entry_atr = atr_val
                self.order = self.buy(size=size)
            elif z >= 2.0:
                self.entry_z = z
                self.entry_price = close
                self.entry_atr = atr_val
                self.order = self.sell(size=size)
        else:
            if self.entry_z is None:
                return

            half_life_met = abs(z) <= 0.5 * abs(self.entry_z)
            entry_p = (
                self.entry_price if self.entry_price is not None else close
            )
            entry_a = (
                self.entry_atr if self.entry_atr is not None else self.atr[0]
            )

            if self.position.size > 0:
                z_cross_up = self.crossover[0] > 0
                z_stop = z <= -3.0
                price_stop = close <= (entry_p - self.p.k_atr * entry_a)

                if z_cross_up or half_life_met or z_stop or price_stop:
                    self.order = self.close()

            elif self.position.size < 0:
                z_cross_dn = self.crossover[0] < 0
                z_stop = z >= 3.0
                price_stop = close >= (entry_p + self.p.k_atr * entry_a)

                if z_cross_dn or half_life_met or z_stop or price_stop:
                    self.order = self.close()
