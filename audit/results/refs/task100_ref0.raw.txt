import backtrader as bt


class DcaRsiStrategy(bt.Strategy):
    params = dict(
        rsi_period=14,
        rsi_oversold=30,
        rsi_overbought=70,
        rsi_exit=50,
        dca_pct=0.017,
        size_pct=0.01,
    )

    def __init__(self):
        self.rsi = bt.indicators.RSI(self.data, period=self.p.rsi_period)
        self.order = None
        self.pending_action = None
        self.primary_price = None
        self.averaged = False

    def _order_size(self):
        return (self.p.size_pct * self.broker.startingcash) / self.data.close[0]

    def next(self):
        if self.order:
            return

        if not self.position:
            if self.rsi[0] < self.p.rsi_oversold:
                self.pending_action = 'primary'
                self.order = self.buy(size=self._order_size())
            elif self.rsi[0] > self.p.rsi_overbought:
                self.pending_action = 'primary'
                self.order = self.sell(size=self._order_size())
            return

        if self.position.size > 0:
            if self.rsi[0] > self.p.rsi_exit:
                self.pending_action = 'exit'
                self.order = self.close()
            elif not self.averaged and self.data.close[0] <= self.primary_price * (1 - self.p.dca_pct):
                self.pending_action = 'averaging'
                self.order = self.buy(size=self._order_size())
        elif self.position.size < 0:
            if self.rsi[0] < self.p.rsi_exit:
                self.pending_action = 'exit'
                self.order = self.close()
            elif not self.averaged and self.data.close[0] >= self.primary_price * (1 + self.p.dca_pct):
                self.pending_action = 'averaging'
                self.order = self.sell(size=self._order_size())

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status == order.Completed:
            if self.pending_action == 'primary':
                self.primary_price = order.executed.price
                self.averaged = False
            elif self.pending_action == 'averaging':
                self.averaged = True
            elif self.pending_action == 'exit':
                self.primary_price = None
                self.averaged = False

        self.order = None
        self.pending_action = None
