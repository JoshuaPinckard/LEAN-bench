import backtrader as bt


class HeuristicStrategy(bt.Strategy):
    params = (
        ('sma_fast_period', 20),
        ('sma_slow_period', 50),
        ('ema_fast_period', 20),
        ('ema_slow_period', 50),
        ('rsi_period', 14),
        ('macd_fast', 12),
        ('macd_slow', 26),
        ('macd_signal', 9),
        ('trend_period', 200),
        ('atr_period', 14),
        ('atr_low_pct', 0.005),
        ('atr_high_pct', 0.05),
        ('take_profit_pct', 0.02),
        ('stop_loss_pct', 0.01),
        ('position_size_pct', 0.02),
        ('use_trend_filter', True),
        ('use_vol_filter', True),
    )

    def __init__(self):
        self.sma_fast = bt.indicators.SMA(self.data.close, period=self.p.sma_fast_period)
        self.sma_slow = bt.indicators.SMA(self.data.close, period=self.p.sma_slow_period)
        self.ema_fast = bt.indicators.EMA(self.data.close, period=self.p.ema_fast_period)
        self.ema_slow = bt.indicators.EMA(self.data.close, period=self.p.ema_slow_period)
        self.rsi = bt.indicators.RSI(self.data.close, period=self.p.rsi_period)
        self.macd = bt.indicators.MACD(
            self.data.close,
            period_me1=self.p.macd_fast,
            period_me2=self.p.macd_slow,
            period_signal=self.p.macd_signal,
        )
        self.sma_trend = bt.indicators.SMA(self.data.close, period=self.p.trend_period)
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.crossover = bt.indicators.CrossOver(self.sma_fast, self.sma_slow)

        self.order = None
        self.entry_price = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return
        if order.status == order.Completed:
            self.entry_price = order.executed.price if self.position else None
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        if self.order:
            return

        price = self.data.close[0]

        if self.position:
            if self.position.size > 0:
                if self.crossover[0] < 0:
                    self.order = self.close()
                    return
                if self.entry_price is not None:
                    if price >= self.entry_price * (1 + self.p.take_profit_pct):
                        self.order = self.close()
                        return
                    if price <= self.entry_price * (1 - self.p.stop_loss_pct):
                        self.order = self.close()
                        return
            elif self.position.size < 0:
                if self.crossover[0] > 0:
                    self.order = self.close()
                    return
                if self.entry_price is not None:
                    if price <= self.entry_price * (1 - self.p.take_profit_pct):
                        self.order = self.close()
                        return
                    if price >= self.entry_price * (1 + self.p.stop_loss_pct):
                        self.order = self.close()
                        return
            return

        if self.crossover[0] > 0:
            if self.p.use_trend_filter and not (price > self.sma_trend[0]):
                return
            if self.p.use_vol_filter:
                atr_pct = self.atr[0] / price
                if not (self.p.atr_low_pct <= atr_pct <= self.p.atr_high_pct):
                    return
            size = int((self.p.position_size_pct * self.broker.getcash()) / price)
            if size > 0:
                self.order = self.buy(size=size)
        elif self.crossover[0] < 0:
            if self.p.use_trend_filter and not (price < self.sma_trend[0]):
                return
            if self.p.use_vol_filter:
                atr_pct = self.atr[0] / price
                if not (self.p.atr_low_pct <= atr_pct <= self.p.atr_high_pct):
                    return
            size = int((self.p.position_size_pct * self.broker.getcash()) / price)
            if size > 0:
                self.order = self.sell(size=size)
