import backtrader as bt


class SyntheticWheelStrategy(bt.Strategy):

    def __init__(self):
        self.bb = bt.indicators.BollingerBands(self.data, period=20, devfactor=2)
        self.atr = bt.indicators.ATR(self.data, period=14)
        self.ema50 = bt.indicators.EMA(self.data, period=50)
        self.ema200 = bt.indicators.EMA(self.data, period=200)
        self.rsi = bt.indicators.RSI(self.data, period=14)

        self.buy_order = None
        self.sell_order = None
        self.close_order = None

        self.tranche_count = 0
        self.position_cost = 0.0
        self.avg_price = 0.0

        self.sell_placed_bar = None
        self.next_entry_bar = 0

        self.realized_profit_total = 0.0
        self.cycle_records = []
        self.current_cycle = None

    def _tranche_size(self, buy_lim, portfolio_value):
        if buy_lim <= 0:
            return 0
        size = int((0.03 * portfolio_value) // buy_lim)
        if size <= 0:
            return 0
        if self.position_cost + buy_lim * size > 0.09 * portfolio_value:
            return 0
        return size

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return

        if order.status == order.Completed:
            dt = self.data.datetime.date(0)
            if order.isbuy():
                fill_price = order.executed.price
                fill_size = order.executed.size
                self.position_cost += fill_price * fill_size
                self.tranche_count += 1
                self.avg_price = self.position_cost / self.position.size
                buy_record = {'date': dt, 'price': fill_price, 'size': fill_size}
                if self.current_cycle is None:
                    self.current_cycle = {
                        'buys': [buy_record],
                        'sell_date': None,
                        'sell_price': None,
                        'realized_profit': 0.0,
                        'tranches': self.tranche_count,
                        'avg_price': self.avg_price,
                        'exit_reason': None,
                    }
                else:
                    self.current_cycle['buys'].append(buy_record)
                    self.current_cycle['tranches'] = self.tranche_count
                    self.current_cycle['avg_price'] = self.avg_price
                if order is self.buy_order:
                    self.buy_order = None
            else:
                fill_price = order.executed.price
                fill_size = abs(order.executed.size)
                realized = (fill_price - self.avg_price) * fill_size
                self.realized_profit_total += realized
                if self.current_cycle is not None:
                    self.current_cycle['sell_date'] = dt
                    self.current_cycle['sell_price'] = fill_price
                    self.current_cycle['realized_profit'] = realized
                    self.current_cycle['exit_reason'] = (
                        'protective_stop' if order is self.close_order else 'take_profit'
                    )
                    self.cycle_records.append(self.current_cycle)
                    self.current_cycle = None
                if order is self.sell_order:
                    self.sell_order = None
                if order is self.close_order:
                    self.close_order = None
                self.position_cost = 0.0
                self.avg_price = 0.0
                self.tranche_count = 0

        elif order.status in (order.Canceled, order.Margin, order.Rejected):
            # identity check: our own reference may already point at a newer
            # replacement order placed the same bar the cancel was requested
            if order is self.buy_order:
                self.buy_order = None
            if order is self.sell_order:
                self.sell_order = None
            if order is self.close_order:
                self.close_order = None

    def next(self):
        close = self.data.close[0]
        upper = self.bb.top[0]
        lower = self.bb.bot[0]
        mid = self.bb.mid[0]

        price_filter = 1.0 <= close <= 15.0
        vol_bb = mid != 0 and ((upper - lower) / mid) > 0.06
        vol_atr = close != 0 and (self.atr[0] / close) > 0.03
        high_vol = vol_bb or vol_atr
        trend_ok = self.ema50[0] >= self.ema200[0]
        filters_ok = price_filter and high_vol and trend_ok

        position_open = self.position.size > 0
        portfolio_value = self.broker.getvalue()

        if position_open:
            catastrophic = (close < self.ema200[0] * 0.9) or \
                            (close <= self.avg_price - 4 * self.atr[0])
            if catastrophic:
                if self.close_order is None:
                    if self.sell_order is not None:
                        self.cancel(self.sell_order)
                        self.sell_order = None
                    if self.buy_order is not None:
                        self.cancel(self.buy_order)
                        self.buy_order = None
                    self.close_order = self.close()
                return

        if position_open and self.sell_order is not None and self.sell_placed_bar is not None:
            if (len(self) - self.sell_placed_bar) >= 20:
                self.cancel(self.sell_order)
                self.sell_order = None

        if (position_open and self.tranche_count < 3 and self.buy_order is None
                and close <= self.avg_price - 1.5 * self.atr[0] and self.rsi[0] < 35):
            buy_lim = min(lower, close * 0.93)
            size = self._tranche_size(buy_lim, portfolio_value)
            if size > 0:
                if self.sell_order is not None:
                    self.cancel(self.sell_order)
                    self.sell_order = None
                self.buy_order = self.buy(exectype=bt.Order.Limit, price=buy_lim, size=size)

        if position_open and self.sell_order is None and self.buy_order is None:
            sell_lim = max(upper, self.avg_price * 1.07)
            self.sell_order = self.sell(exectype=bt.Order.Limit, price=sell_lim, size=self.position.size)
            self.sell_placed_bar = len(self)

        if (not position_open and self.buy_order is None and filters_ok
                and len(self) >= self.next_entry_bar):
            buy_lim = min(lower, close * 0.93)
            size = self._tranche_size(buy_lim, portfolio_value)
            if size > 0:
                self.buy_order = self.buy(exectype=bt.Order.Limit, price=buy_lim, size=size)
                self.next_entry_bar = len(self) + 20
