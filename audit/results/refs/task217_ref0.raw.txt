import backtrader as bt


class MomentumVolumeEmaStrategy(bt.Strategy):
    def __init__(self):
        self.ema9 = bt.indicators.EMA(self.data.close, period=9)
        self.ema21 = bt.indicators.EMA(self.data.close, period=21)
        self.ema50 = bt.indicators.EMA(self.data.close, period=50)
        self.ema200 = bt.indicators.EMA(self.data.close, period=200)
        self.bb = bt.indicators.BollingerBands(self.data.close, period=20, devfactor=2)
        self.macd = bt.indicators.MACD(self.data.close, period_me1=12, period_me2=26, period_signal=9)
        self.atr = bt.indicators.ATR(self.data, period=14)
        self.vol_sma = bt.indicators.SMA(self.data.volume, period=20)

        self.cross_bb_upper = bt.indicators.CrossOver(self.data.close, self.bb.top)
        self.cross_macd = bt.indicators.CrossOver(self.macd.macd, self.macd.signal)
        self.cross_close_ema9 = bt.indicators.CrossOver(self.data.close, self.ema9)
        self.cross_ema21_close = bt.indicators.CrossOver(self.ema21, self.data.close)
        self.cross_signal_macd = bt.indicators.CrossOver(self.macd.signal, self.macd.macd)

        self.order = None
        self.entry_price = None
        self.stop_price = None
        self.tp_price = None
        self.breakeven_triggered = False

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return
        if order.status in (order.Completed, order.Canceled, order.Margin, order.Rejected):
            self.order = None
            if order.status == order.Completed and order.issell():
                self.entry_price = None
                self.stop_price = None
                self.tp_price = None
                self.breakeven_triggered = False

    def next(self):
        if self.order:
            return
        if not self.position:
            self._try_enter()
        else:
            self._manage_position()

    def _try_enter(self):
        close = self.data.close[0]

        price_ok = 0.75 <= close <= 10.00
        rvol = self.data.volume[0] / self.vol_sma[0]
        rvol_ok = rvol >= 1.5
        trend_ok = (close > self.ema9[0] > self.ema21[0] > self.ema50[0]) and (close > self.ema200[0])
        if not (price_ok and rvol_ok and trend_ok):
            return

        macd_cross_recent = (
            self.cross_macd[0] > 0 or self.cross_macd[-1] > 0 or self.cross_macd[-2] > 0
        )
        scenario_a = (self.cross_bb_upper[0] > 0) and macd_cross_recent

        touch_recent = any(
            (self.data.low[-i] <= self.ema21[-i]) or (self.data.close[-i] <= self.bb.mid[-i])
            for i in range(3)
        )
        near_upper = (self.bb.top[0] - close) <= 0.5 * self.atr[0]
        scenario_b = touch_recent and (self.cross_close_ema9[0] > 0) and near_upper

        if not (scenario_a or scenario_b):
            return

        entry = close
        sl = min(self.ema21[0], entry - max(0.02, self.atr[0]))
        tp_offset = max(0.10, min(self.atr[0], 0.20))
        tp = entry + tp_offset

        risk_per_share = entry - sl
        capital = self.broker.getvalue()
        risk_amount = 0.01 * capital
        size = int(risk_amount / risk_per_share)
        if size < 1:
            return

        self.entry_price = entry
        self.stop_price = sl
        self.tp_price = tp
        self.breakeven_triggered = False
        self.order = self.buy(size=size)

    def _manage_position(self):
        close = self.data.close[0]

        if not self.breakeven_triggered and (close - self.entry_price) >= 0.10:
            self.breakeven_triggered = True

        if self.breakeven_triggered:
            candidate = max(self.entry_price, self.ema9[0])
            if candidate > self.stop_price:
                self.stop_price = candidate

        if close <= self.stop_price:
            self.order = self.close()
            return
        if close >= self.tp_price:
            self.order = self.close()
            return
        if self.cross_ema21_close[0] > 0 or self.cross_signal_macd[0] > 0:
            self.order = self.close()
            return
