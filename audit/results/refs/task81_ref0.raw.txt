import backtrader as bt
import collections


class PredictiveImbalanceStrategy(bt.Strategy):
    params = (
        ('event_driven', False),      # False = continuous model (default), True = event-driven model
        ('cont_threshold', 0.2),
        ('event_threshold', 0.5),
        ('event_oi_threshold', 0.6),
        ('atr_period', 14),
        ('atr_pct_filter', 0.001),    # 0.1% of price
        ('trade_imb_period', 20),
        ('vol_sma_period', 20),
        ('session_start_hour', 10),
        ('session_end_hour', 16),
        ('w_oi', 0.5),
        ('w_ti', 0.3),
        ('w_vol', 0.1),
        ('w_atr', 0.1),
    )

    def __init__(self):
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.atr_sma = bt.indicators.SMA(self.atr, period=self.p.atr_period)
        self.vol_sma = bt.indicators.SMA(self.data.volume, period=self.p.vol_sma_period)

        self.trade_signed_vol = collections.deque(maxlen=self.p.trade_imb_period)

        self.setsizer(bt.sizers.PercentSizer(percents=100))

        self.order = None

    def notify_order(self, order):
        if order.status in (order.Completed, order.Canceled, order.Margin, order.Rejected):
            self.order = None

    def next(self):
        high = self.data.high[0]
        low = self.data.low[0]
        close = self.data.close[0]
        open_ = self.data.open[0]
        volume = self.data.volume[0]

        rng = high - low
        order_imbalance = (2.0 * close - high - low) / rng if rng != 0 else 0.0

        signed_vol = volume if close >= open_ else -volume
        self.trade_signed_vol.append(signed_vol)
        total_vol = sum(abs(v) for v in self.trade_signed_vol)
        trade_imbalance = (sum(self.trade_signed_vol) / total_vol) if total_vol != 0 else 0.0

        vol_sma = self.vol_sma[0]
        vol_signal = max(-1.0, min(1.0, (volume - vol_sma) / vol_sma)) if vol_sma != 0 else 0.0

        atr_sma = self.atr_sma[0]
        if atr_sma != 0:
            direction = 1.0 if close >= open_ else -1.0
            atr_signal = direction * max(-1.0, min(1.0, (self.atr[0] - atr_sma) / atr_sma))
        else:
            atr_signal = 0.0

        raw_forecast = (self.p.w_oi * order_imbalance +
                         self.p.w_ti * trade_imbalance +
                         self.p.w_vol * vol_signal +
                         self.p.w_atr * atr_signal)
        forecast = max(-1.0, min(1.0, raw_forecast))

        atr_ok = self.atr[0] >= self.p.atr_pct_filter * close

        bar_time = self.data.datetime.time(0)
        in_session = self.p.session_start_hour <= bar_time.hour < self.p.session_end_hour

        if self.p.event_driven:
            active = abs(order_imbalance) > self.p.event_oi_threshold
            buy_th = self.p.event_threshold
            sell_th = -self.p.event_threshold
        else:
            active = True
            buy_th = self.p.cont_threshold
            sell_th = -self.p.cont_threshold

        pos = self.position.size

        if self.order is None:
            if pos > 0 and forecast < 0:
                self.order = self.close()
            elif pos < 0 and forecast > 0:
                self.order = self.close()
            elif pos == 0 and active and atr_ok and in_session:
                if forecast > buy_th:
                    self.order = self.buy()
                elif forecast < sell_th:
                    self.order = self.sell()
