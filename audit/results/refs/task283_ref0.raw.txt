```python
import backtrader as bt
import math


class DistributedSellStrategy(bt.Strategy):
    def __init__(self):
        self.order = None
        self.entered = False
        self.entry_price = None
        self.initial_investment = None
        self.cost_basis = None
        self.threshold_n = 1

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        if self.order:
            return

        if not self.entered:
            self.entry_price = self.data.close[0]
            self.initial_investment = 400 * self.entry_price
            self.cost_basis = self.initial_investment
            self.order = self.buy(size=400)
            self.entered = True
            return

        if self.position.size > 0:
            position_value = self.position.size * self.data.close[0]
            threshold = self.initial_investment * (2 ** self.threshold_n)
            if position_value >= threshold:
                shares_to_sell = math.floor(self.position.size * 0.5)
                if shares_to_sell > 0:
                    self.order = self.sell(size=shares_to_sell)
                    self.cost_basis = self.cost_basis * 0.5
                    self.threshold_n += 1
```
