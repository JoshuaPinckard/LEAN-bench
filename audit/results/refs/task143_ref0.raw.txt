import backtrader as bt


class _SuperTrend(bt.Indicator):
    lines = ('supertrend', 'trend', 'final_ub', 'final_lb', 'up', 'down')
    params = (
        ('period', 32),
        ('multiplier', 4.35),
    )

    plotinfo = dict(subplot=False)
    plotlines = dict(
        supertrend=dict(_plotskip=True),
        trend=dict(_plotskip=True),
        final_ub=dict(_plotskip=True),
        final_lb=dict(_plotskip=True),
        up=dict(color='green', ls='-'),
        down=dict(color='red', ls='-'),
    )

    def __init__(self):
        self.atr = bt.indicators.ATR(self.data, period=self.p.period)
        self.hl2 = (self.data.high + self.data.low) / 2.0
        self._seeded = False

    def next(self):
        atr = self.atr[0]
        hl2 = self.hl2[0]
        basic_ub = hl2 + self.p.multiplier * atr
        basic_lb = hl2 - self.p.multiplier * atr
        close = self.data.close[0]

        if not self._seeded:
            final_ub = basic_ub
            final_lb = basic_lb
            trend = -1
            st = final_ub
            self._seeded = True
        else:
            prev_final_ub = self.l.final_ub[-1]
            prev_final_lb = self.l.final_lb[-1]
            prev_close = self.data.close[-1]
            prev_st = self.l.supertrend[-1]

            if basic_ub < prev_final_ub or prev_close > prev_final_ub:
                final_ub = basic_ub
            else:
                final_ub = prev_final_ub

            if basic_lb > prev_final_lb or prev_close < prev_final_lb:
                final_lb = basic_lb
            else:
                final_lb = prev_final_lb

            if prev_st == prev_final_ub:
                if close <= final_ub:
                    trend = -1
                    st = final_ub
                else:
                    trend = 1
                    st = final_lb
            else:
                if close >= final_lb:
                    trend = 1
                    st = final_lb
                else:
                    trend = -1
                    st = final_ub

        self.l.final_ub[0] = final_ub
        self.l.final_lb[0] = final_lb
        self.l.supertrend[0] = st
        self.l.trend[0] = trend

        if trend == 1:
            self.l.up[0] = st
            self.l.down[0] = float('nan')
        else:
            self.l.down[0] = st
            self.l.up[0] = float('nan')


class SupertrendLongStrategy(bt.Strategy):
    params = (
        ('atr_period', 32),
        ('atr_multiplier', 4.35),
        ('sma_period', 200),
    )

    def __init__(self):
        self.supertrend = _SuperTrend(
            self.data,
            period=self.p.atr_period,
            multiplier=self.p.atr_multiplier,
        )
        self.sma = bt.indicators.SMA(self.data.close, period=self.p.sma_period)
        self.order = None

    def notify_order(self, order):
        if order.status in (order.Completed, order.Canceled, order.Margin, order.Rejected):
            self.order = None

    def next(self):
        if self.order:
            return

        trend_now = self.supertrend.trend[0]
        trend_prev = self.supertrend.trend[-1]

        if not self.position:
            if trend_prev == -1 and trend_now == 1:
                if self.data.close[0] > self.sma[0]:
                    self.order = self.order_target_percent(target=1.0)
        else:
            if trend_prev == 1 and trend_now == -1:
                self.order = self.close()
