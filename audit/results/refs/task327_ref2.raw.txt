import backtrader as bt


class SpecStrategy(bt.Strategy):
    def __init__(self):
        self.order = None
        body = abs(self.data.close - self.data.open)
        self.sma_body = bt.indicators.SMA(body, period=20)
        self.sma_vol = bt.indicators.SMA(self.data.volume, period=20)

    def notify_order(self, order):
        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            self.order = None

    def next(self):
        if self.order:
            return

        dt = self.data.datetime.time(0)
        time_ok = (dt.hour >= 7) and (
            dt.hour < 17 or (dt.hour == 17 and dt.minute == 0)
        )

        close = self.data.close[0]
        open_price = self.data.open[0]
        body = abs(close - open_price)
        vol = self.data.volume[0]

        sma_body_val = self.sma_body[0]
        sma_vol_val = self.sma_vol[0]

        valid_long_signal = (
            (close > open_price)
            and (body >= 1.5 * sma_body_val)
            and (vol >= 1.5 * sma_vol_val)
        )

        valid_short_signal = (
            (close < open_price)
            and (body >= 1.5 * sma_body_val)
            and (vol >= 1.5 * sma_vol_val)
        )

        if self.position.size > 0:
            if valid_short_signal:
                self.order = self.close()
        elif self.position.size < 0:
            if valid_long_signal:
                self.order = self.close()
        else:
            if time_ok:
                if valid_long_signal:
                    self.order = self.buy()
                elif valid_short_signal:
                    self.order = self.sell()
