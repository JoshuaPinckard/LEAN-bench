import backtrader as bt


class UniversalPortfolioStrategy(bt.Strategy):
    params = dict(
        window=252,
        fractions=tuple(round(i * 0.1, 1) for i in range(11)),
    )

    def __init__(self):
        self.order = None

    def notify_order(self, order):
        if order.status in (order.Completed, order.Canceled, order.Margin, order.Rejected):
            self.order = None

    def next(self):
        if self.order is not None:
            return

        if len(self.data) < self.p.window + 1:
            return

        closes = self.data.close.get(size=self.p.window + 1)
        returns = [closes[i] / closes[i - 1] - 1 for i in range(1, len(closes))]

        numerator = 0.0
        denominator = 0.0
        for w in self.p.fractions:
            wealth = 1.0
            for r in returns:
                wealth *= (1 + w * r)
            numerator += w * wealth
            denominator += wealth

        w_next = numerator / denominator if denominator else 0.0

        self.order = self.order_target_percent(target=w_next)
