import backtrader as bt


class DonchianBreakoutStrategy(bt.Strategy):
    params = (("period", 20),)

    def __init__(self):
        self.upper = bt.indicators.Highest(self.data.high(-1), period=self.p.period)
        self.lower = bt.indicators.Lowest(self.data.low(-1), period=self.p.period)
        self.midpoint = (self.upper + self.lower) / 2.0
        self.order = None
        self.reverse_direction = None

    def next(self):
        if self.order:
            return

        close = self.data.close[0]

        if not self.position:
            if close > self.upper[0]:
                self.order = self.buy()
            elif close < self.lower[0]:
                self.order = self.sell()
            return

        if self.position.size > 0:
            if close < self.lower[0]:
                self.reverse_direction = -1
                self.order = self.close()
            elif close <= self.midpoint[0]:
                self.order = self.close()
        else:
            if close > self.upper[0]:
                self.reverse_direction = 1
                self.order = self.close()
            elif close >= self.midpoint[0]:
                self.order = self.close()

    def notify_order(self, order):
        if order.status in (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            self.order = None

            if order.status == order.Completed and not self.position and self.reverse_direction:
                direction = self.reverse_direction
                self.reverse_direction = None
                self.order = self.buy() if direction > 0 else self.sell()
            elif order.status != order.Completed:
                self.reverse_direction = None
