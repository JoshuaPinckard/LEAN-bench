import backtrader as bt
import math


class QuasiCreditSpreadsStrategy(bt.Strategy):
    params = dict(
        sma_trend_period=50,
        bb_period=20,
        bb_devfactor=2.0,
        atr_period=14,
        rsi_period=14,
        bb_width_sma_period=100,
        vol_sma_period=20,
        vol_mult=1.5,
        gap_atr_mult=1.5,
        rsi_long_thresh=45,
        rsi_short_thresh=55,
        tp_fraction=0.7,
        sl_atr_mult=1.2,
        band_breach_atr_mult=0.5,
        max_hold_bars=6,
        risk_pct=0.015,
    )

    def __init__(self):
        self.sma50 = bt.indicators.SMA(self.data.close, period=self.p.sma_trend_period)
        self.bb = bt.indicators.BollingerBands(
            self.data.close, period=self.p.bb_period, devfactor=self.p.bb_devfactor
        )
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.rsi = bt.indicators.RSI(self.data.close, period=self.p.rsi_period)

        bb_width = (self.bb.top - self.bb.bot) / self.bb.mid
        self.bb_width = bb_width
        self.bb_width_sma = bt.indicators.SMA(bb_width, period=self.p.bb_width_sma_period)
        self.vol_sma = bt.indicators.SMA(self.data.volume, period=self.p.vol_sma_period)

        # bb.mid is itself SMA(20) of close; reused as the profit-target/crossing reference
        self.sma20_cross = bt.indicators.CrossOver(self.data.close, self.bb.mid)

        self.order = None
        self.direction = None
        self.entry_bar = None
        self.target_price = None
        self.stop_price = None

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        if self.order:
            return

        if self.position:
            self._manage_position()
            return

        bb_width_ok = self.bb_width[0] < self.bb_width_sma[0]
        vol_ok = self.data.volume[0] <= self.p.vol_mult * self.vol_sma[0]
        gap_ok = abs(self.data.open[0] - self.data.close[-1]) < self.p.gap_atr_mult * self.atr[0]
        if not (bb_width_ok and vol_ok and gap_ok):
            return

        if (self.data.close[0] <= self.bb.bot[0]
                and self.data.close[0] > self.sma50[0]
                and self.rsi[0] < self.p.rsi_long_thresh):
            self._enter(direction=1)
        elif (self.data.close[0] >= self.bb.top[0]
                and self.data.close[0] < self.sma50[0]
                and self.rsi[0] > self.p.rsi_short_thresh):
            self._enter(direction=-1)

    def _enter(self, direction):
        entry_price = self.data.close[0]
        atr_val = self.atr[0]
        if atr_val <= 0:
            return

        equity = self.broker.getvalue()
        size = math.floor((equity * self.p.risk_pct) / (self.p.sl_atr_mult * atr_val))
        if size <= 0:
            return

        sma20_entry = self.bb.mid[0]
        # 70% of the distance from entry to SMA20-at-entry, direction-agnostic interpolation
        self.target_price = entry_price + self.p.tp_fraction * (sma20_entry - entry_price)
        self.stop_price = entry_price - direction * self.p.sl_atr_mult * atr_val
        self.direction = direction
        self.entry_bar = len(self)

        if direction == 1:
            self.order = self.buy(size=size)
        else:
            self.order = self.sell(size=size)

    def _manage_position(self):
        close = self.data.close[0]
        bars_held = len(self) - self.entry_bar
        time_hit = bars_held >= self.p.max_hold_bars

        if self.direction == 1:
            tp_hit = close >= self.target_price
            cross_hit = self.sma20_cross[0] > 0
            sl_hit = close <= self.stop_price
            band_breach = close <= (self.bb.bot[0] - self.p.band_breach_atr_mult * self.atr[0])
        else:
            tp_hit = close <= self.target_price
            cross_hit = self.sma20_cross[0] < 0
            sl_hit = close >= self.stop_price
            band_breach = close >= (self.bb.top[0] + self.p.band_breach_atr_mult * self.atr[0])

        if tp_hit or cross_hit or sl_hit or band_breach or time_hit:
            self.order = self.close()
            self.direction = None
            self.entry_bar = None
            self.target_price = None
            self.stop_price = None
