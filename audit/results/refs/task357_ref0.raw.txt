import backtrader as bt


class DonchianChannelFlipStrategy(bt.Strategy):
    params = dict(period=20)

    def __init__(self):
        high_prior = self.data.high(-1)
        low_prior = self.data.low(-1)
        self.upper = bt.indicators.Highest(high_prior, period=self.p.period)
        self.lower = bt.indicators.Lowest(low_prior, period=self.p.period)
        self.mid = (self.upper + self.lower) / 2.0

        self.order = None
        self.pending_flip = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return
        if order.status == order.Completed:
            if self.pending_flip is not None:
                direction = self.pending_flip
                self.pending_flip = None
                self.order = self.buy() if direction == 'long' else self.sell()
                return
            self.order = None
        else:
            self.pending_flip = None
            self.order = None

    def next(self):
        if self.order:
            return

        close = self.data.close[0]
        upper = self.upper[0]
        lower = self.lower[0]
        mid = self.mid[0]

        if not self.position:
            if close > upper:
                self.order = self.buy()
            elif close < lower:
                self.order = self.sell()
        elif self.position.size > 0:
            if close < lower:
                self.pending_flip = 'short'
                self.order = self.close()
            elif close <= mid:
                self.order = self.close()
        else:
            if close > upper:
                self.pending_flip = 'long'
                self.order = self.close()
            elif close >= mid:
                self.order = self.close()
