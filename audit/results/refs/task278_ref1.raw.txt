import backtrader as bt


class ProxyOptionIncomeStrategy(bt.Strategy):
    params = (
        ("strategy_mode", "covered_call"),  # covered_call, cash_secured_put, wheel, buy_and_hold
    )

    def __init__(self):
        self.bb = bt.indicators.BollingerBands(self.data.close, period=20, devfactor=2)
        self.sma = bt.indicators.SMA(self.data.close, period=50)
        self.rsi = bt.indicators.RSI(self.data.close, period=14)
        self.atr = bt.indicators.ATR(self.data, period=14)

        self.order = None
        self.cooldown_weeks = 0
        self.position_days = 0

    def notify_order(self, order):
        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            self.order = None

    def next(self):
        if self.order:
            return

        mode = self.p.strategy_mode

        if mode == "buy_and_hold":
            if not self.position:
                self.order = self.buy()
            return

        if self.position:
            self.position_days += 1

        if self.data.datetime.date(0).weekday() != 0:
            return

        close = self.data.close[0]
        upper = self.bb.top[0]
        middle = self.bb.mid[0]
        lower = self.bb.bot[0]
        rsi = self.rsi[0]
        atr = self.atr[0]

        if self.position:
            if mode == "cash_secured_put":
                if (
                    close >= middle
                    or self.position_days >= 20
                    or close <= lower - atr
                ):
                    self.order = self.close()
                    self.position_days = 0
                return

            if mode in ("covered_call", "wheel"):
                if (
                    close >= upper
                    or rsi >= 70
                    or close <= middle - atr
                ):
                    self.order = self.close()
                    self.position_days = 0
                    if mode == "covered_call":
                        self.cooldown_weeks = 1
                return

        if mode == "covered_call":
            if self.cooldown_weeks:
                self.cooldown_weeks -= 1
                return

            if close > self.sma[0] and rsi > 50:
                self.order = self.buy()

        elif mode in ("cash_secured_put", "wheel"):
            if close <= lower or rsi <= 35:
                self.order = self.buy()
