import backtrader as bt


class ArcherChannelBreakout(bt.Strategy):
    def __init__(self):
        self.order = None
        self.weekly_closes = []
        self.weekly_vols = []

        self.current_week_key = None
        self.current_week_vol = 0.0

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        if self.order:
            return

        dt = self.data.datetime.date(0)
        iso_year, iso_week, _ = dt.isocalendar()
        week_key = (iso_year, iso_week)

        if self.current_week_key != week_key:
            self.current_week_key = week_key
            self.current_week_vol = float(self.data.volume[0])
        else:
            self.current_week_vol += float(self.data.volume[0])

        current_week_close = float(self.data.close[0])

        try:
            next_dt = bt.num2date(self.data.datetime[1]).date()
            is_week_end = (next_dt.isocalendar()[:2] != week_key)
        except (IndexError, ValueError):
            is_week_end = True

        if is_week_end:
            self.weekly_closes.append(current_week_close)
            self.weekly_vols.append(self.current_week_vol)

        if self.position:
            # Daily triggers: Take-profit ($13.92) and Second Support ($8.23)
            if self.data.high[0] >= 13.92:
                self.order = self.close()
                return

            if self.data.low[0] < 8.23:
                self.order = self.close()
                return

            # Weekly close triggers
            if is_week_end:
                # Key Support Stop-loss ($8.78)
                if current_week_close < 8.78:
                    self.order = self.close()
                    return

                # Trend Filter Exit (50-week SMA)
                if len(self.weekly_closes) >= 50:
                    sma50_weekly = sum(self.weekly_closes[-50:]) / 50.0
                    if current_week_close < sma50_weekly:
                        self.order = self.close()
                        return
        else:
            if is_week_end:
                if len(self.weekly_closes) < 50 or len(self.weekly_vols) < 20:
                    return

                sma50_weekly = sum(self.weekly_closes[-50:]) / 50.0
                sma20_weekly_vol = sum(self.weekly_vols[-20:]) / 20.0

                # Entry conditions:
                # 1. Clean weekly close above resistance ($9.99)
                # 2. Trend filter: Price above 50-week SMA
                # 3. Volume filter: Weekly volume not 30%+ below 20-week volume SMA (>= 70%)
                if (
                    current_week_close > 9.99
                    and current_week_close > sma50_weekly
                    and self.current_week_vol >= 0.7 * sma20_weekly_vol
                ):
                    entry_price = current_week_close
                    stop_loss_price = 8.78
                    risk_per_share = entry_price - stop_loss_price

                    if risk_per_share > 0:
                        capital = self.broker.getvalue()
                        risk_amount = capital * 0.01  # 1% risk per trade
                        size = int(risk_amount / risk_per_share)
                        cash_size = int(self.broker.getcash() / entry_price)
                        size = min(size, cash_size)

                        if size > 0:
                            self.order = self.buy(size=size)
