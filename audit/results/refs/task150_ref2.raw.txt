import math
import backtrader as bt


class RenkoTrendStrategy(bt.Strategy):
    params = (
        ('brick_size', 50.0),
    )

    def __init__(self):
        self.renko_top = None
        self.renko_bottom = None
        self.renko_bricks = []

        self.entry_order = None
        self.sl_order = None
        self.tp_order = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status == order.Completed:
            if order == self.entry_order:
                self.entry_order = None
                entry_price = order.executed.price
                exec_size = order.executed.size
                if exec_size > 0:
                    stop_price = entry_price - 1.0 * self.p.brick_size
                    take_price = entry_price + 2.0 * self.p.brick_size
                    self.sl_order = self.sell(exectype=bt.Order.Stop, price=stop_price, size=abs(exec_size))
                    self.tp_order = self.sell(exectype=bt.Order.Limit, price=take_price, size=abs(exec_size))
                elif exec_size < 0:
                    stop_price = entry_price + 1.0 * self.p.brick_size
                    take_price = entry_price - 2.0 * self.p.brick_size
                    self.sl_order = self.buy(exectype=bt.Order.Stop, price=stop_price, size=abs(exec_size))
                    self.tp_order = self.buy(exectype=bt.Order.Limit, price=take_price, size=abs(exec_size))
            elif order == self.sl_order:
                self.sl_order = None
                if self.tp_order:
                    self.cancel(self.tp_order)
                    self.tp_order = None
            elif order == self.tp_order:
                self.tp_order = None
                if self.sl_order:
                    self.cancel(self.sl_order)
                    self.sl_order = None

        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            if order == self.entry_order:
                self.entry_order = None
            if order == self.sl_order:
                self.sl_order = None
            if order == self.tp_order:
                self.tp_order = None

    def next(self):
        close_price = self.data.close[0]

        if self.renko_top is None:
            self.renko_top = (math.floor(close_price / self.p.brick_size) + 1) * self.p.brick_size
            self.renko_bottom = self.renko_top - self.p.brick_size
            return

        new_brick = False
        while close_price >= self.renko_top + self.p.brick_size:
            self.renko_bottom = self.renko_top
            self.renko_top += self.p.brick_size
            self.renko_bricks.append(1)
            new_brick = True

        while close_price <= self.renko_bottom - self.p.brick_size:
            self.renko_top = self.renko_bottom
            self.renko_bottom -= self.p.brick_size
            self.renko_bricks.append(-1)
            new_brick = True

        if not new_brick:
            return

        # Exit existing positions on reversal signal
        if self.position.size > 0:
            if self.renko_bricks[-1] == -1:
                if self.sl_order:
                    self.cancel(self.sl_order)
                    self.sl_order = None
                if self.tp_order:
                    self.cancel(self.tp_order)
                    self.tp_order = None
                self.close()
        elif self.position.size < 0:
            if self.renko_bricks[-1] == 1:
                if self.sl_order:
                    self.cancel(self.sl_order)
                    self.sl_order = None
                if self.tp_order:
                    self.cancel(self.tp_order)
                    self.tp_order = None
                self.close()

        # Enter new position if flat and no entry order is pending
        if not self.position and self.entry_order is None:
            if len(self.renko_bricks) >= 2:
                if self.renko_bricks[-1] == 1 and self.renko_bricks[-2] == 1:
                    cash = self.broker.getcash()
                    size = int(cash / close_price)
                    if size > 0:
                        self.entry_order = self.buy(size=size)
                elif self.renko_bricks[-1] == -1 and self.renko_bricks[-2] == -1:
                    cash = self.broker.getcash()
                    size = int(cash / close_price)
                    if size > 0:
                        self.entry_order = self.sell(size=size)
