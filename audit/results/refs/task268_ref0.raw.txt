import backtrader as bt


class BullTrendBounceVolatilityFilter(bt.Strategy):
    params = dict(
        sma_period=20,
        rsi_period=14,
        bb_period=20,
        bb_devfactor=2.0,
        rsi_level=40.0,
        proxy_low=0.05,
        proxy_high=0.10,
        bb_width_max=0.02,
        limit_discount=0.005,
        tp_pct=0.02,
        sl_pct=0.015,
        max_hold_bars=5,
        risk_pct=0.02,
        max_entries=3,
        session_start_min=600,
        session_end_min=960,
    )

    def __init__(self):
        self.sma = bt.indicators.SMA(self.data.close, period=self.p.sma_period)
        self.rsi = bt.indicators.RSI(self.data.close, period=self.p.rsi_period)
        self.bb = bt.indicators.BollingerBands(
            self.data.close, period=self.p.bb_period, devfactor=self.p.bb_devfactor
        )
        self.rsi_cross = bt.indicators.CrossUp(self.rsi, self.p.rsi_level)
        self.entries = []

    def _in_session(self):
        t = self.data.datetime.time(0)
        minutes = t.hour * 60 + t.minute
        return self.p.session_start_min <= minutes <= self.p.session_end_min

    def _find_entry(self, order):
        for e in self.entries:
            if order in (e.get('main'), e.get('stop'), e.get('limit'), e.get('force')):
                return e
        return None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        entry = self._find_entry(order)
        if entry is None:
            return

        if order == entry.get('main'):
            if order.status == order.Completed:
                entry['status'] = 'open'
                entry['entry_price'] = order.executed.price
                entry['bar_open'] = len(self)
                stop_price = entry['entry_price'] * (1 - self.p.sl_pct)
                tp_price = entry['entry_price'] * (1 + self.p.tp_pct)
                stop_order = self.sell(
                    size=entry['size'], exectype=bt.Order.Stop, price=stop_price
                )
                limit_order = self.sell(
                    size=entry['size'], exectype=bt.Order.Limit, price=tp_price,
                    oco=stop_order
                )
                entry['stop'] = stop_order
                entry['limit'] = limit_order
            elif order.status in [order.Canceled, order.Margin, order.Rejected]:
                self.entries.remove(entry)
        elif order in (entry.get('stop'), entry.get('limit')):
            if order.status == order.Completed and entry in self.entries:
                self.entries.remove(entry)
        elif order == entry.get('force'):
            if order.status == order.Completed and entry in self.entries:
                self.entries.remove(entry)

    def next(self):
        for entry in list(self.entries):
            if entry.get('status') == 'open':
                bars_held = len(self) - entry['bar_open']
                if bars_held >= self.p.max_hold_bars:
                    stop_order = entry.get('stop')
                    limit_order = entry.get('limit')
                    if stop_order is not None and stop_order.status in [
                        stop_order.Submitted, stop_order.Accepted
                    ]:
                        self.cancel(stop_order)
                    if limit_order is not None and limit_order.status in [
                        limit_order.Submitted, limit_order.Accepted
                    ]:
                        self.cancel(limit_order)
                    force_order = self.sell(size=entry['size'], exectype=bt.Order.Market)
                    entry['force'] = force_order
                    entry['status'] = 'forced'

        if len(self.entries) >= self.p.max_entries:
            return
        if not self._in_session():
            return

        close = self.data.close[0]
        sma = self.sma[0]

        trend_ok = close > sma
        bounce_ok = self.rsi_cross[0] > 0
        proxy = (close - sma) / sma
        proxy_ok = self.p.proxy_low <= proxy <= self.p.proxy_high
        bb_width = (self.bb.top[0] - self.bb.bot[0]) / self.bb.mid[0]
        vol_ok = bb_width < self.p.bb_width_max

        if trend_ok and bounce_ok and proxy_ok and vol_ok:
            equity = self.broker.getvalue()
            qty = int(self.p.risk_pct * equity / close)
            if qty <= 0:
                return
            limit_price = close * (1 - self.p.limit_discount)
            main_order = self.buy(size=qty, exectype=bt.Order.Limit, price=limit_price)
            self.entries.append({
                'main': main_order,
                'size': qty,
                'status': 'pending',
            })
