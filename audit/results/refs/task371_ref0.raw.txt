import backtrader as bt


class ATRBollingerTrendStrategy(bt.Strategy):
    params = (
        ('p_smooth', 20),
        ('p_atr', 14),
        ('bb_period', 20),
        ('bb_devfactor', 2.0),
        ('atr_mult', 2.0),
    )

    def __init__(self):
        self.ema = bt.indicators.EMA(self.data.close, period=self.p.p_smooth)
        self.bb = bt.indicators.BollingerBands(
            self.data.close, period=self.p.bb_period, devfactor=self.p.bb_devfactor)
        self.atr = bt.indicators.ATR(self.data, period=self.p.p_atr)

        self.entry_order = None
        self.exit_order = None
        self.stop_order = None

    def _long_signal(self):
        return (self.data.close[-1] < self.bb.lines.bot[-1] and
                self.data.close[0] > self.bb.lines.bot[0] and
                self.data.close[0] > self.ema[0])

    def _short_signal(self):
        return (self.data.close[-1] > self.bb.lines.top[-1] and
                self.data.close[0] < self.bb.lines.top[0] and
                self.data.close[0] < self.ema[0])

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return

        if order.status == order.Completed:
            if order == self.entry_order:
                stop_distance = self.atr[0] * self.p.atr_mult
                if order.isbuy():
                    stop_price = order.executed.price - stop_distance
                    self.stop_order = self.sell(exectype=bt.Order.Stop, price=stop_price)
                else:
                    stop_price = order.executed.price + stop_distance
                    self.stop_order = self.buy(exectype=bt.Order.Stop, price=stop_price)
                self.entry_order = None
            elif order == self.stop_order:
                self.stop_order = None
            elif order == self.exit_order:
                self.exit_order = None
        else:
            if order == self.entry_order:
                self.entry_order = None
            elif order == self.stop_order:
                self.stop_order = None
            elif order == self.exit_order:
                self.exit_order = None

    def next(self):
        if not self.position:
            if self.entry_order:
                return
            if self._long_signal():
                self.entry_order = self.buy()
            elif self._short_signal():
                self.entry_order = self.sell()
        else:
            if self.exit_order:
                return
            if self.position.size > 0 and self._short_signal():
                if self.stop_order:
                    self.cancel(self.stop_order)
                self.exit_order = self.close()
            elif self.position.size < 0 and self._long_signal():
                if self.stop_order:
                    self.cancel(self.stop_order)
                self.exit_order = self.close()
