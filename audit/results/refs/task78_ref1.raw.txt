import backtrader as bt
import math


class ORBFibonacciRetracementStrategy(bt.Strategy):
    params = (
        ('risk_fraction', 0.01),
        ('reward_risk', 1.5),
        ('stop_offset', 0.01),
    )

    def __init__(self):
        self.order = None
        self.stop_order = None
        self.target_order = None
        self.current_day = None
        self.orb_high = None
        self.orb_low = None
        self.long_breakout_used = False
        self.short_breakout_used = False
        self.long_setup = None
        self.short_setup = None

    def notify_order(self, order):
        if order.status in (order.Completed, order.Canceled,
                            order.Margin, order.Rejected):
            if order == self.order:
                self.order = None
            if order == self.stop_order:
                self.stop_order = None
            if order == self.target_order:
                self.target_order = None

    def next(self):
        bar_date = self.data.datetime.date(0)
        bar_time = self.data.datetime.time(0)

        if self.current_day != bar_date:
            self.current_day = bar_date
            self.orb_high = self.data.high[0]
            self.orb_low = self.data.low[0]
            self.long_breakout_used = False
            self.short_breakout_used = False
            self.long_setup = None
            self.short_setup = None
            return

        if bar_time.hour == 15 and bar_time.minute == 55:
            if self.position and not self.order:
                if self.stop_order:
                    self.cancel(self.stop_order)
                    self.stop_order = None
                if self.target_order:
                    self.cancel(self.target_order)
                    self.target_order = None
                self.order = self.close()
            return

        prior_orb_high = self.orb_high
        prior_orb_low = self.orb_low

        if (not self.long_breakout_used and
                self.data.high[0] > prior_orb_high):
            self.long_breakout_used = True
            self.long_setup = (prior_orb_low, prior_orb_high)

        if (not self.short_breakout_used and
                self.data.low[0] < prior_orb_low):
            self.short_breakout_used = True
            self.short_setup = (prior_orb_low, prior_orb_high)

        self.orb_high = max(self.orb_high, self.data.high[0])
        self.orb_low = min(self.orb_low, self.data.low[0])

        if self.position or self.order or self.stop_order or self.target_order:
            return

        if self.long_setup:
            orb_low, orb_high = self.long_setup
            level_50 = orb_low + (orb_high - orb_low) * 0.5
            level_618 = orb_low + (orb_high - orb_low) * 0.618
            touched_zone = (self.data.low[0] <= level_618 and
                            self.data.high[0] >= level_50)
            reversal_up = self.data.close[0] > self.data.close[-1]

            if touched_zone and reversal_up:
                entry = self.data.close[0]
                stop = level_618 - self.p.stop_offset
                risk_per_share = entry - stop

                if risk_per_share > 0:
                    size = (self.broker.getvalue() * self.p.risk_fraction /
                            risk_per_share)
                    target = entry + risk_per_share * self.p.reward_risk
                    self.order = self.buy_bracket(
                        size=size,
                        exectype=bt.Order.Market,
                        stopprice=stop,
                        limitprice=target,
                    )[0]
                    bracket = self._orderspending[-2:]
                    self.stop_order = bracket[0] if len(bracket) == 2 else None
                    self.target_order = bracket[1] if len(bracket) == 2 else None
                    self.long_setup = None

        elif self.short_setup:
            orb_low, orb_high = self.short_setup
            level_50 = orb_low + (orb_high - orb_low) * 0.5
            level_618 = orb_low + (orb_high - orb_low) * 0.618
            touched_zone = (self.data.low[0] <= level_618 and
                            self.data.high[0] >= level_50)
            reversal_down = self.data.close[0] < self.data.close[-1]

            if touched_zone and reversal_down:
                entry = self.data.close[0]
                stop = level_618 + self.p.stop_offset
                risk_per_share = stop - entry

                if risk_per_share > 0:
                    size = (self.broker.getvalue() * self.p.risk_fraction /
                            risk_per_share)
                    target = entry - risk_per_share * self.p.reward_risk
                    self.order = self.sell_bracket(
                        size=size,
                        exectype=bt.Order.Market,
                        stopprice=stop,
                        limitprice=target,
                    )[0]
                    bracket = self._orderspending[-2:]
                    self.stop_order = bracket[0] if len(bracket) == 2 else None
                    self.target_order = bracket[1] if len(bracket) == 2 else None
                    self.short_setup = None
