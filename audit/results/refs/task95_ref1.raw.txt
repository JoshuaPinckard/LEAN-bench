import backtrader as bt
import math
from collections import deque


class VaROverlapStrategy(bt.Strategy):
    params = (
        ("var_period", 250),
        ("confidence_level", 0.95),
        ("forecast_horizon_days", 1),
        ("overlap_step_days", 0.5),
    )

    def __init__(self):
        self.returns = deque(maxlen=self.p.var_period)
        self.previous_close = None
        self.exceedances = 0
        self.exceedance_dates = []
        self.exceedance_distribution = {}
        self.exceedance_runs = []
        self._current_run = 0
        self.report = {}

    def next(self):
        close = float(self.data.close[0])

        if self.previous_close is None:
            self.previous_close = close
            return

        daily_return = (close - self.previous_close) / self.previous_close

        if len(self.returns) == self.p.var_period:
            mean_return = sum(self.returns) / self.p.var_period
            variance = sum(
                (value - mean_return) ** 2 for value in self.returns
            ) / (self.p.var_period - 1)
            historical_volatility = math.sqrt(variance)

            z_score = 1.6448536269514722
            var_return = z_score * historical_volatility
            actual_loss_return = -daily_return

            high_volatility_or_black_swan = (
                abs(daily_return) > 3.0 * historical_volatility
            )

            if actual_loss_return > var_return and not high_volatility_or_black_swan:
                date = self.data.datetime.date(0)
                year = str(date.year)

                self.exceedances += 1
                self.exceedance_dates.append(date)
                self.exceedance_distribution[year] = (
                    self.exceedance_distribution.get(year, 0) + 1
                )
                self._current_run += 1
            else:
                if self._current_run:
                    self.exceedance_runs.append(self._current_run)
                    self._current_run = 0

        self.returns.append(daily_return)
        self.previous_close = close

    def stop(self):
        if self._current_run:
            self.exceedance_runs.append(self._current_run)

        clustered_runs = [run for run in self.exceedance_runs if run > 1]
        self.report = {
            "exceedances": self.exceedances,
            "distribution_over_time": self.exceedance_distribution,
            "exceedance_dates": self.exceedance_dates,
            "exceedance_runs": self.exceedance_runs,
            "clustered_runs": clustered_runs,
            "conclusion": (
                "Exceedances were clustered, indicating possible VaR model "
                "misspecification under dependent overlapping intervals."
                if clustered_runs
                else
                "No clustered exceedance sequences were observed; the VaR model "
                "does not show clustering-based evidence of misspecification."
            ),
        }
