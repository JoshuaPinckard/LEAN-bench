# complete strategy class
import backtrader as bt
import math


class PortfolioOptimizationWithCosts(bt.Strategy):
    params = (
        ("lookback", 252),
        ("risk_aversion", 1.0),
        ("impact_coefficient", 0.001),
        ("optimization_steps", 10000),
    )

    def __init__(self):
        self.pending_order = None
        self.addminperiod(self.p.lookback + 1)

    def notify_order(self, order):
        if order.status in (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            self.pending_order = None

    def _current_weight(self):
        value = self.broker.getvalue()
        if value <= 0:
            return 0.0
        return max(0.0, self.position.size * self.data.close[0] / value)

    def _objective(self, weight, current_weight, expected_return, covariance):
        impact_cost = self.p.impact_coefficient * abs(weight - current_weight) ** 1.5
        return (
            (expected_return - impact_cost) * weight
            - self.p.risk_aversion * covariance * weight * weight
            - self.p.impact_coefficient * abs(weight) ** 1.5
        )

    def next(self):
        if self.pending_order is not None:
            return

        volume = self.data.volume[0]
        if not math.isfinite(volume) or volume <= 0:
            target_weight = 0.0
        else:
            returns = []
            for offset in range(self.p.lookback):
                previous_close = self.data.close[-offset - 1]
                current_close = self.data.close[-offset]
                if previous_close > 0:
                    returns.append(current_close / previous_close - 1.0)

            if len(returns) < 2:
                return

            mean_daily_return = sum(returns) / len(returns)
            expected_return = mean_daily_return * self.p.lookback
            covariance = (
                sum((item - mean_daily_return) ** 2 for item in returns)
                / (len(returns) - 1)
                * self.p.lookback
            )

            current_weight = min(1.0, self._current_weight())
            target_weight = current_weight
            best_value = self._objective(
                target_weight, current_weight, expected_return, covariance
            )

            for step in range(self.p.optimization_steps + 1):
                weight = step / self.p.optimization_steps
                value = self._objective(
                    weight, current_weight, expected_return, covariance
                )
                if value > best_value:
                    best_value = value
                    target_weight = weight

        if abs(target_weight - self._current_weight()) > 0:
            self.pending_order = self.order_target_percent(
                target=target_weight
            )
