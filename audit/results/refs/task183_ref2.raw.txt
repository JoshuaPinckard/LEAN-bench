import backtrader as bt


class BreakoutStrategy(bt.Strategy):
    params = (
        ('period', 10),
        ('stop_loss_pct', 0.02),
        ('position_size_pct', 0.01),
    )

    def __init__(self):
        self.hi = bt.indicators.Highest(self.data.close, period=self.p.period)
        self.lo = bt.indicators.Lowest(self.data.close, period=self.p.period)
        self.order = None
        self.stop_order = None
        self.entry_price = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            if order.isbuy():
                self.entry_price = order.executed.price
                stop_price = self.entry_price * (1.0 - self.p.stop_loss_pct)
                self.stop_order = self.sell(
                    exectype=bt.Order.Stop,
                    price=stop_price,
                    size=order.executed.size
                )
                self.order = None
            elif order.issell():
                self.order = None
                self.stop_order = None
                self.entry_price = None

        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            if order == self.order:
                self.order = None
            if order == self.stop_order:
                self.stop_order = None

    def next(self):
        if not self.position:
            if self.order:
                return
            if self.data.close[0] > self.hi[-1]:
                capital = self.broker.getvalue()
                size = (capital * self.p.position_size_pct) / self.data.close[0]
                if size > 0:
                    self.order = self.buy(size=size)
        else:
            if self.order:
                return
            if self.data.close[0] < self.lo[-1]:
                if self.stop_order:
                    self.cancel(self.stop_order)
                    self.stop_order = None
                self.order = self.close()
