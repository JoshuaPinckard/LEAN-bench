import backtrader as bt


class BBStochMeanReversion(bt.Strategy):
    params = dict(
        bb_period=20,
        bb_devfactor=2.0,
        stoch_period=14,
        stoch_period_dfast=3,
        stoch_period_dslow=3,
        oversold=20,
        overbought=80,
        stop_loss_pct=0.02,
    )

    def __init__(self):
        self.bb = bt.indicators.BollingerBands(
            self.data.close,
            period=self.p.bb_period,
            devfactor=self.p.bb_devfactor,
        )
        self.stoch = bt.indicators.Stochastic(
            self.data,
            period=self.p.stoch_period,
            period_dfast=self.p.stoch_period_dfast,
            period_dslow=self.p.stoch_period_dslow,
        )
        self.order = None
        self.entry_price = None

    def notify_order(self, order):
        if order.status == order.Completed:
            if self.position.size == 0:
                self.entry_price = None
            else:
                self.entry_price = order.executed.price
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        if self.order:
            return

        price = self.data.close[0]
        stoch_k = self.stoch.percK[0]

        long_signal = price <= self.bb.bot[0] and stoch_k < self.p.oversold
        short_signal = price >= self.bb.top[0] and stoch_k > self.p.overbought

        if not self.position:
            if long_signal:
                self.order = self.buy()
            elif short_signal:
                self.order = self.sell()
            return

        if self.position.size > 0:
            stop_price = self.entry_price * (1 - self.p.stop_loss_pct)
            if short_signal or price <= stop_price:
                self.order = self.close()
        elif self.position.size < 0:
            stop_price = self.entry_price * (1 + self.p.stop_loss_pct)
            if long_signal or price >= stop_price:
                self.order = self.close()
