import backtrader as bt


class VwapBreakoutStrategy(bt.Strategy):
    params = dict(
        vwap_period=20,
        sma_period=20,
        stop_pct=0.005,
    )

    def __init__(self):
        pv = self.data.close * self.data.volume
        pv_sum = bt.indicators.SumN(pv, period=self.p.vwap_period)
        vol_sum = bt.indicators.SumN(self.data.volume, period=self.p.vwap_period)
        self.vwap20 = pv_sum / vol_sum
        self.sma20 = bt.indicators.SMA(self.data.close, period=self.p.sma_period)

        self.cross_vwap = bt.indicators.CrossOver(self.data.close, self.vwap20)
        self.cross_sma = bt.indicators.CrossOver(self.data.close, self.sma20)

        self.entry_order = None
        self.exit_order = None
        self.stop_order = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order is self.entry_order:
            if order.status == order.Completed:
                entry_price = order.executed.price
                if order.isbuy():
                    stop_price = entry_price * (1 - self.p.stop_pct)
                    self.stop_order = self.sell(exectype=bt.Order.Stop, price=stop_price)
                else:
                    stop_price = entry_price * (1 + self.p.stop_pct)
                    self.stop_order = self.buy(exectype=bt.Order.Stop, price=stop_price)
            if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
                self.entry_order = None

        elif order is self.exit_order:
            if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
                self.exit_order = None

        elif order is self.stop_order:
            if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
                self.stop_order = None

    def next(self):
        if self.entry_order or self.exit_order:
            return

        if not self.position:
            vwap_slope = self.vwap20[0] - self.vwap20[-1]
            if self.cross_vwap[0] > 0 and vwap_slope > 0 and self.data.close[0] > self.sma20[0]:
                self.entry_order = self.buy()
            elif self.cross_vwap[0] < 0 and vwap_slope < 0 and self.data.close[0] < self.sma20[0]:
                self.entry_order = self.sell()
        else:
            exit_signal = (
                (self.position.size > 0 and self.cross_sma[0] < 0) or
                (self.position.size < 0 and self.cross_sma[0] > 0)
            )
            if exit_signal:
                if self.stop_order is not None:
                    self.cancel(self.stop_order)
                self.exit_order = self.close()
