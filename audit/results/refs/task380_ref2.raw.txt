import backtrader as bt


class IntradayMACDATRStrategy(bt.Strategy):
    """
    Intraday trend-following strategy using MACD(12,26,9) histogram crossings
    for long/short entries/exits and ATR(14) for fixed protective stops.
    """
    params = (
        ('stake', 100),
    )

    def __init__(self):
        self.macd = bt.indicators.MACDHisto(
            self.data,
            period_me1=12,
            period_me2=26,
            period_signal=9
        )
        self.atr = bt.indicators.ATR(self.data, period=14)

        self.order = None
        self.stop_order = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            if order == self.order:
                self.order = None

                if self.stop_order:
                    self.cancel(self.stop_order)
                    self.stop_order = None

                if self.position.size > 0:
                    stop_price = order.executed.price - 2.0 * self.atr[0]
                    self.stop_order = self.sell(
                        exectype=bt.Order.Stop,
                        price=stop_price,
                        size=abs(self.position.size)
                    )
                elif self.position.size < 0:
                    stop_price = order.executed.price + 2.0 * self.atr[0]
                    self.stop_order = self.buy(
                        exectype=bt.Order.Stop,
                        price=stop_price,
                        size=abs(self.position.size)
                    )
            elif order == self.stop_order:
                self.stop_order = None

        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            if order == self.order:
                self.order = None
            if order == self.stop_order:
                self.stop_order = None

    def next(self):
        if self.order:
            return

        histo_curr = self.macd.histo[0]
        histo_prev = self.macd.histo[-1]

        cross_above_05 = (histo_curr > 0.5) and (histo_prev <= 0.5)
        cross_below_m05 = (histo_curr < -0.5) and (histo_prev >= -0.5)

        if self.position.size == 0:
            if cross_above_05:
                self.order = self.buy(size=self.p.stake)
            elif cross_below_m05:
                self.order = self.sell(size=self.p.stake)

        elif self.position.size > 0:
            if cross_below_m05:
                if self.stop_order:
                    self.cancel(self.stop_order)
                    self.stop_order = None
                self.order = self.sell(size=self.position.size + self.p.stake)

        elif self.position.size < 0:
            if cross_above_05:
                if self.stop_order:
                    self.cancel(self.stop_order)
                    self.stop_order = None
                self.order = self.buy(size=abs(self.position.size) + self.p.stake)
