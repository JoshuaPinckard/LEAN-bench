import backtrader as bt


class FibGoldenPocketStrategy(bt.Strategy):
    params = dict(
        swing_period=200,
        exit_period=20,
        fib_618=0.618,
        fib_786=0.786,
    )

    def __init__(self):
        self.order = None
        self.highest_exit = bt.indicators.Highest(self.data.high, period=self.p.exit_period)
        self.stage = 0
        self.prev_swing_high = None
        self.prev_swing_low = None

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        if len(self.data) < self.p.swing_period:
            return

        highs = list(self.data.high.get(size=self.p.swing_period))
        lows = list(self.data.low.get(size=self.p.swing_period))

        swing_high = max(highs)
        swing_low = min(lows)
        swing_high_idx = highs.index(swing_high)
        swing_low_idx = lows.index(swing_low)

        if swing_high_idx < swing_low_idx:
            # swing high occurs earlier in the window -> uptrend: retrace from swing low to swing high
            level_618 = swing_high - (swing_high - swing_low) * self.p.fib_618
            level_786 = swing_high - (swing_high - swing_low) * self.p.fib_786
        else:
            # otherwise: retrace from swing high to swing low
            level_618 = swing_low + (swing_high - swing_low) * self.p.fib_618
            level_786 = swing_low + (swing_high - swing_low) * self.p.fib_786

        if swing_high != self.prev_swing_high or swing_low != self.prev_swing_low:
            self.stage = 0
        self.prev_swing_high = swing_high
        self.prev_swing_low = swing_low

        if self.order:
            return

        if self.position:
            if self.data.close[0] >= self.highest_exit[0]:
                self.order = self.close()
            return

        close = self.data.close[0]
        low = self.data.low[0]

        if self.stage == 0 and close < level_618:
            self.stage = 1
        elif self.stage == 1 and low <= level_786:
            self.stage = 2
        elif self.stage == 2 and close > level_618:
            self.order = self.buy()
            self.stage = 3
