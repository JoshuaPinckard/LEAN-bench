import backtrader as bt


class VWAPBreakoutStrategy(bt.Strategy):
    params = (("stop_pct", 0.005),)

    def __init__(self):
        volume_sum = bt.ind.SumN(self.data.volume, period=20)
        close_volume_sum = bt.ind.SumN(
            self.data.close * self.data.volume, period=20
        )
        self.vwap20 = close_volume_sum / volume_sum
        self.vwap_slope = self.vwap20 - self.vwap20(-1)
        self.sma20 = bt.ind.SMA(self.data.close, period=20)

        self.order = None
        self.stop_order = None

    def notify_order(self, order):
        if order is self.order and order.status in (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            self.order = None

        if order is self.stop_order and order.status in (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            self.stop_order = None

    def _cancel_stop(self):
        if self.stop_order is not None and self.stop_order.alive():
            self.cancel(self.stop_order)

    def next(self):
        if self.order is not None:
            return

        close_crosses_above_vwap = (
            self.data.close[0] > self.vwap20[0]
            and self.data.close[-1] <= self.vwap20[-1]
        )
        close_crosses_below_vwap = (
            self.data.close[0] < self.vwap20[0]
            and self.data.close[-1] >= self.vwap20[-1]
        )
        close_crosses_below_sma = (
            self.data.close[0] < self.sma20[0]
            and self.data.close[-1] >= self.sma20[-1]
        )
        close_crosses_above_sma = (
            self.data.close[0] > self.sma20[0]
            and self.data.close[-1] <= self.sma20[-1]
        )

        if not self.position:
            if (
                close_crosses_above_vwap
                and self.vwap_slope[0] > 0
                and self.data.close[0] > self.sma20[0]
            ):
                bracket = self.buy_bracket(
                    exectype=bt.Order.Market,
                    stopprice=self.data.close[0] * (1 - self.p.stop_pct),
                    limitexec=None,
                )
                self.order = bracket[0]
                self.stop_order = bracket[1]

            elif (
                close_crosses_below_vwap
                and self.vwap_slope[0] < 0
                and self.data.close[0] < self.sma20[0]
            ):
                bracket = self.sell_bracket(
                    exectype=bt.Order.Market,
                    stopprice=self.data.close[0] * (1 + self.p.stop_pct),
                    limitexec=None,
                )
                self.order = bracket[0]
                self.stop_order = bracket[1]

        elif self.position.size > 0 and close_crosses_below_sma:
            self._cancel_stop()
            self.order = self.close()

        elif self.position.size < 0 and close_crosses_above_sma:
            self._cancel_stop()
            self.order = self.close()
