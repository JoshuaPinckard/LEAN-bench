import backtrader as bt


class CostAwarePortfolioOptimization(bt.Strategy):
    # Cost-aware portfolio optimization: at each step maximize
    #   (r - I(t))'w - lambda * w'Sigma w - Theta * |w - w0|^(3/2)
    # For this single-asset feed all vectors/matrices collapse to scalars,
    # so w is the AAPL portfolio weight, r a scalar expected return and
    # Sigma a scalar variance.
    params = (
        ('lambda_risk', 1.0),   # lambda: risk aversion coefficient
        ('theta', 0.01),        # Theta: market-impact scaling coefficient
        ('period', 252),        # T = 1 year lookback for r and Sigma
        ('ann_factor', 252),    # bars per year, to annualize r and Sigma
        ('min_weight', 0.0),    # lower bound l_b on the asset weight
        ('max_weight', 1.0),    # upper bound u_b on the asset weight
        ('grid', 200),          # grid resolution for the 1-D optimization
    )

    def __init__(self):
        # daily simple returns
        ret = self.data.close / self.data.close(-1) - 1.0
        # expected return r and risk Sigma from historical returns over T = 1 year
        self.mean_ret = bt.indicators.SMA(ret, period=self.p.period)
        self.std_ret = bt.indicators.StandardDeviation(ret, period=self.p.period)
        self.order = None

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return
        if order.status in (order.Completed, order.Canceled,
                             order.Margin, order.Rejected):
            self.order = None

    def _objective(self, w, w0, r, sigma2):
        # trading volume t = |w - w0| and impact cost I(t) ~ Theta * t^(3/2)
        t = abs(w - w0)
        impact = self.p.theta * (t ** 1.5)
        # (r - I(t)) w - lambda w Sigma w - Theta |w - w0|^(3/2)
        return (r - impact) * w - self.p.lambda_risk * sigma2 * w * w - impact

    def next(self):
        # guard against overlapping orders
        if self.order is not None:
            return

        # expected return r and variance Sigma over the reinvestment period
        r = self.mean_ret[0] * self.p.ann_factor
        sigma2 = (self.std_ret[0] ** 2) * self.p.ann_factor

        # current portfolio weight w0 of the asset
        value = self.broker.getvalue()
        w0 = 0.0
        if value > 0:
            w0 = (self.position.size * self.data.close[0]) / value

        # solve  max_w  (r - I(t))w - lambda w Sigma w - Theta |w - w0|^(3/2)
        # subject to the linear box constraint  min_weight <= w <= max_weight,
        # by grid search (no external optimizers available in this environment).
        lb, ub = self.p.min_weight, self.p.max_weight
        # candidate "do nothing" weight (staying at w0) is the incumbent, so the
        # optimizer only trades when moving beats staying net of impact cost.
        best_w = min(max(w0, lb), ub)
        best_f = self._objective(best_w, w0, r, sigma2)
        n = self.p.grid
        for i in range(n + 1):
            w = lb + (ub - lb) * i / n
            f = self._objective(w, w0, r, sigma2)
            if f > best_f:
                best_f = f
                best_w = w

        # rebalance to the new target weight w* (dynamic rebalancing: the impact
        # term already suppressed trades whose benefit did not exceed their cost)
        if abs(best_w - w0) > 1e-9:
            self.order = self.order_target_percent(target=best_w)
