import backtrader as bt


class ReferenceStrategy(bt.Strategy):
    """
    Reference implementation of a DEMA(200) filtered ATR channel strategy.
    
    Upper Channel = EMA(10) + 3 * ATR(10)
    Lower Channel = EMA(10) - 3 * ATR(10)
    DEMA(200) = 2 * EMA(200) - EMA(EMA(200))
    
    Long Entry: close > DEMA(200) AND price crosses Upper Channel from below
    Long Exit: price crosses Lower Channel downwards OR close < DEMA(200)
    Position Size: 100% of available capital
    """

    def __init__(self):
        self.order = None

        # DEMA(200) calculation: 2 * EMA(200) - EMA(EMA(200))
        self.ema200 = bt.indicators.EMA(self.data.close, period=200)
        self.ema_ema200 = bt.indicators.EMA(self.ema200, period=200)
        self.dema200 = 2.0 * self.ema200 - self.ema_ema200

        # EMA(10) and ATR(10) channel
        self.ema10 = bt.indicators.EMA(self.data.close, period=10)
        self.atr10 = bt.indicators.ATR(self.data, period=10)

        self.upper = self.ema10 + 3.0 * self.atr10
        self.lower = self.ema10 - 3.0 * self.atr10

        # Crossovers against Close price
        self.crossover_upper = bt.indicators.CrossOver(self.data.close, self.upper)
        self.crossover_lower = bt.indicators.CrossOver(self.data.close, self.lower)

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        if self.order:
            return

        if not self.position:
            # Long entry condition: close > DEMA(200) and price crosses Upper channel from below
            if self.data.close[0] > self.dema200[0] and self.crossover_upper[0] > 0:
                cash = self.broker.getcash()
                price = self.data.close[0]
                comm = self.broker.getcommissioninfo(self.data).p.commission
                size = int(cash / (price * (1.0 + comm)))
                if size > 0:
                    self.order = self.buy(size=size)
        else:
            # Long exit condition: price crosses below Lower or close below DEMA(200)
            if self.crossover_lower[0] < 0 or self.data.close[0] < self.dema200[0]:
                self.order = self.close()
