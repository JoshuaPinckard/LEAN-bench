import backtrader as bt


class Strategy(bt.Strategy):
    params = (
        ('rsi_period', 14),
        ('donchian_period', 20),
        ('stop_loss_pct', 0.02),
    )

    def __init__(self):
        self.rsi = bt.indicators.RSI(self.data.close, period=self.p.rsi_period)
        self.donchian_upper = bt.indicators.Highest(
            self.data.high(-1), period=self.p.donchian_period
        )
        self.donchian_lower = bt.indicators.Lowest(
            self.data.low(-1), period=self.p.donchian_period
        )
        self.donchian_middle = (self.donchian_upper + self.donchian_lower) / 2.0

        self._pending_order = None
        self._stop_order = None
        self._exit_after_stop_cancel = False

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return

        if order.status == order.Completed:
            if order == self._stop_order:
                self._stop_order = None
                self._pending_order = None
                return

            if order == self._pending_order:
                self._pending_order = None

            if order.isbuy() and self.position.size > 0:
                self._stop_order = self.sell(
                    exectype=bt.Order.Stop,
                    price=order.executed.price * (1.0 - self.p.stop_loss_pct),
                )
            elif order.issell() and self.position.size < 0:
                self._stop_order = self.buy(
                    exectype=bt.Order.Stop,
                    price=order.executed.price * (1.0 + self.p.stop_loss_pct),
                )

        elif order.status in (order.Canceled, order.Margin, order.Rejected):
            if order == self._stop_order:
                self._stop_order = None
            if order == self._pending_order:
                self._pending_order = None

        if (
            order == self._stop_order
            and order.status == order.Canceled
            and self._exit_after_stop_cancel
        ):
            self._exit_after_stop_cancel = False
            self._pending_order = self.close()

    def next(self):
        if self._pending_order is not None:
            return

        if not self.position:
            if self.rsi[0] < 30 and self.data.close[0] > self.donchian_upper[0]:
                self._pending_order = self.buy()
            elif self.rsi[0] > 70 and self.data.close[0] < self.donchian_lower[0]:
                self._pending_order = self.sell()
            return

        exit_long = self.position.size > 0 and self.data.close[0] < self.donchian_middle[0]
        exit_short = self.position.size < 0 and self.data.close[0] > self.donchian_middle[0]

        if exit_long or exit_short:
            if self._stop_order is not None:
                self._exit_after_stop_cancel = True
                self._pending_order = self._stop_order
                self.cancel(self._stop_order)
            else:
                self._pending_order = self.close()
