```python
import backtrader as bt
import math


class ORBStrategy(bt.Strategy):
    params = (
        ("range_minutes", 60),
        ("risk_pct", 0.01),
        ("min_range_atr", 0.5),
        ("atr_trail_mult", 1.0),
    )

    def __init__(self):
        self.ema = bt.indicators.EMA(self.data.close, period=21)
        self.rsi = bt.indicators.RSI(self.data.close, period=14)
        self.bb = bt.indicators.BollingerBands(
            self.data.close, period=20, devfactor=2.0
        )
        self.atr = bt.indicators.ATR(self.data, period=14)
        self.long_cross = bt.indicators.CrossOver(self.data.close, self.ema)
        self.short_cross = bt.indicators.CrossOver(self.ema, self.data.close)

        self.entry_order = None
        self.stop_order = None
        self.tp_order = None
        self.close_order = None

        self.day = None
        self.orh = None
        self.orl = None
        self.range_complete = False
        self.long_breakout = False
        self.short_breakout = False
        self.long_signaled = False
        self.short_signaled = False
        self.long_bounce_high = None
        self.short_bounce_low = None

        self.entry_price = None
        self.entry_range = None
        self.entry_direction = None
        self.entry_session = None
        self.current_trade_session = None
        self.morning_loss = False
        self.trailing_enabled = False
        self.active_stop_price = None

    def _minutes(self):
        dt = self.data.datetime.datetime(0)
        return dt.hour * 60 + dt.minute

    def _reset_day(self):
        self.orh = None
        self.orl = None
        self.range_complete = False
        self.long_breakout = False
        self.short_breakout = False
        self.long_signaled = False
        self.short_signaled = False
        self.long_bounce_high = None
        self.short_bounce_low = None
        self.morning_loss = False

    def _session(self, minute):
        if minute < 12 * 60:
            return "london"
        if minute >= 13 * 60 + 30:
            return "new_york"
        return None

    def _fib_touched_long(self, level):
        return self.data.low[0] <= level <= self.data.high[0]

    def _fib_touched_short(self, level):
        return self.data.low[0] <= level <= self.data.high[0]

    def _valid_filters(self, direction):
        if direction == "long":
            return (
                self.data.close[0] > self.ema[0]
                and self.data.close[0] > self.bb.mid[0]
                and self.rsi[0] > 50
            )
        return (
            self.data.close[0] < self.ema[0]
            and self.data.close[0] < self.bb.mid[0]
            and self.rsi[0] < 50
        )

    def _quality(self, direction):
        score = 0
        width_now = self.bb.top[0] - self.bb.bot[0]
        width_prev = self.bb.top[-1] - self.bb.bot[-1]

        if direction == "long":
            if self.rsi[0] >= 55:
                score += 1
            if self.ema[0] > self.ema[-1]:
                score += 1
        else:
            if self.rsi[0] <= 45:
                score += 1
            if self.ema[0] < self.ema[-1]:
                score += 1

        if width_now > width_prev:
            score += 1
        return score

    def _submit_entry(self, direction, session):
        stop_price = self.orl if direction == "long" else self.orh
        risk_per_share = abs(self.data.close[0] - stop_price)
        if risk_per_share <= 0:
            return

        size = int(round((self.p.risk_pct * self.broker.getvalue()) / risk_per_share))
        if size <= 0:
            return

        entry_range = self.orh - self.orl
        quality = self._quality(direction)
        patient = quality >= 2 or (
            session == "new_york" and self.morning_loss
        )
        target_multiple = 1.5 if patient else 1.0

        self.entry_price = self.data.close[0]
        self.entry_range = entry_range
        self.entry_direction = direction
        self.entry_session = session
        self.current_trade_session = session
        self.trailing_enabled = False
        self.active_stop_price = stop_price
        self.pending_target_multiple = target_multiple

        if direction == "long":
            self.entry_order = self.buy(size=size)
            self.long_signaled = True
        else:
            self.entry_order = self.sell(size=size)
            self.short_signaled = True

    def _place_exits(self):
        if not self.position or self.entry_range is None:
            return

        size = abs(self.position.size)
        if self.position.size > 0:
            target = self.entry_price + self.pending_target_multiple * self.entry_range
            self.stop_order = self.sell(
                size=size,
                exectype=bt.Order.Stop,
                price=self.active_stop_price,
            )
            self.tp_order = self.sell(
                size=size,
                exectype=bt.Order.Limit,
                price=target,
                oco=self.stop_order,
            )
        else:
            target = self.entry_price - self.pending_target_multiple * self.entry_range
            self.stop_order = self.buy(
                size=size,
                exectype=bt.Order.Stop,
                price=self.active_stop_price,
            )
            self.tp_order = self.buy(
                size=size,
                exectype=bt.Order.Limit,
                price=target,
                oco=self.stop_order,
            )

    def _replace_exits_for_trailing(self, new_stop):
        if new_stop == self.active_stop_price:
            return

        if self.stop_order is not None:
            self.cancel(self.stop_order)
        if self.tp_order is not None:
            self.cancel(self.tp_order)

        self.active_stop_price = new_stop
        self._place_exits()

    def _timeout_close(self):
        if self.stop_order is not None:
            self.cancel(self.stop_order)
        if self.tp_order is not None:
            self.cancel(self.tp_order)
        self.close_order = self.close()

    def next(self):
        dt = self.data.datetime.datetime(0)
        if self.day != dt.date():
            self.day = dt.date()
            self._reset_day()

        minute = self._minutes()
        open_minute = 9 * 60 + 30
        range_end = open_minute + self.p.range_minutes
        timeout_minute = 15 * 60 + 30

        if open_minute <= minute < range_end:
            self.orh = self.data.high[0] if self.orh is None else max(self.orh, self.data.high[0])
            self.orl = self.data.low[0] if self.orl is None else min(self.orl, self.data.low[0])
            return

        if self.orh is None or self.orl is None:
            return

        if not self.range_complete and minute >= range_end:
            self.range_complete = True

        if self.position:
            if minute >= timeout_minute and self.close_order is None:
                self._timeout_close()
                return

            profit = (
                self.data.close[0] - self.entry_price
                if self.position.size > 0
                else self.entry_price - self.data.close[0]
            )
            if profit > self.entry_range:
                self.trailing_enabled = True

            if self.trailing_enabled:
                if self.position.size > 0:
                    new_stop = max(
                        self.active_stop_price,
                        self.data.close[0] - self.p.atr_trail_mult * self.atr[0],
                    )
                else:
                    new_stop = min(
                        self.active_stop_price,
                        self.data.close[0] + self.p.atr_trail_mult * self.atr[0],
                    )
                self._replace_exits_for_trailing(new_stop)
            return

        if self.entry_order is not None or self.close_order is not None:
            return

        if minute >= timeout_minute:
            return

        session = self._session(minute)
        if session is None:
            return

        current_range = self.orh - self.orl
        if current_range < self.p.min_range_atr * self.atr[0]:
            return

        if minute < 12 * 60:
            if not self.long_breakout and self.data.high[0] > self.orh:
                self.long_breakout = True
            if not self.short_breakout and self.data.low[0] < self.orl:
                self.short_breakout = True

        fibs = (
            self.orl + 0.382 * current_range,
            self.orl + 0.500 * current_range,
            self.orl + 0.618 * current_range,
        )

        if self.long_breakout and not self.long_signaled:
            bounced = (
                any(self._fib_touched_long(level) for level in fibs)
                and self.data.close[0] > self.data.open[0]
            )
            if bounced:
                self.long_bounce_high = self.data.high[0]

            triggered = (
                self.long_cross[0] > 0
                or (
                    self.long_bounce_high is not None
                    and self.data.high[0] > self.long_bounce_high
                )
            )
            if triggered and self._valid_filters("long"):
                self._submit_entry("long", session)
                return

        if self.short_breakout and not self.short_signaled:
            bounced = (
                any(self._fib_touched_short(level) for level in fibs)
                and self.data.close[0] < self.data.open[0]
            )
            if bounced:
                self.short_bounce_low = self.data.low[0]

            triggered = (
                self.short_cross[0] > 0
                or (
                    self.short_bounce_low is not None
                    and self.data.low[0] < self.short_bounce_low
                )
            )
            if triggered and self._valid_filters("short"):
                self._submit_entry("short", session)

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return

        if order is self.entry_order:
            if order.status == order.Completed:
                self.entry_price = order.executed.price
                self.entry_order = None
                self._place_exits()
            elif order.status in (
                order.Canceled, order.Margin, order.Rejected, order.Expired
            ):
                self.entry_order = None
            return

        if order is self.close_order:
            if order.status in (
                order.Completed, order.Canceled, order.Margin, order.Rejected, order.Expired
            ):
                self.close_order = None
            return

        if order is self.stop_order:
            if order.status == order.Completed and self.tp_order is not None:
                self.cancel(self.tp_order)
            if order.status in (
                order.Completed, order.Canceled, order.Margin, order.Rejected, order.Expired
            ):
                self.stop_order = None
            return

        if order is self.tp_order:
            if order.status == order.Completed and self.stop_order is not None:
                self.cancel(self.stop_order)
            if order.status in (
                order.Completed, order.Canceled, order.Margin, order.Rejected, order.Expired
            ):
                self.tp_order = None

    def notify_trade(self, trade):
        if not trade.isclosed:
            return

        if self.current_trade_session == "london" and trade.pnlcomm < 0:
            self.morning_loss = True

        self.entry_price = None
        self.entry_range = None
        self.entry_direction = None
        self.entry_session = None
        self.current_trade_session = None
        self.trailing_enabled = False
        self.active_stop_price = None
```
