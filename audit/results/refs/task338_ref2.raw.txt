import backtrader as bt


class JanuaryEffectStrategy(bt.Strategy):
    def __init__(self):
        self.sma = bt.indicators.SimpleMovingAverage(self.data.close, period=20)
        self.order = None
        self.stop_order = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            if order == self.order:
                self.order = None
                if order.isbuy():
                    stop_price = order.executed.price * 0.98
                    self.stop_order = self.sell(
                        exectype=bt.Order.Stop,
                        price=stop_price,
                        size=order.executed.size,
                    )
            elif order == self.stop_order:
                self.stop_order = None

        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            if order == self.order:
                self.order = None
            if order == self.stop_order:
                self.stop_order = None

    def next(self):
        curr_month = self.data.datetime.date(0).month
        prev_month = (
            self.data.datetime.date(-1).month if len(self) > 1 else None
        )

        is_first_jan = (curr_month == 1) and (prev_month != 1)
        is_first_feb = (curr_month == 2) and (prev_month != 2)

        if self.order:
            return

        if not self.position:
            if is_first_jan:
                cash = self.broker.getcash()
                price = self.data.close[0]
                size = int(cash / price)
                if size > 0:
                    self.order = self.buy(size=size)
        else:
            if is_first_feb:
                if self.stop_order:
                    self.cancel(self.stop_order)
                self.order = self.close()
