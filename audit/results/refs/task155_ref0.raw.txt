import backtrader as bt


class VolatilitySpikeStrategy(bt.Strategy):
    params = dict(
        sma_fast_period=20,
        sma_slow_period=50,
        bb_period=20,
        bb_devfactor=2,
        bbwidth_sma_period=20,
        bbwidth_mult=1.3,
        atr_period=14,
        volume_sma_period=20,
        volume_mult=1.5,
        stop_atr_mult=1.0,
        target_atr_mult=2.0,
        time_exit_bars=3,
    )

    def __init__(self):
        self.sma_fast = bt.indicators.SMA(self.data.close, period=self.p.sma_fast_period)
        self.sma_slow = bt.indicators.SMA(self.data.close, period=self.p.sma_slow_period)
        self.bb = bt.indicators.BollingerBands(
            self.data.close, period=self.p.bb_period, devfactor=self.p.bb_devfactor
        )
        self.bbwidth = self.bb.lines.top - self.bb.lines.bot
        self.bbwidth_sma = bt.indicators.SMA(self.bbwidth, period=self.p.bbwidth_sma_period)
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.vol_sma = bt.indicators.SMA(self.data.volume, period=self.p.volume_sma_period)
        self.cross_up = bt.indicators.CrossOver(self.data.close, self.bb.lines.top)
        self.cross_down = bt.indicators.CrossOver(self.bb.lines.bot, self.data.close)

        self.order = None
        self.entry_bar = None
        self.stop_price = None
        self.take_price = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return
        if order.status == order.Completed:
            if self.position:
                entry_price = order.executed.price
                entry_atr = self.atr[0]
                self.entry_bar = len(self)
                if order.isbuy():
                    self.stop_price = entry_price - self.p.stop_atr_mult * entry_atr
                    self.take_price = entry_price + self.p.target_atr_mult * entry_atr
                else:
                    self.stop_price = entry_price + self.p.stop_atr_mult * entry_atr
                    self.take_price = entry_price - self.p.target_atr_mult * entry_atr
            else:
                self.entry_bar = None
                self.stop_price = None
                self.take_price = None
        self.order = None

    def next(self):
        if self.order:
            return

        if self.position:
            if self.position.size > 0:
                if self.data.low[0] <= self.stop_price or self.data.high[0] >= self.take_price:
                    self.order = self.close()
                    return
            elif self.position.size < 0:
                if self.data.high[0] >= self.stop_price or self.data.low[0] <= self.take_price:
                    self.order = self.close()
                    return

            if (len(self) - self.entry_bar) >= self.p.time_exit_bars:
                self.order = self.close()
            return

        trend_up = self.sma_fast[0] > self.sma_slow[0]
        trend_down = self.sma_fast[0] < self.sma_slow[0]
        vol_expansion = self.bbwidth[0] > self.bbwidth_sma[0] * self.p.bbwidth_mult
        volume_spike = self.data.volume[0] > self.p.volume_mult * self.vol_sma[0]

        if trend_up and vol_expansion and volume_spike and self.cross_up[0] > 0:
            self.order = self.buy()
        elif trend_down and vol_expansion and volume_spike and self.cross_down[0] < 0:
            self.order = self.sell()
