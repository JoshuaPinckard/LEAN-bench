import datetime
import backtrader as bt


class IntradayMeanReversion(bt.Strategy):
    def __init__(self):
        self.bb = bt.indicators.BollingerBands(self.data.close, period=20, devfactor=2)
        self.sma = bt.indicators.SMA(self.data.close, period=20)
        self.rsi = bt.indicators.RSI(self.data.close, period=14)
        self.atr = bt.indicators.ATR(self.data, period=14)
        self.crossover = bt.indicators.CrossOver(self.data.close, self.bb.lines.bot)

        self.order = None
        self.last_date = None
        self.no_trade_today = False
        self.tp_taken = False
        self.eod_reduced = False
        self.max_price_since_entry = None
        self.base_size = 0

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            if not self.position:
                self.max_price_since_entry = None
                self.tp_taken = False
                self.eod_reduced = False
                self.base_size = 0

        self.order = None

    def next(self):
        dt = self.data.datetime.datetime(0)
        current_date = dt.date()
        current_time = dt.time()

        # Check for start of a new day
        if current_date != self.last_date:
            self.last_date = current_date
            if len(self) > 1:
                open_bb_width = (self.bb.top[0] - self.bb.bot[0]) / self.bb.mid[0]
                prev_atr_pct = self.atr[-1] / self.data.close[-1]
                if open_bb_width > 0.015 or prev_atr_pct > 0.015:
                    self.no_trade_today = True
                else:
                    self.no_trade_today = False
            else:
                self.no_trade_today = False

        # Position management
        if self.position:
            if self.max_price_since_entry is None:
                self.max_price_since_entry = self.data.high[0]
            else:
                self.max_price_since_entry = max(self.max_price_since_entry, self.data.high[0])

            if self.order is not None:
                return

            # Stop-loss: 2% below highest price reached since entry
            if self.data.close[0] <= self.max_price_since_entry * 0.98:
                self.order = self.close()
                return

            # Take-profit: +0.3% from average entry price, close 50% of position
            if not self.tp_taken and self.data.close[0] >= self.position.price * 1.003:
                tp_size = max(1, int(self.position.size * 0.5))
                self.order = self.sell(size=tp_size)
                self.tp_taken = True
                return

            # EOD Reduction: 30-60 minutes before close (15:00 to 15:30)
            if datetime.time(15, 0) <= current_time <= datetime.time(15, 30) and not self.eod_reduced:
                target_size = int(0.5 * self.base_size)
                if self.position.size > target_size:
                    reduce_size = self.position.size - target_size
                    if reduce_size > 0:
                        self.order = self.sell(size=reduce_size)
                self.eod_reduced = True
                return

        # Entry logic
        else:
            if self.order is not None:
                return

            if self.no_trade_today:
                return

            # First 30 minutes after open (09:30 to 10:00)
            if datetime.time(9, 30) <= current_time <= datetime.time(10, 0):
                if self.crossover[0] > 0 and self.rsi[0] < 45:
                    band_width = (self.bb.top[0] - self.bb.bot[0]) / self.bb.mid[0]
                    atr_pct = self.atr[0] / self.data.close[0]

                    if band_width <= 0.005 or atr_pct <= 0.008:
                        capital = self.broker.getvalue()
                        base_size = int(capital / self.data.close[0])
                        leverage = 1.25 if atr_pct > 0.01 else 1.50
                        target_size = int((capital * leverage) / self.data.close[0])

                        if target_size > 0:
                            self.base_size = base_size
                            self.max_price_since_entry = self.data.high[0]
                            self.tp_taken = False
                            self.eod_reduced = False
                            self.order = self.buy(size=target_size)
