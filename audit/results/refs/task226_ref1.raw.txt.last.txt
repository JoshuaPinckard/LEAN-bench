import backtrader as bt


class ImpulseStrategy12_1(bt.Strategy):
    params = (
        ('risk_fraction', 0.10),
        ('stop_loss', 0.02),
        ('take_profit', 0.05),
    )

    def __init__(self):
        self.sma = bt.indicators.SMA(self.data.close, period=252)
        self.atr_percent = bt.indicators.SMA(
            bt.indicators.ATR(self.data, period=14),
            period=252
        ) / self.data.close

        self.order = None
        self.stop_order = None
        self.take_profit_order = None

    def next(self):
        if self.order:
            return

        if not self.position:
            if self.data.close[0] > self.sma[0] and self.atr_percent[0] <= 0.05:
                capital_to_invest = self.broker.getvalue() * self.p.risk_fraction
                size = int(capital_to_invest / self.data.close[0])
                if size > 0:
                    self.order = self.buy(size=size)
            return

        if self.data.close[0] < self.sma[0]:
            if self.stop_order:
                self.cancel(self.stop_order)
            if self.take_profit_order:
                self.cancel(self.take_profit_order)
            self.order = self.close()

    def notify_order(self, order):
        if order.status in [order.Completed]:
            if order.isbuy():
                entry_price = order.executed.price
                self.stop_order = self.sell(
                    size=order.executed.size,
                    exectype=bt.Order.Stop,
                    price=entry_price * (1.0 - self.p.stop_loss)
                )
                self.take_profit_order = self.sell(
                    size=order.executed.size,
                    exectype=bt.Order.Limit,
                    price=entry_price * (1.0 + self.p.take_profit_order)
                )

            elif order == self.stop_order:
                if self.take_profit_order:
                    self.cancel(self.take_profit_order)
                self.stop_order = None
                self.take_profit_order = None

            elif order == self.take_profit_order:
                if self.stop_order:
                    self.cancel(self.stop_order)
                self.stop_order = None
                self.take_profit_order = None

        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            if order == self.order:
                self.order = None
