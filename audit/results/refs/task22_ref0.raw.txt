import backtrader as bt


class TrendRSIFilterHardStopStrategy(bt.Strategy):
    params = (
        ('rsi_period', 14),
        ('rsi_oversold', 30.0),
        ('rsi_overbought', 70.0),
        ('stop_loss_pct', 0.001),
        ('take_profit_pct', 0.01),
        ('risk_pct', 0.01),
        ('pip', 0.0001),
        ('spread_pips', 1),
    )

    def __init__(self):
        self.spread = self.p.pip * self.p.spread_pips

        self.current_day = None
        self.day_last_close = None
        self.completed_daily_closes = []
        self.avg_gain = None
        self.avg_loss = None
        self.daily_rsi = None
        self.pending_signal = None

        self.order = None
        self.stop_order = None
        self.limit_order = None
        self.entry_price = None
        self.trade_direction = None

    def _finalize_day(self, close_price):
        self.completed_daily_closes.append(close_price)
        n = len(self.completed_daily_closes)
        period = self.p.rsi_period

        if n < period + 1:
            return
        elif n == period + 1:
            gains = []
            losses = []
            closes = self.completed_daily_closes
            for i in range(1, n):
                diff = closes[i] - closes[i - 1]
                gains.append(diff if diff > 0 else 0.0)
                losses.append(-diff if diff < 0 else 0.0)
            self.avg_gain = sum(gains) / period
            self.avg_loss = sum(losses) / period
        else:
            diff = close_price - self.completed_daily_closes[-2]
            gain = diff if diff > 0 else 0.0
            loss = -diff if diff < 0 else 0.0
            self.avg_gain = (self.avg_gain * (period - 1) + gain) / period
            self.avg_loss = (self.avg_loss * (period - 1) + loss) / period

        if self.avg_loss == 0:
            self.daily_rsi = 100.0
        elif self.avg_gain == 0:
            self.daily_rsi = 0.0
        else:
            rs = self.avg_gain / self.avg_loss
            self.daily_rsi = 100.0 - (100.0 / (1.0 + rs))

        if self.daily_rsi < self.p.rsi_oversold:
            self.pending_signal = 'long'
        elif self.daily_rsi > self.p.rsi_overbought:
            self.pending_signal = 'short'
        else:
            self.pending_signal = None

    def _position_size(self, reference_price):
        capital = self.broker.getvalue()
        risk_amount = capital * self.p.risk_pct
        sl_distance = reference_price * self.p.stop_loss_pct
        return risk_amount / sl_distance

    def next(self):
        dt = self.data.datetime.date(0)
        close = self.data.close[0]

        if self.current_day is None:
            self.current_day = dt
            self.day_last_close = close
            return

        if dt != self.current_day:
            self._finalize_day(self.day_last_close)
            self.current_day = dt
            self.day_last_close = close

            if (self.pending_signal is not None and not self.position
                    and self.order is None):
                size = self._position_size(close)
                if self.pending_signal == 'long':
                    self.trade_direction = 'long'
                    self.order = self.buy(size=size)
                elif self.pending_signal == 'short':
                    self.trade_direction = 'short'
                    self.order = self.sell(size=size)
            self.pending_signal = None
        else:
            self.day_last_close = close

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status == order.Completed:
            if order == self.order:
                self.entry_price = order.executed.price
                size = abs(order.executed.size)
                if self.trade_direction == 'long':
                    sl_price = self.entry_price * (1 - self.p.stop_loss_pct)
                    tp_price = self.entry_price * (1 + self.p.take_profit_pct) + self.spread
                    self.stop_order = self.sell(exectype=bt.Order.Stop, price=sl_price, size=size)
                    self.limit_order = self.sell(exectype=bt.Order.Limit, price=tp_price, size=size,
                                                  oco=self.stop_order)
                else:
                    sl_price = self.entry_price * (1 + self.p.stop_loss_pct)
                    tp_price = self.entry_price * (1 - self.p.take_profit_pct) - self.spread
                    self.stop_order = self.buy(exectype=bt.Order.Stop, price=sl_price, size=size)
                    self.limit_order = self.buy(exectype=bt.Order.Limit, price=tp_price, size=size,
                                                 oco=self.stop_order)
                self.order = None
            elif order in (self.stop_order, self.limit_order):
                self.stop_order = None
                self.limit_order = None
                self.entry_price = None
                self.trade_direction = None

        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            if order == self.order:
                self.order = None
            elif order == self.stop_order:
                self.stop_order = None
            elif order == self.limit_order:
                self.limit_order = None
