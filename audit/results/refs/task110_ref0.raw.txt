import backtrader as bt


class RSIReversalStrategy(bt.Strategy):
    params = (
        ('rsi_period', 14),
        ('rsi_lower', 30),
        ('rsi_upper', 70),
    )

    def __init__(self):
        self.rsi = bt.indicators.RSI(self.data, period=self.p.rsi_period)
        self.cross_lower = bt.indicators.CrossOver(self.rsi, self.p.rsi_lower)
        self.cross_upper = bt.indicators.CrossOver(self.rsi, self.p.rsi_upper)
        self.order = None

    def notify_order(self, order):
        if order.status in (order.Completed, order.Canceled, order.Margin, order.Rejected):
            self.order = None

    def next(self):
        if self.order:
            return

        if not self.position:
            if self.cross_lower > 0:
                self.order = self.buy()
        else:
            if self.cross_upper < 0:
                self.order = self.close()
