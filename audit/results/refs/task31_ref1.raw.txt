import backtrader as bt
import math


class Trading0DTEOpeningRangeStrategy(bt.Strategy):
    params = (
        ("risk_fraction", 0.01),
        ("stop_offset_fraction", 0.001),
    )

    def __init__(self):
        self.rsi = bt.indicators.RSI(self.data.close, period=14)
        self.order = None
        self.stop_order = None
        self.current_date = None
        self.upper_channel = None
        self.lower_channel = None
        self.range_complete = False

    def prenext(self):
        self._process_bar()

    def next(self):
        self._process_bar()

    def _reset_day(self, bar_date):
        self.current_date = bar_date
        self.upper_channel = None
        self.lower_channel = None
        self.range_complete = False

    def _process_bar(self):
        bar_date = self.data.datetime.date(0)
        bar_time = self.data.datetime.time(0)

        if bar_date != self.current_date:
            self._reset_day(bar_date)

        hour = bar_time.hour
        minute = bar_time.minute
        minutes = hour * 60 + minute

        range_start = 9 * 60 + 30
        range_end = 10 * 60 + 10
        final_entry_time = 15 * 60 + 50

        if range_start <= minutes < range_end:
            high = float(self.data.high[0])
            low = float(self.data.low[0])

            if self.upper_channel is None or high > self.upper_channel:
                self.upper_channel = high
            if self.lower_channel is None or low < self.lower_channel:
                self.lower_channel = low
            return

        if self.upper_channel is None or self.lower_channel is None:
            return

        if minutes >= range_end:
            self.range_complete = True

        if not self.range_complete:
            return

        if self.position and minutes == final_entry_time and self.order is None:
            self._cancel_stop()
            self.order = self.close(exectype=bt.Order.Close)
            return

        if self.position and self.order is None:
            close = float(self.data.close[0])

            if ((self.position.size > 0 and close <= self.upper_channel) or
                    (self.position.size < 0 and close >= self.lower_channel)):
                self._cancel_stop()
                self.order = self.close()
            return

        if self.position or self.order is not None or minutes >= final_entry_time:
            return

        if len(self.data) < 2:
            return

        close = float(self.data.close[0])
        rsi = float(self.rsi[0])
        volume_increasing = self.data.volume[0] > self.data.volume[-1]

        if math.isnan(rsi):
            return

        if (close > self.upper_channel and
                (volume_increasing or rsi > 50) and
                rsi <= 70):
            stop_price = self.lower_channel * (1 - self.p.stop_offset_fraction)
            size = self._risk_size(close, stop_price)
            if size > 0:
                self.order = self.buy(size=size)

        elif (close < self.lower_channel and
                (volume_increasing or rsi < 50) and
                rsi >= 30):
            stop_price = self.upper_channel * (1 + self.p.stop_offset_fraction)
            size = self._risk_size(close, stop_price)
            if size > 0:
                self.order = self.sell(size=size)

    def _risk_size(self, entry_price, stop_price):
        risk_per_share = abs(entry_price - stop_price)
        if risk_per_share <= 0:
            return 0

        risk_capital = self.broker.getvalue() * self.p.risk_fraction
        return int(risk_capital / risk_per_share)

    def _cancel_stop(self):
        if self.stop_order is not None:
            self.cancel(self.stop_order)
            self.stop_order = None

    def notify_order(self, order):
        if order.status in (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            if order is self.order:
                self.order = None

            if order is self.stop_order:
                self.stop_order = None

        if order.status != order.Completed:
            return

        if order.isbuy() and self.position.size > 0 and self.stop_order is None:
            stop_price = self.lower_channel * (1 - self.p.stop_offset_fraction)
            self.stop_order = self.sell(
                size=abs(self.position.size),
                exectype=bt.Order.Stop,
                price=stop_price,
            )

        elif order.issell() and self.position.size < 0 and self.stop_order is None:
            stop_price = self.upper_channel * (1 + self.p.stop_offset_fraction)
            self.stop_order = self.buy(
                size=abs(self.position.size),
                exectype=bt.Order.Stop,
                price=stop_price,
            )
