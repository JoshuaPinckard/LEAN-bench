import backtrader as bt


class EMANeuronMeanReversion(bt.Strategy):
    def __init__(self):
        self.ema20 = bt.indicators.EMA(self.data.close, period=20)
        self.ema_mean = bt.indicators.SMA(self.ema20, period=20)
        self.ema_std = bt.indicators.StdDev(self.ema20, period=20)
        self.upper_band = self.ema_mean + 2.0 * self.ema_std
        self.lower_band = self.ema_mean - 2.0 * self.ema_std

        self.pending_order = None
        self.stop_order = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            if order == self.pending_order:
                self.pending_order = None
                entry_price = order.executed.price
                if self.position.size > 0:
                    stop_price = entry_price * 0.98
                    self.stop_order = self.sell(
                        exectype=bt.Order.Stop, price=stop_price
                    )
                elif self.position.size < 0:
                    stop_price = entry_price * 1.02
                    self.stop_order = self.buy(
                        exectype=bt.Order.Stop, price=stop_price
                    )
            elif order == self.stop_order:
                self.stop_order = None

        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            if order == self.pending_order:
                self.pending_order = None
            if order == self.stop_order:
                self.stop_order = None

    def next(self):
        if self.pending_order:
            return

        if not self.position:
            # Long setup: EMA20 was below lower_band on prior bar and crosses back above on current bar
            if (
                self.ema20[-1] < self.lower_band[-1]
                and self.ema20[0] >= self.lower_band[0]
            ):
                self.pending_order = self.buy()
            # Short setup: EMA20 was above upper_band on prior bar and crosses back below on current bar
            elif (
                self.ema20[-1] > self.upper_band[-1]
                and self.ema20[0] <= self.upper_band[0]
            ):
                self.pending_order = self.sell()
        else:
            if self.position.size > 0:
                # Exit long when EMA20 crosses EMA_mean
                if (
                    self.ema20[-1] < self.ema_mean[-1]
                    and self.ema20[0] >= self.ema_mean[0]
                ):
                    if self.stop_order:
                        self.cancel(self.stop_order)
                    self.pending_order = self.close()
            elif self.position.size < 0:
                # Exit short when EMA20 crosses EMA_mean
                if (
                    self.ema20[-1] > self.ema_mean[-1]
                    and self.ema20[0] <= self.ema_mean[0]
                ):
                    if self.stop_order:
                        self.cancel(self.stop_order)
                    self.pending_order = self.close()
