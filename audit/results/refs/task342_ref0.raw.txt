import math

import backtrader as bt


class StochasticFullReversalStrategy(bt.Strategy):
    params = (
        ('stoch_period', 14),
        ('period_dfast', 3),
        ('period_dslow', 3),
        ('atr_period', 14),
        ('oversold', 20),
        ('overbought', 80),
        ('exit_level', 50),
        ('stop_pct', 0.02),
        ('risk_per_trade', 0.01),
    )

    def __init__(self):
        self.stoch = bt.indicators.StochasticFull(
            self.data,
            period=self.p.stoch_period,
            period_dfast=self.p.period_dfast,
            period_dslow=self.p.period_dslow,
        )
        self.percK = self.stoch.percK
        self.percD = self.stoch.percD
        self.crossover = bt.indicators.CrossOver(self.percK, self.percD)
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)

        self.order = None
        self.order_action = None
        self.entry_price = None
        self.stop_price = None
        self.pending_direction = None

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return

        if order.status == order.Completed:
            if self.order_action == 'enter_long':
                self.entry_price = order.executed.price
                self.stop_price = self.entry_price * (1 - self.p.stop_pct)
            elif self.order_action == 'enter_short':
                self.entry_price = order.executed.price
                self.stop_price = self.entry_price * (1 + self.p.stop_pct)
            elif self.order_action == 'close':
                self.entry_price = None
                self.stop_price = None

        self.order = None
        self.order_action = None

    def _enter_long(self):
        price_est = self.data.close[0]
        stop_distance = price_est * self.p.stop_pct
        equity = self.broker.getvalue()
        size = math.floor((equity * self.p.risk_per_trade) / stop_distance)
        if size > 0:
            self.order = self.buy(size=size)
            self.order_action = 'enter_long'

    def _enter_short(self):
        price_est = self.data.close[0]
        stop_distance = price_est * self.p.stop_pct
        equity = self.broker.getvalue()
        size = math.floor((equity * self.p.risk_per_trade) / stop_distance)
        if size > 0:
            self.order = self.sell(size=size)
            self.order_action = 'enter_short'

    def next(self):
        if self.order:
            return

        k0 = self.percK[0]
        k1 = self.percK[-1]
        cross = self.crossover[0]

        long_signal = cross > 0 and k1 < self.p.oversold
        short_signal = cross < 0 and k1 > self.p.overbought

        if self.position:
            if self.position.size > 0 and self.data.low[0] <= self.stop_price:
                self.order = self.close()
                self.order_action = 'close'
                return
            if self.position.size < 0 and self.data.high[0] >= self.stop_price:
                self.order = self.close()
                self.order_action = 'close'
                return

            crossed_down_50 = k1 >= self.p.exit_level and k0 < self.p.exit_level
            crossed_up_50 = k1 <= self.p.exit_level and k0 > self.p.exit_level

            if self.position.size > 0 and crossed_down_50:
                self.order = self.close()
                self.order_action = 'close'
                return
            if self.position.size < 0 and crossed_up_50:
                self.order = self.close()
                self.order_action = 'close'
                return

            if self.position.size > 0 and short_signal:
                self.order = self.close()
                self.order_action = 'close'
                self.pending_direction = 'short'
                return
            if self.position.size < 0 and long_signal:
                self.order = self.close()
                self.order_action = 'close'
                self.pending_direction = 'long'
                return
        else:
            # Opposite-signal closes remember the intended new direction; the
            # crossover condition is a single-bar event and won't re-trigger here.
            if self.pending_direction == 'long':
                self.pending_direction = None
                self._enter_long()
                return
            if self.pending_direction == 'short':
                self.pending_direction = None
                self._enter_short()
                return

            if long_signal:
                self._enter_long()
            elif short_signal:
                self._enter_short()
