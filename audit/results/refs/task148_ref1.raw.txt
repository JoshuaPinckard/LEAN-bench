import backtrader as bt


class ConvertibleBondConversionStrategy(bt.Strategy):
    params = (
        ('sma_period', 50),
        ('rsi_period', 14),
        ('capital_fraction', 0.02),
    )

    def __init__(self):
        self.sma = bt.indicators.SimpleMovingAverage(
            self.data.close, period=self.p.sma_period
        )
        self.rsi = bt.indicators.RSI(self.data.close, period=self.p.rsi_period)

        self.order = None
        self.stop_order = None
        self.entry_price = None
        self.entry_bar = None

    def notify_order(self, order):
        if order.status == order.Completed:
            if order.isbuy() and self.position.size < 0 and order == self.stop_order:
                self.stop_order = None
                self.entry_price = None
                self.entry_bar = None
            elif order.issell() and self.position.size < 0:
                self.entry_price = order.executed.price
                self.entry_bar = len(self)
                self.stop_order = self.buy(
                    exectype=bt.Order.Stop,
                    price=self.entry_price * 1.05,
                    size=abs(self.position.size),
                )

            if order == self.order:
                self.order = None

        elif order.status in (
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            if order == self.order:
                self.order = None
            if order == self.stop_order:
                self.stop_order = None

    def next(self):
        if self.order:
            return

        if self.position:
            exit_signal = (
                self.data.close[0] <= 160.00
                or self.data.close[0] > self.sma[0]
                or (self.entry_bar is not None and len(self) - self.entry_bar >= 5)
            )

            if exit_signal:
                if self.stop_order:
                    self.cancel(self.stop_order)
                self.order = self.close()
            return

        current_date = self.data.datetime.date(0)
        on_or_after_october_1 = (current_date.month, current_date.day) >= (10, 1)

        if (
            on_or_after_october_1
            and self.data.close[-1] > 200.00
            and self.rsi[-1] > 50
            and self.data.open[0] < self.data.close[-1]
            and self.data.open[0] >= 160.00
        ):
            size = int((self.broker.getvalue() * self.p.capital_fraction) / self.data.open[0])
            if size > 0:
                self.order = self.sell(size=size)
