import backtrader as bt


class AbsoluteMomentumStrategy(bt.Strategy):
    params = (
        ('period', 252),
        ('threshold', 100.0),
        ('cash_pct', 0.95),
    )

    def __init__(self):
        self.mom = bt.indicators.MomentumOscillator(
            self.data.close, period=self.p.period, band=self.p.threshold
        )
        self.order = None

    def notify_order(self, order):
        if order.status in (order.Completed, order.Canceled, order.Margin, order.Rejected):
            self.order = None

    def next(self):
        if self.order:
            return

        crossed_up = self.mom[0] > self.p.threshold and self.mom[-1] <= self.p.threshold
        crossed_down = self.mom[0] < self.p.threshold and self.mom[-1] >= self.p.threshold

        if not self.position:
            if crossed_up:
                cash = self.broker.getcash()
                price = self.data.close[0]
                size = int((cash * self.p.cash_pct) / price)
                if size > 0:
                    self.order = self.buy(size=size)
        else:
            if crossed_down:
                self.order = self.close()
