import backtrader as bt


class SmaRegimeStrategy(bt.Strategy):
    params = (("period", 200),)

    def log(self, txt, dt=None):
        dt = dt or self.data.datetime.date(0)
        print(f"{dt.isoformat()}, {txt}")

    def __init__(self):
        self.sma = bt.indicators.SimpleMovingAverage(
            self.data.close, period=self.p.period
        )
        self.order = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            if order.isbuy():
                self.log(
                    f"BUY EXECUTED, Price: {order.executed.price:.2f}, Cost: {order.executed.value:.2f}, Comm: {order.executed.comm:.2f}"
                )
            else:
                self.log(
                    f"SELL EXECUTED, Price: {order.executed.price:.2f}, Cost: {order.executed.value:.2f}, Comm: {order.executed.comm:.2f}"
                )
        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            self.log(
                f"Order Canceled/Margin/Rejected: Status {order.getstatusname()}"
            )

        self.order = None

    def next(self):
        self.log(
            f"Portfolio Value: {self.broker.getvalue():.2f}, Position Size: {self.position.size}"
        )

        if self.order:
            return

        # Check once a week at Friday close
        if self.data.datetime.date(0).weekday() == 4:
            if not self.position and self.data.close[0] > self.sma[0]:
                cash = self.broker.getcash()
                price = self.data.close[0]
                comm_info = self.broker.getcommissioninfo(self.data)
                comm_rate = getattr(comm_info, "commission", 0.001) or 0.001
                size = int(cash / (price * (1.0 + comm_rate)))
                if size > 0:
                    self.order = self.buy(size=size)
                    self.log(f"BUY CREATE, Price: {price:.2f}, Size: {size}")
            elif self.position and self.data.close[0] < self.sma[0]:
                self.order = self.close()
                self.log(
                    f"SELL CREATE, Price: {self.data.close[0]:.2f}, Size: {self.position.size}"
                )
