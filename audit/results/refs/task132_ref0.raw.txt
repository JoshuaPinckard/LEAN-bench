import backtrader as bt
import collections


class DailySmaLimitOcoStrategy(bt.Strategy):
    params = (
        ('sma_period', 20),
        ('tp_pct', 0.02),
        ('sl_pct', 0.015),
    )

    def __init__(self):
        self.daily_closes = collections.deque(maxlen=self.p.sma_period)
        self.current_date = None
        self.last_close = None
        self.prev_close = None
        self.prev_sma = None
        self.entry_order = None
        self.tp_order = None
        self.sl_order = None

    def _finalize_day(self, close_price):
        self.daily_closes.append(close_price)
        sma = None
        if len(self.daily_closes) >= self.p.sma_period:
            sma = sum(self.daily_closes) / self.p.sma_period

        signal = None
        if self.prev_close is not None and self.prev_sma is not None and sma is not None:
            if self.prev_close <= self.prev_sma and close_price > sma:
                signal = 'buy'
            elif self.prev_close >= self.prev_sma and close_price < sma:
                signal = 'sell'

        self.prev_close = close_price
        self.prev_sma = sma
        return signal

    def next(self):
        dt = self.data.datetime.date(0)
        o = self.data.open[0]
        c = self.data.close[0]

        if self.current_date is None:
            self.current_date = dt
        elif dt != self.current_date:
            if self.entry_order is not None:
                self.cancel(self.entry_order)
                self.entry_order = None

            close_of_prev_day = self.last_close
            signal = self._finalize_day(close_of_prev_day)
            self.current_date = dt

            if signal is not None and not self.position and self.entry_order is None:
                if signal == 'buy':
                    self.entry_order = self.buy(exectype=bt.Order.Limit, price=close_of_prev_day)
                else:
                    self.entry_order = self.sell(exectype=bt.Order.Limit, price=o)

        self.last_close = c

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return

        if order.status == order.Completed:
            if order == self.entry_order:
                self.entry_order = None
                entry_price = order.executed.price
                if order.isbuy():
                    tp_price = entry_price * (1.0 + self.p.tp_pct)
                    sl_price = entry_price * (1.0 - self.p.sl_pct)
                    tp = self.sell(exectype=bt.Order.Limit, price=tp_price)
                    sl = self.sell(exectype=bt.Order.Stop, price=sl_price, oco=tp)
                else:
                    tp_price = entry_price * (1.0 - self.p.tp_pct)
                    sl_price = entry_price * (1.0 + self.p.sl_pct)
                    tp = self.buy(exectype=bt.Order.Limit, price=tp_price)
                    sl = self.buy(exectype=bt.Order.Stop, price=sl_price, oco=tp)
                self.tp_order = tp
                self.sl_order = sl
            elif order == self.tp_order:
                self.tp_order = None
            elif order == self.sl_order:
                self.sl_order = None
        elif order.status in (order.Canceled, order.Margin, order.Rejected):
            if order == self.entry_order:
                self.entry_order = None
            elif order == self.tp_order:
                self.tp_order = None
            elif order == self.sl_order:
                self.sl_order = None
