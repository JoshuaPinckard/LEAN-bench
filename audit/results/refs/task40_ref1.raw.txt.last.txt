import backtrader as bt
import math


class PowerHourReversal(bt.Strategy):
    params = (
        ('rsi_period', 14),
        ('ema_period', 20),
        ('take_profit', 0.015),
        ('stop_loss', 0.008),
        ('volatility_threshold', 0.015),
        ('capital_fraction', 0.02),
    )

    def __init__(self):
        self.rsi = bt.indicators.RSI(self.data.close, period=self.p.rsi_period)
        self.ema = bt.indicators.EMA(self.data.close, period=self.p.ema_period)

        self.order = None
        self.entry_price = None
        self.current_date = None
        self.daily_high = None
        self.daily_low = None

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            if order.status == order.Completed:
                if order.info.get('entry', False):
                    self.entry_price = order.executed.price
                elif not self.position:
                    self.entry_price = None

            self.order = None

    def next(self):
        current_date = self.data.datetime.date(0)
        if current_date != self.current_date:
            self.current_date = current_date
            self.daily_high = self.data.high[0]
            self.daily_low = self.data.low[0]
        else:
            self.daily_high = max(self.daily_high, self.data.high[0])
            self.daily_low = min(self.daily_low, self.data.low[0])

        if self.order:
            return

        hour = self.data.datetime.time(0).hour
        minute = self.data.datetime.time(0).minute
        in_last_hour = hour == 15
        session_end = hour == 15 and minute == 30

        if self.position:
            if session_end:
                self.order = self.close()
                return

            if self.position.size > 0:
                if (self.data.close[0] >= self.entry_price * (1 + self.p.take_profit) or
                        self.data.close[0] <= self.entry_price * (1 - self.p.stop_loss)):
                    self.order = self.close()
            else:
                if (self.data.close[0] <= self.entry_price * (1 - self.p.take_profit) or
                        self.data.close[0] >= self.entry_price * (1 + self.p.stop_loss)):
                    self.order = self.close()
            return

        if not in_last_hour:
            return

        volatility = (self.daily_high - self.daily_low) / self.daily_low
        if volatility <= self.p.volatility_threshold:
            return

        size = math.floor(
            (self.broker.getvalue() * self.p.capital_fraction) / self.data.close[0]
        )
        if size <= 0:
            return

        if self.data.close[0] < self.ema[0] and self.rsi[-1] < 30 and self.rsi[0] > 30:
            self.order = self.buy(size=size)
            self.order.addinfo(entry=True)
        elif self.data.close[0] > self.ema[0] and self.rsi[-1] > 70 and self.rsi[0] < 70:
            self.order = self.sell(size=size)
            self.order.addinfo(entry=True)
