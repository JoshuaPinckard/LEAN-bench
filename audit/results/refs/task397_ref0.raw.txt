import backtrader as bt


class WilliamsRBandStrategy(bt.Strategy):
    params = (
        ('wr_period', 14),
        ('band_period', 20),
        ('band_mult', 2.0),
        ('stop_pct', 1.0),
    )

    def __init__(self):
        self.wr = bt.indicators.WilliamsR(self.data, period=self.p.wr_period)
        self.wr_mean = bt.indicators.SMA(self.wr, period=self.p.band_period)
        self.wr_std = bt.indicators.StdDev(self.wr, period=self.p.band_period)
        self.upper_band = self.wr_mean + self.p.band_mult * self.wr_std
        self.lower_band = self.wr_mean - self.p.band_mult * self.wr_std
        self.order = None

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        if self.order:
            return

        wr0, wr1 = self.wr[0], self.wr[-1]
        mean0, mean1 = self.wr_mean[0], self.wr_mean[-1]
        upper0, upper1 = self.upper_band[0], self.upper_band[-1]
        lower0, lower1 = self.lower_band[0], self.lower_band[-1]

        cross_up_upper = wr1 <= upper1 and wr0 > upper0
        cross_down_mean = wr1 >= mean1 and wr0 < mean0
        cross_down_lower = wr1 >= lower1 and wr0 < lower0
        cross_up_mean = wr1 <= mean1 and wr0 > mean0

        if not self.position:
            if cross_up_upper:
                self.order = self.buy()
            elif cross_down_lower:
                self.order = self.sell()
        elif self.position.size > 0:
            stop_price = self.position.price * (1 - self.p.stop_pct / 100.0)
            if cross_down_mean or self.data.low[0] <= stop_price:
                self.order = self.close()
        elif self.position.size < 0:
            stop_price = self.position.price * (1 + self.p.stop_pct / 100.0)
            if cross_up_mean or self.data.high[0] >= stop_price:
                self.order = self.close()
