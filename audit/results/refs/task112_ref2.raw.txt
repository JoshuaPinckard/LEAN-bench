import backtrader as bt


class SOFIMeanReversionFlag(bt.Strategy):
    params = (
        ('ema_short', 10),
        ('ema_medium', 20),
        ('within_high_pct', 0.01),
        ('stop_loss_pct', 0.015),
        ('take_profit_pct', 0.04),
        ('risk_pct', 0.015),
        ('key_level_vrvp', 26.0),
    )

    def __init__(self):
        self.ema10 = bt.indicators.EMA(self.data.close, period=self.params.ema_short)
        self.ema20 = bt.indicators.EMA(self.data.close, period=self.params.ema_medium)
        self.order = None
        self.entry_price = None
        self.highest_high = 0.0

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return
        if order.status in [order.Completed]:
            if order.isbuy():
                self.entry_price = order.executed.price
            elif order.issell():
                self.entry_price = None
        self.order = None

    def next(self):
        if self.data.high[0] > self.highest_high:
            self.highest_high = self.data.high[0]

        if self.order:
            return

        if self.position:
            if self.entry_price is not None:
                if self.data.low[0] <= self.entry_price * (1.0 - self.params.stop_loss_pct):
                    self.order = self.close()
                    return
                if self.data.high[0] >= self.entry_price * (1.0 + self.params.take_profit_pct):
                    self.order = self.close()
                    return

            is_last_bar_of_day = False
            try:
                if self.data.datetime.date(0) != self.data.datetime.date(1):
                    is_last_bar_of_day = True
            except IndexError:
                is_last_bar_of_day = True

            if is_last_bar_of_day:
                if (self.data.close[0] < self.ema10[0]) or (self.data.close[0] < self.ema20[0]):
                    self.order = self.close()
                    return
        else:
            trend_ok = (self.data.close[0] > self.ema10[0]) and (self.data.close[0] > self.ema20[0])
            position_ok = (self.highest_high > 0) and (self.data.close[0] >= self.highest_high * (1.0 - self.params.within_high_pct))
            vrvp_ok = self.data.close[0] >= self.params.key_level_vrvp

            if not (trend_ok and position_ok and vrvp_ok):
                return

            ema_rising = (len(self) > 1) and ((self.ema10[0] > self.ema10[-1]) or (self.ema20[0] > self.ema20[-1]))
            pullback = (self.data.low[0] <= self.ema10[0]) or (self.data.low[0] <= self.ema20[0])
            reversing = self.data.close[0] > self.data.open[0]
            option_a = ema_rising and pullback and reversing

            is_first_bar = (len(self) == 1) or (self.data.datetime.date(0) != self.data.datetime.date(-1))
            option_b = False
            if not is_first_bar and len(self) >= 4:
                flag_upper = max(self.data.high[-1], self.data.high[-2], self.data.high[-3])
                if self.data.close[0] > flag_upper:
                    option_b = True

            if option_a or option_b:
                risk_amount = self.broker.getvalue() * self.params.risk_pct
                risk_per_share = self.data.close[0] * self.params.stop_loss_pct
                if risk_per_share > 0:
                    size = int(risk_amount / risk_per_share)
                else:
                    size = 1
                size = max(1, size)
                self.order = self.buy(size=size)
