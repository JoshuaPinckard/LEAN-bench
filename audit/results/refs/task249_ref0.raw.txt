import backtrader as bt


class IchimokuStyleSMAMomentum(bt.Strategy):
    params = dict(
        tenkan_period=9,   # SMA analogue of Tenkan
        kijun_period=26,   # SMA analogue of Kijun
        senkou_period=52,  # SMA analogue of long-term cloud support
    )

    def __init__(self):
        # Indicators named by the spec: SMA(9), SMA(26), SMA(52), CrossOver
        self.sma9 = bt.indicators.SMA(self.data.close, period=self.p.tenkan_period)
        self.sma26 = bt.indicators.SMA(self.data.close, period=self.p.kijun_period)
        self.sma52 = bt.indicators.SMA(self.data.close, period=self.p.senkou_period)
        # CrossOver of SMA(9) vs SMA(26): +1 = 9 crosses above 26, -1 = 9 crosses below 26
        self.crossover = bt.indicators.CrossOver(self.sma9, self.sma26)

        self.order = None

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return
        if order.status in (order.Completed, order.Canceled, order.Margin, order.Rejected):
            self.order = None

    def _full_capital_size(self):
        # 100% of available capital per trade, sized off the confirmed close,
        # leaving room for commission so the next-open fill is not margin-rejected.
        price = self.data.close[0]
        if price <= 0:
            return 0
        cash = self.broker.get_cash()
        comm = self.broker.getcommissioninfo(self.data).p.commission
        return int(cash / (price * (1.0 + comm)))

    def next(self):
        if self.order:
            return

        close = self.data.close[0]
        sma26 = self.sma26[0]
        sma52 = self.sma52[0]
        cross = self.crossover[0]

        if self.position:
            if self.position.size > 0:
                # Exit long: reverse cross (9 below 26) OR close below lower of SMA26/SMA52
                if cross < 0 or close < min(sma26, sma52):
                    self.order = self.close()
            elif self.position.size < 0:
                # Exit short: reverse cross (9 above 26) OR close above higher of SMA26/SMA52
                if cross > 0 or close > max(sma26, sma52):
                    self.order = self.close()
        else:
            # Enter long: 9 crosses above 26, close above both SMA26 & SMA52, SMA26 > SMA52
            if cross > 0 and close > sma26 and close > sma52 and sma26 > sma52:
                size = self._full_capital_size()
                if size > 0:
                    self.order = self.buy(size=size)
            # Enter short: 9 crosses below 26, close below both SMA26 & SMA52, SMA26 < SMA52
            elif cross < 0 and close < sma26 and close < sma52 and sma26 < sma52:
                size = self._full_capital_size()
                if size > 0:
                    self.order = self.sell(size=size)
