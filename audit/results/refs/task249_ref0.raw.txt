import backtrader as bt


class MomentumIchimokuSMAStrategy(bt.Strategy):
    def __init__(self):
        self.tenkan = bt.indicators.SMA(self.data.close, period=9)
        self.kijun = bt.indicators.SMA(self.data.close, period=26)
        self.senkou = bt.indicators.SMA(self.data.close, period=52)
        self.crossover = bt.indicators.CrossOver(self.tenkan, self.kijun)
        self.order = None

    def notify_order(self, order):
        if order.status in (order.Completed, order.Canceled, order.Margin, order.Rejected):
            self.order = None

    def next(self):
        if self.order:
            return

        close = self.data.close[0]
        kijun = self.kijun[0]
        senkou = self.senkou[0]

        if not self.position:
            if (self.crossover[0] > 0 and close > kijun and close > senkou
                    and kijun > senkou):
                cash = self.broker.getcash()
                size = int(cash / close)
                if size > 0:
                    self.order = self.buy(size=size)
            elif (self.crossover[0] < 0 and close < kijun and close < senkou
                    and kijun < senkou):
                cash = self.broker.getcash()
                size = int(cash / close)
                if size > 0:
                    self.order = self.sell(size=size)
        elif self.position.size > 0:
            if self.crossover[0] < 0 or close < min(kijun, senkou):
                self.order = self.close()
        elif self.position.size < 0:
            if self.crossover[0] > 0 or close > max(kijun, senkou):
                self.order = self.close()
