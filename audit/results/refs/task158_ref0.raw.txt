import backtrader as bt


class RiskParityVolTargetStrategy(bt.Strategy):
    params = dict(
        vtgt=0.01,
        atr_period=252,
        sma_period=252,
        w_min=0.01,
        w_max=0.10,
    )

    def __init__(self):
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.sma = bt.indicators.SMA(self.data.close, period=self.p.sma_period)
        self.order = None

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        if self.order:
            return

        dt = self.data.datetime.date(0)
        if len(self.data) >= self.data.buflen():
            month_end = True
        else:
            month_end = self.data.datetime.date(1).month != dt.month

        if not month_end:
            return

        vol = self.atr[0] / self.sma[0]
        w_star = self.p.vtgt / vol
        w = min(max(w_star, self.p.w_min), self.p.w_max)

        self.order = self.order_target_percent(target=w)
