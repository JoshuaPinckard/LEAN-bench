import math
import backtrader as bt


class TechnicalPatternStrategy(bt.Strategy):
    def __init__(self):
        self.sma20 = bt.indicators.SMA(self.data.close, period=20)
        self.vol_sma20 = bt.indicators.SMA(self.data.volume, period=20)

        self.order = None
        self.sl = None
        self.tp = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return
        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            self.order = None

    def next(self):
        if len(self.data) < 30:
            return

        if self.order:
            return

        if self.position:
            if self.position.size > 0:
                if (
                    self.data.low[0] <= self.sl
                    or self.data.high[0] >= self.tp
                ):
                    self.close()
            elif self.position.size < 0:
                if (
                    self.data.high[0] >= self.sl
                    or self.data.low[0] <= self.tp
                ):
                    self.close()
            return

        # Check patterns for signal generation
        signal = None  # 'BUY' or 'SELL'
        sl_price = None
        tp_price = None

        uptrend = self.data.close[0] > self.sma20[0]

        # 1. Double Top Pattern (Sell / Short)
        if uptrend and not signal:
            peaks = []
            for i in range(-20, -1):
                if (
                    self.data.high[i] > self.data.high[i - 1]
                    and self.data.high[i] > self.data.high[i + 1]
                ):
                    peaks.append((i, self.data.high[i]))

            if len(peaks) >= 2:
                p1, p2 = peaks[-2], peaks[-1]
                if (p2[0] - p1[0] >= 3) and (abs(p1[1] - p2[1]) / p1[1] <= 0.02):
                    trough_low = min(
                        [self.data.low[k] for k in range(p1[0], p2[0] + 1)]
                    )
                    if (
                        self.data.close[-1] >= trough_low
                        and self.data.close[0] < trough_low
                    ):
                        signal = "SELL"
                        sl_price = max(p1[1], p2[1])
                        pattern_height = max(p1[1], p2[1]) - trough_low
                        tp_price = self.data.close[0] - pattern_height

        # 2. Head and Shoulders Pattern (Sell / Short)
        if uptrend and not signal:
            peaks = []
            for i in range(-25, -1):
                if (
                    self.data.high[i] > self.data.high[i - 1]
                    and self.data.high[i] > self.data.high[i + 1]
                ):
                    peaks.append((i, self.data.high[i]))

            if len(peaks) >= 3:
                p1, p2, p3 = peaks[-3], peaks[-2], peaks[-1]
                if (
                    p2[1] > p1[1]
                    and p2[1] > p3[1]
                    and (abs(p1[1] - p3[1]) / p1[1] <= 0.04)
                ):
                    v1_val = min(
                        [self.data.low[k] for k in range(p1[0], p2[0] + 1)]
                    )
                    v1_idx = [
                        k
                        for k in range(p1[0], p2[0] + 1)
                        if self.data.low[k] == v1_val
                    ][0]

                    v2_val = min(
                        [self.data.low[k] for k in range(p2[0], p3[0] + 1)]
                    )
                    v2_idx = [
                        k
                        for k in range(p2[0], p3[0] + 1)
                        if self.data.low[k] == v2_val
                    ][0]

                    if v2_idx != v1_idx:
                        m = (v2_val - v1_val) / (v2_idx - v1_idx)
                        neckline_prev = v2_val + m * (-1 - v2_idx)
                        neckline_curr = v2_val + m * (0 - v2_idx)

                        if (
                            self.data.close[-1] >= neckline_prev
                            and self.data.close[0] < neckline_curr
                        ):
                            head_neckline = v2_val + m * (p2[0] - v2_idx)
                            pattern_height = p2[1] - head_neckline
                            signal = "SELL"
                            sl_price = max(p1[1], p2[1], p3[1])
                            tp_price = self.data.close[0] - max(
                                pattern_height, 0.01
                            )

        # 3. Flag Pattern (Buy / Long)
        if not signal:
            pole_move = (
                self.data.close[-5] - self.data.low[-15]
            ) / self.data.low[-15]
            if pole_move > 0.03:
                consol_highs = [self.data.high[k] for k in range(-5, 0)]
                consol_lows = [self.data.low[k] for k in range(-5, 0)]
                max_c, min_c = max(consol_highs), min(consol_lows)

                if (max_c - min_c) / self.data.close[-5] < 0.035:
                    vol_consol = sum(
                        [self.data.volume[k] for k in range(-5, 0)]
                    ) / 5.0
                    vol_pole = sum(
                        [self.data.volume[k] for k in range(-15, -5)]
                    ) / 10.0

                    if vol_consol < vol_pole:
                        if (
                            self.data.close[0] > max_c
                            and self.data.volume[0] > 1.3 * self.vol_sma20[0]
                        ):
                            signal = "BUY"
                            sl_price = min_c
                            pole_height = (
                                self.data.close[-5] - self.data.low[-15]
                            )
                            tp_price = self.data.close[0] + pole_height

        # 4. Pennant Pattern (Buy / Long)
        if not signal:
            pole_move = (
                self.data.close[-5] - self.data.low[-15]
            ) / self.data.low[-15]
            if pole_move > 0.03:
                h_desc = (
                    self.data.high[-5]
                    > self.data.high[-3]
                    > self.data.high[-1]
                )
                l_asc = (
                    self.data.low[-5] < self.data.low[-3] < self.data.low[-1]
                )

                if h_desc and l_asc:
                    vol_consol = sum(
                        [self.data.volume[k] for k in range(-5, 0)]
                    ) / 5.0
                    vol_pole = sum(
                        [self.data.volume[k] for k in range(-15, -5)]
                    ) / 10.0

                    if vol_consol < vol_pole:
                        max_c = max([self.data.high[k] for k in range(-5, 0)])
                        min_c = min([self.data.low[k] for k in range(-5, 0)])
                        if (
                            self.data.close[0] > max_c
                            and self.data.volume[0] > 1.3 * self.vol_sma20[0]
                        ):
                            signal = "BUY"
                            sl_price = min_c
                            pole_height = (
                                self.data.close[-5] - self.data.low[-15]
                            )
                            tp_price = self.data.close[0] + pole_height

        # Execute orders based on risk management
        if signal and sl_price and tp_price:
            equity = self.broker.getvalue()
            risk_amount = equity * 0.015
            risk_per_share = abs(self.data.close[0] - sl_price)

            if risk_per_share > 0:
                size = int(risk_amount / risk_per_share)
                if signal == "BUY":
                    max_affordable = int(
                        self.broker.getcash() / self.data.close[0]
                    )
                    size = min(size, max_affordable)
                    if size > 0:
                        self.sl = sl_price
                        self.tp = tp_price
                        self.order = self.buy(size=size)
                elif signal == "SELL":
                    max_affordable = int(
                        self.broker.getcash() / self.data.close[0]
                    )
                    size = min(size, max_affordable)
                    if size > 0:
                        self.sl = sl_price
                        self.tp = tp_price
                        self.order = self.sell(size=size)
