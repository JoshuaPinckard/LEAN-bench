import math
import statistics

import backtrader as bt


class FinancialTurbulenceStrategy(bt.Strategy):
    params = dict(
        lookback=24,
        exit_ma_period=12,
        threshold_pct=95,
        stop_loss_pct=0.02,
        position_pct=0.05,
    )

    def __init__(self):
        self.order = None
        self.month_key = None
        self.month_end_closes = []
        self.monthly_returns = []
        self.ft_values = []
        self.ft_thresholds = []
        self.ft_ma = None
        self.entry_price = None
        self.stop_price = None

    def _percentile(self, data, pct):
        s = sorted(data)
        n = len(s)
        if n == 1:
            return s[0]
        k = (n - 1) * (pct / 100.0)
        f = math.floor(k)
        c = math.ceil(k)
        if f == c:
            return s[int(k)]
        return s[int(f)] * (c - k) + s[int(c)] * (k - f)

    def _calc_size(self, price):
        value = self.broker.getvalue() * self.p.position_pct
        return int(value / price)

    def notify_order(self, order):
        if order.status == order.Completed:
            if order.isbuy() and self.position.size > 0:
                self.entry_price = order.executed.price
                self.stop_price = self.entry_price * (1 - self.p.stop_loss_pct)
            elif order.issell() and self.position.size < 0:
                self.entry_price = order.executed.price
                self.stop_price = self.entry_price * (1 + self.p.stop_loss_pct)
            elif not self.position:
                self.entry_price = None
                self.stop_price = None
            self.order = None
        elif order.status in (order.Canceled, order.Margin, order.Rejected):
            self.order = None

    def _on_month_end(self, close_price):
        self.month_end_closes.append(close_price)
        if len(self.month_end_closes) < 2:
            return

        r_t = (self.month_end_closes[-1] - self.month_end_closes[-2]) / self.month_end_closes[-2]
        self.monthly_returns.append(r_t)

        N = self.p.lookback
        if len(self.monthly_returns) < N + 1:
            return

        window = self.monthly_returns[-(N + 1):-1]
        mu = statistics.mean(window)
        var = statistics.variance(window)
        d_t = ((r_t - mu) ** 2) / var if var > 0 else 0.0

        prev_ft = self.ft_values[-1] if self.ft_values else None
        prev_threshold = self.ft_thresholds[-1] if self.ft_thresholds else None

        hist_window = self.ft_values[-N:] if self.ft_values else []
        current_threshold = self._percentile(hist_window, self.p.threshold_pct) if hist_window else None

        self.ft_values.append(d_t)
        self.ft_thresholds.append(current_threshold)

        if len(self.ft_values) >= self.p.exit_ma_period:
            self.ft_ma = statistics.mean(self.ft_values[-self.p.exit_ma_period:])
        else:
            self.ft_ma = None

        if self.order is not None:
            return

        if not self.position:
            crossed_above = (
                prev_ft is not None and prev_threshold is not None
                and prev_ft <= prev_threshold
                and current_threshold is not None and d_t > current_threshold
            )
            if crossed_above:
                price = self.data.close[0]
                size = self._calc_size(price)
                if size > 0:
                    if r_t < 0:
                        self.order = self.buy(size=size)
                    elif r_t > 0:
                        self.order = self.sell(size=size)
        else:
            if self.ft_ma is not None and d_t < self.ft_ma:
                self.order = self.close()

    def next(self):
        dt = self.data.datetime.date(0)
        current_key = (dt.year, dt.month)
        if self.month_key is None:
            self.month_key = current_key
        elif current_key != self.month_key:
            self._on_month_end(self.data.close[-1])
            self.month_key = current_key

        if self.position and self.order is None and self.stop_price is not None:
            if self.position.size > 0 and self.data.low[0] <= self.stop_price:
                self.order = self.close()
            elif self.position.size < 0 and self.data.high[0] >= self.stop_price:
                self.order = self.close()
