import backtrader as bt


class MACDBollingerStrategy(bt.Strategy):
    params = (
        ('bb_period', 20),
        ('bb_dev', 2.0),
        ('sma_period', 50),
        ('macd_fast', 12),
        ('macd_slow', 26),
        ('macd_signal', 9),
    )

    def __init__(self):
        self.order = None

        # Bollinger Bands (20, 2)
        self.bb = bt.indicators.BollingerBands(
            self.data,
            period=self.params.bb_period,
            devfactor=self.params.bb_dev,
        )

        # Band width: upper - lower
        self.width = self.bb.top - self.bb.bot

        # SMA(50) of the band width
        self.sma_width = bt.indicators.SMA(
            self.width, period=self.params.sma_period
        )

        # MACD(12, 26, 9)
        self.macd = bt.indicators.MACD(
            self.data,
            period_me1=self.params.macd_fast,
            period_me2=self.params.macd_slow,
            period_signal=self.params.macd_signal,
        )

        # CrossOver of MACD line and Signal line
        self.crossover = bt.indicators.CrossOver(
            self.macd.macd, self.macd.signal
        )

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            self.order = None

    def next(self):
        if self.order:
            return

        filter_active = self.width[0] > self.sma_width[0]

        if not self.position:
            if filter_active:
                if self.crossover[0] > 0:
                    size = int(self.broker.getcash() / self.data.close[0])
                    if size > 0:
                        self.order = self.buy(size=size)
                elif self.crossover[0] < 0:
                    size = int(self.broker.getcash() / self.data.close[0])
                    if size > 0:
                        self.order = self.sell(size=size)
        else:
            if self.position.size > 0:
                if self.crossover[0] < 0:
                    self.order = self.close()
            elif self.position.size < 0:
                if self.crossover[0] > 0:
                    self.order = self.close()
