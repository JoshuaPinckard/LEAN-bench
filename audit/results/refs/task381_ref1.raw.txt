import backtrader as bt


class EMASetupBreakoutStrategy(bt.Strategy):
    def __init__(self):
        self.ema = bt.indicators.EMA(self.data.close, period=5)

        self.entry_order = None
        self.stop_order = None
        self.target_order = None

        self.entry_side = None
        self.setup_high = None
        self.setup_low = None
        self.entry_canceling = False

    def notify_order(self, order):
        if self.entry_order is not None and order.ref == self.entry_order.ref:
            if order.status == order.Completed:
                side = self.entry_side
                stop_price = self.setup_low if side == "long" else self.setup_high
                entry_price = order.executed.price
                exit_size = abs(self.position.size)

                self.entry_order = None
                self.entry_side = None
                self.entry_canceling = False

                if side == "long":
                    target_price = entry_price + 3 * (entry_price - stop_price)
                    self.stop_order = self.sell(
                        size=exit_size,
                        exectype=bt.Order.Stop,
                        price=stop_price,
                    )
                    self.target_order = self.sell(
                        size=exit_size,
                        exectype=bt.Order.Limit,
                        price=target_price,
                        oco=self.stop_order,
                    )
                else:
                    target_price = entry_price - 3 * (stop_price - entry_price)
                    self.stop_order = self.buy(
                        size=exit_size,
                        exectype=bt.Order.Stop,
                        price=stop_price,
                    )
                    self.target_order = self.buy(
                        size=exit_size,
                        exectype=bt.Order.Limit,
                        price=target_price,
                        oco=self.stop_order,
                    )
                return

            if order.status in [order.Canceled, order.Margin, order.Rejected]:
                self.entry_order = None
                self.entry_side = None
                self.entry_canceling = False
                return

        if self.stop_order is not None and order.ref == self.stop_order.ref:
            if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
                self.stop_order = None
                return

        if self.target_order is not None and order.ref == self.target_order.ref:
            if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
                self.target_order = None

    def next(self):
        long_setup = self.data.high[0] < self.ema[0]
        short_setup = self.data.low[0] > self.ema[0]

        if self.entry_order is not None:
            opposite_setup = (
                (self.entry_side == "long" and short_setup)
                or (self.entry_side == "short" and long_setup)
            )
            if opposite_setup and not self.entry_canceling:
                self.cancel(self.entry_order)
                self.entry_canceling = True
            return

        if self.position.size or self.stop_order is not None or self.target_order is not None:
            return

        equity = self.broker.getvalue()

        if long_setup:
            stop_distance = self.data.high[0] - self.data.low[0]
            if stop_distance > 0:
                size = (0.05 * equity) / stop_distance
                self.setup_high = self.data.high[0]
                self.setup_low = self.data.low[0]
                self.entry_side = "long"
                self.entry_order = self.buy(
                    size=size,
                    exectype=bt.Order.Stop,
                    price=self.setup_high,
                )

        elif short_setup:
            stop_distance = self.data.high[0] - self.data.low[0]
            if stop_distance > 0:
                size = (0.05 * equity) / stop_distance
                self.setup_high = self.data.high[0]
                self.setup_low = self.data.low[0]
                self.entry_side = "short"
                self.entry_order = self.sell(
                    size=size,
                    exectype=bt.Order.Stop,
                    price=self.setup_low,
                )
