import backtrader as bt
import math
import statistics


class MeanReversionAutocorrelation(bt.Strategy):
    params = (
        ('lookback', 20),
        ('stop_loss_pct', 0.02),
        ('z_thresh', 2.0),
    )

    def __init__(self):
        self.order = None
        self.stop_order = None

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            if self.order == order:
                self.order = None
            if self.stop_order == order:
                self.stop_order = None

        if order.status == order.Completed:
            if self.position.size > 0 and self.stop_order is None:
                stop_price = order.executed.price * (1.0 - self.p.stop_loss_pct)
                self.stop_order = self.sell(exectype=bt.Order.Stop, price=stop_price)
            elif self.position.size < 0 and self.stop_order is None:
                stop_price = order.executed.price * (1.0 + self.p.stop_loss_pct)
                self.stop_order = self.buy(exectype=bt.Order.Stop, price=stop_price)

    def _pearson(self, x, y):
        n = len(x)
        mean_x = sum(x) / float(n)
        mean_y = sum(y) / float(n)
        cov = sum((x[i] - mean_x) * (y[i] - mean_y) for i in range(n))
        var_x = sum((x[i] - mean_x) ** 2 for i in range(n))
        var_y = sum((y[i] - mean_y) ** 2 for i in range(n))
        denom = math.sqrt(var_x * var_y)
        return cov / denom if denom > 1e-12 else 0.0

    def next(self):
        p = self.p.lookback
        min_bars = p * 2 + 1
        if len(self.data) < min_bars:
            return

        corrs = []
        for k in range(p + 1):
            x = [self.data.close[-k - i] for i in range(p)]
            y = [self.data.close[-k - i - 1] for i in range(p)]
            corrs.append(self._pearson(x, y))

        corrs_curr = corrs[0:p]
        mean_0 = sum(corrs_curr) / float(p)
        std_0 = statistics.stdev(corrs_curr)
        z_0 = (corrs_curr[0] - mean_0) / std_0 if std_0 > 1e-12 else 0.0

        corrs_prev = corrs[1:p + 1]
        mean_1 = sum(corrs_prev) / float(p)
        std_1 = statistics.stdev(corrs_prev)
        z_1 = (corrs_prev[0] - mean_1) / std_1 if std_1 > 1e-12 else 0.0

        if self.order:
            return

        if not self.position:
            if z_1 < -self.p.z_thresh and z_0 > z_1:
                self.order = self.buy()
            elif z_1 > self.p.z_thresh and z_0 < z_1:
                self.order = self.sell()
        else:
            if self.position.size > 0:
                if z_0 >= 0.0:
                    if self.stop_order:
                        self.cancel(self.stop_order)
                        self.stop_order = None
                    self.order = self.close()
            elif self.position.size < 0:
                if z_0 <= 0.0:
                    if self.stop_order:
                        self.cancel(self.stop_order)
                        self.stop_order = None
                    self.order = self.close()
