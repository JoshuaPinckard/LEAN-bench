import backtrader as bt


class MeanReversionFixedLevels(bt.Strategy):
    params = (
        ('fixed_low', 445.00),
        ('fixed_high', 450.00),
        ('tp_mid_short', 447.5),
        ('tp_close_short', 446.0),
        ('tp_high_long', 449.5),
        ('tp_close_long', 449.0),
        ('sl_short_init', 450.25),
        ('sl_long_init', 444.75),
        ('size', 2),
    )

    def __init__(self):
        self.order = None
        self.sl_price = None

        self.atr = bt.indicators.ATR(self.data, period=14)
        self.atr_sma = bt.indicators.SMA(self.atr, period=50)
        self.rsi = bt.indicators.RSI(self.data.close, period=14)
        self.bb = bt.indicators.BollingerBands(self.data.close, period=20, devfactor=2.0)

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None
        if self.position.size == 0:
            self.sl_price = None

    def next(self):
        if self.order:
            return

        pos = self.position.size

        if pos == 0:
            vol_ok = self.atr[0] > self.atr_sma[0]
            if not vol_ok:
                return

            short_cond = (
                self.data.high[0] > self.p.fixed_high and
                self.data.close[0] < self.p.fixed_high and
                self.rsi[0] > 60 and
                self.data.high[0] >= self.bb.top[0]
            )

            long_cond = (
                self.data.low[0] < self.p.fixed_low and
                self.data.close[0] > self.p.fixed_low and
                self.rsi[0] < 40 and
                self.data.low[0] <= self.bb.bot[0]
            )

            if short_cond:
                self.order = self.sell(size=self.p.size)
                self.sl_price = self.p.sl_short_init
            elif long_cond:
                self.order = self.buy(size=self.p.size)
                self.sl_price = self.p.sl_long_init

        else:
            entry_p = self.position.price
            atr_val = self.atr[0]

            if pos > 0:
                if self.data.high[0] - entry_p >= 0.75 * atr_val:
                    new_sl = entry_p + 0.5 * atr_val
                    if self.sl_price is None or new_sl > self.sl_price:
                        self.sl_price = new_sl

                if self.data.low[0] <= self.sl_price:
                    self.order = self.close()
                    return

                if self.data.high[0] >= self.p.tp_high_long or self.data.close[0] >= self.p.tp_close_long:
                    self.order = self.close()
                    return

                self.order = self.close()

            elif pos < 0:
                if entry_p - self.data.low[0] >= 0.75 * atr_val:
                    new_sl = entry_p - 0.5 * atr_val
                    if self.sl_price is None or new_sl < self.sl_price:
                        self.sl_price = new_sl

                if self.data.high[0] >= self.sl_price:
                    self.order = self.close()
                    return

                if (self.data.low[0] <= self.p.tp_mid_short or
                    self.data.close[0] <= self.p.tp_close_short or
                    self.data.low[0] <= self.bb.mid[0]):
                    self.order = self.close()
                    return

                self.order = self.close()
