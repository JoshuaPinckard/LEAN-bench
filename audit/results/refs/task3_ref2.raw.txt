import backtrader as bt


class SwingRSIFilter(bt.Strategy):
    def __init__(self):
        self.sma = bt.indicators.SMA(self.data.close, period=50)
        self.rsi = bt.indicators.RSI(self.data.close, period=14)

        self.order = None
        self.target_high = None

    def notify_order(self, order):
        if order.status in [order.Completed]:
            if order.isbuy():
                self.order = self.close(
                    exectype=bt.Order.Limit, price=self.target_high
                )
            else:
                self.order = None
        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        if self.order or self.position:
            return

        if self.data.close[0] > self.sma[0] and self.rsi[0] < 50:
            self.target_high = max(self.data.high.get(size=20))
            capital = self.broker.getvalue()
            size = (capital * 0.01) / self.data.close[0]
            self.order = self.buy(size=size)
