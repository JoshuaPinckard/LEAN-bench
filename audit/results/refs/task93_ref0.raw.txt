import backtrader as bt


class MomentumVolatilityFilterStrategy(bt.Strategy):
    def __init__(self):
        self.fast_sma = bt.indicators.SMA(self.data.open, period=20)
        self.slow_sma = bt.indicators.SMA(self.data.close, period=50)
        self.short_vol = bt.indicators.StandardDeviation(self.data.close, period=10)
        self.long_vol = bt.indicators.StandardDeviation(self.data.close, period=30)
        self.crossover = bt.indicators.CrossOver(self.fast_sma, self.slow_sma)
        self.order = None

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        if self.order:
            return

        if not self.position:
            if self.crossover[0] > 0 and self.long_vol[0] > self.short_vol[0]:
                size = int(self.broker.getcash() / self.data.close[0])
                if size > 0:
                    self.order = self.buy(size=size)
        else:
            if self.crossover[0] < 0 or self.long_vol[0] <= self.short_vol[0]:
                self.order = self.close()
