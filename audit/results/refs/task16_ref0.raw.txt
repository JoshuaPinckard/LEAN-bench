import math

import backtrader as bt


class PartialTakeProfitTrailingStop(bt.Strategy):
    params = (
        ('risk_pct', 0.01),
        ('partial_pct', 0.5),
        ('tp_r_mult', 1.5),
        ('trail_pct', 0.0075),
    )

    def __init__(self):
        self.order = None
        self.exit_reason = None
        self._pending_stop = None

        self.entry_price = None
        self.stop_price = None
        self.r = None
        self.entry_size = 0
        self.partial_size = 0
        self.partial_tp_price = None
        self.partial_taken = False

        self.trail_active = False
        self.trail_extreme = None
        self.trail_stop_price = None

    def _reset_state(self):
        self.entry_price = None
        self.stop_price = None
        self.r = None
        self.entry_size = 0
        self.partial_size = 0
        self.partial_tp_price = None
        self.partial_taken = False
        self.trail_active = False
        self.trail_extreme = None
        self.trail_stop_price = None

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return

        if order.status == order.Completed:
            if order.isbuy():
                self.entry_price = order.executed.price
                self.stop_price = self._pending_stop
                self.r = self.entry_price - self.stop_price
                self.entry_size = order.executed.size
                psize = math.ceil(self.entry_size * self.p.partial_pct)
                psize = min(psize, self.entry_size - 1)
                self.partial_size = psize
                self.partial_tp_price = self.entry_price + self.p.tp_r_mult * self.r
                self.partial_taken = False
                self.trail_active = False
            else:
                if self.exit_reason == 'partial_tp':
                    self.partial_taken = True
                    self.trail_active = True
                    self.trail_extreme = order.executed.price
                    self.trail_stop_price = self.trail_extreme * (1 - self.p.trail_pct)
                else:
                    self._reset_state()

        if order.status in (order.Completed, order.Canceled, order.Margin, order.Rejected):
            self.order = None
            self.exit_reason = None

    def next(self):
        if self.order:
            return

        if not self.position:
            equity = self.broker.getvalue()
            risk_amount = equity * self.p.risk_pct
            stop_ref = self.data.low[0]
            r_est = self.data.close[0] - stop_ref
            if r_est <= 0:
                return
            size = int(risk_amount / r_est)
            if size < 2:
                return
            self._pending_stop = stop_ref
            self.order = self.buy(size=size)
            return

        if not self.partial_taken:
            if self.data.low[0] <= self.stop_price:
                self.exit_reason = 'stop'
                self.order = self.close()
            elif self.data.high[0] >= self.partial_tp_price:
                self.exit_reason = 'partial_tp'
                self.order = self.sell(size=self.partial_size)
            return

        self.trail_extreme = max(self.trail_extreme, self.data.high[0])
        self.trail_stop_price = self.trail_extreme * (1 - self.p.trail_pct)
        if self.data.low[0] <= self.trail_stop_price:
            self.exit_reason = 'trailing'
            self.order = self.close()
