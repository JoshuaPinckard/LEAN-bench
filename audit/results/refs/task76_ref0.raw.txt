import math

import backtrader as bt


class EntropyPoolingStressStrategy(bt.Strategy):
    params = (
        ('lookback', 60),
        ('target_return', 1.00),
        ('target_vol', 0.20),
        ('shift_threshold', 0.05),
        ('exit_threshold', 0.05),
        ('spike_ratio', 0.10),
        ('volume_period', 20),
        ('max_iter', 50),
        ('tol', 1e-8),
        ('periods_per_year', 252),
    )

    def __init__(self):
        self.order = None
        self.stop_price = None
        self.target_price = None
        self.vol_sma = bt.indicators.SMA(self.data.volume, period=self.p.volume_period)
        self.daily_mu = self.p.target_return / self.p.periods_per_year
        self.daily_sigma = self.p.target_vol / math.sqrt(self.p.periods_per_year)

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def _entropy_pool(self, returns, prior_probs, mu_target, sigma_target):
        n = len(returns)
        lam1, lam2 = 0.0, 0.0
        m2_target = sigma_target ** 2 + mu_target ** 2
        for _ in range(self.p.max_iter):
            try:
                exps = [math.exp(-lam1 * r - lam2 * r * r) for r in returns]
            except OverflowError:
                return None, False
            weighted = [prior_probs[i] * exps[i] for i in range(n)]
            z = sum(weighted)
            if not z or math.isnan(z) or math.isinf(z):
                return None, False
            w = [x / z for x in weighted]
            mean_r = sum(w[i] * returns[i] for i in range(n))
            mean_r2 = sum(w[i] * returns[i] ** 2 for i in range(n))
            g1 = mean_r - mu_target
            g2 = mean_r2 - m2_target
            if abs(g1) < self.p.tol and abs(g2) < self.p.tol:
                return w, True
            mean_r3 = sum(w[i] * returns[i] ** 3 for i in range(n))
            mean_r4 = sum(w[i] * returns[i] ** 4 for i in range(n))
            # Newton step on the dual: J = -Cov_w[(r, r^2)], solved via 2x2 closed form.
            cov11 = mean_r2 - mean_r * mean_r
            cov12 = mean_r3 - mean_r * mean_r2
            cov22 = mean_r4 - mean_r2 * mean_r2
            det = cov11 * cov22 - cov12 * cov12
            if abs(det) < 1e-14:
                return None, False
            lam1 += (cov22 * g1 - cov12 * g2) / det
            lam2 += (-cov12 * g1 + cov11 * g2) / det
            if abs(lam1) > 1e6 or abs(lam2) > 1e6:
                return None, False
        return None, False

    def _quality_ok(self, probs, n):
        if probs is None:
            return False
        ens = 1.0 / sum(p * p for p in probs)
        return ens >= self.p.spike_ratio * n

    def _weighted_quantile(self, values, probs, q):
        pairs = sorted(zip(values, probs), key=lambda x: x[0])
        cum = 0.0
        for v, p in pairs:
            cum += p
            if cum >= q:
                return v
        return pairs[-1][0]

    def _weighted_std(self, values, probs):
        mean = sum(v * p for v, p in zip(values, probs))
        var = sum(p * (v - mean) ** 2 for v, p in zip(values, probs))
        return math.sqrt(var) if var > 0 else 0.0

    def _size_from_vol(self, vol):
        if vol <= 0:
            return 0
        return max(1, int(round(1.0 / vol)))

    def next(self):
        if self.order:
            return
        if len(self.data) < self.p.lookback + 1:
            return

        closes = list(self.data.close.get(size=self.p.lookback + 1))
        if len(closes) < self.p.lookback + 1:
            return
        returns = [(closes[i] - closes[i - 1]) / closes[i - 1] for i in range(1, len(closes))]
        n = len(returns)
        prior_probs = [1.0 / n] * n
        prior_median = self._weighted_quantile(returns, prior_probs, 0.5)

        high_volume = self.data.volume[0] > self.vol_sma[0]

        pos_probs, pos_conv = self._entropy_pool(returns, prior_probs, self.daily_mu, self.daily_sigma)
        neg_probs, neg_conv = self._entropy_pool(returns, prior_probs, -self.daily_mu, self.daily_sigma)
        pos_ok = pos_conv and self._quality_ok(pos_probs, n)
        neg_ok = neg_conv and self._quality_ok(neg_probs, n)
        pos_median = self._weighted_quantile(returns, pos_probs, 0.5) if pos_ok else None
        neg_median = self._weighted_quantile(returns, neg_probs, 0.5) if neg_ok else None

        if not self.position:
            if pos_ok and high_volume and (pos_median - prior_median) > self.p.shift_threshold:
                post_vol = self._weighted_std(returns, pos_probs)
                size = self._size_from_vol(post_vol)
                if size >= 1:
                    p05 = self._weighted_quantile(returns, pos_probs, 0.05)
                    p95 = self._weighted_quantile(returns, pos_probs, 0.95)
                    entry_price = self.data.close[0]
                    self.stop_price = entry_price * (1 + p05)
                    self.target_price = entry_price * (1 + p95)
                    self.order = self.buy(size=size)
            elif neg_ok and high_volume and (prior_median - neg_median) > self.p.shift_threshold:
                post_vol = self._weighted_std(returns, neg_probs)
                size = self._size_from_vol(post_vol)
                if size >= 1:
                    p05 = self._weighted_quantile(returns, neg_probs, 0.05)
                    p95 = self._weighted_quantile(returns, neg_probs, 0.95)
                    entry_price = self.data.close[0]
                    self.stop_price = entry_price * (1 + p95)
                    self.target_price = entry_price * (1 + p05)
                    self.order = self.sell(size=size)
            return

        price = self.data.close[0]
        if self.position.size > 0:
            if price <= self.stop_price or price >= self.target_price:
                self.order = self.close()
                return
            reverted = pos_ok and abs(pos_median - prior_median) < self.p.exit_threshold
            opposite = neg_ok and (prior_median - neg_median) > self.p.shift_threshold
            if reverted or opposite:
                self.order = self.close()
        elif self.position.size < 0:
            if price >= self.stop_price or price <= self.target_price:
                self.order = self.close()
                return
            reverted = neg_ok and abs(neg_median - prior_median) < self.p.exit_threshold
            opposite = pos_ok and (pos_median - prior_median) > self.p.shift_threshold
            if reverted or opposite:
                self.order = self.close()
