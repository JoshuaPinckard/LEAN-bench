import backtrader as bt


class BBEmaIntradayStrategy(bt.Strategy):
    params = (
        ("bb_period", 20),
        ("bb_devfactor", 2),
        ("ema_period", 50),
    )

    def __init__(self):
        self.bb = bt.indicators.BollingerBands(
            self.data.close,
            period=self.p.bb_period,
            devfactor=self.p.bb_devfactor,
        )
        self.ema = bt.indicators.EMA(
            self.data.close,
            period=self.p.ema_period,
        )
        self.cross_bb_top = bt.indicators.CrossOver(self.data.close, self.bb.top)
        self.cross_bb_bot = bt.indicators.CrossOver(self.data.close, self.bb.bot)
        self.cross_ema = bt.indicators.CrossOver(self.data.close, self.ema)
        self.order = None

    def notify_order(self, order):
        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            self.order = None

    def next(self):
        if self.order:
            return

        long_entry = (self.cross_bb_top[0] > 0) and (self.data.close[0] > self.ema[0])
        short_entry = (self.cross_bb_bot[0] < 0) and (self.data.close[0] < self.ema[0])
        exit_long = self.cross_ema[0] < 0
        exit_short = self.cross_ema[0] > 0

        if self.position.size > 0:
            if short_entry:
                self.close()
                self.order = self.sell()
            elif exit_long:
                self.order = self.close()
        elif self.position.size < 0:
            if long_entry:
                self.close()
                self.order = self.buy()
            elif exit_short:
                self.order = self.close()
        else:
            if long_entry:
                self.order = self.buy()
            elif short_entry:
                self.order = self.sell()
