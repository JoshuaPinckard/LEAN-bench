import backtrader as bt


class KeltnerWilliamsStrategy(bt.Strategy):
    params = (
        ('ema_period', 20),
        ('atr_period', 14),
        ('keltner_mult', 2.0),
        ('willr_period', 14),
        ('stop_atr_mult', 1.5),
    )

    def __init__(self):
        self.ema = bt.indicators.EMA(self.data.close, period=self.p.ema_period)
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.lower_band = self.ema - self.p.keltner_mult * self.atr
        self.upper_band = self.ema + self.p.keltner_mult * self.atr
        self.willr = bt.indicators.WilliamsR(self.data, period=self.p.willr_period)

        self.order = None
        self.entry_order = None
        self.exit_order = None
        self.stop_order = None

    def next(self):
        if self.order:
            return

        if self.position:
            if ((self.position.size > 0 and self.data.close[0] >= self.ema[0]) or
                    (self.position.size < 0 and self.data.close[0] <= self.ema[0])):
                if self.stop_order:
                    self.cancel(self.stop_order)
                self.exit_order = self.close()
                self.order = self.exit_order
            return

        if self.data.close[0] < self.lower_band[0] and self.willr[0] < -80:
            self.entry_order = self.buy()
            self.order = self.entry_order
        elif self.data.close[0] > self.upper_band[0] and self.willr[0] > -20:
            self.entry_order = self.sell()
            self.order = self.entry_order

    def notify_order(self, order):
        if order.status in (order.Completed, order.Canceled, order.Margin, order.Rejected):
            if order is self.order:
                self.order = None

            if order is self.entry_order:
                self.entry_order = None
                if order.status == order.Completed:
                    if self.position.size > 0:
                        self.stop_order = self.sell(
                            exectype=bt.Order.Stop,
                            price=order.executed.price - self.p.stop_atr_mult * self.atr[0],
                        )
                    elif self.position.size < 0:
                        self.stop_order = self.buy(
                            exectype=bt.Order.Stop,
                            price=order.executed.price + self.p.stop_atr_mult * self.atr[0],
                        )

            if order is self.exit_order:
                self.exit_order = None
                if order.status == order.Completed and not self.position and self.stop_order:
                    self.cancel(self.stop_order)

            if order is self.stop_order:
                self.stop_order = None
                if order.status == order.Completed and not self.position:
                    self.order = None
