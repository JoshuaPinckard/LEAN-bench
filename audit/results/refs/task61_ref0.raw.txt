import backtrader as bt


class OverheatingIndicator(bt.Indicator):
    lines = ('state',)
    params = (
        ('sma_period', 200),
        ('heavy_threshold', 1.00),
        ('light_threshold', 1.20),
    )

    STATE_INITIAL = 0
    STATE_HEAVY = 1
    STATE_LIGHT = 2

    def __init__(self):
        self.sma = bt.indicators.SimpleMovingAverage(self.data.close, period=self.p.sma_period)

    def next(self):
        ratio = self.data.close[0] / self.sma[0]
        if ratio <= self.p.heavy_threshold:
            self.lines.state[0] = self.STATE_HEAVY
        elif ratio >= self.p.light_threshold:
            self.lines.state[0] = self.STATE_LIGHT
        else:
            self.lines.state[0] = self.STATE_INITIAL


class MarketOverheatingPositionalStrategy(bt.Strategy):
    params = (
        ('sma_period', 200),
        ('heavy_threshold', 1.00),
        ('light_threshold', 1.20),
        ('qld_weight', 0.5),
        ('tqqq_weight', 0.5),
        ('heavy_target_percent', 1.0),
        ('light_target_percent', 0.0),
    )

    def __init__(self):
        self.overheat = OverheatingIndicator(
            self.data,
            sma_period=self.p.sma_period,
            heavy_threshold=self.p.heavy_threshold,
            light_threshold=self.p.light_threshold,
        )
        self.order = None
        self.prev_state = None

    def notify_order(self, order):
        if order.status in (order.Completed, order.Canceled, order.Margin, order.Rejected):
            self.order = None

    def next(self):
        if self.order:
            return

        state = int(self.overheat.state[0])

        if self.prev_state is None:
            self.prev_state = state
            return

        if state != self.prev_state:
            if state == OverheatingIndicator.STATE_HEAVY:
                self.order = self.order_target_percent(target=self.p.heavy_target_percent)
            elif state == OverheatingIndicator.STATE_LIGHT:
                self.order = self.order_target_percent(target=self.p.light_target_percent)

        self.prev_state = state
