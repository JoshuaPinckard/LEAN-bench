import backtrader as bt


class SupplyDemandSwingStrategy(bt.Strategy):
    params = (
        ('consol_window', 3),
        ('consol_max_pct', 0.01),
        ('impulse_min_pct', 0.01),
        ('min_zone_height_pct', 0.005),
        ('risk_pct', 0.01),
        ('sl_buffer_pct', 0.005),
        ('rr_default', 2.0),
        ('trend_sma_period', 20),
        ('rsi_period', 14),
    )

    def __init__(self):
        self.rsi = bt.indicators.RSI(self.data.close, period=self.p.rsi_period)

        self._day_date = None
        self._day_o = None
        self._day_h = None
        self._day_l = None
        self._day_c = None
        self.daily_bars = []
        self.daily_closes = []

        self._h4_count = 0
        self._h4_o = None
        self._h4_h = None
        self._h4_l = None
        self._h4_c = None
        self.h4_bars = []

        self.demand_zones = []
        self.supply_zones = []

        self.order = None
        self.stop_order = None
        self.limit_order = None
        self._pending_stop = None
        self._pending_target = None

    def _update_daily(self):
        dt = self.data.datetime.date(0)
        o, h, l, c = self.data.open[0], self.data.high[0], self.data.low[0], self.data.close[0]
        if self._day_date is None:
            self._day_date = dt
            self._day_o, self._day_h, self._day_l, self._day_c = o, h, l, c
        elif dt == self._day_date:
            self._day_h = max(self._day_h, h)
            self._day_l = min(self._day_l, l)
            self._day_c = c
        else:
            self.daily_bars.append((self._day_o, self._day_h, self._day_l, self._day_c))
            self.daily_closes.append(self._day_c)
            self._detect_zone(self.daily_bars)
            self._day_date = dt
            self._day_o, self._day_h, self._day_l, self._day_c = o, h, l, c

    def _update_h4(self):
        o, h, l, c = self.data.open[0], self.data.high[0], self.data.low[0], self.data.close[0]
        if self._h4_count == 0:
            self._h4_o, self._h4_h, self._h4_l, self._h4_c = o, h, l, c
        else:
            self._h4_h = max(self._h4_h, h)
            self._h4_l = min(self._h4_l, l)
            self._h4_c = c
        self._h4_count += 1
        if self._h4_count == 4:
            self.h4_bars.append((self._h4_o, self._h4_h, self._h4_l, self._h4_c))
            self._detect_zone(self.h4_bars)
            self._h4_count = 0

    def _detect_zone(self, bars):
        w = self.p.consol_window
        if len(bars) < w + 1:
            return
        window = bars[-(w + 1):-1]
        i_open, i_high, i_low, i_close = bars[-1]
        zone_high = max(b[1] for b in window)
        zone_low = min(b[2] for b in window)
        ref_price = i_close
        if ref_price <= 0:
            return
        if (zone_high - zone_low) / ref_price > self.p.consol_max_pct:
            return
        if (zone_high - zone_low) / ref_price < self.p.min_zone_height_pct:
            return
        if abs(i_close - i_open) / ref_price < self.p.impulse_min_pct:
            return
        if i_close > i_open and i_close > zone_high:
            self.demand_zones.append({'low': zone_low, 'high': zone_high})
        elif i_close < i_open and i_close < zone_low:
            self.supply_zones.append({'low': zone_low, 'high': zone_high})

    def _enter_long(self, zone):
        entry_price = self.data.close[0]
        stop_price = zone['low'] * (1 - self.p.sl_buffer_pct)
        risk_per_unit = entry_price - stop_price
        if risk_per_unit <= 0:
            return
        target = None
        for sz in self.supply_zones:
            if sz['low'] > entry_price and (target is None or sz['low'] < target):
                target = sz['low']
        if target is None:
            target = entry_price + self.p.rr_default * risk_per_unit
        size = (self.broker.getvalue() * self.p.risk_pct) / risk_per_unit
        if size <= 0:
            return
        self._pending_stop = stop_price
        self._pending_target = target
        self.order = self.buy(size=size)

    def _enter_short(self, zone):
        entry_price = self.data.close[0]
        stop_price = zone['high'] * (1 + self.p.sl_buffer_pct)
        risk_per_unit = stop_price - entry_price
        if risk_per_unit <= 0:
            return
        target = None
        for dz in self.demand_zones:
            if dz['high'] < entry_price and (target is None or dz['high'] > target):
                target = dz['high']
        if target is None:
            target = entry_price - self.p.rr_default * risk_per_unit
        size = (self.broker.getvalue() * self.p.risk_pct) / risk_per_unit
        if size <= 0:
            return
        self._pending_stop = stop_price
        self._pending_target = target
        self.order = self.sell(size=size)

    def next(self):
        self._update_daily()
        self._update_h4()

        if self.order or self.position:
            return

        if len(self.daily_closes) < self.p.trend_sma_period:
            return

        sma = sum(self.daily_closes[-self.p.trend_sma_period:]) / self.p.trend_sma_period
        last_daily_close = self.daily_closes[-1]
        uptrend = last_daily_close > sma
        downtrend = last_daily_close < sma

        price_open = self.data.open[0]
        price_high = self.data.high[0]
        price_low = self.data.low[0]
        price_close = self.data.close[0]

        if uptrend:
            for zone in reversed(self.demand_zones):
                zlow, zhigh = zone['low'], zone['high']
                zmid = (zlow + zhigh) / 2.0
                if zlow <= price_low <= zhigh and price_low <= zmid:
                    if price_close > price_low and price_close > price_open:
                        if self.rsi[0] < 50:
                            self._enter_long(zone)
                            break
        elif downtrend:
            for zone in reversed(self.supply_zones):
                zlow, zhigh = zone['low'], zone['high']
                zmid = (zlow + zhigh) / 2.0
                if zlow <= price_high <= zhigh and price_high >= zmid:
                    if price_close < price_high and price_close < price_open:
                        if self.rsi[0] > 50:
                            self._enter_short(zone)
                            break

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order == self.order:
            self.order = None
            if order.status == order.Completed:
                size = abs(order.executed.size)
                if order.isbuy():
                    self.stop_order = self.sell(exectype=bt.Order.Stop, price=self._pending_stop, size=size)
                    self.limit_order = self.sell(exectype=bt.Order.Limit, price=self._pending_target, size=size)
                else:
                    self.stop_order = self.buy(exectype=bt.Order.Stop, price=self._pending_stop, size=size)
                    self.limit_order = self.buy(exectype=bt.Order.Limit, price=self._pending_target, size=size)
            return

        if order == self.stop_order or order == self.limit_order:
            if order.status == order.Completed:
                other = self.limit_order if order == self.stop_order else self.stop_order
                if other is not None and other.status in [other.Submitted, other.Accepted]:
                    self.cancel(other)
            if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
                if order == self.stop_order:
                    self.stop_order = None
                else:
                    self.limit_order = None
