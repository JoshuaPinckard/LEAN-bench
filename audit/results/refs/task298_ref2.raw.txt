import backtrader as bt


class SniperPullback(bt.Strategy):
    params = (
        ("bb_period", 20),
        ("bb_dev", 2.0),
        ("atr_period", 14),
        ("vol_sma_period", 20),
        ("rsi_period", 14),
        ("drop_min", 0.09),
        ("drop_max", 0.13),
        ("vol_mult", 1.5),
        ("rsi_threshold", 45.0),
        ("tp_pct", 0.05),
        ("sl_pct", 0.03),
        ("position_pct", 0.02),
        ("ref_lookback", 6),
    )

    def __init__(self):
        self.bb = bt.indicators.BollingerBands(
            self.data.close,
            period=self.p.bb_period,
            devfactor=self.p.bb_dev,
        )
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.vol_sma = bt.indicators.SMA(
            self.data.volume, period=self.p.vol_sma_period
        )
        self.rsi = bt.indicators.RSI(
            self.data.close, period=self.p.rsi_period
        )

        self.order = None
        self.tp_order = None
        self.sl_order = None

        self.h_ref = None
        self.h_ref_bar = -1000

    def notify_order(self, order):
        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            if self.order == order:
                self.order = None

        if order.status == order.Completed:
            if order.isbuy():
                entry_price = order.executed.price
                tp_price = entry_price * (1.0 + self.p.tp_pct)
                sl_price = entry_price * (1.0 - self.p.sl_pct)
                self.tp_order = self.sell(
                    size=order.executed.size,
                    price=tp_price,
                    exectype=bt.Order.Limit,
                )
                self.sl_order = self.sell(
                    size=order.executed.size,
                    price=sl_price,
                    exectype=bt.Order.Stop,
                    oco=self.tp_order,
                )

    def next(self):
        # Update H_ref whenever a new close above the upper BB appears
        if self.data.close[0] > self.bb.top[0]:
            self.h_ref = self.data.close[0]
            self.h_ref_bar = len(self)

        # Check for entry if no open position and no pending entry order
        if not self.position and self.order is None:
            if (
                self.h_ref is not None
                and (len(self) - self.h_ref_bar) < self.p.ref_lookback
            ):
                drop = (self.h_ref - self.data.close[0]) / self.h_ref
                if self.p.drop_min <= drop <= self.p.drop_max:
                    if self.data.volume[0] > self.p.vol_mult * self.vol_sma[0]:
                        if self.rsi[0] < self.p.rsi_threshold:
                            size = (
                                self.broker.getvalue() * self.p.position_pct
                            ) / self.data.close[0]
                            if size > 0:
                                self.order = self.buy(size=size)
