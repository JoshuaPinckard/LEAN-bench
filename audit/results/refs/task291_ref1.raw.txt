import backtrader as bt


class BreakoutRetestStrategy(bt.Strategy):
    params = (
        ('ema_period', 10),          # 10-day EMA (support after breakout)
        ('breakout_period', 50),     # highest high over last 50 days
        ('retest_window', 5),        # retest must occur within 5 days
        ('vol_period', 50),          # average daily volume over last 50 days
        ('atr_period', 14),          # ATR over 14 days (volatility filter)
        ('take_profit', 0.10),       # 10% take-profit from entry
        ('capital_pct', 0.02),       # 2% of total capital per trade
    )

    def __init__(self):
        # 10-day Exponential Moving Average -> support/stop reference
        self.ema = bt.indicators.EMA(self.data.close, period=self.p.ema_period)

        # Breakout level: highest HIGH over the last 50 days PRIOR to current day.
        # high(-1) excludes the current bar from the 50-day lookback.
        self.breakout_level = bt.indicators.Highest(
            self.data.high(-1), period=self.p.breakout_period)

        # Average daily volume over the last 50 days (demand confirmation).
        self.vol_avg = bt.indicators.SMA(
            self.data.volume, period=self.p.vol_period)

        # ATR(14): named by the volatility filter (see assumptions: no market avg).
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)

        self.order = None
        self.watching = False          # inside a post-breakout retest window
        self.days_since_breakout = 0
        self.entry_price = None
        self.stop_level = None         # 10-day EMA at time of entry

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return
        if order.status == order.Completed and order.isbuy():
            self.entry_price = order.executed.price
        if order.status in (order.Completed, order.Canceled,
                            order.Margin, order.Rejected):
            self.order = None

    def next(self):
        if self.order:
            return

        # ---- Manage an open position (exits) ----
        if self.position:
            # Exit 1 - Stop-loss: close below the 10-day EMA (level fixed at entry)
            if self.data.close[0] < self.stop_level:
                self.order = self.close()
                return
            # Exit 2 - Take-profit: reaching +10% from the entry price
            if self.data.high[0] >= self.entry_price * (1.0 + self.p.take_profit):
                self.order = self.close()
                return
            return

        # ---- Flat: process an active retest window ----
        if self.watching:
            self.days_since_breakout += 1
            # Retest: pull back to (touch) the 10-day EMA but not close below it,
            # with volume above the 50-day average (demand confirmation).
            pulled_back = self.data.low[0] <= self.ema[0]
            not_below = self.data.close[0] >= self.ema[0]
            volume_ok = self.data.volume[0] > self.vol_avg[0]
            if pulled_back and not_below and volume_ok:
                value = self.broker.getvalue()
                size = (value * self.p.capital_pct) / self.data.close[0]
                self.stop_level = self.ema[0]      # stop = EMA at time of entry
                self.order = self.buy(size=size)
                self.watching = False
                return
            # Window elapsed without a valid retest -> abandon the setup.
            if self.days_since_breakout >= self.p.retest_window:
                self.watching = False
            if self.watching:
                return

        # ---- Look for a new breakout ----
        # Breakout: today's close exceeds the 50-day breakout level.
        if self.data.close[0] > self.breakout_level[0]:
            self.watching = True
            self.days_since_breakout = 0
