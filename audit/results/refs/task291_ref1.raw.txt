import backtrader as bt


class BreakoutRetest(bt.Strategy):
    params = (
        ("ema_period", 10),
        ("breakout_period", 50),
        ("retest_days", 5),
        ("atr_period", 14),
        ("take_profit", 0.10),
        ("risk_fraction", 0.02),
    )

    def __init__(self):
        self.ema = bt.indicators.EMA(self.data.close, period=self.p.ema_period)
        self.breakout_level = bt.indicators.Highest(
            self.data.high(-1), period=self.p.breakout_period
        )
        self.volume_average = bt.indicators.SMA(
            self.data.volume, period=self.p.breakout_period
        )
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)

        self.order = None
        self.breakout_bars = []
        self.entry_price = None
        self.entry_ema_stop = None

    def notify_order(self, order):
        if order.status == order.Completed:
            if order.isbuy():
                self.entry_price = order.executed.price
                self.entry_ema_stop = self.ema[0]
            else:
                self.entry_price = None
                self.entry_ema_stop = None
            self.order = None

        elif order.status in (order.Canceled, order.Margin, order.Rejected):
            self.order = None

    def next(self):
        if self.order:
            return

        if self.position:
            take_profit_price = self.entry_price * (1.0 + self.p.take_profit)

            if (
                self.data.close[0] < self.ema[0]
                or self.data.close[0] < self.entry_ema_stop
                or self.data.close[0] >= take_profit_price
            ):
                self.order = self.close()
            return

        current_bar = len(self)
        self.breakout_bars = [
            bar for bar in self.breakout_bars
            if current_bar - bar <= self.p.retest_days
        ]

        is_retest = (
            bool(self.breakout_bars)
            and self.data.low[0] <= self.ema[0]
            and self.data.close[0] >= self.ema[0]
            and self.data.volume[0] > self.volume_average[0]
        )

        if is_retest:
            capital_at_risk = self.broker.getvalue() * self.p.risk_fraction
            size = capital_at_risk / self.data.close[0]
            self.order = self.buy(size=size)
            self.breakout_bars = []
            return

        if self.data.close[0] > self.breakout_level[0]:
            self.breakout_bars.append(current_bar)
