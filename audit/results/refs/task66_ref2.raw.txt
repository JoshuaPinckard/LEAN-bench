import backtrader as bt


class DoubleBottomStrategy(bt.Strategy):

    params = (
        ('sma_period', 200),
        ('vol_avg_period', 20),
        ('bottom_tolerance', 0.01),   # 1% max difference between bottoms
        ('min_bars_between', 10),      # min bars separating two bottoms
        ('stop_pct', 0.005),           # 0.5% below second bottom
        ('local_n', 5),               # bars on each side for local extremum detection
        ('max_lookback', 200),        # max bars to scan backward for pattern
    )

    def __init__(self):
        self.sma200 = bt.indicators.SMA(self.data.close, period=self.p.sma_period)
        self.vol_sma20 = bt.indicators.SMA(self.data.volume, period=self.p.vol_avg_period)
        self.order = None
        self.stop_price = None
        self.take_profit_price = None
        self._pending_stop = None

    def notify_order(self, order):
        if order.status == order.Completed:
            if order.isbuy():
                fill = order.executed.price
                risk = fill - self._pending_stop
                self.stop_price = self._pending_stop
                self.take_profit_price = fill + 2.0 * risk
            self.order = None
        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            self.order = None
            self._pending_stop = None

    def next(self):
        if self.order:
            return

        if self.position:
            close = self.data.close[0]
            if close <= self.stop_price:
                self.order = self.close()
                return
            if close >= self.take_profit_price:
                self.order = self.close()
                return
            if close < self.sma200[0]:
                self.order = self.close()
                return
            return

        if self.data.close[0] <= self.sma200[0]:
            return

        result = self._find_double_bottom()
        if result is not None:
            _, low2 = result
            self._pending_stop = low2 * (1.0 - self.p.stop_pct)
            self.order = self.buy()

    def _find_double_bottom(self):
        n = self.p.local_n
        available = len(self.data)
        lb = min(self.p.max_lookback, available - 1)
        total = min(lb + n + 1, available)

        # Need enough history for two bottoms (each n-confirmed) plus minimum gap plus prev bar
        if total < 2 * n + self.p.min_bars_between + 2:
            return None

        # Volume must exceed 20-period average to confirm breakout
        if self.data.volume[0] <= self.vol_sma20[0]:
            return None

        current_close = self.data.close[0]
        prev_close = self.data.close[-1]

        # lows[i] / highs[i] = value i bars ago (0 = current bar)
        lows = [self.data.low[-i] for i in range(total)]
        highs = [self.data.high[-i] for i in range(total)]

        # Only consider indices with n confirmed bars on each side
        valid_range = range(n, total - n)

        local_mins = [
            i for i in valid_range
            if all(lows[i] <= lows[j] for j in range(i - n, i + n + 1) if j != i)
        ]
        local_maxs = [
            i for i in valid_range
            if all(highs[i] >= highs[j] for j in range(i - n, i + n + 1) if j != i)
        ]

        # Pattern (in bars-ago indexing, larger index = older bar):
        #   b1 (oldest bottom) > nl (neckline) > b2 (newer bottom) > 0 (current breakout)
        for b2 in local_mins:
            low2 = lows[b2]
            for nl in local_maxs:
                if nl <= b2:
                    continue  # neckline must be older than second bottom
                neckline = highs[nl]
                if neckline <= low2:
                    continue  # neckline must be above second bottom
                if current_close <= neckline:
                    continue  # require current close to break above neckline
                if prev_close > neckline:
                    continue  # require fresh crossover (previous close at or below neckline)
                for b1 in local_mins:
                    if b1 <= nl:
                        continue  # first bottom must be older than neckline
                    if b1 - b2 < self.p.min_bars_between:
                        continue  # at least 10 bars between the two bottoms
                    low1 = lows[b1]
                    if abs(low1 - low2) / max(low1, low2) > self.p.bottom_tolerance:
                        continue  # bottoms must be within 1% of each other
                    if neckline <= low1:
                        continue  # neckline must be above first bottom
                    return (neckline, low2)

        return None
