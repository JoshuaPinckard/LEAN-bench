import backtrader as bt


class DoubleBottomStrategy(bt.Strategy):
    def __init__(self):
        self.sma200 = bt.indicators.SimpleMovingAverage(self.data.close, period=200)
        self.sma_vol = bt.indicators.SimpleMovingAverage(self.data.volume, period=20)
        self.order = None
        self.second_bottom_low = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def _detect_double_bottom(self):
        if not (self.data.close[0] > self.sma200[0] and self.data.volume[0] > self.sma_vol[0]):
            return None

        max_lookback = min(len(self) - 1, 100)

        for k2 in range(1, min(60, max_lookback - 11)):
            for k1 in range(k2 + 10, max_lookback - 1):
                peaks = [(self.data.high[-k], k) for k in range(k2 + 1, k1)]
                if not peaks:
                    continue
                h_peak, k_peak = max(peaks, key=lambda x: x[0])

                l1 = self.data.low[-k1]
                l2 = self.data.low[-k2]

                if abs(l2 - l1) / l1 > 0.01:
                    continue

                min_l1_window = min(self.data.low[-k] for k in range(k_peak, min(k1 + 4, max_lookback)))
                if l1 > min_l1_window:
                    continue

                min_l2_window = min(self.data.low[-k] for k in range(max(0, k2 - 3), k_peak + 1))
                if l2 > min_l2_window:
                    continue

                if self.data.close[0] <= h_peak:
                    continue

                if self.data.close[-1] > h_peak:
                    continue

                if any(self.data.close[-k] > h_peak for k in range(1, k_peak)):
                    continue

                return l2

        return None

    def next(self):
        if self.order is not None:
            return

        if not self.position:
            l2 = self._detect_double_bottom()
            if l2 is not None:
                self.second_bottom_low = l2
                self.order = self.buy()
        else:
            entry_price = self.position.price
            stop_loss = 0.995 * self.second_bottom_low
            risk = entry_price - stop_loss

            if risk > 0:
                take_profit = entry_price + 2.0 * risk
            else:
                take_profit = entry_price

            if self.data.low[0] <= stop_loss:
                self.order = self.close()
            elif self.data.high[0] >= take_profit:
                self.order = self.close()
            elif self.data.close[0] < self.sma200[0]:
                self.order = self.close()
