import backtrader as bt
import math
from collections import namedtuple


class DonchianADXATRStrategy(bt.Strategy):
    params = (
        ('donchian_period', 20),
        ('adx_period', 14),
        ('atr_period', 14),
    )

    def __init__(self):
        DonchianChannel = namedtuple('DonchianChannel', ['top', 'bottom'])
        self.donchian = DonchianChannel(
            top=bt.indicators.Highest(self.data.high(-1), period=self.p.donchian_period),
            bottom=bt.indicators.Lowest(self.data.low(-1), period=self.p.donchian_period),
        )
        self.adx = bt.indicators.ADX(self.data, period=self.p.adx_period)
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)

        self.order = None
        self.stop_order = None

    def _stop_distance(self, entry_price):
        return max(2 * self.atr[0], 0.02 * entry_price)

    def _cancel_stop(self):
        if self.stop_order is not None:
            self.cancel(self.stop_order)

    def next(self):
        if self.order is not None:
            return

        if self.position:
            if (self.position.size > 0 and
                    self.data.close[0] < self.donchian.bottom[0]):
                self._cancel_stop()
                self.order = self.close()
            elif (self.position.size < 0 and
                    self.data.close[0] > self.donchian.top[0]):
                self._cancel_stop()
                self.order = self.close()
            return

        if self.stop_order is not None:
            return

        entry_price = self.data.close[0]
        stop_dist = self._stop_distance(entry_price)
        size = math.floor((self.broker.getvalue() * 0.01) / stop_dist)

        if size <= 0:
            return

        if self.data.close[0] > self.donchian.top[0] and self.adx[0] > 25:
            self.order = self.buy(size=size)
        elif self.data.close[0] < self.donchian.bottom[0] and self.adx[0] > 25:
            self.order = self.sell(size=size)

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return

        if order.status == order.Completed:
            if order == self.order:
                self.order = None

                if order.isbuy() and self.position.size > 0:
                    stop_price = order.executed.price - self._stop_distance(order.executed.price)
                    self.stop_order = self.sell(
                        size=order.executed.size,
                        exectype=bt.Order.Stop,
                        price=stop_price,
                    )
                elif order.issell() and self.position.size < 0:
                    stop_price = order.executed.price + self._stop_distance(order.executed.price)
                    self.stop_order = self.buy(
                        size=abs(order.executed.size),
                        exectype=bt.Order.Stop,
                        price=stop_price,
                    )

            if order == self.stop_order:
                self.stop_order = None

        elif order.status in (order.Canceled, order.Margin, order.Rejected):
            if order == self.order:
                self.order = None
            if order == self.stop_order:
                self.stop_order = None
