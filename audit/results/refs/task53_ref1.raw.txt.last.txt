import backtrader as bt


class SMTC_Pattern_Strategy(bt.Strategy):
    def __init__(self):
        self.ema9 = bt.indicators.ExponentialMovingAverage(
            self.data.close, period=9
        )
        self.order = None
        self.take_profit_order = None
        self.trailing_active = False
        self.trailing_stop = None

    def notify_order(self, order):
        if order.status == order.Completed:
            if order == self.take_profit_order:
                self.trailing_active = True
                self.trailing_stop = float(self.ema9[0])

        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            if order == self.order:
                self.order = None

    def next(self):
        if self.order:
            return

        close = self.data.close[0]

        if self.position:
            if close < 59.16:
                self.order = self.close()
                return

            if not self.trailing_active:
                if close >= 65.33:
                    self.order = self.close(size=self.position.size * 0.5)
                    self.take_profit_order = self.order
                return

            self.trailing_stop = max(self.trailing_stop, float(self.ema9[0]))
            if close < self.trailing_stop:
                self.order = self.close()
            return

        previous_bearish = self.data.close[-1] < self.data.open[-1]
        current_bullish = close > self.data.open[0]

        piercing_pattern = (
            previous_bearish
            and current_bullish
            and self.data.open[0] < self.data.low[-1]
            and close >= (self.data.open[-1] + self.data.close[-1]) / 2.0
        )

        bullish_engulfing = (
            previous_bearish
            and current_bullish
            and self.data.open[0] <= self.data.close[-1]
            and close >= self.data.open[-1]
        )

        if (piercing_pattern or bullish_engulfing) and close > 62.17:
            self.order = self.buy()
