import backtrader as bt


class EmaBollingerMeanReversion(bt.Strategy):
    params = (
        ('period', 50),
        ('devfactor', 2.0),
        ('size', 100),
    )

    def __init__(self):
        self.ema = bt.indicators.EMA(self.data.close, period=self.p.period)
        self.bband = bt.indicators.BollingerBands(
            self.data.close,
            period=self.p.period,
            devfactor=self.p.devfactor,
            movav=bt.indicators.EMA
        )
        self.order = None
        self.db = []

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            trans_type = 'bought' if order.isbuy() else 'sold'
            dt = self.data.num2date(order.executed.dt).date() if hasattr(order.executed, 'dt') else self.data.datetime.date(0)
            ticker = getattr(self.data, '_name', '') or 'AAPL'
            price = order.executed.price
            size = abs(order.executed.size)
            ema_val = self.ema[0]

            self.db.append({
                'date': dt,
                'ticker': ticker,
                'type': trans_type,
                'price': price,
                'ema': ema_val,
                'size': size
            })

        self.order = None

    def next(self):
        if self.order:
            return

        price = self.data.close[0]
        lower = self.bband.bot[0]
        upper = self.bband.top[0]

        if not self.position:
            if price <= lower:
                self.order = self.buy(size=self.p.size)
            elif price >= upper:
                self.order = self.sell(size=self.p.size)
        elif self.position.size > 0:
            if price >= upper:
                self.order = self.close()
        elif self.position.size < 0:
            if price <= lower:
                self.order = self.close()
