import math
import backtrader as bt


class BreakoutSqueezeStrategy(bt.Strategy):
    def __init__(self):
        self.sma200 = bt.indicators.SMA(self.data.close, period=200)
        self.ema20 = bt.indicators.EMA(self.data.close, period=20)
        self.ema50 = bt.indicators.EMA(self.data.close, period=50)

        self.boll = bt.indicators.BollingerBands(self.data.close, period=20, devfactor=2)
        self.bbwidth = (self.boll.top - self.boll.bot) / self.boll.mid
        self.bbwidth_sma100 = bt.indicators.SMA(self.bbwidth, period=100)

        self.sma_vol20 = bt.indicators.SMA(self.data.volume, period=20)
        self.sma_vol50 = bt.indicators.SMA(self.data.volume, period=50)
        self.rvol = self.data.volume / self.sma_vol20

        self.rsi = bt.indicators.RSI(self.data.close, period=14)
        self.atr = bt.indicators.ATR(self.data, period=14)

        self.crossover = bt.indicators.CrossOver(self.data.close, self.boll.top)

        self.order = None
        self.pending_entry_atr = None
        self.entry_price = None
        self.entry_atr = None
        self.initial_stop = None
        self.take_profit = None
        self.highest_close = None

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return

        if order.status == order.Completed:
            if order.isbuy():
                self.entry_price = order.executed.price
                self.entry_atr = self.pending_entry_atr
                self.initial_stop = self.entry_price - 1.5 * self.entry_atr
                self.take_profit = self.entry_price + 2 * self.entry_atr
                self.highest_close = self.entry_price
            else:
                self.entry_price = None
                self.entry_atr = None
                self.initial_stop = None
                self.take_profit = None
                self.highest_close = None

        if order.status in (order.Completed, order.Canceled, order.Margin, order.Rejected):
            self.order = None

    def next(self):
        if self.order:
            return

        if not self.position:
            liquidity_ok = self.sma_vol20[0] >= self.sma_vol50[0] * 0.7
            uptrend = self.data.close[0] > self.sma200[0] and self.ema20[0] > self.ema50[0]
            squeeze = self.bbwidth[0] < 0.8 * self.bbwidth_sma100[0]
            vol_spike = self.rvol[0] >= 5
            breakout = self.crossover[0] > 0
            momentum = self.rsi[0] > 55

            if liquidity_ok and uptrend and squeeze and vol_spike and breakout and momentum:
                atr_val = self.atr[0]
                risk_per_share = 1.5 * atr_val
                if risk_per_share > 0:
                    equity = self.broker.getvalue()
                    size = math.floor(0.02 * equity / risk_per_share)
                    if size > 0:
                        self.pending_entry_atr = atr_val
                        self.order = self.buy(size=size)
        else:
            self.highest_close = max(self.highest_close, self.data.close[0])

            take_profit_hit = self.data.close[0] >= self.take_profit
            breakdown = self.data.close[0] < self.ema20[0]
            trailing_stop_hit = self.data.close[0] <= (self.highest_close - 2 * self.entry_atr)
            stop_loss_hit = self.data.close[0] <= self.initial_stop

            if take_profit_hit or breakdown or trailing_stop_hit or stop_loss_hit:
                self.order = self.close()
