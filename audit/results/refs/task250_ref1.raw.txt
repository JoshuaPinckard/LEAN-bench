import backtrader as bt
import math


class MomentumStrategy(bt.Strategy):
    params = (
        ('fastperiod', 12),
        ('slowperiod', 26),
        ('signalperiod', 9),
    )

    def __init__(self):
        macd = bt.indicators.MACD(
            self.data,
            period_me1=self.p.fastperiod,
            period_me2=self.p.slowperiod,
            period_signal=self.p.signalperiod,
        )
        self.crossover = bt.indicators.CrossOver(macd.macd, macd.signal)

        self.capital = 100.0
        self.capital_dynamics = []
        self.pending_orders = []
        self.entry_order = None
        self.exit_order = None

    def next(self):
        if self.pending_orders or self.position:
            return

        if self.crossover[0] > 0:
            try:
                next_open = self.data.open[1]
            except IndexError:
                return

            if not math.isfinite(next_open) or next_open <= 0:
                return

            size = self.capital / next_open

            self.entry_order = self.buy(size=size)
            self.exit_order = self.sell(size=size, exectype=bt.Order.Close)
            self.pending_orders = [self.entry_order, self.exit_order]

    def notify_order(self, order):
        if order.status == order.Completed:
            if order is self.exit_order and self.entry_order is not None:
                entry_price = self.entry_order.executed.price
                exit_price = order.executed.price
                self.capital *= 1.0 + (exit_price - entry_price) / entry_price
                self.capital_dynamics.append(self.capital)

        if order.status in (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            if order in self.pending_orders:
                self.pending_orders.remove(order)

            if not self.pending_orders:
                self.entry_order = None
                self.exit_order = None
