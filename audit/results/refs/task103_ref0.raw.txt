import backtrader as bt


class DirectionalMovementEMA200Filter(bt.Strategy):
    params = dict(ema_period=200)

    def __init__(self):
        self.ema = bt.indicators.ExponentialMovingAverage(
            self.data.close, period=self.p.ema_period
        )

        self.totaldays = 0
        self.totalup = 0
        self.totaldown = 0
        self.totaldojis = 0
        self.totalabove = 0
        self.totalbelow = 0

        self.upabove = 0
        self.downabove = 0
        self.dojisabove = 0

        self.upbelow = 0
        self.downbelow = 0
        self.dojisbelow = 0

    def next(self):
        close = self.data.close[0]
        open_ = self.data.open[0]

        self.totaldays += 1

        if close > open_:
            self.totalup += 1
            is_up, is_down, is_doji = True, False, False
        elif close < open_:
            self.totaldown += 1
            is_up, is_down, is_doji = False, True, False
        else:
            self.totaldojis += 1
            is_up, is_down, is_doji = False, False, True

        if close > self.ema[0]:
            self.totalabove += 1
            if is_up:
                self.upabove += 1
            elif is_down:
                self.downabove += 1
            else:
                self.dojisabove += 1
        elif close < self.ema[0]:
            self.totalbelow += 1
            if is_up:
                self.upbelow += 1
            elif is_down:
                self.downbelow += 1
            else:
                self.dojisbelow += 1

    def stop(self):
        td = self.totaldays if self.totaldays else 1
        ta = self.totalabove if self.totalabove else 1
        tb = self.totalbelow if self.totalbelow else 1

        lines = []
        lines.append("Directional Movement Filtering by EMA200 - Statistical Report")
        lines.append("=" * 62)
        lines.append("Totaldays:   %d (100.00%%)" % self.totaldays)
        lines.append("Totalup:     %d (%.2f%%)" % (self.totalup, 100.0 * self.totalup / td))
        lines.append("Totaldown:   %d (%.2f%%)" % (self.totaldown, 100.0 * self.totaldown / td))
        lines.append("Totaldojis:  %d (%.2f%%)" % (self.totaldojis, 100.0 * self.totaldojis / td))
        lines.append("Totalabove:  %d (%.2f%%)" % (self.totalabove, 100.0 * self.totalabove / td))
        lines.append("Totalbelow:  %d (%.2f%%)" % (self.totalbelow, 100.0 * self.totalbelow / td))
        lines.append("-" * 62)
        lines.append("When close is above EMA200 (base = Totalabove):")
        lines.append("  Upabove:    %d (%.2f%%)" % (self.upabove, 100.0 * self.upabove / ta))
        lines.append("  Downabove:  %d (%.2f%%)" % (self.downabove, 100.0 * self.downabove / ta))
        lines.append("  Dojisabove: %d (%.2f%%)" % (self.dojisabove, 100.0 * self.dojisabove / ta))
        lines.append("-" * 62)
        lines.append("When close is below EMA200 (base = Totalbelow):")
        lines.append("  Upbelow:    %d (%.2f%%)" % (self.upbelow, 100.0 * self.upbelow / tb))
        lines.append("  Downbelow:  %d (%.2f%%)" % (self.downbelow, 100.0 * self.downbelow / tb))
        lines.append("  Dojisbelow: %d (%.2f%%)" % (self.dojisbelow, 100.0 * self.dojisbelow / tb))

        self.summary_report = "\n".join(lines)
        print(self.summary_report)
