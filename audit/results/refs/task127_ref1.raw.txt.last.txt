import backtrader as bt
import math


class MSTRSignalConflict(bt.Strategy):
    def __init__(self):
        self.sma365 = bt.indicators.SimpleMovingAverage(self.data.close, period=365)
        self.sma200 = bt.indicators.SimpleMovingAverage(self.data.close, period=200)
        self.sma10 = bt.indicators.SimpleMovingAverage(self.data.close, period=10)
        self.pending_order = None
        self.sma365_history = []

    @staticmethod
    def _percentile(values, percentile):
        ordered = sorted(values)
        index = (len(ordered) - 1) * percentile
        lower = int(math.floor(index))
        upper = int(math.ceil(index))
        if lower == upper:
            return ordered[lower]
        return ordered[lower] + (ordered[upper] - ordered[lower]) * (index - lower)

    def next(self):
        current_sma365 = self.sma365[0]
        if math.isfinite(current_sma365):
            self.sma365_history.append(current_sma365)

        if self.pending_order is not None:
            return

        bullish_cross = (
            self.data.close[-1] < self.sma365[-1]
            and self.data.close[0] > self.sma365[0]
        )
        bearish_cross = (
            self.data.close[-1] > self.sma365[-1]
            and self.data.close[0] < self.sma365[0]
        )
        sma365_above_percentile = (
            self.sma365[0] > self._percentile(self.sma365_history, 0.44)
        )
        sma200_sell_signal = self.data.close[0] < self.sma200[0]

        if not self.position:
            if bullish_cross and sma365_above_percentile:
                self.pending_order = self.buy()
        elif bearish_cross or sma200_sell_signal:
            self.pending_order = self.close()

    def notify_order(self, order):
        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            self.pending_order = None
