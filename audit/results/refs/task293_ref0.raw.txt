import backtrader as bt


class MeanReversionBBStrategy(bt.Strategy):
    params = (
        ('bb_period', 20),
        ('bb_devfactor', 2.0),
        ('bb_stop_devfactor', 3.0),
        ('rsi_period', 14),
        ('stoch_period', 14),
        ('stoch_period_dfast', 3),
        ('ema_fast_period', 5),
        ('ema_slow_period', 10),
        ('atr_period', 14),
        ('vol_sma_fast', 20),
        ('vol_sma_slow', 50),
        ('rsi_long_entry', 35),
        ('rsi_short_entry', 65),
        ('stoch_long_entry', 20),
        ('stoch_short_entry', 80),
        ('rsi_neutral', 50),
        ('bbwidth_min', 0.03),
        ('bbwidth_max', 0.20),
        ('noise_atr_mult', 3.0),
        ('atr_stop_mult', 1.5),
        ('risk_pct', 0.02),
        ('dev_low', 1.5),
        ('dev_high', 2.5),
        ('mult_low', 0.5),
        ('mult_high', 1.5),
    )

    def __init__(self):
        self.sma20 = bt.indicators.SMA(self.data.close, period=self.p.bb_period)
        self.bb2 = bt.indicators.BollingerBands(
            self.data.close, period=self.p.bb_period, devfactor=self.p.bb_devfactor)
        self.bb3 = bt.indicators.BollingerBands(
            self.data.close, period=self.p.bb_period, devfactor=self.p.bb_stop_devfactor)
        self.rsi = bt.indicators.RSI(self.data.close, period=self.p.rsi_period)
        self.stoch = bt.indicators.Stochastic(
            self.data, period=self.p.stoch_period, period_dfast=self.p.stoch_period_dfast)
        self.ema_fast = bt.indicators.EMA(self.data.close, period=self.p.ema_fast_period)
        self.ema_slow = bt.indicators.EMA(self.data.close, period=self.p.ema_slow_period)
        self.ema_cross = bt.indicators.CrossOver(self.ema_fast, self.ema_slow)
        self.rsi_cross = bt.indicators.CrossOver(self.rsi, self.p.rsi_neutral)
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.vol_sma_fast_line = bt.indicators.SMA(self.data.volume, period=self.p.vol_sma_fast)
        self.vol_sma_slow_line = bt.indicators.SMA(self.data.volume, period=self.p.vol_sma_slow)

        self.order = None
        self.stop_price = None
        self.pending_stop_distance = None

    def notify_order(self, order):
        if order.status == order.Completed:
            if order.isbuy() and self.position.size > 0 and self.pending_stop_distance is not None:
                self.stop_price = order.executed.price - self.pending_stop_distance
                self.pending_stop_distance = None
            elif order.issell() and self.position.size < 0 and self.pending_stop_distance is not None:
                self.stop_price = order.executed.price + self.pending_stop_distance
                self.pending_stop_distance = None
            self.order = None
        elif order.status in (order.Canceled, order.Margin, order.Rejected):
            self.order = None
            self.pending_stop_distance = None

    def next(self):
        if self.order:
            return

        sma20 = self.sma20[0]
        bb2_top = self.bb2.top[0]
        bb2_bot = self.bb2.bot[0]
        bb3_top = self.bb3.top[0]
        bb3_bot = self.bb3.bot[0]
        close = self.data.close[0]
        high = self.data.high[0]
        low = self.data.low[0]
        atr = self.atr[0]

        if not self.position:
            if high - low > self.p.noise_atr_mult * atr:
                return

            bbwidth = (bb2_top - bb2_bot) / sma20 if sma20 else 0.0
            if not (self.p.bbwidth_min <= bbwidth <= self.p.bbwidth_max):
                return
            if not (self.vol_sma_fast_line[0] > self.vol_sma_slow_line[0]):
                return

            std = (bb2_top - sma20) / self.p.bb_devfactor
            if std <= 0:
                return
            deviation = abs(close - sma20) / std
            multiplier = self.p.mult_low + (deviation - self.p.dev_low) * (
                self.p.mult_high - self.p.mult_low) / (self.p.dev_high - self.p.dev_low)
            multiplier = max(self.p.mult_low, min(self.p.mult_high, multiplier))

            portfolio_value = self.broker.getvalue()
            risk_amount = self.p.risk_pct * portfolio_value

            long_signal = (close < bb2_bot and self.rsi[0] < self.p.rsi_long_entry
                           and self.stoch.percK[0] < self.p.stoch_long_entry)
            short_signal = (close > bb2_top and self.rsi[0] > self.p.rsi_short_entry
                            and self.stoch.percK[0] > self.p.stoch_short_entry)

            if long_signal:
                stop_distance = max(close - bb3_bot, atr * self.p.atr_stop_mult)
                if stop_distance <= 0:
                    return
                size = int((risk_amount / stop_distance) * multiplier)
                if size >= 1:
                    self.pending_stop_distance = stop_distance
                    self.order = self.buy(size=size)
            elif short_signal:
                stop_distance = max(bb3_top - close, atr * self.p.atr_stop_mult)
                if stop_distance <= 0:
                    return
                size = int((risk_amount / stop_distance) * multiplier)
                if size >= 1:
                    self.pending_stop_distance = stop_distance
                    self.order = self.sell(size=size)
        else:
            if self.position.size > 0:
                if self.stop_price is not None and low <= self.stop_price:
                    self.order = self.close()
                    return
                if high >= sma20:
                    self.order = self.close()
                    return
                if self.ema_cross[0] < 0 or self.rsi_cross[0] > 0:
                    self.order = self.close()
                    return
            elif self.position.size < 0:
                if self.stop_price is not None and high >= self.stop_price:
                    self.order = self.close()
                    return
                if low <= sma20:
                    self.order = self.close()
                    return
                if self.ema_cross[0] > 0 or self.rsi_cross[0] < 0:
                    self.order = self.close()
                    return
