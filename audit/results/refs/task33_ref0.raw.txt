import backtrader as bt


class BullishReversalOffSellingClimax(bt.Strategy):
    params = dict(
        sma_period=200,
        atr_period=14,
        vol_avg_period=50,
        vol_spike_mult=1.5,
        climax_lookback=10,
        stop_atr_mult=1.5,
        reward_risk_mult=2.0,
        risk_pct=0.01,
        body_atr_mult=1.0,
    )

    def __init__(self):
        self.sma200 = bt.indicators.SMA(self.data.close, period=self.p.sma_period)
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.vol_avg50 = bt.indicators.SMA(self.data.volume, period=self.p.vol_avg_period)

        self.order = None
        self.stop_price = None
        self.target_price = None
        self.entry_atr = None

    def _find_climax_ago(self):
        best_ago = None
        best_low = None
        for ago in range(1, self.p.climax_lookback + 1):
            down_day = self.data.close[-ago] < self.data.close[-ago - 1]
            vol_spike = self.data.volume[-ago] >= self.p.vol_spike_mult * self.vol_avg50[-ago]
            if down_day and vol_spike:
                low = self.data.low[-ago]
                if best_low is None or low < best_low:
                    best_low = low
                    best_ago = ago
        return best_ago

    def next(self):
        if self.order:
            return

        if self.position:
            if self.stop_price is not None and self.data.low[0] <= self.stop_price:
                self.order = self.close()
            elif self.target_price is not None and self.data.high[0] >= self.target_price:
                self.order = self.close()
            elif self.data.close[0] < self.sma200[0]:
                self.order = self.close()
            return

        if self.data.close[0] <= self.sma200[0]:
            return

        climax_ago = self._find_climax_ago()
        if climax_ago is None:
            return

        climax_high = self.data.high[-climax_ago]
        if self.data.close[0] <= climax_high:
            return

        bullish_candle = self.data.close[0] > self.data.open[0]
        if not bullish_candle:
            return

        big_body = (self.data.close[0] - self.data.open[0]) >= self.p.body_atr_mult * self.atr[0]
        prior_bearish = self.data.close[-1] < self.data.open[-1]
        engulfing = (
            prior_bearish
            and self.data.open[0] <= self.data.close[-1]
            and self.data.close[0] >= self.data.open[-1]
        )
        if not (big_body or engulfing):
            return

        per_share_risk = self.p.stop_atr_mult * self.atr[0]
        if per_share_risk <= 0:
            return
        risk_amount = self.p.risk_pct * self.broker.getvalue()
        size = int(risk_amount / per_share_risk)
        if size <= 0:
            return

        self.entry_atr = self.atr[0]
        self.order = self.buy(size=size)

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status == order.Completed and order.isbuy():
            entry_price = order.executed.price
            atr_val = self.entry_atr if self.entry_atr else self.atr[0]
            self.stop_price = entry_price - self.p.stop_atr_mult * atr_val
            risk = entry_price - self.stop_price
            self.target_price = entry_price + self.p.reward_risk_mult * risk

        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None
            if not self.position:
                self.stop_price = None
                self.target_price = None
                self.entry_atr = None
