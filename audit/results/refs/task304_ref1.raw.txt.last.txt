import backtrader as bt


class TweezerBottomStrategy(bt.Strategy):
    params = (
        ("sma_period", 20),
    )

    def __init__(self):
        self.sma = bt.indicators.SimpleMovingAverage(
            self.data.close, period=self.p.sma_period
        )
        self.order = None
        self.stop_order = None
        self.broker.set_coc(True)

    def notify_order(self, order):
        if order.status in (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            if order == self.order:
                self.order = None
            if order == self.stop_order:
                self.stop_order = None

    def next(self):
        if self.order:
            return

        if self.position:
            bearish_engulfing = (
                self.data.close[0] < self.data.open[0]
                and self.data.open[0] >= self.data.close[-1]
                and self.data.close[0] <= self.data.open[-1]
            )

            if bearish_engulfing:
                if self.stop_order:
                    self.cancel(self.stop_order)
                self.order = self.close()
            return

        decline = (
            self.data.close[0] < self.sma[0]
            and self.data.close[0] < self.data.close[-5]
        )

        common_low = min(self.data.low[-2], self.data.low[-1])
        tweezer_bottom = (
            abs(self.data.low[-2] - self.data.low[-1])
            <= 0.001 * self.data.close[-1]
            and self.data.close[-2] < self.data.open[-2]
            and self.data.close[-1] > self.data.open[-1]
        )

        confirmation = self.data.close[0] > self.data.high[-1]

        if decline and tweezer_bottom and confirmation:
            self.order = self.buy(transmit=False)
            self.stop_order = self.sell(
                exectype=bt.Order.Stop,
                price=common_low * 0.98,
                parent=self.order,
                transmit=True,
            )
