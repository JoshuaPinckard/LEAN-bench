import backtrader as bt
import math
import statistics


class VolatilityRankingStrategy(bt.Strategy):
    params = (
        ('start_time', (8, 0)),
        ('end_time', (9, 0)),
    )

    def __init__(self):
        self._current_day = None
        self._log_returns = []
        self.daily_volatility = {}
        self.sorted_volatility_days = []

    def _store_current_day(self):
        if self._current_day is not None and len(self._log_returns) >= 2:
            self.daily_volatility[self._current_day] = statistics.pstdev(
                self._log_returns
            )
        self._log_returns = []

    def next(self):
        current_datetime = self.data.datetime.datetime(0)
        current_day = current_datetime.date()

        if self._current_day is None:
            self._current_day = current_day
        elif current_day != self._current_day:
            self._store_current_day()
            self._current_day = current_day

        current_time = current_datetime.time()
        start_hour, start_minute = self.p.start_time
        end_hour, end_minute = self.p.end_time

        if ((current_time.hour, current_time.minute) >= (start_hour, start_minute)
                and (current_time.hour, current_time.minute) < (end_hour, end_minute)):
            candle_open = self.data.open[0]
            candle_close = self.data.close[0]
            if candle_open > 0 and candle_close > 0:
                self._log_returns.append(math.log(candle_close / candle_open))

    def stop(self):
        self._store_current_day()
        self.sorted_volatility_days = sorted(
            self.daily_volatility.items(),
            key=lambda item: item[1],
            reverse=True,
        )
