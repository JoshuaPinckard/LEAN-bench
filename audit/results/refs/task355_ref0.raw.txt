import math
import backtrader as bt


class OutsideBarReversalStrategy(bt.Strategy):
    params = dict(
        lookback=6,
        risk_pct=0.01,
        tick_size=0.01,
    )

    def __init__(self):
        self.order = None
        self.stop_order = None
        self.stop_price = None
        self.trigger_level = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order is self.stop_order:
            if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
                self.stop_order = None
            return

        if order is self.order:
            if order.status == order.Completed:
                self.order = None
                if self.position:
                    if order.isbuy():
                        self.stop_order = self.sell(
                            exectype=bt.Order.Stop,
                            price=self.stop_price,
                            size=order.executed.size,
                        )
                    else:
                        self.stop_order = self.buy(
                            exectype=bt.Order.Stop,
                            price=self.stop_price,
                            size=abs(order.executed.size),
                        )
            else:
                self.order = None

    def next(self):
        if len(self.data) <= self.p.lookback:
            return

        if self.order:
            return

        if self.position:
            if self.position.size > 0 and self.data.close[0] < self.trigger_level:
                if self.stop_order:
                    self.cancel(self.stop_order)
                self.order = self.close()
            elif self.position.size < 0 and self.data.close[0] > self.trigger_level:
                if self.stop_order:
                    self.cancel(self.stop_order)
                self.order = self.close()
            return

        is_outside_bar = (
            self.data.high[0] > self.data.high[-1]
            and self.data.low[0] < self.data.low[-1]
        )
        if not is_outside_bar:
            return

        bullish = self.data.close[0] > self.data.open[0]
        bearish = self.data.close[0] < self.data.open[0]
        after_decline = self.data.close[0] < self.data.close[-self.p.lookback]
        after_rally = self.data.close[0] > self.data.close[-self.p.lookback]

        entry_price = self.data.close[0]
        risk_budget = self.p.risk_pct * self.broker.getvalue()

        if bullish and after_decline:
            stop_price = self.data.low[0] - self.p.tick_size
            risk_per_share = abs(entry_price - stop_price)
            if risk_per_share > 0:
                size = int(math.floor(risk_budget / risk_per_share))
                if size > 0:
                    self.trigger_level = self.data.low[0]
                    self.stop_price = stop_price
                    self.order = self.buy(size=size)
        elif bearish and after_rally:
            stop_price = self.data.high[0] + self.p.tick_size
            risk_per_share = abs(entry_price - stop_price)
            if risk_per_share > 0:
                size = int(math.floor(risk_budget / risk_per_share))
                if size > 0:
                    self.trigger_level = self.data.high[0]
                    self.stop_price = stop_price
                    self.order = self.sell(size=size)
