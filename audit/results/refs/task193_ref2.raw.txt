import backtrader as bt


class VolatilityTargetingStrategy(bt.Strategy):
    def __init__(self):
        self.atr21 = bt.indicators.ATR(self.data, period=21)
        self.atr5 = bt.indicators.ATR(self.data, period=5)
        self.order = None

    def notify_order(self, order):
        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            self.order = None

    def next(self):
        if self.order:
            return

        # Check for month rollover (first trading bar of a new month)
        if self.data.datetime.date(0).month != self.data.datetime.date(-1).month:
            close = self.data.close[0]
            if close <= 0:
                return

            atr5_pct = self.atr5[0] / close
            atr21_pct = self.atr21[0] / close

            # Filter condition: if ATR(5)/Close > 2%, target weight = 0 (cash)
            if atr5_pct > 0.02:
                target_w = 0.0
            else:
                if atr21_pct > 0:
                    target_w = min(0.4, 0.01 / atr21_pct)
                else:
                    target_w = 0.4

            self.order = self.order_target_percent(target=target_w)
