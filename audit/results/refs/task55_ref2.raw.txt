import backtrader as bt


class RSIBearishDivergenceLiquidityExhaustion(bt.Strategy):
    params = (
        ('rsi_period', 14),
        ('sma_htf_period', 200),
        ('risk_pct', 0.01),
        ('sl_pct', 0.01),
        ('tp_pct', 0.02),
    )

    def __init__(self):
        self.rsi = bt.indicators.RSI(self.data.close, period=self.p.rsi_period)
        self.sma_htf = bt.indicators.SMA(self.data.close, period=self.p.sma_htf_period)

        self.order = None
        self.prev_peak_high = None
        self.prev_peak_rsi = None
        self.prev_peak_volume = None

        self.entry_price = None
        self.sl_price = None
        self.tp_price = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            if order.issell():
                self.entry_price = order.executed.price
                self.sl_price = self.entry_price * (1.0 + self.p.sl_pct)
                self.tp_price = self.entry_price * (1.0 - self.p.tp_pct)
            elif order.isbuy():
                self.entry_price = None
                self.sl_price = None
                self.tp_price = None

        self.order = None

    def next(self):
        if self.order:
            return

        if self.position:
            if self.rsi[0] > 50:
                self.order = self.close()
                return
            if self.sl_price and self.data.high[0] >= self.sl_price:
                self.order = self.close()
                return
            if self.tp_price and self.data.low[0] <= self.tp_price:
                self.order = self.close()
                return
            return

        if len(self.data) < 3:
            return

        is_peak = (self.data.high[-1] > self.data.high[-2]) and (self.data.high[-1] > self.data.high[0])

        if is_peak:
            curr_peak_high = self.data.high[-1]
            curr_peak_rsi = self.rsi[-1]
            curr_peak_volume = self.data.volume[-1]

            if self.prev_peak_high is not None:
                cond1 = curr_peak_high > self.prev_peak_high
                cond2 = curr_peak_rsi < self.prev_peak_rsi
                cond3 = curr_peak_volume < self.prev_peak_volume
                htf_upward = self.data.close[0] > self.sma_htf[0]
                cond4 = not htf_upward

                if cond1 and cond2 and cond3 and cond4:
                    risk_amount = self.broker.getvalue() * self.p.risk_pct
                    risk_per_share = self.data.close[0] * self.p.sl_pct
                    if risk_per_share > 0:
                        size = int(risk_amount / risk_per_share)
                        max_size = int(self.broker.getcash() / self.data.close[0])
                        size = min(size, max_size)
                        if size > 0:
                            self.order = self.sell(size=size)

            self.prev_peak_high = curr_peak_high
            self.prev_peak_rsi = curr_peak_rsi
            self.prev_peak_volume = curr_peak_volume
