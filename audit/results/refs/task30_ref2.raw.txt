import backtrader as bt


class SequentialBuyOrderStrategy(bt.Strategy):
    params = (
        ("pct_first", 0.50),
        ("pct_second", 0.50),
    )

    def __init__(self):
        self.order1 = None
        self.order2 = None
        self.size2 = 0

    def notify_order(self, order):
        if order.status in [order.Completed]:
            if order == self.order1:
                self.order1 = None
                cash = self.broker.getcash()
                if self.data.close[0] > 0:
                    exec_size2 = min(
                        self.size2, int(cash / self.data.close[0])
                    )
                    if exec_size2 > 0:
                        self.order2 = self.buy(
                            exectype=bt.Order.Market, size=exec_size2
                        )
            elif order == self.order2:
                self.order2 = None
        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            if order == self.order1:
                self.order1 = None
            if order == self.order2:
                self.order2 = None

    def next(self):
        if (
            self.position.size > 0
            or self.order1 is not None
            or self.order2 is not None
        ):
            return

        price = self.data.close[0]
        if price <= 0:
            return

        cash = self.broker.getcash()
        size1 = int((cash * self.p.pct_first) / price)
        size2 = int((cash * self.p.pct_second) / price)

        if size1 > 0 and size2 > 0 and (size1 + size2) * price <= cash:
            self.size2 = size2
            self.order1 = self.buy(
                exectype=bt.Order.Limit, price=price, size=size1
            )
