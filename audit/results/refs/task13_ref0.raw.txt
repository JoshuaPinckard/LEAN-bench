import backtrader as bt


class PyramidingInTrend(bt.Strategy):
    params = dict(
        ema_period=50,
        rsi_period=14,
        max_lots=4,
        size_pct=0.01,
        partial_close_pct=0.25,
        tp_close_pct=0.50,
        tp_threshold=0.05,
    )

    def __init__(self):
        self.ema = bt.indicators.ExponentialMovingAverage(
            self.data.close, period=self.p.ema_period)
        self.rsi = bt.indicators.RSI(
            self.data.close, period=self.p.rsi_period)

        self.broker.set_coc(True)

        self.order = None
        self.lot_count = 0

        self.rsi_dipped_below_50 = False
        self.rsi_reached_70 = False
        self.rsi_above_60 = False

        self.partial_exit_done = False
        self.tp_exit_done = False

    def notify_order(self, order):
        if order.status in (order.Completed, order.Canceled, order.Margin, order.Rejected):
            self.order = None

    def notify_trade(self, trade):
        if trade.isclosed:
            self.lot_count = 0
            self.rsi_dipped_below_50 = False
            self.rsi_reached_70 = False
            self.rsi_above_60 = False
            self.partial_exit_done = False
            self.tp_exit_done = False

    def _entry_size(self):
        return (self.p.size_pct * self.broker.startingcash) / self.data.close[0]

    def next(self):
        if self.order is not None:
            return

        close = self.data.close[0]
        ema = self.ema[0]
        rsi = self.rsi[0]

        if rsi < 50:
            self.rsi_dipped_below_50 = True
        if rsi > 60:
            self.rsi_above_60 = True
        if rsi >= 70:
            self.rsi_reached_70 = True

        if not self.position:
            crossed_up = False
            if self.rsi_dipped_below_50 and rsi > 50:
                crossed_up = True
                self.rsi_dipped_below_50 = False

            if crossed_up and close > ema:
                self.order = self.buy(size=self._entry_size())
                self.lot_count = 1
            return

        exit_pullback = False
        if self.rsi_above_60 and rsi < 40:
            exit_pullback = True
            self.rsi_above_60 = False

        if close < ema or exit_pullback:
            self.order = self.close()
            return

        pos_price = self.position.price
        pos_size = self.position.size

        if not self.partial_exit_done and close >= pos_price:
            self.order = self.sell(size=pos_size * self.p.partial_close_pct)
            self.partial_exit_done = True
            return

        if self.partial_exit_done and not self.tp_exit_done:
            profit_pct = (close - pos_price) / pos_price
            if profit_pct >= self.p.tp_threshold:
                self.order = self.sell(size=pos_size * self.p.tp_close_pct)
                self.tp_exit_done = True
                return

        pyramid_trigger = False
        if self.rsi_reached_70 and rsi <= 60:
            pyramid_trigger = True
            self.rsi_reached_70 = False

        if pyramid_trigger and self.lot_count < self.p.max_lots and close > ema:
            self.order = self.buy(size=self._entry_size())
            self.lot_count += 1
