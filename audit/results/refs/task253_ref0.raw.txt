import backtrader as bt


class PanicEuphoriaStrategy(bt.Strategy):
    def __init__(self):
        self.sma200 = bt.indicators.SMA(self.data.close, period=200)
        self.bb = bt.indicators.BollingerBands(self.data.close, period=20, devfactor=2)
        self.rsi = bt.indicators.RSI(self.data.close, period=14)
        self.atr = bt.indicators.ATR(self.data, period=14)
        self.atr_sma = bt.indicators.SMA(self.atr, period=50)
        self.bb_width = self.bb.lines.top - self.bb.lines.bot
        self.width_sma = bt.indicators.SMA(self.bb_width, period=50)
        self.order = None

    def notify_order(self, order):
        if order.status in (order.Completed, order.Canceled, order.Margin, order.Rejected):
            self.order = None

    def next(self):
        if self.order:
            return

        close = self.data.close[0]
        uptrend = close > self.sma200[0]
        panic = (
            close < self.bb.lines.bot[0]
            and self.rsi[0] <= 30
            and (self.atr[0] > self.atr_sma[0] or self.bb_width[0] > self.width_sma[0])
        )
        euphoria = (
            close > self.bb.lines.top[0]
            and self.rsi[0] >= 65
            and (self.atr[0] < self.atr_sma[0] or self.bb_width[0] < self.width_sma[0])
        )

        if not self.position:
            if uptrend and panic:
                self.order = self.order_target_percent(target=1.0)
        else:
            if euphoria:
                self.order = self.close()
