import backtrader as bt


class CupAndHandleQuarterlyStrategy(bt.Strategy):
    def __init__(self):
        self.order = None
        self.target_order = None
        self.stop_requested = False

        self.current_week = None
        self.current_quarter = None
        self.week_bar = None
        self.quarter_bar = None

        self.weekly_closes = []
        self.quarterly_bars = []
        self.all_quarterly_closes = []

        self.last_week_close = None
        self.last_week_sma20 = None
        self.breakout_candidate = None

    def _week_key(self, date):
        iso = date.isocalendar()
        return iso.year, iso.week

    def _quarter_key(self, date):
        return date.year, (date.month - 1) // 3 + 1

    def _new_bar(self, close, high, low):
        return {"close": close, "high": high, "low": low}

    def _update_bar(self, bar, close, high, low):
        bar["close"] = close
        bar["high"] = max(bar["high"], high)
        bar["low"] = min(bar["low"], low)

    def _finalize_week(self):
        close = self.week_bar["close"]
        self.weekly_closes.append(close)
        self.last_week_close = close

        if len(self.weekly_closes) >= 20:
            self.last_week_sma20 = sum(self.weekly_closes[-20:]) / 20.0

        self._try_enter()

    def _finalize_quarter(self):
        quarter = self.quarter_bar
        history = self.quarterly_bars

        ath = (
            not self.all_quarterly_closes
            or quarter["close"] >= max(self.all_quarterly_closes)
        )

        if len(history) >= 6:
            left_rim = history[-6]
            cup_section = history[-5:-2]
            right_rim = history[-2]
            handle = history[-1]

            cup_low = min(bar["low"] for bar in cup_section)
            resistance = max(left_rim["high"], right_rim["high"])
            handle_low = handle["low"]

            cup_and_handle_formed = (
                cup_low < left_rim["low"]
                and cup_low < right_rim["low"]
                and handle_low >= cup_low
                and handle_low < right_rim["high"]
            )

            sma200_ready = len(history) >= 200
            if sma200_ready:
                sma200 = sum(bar["close"] for bar in history[-200:]) / 200.0
                bullish_trend = quarter["close"] > sma200
            else:
                bullish_trend = False

            breakout = quarter["close"] > resistance

            if cup_and_handle_formed and breakout and bullish_trend and ath:
                self.breakout_candidate = {
                    "breakout": resistance,
                    "target": resistance + (resistance - cup_low),
                    "stop": handle_low,
                }

        history.append(quarter)
        self.all_quarterly_closes.append(quarter["close"])
        self._try_enter()

    def _try_enter(self):
        if (
            self.order is not None
            or self.target_order is not None
            or self.position
            or self.breakout_candidate is None
            or self.last_week_sma20 is None
        ):
            return

        if self.last_week_close <= self.last_week_sma20:
            return

        stop = self.breakout_candidate["stop"]
        reference_entry = self.data.close[0]
        loss_per_share = reference_entry - stop

        if loss_per_share <= 0:
            self.breakout_candidate = None
            return

        capital = self.broker.getvalue()
        risk_size = int((capital * 0.02) / loss_per_share)
        affordable_size = int(self.broker.getcash() / (reference_entry * 1.001))
        size = min(risk_size, affordable_size)

        if size > 0:
            self.order = self.buy(size=size)
        self.breakout_candidate = None

    def next(self):
        date = self.data.datetime.date(0)
        close = self.data.close[0]
        high = self.data.high[0]
        low = self.data.low[0]

        week_key = self._week_key(date)
        quarter_key = self._quarter_key(date)

        if self.current_week is None:
            self.current_week = week_key
            self.current_quarter = quarter_key
            self.week_bar = self._new_bar(close, high, low)
            self.quarter_bar = self._new_bar(close, high, low)
            return

        week_changed = week_key != self.current_week
        quarter_changed = quarter_key != self.current_quarter

        if week_changed:
            self._finalize_week()
            self.current_week = week_key
            self.week_bar = self._new_bar(close, high, low)
        else:
            self._update_bar(self.week_bar, close, high, low)

        if quarter_changed:
            self._finalize_quarter()
            self.current_quarter = quarter_key
            self.quarter_bar = self._new_bar(close, high, low)
        else:
            self._update_bar(self.quarter_bar, close, high, low)

        if (
            self.position
            and self.target_order is not None
            and self.last_week_close is not None
            and self.last_week_close < self.breakout_candidate["stop"]
            if self.breakout_candidate is not None
            else False
        ):
            self.stop_requested = True
            self.cancel(self.target_order)

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order is self.order:
            if order.status == order.Completed and order.isbuy():
                self.order = None
                self.target_order = self.sell(
                    size=self.position.size,
                    exectype=bt.Order.Limit,
                    price=self.breakout_candidate["target"]
                    if self.breakout_candidate is not None
                    else order.executed.price,
                )
            elif order.status in [
                order.Completed,
                order.Canceled,
                order.Margin,
                order.Rejected,
            ]:
                self.order = None

        elif order is self.target_order:
            if order.status in [
                order.Completed,
                order.Canceled,
                order.Margin,
                order.Rejected,
            ]:
                self.target_order = None
                if self.stop_requested and self.position:
                    self.stop_requested = False
                    self.order = self.close()
