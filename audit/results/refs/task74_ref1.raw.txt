import backtrader as bt
import math


class DynamicTrailingStopStrategy(bt.Strategy):
    params = (
        ("size", 0.1),
        ("stop_distance", 10.0),
        ("take_profit_distance", 50.0),
    )

    def __init__(self):
        self.entry_order = None
        self.stop_order = None
        self.take_profit_order = None
        self.entry_price = None
        self.stop_price = None

    def next(self):
        if self.entry_order:
            return

        if not self.position:
            self.entry_order = self.buy(size=self.p.size)
            return

        if not self.entry_price or self.stop_order:
            if self.entry_price:
                profit_points = self.data.close[0] - self.entry_price
                reached_steps = math.floor(profit_points / self.p.stop_distance)
                if reached_steps > 0:
                    new_stop = self.entry_price + (
                        reached_steps * self.p.stop_distance
                    )
                    if new_stop > self.stop_price and new_stop < self.data.close[0]:
                        self.stop_order = self.sell(
                            size=self.position.size,
                            exectype=bt.Order.Stop,
                            price=new_stop,
                            oco=self.take_profit_order,
                        )
                        self.stop_price = new_stop

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return

        if order.status == order.Completed:
            if order == self.entry_order:
                self.entry_price = order.executed.price
                self.stop_price = self.entry_price - self.p.stop_distance
                self.stop_order = self.sell(
                    size=self.position.size,
                    exectype=bt.Order.Stop,
                    price=self.stop_price,
                )
                self.take_profit_order = self.sell(
                    size=self.position.size,
                    exectype=bt.Order.Limit,
                    price=self.entry_price + self.p.take_profit_distance,
                    oco=self.stop_order,
                )
                self.entry_order = None
                return

            if order == self.stop_order or order == self.take_profit_order:
                self.stop_order = None
                self.take_profit_order = None
                self.entry_price = None
                self.stop_price = None
                return

        if order.status in (
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            if order == self.entry_order:
                self.entry_order = None
            elif order == self.stop_order:
                self.stop_order = None
            elif order == self.take_profit_order:
                self.take_profit_order = None
