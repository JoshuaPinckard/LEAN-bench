import backtrader as bt


class MeanReversionStrategy(bt.Strategy):
    params = (
        ('risk_fraction', 0.01),
    )

    def __init__(self):
        self.roc21 = bt.indicators.RateOfChange(self.data.close, period=21)
        self.bbands = bt.indicators.BollingerBands(
            self.roc21, period=252, devfactor=2
        )
        self.atr = bt.indicators.ATR(self.data, period=14)

        self.order = None
        self.stop_order = None
        self.exit_order = None
        self.entry_bar = None
        self.entry_atr = None

    def next(self):
        if len(self) < 273:
            return

        if self.position:
            if (
                self.entry_bar is not None
                and len(self) >= self.entry_bar + 5
                and self.exit_order is None
            ):
                if self.stop_order is not None:
                    self.cancel(self.stop_order)
                self.exit_order = self.close()
            return

        if self.order is not None or self.exit_order is not None:
            return

        if self.data.datetime.date(0).weekday() != 0:
            return

        atr_value = self.atr[0]
        if atr_value <= 0:
            return

        size = (self.broker.getvalue() * self.p.risk_fraction) / (2 * atr_value)

        if self.roc21[0] < self.bbands.bot[0]:
            self.entry_atr = atr_value
            self.order = self.buy(size=size)
        elif self.roc21[0] > self.bbands.top[0]:
            self.entry_atr = atr_value
            self.order = self.sell(size=size)

    def notify_order(self, order):
        if order.status in (
            order.Submitted,
            order.Accepted,
        ):
            return

        if order.status == order.Completed:
            if order == self.order:
                self.entry_bar = len(self)

                if order.isbuy():
                    self.stop_order = self.sell(
                        exectype=bt.Order.Stop,
                        price=order.executed.price - (2 * self.entry_atr),
                        size=order.executed.size,
                    )
                else:
                    self.stop_order = self.buy(
                        exectype=bt.Order.Stop,
                        price=order.executed.price + (2 * self.entry_atr),
                        size=abs(order.executed.size),
                    )

                self.order = None
                self.entry_atr = None
            elif order == self.stop_order:
                self.stop_order = None
                self.entry_bar = None
            elif order == self.exit_order:
                self.exit_order = None
                self.entry_bar = None
                self.stop_order = None

        elif order.status in (
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            if order == self.order:
                self.order = None
                self.entry_atr = None
            elif order == self.stop_order:
                self.stop_order = None
            elif order == self.exit_order:
                self.exit_order = None
