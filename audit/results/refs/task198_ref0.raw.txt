import backtrader as bt


class BreakoutStochasticStrategy(bt.Strategy):
    params = (
        ('stoch_period', 260),
        ('entry_level', 85),
        ('exit_level', 1),
    )

    def __init__(self):
        self.highest = bt.indicators.Highest(self.data.high, period=self.p.stoch_period)
        self.lowest = bt.indicators.Lowest(self.data.low, period=self.p.stoch_period)
        self.prev_k = None
        self.order = None

    def notify_order(self, order):
        if order.status in (order.Completed, order.Canceled, order.Margin, order.Rejected):
            self.order = None

    def next(self):
        highest_high = self.highest[0]
        lowest_low = self.lowest[0]
        denom = highest_high - lowest_low
        k = (self.data.close[0] - lowest_low) / denom * 100.0 if denom != 0 else 0.0

        if self.order:
            self.prev_k = k
            return

        if self.prev_k is not None:
            if not self.position:
                if self.prev_k <= self.p.entry_level and k > self.p.entry_level:
                    size = int(self.broker.getcash() / self.data.close[0])
                    if size > 0:
                        self.order = self.buy(size=size)
            else:
                if k < self.p.exit_level:
                    self.order = self.close()

        self.prev_k = k
