import backtrader as bt


class OpeningRangeBreakoutStrategy(bt.Strategy):
    params = dict(
        or_bars=4,
        rvol_period=20,
        rvol_threshold=1.5,
        atr_period=14,
        atr_mult=1.5,
        rsi_period=14,
        rsi_level=50,
        session_start=(9, 30),
        session_end=(16, 0),
    )

    def __init__(self):
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.rsi = bt.indicators.RSI(self.data, period=self.p.rsi_period)
        self.vol_sma = bt.indicators.SMA(self.data.volume, period=self.p.rvol_period)

        self.order = None
        self.stop_order = None

        self.current_day = None
        self.bar_of_day = 0
        self.or_high = None
        self.or_low = None
        self.or_ready = False

    def _update_session(self):
        dt = self.data.datetime.datetime(0)
        bar_date = dt.date()
        bar_time = dt.time()

        if bar_date != self.current_day:
            self.current_day = bar_date
            self.bar_of_day = 0
            self.or_high = None
            self.or_low = None
            self.or_ready = False

        self.bar_of_day += 1

        in_session = self.p.session_start <= (bar_time.hour, bar_time.minute) < self.p.session_end

        forming_or = False
        if in_session and self.bar_of_day <= self.p.or_bars:
            forming_or = True
            if self.or_high is None:
                self.or_high = self.data.high[0]
                self.or_low = self.data.low[0]
            else:
                self.or_high = max(self.or_high, self.data.high[0])
                self.or_low = min(self.or_low, self.data.low[0])
            if self.bar_of_day == self.p.or_bars:
                self.or_ready = True

        return in_session, forming_or

    def prenext(self):
        self._update_session()

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return

        if order.status == order.Completed:
            if order is self.stop_order:
                self.stop_order = None
            elif order is self.order:
                self.order = None
                if self.position and not self.stop_order:
                    trail = self.p.atr_mult * self.atr[0]
                    if self.position.size > 0:
                        self.stop_order = self.sell(
                            exectype=bt.Order.StopTrail,
                            trailamount=trail,
                            size=self.position.size,
                        )
                    else:
                        self.stop_order = self.buy(
                            exectype=bt.Order.StopTrail,
                            trailamount=trail,
                            size=abs(self.position.size),
                        )
            return

        if order.status in (order.Canceled, order.Margin, order.Rejected):
            if order is self.stop_order:
                self.stop_order = None
            elif order is self.order:
                self.order = None

    def next(self):
        in_session, forming_or = self._update_session()

        if not in_session:
            return

        if self.order:
            return

        rsi_cross_down = self.rsi[-1] >= self.p.rsi_level and self.rsi[0] < self.p.rsi_level
        rsi_cross_up = self.rsi[-1] <= self.p.rsi_level and self.rsi[0] > self.p.rsi_level
        rvol = self.data.volume[0] / self.vol_sma[0]

        if self.position:
            if self.position.size > 0:
                exit_signal = rsi_cross_down
                if not forming_or and self.or_ready:
                    if self.data.close[0] < self.or_low and rvol > self.p.rvol_threshold:
                        exit_signal = True
            else:
                exit_signal = rsi_cross_up
                if not forming_or and self.or_ready:
                    if self.data.close[0] > self.or_high and rvol > self.p.rvol_threshold:
                        exit_signal = True

            if exit_signal:
                if self.stop_order:
                    self.cancel(self.stop_order)
                    self.stop_order = None
                self.order = self.close()
            return

        if forming_or or not self.or_ready:
            return

        long_signal = self.data.close[0] > self.or_high and rvol > self.p.rvol_threshold
        short_signal = self.data.close[0] < self.or_low and rvol > self.p.rvol_threshold

        if long_signal:
            self.order = self.buy()
        elif short_signal:
            self.order = self.sell()
