import backtrader as bt


class ReferenceStrategy(bt.Strategy):
    def __init__(self):
        self.sma = bt.indicators.SimpleMovingAverage(self.data.close, period=20)
        self.swing_high = bt.indicators.Highest(self.data.high(-1), period=5)
        self.swing_low = bt.indicators.Lowest(self.data.low(-1), period=5)
        self.order = None

    def notify_order(self, order):
        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            self.order = None

    def next(self):
        if self.order:
            return

        crossed_above_high = (
            self.data.close[-1] <= self.swing_high[-1]
            and self.data.close[0] > self.swing_high[0]
        )
        crossed_below_low = (
            self.data.close[-1] >= self.swing_low[-1]
            and self.data.close[0] < self.swing_low[0]
        )

        if not self.position:
            if self.data.close[0] > self.sma[0] and crossed_above_high:
                self.order = self.buy()
            elif self.data.close[0] < self.sma[0] and crossed_below_low:
                self.order = self.sell()
        elif self.position.size > 0:
            if crossed_below_low:
                self.order = self.close()
        elif self.position.size < 0:
            if crossed_above_high:
                self.order = self.close()
