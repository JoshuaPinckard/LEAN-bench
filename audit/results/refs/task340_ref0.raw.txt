import backtrader as bt


class SessionVWAP(bt.Indicator):
    lines = ('vwap',)

    def __init__(self):
        self._session_date = None
        self._cum_pv = 0.0
        self._cum_vol = 0.0
        self.addminperiod(1)

    def next(self):
        dt = self.data.datetime.date(0)
        if dt != self._session_date:
            self._session_date = dt
            self._cum_pv = 0.0
            self._cum_vol = 0.0
        typical_price = (self.data.high[0] + self.data.low[0] + self.data.close[0]) / 3.0
        self._cum_pv += typical_price * self.data.volume[0]
        self._cum_vol += self.data.volume[0]
        self.lines.vwap[0] = self._cum_pv / self._cum_vol


class VWAPWilliamsRMeanReversion(bt.Strategy):
    params = (
        ('wr_period', 14),
        ('stop_distance', 2.0),
    )

    def __init__(self):
        self.vwap = SessionVWAP(self.data)
        self.wr = bt.indicators.WilliamsR(self.data, period=self.p.wr_period)
        self.order = None
        self.stop_order = None

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return

        if order.status == order.Completed:
            if order == self.order:
                self.order = None
                if order.isbuy() and self.position.size > 0 and self.stop_order is None:
                    entry_price = order.executed.price
                    entry_vwap = self.vwap[0]
                    stop_price = max(entry_price - self.p.stop_distance,
                                      entry_vwap - self.p.stop_distance)
                    self.stop_order = self.sell(exectype=bt.Order.Stop,
                                                 price=stop_price,
                                                 size=self.position.size)
                elif order.issell() and self.position.size < 0 and self.stop_order is None:
                    entry_price = order.executed.price
                    entry_vwap = self.vwap[0]
                    stop_price = min(entry_price + self.p.stop_distance,
                                      entry_vwap + self.p.stop_distance)
                    self.stop_order = self.buy(exectype=bt.Order.Stop,
                                                price=stop_price,
                                                size=abs(self.position.size))
            elif order == self.stop_order:
                self.stop_order = None
        elif order.status in (order.Canceled, order.Margin, order.Rejected):
            if order == self.order:
                self.order = None
            elif order == self.stop_order:
                self.stop_order = None

    def next(self):
        if self.order:
            return

        if not self.position:
            wr_cross_down = self.wr[-1] >= -80 and self.wr[0] < -80
            wr_cross_up = self.wr[-1] <= -20 and self.wr[0] > -20

            if self.data.close[0] < self.vwap[0] and wr_cross_down:
                self.order = self.buy()
            elif self.data.close[0] > self.vwap[0] and wr_cross_up:
                self.order = self.sell()
        else:
            cross_above = self.data.close[-1] <= self.vwap[-1] and self.data.close[0] > self.vwap[0]
            cross_below = self.data.close[-1] >= self.vwap[-1] and self.data.close[0] < self.vwap[0]

            if self.position.size > 0 and cross_above:
                if self.stop_order:
                    self.cancel(self.stop_order)
                self.order = self.close()
            elif self.position.size < 0 and cross_below:
                if self.stop_order:
                    self.cancel(self.stop_order)
                self.order = self.close()
