import backtrader as bt
import math
import statistics
import collections


class PortfolioOptimizationTransactionCosts(bt.Strategy):
    params = (
        ('risk_coefficient', 2.5),
        ('turnover_limit', 0.20),
        ('transaction_cost_rate', 0.001),
        ('lookback', 60),
    )

    def __init__(self):
        self.order = None
        self.daily_returns = collections.deque(maxlen=self.p.lookback)
        self.portfolio_values = []
        self.rebalance_log = []
        self.transaction_log = []
        self.last_rebalance_month = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return
        if order.status == order.Completed:
            txn = {
                'date': self.data.datetime.date(0).isoformat(),
                'side': 'BUY' if order.isbuy() else 'SELL',
                'size': order.executed.size,
                'price': order.executed.price,
                'value': order.executed.value,
                'commission': order.executed.comm,
            }
            self.transaction_log.append(txn)
            print(f"[TRANSACTION] {txn}")
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def _compute_performance_metrics(self):
        if len(self.portfolio_values) < 2:
            return 0.0, 0.0, 0.0
        rets = []
        for i in range(1, len(self.portfolio_values)):
            prev = self.portfolio_values[i - 1]
            curr = self.portfolio_values[i]
            if prev != 0:
                rets.append((curr - prev) / prev)
        if len(rets) < 2:
            return 0.0, 0.0, 0.0
        total_return = (self.portfolio_values[-1] - self.portfolio_values[0]) / self.portfolio_values[0]
        vol = statistics.stdev(rets) * math.sqrt(252)
        mean_ret = statistics.mean(rets)
        sharpe = (mean_ret * 252) / vol if vol > 0 else 0.0
        return total_return, vol, sharpe

    def _rebalance(self):
        if self.order is not None:
            return
        if len(self.daily_returns) < 2:
            return

        port_value = self.broker.getvalue()
        if port_value <= 0:
            return

        position_value = self.position.size * self.data.close[0]
        current_weight = position_value / port_value

        alpha = statistics.mean(self.daily_returns)
        variance = statistics.variance(self.daily_returns)

        lo = max(0.0, current_weight - self.p.turnover_limit)
        hi = min(1.0, current_weight + self.p.turnover_limit)

        if variance <= 0:
            if alpha > self.p.transaction_cost_rate:
                target_weight = hi
            elif alpha < -self.p.transaction_cost_rate:
                target_weight = lo
            else:
                target_weight = current_weight
        else:
            w_hat = alpha / (2.0 * self.p.risk_coefficient * variance)
            threshold = self.p.transaction_cost_rate / (2.0 * self.p.risk_coefficient * variance)
            if w_hat > current_weight + threshold:
                target_weight = w_hat - threshold
            elif w_hat < current_weight - threshold:
                target_weight = w_hat + threshold
            else:
                target_weight = current_weight
            target_weight = min(max(target_weight, lo), hi)

        target_weight = min(max(target_weight, 0.0), 1.0)

        turnover = abs(target_weight - current_weight)
        est_txn_cost = self.p.transaction_cost_rate * turnover * port_value
        total_risk = variance * (target_weight ** 2)
        risk_contribution_pct = 100.0 if total_risk > 0 else 0.0
        total_return, vol, sharpe = self._compute_performance_metrics()

        report = {
            'date': self.data.datetime.date(0).isoformat(),
            'current_weight': current_weight,
            'target_weight': target_weight,
            'alpha_forecast': alpha,
            'variance_forecast': variance,
            'turnover': turnover,
            'estimated_transaction_cost': est_txn_cost,
            'portfolio_return': total_return,
            'portfolio_volatility': vol,
            'sharpe_ratio': sharpe,
            'total_portfolio_risk': total_risk,
            'aapl_risk_contribution_pct': risk_contribution_pct,
        }
        self.rebalance_log.append(report)
        print(f"[REBALANCE REPORT] {report}")

        self.order = self.order_target_percent(target=target_weight)

    def next(self):
        self.portfolio_values.append(self.broker.getvalue())
        if len(self) > 1:
            ret = (self.data.close[0] - self.data.close[-1]) / self.data.close[-1]
            self.daily_returns.append(ret)

        dt = self.data.datetime.date(0)
        if self.last_rebalance_month != dt.month:
            self.last_rebalance_month = dt.month
            self._rebalance()

    def stop(self):
        total_return, vol, sharpe = self._compute_performance_metrics()
        final_position_value = self.position.size * self.data.close[0]
        final_value = self.broker.getvalue()
        final_weight = final_position_value / final_value if final_value > 0 else 0.0
        total_txn_costs = sum(t['commission'] for t in self.transaction_log)
        final_report = {
            'final_weight': final_weight,
            'total_return': total_return,
            'volatility': vol,
            'sharpe_ratio': sharpe,
            'total_transaction_costs': total_txn_costs,
            'num_rebalances': len(self.rebalance_log),
            'num_transactions': len(self.transaction_log),
        }
        print(f"[FINAL REPORT] {final_report}")
