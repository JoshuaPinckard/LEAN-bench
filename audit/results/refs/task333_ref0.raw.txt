import math
import backtrader as bt


class HurstExponent(bt.Indicator):
    lines = ('hurst',)
    params = (('period', 100),)

    def __init__(self):
        self.addminperiod(self.p.period)

    def next(self):
        n = self.p.period
        prices = self.data.get(size=n)
        mean_price = sum(prices) / n
        deviations = [p - mean_price for p in prices]

        cumulative = []
        running = 0.0
        for d in deviations:
            running += d
            cumulative.append(running)
        r = max(cumulative) - min(cumulative)

        variance = sum(d * d for d in deviations) / n
        s = math.sqrt(variance)

        if s == 0.0 or r == 0.0:
            self.lines.hurst[0] = 0.5
        else:
            self.lines.hurst[0] = math.log(r / s) / math.log(n)


class HurstMeanReversion(bt.Strategy):
    params = (
        ('hurst_period', 100),
        ('sma_period', 20),
        ('stop_pct', 0.02),
    )

    def __init__(self):
        self.sma = bt.indicators.SMA(self.data.close, period=self.p.sma_period)
        self.hurst = HurstExponent(self.data.close, period=self.p.hurst_period)
        self.crossover = bt.indicators.CrossOver(self.data.close, self.sma)

        self.entry_order = None
        self.exit_order = None
        self.stop_order = None

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return

        if order.status == order.Completed:
            if order == self.entry_order:
                entry_price = order.executed.price
                size = abs(order.executed.size)
                if order.isbuy():
                    stop_price = entry_price * (1.0 - self.p.stop_pct)
                    self.stop_order = self.sell(
                        exectype=bt.Order.Stop, price=stop_price, size=size)
                else:
                    stop_price = entry_price * (1.0 + self.p.stop_pct)
                    self.stop_order = self.buy(
                        exectype=bt.Order.Stop, price=stop_price, size=size)
                self.entry_order = None
            elif order == self.stop_order:
                self.stop_order = None
                self.exit_order = None
            elif order == self.exit_order:
                if self.stop_order is not None:
                    self.cancel(self.stop_order)
                self.exit_order = None
        elif order.status in (order.Canceled, order.Margin, order.Rejected):
            if order == self.entry_order:
                self.entry_order = None
            elif order == self.stop_order:
                self.stop_order = None
            elif order == self.exit_order:
                self.exit_order = None

    def next(self):
        if self.entry_order or self.exit_order:
            return

        if not self.position:
            if self.hurst[0] < 0.5 and self.data.close[0] < self.sma[0]:
                self.entry_order = self.buy()
            elif self.hurst[0] < 0.5 and self.data.close[0] > self.sma[0]:
                self.entry_order = self.sell()
        else:
            if self.position.size > 0:
                if self.crossover[0] > 0 or self.hurst[0] >= 0.5:
                    if self.stop_order is not None:
                        self.cancel(self.stop_order)
                    self.exit_order = self.close()
            elif self.position.size < 0:
                if self.crossover[0] < 0 or self.hurst[0] >= 0.5:
                    if self.stop_order is not None:
                        self.cancel(self.stop_order)
                    self.exit_order = self.close()
