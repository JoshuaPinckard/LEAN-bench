import backtrader as bt


class BollingerBandsBreakoutTradingForBitcoin(bt.Strategy):
    params = (
        ("period", 20),
        ("devfactor", 2),
        ("take_profit", 0.06),
        ("stop_loss", 0.03),
    )

    def __init__(self):
        self.bollinger = bt.indicators.BollingerBands(
            self.data.close,
            period=self.p.period,
            devfactor=self.p.devfactor,
        )
        self.order = None
        self.entry_price = None

    def notify_order(self, order):
        if order.status == order.Completed:
            if order.isbuy():
                self.entry_price = order.executed.price
            elif order.issell():
                self.entry_price = None

        if order.status in (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            self.order = None

    def next(self):
        if self.order:
            return

        if not self.position:
            if self.data.close[0] > self.bollinger.top[0]:
                self.order = self.buy(exectype=bt.Order.Close)
        elif (
            self.data.close[0] >= self.entry_price * (1.0 + self.p.take_profit)
            or self.data.close[0] <= self.entry_price * (1.0 - self.p.stop_loss)
        ):
            self.order = self.close()
