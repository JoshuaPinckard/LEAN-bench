import backtrader as bt
import math


class SwingMomentumStrategy(bt.Strategy):
    params = (
        ('commission_rate', 0.001),
    )

    def __init__(self):
        self.sma200 = bt.indicators.SMA(self.data.close, period=200)
        self.ema20 = bt.indicators.EMA(self.data.close, period=20)
        self.ema50 = bt.indicators.EMA(self.data.close, period=50)
        self.rsi14 = bt.indicators.RSI(self.data.close, period=14)
        self.atr14 = bt.indicators.ATR(self.data, period=14)
        self.volume_sma20 = bt.indicators.SMA(self.data.volume, period=20)

        self.order = None
        self.added_once = False
        self.initial_stop = None
        self.highest_close_since_entry = None

    def notify_order(self, order):
        if order.status in [
            order.Completed, order.Canceled, order.Margin, order.Rejected
        ]:
            if order.status == order.Completed and order.isbuy():
                if not self.position or order.executed.size >= self.position.size:
                    self.added_once = False
                    self.highest_close_since_entry = self.data.close[0]
                    self.initial_stop = (
                        order.executed.price
                        - max(0.08 * order.executed.price, 2 * self.atr14[0])
                    )
            self.order = None

    def _buy_size(self, allocation):
        capital = self.broker.getvalue()
        price = self.data.close[0]
        if price <= 0:
            return 0

        allocation_size = math.floor((capital * allocation) / price)
        affordable_size = math.floor(
            self.broker.getcash() / (price * (1 + self.p.commission_rate))
        )
        return max(0, min(allocation_size, affordable_size))

    def next(self):
        if self.order:
            return

        ema_bull_cross = self.ema20[-1] < self.ema50[-1] and self.ema20[0] > self.ema50[0]
        ema_bear_cross = self.ema20[-1] > self.ema50[-1] and self.ema20[0] < self.ema50[0]
        rsi_bull_cross = self.rsi14[-1] < 50 and self.rsi14[0] > 50
        rsi_bear_cross = self.rsi14[-1] > 50 and self.rsi14[0] < 50

        if not self.position:
            if self.data.close[0] > self.sma200[0] and (ema_bull_cross or rsi_bull_cross):
                size = self._buy_size(0.60)
                if size > 0:
                    self.order = self.buy(size=size)
            return

        if self.highest_close_since_entry is None:
            self.highest_close_since_entry = self.data.close[0]
            self.initial_stop = (
                self.position.price
                - max(0.08 * self.position.price, 2 * self.atr14[0])
            )

        self.highest_close_since_entry = max(
            self.highest_close_since_entry, self.data.close[0]
        )
        trailing_stop = max(
            self.initial_stop,
            self.highest_close_since_entry - 3 * self.atr14[0],
        )

        if (
            self.data.close[0] >= self.position.price * 1.18
            or self.data.close[0] <= trailing_stop
            or ema_bear_cross
            or rsi_bear_cross
        ):
            self.order = self.close()
            return

        pullback = (
            (self.highest_close_since_entry - self.data.close[0])
            / self.highest_close_since_entry
        )

        if (
            not self.added_once
            and 0.10 <= pullback <= 0.15
            and self.data.close[0] > self.sma200[0]
            and self.ema20[0] > self.ema50[0]
        ):
            capital = self.broker.getvalue()
            max_position_size = math.floor((capital * 0.80) / self.data.close[0])
            add_size = min(
                self._buy_size(0.20),
                max(0, max_position_size - self.position.size),
            )
            if add_size > 0:
                self.added_once = True
                self.order = self.buy(size=add_size)
