import backtrader as bt
import math


class ScalpingPullbacks(bt.Strategy):
    params = (
        ("trend_period", 50),
        ("rsi_period", 14),
        ("atr_period", 14),
        ("bb_period", 20),
        ("volume_period", 20),
    )

    def __init__(self):
        self.trend_sma = bt.indicators.SMA(self.data.close, period=50)
        self.rsi = bt.indicators.RSI(self.data.close, period=14)
        self.atr = bt.indicators.ATR(self.data, period=14)
        self.atr_sma = bt.indicators.SMA(self.atr, period=50)
        self.bbands = bt.indicators.BollingerBands(
            self.data.close, period=20, devfactor=2
        )
        self.volume_sma = bt.indicators.SMA(self.data.volume, period=20)

        self.pending_order = None
        self.entry_price = None
        self.entry_atr = None
        self.stop_price = None
        self.target_price = None
        self.entry_bar = None

    def notify_order(self, order):
        if order.status in (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            if order.status == order.Completed and order.isbuy():
                self.entry_price = order.executed.price
                self.entry_atr = float(self.atr[0])
                self.stop_price = self.entry_price - (1.5 * self.entry_atr)
                self.target_price = self.entry_price + (1.5 * self.entry_atr)
                self.entry_bar = len(self)

            elif order.status == order.Completed and order.issell():
                self.entry_price = None
                self.entry_atr = None
                self.stop_price = None
                self.target_price = None
                self.entry_bar = None

            if order == self.pending_order:
                self.pending_order = None

    def _recent_pullback(self):
        for bars_ago in range(1, 6):
            if self.rsi[-bars_ago] < 50 and self.rsi[-bars_ago - 1] > 50:
                return True
        return False

    def next(self):
        if self.pending_order:
            return

        if self.position:
            bars_since_entry = len(self) - self.entry_bar + 1

            if self.data.low[0] <= self.stop_price:
                self.pending_order = self.close()
                return

            if self.data.high[0] >= self.target_price:
                self.pending_order = self.close()
                return

            if (
                bars_since_entry <= 5
                and self.data.high[0] >= self.bbands.top[0]
            ):
                self.pending_order = self.close()
                return

            if (
                bars_since_entry <= 3
                and self.data.high[0] - self.entry_price >= self.entry_atr
            ):
                self.pending_order = self.close()
                return

            if self.rsi[0] < 40:
                self.pending_order = self.close()
            return

        if (
            self.data.close[0] > self.trend_sma[0]
            and self._recent_pullback()
            and self.rsi[0] > 50
            and self.rsi[-1] <= 50
            and self.data.volume[0] > self.volume_sma[0]
            and self.atr[0] <= self.atr_sma[0] * 1.3
            and abs(self.data.open[0] - self.data.close[-1]) <= self.atr[0]
        ):
            entry = float(self.data.close[0])
            stop = entry - (1.5 * float(self.atr[0]))

            if entry > stop:
                size = math.floor(
                    (0.02 * self.broker.getvalue()) / (entry - stop)
                )
                if size > 0:
                    self.pending_order = self.buy(size=size)
