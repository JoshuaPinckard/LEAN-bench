import backtrader as bt


class SlopeIndicator(bt.Indicator):
    lines = ('slope',)
    params = (('lag', 20),)

    def __init__(self):
        self.lines.slope = (self.data - self.data(-self.p.lag)) / self.p.lag


class SmaSlopeReversalStrategy(bt.Strategy):
    params = (
        ('sma_period', 20),
        ('slope_lag', 20),
        ('std_lookback', 100),
        ('extreme_mult', 2.0),
        ('stop_pct', 0.02),
        ('neutral_mult', 0.2),
    )

    def __init__(self):
        self.sma = bt.indicators.SMA(self.data.close, period=self.p.sma_period)
        self.slope = SlopeIndicator(self.sma, lag=self.p.slope_lag)
        self.slope_std = bt.indicators.StdDev(self.slope, period=self.p.std_lookback)
        self.pos_th = self.p.extreme_mult * self.slope_std
        self.neg_th = -self.p.extreme_mult * self.slope_std
        self.neutral_band = self.p.neutral_mult * self.slope_std

        self.order = None
        self.stop_price = None

    def notify_order(self, order):
        if order.status == order.Completed:
            if self.position:
                if order.isbuy():
                    self.stop_price = order.executed.price * (1 - self.p.stop_pct)
                else:
                    self.stop_price = order.executed.price * (1 + self.p.stop_pct)
            else:
                self.stop_price = None
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        if self.order:
            return

        slope0 = self.slope[0]
        slope1 = self.slope[-1]
        neg_th0 = self.neg_th[0]
        neg_th1 = self.neg_th[-1]
        pos_th0 = self.pos_th[0]
        pos_th1 = self.pos_th[-1]
        neutral_band0 = self.neutral_band[0]

        if not self.position:
            if slope1 <= neg_th1 and slope0 > neg_th0:
                self.order = self.buy()
            elif slope1 >= pos_th1 and slope0 < pos_th0:
                self.order = self.sell()
            return

        cross_zero = (slope1 <= 0 and slope0 > 0) or (slope1 >= 0 and slope0 < 0)
        neutral = abs(slope0) < neutral_band0

        if self.position.size > 0:
            if self.data.close[0] <= self.stop_price:
                self.order = self.close()
            elif cross_zero or neutral:
                self.order = self.close()
        elif self.position.size < 0:
            if self.data.close[0] >= self.stop_price:
                self.order = self.close()
            elif cross_zero or neutral:
                self.order = self.close()
