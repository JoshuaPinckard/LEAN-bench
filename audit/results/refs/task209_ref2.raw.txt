import backtrader as bt


class MicroMomentumStrategy(bt.Strategy):
    params = dict()

    def __init__(self):
        self.ema5 = bt.indicators.EMA(self.data.close, period=5)
        self.ema10 = bt.indicators.EMA(self.data.close, period=10)
        self.rsi7 = bt.indicators.RSI(self.data.close, period=7)
        self.vol_sma20 = bt.indicators.SMA(self.data.volume, period=20)
        self.atr14 = bt.indicators.ATR(self.data, period=14)
        self.crossover = bt.indicators.CrossOver(self.ema5, self.ema10)

        self.order = None
        self.entry_price = None
        self.entry_bar = None
        self.tp_price = None
        self.sl_price = None

        self.broker.set_slippage_perc(0.0005)

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            if self.position:
                self.entry_price = order.executed.price
                self.entry_bar = len(self)
                atr_val = self.atr14[0]
                if self.position.size > 0:
                    self.tp_price = self.entry_price + 0.15 * atr_val
                    self.sl_price = self.entry_price - 0.10 * atr_val
                else:
                    self.tp_price = self.entry_price - 0.15 * atr_val
                    self.sl_price = self.entry_price + 0.10 * atr_val
            self.order = None
        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        if self.order:
            return

        dt_time = self.data.datetime.time(0)
        is_eod = (dt_time.hour == 15 and dt_time.minute >= 55) or (
            dt_time.hour > 15
        )

        if self.position:
            if is_eod:
                self.order = self.close()
                return

            if (
                self.entry_bar is not None
                and (len(self) - self.entry_bar) >= 3
            ):
                self.order = self.close()
                return

            if self.position.size > 0:
                if (
                    self.tp_price is not None
                    and self.data.high[0] >= self.tp_price
                ):
                    self.order = self.close()
                    return
                if (
                    self.sl_price is not None
                    and self.data.low[0] <= self.sl_price
                ):
                    self.order = self.close()
                    return
            elif self.position.size < 0:
                if (
                    self.tp_price is not None
                    and self.data.low[0] <= self.tp_price
                ):
                    self.order = self.close()
                    return
                if (
                    self.sl_price is not None
                    and self.data.high[0] >= self.sl_price
                ):
                    self.order = self.close()
                    return
        else:
            if is_eod:
                return

            vol_filter = self.data.volume[0] > self.vol_sma20[0]
            atr_filter = (self.atr14[0] / self.data.close[0]) > 0.001

            if vol_filter and atr_filter:
                size = (self.broker.getvalue() * 0.001) / self.data.close[0]
                if self.crossover[0] > 0 and self.rsi7[0] > 55:
                    self.order = self.buy(size=size)
                elif self.crossover[0] < 0 and self.rsi7[0] < 45:
                    self.order = self.sell(size=size)
