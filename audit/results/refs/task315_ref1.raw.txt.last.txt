import backtrader as bt
import math


class MonthlySMAStopStrategy(bt.Strategy):
    params = (('stop_loss_pct', 0.02),)

    def __init__(self):
        self.sma = bt.indicators.SMA(self.data.close, period=20)
        self.entry_order = None
        self.stop_order = None
        self.exit_order = None
        self.entry_year = None
        self.entry_month = None
        self.checked_entry_month = None

    def next(self):
        current_date = self.data.datetime.date(0)
        current_month = (current_date.year, current_date.month)

        if not self.position and self.entry_order is None and self.exit_order is None:
            if current_month != self.checked_entry_month and current_date.day >= 25:
                self.checked_entry_month = current_month

                if self.data.close[0] > self.sma[0]:
                    commission = self.broker.getcommissioninfo(self.data).p.commission
                    size = math.floor(
                        self.broker.getcash() /
                        (self.data.close[0] * (1.0 + commission))
                    )
                    if size > 0:
                        self.entry_order = self.buy(size=size)

        if self.position and self.exit_order is None:
            following_month = self.entry_month + 1
            following_year = self.entry_year
            if following_month == 13:
                following_month = 1
                following_year += 1

            if (current_date.year, current_date.month) == (following_year, following_month):
                if current_date.day >= 5:
                    if self.stop_order is not None:
                        self.cancel(self.stop_order)
                    self.exit_order = self.close()

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            if order is self.entry_order:
                self.entry_order = None

                if order.status == order.Completed:
                    entry_date = bt.num2date(order.executed.dt).date()
                    self.entry_year = entry_date.year
                    self.entry_month = entry_date.month
                    self.stop_order = self.sell(
                        exectype=bt.Order.Stop,
                        price=order.executed.price * (1.0 - self.p.stop_loss_pct)
                    )

            elif order is self.stop_order:
                self.stop_order = None

            elif order is self.exit_order:
                self.exit_order = None
