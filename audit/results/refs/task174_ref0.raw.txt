import math
import backtrader as bt


class ProfitBackedExpansion(bt.Strategy):

    def __init__(self):
        self.highest_high = bt.indicators.Highest(self.data.high, period=252)
        self.rsi = bt.indicators.RSI(self.data.close, period=14)
        self.vol_sma20 = bt.indicators.SMA(self.data.volume, period=20)
        self.vol_sma100 = bt.indicators.SMA(self.data.volume, period=100)

        self.order = None
        self.pending_type = None

        self.base_active = False
        self.base_entry_price = None
        self.base_shares = 0

        self.tranche_active = False
        self.tranche_entry_price = None
        self.tranche_shares = 0
        self.tranche_entry_bar = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status == order.Completed:
            if self.pending_type == 'base_entry':
                self.base_entry_price = order.executed.price
                self.base_shares = order.executed.size
                self.base_active = True
            elif self.pending_type == 'tranche_entry':
                self.tranche_entry_price = order.executed.price
                self.tranche_shares = order.executed.size
                self.tranche_entry_bar = len(self)
                self.tranche_active = True
            elif self.pending_type == 'tranche_exit':
                self.tranche_active = False
                self.tranche_shares = 0
                self.tranche_entry_price = None
                self.tranche_entry_bar = None

        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None
            self.pending_type = None

    def next(self):
        if self.order:
            return

        close = self.data.close[0]

        if not self.base_active:
            cond1 = close >= 0.9 * self.highest_high[0]
            cond2 = self.rsi[0] > 50
            cond4 = self.vol_sma20[0] > self.vol_sma100[0]
            if cond1 and cond2 and cond4:
                self.order = self.buy()
                self.pending_type = 'base_entry'
            return

        if self.tranche_active:
            tp_level = self.tranche_entry_price * 1.12
            drawdown_level = 0.8 * self.base_entry_price
            bars_since_entry = len(self) - self.tranche_entry_bar

            if close >= tp_level or close <= drawdown_level or bars_since_entry >= 126:
                self.order = self.sell(size=self.tranche_shares)
                self.pending_type = 'tranche_exit'
        else:
            cond1 = close >= 0.9 * self.highest_high[0]
            cond2 = self.rsi[0] > 50
            cond3 = (close / self.base_entry_price - 1) >= 0.4
            cond4 = self.vol_sma20[0] > self.vol_sma100[0]

            if cond1 and cond2 and cond3 and cond4:
                unrealized_profit_base = (close - self.base_entry_price) * self.base_shares
                buy_shares = math.floor(0.5 * unrealized_profit_base / close)
                if buy_shares > 0:
                    self.order = self.buy(size=buy_shares)
                    self.pending_type = 'tranche_entry'
