== 'entry':
                self.entry_session_index = self.session_count
            elif self.order_type == 'exit':
                self._reset_trade_state()
        elif order.status in (order.Canceled, order.Margin, order.Rejected):
            if self.order_type == 'entry':
                self._reset_trade_state()
            elif self.order_type == 'retest':
                self.retest_done = False
        self.order = None
        self.order_type = None

    def _reset_trade_state(self):
        self.breakout_low = None
        self.breakout_high = None
        self.breakout_volume = None
        self.retest_done = False
        self.entry_session_index = None
        self.trail_active = False
        self.trail_stop = None

    def _position_size(self, entry_price, stop_price):
        risk_amount = self.broker.getvalue() * self.p.risk_pct
        risk_per_share = abs(entry_price - stop_price)
        if risk_per_share <= 0:
            return 0
        return int(risk_amount / risk_per_share)

    def next(self):
        cur_date = self.data.datetime.date(0)
        if self._cur_date is None:
            self._cur_date = cur_date
        elif cur_date != self._cur_date:
            self.daily_closes.append(self._prev_close)
            self._cur_date = cur_date
            self.session_count += 1
        self._prev_close = self.data.close[0]

        if len(self.daily_closes) >= self.p.market_sma_period:
            self.daily_sma = statistics.mean(self.daily_closes)
        else:
            self.daily_sma = None

        if self.order:
            return

        close = self.data.close[0]
        high = self.data.high[0]
        low = self.data.low[0]
        vol = self.data.volume[0]
        upper = self.bb.top[0]
        lower = self.bb.bot[0]
        mid = self.bb.mid[0]
        avgvol = self.vol_sma[0]

        if self.position.size == 0:
            if self.daily_sma is None:
                return

            if close > upper and vol >= self.p.vol_mult * avgvol and close > self.daily_sma:
                stop = low
                size = self._position_size(close, stop)
                if size > 0:
                    self.breakout_low = stop
                    self.breakout_volume = vol
                    self.order_type = 'entry'
                    self.order = self.buy(size=size)
                return

            if close < lower and vol >= self.p.vol_mult * avgvol and close < self.daily_sma:
                stop = high
                size = self._position_size(close, stop)
                if size > 0:
                    self.breakout_high = stop
                    self.breakout_volume = vol
                    self.order_type = 'entry'
                    self.order = self.sell(size=size)
                return

        elif self.position.size > 0:
            stop = self.trail_stop if self.trail_active else self.breakout_low
            if not self.trail_active and close > self.position.price:
                self.trail_active = True
                self.trail_stop = self.position.price + self.atr[0]
                stop = self.trail_stop

            if low <= stop:
                self.order_type = 'exit'
                self.order = self.close()
            elif low <= mid:
                self.order_type = 'exit'
                self.order = self.close()
            elif self.entry_session_index is not None and \
                    (self.session_count - self.entry_session_index) >= self.p.max_hold_days:
                self.order_type = 'exit'
                self.order = self.close()
            elif not self.retest_done and low <= upper and vol < self.breakout_volume:
                size = self._position_size(close, self.breakout_low)
                if size > 0:
                    self.retest_done = True
                    self.order_type = 'retest'
                    self.order = self.buy(size=size)

        elif self.position.size < 0:
            stop = self.trail_stop if self.trail_active else self.breakout_high
            if not self.trail_active and close < self.position.price:
                self.trail_active = True
                self.trail_stop = self.position.price - self.atr[0]
                stop = self.trail_stop

            if high >= stop:
                self.order_type = 'exit'
                self.order = self.close()
            elif high >= mid:
                self.order_type = 'exit'
                self.order = self.close()
            elif self.entry_session_index is not None and \
                    (self.session_count - self.entry_session_index) >= self.p.max_hold_days:
                self.order_type = 'exit'
                self.order = self.close()
            elif not self.retest_done and high >= lower and vol < self.breakout_volume:
                size = self._position_size(close, self.breakout_high)
                if size > 0:
                    self.retest_done = True
                    self.order_type = 'retest'
                    self.order = self.sell(size=size)
