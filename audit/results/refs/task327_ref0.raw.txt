import backtrader as bt


class CandleBodyVolumeBreakoutStrategy(bt.Strategy):
    params = (
        ('body_period', 20),
        ('volume_period', 20),
        ('body_mult', 1.5),
        ('volume_mult', 1.5),
        ('enable_short', True),
        ('session_start_hour', 7),
        ('session_start_minute', 0),
        ('session_end_hour', 17),
        ('session_end_minute', 0),
    )

    def __init__(self):
        self.body = abs(self.data.close - self.data.open)
        self.sma_body = bt.indicators.SMA(self.body, period=self.p.body_period)
        self.sma_volume = bt.indicators.SMA(self.data.volume, period=self.p.volume_period)
        self.order = None

    def notify_order(self, order):
        if order.status in (order.Completed, order.Canceled, order.Margin, order.Rejected):
            self.order = None

    def next(self):
        if self.order:
            return

        close = self.data.close[0]
        open_ = self.data.open[0]
        volume = self.data.volume[0]
        body = self.body[0]
        sma_body = self.sma_body[0]
        sma_volume = self.sma_volume[0]

        long_signal = (close > open_) and (body >= self.p.body_mult * sma_body) and (volume >= self.p.volume_mult * sma_volume)
        short_signal = (close < open_) and (body >= self.p.body_mult * sma_body) and (volume >= self.p.volume_mult * sma_volume)

        t = self.data.datetime.time(0)
        start = (self.p.session_start_hour, self.p.session_start_minute)
        end = (self.p.session_end_hour, self.p.session_end_minute)
        time_ok = start <= (t.hour, t.minute) <= end

        if self.position:
            if self.position.size > 0 and short_signal:
                self.order = self.close()
            elif self.position.size < 0 and long_signal:
                self.order = self.close()
        else:
            if time_ok and long_signal:
                self.order = self.buy()
            elif self.p.enable_short and time_ok and short_signal:
                self.order = self.sell()
