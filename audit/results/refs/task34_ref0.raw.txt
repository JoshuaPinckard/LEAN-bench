import backtrader as bt


class WyckoffRangeBreakoutStrategy(bt.Strategy):
    params = (
        ('sma_period', 20),
        ('vol_period', 20),
        ('vol_mult', 1.2),
        ('range_min_pct', 0.02),
        ('stop_pct', 0.005),
        ('reentry_bars', 5),
        ('risk_pct', 0.01),
        ('range_lookback', 30),
        ('swing_order', 2),
        ('touch_tolerance', 0.005),
    )

    def __init__(self):
        self.sma = bt.indicators.SimpleMovingAverage(self.data.close, period=self.p.sma_period)
        self.vol_sma = bt.indicators.SimpleMovingAverage(self.data.volume, period=self.p.vol_period)

        self.order = None
        self.stop_order = None
        self.tp_order = None
        self.close_order = None

        self.last_entry_bar = None
        self.active_support = None
        self.active_resistance = None

        self.pending_stop = None
        self.pending_tp = None
        self.pending_support = None
        self.pending_resistance = None

    def _find_swings(self):
        n = len(self)
        so = self.p.swing_order
        max_k = min(self.p.range_lookback, n - 1 - so)
        highs = []
        lows = []
        if max_k < so:
            return highs, lows
        for k in range(so, max_k + 1):
            window_high = [self.data.high[-(k + j)] for j in range(-so, so + 1)]
            if self.data.high[-k] >= max(window_high):
                highs.append(self.data.high[-k])
            window_low = [self.data.low[-(k + j)] for j in range(-so, so + 1)]
            if self.data.low[-k] <= min(window_low):
                lows.append(self.data.low[-k])
        return highs, lows

    def _cluster_level(self, values, want_max):
        tol = self.p.touch_tolerance
        best = None
        for v in values:
            count = sum(1 for other in values if abs(other - v) <= tol * v)
            if count >= 2:
                if best is None or (want_max and v > best) or (not want_max and v < best):
                    best = v
        return best

    def _find_range(self):
        highs, lows = self._find_swings()
        resistance = self._cluster_level(highs, want_max=True)
        support = self._cluster_level(lows, want_max=False)
        return support, resistance

    def _manage_exit(self):
        if self.close_order:
            return
        if self.active_support is None or self.active_resistance is None:
            return
        price = self.data.close[0]
        if self.active_support <= price <= self.active_resistance:
            if self.stop_order:
                self.cancel(self.stop_order)
            if self.tp_order:
                self.cancel(self.tp_order)
            self.close_order = self.close()

    def next(self):
        if self.order:
            return

        if self.position:
            self._manage_exit()
            return

        support, resistance = self._find_range()
        if support is None or resistance is None:
            return

        price = self.data.close[0]
        if price <= 0 or (resistance - support) / price < self.p.range_min_pct:
            return

        if price < self.sma[0]:
            return

        if self.last_entry_bar is not None and (len(self) - self.last_entry_bar) < self.p.reentry_bars:
            return

        if self.data.volume[0] < self.p.vol_mult * self.vol_sma[0]:
            return

        range_height = resistance - support

        if price > resistance:
            stop_price = support * (1 - self.p.stop_pct)
            tp_price = resistance + range_height
            risk_per_share = price - stop_price
            if risk_per_share <= 0:
                return
            size = int((self.p.risk_pct * self.broker.getvalue()) / risk_per_share)
            if size <= 0:
                return
            self.pending_stop = stop_price
            self.pending_tp = tp_price
            self.pending_support = support
            self.pending_resistance = resistance
            self.order = self.buy(size=size)

        elif price < support:
            stop_price = resistance * (1 + self.p.stop_pct)
            tp_price = support - range_height
            risk_per_share = stop_price - price
            if risk_per_share <= 0:
                return
            size = int((self.p.risk_pct * self.broker.getvalue()) / risk_per_share)
            if size <= 0:
                return
            self.pending_stop = stop_price
            self.pending_tp = tp_price
            self.pending_support = support
            self.pending_resistance = resistance
            self.order = self.sell(size=size)

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status == order.Completed:
            if self.order is not None and order.ref == self.order.ref:
                self.order = None
                self.last_entry_bar = len(self)
                self.active_support = self.pending_support
                self.active_resistance = self.pending_resistance
                sz = abs(order.executed.size)
                if order.isbuy():
                    self.stop_order = self.sell(exectype=bt.Order.Stop, price=self.pending_stop, size=sz)
                    self.tp_order = self.sell(exectype=bt.Order.Limit, price=self.pending_tp, size=sz,
                                               oco=self.stop_order)
                else:
                    self.stop_order = self.buy(exectype=bt.Order.Stop, price=self.pending_stop, size=sz)
                    self.tp_order = self.buy(exectype=bt.Order.Limit, price=self.pending_tp, size=sz,
                                              oco=self.stop_order)
            elif self.stop_order is not None and order.ref == self.stop_order.ref:
                self.stop_order = None
                self.tp_order = None
                self.active_support = None
                self.active_resistance = None
            elif self.tp_order is not None and order.ref == self.tp_order.ref:
                self.stop_order = None
                self.tp_order = None
                self.active_support = None
                self.active_resistance = None
            elif self.close_order is not None and order.ref == self.close_order.ref:
                self.close_order = None
                self.stop_order = None
                self.tp_order = None
                self.active_support = None
                self.active_resistance = None

        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            if self.order is not None and order.ref == self.order.ref:
                self.order = None
            if self.stop_order is not None and order.ref == self.stop_order.ref:
                self.stop_order = None
            if self.tp_order is not None and order.ref == self.tp_order.ref:
                self.tp_order = None
            if self.close_order is not None and order.ref == self.close_order.ref:
                self.close_order = None
