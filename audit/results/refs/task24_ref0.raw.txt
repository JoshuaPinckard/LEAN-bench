import backtrader as bt


class InstantTradingDeltaTriggers(bt.Strategy):
    params = (
        ('buy_position_delta', 0.02),
        ('sell_position_delta', 0.02),
    )

    def __init__(self):
        self.order = None
        self.entry_price = None

    def notify_order(self, order):
        if order.status == order.Completed:
            self.entry_price = order.executed.price
        if order.status in (order.Completed, order.Canceled, order.Margin, order.Rejected):
            self.order = None

    def next(self):
        if self.order:
            return
        if len(self) < 2:
            return

        prev_close = self.data.close[-1]
        cur_close = self.data.close[0]

        if not self.position:
            if cur_close > prev_close * (1 + self.p.buy_position_delta):
                size = int(self.broker.getcash() / cur_close)
                if size > 0:
                    self.order = self.buy(size=size)
            elif cur_close < prev_close * (1 - self.p.sell_position_delta):
                size = int(self.broker.getcash() / cur_close)
                if size > 0:
                    self.order = self.sell(size=size)
        else:
            if self.position.size > 0:
                exit_price = self.entry_price * (1 - self.p.sell_position_delta)
                if cur_close <= exit_price and cur_close > self.entry_price:
                    self.order = self.close()
            elif self.position.size < 0:
                exit_price = self.entry_price * (1 + self.p.buy_position_delta)
                if cur_close >= exit_price and cur_close < self.entry_price:
                    self.order = self.close()
