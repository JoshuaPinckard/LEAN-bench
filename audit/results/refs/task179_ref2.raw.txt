import backtrader as bt


class BuyingSpikeMomentum(bt.Strategy):
    def __init__(self):
        self.sma20_price = bt.indicators.SMA(self.data.close, period=20)
        self.sma5_vol = bt.indicators.SMA(self.data.volume, period=5)
        self.sma20_vol = bt.indicators.SMA(self.data.volume, period=20)
        self.stoch = bt.indicators.Stochastic(
            self.data, period=14, period_dfast=3, period_dslow=3
        )
        self.atr14 = bt.indicators.ATR(self.data, period=14)

        self.order = None
        self.stop_order = None
        self.buy_bar = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            if order.isbuy():
                self.buy_bar = len(self)
                entry_price = order.executed.price
                stop_price = entry_price * 0.98
                self.stop_order = self.sell(
                    exectype=bt.Order.Stop,
                    price=stop_price,
                    size=order.executed.size,
                )
            else:
                if self.stop_order and order.ref == self.stop_order.ref:
                    self.stop_order = None
                if not self.position:
                    self.buy_bar = None

            self.order = None

        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            if self.order and order.ref == self.order.ref:
                self.order = None
            if self.stop_order and order.ref == self.stop_order.ref:
                self.stop_order = None

    def next(self):
        if self.order:
            return

        if not self.position:
            cond1 = self.data.close[0] > self.sma20_price[0]
            cond2 = self.sma5_vol[0] >= 3.0 * self.sma20_vol[0]

            k_val = self.stoch.percK[0]
            d_val = self.stoch.percD[0]
            k_prev = self.stoch.percK[-1]
            d_prev = self.stoch.percD[-1]

            crossover_up = (k_val > d_val) and (k_prev <= d_prev)
            close_higher = self.data.close[0] > self.data.close[-1]
            cond3 = (k_val > 80) and (crossover_up or close_higher)

            filter_atr = (self.atr14[0] / self.data.close[0]) > 0.015
            filter_liq = (
                self.sma20_vol[0] * self.sma20_price[0]
            ) >= 10000000.0

            if cond1 and cond2 and cond3 and filter_atr and filter_liq:
                size = int(
                    (0.02 * self.broker.getvalue()) / self.data.close[0]
                )
                if size > 0:
                    self.order = self.buy(size=size)
        else:
            bars_held = (
                len(self) - self.buy_bar if self.buy_bar is not None else 0
            )
            exit1 = self.data.close[0] < self.sma20_price[0]
            exit2 = bars_held >= 10

            if exit1 or exit2:
                if self.stop_order:
                    self.cancel(self.stop_order)
                    self.stop_order = None
                self.order = self.close()
