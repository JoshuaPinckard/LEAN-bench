import backtrader as bt


class NVDAMeanReversionStrategy(bt.Strategy):
    params = (
        ("rsi_period", 14),
        ("sma_period", 50),
        ("rsi_oversold", 31),
        ("rsi_exit", 50),
        ("hold_days", 5),
    )

    def __init__(self):
        self.rsi = bt.indicators.RSI(self.data.close, period=self.p.rsi_period)
        self.sma = bt.indicators.SMA(self.data.close, period=self.p.sma_period)
        self.order = None
        self.buy_bar = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            if order.isbuy():
                self.buy_bar = len(self)

        self.order = None

    def next(self):
        if self.order:
            return

        if not self.position:
            if self.data.close[0] < self.sma[0] and self.rsi[0] < self.p.rsi_oversold:
                self.order = self.buy()
        else:
            days_held = len(self) - self.buy_bar if self.buy_bar is not None else 0
            if days_held >= self.p.hold_days or self.rsi[0] > self.p.rsi_exit:
                self.order = self.close()
