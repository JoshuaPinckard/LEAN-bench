import backtrader as bt
import collections


class LevelRetestEntryStrategy(bt.Strategy):
    params = dict(
        level_lookback=20,
        rsi_period=14,
        atr_period=14,
        tp_atr_mult=2.0,
        risk_pct=0.01,
        htf_multiple=4,
        htf_trend_sma_period=20,
        candle_lower_shadow_mult=2.0,
        candle_upper_shadow_max_mult=1.0,
    )

    def __init__(self):
        self.rsi = bt.indicators.RSI(self.data.close, period=self.p.rsi_period)
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        # level = extreme of the 20 bars strictly before the current one, so the
        # current bar can "touch" it rather than always defining it
        self.support_level = bt.indicators.Lowest(self.data.low(-1), period=self.p.level_lookback)
        self.resistance_level = bt.indicators.Highest(self.data.high(-1), period=self.p.level_lookback)

        self.order = None
        self.entry_level = None
        self.tp_price = None

        # manual 4x aggregation of the 1H feed into synthetic 4H bars, since only
        # a single 1H data feed is available (no resampled feed provided)
        self.htf_count = 0
        self.htf_close_history = collections.deque(maxlen=self.p.htf_trend_sma_period)
        self.htf_trend = 0

    def notify_order(self, order):
        if order.status in (order.Completed, order.Canceled, order.Margin, order.Rejected):
            self.order = None

    def _is_bullish_engulfing(self):
        o0, c0 = self.data.open[0], self.data.close[0]
        o1, c1 = self.data.open[-1], self.data.close[-1]
        return (c1 < o1) and (c0 > o0) and (o0 <= c1) and (c0 >= o1)

    def _is_bearish_engulfing(self):
        o0, c0 = self.data.open[0], self.data.close[0]
        o1, c1 = self.data.open[-1], self.data.close[-1]
        return (c1 > o1) and (c0 < o0) and (o0 >= c1) and (c0 <= o1)

    def _is_small_body_long_lower_shadow(self):
        o, h, l, c = self.data.open[0], self.data.high[0], self.data.low[0], self.data.close[0]
        body = abs(c - o)
        if body <= 0:
            return False
        lower_shadow = min(o, c) - l
        upper_shadow = h - max(o, c)
        return (lower_shadow >= self.p.candle_lower_shadow_mult * body and
                upper_shadow <= self.p.candle_upper_shadow_max_mult * body)

    def _update_htf_trend(self):
        if self.htf_count == 0:
            pass
        self.htf_count += 1
        if self.htf_count == self.p.htf_multiple:
            htf_close = self.data.close[0]
            self.htf_close_history.append(htf_close)
            if len(self.htf_close_history) == self.htf_close_history.maxlen:
                htf_sma = sum(self.htf_close_history) / len(self.htf_close_history)
                if htf_close > htf_sma:
                    self.htf_trend = 1
                elif htf_close < htf_sma:
                    self.htf_trend = -1
                else:
                    self.htf_trend = 0
            self.htf_count = 0

    def _manage_open_position(self):
        if self.position.size > 0:
            if self.data.high[0] >= self.tp_price:
                self.order = self.close()
            elif self.data.close[0] < self.entry_level:
                self.order = self.close()
        elif self.position.size < 0:
            if self.data.low[0] <= self.tp_price:
                self.order = self.close()
            elif self.data.close[0] > self.entry_level:
                self.order = self.close()

    def _check_entries(self):
        support = self.support_level[0]
        resistance = self.resistance_level[0]

        touches_support = self.data.low[0] <= support
        touches_resistance = self.data.high[0] >= resistance

        bullish_pattern = self._is_small_body_long_lower_shadow() or self._is_bullish_engulfing()
        bearish_pattern = self._is_small_body_long_lower_shadow() or self._is_bearish_engulfing()

        if touches_support and bullish_pattern and self.rsi[0] > 50 and self.htf_trend == 1:
            entry_price = self.data.close[0]
            risk_per_share = entry_price - support
            if risk_per_share > 0:
                risk_amount = self.p.risk_pct * self.broker.getvalue()
                size = risk_amount / risk_per_share
                if size > 0:
                    self.entry_level = support
                    self.tp_price = entry_price + self.p.tp_atr_mult * self.atr[0]
                    self.order = self.buy(size=size)
        elif touches_resistance and bearish_pattern and self.rsi[0] < 50 and self.htf_trend == -1:
            entry_price = self.data.close[0]
            risk_per_share = resistance - entry_price
            if risk_per_share > 0:
                risk_amount = self.p.risk_pct * self.broker.getvalue()
                size = risk_amount / risk_per_share
                if size > 0:
                    self.entry_level = resistance
                    self.tp_price = entry_price - self.p.tp_atr_mult * self.atr[0]
                    self.order = self.sell(size=size)

    def next(self):
        self._update_htf_trend()

        if self.order:
            return

        if self.position:
            self._manage_open_position()
            return

        self._check_entries()
