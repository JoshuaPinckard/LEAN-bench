import backtrader as bt


class DynamicProfitExitStrategy(bt.Strategy):
    params = (
        ('fast_period', 20),
        ('slow_period', 50),
        ('profit_threshold', 10.0),
    )

    def __init__(self):
        self.sma_fast = bt.indicators.SMA(self.data.close, period=self.p.fast_period)
        self.sma_slow = bt.indicators.SMA(self.data.close, period=self.p.slow_period)
        self.crossover = bt.indicators.CrossOver(self.sma_fast, self.sma_slow)
        self.order = None

    def notify_order(self, order):
        if order.status in (order.Completed, order.Canceled, order.Margin, order.Rejected):
            self.order = None

    def sell_below10(self):
        return self.close()

    def sell_abv10(self):
        return self.close()

    def next(self):
        if self.order:
            return

        if not self.position:
            if self.crossover > 0:
                comminfo = self.broker.getcommissioninfo(self.data)
                price = self.data.close[0]
                cash = self.broker.getcash()
                size = int(cash / (price * (1 + comminfo.p.commission)))
                if size > 0:
                    self.order = self.buy(size=size)
        else:
            entry_price = self.position.price
            profit_pct = (self.data.close[0] - entry_price) / entry_price * 100.0
            if profit_pct < self.p.profit_threshold:
                self.order = self.sell_below10()
            else:
                self.order = self.sell_abv10()
