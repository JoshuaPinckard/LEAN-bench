import backtrader as bt


class PriceRangeVolumeBreakout(bt.Strategy):
    params = (
        ('range_period', 20),
        ('volume_period', 20),
        ('volume_multiplier', 2.15),
        ('target_price', 28.00),
    )

    def __init__(self):
        self.order = None
        self.range_high = bt.indicators.Highest(self.data.high(-1), period=self.p.range_period)
        self.average_volume = bt.indicators.SimpleMovingAverage(
            self.data.volume, period=self.p.volume_period
        )

    def notify_order(self, order):
        if order.status in (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            self.order = None

    def next(self):
        if self.order:
            return

        if self.position:
            if self.data.high[0] >= self.p.target_price:
                self.order = self.close()
            return

        if (
            self.data.open[0] > self.range_high[0]
            and self.data.volume[0] > self.p.volume_multiplier * self.average_volume[0]
        ):
            self.order = self.order_target_percent(target=1.0)
