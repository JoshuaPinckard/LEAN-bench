import math
import backtrader as bt


class DonchianChannel(bt.Indicator):
    lines = ('top', 'bottom', 'middle')
    params = (('period', 20),)

    def __init__(self):
        self.lines.top = bt.indicators.Highest(self.data.high, period=self.p.period)(-1)
        self.lines.bottom = bt.indicators.Lowest(self.data.low, period=self.p.period)(-1)
        self.lines.middle = (self.lines.top + self.lines.bottom) / 2.0


class DonchianBreakoutStrategy(bt.Strategy):
    params = (
        ('donchian_period', 20),
        ('adx_period', 14),
        ('atr_period', 14),
        ('adx_threshold', 25.0),
        ('risk_pct', 0.01),
    )

    def __init__(self):
        self.donchian = DonchianChannel(self.data, period=self.p.donchian_period)
        self.adx = bt.indicators.ADX(self.data, period=self.p.adx_period)
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)

        self.order = None
        self.stop_order = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            if order == self.order:
                self.order = None
                if self.position:
                    entry_price = order.executed.price
                    stop_dist = max(2.0 * self.atr[0], 0.02 * entry_price)
                    if self.position.size > 0:
                        stop_price = entry_price - stop_dist
                        self.stop_order = self.sell(
                            size=abs(self.position.size),
                            exectype=bt.Order.Stop,
                            price=stop_price
                        )
                    elif self.position.size < 0:
                        stop_price = entry_price + stop_dist
                        self.stop_order = self.buy(
                            size=abs(self.position.size),
                            exectype=bt.Order.Stop,
                            price=stop_price
                        )
                else:
                    if self.stop_order:
                        self.cancel(self.stop_order)
                        self.stop_order = None

            elif order == self.stop_order:
                self.stop_order = None
                if self.order:
                    self.cancel(self.order)
                    self.order = None

        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            if order == self.order:
                self.order = None
            if order == self.stop_order:
                self.stop_order = None

    def next(self):
        if not self.position:
            if self.order is not None:
                return

            if self.data.close[0] > self.donchian.top[0] and self.adx[0] > self.p.adx_threshold:
                ref_price = self.data.close[0]
                stop_dist = max(2.0 * self.atr[0], 0.02 * ref_price)
                if stop_dist > 0:
                    equity = self.broker.getvalue()
                    size = int(math.floor((equity * self.p.risk_pct) / stop_dist))
                    if size > 0:
                        self.order = self.buy(size=size)

            elif self.data.close[0] < self.donchian.bottom[0] and self.adx[0] > self.p.adx_threshold:
                ref_price = self.data.close[0]
                stop_dist = max(2.0 * self.atr[0], 0.02 * ref_price)
                if stop_dist > 0:
                    equity = self.broker.getvalue()
                    size = int(math.floor((equity * self.p.risk_pct) / stop_dist))
                    if size > 0:
                        self.order = self.sell(size=size)

        else:
            if self.order is not None:
                return

            if self.position.size > 0:
                if self.data.close[0] < self.donchian.bottom[0]:
                    if self.stop_order:
                        self.cancel(self.stop_order)
                        self.stop_order = None
                    self.order = self.close()
            elif self.position.size < 0:
                if self.data.close[0] > self.donchian.top[0]:
                    if self.stop_order:
                        self.cancel(self.stop_order)
                        self.stop_order = None
                    self.order = self.close()
