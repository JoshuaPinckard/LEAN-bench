import backtrader as bt


class ImpulseStrategy(bt.Strategy):
    def __init__(self):
        self.sma252 = bt.indicators.SimpleMovingAverage(self.data.close, period=252)
        self.atr14 = bt.indicators.ATR(self.data, period=14)
        self.sma_atr = bt.indicators.SimpleMovingAverage(self.atr14, period=252)
        self.atr_pct = self.sma_atr / self.data.close

        self.entry_order = None
        self.stop_order = None
        self.limit_order = None
        self.close_order = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            if order == self.entry_order:
                self.entry_order = None
                entry_price = order.executed.price
                stop_price = entry_price * 0.98
                limit_price = entry_price * 1.05
                self.stop_order = self.sell(
                    size=order.executed.size,
                    exectype=bt.Order.Stop,
                    price=stop_price,
                )
                self.limit_order = self.sell(
                    size=order.executed.size,
                    exectype=bt.Order.Limit,
                    price=limit_price,
                    oco=self.stop_order,
                )
            elif order in [self.stop_order, self.limit_order]:
                self.stop_order = None
                self.limit_order = None
            elif order == self.close_order:
                self.close_order = None

        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            if order == self.entry_order:
                self.entry_order = None
            if order == self.stop_order:
                self.stop_order = None
            if order == self.limit_order:
                self.limit_order = None
            if order == self.close_order:
                self.close_order = None

    def next(self):
        if self.entry_order:
            return

        if not self.position:
            if self.data.close[0] > self.sma252[0] and self.atr_pct[0] <= 0.05:
                capital = self.broker.getvalue()
                size = int((capital * 0.10) / self.data.close[0])
                if size > 0:
                    self.entry_order = self.buy(size=size)
        else:
            if self.close_order:
                return

            if self.data.close[0] < self.sma252[0]:
                if self.stop_order:
                    self.cancel(self.stop_order)
                if self.limit_order:
                    self.cancel(self.limit_order)
                self.close_order = self.close()
