import backtrader as bt
import math


class SellOffHedgeShort(bt.Strategy):
    params = dict(
        risk_dollars=200,
    )

    ENTRY_START = (9, 30)
    ENTRY_END = (11, 30)
    FORCED_EXIT = (15, 45)

    def __init__(self):
        self.sma20 = bt.indicators.SMA(self.data.close, period=20)
        self.sma50 = bt.indicators.SMA(self.data.close, period=50)
        self.ema9 = bt.indicators.EMA(self.data.close, period=9)
        self.rsi14 = bt.indicators.RSI(self.data.close, period=14)
        self.atr14 = bt.indicators.ATR(self.data, period=14)
        self.bb20 = bt.indicators.BollingerBands(self.data, period=20, devfactor=2)
        self.crossover = bt.indicators.CrossOver(self.ema9, self.sma20)

        self.order = None
        self.stop_order = None
        self.limit_order = None

        self.current_date = None
        self.session_open = None
        self.gap_filter_ok = True

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            if order is self.order:
                self.order = None
            if order is self.stop_order:
                self.stop_order = None
            if order is self.limit_order:
                self.limit_order = None

    def _start_new_session(self, bar_date):
        if self.current_date is not None and len(self.data) > 1:
            prev_close = self.data.close[-1]
            session_open = self.data.open[0]
            if prev_close != 0:
                gap = abs(session_open - prev_close) / prev_close
                self.gap_filter_ok = gap <= 0.015
            else:
                self.gap_filter_ok = True
        else:
            self.gap_filter_ok = True
        self.current_date = bar_date
        self.session_open = self.data.open[0]

    def next(self):
        bar_date = self.data.datetime.date(0)
        if bar_date != self.current_date:
            self._start_new_session(bar_date)

        bar_time = self.data.datetime.time(0)
        bar_hm = (bar_time.hour, bar_time.minute)

        if self.position:
            if bar_hm >= self.FORCED_EXIT:
                if self.stop_order is not None:
                    self.cancel(self.stop_order)
                if self.limit_order is not None:
                    self.cancel(self.limit_order)
                self.close()
            return

        if self.order is not None:
            return

        if not (self.ENTRY_START <= bar_hm <= self.ENTRY_END):
            return

        if self.session_open is None or self.session_open == 0:
            return

        close = self.data.close[0]
        atr = self.atr14[0]

        if atr <= 0:
            return

        price_drop_ok = (close / self.session_open - 1) <= -0.005
        downtrend_ok = close < self.sma50[0] and self.rsi14[0] < 45
        vol_not_low_ok = (atr / close) >= 0.005
        low_vol_filter_ok = (atr / close) >= 0.004
        gap_ok = self.gap_filter_ok
        crossover_ok = self.crossover[0] < 0

        if price_drop_ok and downtrend_ok and vol_not_low_ok and low_vol_filter_ok and gap_ok and crossover_ok:
            size = int(math.floor(self.p.risk_dollars / (1.5 * atr)))
            if size < 1:
                return
            stop_price = close + 1.5 * atr
            target_price = close - 2 * atr
            orders = self.sell_bracket(
                size=size,
                exectype=bt.Order.Market,
                stopprice=stop_price,
                limitprice=target_price,
            )
            self.order = orders[0]
            self.stop_order = orders[1]
            self.limit_order = orders[2]
