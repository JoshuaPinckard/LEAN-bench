import backtrader as bt


class DemaAtrChannelStrategy(bt.Strategy):
    def __init__(self):
        self.ema10 = bt.indicators.EMA(self.data.close, period=10)
        self.atr10 = bt.indicators.ATR(self.data, period=10)

        self.ema200 = bt.indicators.EMA(self.data.close, period=200)
        self.ema200_ema = bt.indicators.EMA(self.ema200, period=200)
        self.dema200 = 2.0 * self.ema200 - self.ema200_ema

        self.upper = self.ema10 + 3.0 * self.atr10
        self.lower = self.ema10 - 3.0 * self.atr10

        self.cross_upper = bt.indicators.CrossOver(self.data.close, self.upper)
        self.cross_lower = bt.indicators.CrossOver(self.data.close, self.lower)

        self.order = None

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        if self.order:
            return

        if not self.position:
            if self.data.close[0] > self.dema200[0] and self.cross_upper[0] > 0:
                size = int(self.broker.getcash() / self.data.close[0])
                if size > 0:
                    self.order = self.buy(size=size)
        else:
            if self.cross_lower[0] < 0 or self.data.close[0] < self.dema200[0]:
                self.order = self.close()
