import backtrader as bt


class TrendFollowingDEMAATRChannel(bt.Strategy):
    def __init__(self):
        self.order = None

        ema_200 = bt.indicators.EMA(self.data.close, period=200)
        self.dema_200 = 2.0 * ema_200 - bt.indicators.EMA(ema_200, period=200)

        self.ema_10 = bt.indicators.EMA(self.data.close, period=10)
        self.atr_10 = bt.indicators.ATR(self.data, period=10)
        self.upper = self.ema_10 + 3.0 * self.atr_10
        self.lower = self.ema_10 - 3.0 * self.atr_10

        self.cross_upper = bt.indicators.CrossOver(self.data.close, self.upper)
        self.cross_lower = bt.indicators.CrossOver(self.data.close, self.lower)

    def notify_order(self, order):
        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            self.order = None

    def next(self):
        if self.order:
            return

        if not self.position:
            if self.data.close[0] > self.dema_200[0] and self.cross_upper[0] > 0:
                size = self.broker.getcash() / (
                    self.data.close[0] * (1.0 + self.broker.getcommissioninfo(self.data).p.commission)
                )
                self.order = self.buy(size=size)
        elif self.cross_lower[0] < 0 or self.data.close[0] < self.dema_200[0]:
            self.order = self.close()
