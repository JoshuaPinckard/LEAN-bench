import backtrader as bt


class ATRTrailingStopSAR(bt.Strategy):
    params = (
        ("atr_period", 14),
        ("k", 2.5),
    )

    def __init__(self):
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.order = None
        self.stop_order = None

        self.sar_history = []
        self.up_stop_history = []
        self.down_stop_history = []
        self.entry_sar = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            if order == self.order:
                self.order = None
                if self.position and self.entry_sar is not None:
                    if self.position.size > 0:
                        self.stop_order = self.sell(
                            exectype=bt.Order.Stop, price=self.entry_sar
                        )
                    elif self.position.size < 0:
                        self.stop_order = self.buy(
                            exectype=bt.Order.Stop, price=self.entry_sar
                        )
            elif order == self.stop_order:
                self.stop_order = None

        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            if order == self.order:
                self.order = None
            if order == self.stop_order:
                self.stop_order = None

    def next(self):
        c = self.data.close[0]
        a = self.atr[0]
        k = self.p.k

        candidate_up = c - k * a
        candidate_down = c + k * a

        if len(self.sar_history) == 0:
            if c >= self.data.close[-1]:
                sar_curr = candidate_up
            else:
                sar_curr = candidate_down
            up_stop = candidate_up
            down_stop = candidate_down
        else:
            prev_sar = self.sar_history[-1]
            prev_up = self.up_stop_history[-1]
            prev_down = self.down_stop_history[-1]

            if self.data.close[-1] > prev_sar:
                up_stop = max(prev_up, candidate_up)
                down_stop = candidate_down
            else:
                up_stop = candidate_up
                down_stop = min(prev_down, candidate_down)

            if c > prev_sar:
                sar_curr = up_stop
            else:
                sar_curr = down_stop

        self.sar_history.append(sar_curr)
        self.up_stop_history.append(up_stop)
        self.down_stop_history.append(down_stop)

        if len(self.sar_history) < 2:
            return

        if self.order:
            return

        prev_c = self.data.close[-1]
        prev_s = self.sar_history[-2]
        curr_c = c
        curr_s = sar_curr

        crossover = 0
        if prev_c <= prev_s and curr_c > curr_s:
            crossover = 1
        elif prev_c >= prev_s and curr_c < curr_s:
            crossover = -1

        if crossover > 0:
            if self.stop_order:
                self.cancel(self.stop_order)
                self.stop_order = None
            self.entry_sar = curr_s
            if self.position.size < 0:
                self.close()
                self.order = self.buy()
            elif self.position.size == 0:
                self.order = self.buy()

        elif crossover < 0:
            if self.stop_order:
                self.cancel(self.stop_order)
                self.stop_order = None
            self.entry_sar = curr_s
            if self.position.size > 0:
                self.close()
                self.order = self.sell()
            elif self.position.size == 0:
                self.order = self.sell()
