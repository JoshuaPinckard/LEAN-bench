import backtrader as bt


class BollingerMeanReversionStrategy(bt.Strategy):
    params = (
        ('initial_capital', 10000),
        ('position_size', 100),
    )

    def __init__(self):
        self.ema = bt.indicators.ExponentialMovingAverage(
            self.data.close,
            period=50,
        )
        self.bollinger = bt.indicators.BollingerBands(
            self.data.close,
            period=50,
            devfactor=2,
            movav=bt.indicators.ExponentialMovingAverage,
        )
        self.order = None
        self.transactions = []

    def next(self):
        if self.order:
            return

        price = self.data.close[0]

        if not self.position:
            if price <= self.bollinger.bot[0]:
                self.order = self.buy(size=self.p.position_size)
            elif price >= self.bollinger.top[0]:
                self.order = self.sell(size=self.p.position_size)

        elif self.position.size > 0:
            if price >= self.bollinger.top[0]:
                self.order = self.close()

        elif self.position.size < 0:
            if price <= self.bollinger.bot[0]:
                self.order = self.close()

    def notify_order(self, order):
        if order.status == order.Completed:
            self.transactions.append({
                'date': self.data.datetime.date(0),
                'ticker': self.data._name or 'AAPL',
                'transaction_type': 'bought' if order.isbuy() else 'sold',
                'price': order.executed.price,
                'ema': self.ema[0],
                'shares': abs(order.executed.size),
            })

        if order.status in (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            self.order = None
