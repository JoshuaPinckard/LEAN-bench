import math
import backtrader as bt


class PostDisasterCapitulationBounce(bt.Strategy):
    params = (
        ("sma20_period", 20),
        ("sma50_period", 50),
        ("rsi_period", 14),
        ("atr_period", 14),
        ("volsma_period", 20),
        ("cap_window", 5),
        ("active_window", 10),
        ("cap_drop_pct", 0.15),
        ("cap_vol_mult", 7.0),
        ("vol_ease_mult", 0.8),
        ("min_volatility", 0.03),
        ("risk_pct", 0.02),
    )

    def __init__(self):
        self.sma20 = bt.indicators.SMA(self.data.close, period=self.params.sma20_period)
        self.sma50 = bt.indicators.SMA(self.data.close, period=self.params.sma50_period)
        self.rsi = bt.indicators.RSI(self.data.close, period=self.params.rsi_period)
        self.atr = bt.indicators.ATR(self.data, period=self.params.atr_period)
        self.volsma = bt.indicators.SMA(self.data.volume, period=self.params.volsma_period)
        self.rsi_crossover = bt.indicators.CrossOver(self.rsi, 30.0)

        self.order = None
        self.cap_active = False
        self.cap_bar = -1
        self.cap_low = None
        self.cap_H = None
        self.cap_vol = 0.0

        self.entry_stop = None
        self.entry_tp2 = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        # 1. Check for capitulation event over the last 5 days
        drop_15 = self.data.close[0] <= self.data.close[-5] * (1.0 - self.params.cap_drop_pct)

        candidate_cap_days = []
        for i in range(0, -self.params.cap_window, -1):
            if self.data.close[i] < self.data.open[i] and self.data.volume[i] > self.params.cap_vol_mult * self.volsma[i]:
                candidate_cap_days.append(i)

        if drop_15 and candidate_cap_days:
            best_i = max(candidate_cap_days, key=lambda idx: self.data.volume[idx])
            cand_bar = len(self) + best_i
            cand_vol = self.data.volume[best_i]
            cand_low = self.data.low[best_i]
            cand_H = max([self.data.high[best_i - k] for k in range(20)])

            if not self.cap_active:
                self.cap_active = True
                self.cap_bar = cand_bar
                self.cap_low = cand_low
                self.cap_H = cand_H
                self.cap_vol = cand_vol
            else:
                if cand_bar > self.cap_bar or (cand_bar == self.cap_bar and cand_vol > self.cap_vol):
                    self.cap_bar = cand_bar
                    self.cap_low = cand_low
                    self.cap_H = cand_H
                    self.cap_vol = cand_vol

        # 2. Expire capitulation window if active window has passed
        if self.cap_active:
            bars_since_cap = len(self) - self.cap_bar
            if bars_since_cap > self.params.active_window:
                self.cap_active = False

        # 3. Handle open position exits
        if self.position:
            if self.order is None:
                # Stop Loss check
                if self.data.low[0] <= self.entry_stop:
                    self.order = self.close()
                # Take Profit check (TP2 50% retracement or touching SMA50)
                elif self.data.high[0] >= self.entry_tp2 or self.data.high[0] >= self.sma50[0]:
                    self.order = self.close()
            return

        # 4. Handle entry signals
        if self.order is not None:
            return

        if self.cap_active:
            bars_since_cap = len(self) - self.cap_bar
            if 1 <= bars_since_cap <= self.params.active_window:
                positive_day = self.data.close[0] > self.data.open[0]
                rsi_cross_up = self.rsi_crossover[0] > 0
                vol_normal = self.data.volume[0] >= self.params.vol_ease_mult * self.volsma[0]
                volatility_sufficient = ((self.data.high[0] - self.data.low[0]) / self.data.open[0]) >= self.params.min_volatility
                trend_filter = self.data.close[0] > self.sma20[0]

                if positive_day and rsi_cross_up and vol_normal and volatility_sufficient and trend_filter:
                    equity = self.broker.getvalue()
                    stop_loss = self.cap_low - 1.0 * self.atr[0]
                    entry_price = self.data.close[0]
                    risk_per_share = entry_price - stop_loss

                    if risk_per_share > 0:
                        size = math.floor((self.params.risk_pct * equity) / risk_per_share)
                        if size >= 1:
                            self.entry_stop = stop_loss
                            self.entry_tp2 = self.cap_low + 0.500 * (self.cap_H - self.cap_low)
                            self.order = self.buy(size=size)
