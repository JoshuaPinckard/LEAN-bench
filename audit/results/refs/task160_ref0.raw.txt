import backtrader as bt


class MeanReversionTrendStrategy(bt.Strategy):
    params = dict(
        ema_period=21,
        bb_period=50,
        bb_devfactor=1.5,
        ema_prox_threshold=0.02,
        risk_pct=0.02,
    )

    def __init__(self):
        self.ema = bt.indicators.EMA(self.data.close, period=self.p.ema_period)
        self.bb = bt.indicators.BollingerBands(
            self.data.close, period=self.p.bb_period, devfactor=self.p.bb_devfactor
        )
        self.cross_lower = bt.indicators.CrossOver(self.data.close, self.bb.lines.bot)
        self.cross_upper = bt.indicators.CrossOver(self.data.close, self.bb.lines.top)
        self.cross_ema = bt.indicators.CrossOver(self.data.close, self.ema)
        self.order = None

    def notify_order(self, order):
        if order.status in (order.Completed, order.Canceled, order.Margin, order.Rejected):
            self.order = None

    def next(self):
        if self.order:
            return

        close = self.data.close[0]
        ema = self.ema[0]
        ema_prox = abs(close - ema) / ema

        if not self.position:
            if self.cross_lower[0] > 0 and close > ema and ema_prox <= self.p.ema_prox_threshold:
                size = int((self.broker.getvalue() * self.p.risk_pct) / close)
                if size > 0:
                    self.order = self.buy(size=size)
            elif self.cross_upper[0] < 0 and close < ema and ema_prox <= self.p.ema_prox_threshold:
                size = int((self.broker.getvalue() * self.p.risk_pct) / close)
                if size > 0:
                    self.order = self.sell(size=size)
        elif self.position.size > 0:
            touch_upper = close >= self.bb.lines.top[0]
            touch_lower = close <= self.bb.lines.bot[0]
            cross_down = self.cross_ema[0] < 0
            if touch_upper or cross_down or touch_lower:
                self.order = self.close()
        elif self.position.size < 0:
            touch_lower = close <= self.bb.lines.bot[0]
            touch_upper = close >= self.bb.lines.top[0]
            cross_up = self.cross_ema[0] > 0
            if touch_lower or cross_up or touch_upper:
                self.order = self.close()
