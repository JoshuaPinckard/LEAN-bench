import backtrader as bt


class SwingTradingStrategy(bt.Strategy):
    params = dict(
        ema_fast_period=21,
        ema_slow_period=50,
        rsi_period=14,
        rsi_overbought=70,
        rsi_oversold=30,
        rsi_mid=50,
        swing_lookback=10,
        risk_pct=0.01,
    )

    def __init__(self):
        self.ema_fast = bt.indicators.EMA(self.data.close, period=self.p.ema_fast_period)
        self.ema_slow = bt.indicators.EMA(self.data.close, period=self.p.ema_slow_period)
        self.rsi = bt.indicators.RSI(self.data.close, period=self.p.rsi_period)
        self.crossover = bt.indicators.CrossOver(self.ema_fast, self.ema_slow)
        self.swing_low = bt.indicators.Lowest(self.data.low, period=self.p.swing_lookback)
        self.swing_high = bt.indicators.Highest(self.data.high, period=self.p.swing_lookback)

        self.order = None

    def notify_order(self, order):
        if order.status in (order.Completed, order.Canceled, order.Margin, order.Rejected):
            if order is self.order:
                self.order = None

    def next(self):
        if self.order or self.position:
            return

        price = self.data.close[0]
        rsi = self.rsi[0]

        if rsi > self.p.rsi_overbought or rsi < self.p.rsi_oversold:
            return

        long_signal = (
            price > self.ema_slow[0]
            and self.crossover[0] > 0
            and rsi < self.p.rsi_mid
        )
        short_signal = (
            price < self.ema_slow[0]
            and self.crossover[0] < 0
            and rsi > self.p.rsi_mid
        )

        if long_signal:
            stop_loss = min(self.swing_low[0], self.ema_slow[0])
            stop_distance = price - stop_loss
            if stop_distance <= 0:
                return
            take_profit = price + 2 * stop_distance
            risk_amount = self.p.risk_pct * self.broker.getvalue()
            size = int(risk_amount / stop_distance)
            if size < 1:
                return
            orders = self.buy_bracket(
                size=size,
                exectype=bt.Order.Market,
                stopprice=stop_loss,
                limitprice=take_profit,
            )
            self.order = orders[0]

        elif short_signal:
            stop_loss = max(self.swing_high[0], self.ema_slow[0])
            stop_distance = stop_loss - price
            if stop_distance <= 0:
                return
            take_profit = price - 2 * stop_distance
            risk_amount = self.p.risk_pct * self.broker.getvalue()
            size = int(risk_amount / stop_distance)
            if size < 1:
                return
            orders = self.sell_bracket(
                size=size,
                exectype=bt.Order.Market,
                stopprice=stop_loss,
                limitprice=take_profit,
            )
            self.order = orders[0]
