import math
import backtrader as bt


class QuasiCreditSpreads(bt.Strategy):
    params = (("risk_pct", 0.015),)

    def __init__(self):
        self.sma50 = bt.indicators.SMA(self.data.close, period=50)
        self.bb = bt.indicators.BollingerBands(
            self.data.close, period=20, devfactor=2.0
        )
        self.atr = bt.indicators.ATR(self.data, period=14)
        self.rsi = bt.indicators.RSI(self.data.close, period=14)
        self.sma_vol = bt.indicators.SMA(self.data.volume, period=20)

        self.bb_width = (self.bb.top - self.bb.bot) / self.bb.mid
        self.sma_bb_width = bt.indicators.SMA(self.bb_width, period=100)

        self.order = None
        self.entry_price = None
        self.entry_sma20 = None
        self.entry_atr = None
        self.bars_in_trade = 0

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            if self.position:
                if self.entry_price is None:
                    self.entry_price = order.executed.price
                    self.entry_sma20 = self.bb.mid[0]
                    self.entry_atr = self.atr[0]
                    self.bars_in_trade = 0
            else:
                self.entry_price = None
                self.entry_sma20 = None
                self.entry_atr = None
                self.bars_in_trade = 0

        self.order = None

    def next(self):
        if self.order:
            return

        if not self.position:
            low_vol = self.bb_width[0] < self.sma_bb_width[0]
            normal_vol = self.data.volume[0] <= 1.5 * self.sma_vol[0]
            no_big_gap = (
                abs(self.data.open[0] - self.data.close[-1])
                < 1.5 * self.atr[0]
            )

            filters_met = low_vol and normal_vol and no_big_gap

            if not filters_met:
                return

            long_cond = (
                self.data.close[0] <= self.bb.bot[0]
                and self.data.close[0] > self.sma50[0]
                and self.rsi[0] < 45
            )

            short_cond = (
                self.data.close[0] >= self.bb.top[0]
                and self.data.close[0] < self.sma50[0]
                and self.rsi[0] > 55
            )

            if long_cond or short_cond:
                equity = self.broker.getvalue()
                risk_amt = equity * self.p.risk_pct
                per_share_risk = 1.2 * self.atr[0]
                if per_share_risk <= 0:
                    return
                size = math.floor(risk_amt / per_share_risk)
                if size <= 0:
                    return

                if long_cond:
                    self.order = self.buy(size=size)
                elif short_cond:
                    self.order = self.sell(size=size)

        else:
            self.bars_in_trade += 1

            if (
                self.entry_price is None
                or self.entry_sma20 is None
                or self.entry_atr is None
            ):
                return

            entry = self.entry_price
            sma20_entry = self.entry_sma20
            target_price = entry - 0.7 * (entry - sma20_entry)
            stop_loss = (
                entry - 1.2 * self.entry_atr
                if self.position.size > 0
                else entry + 1.2 * self.entry_atr
            )

            close_price = self.data.close[0]

            if self.position.size > 0:
                tp_triggered = (close_price >= target_price) or (
                    close_price >= self.bb.mid[0]
                )
                sl_triggered = close_price <= stop_loss
                early_assignment = close_price < (
                    self.bb.bot[0] - 0.5 * self.atr[0]
                )
                time_exit = self.bars_in_trade >= 6

                if tp_triggered or sl_triggered or early_assignment or time_exit:
                    self.order = self.close()

            elif self.position.size < 0:
                tp_triggered = (close_price <= target_price) or (
                    close_price <= self.bb.mid[0]
                )
                sl_triggered = close_price >= stop_loss
                early_assignment = close_price > (
                    self.bb.top[0] + 0.5 * self.atr[0]
                )
                time_exit = self.bars_in_trade >= 6

                if tp_triggered or sl_triggered or early_assignment or time_exit:
                    self.order = self.close()
