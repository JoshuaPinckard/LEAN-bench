```python
import datetime
import backtrader as bt


class CreditSpreadProxyStrategy(bt.Strategy):
    params = dict(
        sma_period=20,
        trend_lookback=30,
        boll_period=20,
        boll_devfactor=2.0,
        atr_period=14,
        tp_atr_mult=0.25,
        sl_atr_mult=1.25,
        risk_pct=0.01,
        risk_capital=4000.0,
        vol_filter_pct=0.02,
        trade_start_time=datetime.time(11, 0),
        flatten_time=datetime.time(15, 30),
    )

    def __init__(self):
        self.sma = bt.indicators.SMA(self.data.close, period=self.p.sma_period)
        self.sma_prior = self.sma(-self.p.trend_lookback)
        self.boll = bt.indicators.BollingerBands(
            self.data.close, period=self.p.boll_period, devfactor=self.p.boll_devfactor
        )
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.cross_bot = bt.indicators.CrossOver(self.data.close, self.boll.bot)
        self.cross_top = bt.indicators.CrossOver(self.data.close, self.boll.top)

        self.order = None
        self.tp_price = None
        self.sl_price = None

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return
        if order.status in (order.Completed, order.Canceled, order.Margin, order.Rejected):
            self.order = None

    def notify_trade(self, trade):
        if trade.isclosed:
            self.tp_price = None
            self.sl_price = None

    def next(self):
        bar_time = self.data.datetime.time(0)

        if self.position:
            if self.order:
                return
            if self.position.size > 0:
                if self.data.low[0] <= self.sl_price:
                    self.order = self.close()
                    return
                if self.data.high[0] >= self.tp_price:
                    self.order = self.close()
                    return
            else:
                if self.data.high[0] >= self.sl_price:
                    self.order = self.close()
                    return
                if self.data.low[0] <= self.tp_price:
                    self.order = self.close()
                    return
            if bar_time >= self.p.flatten_time:
                self.order = self.close()
            return

        if self.order:
            return

        if bar_time < self.p.trade_start_time or bar_time >= self.p.flatten_time:
            return

        close = self.data.close[0]
        sma = self.sma[0]
        sma_prior = self.sma_prior[0]

        bullish = close > sma and sma > sma_prior
        bearish = close < sma and sma < sma_prior
        if not bullish and not bearish:
            return

        atr_val = self.atr[0]
        if (atr_val / close) > self.p.vol_filter_pct:
            return

        mid = self.boll.mid[0]
        top = self.boll.top[0]
        bot = self.boll.bot[0]
        risk_amount = self.p.risk_pct * self.p.risk_capital

        if bullish and self.cross_bot[0] > 0:
            entry_est = close
            tp_distance = min(self.p.tp_atr_mult * atr_val, abs(mid - entry_est))
            sl_distance = max(self.p.sl_atr_mult * atr_val, abs(top - entry_est))
            size = int(risk_amount // sl_distance)
            if size < 1:
                return
            self.tp_price = entry_est + tp_distance
            self.sl_price = entry_est - sl_distance
            self.order = self.buy(size=size)
        elif bearish and self.cross_top[0] < 0:
            entry_est = close
            tp_distance = min(self.p.tp_atr_mult * atr_val, abs(entry_est - mid))
            sl_distance = max(self.p.sl_atr_mult * atr_val, abs(entry_est - bot))
            size = int(risk_amount // sl_distance)
            if size < 1:
                return
            self.tp_price = entry_est - tp_distance
            self.sl_price = entry_est + sl_distance
            self.order = self.sell(size=size)
```
