import backtrader as bt


class TrendFollowingPullbackStrategy(bt.Strategy):
    params = (
        ('atr_period', 14),
        ('ema_period', 50),
        ('pullback_atr', 1.5),
        ('take_profit_atr', 2.5),
        ('max_entries', 4),
        ('capital_fraction', 0.01),
    )

    def __init__(self):
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.ema = bt.indicators.EMA(self.data.close, period=self.p.ema_period)

        self.entry_order = None
        self.exit_order = None
        self.stop_orders = []
        self.entries = []
        self.recent_high = None
        self.entry_high = None
        self.entries_in_direction = 0

    def notify_order(self, order):
        if order.status in (order.Completed, order.Canceled,
                            order.Margin, order.Rejected):
            if order is self.entry_order:
                self.entry_order = None

                if order.status == order.Completed:
                    atr_at_entry = float(self.atr[0])
                    entry_price = order.executed.price
                    size = order.executed.size
                    stop_price = entry_price - self.p.pullback_atr * atr_at_entry

                    self.entries.append({
                        'price': entry_price,
                        'size': size,
                        'atr': atr_at_entry,
                        'stop_price': stop_price,
                    })

                    stop_order = self.sell(
                        size=size,
                        exectype=bt.Order.Stop,
                        price=stop_price
                    )
                    self.stop_orders.append(stop_order)
                    self.entry_high = float(self.data.high[0])

            if order is self.exit_order:
                self.exit_order = None

            if order in self.stop_orders:
                self.stop_orders.remove(order)

                if order.status == order.Completed:
                    remaining = abs(order.executed.size)
                    for entry in self.entries:
                        if remaining <= 0:
                            break
                        removed = min(entry['size'], remaining)
                        entry['size'] -= removed
                        remaining -= removed
                    self.entries = [e for e in self.entries if e['size'] > 0]

    def _cancel_stops(self):
        for order in list(self.stop_orders):
            self.cancel(order)
        self.stop_orders = []

    def _close_entire_position(self):
        if self.exit_order is None and self.position.size:
            self._cancel_stops()
            self.exit_order = self.close()
            self.entries = []
            self.entries_in_direction = 0
            self.recent_high = None
            self.entry_high = None

    def next(self):
        close = float(self.data.close[0])
        high = float(self.data.high[0])
        atr = float(self.atr[0])

        if self.position.size:
            if close < float(self.ema[0]):
                self._close_entire_position()
                return

            for entry in self.entries:
                take_profit = entry['price'] + self.p.take_profit_atr * entry['atr']
                if high >= take_profit:
                    self._close_entire_position()
                    return

            if self.entry_high is None or high > self.entry_high:
                self.entry_high = high

            if (self.entry_order is None and
                    self.exit_order is None and
                    self.entries_in_direction < self.p.max_entries and
                    close <= self.entry_high - self.p.pullback_atr * atr):
                capital = self.broker.getvalue()
                size = (capital * self.p.capital_fraction) / close
                if size > 0:
                    self.entry_order = self.buy(size=size)
                    self.entries_in_direction += 1
            return

        if self.entry_order is not None or self.exit_order is not None:
            return

        self.entries = []
        self.entries_in_direction = 0
        self.entry_high = None

        if close > float(self.ema[0]):
            if self.recent_high is None or high > self.recent_high:
                self.recent_high = high

            if close <= self.recent_high - self.p.pullback_atr * atr:
                capital = self.broker.getvalue()
                size = (capital * self.p.capital_fraction) / close
                if size > 0:
                    self.entry_order = self.buy(size=size)
                    self.entries_in_direction = 1
        else:
            self.recent_high = None
