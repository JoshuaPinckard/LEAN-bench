import backtrader as bt


class ATRDailyRangeReversalStrategy(bt.Strategy):
    params = (
        ('atr_period', 14),
        ('first_window_bars', 24),
        ('range_threshold_pct', 0.8),
        ('zone_pct', 0.15),
        ('stop_buffer_mult', 0.1),
        ('risk_pct', 0.01),
    )

    def __init__(self):
        self.order = None

        self.current_day = None
        self.day_high = None
        self.day_low = None
        self.bars_today = 0
        self.day_activated = False

        self.daily_atr = None
        self.prev_daily_close = None
        self.tr_seed = []

        self.stop_price = None
        self.target_price = None

    def _finalize_previous_day(self):
        if self.current_day is None:
            return
        prev_high = self.day_high
        prev_low = self.day_low
        prev_close = self.data.close[-1]

        if self.prev_daily_close is None:
            tr = prev_high - prev_low
        else:
            tr = max(
                prev_high - prev_low,
                abs(prev_high - self.prev_daily_close),
                abs(prev_low - self.prev_daily_close),
            )

        if self.daily_atr is None:
            self.tr_seed.append(tr)
            if len(self.tr_seed) >= self.p.atr_period:
                self.daily_atr = sum(self.tr_seed) / float(self.p.atr_period)
        else:
            self.daily_atr = (self.daily_atr * (self.p.atr_period - 1) + tr) / float(self.p.atr_period)

        self.prev_daily_close = prev_close

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def _enter_long(self):
        entry_price = self.data.close[0]
        stop_price = self.day_low - self.p.stop_buffer_mult * self.daily_atr
        target_price = self.day_high
        stop_distance = entry_price - stop_price
        if stop_distance <= 0:
            return
        risk_amount = self.p.risk_pct * self.broker.getvalue()
        size = int(risk_amount / stop_distance)
        if size < 1:
            return
        self.stop_price = stop_price
        self.target_price = target_price
        self.order = self.buy(size=size)

    def _enter_short(self):
        entry_price = self.data.close[0]
        stop_price = self.day_high + self.p.stop_buffer_mult * self.daily_atr
        target_price = self.day_low
        stop_distance = stop_price - entry_price
        if stop_distance <= 0:
            return
        risk_amount = self.p.risk_pct * self.broker.getvalue()
        size = int(risk_amount / stop_distance)
        if size < 1:
            return
        self.stop_price = stop_price
        self.target_price = target_price
        self.order = self.sell(size=size)

    def next(self):
        dt = self.data.datetime.date(0)
        new_day = (self.current_day is None or dt != self.current_day)

        if new_day:
            self._finalize_previous_day()
            self.current_day = dt
            self.day_high = self.data.high[0]
            self.day_low = self.data.low[0]
            self.bars_today = 1
            self.day_activated = False
        else:
            self.day_high = max(self.day_high, self.data.high[0])
            self.day_low = min(self.day_low, self.data.low[0])
            self.bars_today += 1

        bar_time = self.data.datetime.time(0)
        is_eod = (bar_time.hour, bar_time.minute) >= (15, 55)

        if self.position:
            if self.order is None:
                if is_eod:
                    self.order = self.close()
                elif self.position.size > 0:
                    if self.data.low[0] <= self.stop_price or self.data.high[0] >= self.target_price:
                        self.order = self.close()
                elif self.position.size < 0:
                    if self.data.high[0] >= self.stop_price or self.data.low[0] <= self.target_price:
                        self.order = self.close()
            return

        if self.order is not None:
            return

        if is_eod:
            return

        if self.daily_atr is not None and not self.day_activated and self.bars_today <= self.p.first_window_bars:
            if (self.day_high - self.day_low) >= self.p.range_threshold_pct * self.daily_atr:
                self.day_activated = True

        if not self.day_activated or self.daily_atr is None:
            return

        day_range = self.day_high - self.day_low
        if day_range <= 0:
            return

        support_zone_top = self.day_low + self.p.zone_pct * day_range
        resistance_zone_bottom = self.day_high - self.p.zone_pct * day_range

        o = self.data.open[0]
        h = self.data.high[0]
        l = self.data.low[0]
        c = self.data.close[0]

        body = abs(c - o)
        upper_wick = h - max(c, o)
        lower_wick = min(c, o) - l

        bullish_pin = body > 0 and lower_wick >= 2 * body and upper_wick <= body and c > o
        bearish_pin = body > 0 and upper_wick >= 2 * body and lower_wick <= body and c < o

        if l <= support_zone_top and bullish_pin:
            self._enter_long()
        elif h >= resistance_zone_bottom and bearish_pin:
            self._enter_short()
