import backtrader as bt
import math
import collections
import statistics


class FutureReturnDistributionStrategy(bt.Strategy):
    params = (
        ('fast_ma', 13),
        ('slow_ma', 40),
        ('rsi_period', 14),
        ('macd_fast', 12),
        ('macd_slow', 26),
        ('macd_signal', 9),
        ('bars_per_window', 6),        # 6 x 5min = one 30-minute candle
        ('ma_crosses_required', 1),    # ma_crosses must equal 1
        ('rsi_threshold', 70.0),       # avg_rsi must be above 70
        ('hold_periods', 6),           # close after 6 x 30-minute periods
        ('stop_loss', 0.015),          # 1.5% below entry
        ('take_profit', 0.030),        # 3.0% above entry
        ('capital_fraction', 0.99),    # ~100% of available capital
    )

    def __init__(self):
        # --- Indicators computed on the 5-minute feed ---
        # Two moving averages: fast SMA(13), slow SMA(40)
        self.fast_sma = bt.indicators.SMA(self.data.close, period=self.p.fast_ma)
        self.slow_sma = bt.indicators.SMA(self.data.close, period=self.p.slow_ma)
        # +1 when fast crosses above slow (used to count bullish crossovers)
        self.crossover = bt.indicators.CrossOver(self.fast_sma, self.slow_sma)
        # RSI with the standard period of 14
        self.rsi = bt.indicators.RSI(self.data.close, period=self.p.rsi_period)
        # MACD with default settings (12/26/9) -- required indicator in the set.
        self.macd = bt.indicators.MACD(
            self.data.close,
            period_me1=self.p.macd_fast,
            period_me2=self.p.macd_slow,
            period_signal=self.p.macd_signal,
        )

        # --- Order / position bookkeeping ---
        self.order = None
        self.entry_price = None
        self.entry_bucket_id = None

        # --- 30-minute window aggregation state ---
        self.cur_bucket_key = None
        self.bucket_id = -1
        self.window_crosses = 0
        self.window_rsi = []

    def _bucket_key(self, dt):
        # Identify the clock-aligned 30-minute window this 5-minute bar belongs to.
        minutes = dt.hour * 60 + dt.minute
        return (dt.year, dt.month, dt.day, minutes // 30)

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return
        if order.status == order.Completed:
            if order.isbuy():
                self.entry_price = order.executed.price
                self.entry_bucket_id = self.bucket_id
        if order.status in (order.Completed, order.Canceled,
                            order.Margin, order.Rejected):
            self.order = None

    def notify_trade(self, trade):
        if trade.isclosed:
            self.entry_price = None
            self.entry_bucket_id = None

    def _finalize_window(self):
        # Evaluate the just-completed 30-minute candle for an entry.
        n = len(self.window_rsi)
        if n < self.p.bars_per_window:
            # Only act on full 30-minute candles (ignore partial warm-up windows).
            return

        ma_crosses = self.window_crosses
        avg_rsi = statistics.mean(self.window_rsi)

        if self.order or self.position:
            return

        # Entry filter: exactly one bullish MA crossover and average RSI above 70.
        if ma_crosses == self.p.ma_crosses_required and avg_rsi > self.p.rsi_threshold:
            price = self.data.close[0]
            cash = self.broker.get_cash()
            size = int((cash * self.p.capital_fraction) / price)
            if size > 0:
                self.order = self.buy(size=size)

    def next(self):
        dt = self.data.datetime.datetime(0)
        key = self._bucket_key(dt)

        # Detect a 30-minute window boundary.
        if key != self.cur_bucket_key:
            if self.cur_bucket_key is not None:
                self._finalize_window()
            self.cur_bucket_key = key
            self.bucket_id += 1
            self.window_crosses = 0
            self.window_rsi = []

        # Accumulate this 5-minute bar into the current 30-minute window.
        if self.crossover[0] > 0:
            self.window_crosses += 1
        self.window_rsi.append(self.rsi[0])

        # --- Position management (checked on every 5-minute bar) ---
        if self.position and not self.order and self.entry_price is not None:
            stop_price = self.entry_price * (1.0 - self.p.stop_loss)
            take_price = self.entry_price * (1.0 + self.p.take_profit)

            if self.data.low[0] <= stop_price:
                # Stop-loss: price dropped 1.5% below entry.
                self.order = self.close()
            elif self.data.high[0] >= take_price:
                # Take-profit: price rose 3.0% above entry.
                self.order = self.close()
            elif (self.entry_bucket_id is not None and
                  (self.bucket_id - self.entry_bucket_id) >= self.p.hold_periods):
                # Fixed holding period: 6 x 30-minute periods after opening.
                self.order = self.close()
