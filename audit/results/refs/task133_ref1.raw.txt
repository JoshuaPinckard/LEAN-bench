import backtrader as bt


class StopLossATRStrategy(bt.Strategy):
    params = (
        ("sma_period", 20),
        ("atr_period", 14),
        ("take_profit_pct", 0.02),
        ("atr_stop_multiple", 2.0),
        ("capital_fraction", 0.01),
    )

    def __init__(self):
        self.sma = bt.indicators.SimpleMovingAverage(
            self.data.close, period=self.p.sma_period
        )
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.entry_order = None
        self.exit_orders = []
        self.entry_atr = None

    def next(self):
        if self.entry_order is not None or self.exit_orders:
            return

        if self.position.size:
            return

        if self.data.close[0] > self.sma[0]:
            size = (
                self.broker.getvalue()
                * self.p.capital_fraction
                / self.data.close[0]
            )
            self.entry_atr = self.atr[0]
            self.entry_order = self.buy(size=size)

    def notify_order(self, order):
        terminal = (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        )

        if order is self.entry_order:
            if order.status == order.Completed:
                entry_price = order.executed.price
                stop_price = entry_price - (
                    self.p.atr_stop_multiple * self.entry_atr
                )
                take_profit_price = entry_price * (1.0 + self.p.take_profit_pct)

                stop_order = self.sell(
                    size=order.executed.size,
                    exectype=bt.Order.Stop,
                    price=stop_price,
                )
                take_profit_order = self.sell(
                    size=order.executed.size,
                    exectype=bt.Order.Limit,
                    price=take_profit_price,
                    oco=stop_order,
                )
                self.exit_orders = [stop_order, take_profit_order]
                self.entry_order = None
            elif order.status in terminal:
                self.entry_order = None

        if order in self.exit_orders and order.status in terminal:
            if all(exit_order.status in terminal for exit_order in self.exit_orders):
                self.exit_orders = []
