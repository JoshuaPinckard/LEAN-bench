import backtrader as bt


class EMACrossoverStrategy(bt.Strategy):
    params = (
        ("FAST", 21),
        ("SLOW", 50),
        ("UNITS", 10000),
    )

    def __init__(self):
        self.fast_ema = bt.indicators.EMA(self.data.close, period=self.p.FAST)
        self.slow_ema = bt.indicators.EMA(self.data.close, period=self.p.SLOW)
        self.crossover = bt.indicators.CrossOver(self.fast_ema, self.slow_ema)
        self.order = None
        self.exit_price = None
        self.exit_datetime = None

    def log(self, text):
        print(text)

    def next(self):
        if self.order:
            return

        if self.position.size > 0:
            if self.crossover[0] < 0:
                self.order = self.close()
            return

        if self.position.size < 0:
            if self.crossover[0] > 0:
                self.order = self.close()
            return

        if self.crossover[0] > 0:
            self.order = self.buy(size=self.p.UNITS)
        elif self.crossover[0] < 0:
            self.order = self.sell(size=self.p.UNITS)

    def notify_order(self, order):
        if order.status == order.Completed:
            if order == self.order and order.size:
                if ((order.isbuy() and order.executed.size > 0 and
                     self.position.size == 0) or
                        (order.issell() and order.executed.size < 0 and
                         self.position.size == 0)):
                    self.exit_price = order.executed.price
                    self.exit_datetime = bt.num2date(order.executed.dt)

            self.order = None

        elif order.status in (order.Canceled, order.Margin, order.Rejected):
            self.order = None

    def notify_trade(self, trade):
        if trade.isclosed:
            timestamp = self.exit_datetime or self.data.datetime.datetime(0)
            price = self.exit_price if self.exit_price is not None else self.data.close[0]
            self.log(
                f"{timestamp.isoformat()}, CLOSE, Price: {price:.2f}, "
                f"PnL: {trade.pnlcomm:.2f}"
            )
            self.exit_price = None
            self.exit_datetime = None
