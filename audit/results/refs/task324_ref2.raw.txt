import backtrader as bt


class IntradayVWAPBreakout(bt.Strategy):
    params = (
        ("N", 3),
        ("stop_loss_pct", 0.005),
    )

    def __init__(self):
        self.order = None
        self.stop_order = None
        self.cum_vol = 0.0
        self.cum_pv = 0.0
        self.last_date = None
        self.vwap_vals = []

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            if order == self.order:
                self.order = None
                if self.position.size > 0:
                    stop_price = order.executed.price * (1.0 - self.p.stop_loss_pct)
                    self.stop_order = self.sell(
                        exectype=bt.Order.Stop, price=stop_price
                    )
                elif self.position.size < 0:
                    stop_price = order.executed.price * (1.0 + self.p.stop_loss_pct)
                    self.stop_order = self.buy(
                        exectype=bt.Order.Stop, price=stop_price
                    )
            elif order == self.stop_order:
                self.stop_order = None

        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            if order == self.order:
                self.order = None
            if order == self.stop_order:
                self.stop_order = None

    def next(self):
        cur_date = self.data.datetime.date(0)
        if self.last_date != cur_date:
            self.cum_vol = 0.0
            self.cum_pv = 0.0
            self.last_date = cur_date

        typical_price = (
            self.data.high[0] + self.data.low[0] + self.data.close[0]
        ) / 3.0
        volume = self.data.volume[0]

        self.cum_vol += volume
        self.cum_pv += typical_price * volume

        vwap = (
            self.cum_pv / self.cum_vol if self.cum_vol > 0.0 else typical_price
        )
        self.vwap_vals.append(vwap)

        if len(self.vwap_vals) < self.p.N + 1:
            return

        if self.order is not None:
            return

        if self.position.size > 0:
            if self.data.close[0] < vwap:
                if self.stop_order:
                    self.cancel(self.stop_order)
                    self.stop_order = None
                self.order = self.close()
            return

        if self.position.size < 0:
            if self.data.close[0] > vwap:
                if self.stop_order:
                    self.cancel(self.stop_order)
                    self.stop_order = None
                self.order = self.close()
            return

        long_cond = (self.data.close[0] > vwap) and all(
            self.data.close[-i] < self.vwap_vals[-1 - i]
            for i in range(1, self.p.N + 1)
        )

        short_cond = (self.data.close[0] < vwap) and all(
            self.data.close[-i] > self.vwap_vals[-1 - i]
            for i in range(1, self.p.N + 1)
        )

        if long_cond:
            self.order = self.buy()
        elif short_cond:
            self.order = self.sell()
