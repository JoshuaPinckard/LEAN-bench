import backtrader as bt
import statistics


class BollingerMeanReversion(bt.Strategy):
    params = (
        ("bb_period", 20),
        ("bb_dev", 1.5),
        ("sma_period", 200),
        ("atr_period", 14),
        ("atr_lookback", 252),
        ("time_stop_bars", 5),
        ("trade_value", 1000.0),
        ("max_loss_pct", 0.02),
    )

    def __init__(self):
        self.boll = bt.indicators.BollingerBands(
            self.data.close, period=self.p.bb_period, devfactor=self.p.bb_dev
        )
        self.sma200 = bt.indicators.SMA(self.data.close, period=self.p.sma_period)
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)

        self.crossover_bot = bt.indicators.CrossOver(
            self.data.close, self.boll.lines.bot
        )
        self.crossover_top = bt.indicators.CrossOver(
            self.data.close, self.boll.lines.top
        )
        self.crossover_mid = bt.indicators.CrossOver(
            self.data.close, self.boll.lines.mid
        )

        self.order = None
        self.bar_entered = 0

    def notify_order(self, order):
        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            self.order = None
            if order.status == order.Completed:
                self.bar_entered = len(self)

    def next(self):
        if self.order:
            return

        if self.position:
            capital = self.broker.getvalue()
            pnl = self.position.size * (self.data.close[0] - self.position.price)
            if pnl < -self.p.max_loss_pct * capital:
                self.order = self.close()
                return

            bars_in_trade = len(self) - self.bar_entered
            if self.crossover_mid[0] != 0 or bars_in_trade >= self.p.time_stop_bars:
                self.order = self.close()
                return
        else:
            if len(self.atr) < self.p.atr_lookback:
                return

            atr_vals = self.atr.get(size=self.p.atr_lookback)
            p20 = statistics.quantiles(atr_vals, n=5)[0]
            atr_ok = self.atr[0] >= p20

            if (
                self.crossover_bot[0] == -1.0
                and atr_ok
                and self.data.close[0] <= self.sma200[0]
            ):
                size = int(self.p.trade_value / self.data.close[0])
                if size > 0:
                    self.order = self.buy(size=size)

            elif (
                self.crossover_top[0] == 1.0
                and atr_ok
                and self.data.close[0] >= self.sma200[0]
            ):
                size = int(self.p.trade_value / self.data.close[0])
                if size > 0:
                    self.order = self.sell(size=size)
