import backtrader as bt


class OptionIncomeProxyStrategy(bt.Strategy):
    params = (
        # 'covered_call', 'cash_secured_put', 'wheel', 'buy_hold'
        ('mode', 'wheel'),
    )

    def __init__(self):
        self.order = None
        self.entry_bar = None
        self.cc_cooldown_weeks = 0
        self.bh_done = False

        if self.p.mode != 'buy_hold':
            self.bb = bt.indicators.BollingerBands(self.data, period=20, devfactor=2)
            self.sma50 = bt.indicators.SMA(self.data, period=50)
            self.rsi = bt.indicators.RSI(self.data, period=14)
            self.atr = bt.indicators.ATR(self.data, period=14)

    def notify_order(self, order):
        if order.status in (order.Completed, order.Canceled, order.Margin, order.Rejected):
            if order.status == order.Completed and order.isbuy():
                self.entry_bar = len(self)
            self.order = None

    def next(self):
        if self.p.mode == 'buy_hold':
            self._next_buy_hold()
            return

        if self.data.datetime.date(0).weekday() != 0:
            return
        if self.order is not None:
            return

        if self.p.mode == 'covered_call':
            self._next_covered_call()
        elif self.p.mode == 'cash_secured_put':
            self._next_cash_secured_put()
        elif self.p.mode == 'wheel':
            self._next_wheel()

    def _next_buy_hold(self):
        if not self.bh_done and self.order is None:
            self.order = self.buy()
            self.bh_done = True

    def _next_covered_call(self):
        close = self.data.close[0]
        if not self.position:
            if self.cc_cooldown_weeks > 0:
                self.cc_cooldown_weeks -= 1
                return
            if close > self.sma50[0] and self.rsi[0] > 50:
                self.order = self.buy()
        else:
            if close >= self.bb.lines.top[0] or self.rsi[0] >= 70:
                self.order = self.close()
                self.cc_cooldown_weeks = 1
            elif close <= self.bb.lines.mid[0] - self.atr[0]:
                self.order = self.close()
                self.cc_cooldown_weeks = 1

    def _next_cash_secured_put(self):
        close = self.data.close[0]
        if not self.position:
            if close <= self.bb.lines.bot[0] or self.rsi[0] <= 35:
                self.order = self.buy()
        else:
            bars_held = len(self) - self.entry_bar
            if close >= self.bb.lines.mid[0] or bars_held >= 20:
                self.order = self.close()
            elif close <= self.bb.lines.bot[0] - self.atr[0]:
                self.order = self.close()

    def _next_wheel(self):
        close = self.data.close[0]
        if not self.position:
            if close <= self.bb.lines.bot[0] or self.rsi[0] <= 35:
                self.order = self.buy()
        else:
            if close >= self.bb.lines.top[0] or self.rsi[0] >= 70:
                self.order = self.close()
            elif close <= self.bb.lines.mid[0] - self.atr[0]:
                self.order = self.close()
