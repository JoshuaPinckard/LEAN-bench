import backtrader as bt


class FutureReturnDistributionStrategy(bt.Strategy):
    params = dict(
        fast_period=13,          # Fast SMA
        slow_period=40,          # Slow SMA
        rsi_period=14,           # RSI standard period
        macd_fast=12,            # MACD fast EMA
        macd_slow=26,            # MACD slow EMA
        macd_signal=9,           # MACD signal SMA
        bars_per_window=6,       # 6 x 5-minute bars = one 30-minute candle
        hold_periods=6,          # exit after 6 completed 30-minute periods (return_6)
        ma_crosses_required=1,   # entry: exactly one upward SMA cross in the window
        avg_rsi_threshold=70.0,  # entry: average RSI above 70
        stop_loss=0.015,         # 1.5% stop-loss
        take_profit=0.03,        # 3.0% take-profit
        capital_fraction=0.99,   # ~100% of available capital (buffered)
    )

    def __init__(self):
        # Indicators are computed on the 5-minute feed; the two 30-minute
        # conditions are aggregated from these in next() (see assumptions).
        self.fast_sma = bt.indicators.SMA(self.data.close, period=self.p.fast_period)
        self.slow_sma = bt.indicators.SMA(self.data.close, period=self.p.slow_period)
        self.crossover = bt.indicators.CrossOver(self.fast_sma, self.slow_sma)
        self.rsi = bt.indicators.RSI(self.data.close, period=self.p.rsi_period)
        # MACD is specified among the indicators; instantiated for fidelity.
        self.macd = bt.indicators.MACD(
            self.data.close,
            period_me1=self.p.macd_fast,
            period_me2=self.p.macd_slow,
            period_signal=self.p.macd_signal,
        )

        # 30-minute window aggregation state
        self.cur_bucket = None
        self.bar_in_window = 0
        self.ma_crosses = 0
        self.rsi_sum = 0.0
        self.rsi_count = 0

        # position / order state
        self.order = None
        self.entry_price = None
        self.hold_count = 0

    def _bucket(self, dt):
        # Group 5-minute bars by calendar half-hour (open-time labeled).
        return (dt.year, dt.month, dt.day, (dt.hour * 60 + dt.minute) // 30)

    def _check_risk(self):
        if self.entry_price is None:
            return
        price = self.data.close[0]
        if price <= self.entry_price * (1.0 - self.p.stop_loss):
            self.order = self.close()
        elif price >= self.entry_price * (1.0 + self.p.take_profit):
            self.order = self.close()

    def _finalize_window(self):
        full = (self.bar_in_window == self.p.bars_per_window)
        avg_rsi = self.rsi_sum / self.rsi_count if self.rsi_count else 0.0
        ma_crosses = self.ma_crosses

        # Fixed holding period: close after 6 completed 30-minute periods.
        if self.position and self.order is None:
            self.hold_count += 1
            if self.hold_count >= self.p.hold_periods:
                self.order = self.close()
                return

        # Entry filter evaluated on the just-completed 30-minute candle.
        if full and not self.position and self.order is None:
            if (ma_crosses == self.p.ma_crosses_required
                    and avg_rsi > self.p.avg_rsi_threshold):
                cash = self.broker.get_cash()
                price = self.data.close[0]
                size = int(cash * self.p.capital_fraction / price)
                if size > 0:
                    self.order = self.buy(size=size)
                    self.hold_count = 0

    def next(self):
        dt = self.data.datetime.datetime(0)
        bucket = self._bucket(dt)

        # 1) Risk management evaluated on every 5-minute bar.
        if self.position and self.order is None:
            self._check_risk()

        # 2) 30-minute window boundary: finalize the completed candle.
        if self.cur_bucket is None:
            self.cur_bucket = bucket
        elif bucket != self.cur_bucket:
            self._finalize_window()
            self.cur_bucket = bucket
            self.bar_in_window = 0
            self.ma_crosses = 0
            self.rsi_sum = 0.0
            self.rsi_count = 0

        # 3) Accumulate the current 5-minute bar into the current 30-min window.
        self.bar_in_window += 1
        if self.crossover[0] > 0:          # fast SMA crossed above slow SMA
            self.ma_crosses += 1
        self.rsi_sum += float(self.rsi[0])
        self.rsi_count += 1

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return
        if order.status == order.Completed:
            if order.isbuy():
                self.entry_price = order.executed.price
            else:
                self.entry_price = None
                self.hold_count = 0
        # Completed / Canceled / Margin / Rejected -> clear the pending order.
        self.order = None
