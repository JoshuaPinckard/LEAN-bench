```python
import backtrader as bt


class ReturnDistributionMACrossRSIStrategy(bt.Strategy):
    params = dict(
        fast_period=13,
        slow_period=40,
        rsi_period=14,
        agg_bars=6,       # number of 5-minute bars per 30-minute candle
        hold_periods=6,   # 30-minute periods to hold after entry
        stop_loss_pct=0.015,
        take_profit_pct=0.03,
    )

    def __init__(self):
        self.fast_sma = bt.indicators.SMA(self.data.close, period=self.p.fast_period)
        self.slow_sma = bt.indicators.SMA(self.data.close, period=self.p.slow_period)
        self.rsi = bt.indicators.RSI(self.data.close, period=self.p.rsi_period)
        self.macd = bt.indicators.MACD(self.data.close)
        self.crossover = bt.indicators.CrossOver(self.fast_sma, self.slow_sma)

        self.order = None
        self.entry_price = None
        self.periods_held = 0

        self.group_cross_count = 0
        self.group_rsi_sum = 0.0
        self.group_bar_count = 0

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return
        if order.status == order.Completed:
            if order.isbuy():
                self.entry_price = order.executed.price
            else:
                self.entry_price = None
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        bar_index = len(self.data) - 1
        group_position = bar_index % self.p.agg_bars

        if group_position == 0:
            self.group_cross_count = 0
            self.group_rsi_sum = 0.0
            self.group_bar_count = 0

        if self.crossover[0] > 0:
            self.group_cross_count += 1
        self.group_rsi_sum += self.rsi[0]
        self.group_bar_count += 1

        group_complete = (group_position == self.p.agg_bars - 1)

        if self.position and self.entry_price is not None and not self.order:
            price = self.data.close[0]
            if price <= self.entry_price * (1.0 - self.p.stop_loss_pct):
                self.order = self.close()
            elif price >= self.entry_price * (1.0 + self.p.take_profit_pct):
                self.order = self.close()

        if group_complete:
            avg_rsi = self.group_rsi_sum / self.group_bar_count
            ma_crosses = self.group_cross_count

            if self.position:
                if not self.order:
                    self.periods_held += 1
                    if self.periods_held >= self.p.hold_periods:
                        self.order = self.close()
            elif not self.order:
                if ma_crosses == 1 and avg_rsi > 70:
                    size = int(self.broker.getcash() / self.data.close[0])
                    if size > 0:
                        self.order = self.buy(size=size)
                        self.periods_held = 0
```
