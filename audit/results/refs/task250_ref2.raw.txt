import backtrader as bt


class MomentumMACDStrategy(bt.Strategy):
    params = (
        ("period_me1", 12),
        ("period_me2", 26),
        ("period_signal", 9),
    )

    def __init__(self):
        self.macd = bt.ind.MACD(
            self.data.close,
            period_me1=self.p.period_me1,
            period_me2=self.p.period_me2,
            period_signal=self.p.period_signal,
        )
        self.crossover = bt.ind.CrossOver(self.macd.macd, self.macd.signal)

        self.order_buy = None
        self.order_sell = None
        self.buy_price = None

        # Track capital dynamics starting at initial $100 as specified
        self.capital = 100.0
        self.capital_history = [100.0]

    def notify_order(self, order):
        if order.status in [order.Canceled, order.Margin, order.Rejected]:
            if order == self.order_buy:
                self.order_buy = None
                if self.order_sell:
                    self.cancel(self.order_sell)
                    self.order_sell = None
            elif order == self.order_sell:
                self.order_sell = None
                if self.order_buy:
                    self.cancel(self.order_buy)
                    self.order_buy = None
            return

        if order.status == order.Completed:
            if order == self.order_buy:
                self.buy_price = order.executed.price
            elif order == self.order_sell:
                close_price = order.executed.price
                if self.buy_price is not None and self.buy_price > 0:
                    ret = (close_price - self.buy_price) / self.buy_price
                    self.capital = self.capital * (1.0 + ret)
                    self.capital_history.append(self.capital)

                self.order_buy = None
                self.order_sell = None
                self.buy_price = None

    def next(self):
        # Guard against pending orders or active positions
        if self.order_buy is not None or self.order_sell is not None:
            return

        if self.position.size != 0:
            return

        # Entry signal: MACD line crosses signal line from below to above on day t
        if self.crossover[0] > 0:
            cash = self.broker.getcash()
            price = self.data.close[0]

            if price > 0 and cash > 0:
                # Use 98% of cash to accommodate commission and open price gaps
                size = (cash * 0.98) / price
                # Open long at Open of day t+1, close at Close of day t+1
                self.order_buy = self.buy(size=size, exectype=bt.Order.Market)
                self.order_sell = self.sell(size=size, exectype=bt.Order.Close)
