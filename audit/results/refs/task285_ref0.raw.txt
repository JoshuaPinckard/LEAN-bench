import backtrader as bt


class SARATRStrategy(bt.Strategy):
    params = dict(
        atr_period=14,
        mult=2.5,
    )

    def __init__(self):
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.pending_orders = []
        self.up_stop = None
        self.down_stop = None
        self.sar = None
        self.prev_close = None
        self.stop_price = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return
        if order in self.pending_orders:
            self.pending_orders.remove(order)

    def next(self):
        close = self.data.close[0]
        atr = self.atr[0]

        if self.sar is None:
            self.up_stop = close - self.p.mult * atr
            self.down_stop = close + self.p.mult * atr
            self.sar = self.up_stop
            self.prev_close = close
            return

        new_up_stop = max(self.up_stop, close - self.p.mult * atr)
        new_down_stop = min(self.down_stop, close + self.p.mult * atr)
        prev_sar = self.sar
        new_sar = new_up_stop if close > prev_sar else new_down_stop

        cross_up = close > new_sar and self.prev_close <= prev_sar
        cross_down = close < new_sar and self.prev_close >= prev_sar

        if not self.pending_orders:
            if self.position.size > 0:
                if cross_down:
                    self.pending_orders.append(self.close())
                    self.pending_orders.append(self.sell())
                    self.stop_price = new_sar
                elif close <= self.stop_price:
                    self.pending_orders.append(self.close())
            elif self.position.size < 0:
                if cross_up:
                    self.pending_orders.append(self.close())
                    self.pending_orders.append(self.buy())
                    self.stop_price = new_sar
                elif close >= self.stop_price:
                    self.pending_orders.append(self.close())
            else:
                if cross_up:
                    self.pending_orders.append(self.buy())
                    self.stop_price = new_sar
                elif cross_down:
                    self.pending_orders.append(self.sell())
                    self.stop_price = new_sar

        self.up_stop = new_up_stop
        self.down_stop = new_down_stop
        self.sar = new_sar
        self.prev_close = close
