import backtrader as bt


class SessionVWAP(bt.Indicator):
    lines = ('vwap',)

    def __init__(self):
        self._cum_pv = 0.0
        self._cum_v = 0.0
        self._last_date = None

    def next(self):
        dt = self.data.datetime.date(0)
        if dt != self._last_date:
            self._cum_pv = 0.0
            self._cum_v = 0.0
            self._last_date = dt
        typical_price = (self.data.high[0] + self.data.low[0] + self.data.close[0]) / 3.0
        vol = self.data.volume[0]
        self._cum_pv += typical_price * vol
        self._cum_v += vol
        self.lines.vwap[0] = self._cum_pv / self._cum_v if self._cum_v else typical_price


class SlopeIndicator(bt.Indicator):
    lines = ('slope',)
    params = (('period', 20),)

    def __init__(self):
        self.lines.slope = self.data - self.data(-self.p.period)


class VWAPMeanReversion(bt.Strategy):
    params = (
        ('slope_period', 20),
        ('stddev_period', 20),
        ('vwap_base_period', 20),
        ('threshold_mult', 2.0),
        ('stop_loss_pct', 0.02),
    )

    def __init__(self):
        self.vwap = SessionVWAP(self.data)
        self.slope = SlopeIndicator(self.vwap, period=self.p.slope_period)
        self.slope_std = bt.indicators.StdDev(self.slope, period=self.p.stddev_period)
        self.threshold = self.slope_std * self.p.threshold_mult
        self.vwap_base = bt.indicators.SMA(self.vwap, period=self.p.vwap_base_period)
        self.crossover = bt.indicators.CrossOver(self.vwap, self.vwap_base)

        self.entry_order = None
        self.exit_order = None
        self.stop_order = None

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return

        if order.status == order.Completed:
            if order is self.entry_order:
                entry_price = order.executed.price
                if order.isbuy():
                    stop_price = entry_price * (1.0 - self.p.stop_loss_pct)
                    self.stop_order = self.sell(
                        exectype=bt.Order.Stop, price=stop_price, size=order.executed.size)
                else:
                    stop_price = entry_price * (1.0 + self.p.stop_loss_pct)
                    self.stop_order = self.buy(
                        exectype=bt.Order.Stop, price=stop_price, size=abs(order.executed.size))
                self.entry_order = None
            elif order is self.stop_order:
                self.stop_order = None
                self.exit_order = None
            elif order is self.exit_order:
                if self.stop_order is not None:
                    self.cancel(self.stop_order)
                    self.stop_order = None
                self.exit_order = None

        elif order.status in (order.Canceled, order.Margin, order.Rejected):
            if order is self.entry_order:
                self.entry_order = None
            elif order is self.stop_order:
                self.stop_order = None
            elif order is self.exit_order:
                self.exit_order = None

    def next(self):
        if self.entry_order is not None or self.exit_order is not None:
            return

        prior_slope = self.slope[-1]
        prior_threshold = self.threshold[-1]
        cur_slope = self.slope[0]

        if not self.position:
            long_setup = prior_slope < -prior_threshold and cur_slope > prior_slope
            short_setup = prior_slope > prior_threshold and cur_slope < prior_slope

            if long_setup:
                self.entry_order = self.buy()
            elif short_setup:
                self.entry_order = self.sell()
        else:
            if self.position.size > 0 and self.crossover[0] > 0:
                if self.stop_order is not None:
                    self.cancel(self.stop_order)
                self.exit_order = self.close()
            elif self.position.size < 0 and self.crossover[0] < 0:
                if self.stop_order is not None:
                    self.cancel(self.stop_order)
                self.exit_order = self.close()
