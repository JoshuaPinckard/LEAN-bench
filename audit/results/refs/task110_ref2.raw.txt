import backtrader as bt


class RSIReversal(bt.Strategy):
    params = (
        ('rsi_period', 14),
        ('oversold_level', 30.0),
        ('overbought_level', 70.0),
    )

    def __init__(self):
        self.rsi = bt.indicators.RSI(self.data.close, period=self.p.rsi_period)
        self.order = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return
        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            self.order = None

    def next(self):
        if self.order:
            return

        cross_above_30 = (
            self.rsi[0] > self.p.oversold_level
            and self.rsi[-1] <= self.p.oversold_level
        )
        cross_below_70 = (
            self.rsi[0] < self.p.overbought_level
            and self.rsi[-1] >= self.p.overbought_level
        )

        if not self.position:
            if cross_above_30:
                self.order = self.buy()
        else:
            if cross_below_70:
                self.order = self.close()
