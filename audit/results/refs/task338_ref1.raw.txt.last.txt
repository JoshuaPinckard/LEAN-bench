import backtrader as bt
import math


class JanuaryEffectStrategy(bt.Strategy):
    params = (("stop_loss_pct", 0.02),)

    def __init__(self):
        self.sma = bt.indicators.SimpleMovingAverage(self.data.close, period=20)
        self.order = None
        self.stop_order = None

    def notify_order(self, order):
        if order.status == order.Completed:
            if order == self.order:
                if order.isbuy():
                    self.stop_order = self.sell(
                        exectype=bt.Order.Stop,
                        price=order.executed.price * (1 - self.p.stop_loss_pct),
                    )
                self.order = None
            elif order == self.stop_order:
                self.stop_order = None

        elif order.status in (order.Canceled, order.Margin, order.Rejected):
            if order == self.order:
                self.order = None
            elif order == self.stop_order:
                self.stop_order = None

    def next(self):
        current_date = self.data.datetime.date(0)
        previous_date = self.data.datetime.date(-1) if len(self) > 1 else None

        first_january_trading_day = (
            current_date.month == 1
            and (previous_date is None or previous_date.month != 1)
        )
        first_february_trading_day = (
            current_date.month == 2
            and (previous_date is None or previous_date.month != 2)
        )

        if first_january_trading_day and not self.position and self.order is None:
            commission = self.broker.getcommissioninfo(self.data).p.commission
            size = math.floor(
                self.broker.getcash() / (self.data.close[0] * (1 + commission))
            )
            if size > 0:
                self.order = self.buy(size=size)

        elif first_february_trading_day and self.position and self.order is None:
            if self.stop_order is not None:
                self.cancel(self.stop_order)
            self.order = self.close()
