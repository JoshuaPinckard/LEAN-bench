import backtrader as bt


class VolatilityTrendBandStrategy(bt.Strategy):
    params = dict(
        atr_period=10,
        atr_multiplier=3,
        ma_period=20,
        band_period=20,
        band_mult=2.0,
        expansion_mult=1.0,
    )

    def __init__(self):
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.ma = bt.indicators.EMA(self.data.close, period=self.p.ma_period)
        self.signal = self.data.close - self.ma
        self.signal_mean = bt.indicators.SMA(self.signal, period=self.p.band_period)
        self.signal_std = bt.indicators.StdDev(self.signal, period=self.p.band_period)
        self.upper_band = self.signal_mean + self.p.band_mult * self.signal_std
        self.lower_band = self.signal_mean - self.p.band_mult * self.signal_std

        self.cross_upper = bt.indicators.CrossOver(self.signal, self.upper_band)
        self.cross_lower = bt.indicators.CrossOver(self.signal, self.lower_band)
        self.cross_mean = bt.indicators.CrossOver(self.signal, self.signal_mean)

        self.order = None
        self.stop_order = None
        self.entry_atr = None

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return

        if order == self.order:
            if order.status == order.Completed:
                if order.isbuy():
                    stop_price = order.executed.price - self.p.atr_multiplier * self.entry_atr
                    self.stop_order = self.sell(exectype=bt.Order.Stop, price=stop_price)
                else:
                    stop_price = order.executed.price + self.p.atr_multiplier * self.entry_atr
                    self.stop_order = self.buy(exectype=bt.Order.Stop, price=stop_price)
            self.order = None

        elif order == self.stop_order:
            if order.status in (order.Completed, order.Canceled, order.Margin, order.Rejected):
                self.stop_order = None

    def next(self):
        if self.order:
            return

        if not self.position:
            expansion = abs(self.signal[0] - self.signal[-1]) > self.p.expansion_mult * self.signal_std[0]

            if self.cross_upper[0] > 0 and expansion:
                self.entry_atr = self.atr[0]
                self.order = self.buy()
            elif self.cross_lower[0] < 0 and expansion:
                self.entry_atr = self.atr[0]
                self.order = self.sell()

        else:
            if self.position.size > 0 and self.cross_mean[0] < 0:
                if self.stop_order:
                    self.cancel(self.stop_order)
                self.order = self.close()
            elif self.position.size < 0 and self.cross_mean[0] > 0:
                if self.stop_order:
                    self.cancel(self.stop_order)
                self.order = self.close()
