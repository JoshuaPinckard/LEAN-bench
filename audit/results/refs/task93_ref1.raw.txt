import backtrader as bt


class MomentumVolatilityStrategy(bt.Strategy):
    params = (
        ('fast_period', 20),
        ('slow_period', 50),
        ('short_vol_period', 10),
        ('long_vol_period', 30),
    )

    def __init__(self):
        self.order = None

        self.fast_sma = bt.indicators.SimpleMovingAverage(
            self.data.open, period=self.p.fast_period
        )
        self.slow_sma = bt.indicators.SimpleMovingAverage(
            self.data.close, period=self.p.slow_period
        )
        self.short_volatility = bt.indicators.StandardDeviation(
            self.data.close, period=self.p.short_vol_period
        )
        self.long_volatility = bt.indicators.StandardDeviation(
            self.data.close, period=self.p.long_vol_period
        )
        self.crossover = bt.indicators.CrossOver(self.fast_sma, self.slow_sma)

    def next(self):
        if self.order:
            return

        if not self.position:
            if (self.crossover[0] > 0 and
                    self.long_volatility[0] > self.short_volatility[0]):
                self.order = self.order_target_percent(target=1.0)
        else:
            if (self.crossover[0] < 0 or
                    self.long_volatility[0] <= self.short_volatility[0]):
                self.order = self.close()

    def notify_order(self, order):
        if order.status in (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            self.order = None
