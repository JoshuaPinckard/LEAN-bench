import backtrader as bt
import statistics
from collections import deque


class RegimeFilteredScalper(bt.Strategy):
    def __init__(self):
        self.bbands = bt.indicators.BollingerBands(self.data.close, period=20, devfactor=2.0)
        self.rsi = bt.indicators.RSI(self.data.close, period=14)
        self.ema9 = bt.indicators.EMA(self.data.close, period=9)
        self.ema21 = bt.indicators.EMA(self.data.close, period=21)
        self.crossover_trend = bt.indicators.CrossOver(self.ema9, self.ema21)
        self.atr = bt.indicators.ATR(self.data, period=14)

        self.order = None
        self.position_type = None
        self.entry_price = None
        self.entry_atr = None
        self.mr_tp_mult = None
        self.pending_position_type = None
        self.pending_mr_tp_mult = None

        self.session_date = None
        self.bar_idx_in_session = 0
        self.first_bar_open = None
        self.first6_close = None
        self.bb_width_sum = 0.0
        self.spotvol_value = None
        self.day_direction = None
        self.current_regime = None
        self.regime_ready = False

        self.gamma_history = deque(maxlen=20)
        self.spotvol_history = deque(maxlen=20)

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return
        if order.status == order.Completed:
            if self.position.size == 0:
                self.position_type = None
                self.entry_price = None
                self.entry_atr = None
                self.mr_tp_mult = None
            else:
                self.position_type = self.pending_position_type
                self.entry_price = order.executed.price
                self.entry_atr = self.atr[0]
                if self.position_type == 'mr':
                    self.mr_tp_mult = self.pending_mr_tp_mult
        self.order = None

    def next(self):
        current_date = self.data.datetime.date(0)
        if current_date != self.session_date:
            self.session_date = current_date
            self.bar_idx_in_session = 0
            self.bb_width_sum = 0.0
            self.first6_close = None
            self.regime_ready = False
            self.first_bar_open = self.data.open[0]
            if len(self) > 1:
                prev_close = self.data.close[-1]
                self.spotvol_value = abs(self.data.open[0] - prev_close) / prev_close
            else:
                self.spotvol_value = None

        self.bar_idx_in_session += 1

        if self.bar_idx_in_session <= 6:
            self.bb_width_sum += (self.bbands.top[0] - self.bbands.bot[0])
            if self.bar_idx_in_session == 6:
                self.first6_close = self.data.close[0]
                gamma_value = self.bb_width_sum / 6.0
                agg_return = (self.first6_close - self.first_bar_open) / self.first_bar_open
                self.day_direction = 1 if agg_return > 0 else -1

                if len(self.gamma_history) > 0 and self.spotvol_value is not None:
                    gamma_median = statistics.median(self.gamma_history)
                    spotvol_median = statistics.median(self.spotvol_history)
                    gamma_class = 'H' if gamma_value < gamma_median else 'L'
                    spotvol_class = 'H' if self.spotvol_value > spotvol_median else 'L'
                    self.current_regime = gamma_class + spotvol_class
                    self.regime_ready = True

                if self.spotvol_value is not None:
                    self.gamma_history.append(gamma_value)
                    self.spotvol_history.append(self.spotvol_value)

        if self.position:
            if self.order is None:
                self._manage_exit()
            return

        if self.order is not None or not self.regime_ready or self.bar_idx_in_session <= 6:
            return

        self._check_entries()

    def _manage_exit(self):
        if self.position.size > 0:
            if self.position_type == 'mr':
                stop = self.entry_price - 1.0 * self.entry_atr
                tp = self.entry_price + self.mr_tp_mult * self.entry_atr
                if (self.data.close[0] >= self.bbands.mid[0] or self.rsi[0] > 70
                        or self.data.close[0] <= stop or self.data.close[0] >= tp):
                    self.order = self.close()
            elif self.position_type == 'trend':
                stop = self.entry_price - 1.5 * self.entry_atr
                tp = self.entry_price + 2.0 * self.entry_atr
                if (self.crossover_trend[0] < 0
                        or self.data.close[0] <= stop or self.data.close[0] >= tp):
                    self.order = self.close()
        elif self.position.size < 0:
            if self.position_type == 'mr':
                stop = self.entry_price + 1.0 * self.entry_atr
                tp = self.entry_price - self.mr_tp_mult * self.entry_atr
                if (self.data.close[0] <= self.bbands.mid[0] or self.rsi[0] < 30
                        or self.data.close[0] >= stop or self.data.close[0] <= tp):
                    self.order = self.close()
            elif self.position_type == 'trend':
                stop = self.entry_price + 1.5 * self.entry_atr
                tp = self.entry_price - 2.0 * self.entry_atr
                if (self.crossover_trend[0] > 0
                        or self.data.close[0] >= stop or self.data.close[0] <= tp):
                    self.order = self.close()

    def _check_entries(self):
        regime = self.current_regime
        rsi_cross_up_30 = self.rsi[-1] <= 30 and self.rsi[0] > 30
        rsi_cross_down_70 = self.rsi[-1] >= 70 and self.rsi[0] < 70
        bb_touch_low = self.data.low[0] <= self.bbands.bot[0]
        bb_touch_high = self.data.high[0] >= self.bbands.top[0]

        mr_long = False
        mr_short = False
        mr_long_tp = 1.0
        mr_short_tp = 1.0

        if regime[0] == 'H':
            if rsi_cross_up_30:
                mr_long = True
            if rsi_cross_down_70:
                mr_short = True
            if regime == 'HH':
                if bb_touch_low:
                    mr_long = True
                    mr_long_tp = 0.75
                if bb_touch_high:
                    mr_short = True
                    mr_short_tp = 0.75
        else:
            if rsi_cross_up_30 and bb_touch_low:
                mr_long = True
            if rsi_cross_down_70 and bb_touch_high:
                mr_short = True
            if regime == 'LL':
                if mr_long and not (self.crossover_trend[0] > 0):
                    mr_long = False
                if mr_short and not (self.crossover_trend[0] < 0):
                    mr_short = False

        trend_long = False
        trend_short = False
        if regime == 'HH':
            trend_long = self.crossover_trend[0] > 0
            trend_short = self.crossover_trend[0] < 0
        elif regime == 'LH':
            if self.day_direction > 0:
                trend_long = self.crossover_trend[0] > 0
            else:
                trend_short = self.crossover_trend[0] < 0

        if mr_long:
            self.pending_position_type = 'mr'
            self.pending_mr_tp_mult = mr_long_tp
            self.order = self.buy()
        elif mr_short:
            self.pending_position_type = 'mr'
            self.pending_mr_tp_mult = mr_short_tp
            self.order = self.sell()
        elif trend_long:
            self.pending_position_type = 'trend'
            self.order = self.buy()
        elif trend_short:
            self.pending_position_type = 'trend'
            self.order = self.sell()
