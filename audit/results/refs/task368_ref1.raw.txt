import backtrader as bt


class WilliamsRMeanReversionStrategy(bt.Strategy):
    def __init__(self):
        self.wr = bt.indicators.WilliamsR(self.data, period=14)
        self.sma_wr = bt.indicators.SimpleMovingAverage(self.wr, period=20)
        self.dev = self.wr - self.sma_wr
        self.stddev = bt.indicators.StdDev(self.dev, period=20)

        self.order = None
        self.stop_order = None

    def next(self):
        if self.order:
            return

        if not self.position:
            entry_filter = abs(self.dev[0]) > 2 * self.stddev[0]
            if not entry_filter:
                return

            if self.wr[-1] <= -80 and self.wr[0] > -80:
                self.order = self.buy()
            elif self.wr[-1] >= -20 and self.wr[0] < -20:
                self.order = self.sell()
            return

        if self.position.size > 0:
            if self.wr[-1] >= self.sma_wr[-1] and self.wr[0] < self.sma_wr[0]:
                if self.stop_order:
                    self.cancel(self.stop_order)
                self.order = self.close()
        else:
            if self.wr[-1] <= self.sma_wr[-1] and self.wr[0] > self.sma_wr[0]:
                if self.stop_order:
                    self.cancel(self.stop_order)
                self.order = self.close()

    def notify_order(self, order):
        if order.status in [
            order.Submitted,
            order.Accepted,
        ]:
            return

        if order is self.stop_order:
            if order.status in [
                order.Completed,
                order.Canceled,
                order.Margin,
                order.Rejected,
            ]:
                self.stop_order = None
            return

        if order is self.order:
            if order.status == order.Completed:
                if self.position.size > 0 and order.isbuy():
                    self.stop_order = self.close(
                        exectype=bt.Order.Stop,
                        price=order.executed.price * 0.98,
                    )
                elif self.position.size < 0 and order.issell():
                    self.stop_order = self.close(
                        exectype=bt.Order.Stop,
                        price=order.executed.price * 1.02,
                    )
                elif not self.position and self.stop_order:
                    self.cancel(self.stop_order)

            if order.status in [
                order.Completed,
                order.Canceled,
                order.Margin,
                order.Rejected,
            ]:
                self.order = None
