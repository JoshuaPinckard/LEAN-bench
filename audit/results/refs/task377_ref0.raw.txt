import datetime

import backtrader as bt


class IntradayMiddayReversal(bt.Strategy):
    params = dict(
        session_start=datetime.time(9, 30),
        session_end=datetime.time(16, 0),
        morning_start=datetime.time(9, 30),
        morning_end=datetime.time(11, 30),
        midday_start=datetime.time(11, 30),
        midday_end=datetime.time(13, 30),
        eod_time=datetime.time(15, 45),
        sma_period=20,
        rsi_period=14,
        trend_lookback=4,
        stop_pct=0.02,
    )

    def __init__(self):
        self.sma = bt.indicators.SMA(self.data.close, period=self.p.sma_period)
        self.rsi = bt.indicators.RSI(self.data.close, period=self.p.rsi_period)

        self.order = None
        self.entry_price = None
        self.stop_price = None

        self.current_day = None
        self.trend_computed = False
        self.morning_trend = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return
        if order.status == order.Completed:
            if self.position:
                if self.entry_price is None:
                    self.entry_price = order.executed.price
                    if self.position.size > 0:
                        self.stop_price = self.entry_price * (1 - self.p.stop_pct)
                    else:
                        self.stop_price = self.entry_price * (1 + self.p.stop_pct)
            else:
                self.entry_price = None
                self.stop_price = None
        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            pass
        self.order = None

    def next(self):
        current_time = self.data.datetime.time(0)
        current_date = self.data.datetime.date(0)

        if self.current_day != current_date:
            self.current_day = current_date
            self.trend_computed = False
            self.morning_trend = None

        if not self.trend_computed and current_time == self.p.morning_end:
            self.trend_computed = True
            if len(self) >= self.p.sma_period + self.p.trend_lookback:
                close = self.data.close[0]
                sma_now = self.sma[0]
                sma_prev = self.sma[-self.p.trend_lookback]
                if close > sma_now and sma_now > sma_prev:
                    self.morning_trend = 'up'
                elif close < sma_now and sma_now < sma_prev:
                    self.morning_trend = 'down'
                else:
                    self.morning_trend = None
            else:
                self.morning_trend = None

        if self.order:
            return

        if self.position:
            have_prev_rsi = len(self.rsi) > 1
            if self.position.size > 0:
                if self.data.low[0] <= self.stop_price:
                    self.order = self.close()
                    return
                if have_prev_rsi and self.rsi[-1] >= 50 and self.rsi[0] < 50:
                    self.order = self.close()
                    return
                if current_time >= self.p.eod_time:
                    self.order = self.close()
                    return
            else:
                if self.data.high[0] >= self.stop_price:
                    self.order = self.close()
                    return
                if have_prev_rsi and self.rsi[-1] <= 50 and self.rsi[0] > 50:
                    self.order = self.close()
                    return
                if current_time >= self.p.eod_time:
                    self.order = self.close()
                    return
            return

        if self.morning_trend is None:
            return

        if not (self.p.midday_start <= current_time < self.p.midday_end):
            return

        if len(self.rsi) <= 1:
            return

        if self.morning_trend == 'up' and self.rsi[-1] >= 50 and self.rsi[0] < 50:
            self.order = self.sell()
        elif self.morning_trend == 'down' and self.rsi[-1] <= 50 and self.rsi[0] > 50:
            self.order = self.buy()
