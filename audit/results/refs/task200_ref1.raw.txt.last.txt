import backtrader as bt


class TrendFollowingStrategy(bt.Strategy):
    params = (
        ('bb_period', 20),
        ('bb_devfactor', 2),
        ('width_sma_period', 50),
        ('macd_me1', 12),
        ('macd_me2', 26),
        ('macd_signal', 9),
    )

    def __init__(self):
        self.order = None

        self.bollinger = bt.indicators.BollingerBands(
            self.data.close,
            period=self.p.bb_period,
            devfactor=self.p.bb_devfactor,
        )
        self.band_width = self.bollinger.top - self.bollinger.bot
        self.width_sma = bt.indicators.SMA(
            self.band_width,
            period=self.p.width_sma_period,
        )

        self.macd = bt.indicators.MACD(
            self.data.close,
            period_me1=self.p.macd_me1,
            period_me2=self.p.macd_me2,
            period_signal=self.p.macd_signal,
        )
        self.crossover = bt.indicators.CrossOver(self.macd.macd, self.macd.signal)

    def next(self):
        if self.order:
            return

        if self.band_width[0] <= self.width_sma[0]:
            return

        if not self.position:
            if self.crossover[0] > 0:
                self.order = self.order_target_percent(target=1.0)
            elif self.crossover[0] < 0:
                self.order = self.order_target_percent(target=-1.0)
        elif self.position.size > 0:
            if self.crossover[0] < 0:
                self.order = self.close()
        elif self.position.size < 0:
            if self.crossover[0] > 0:
                self.order = self.close()

    def notify_order(self, order):
        if order.status in (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            self.order = None
