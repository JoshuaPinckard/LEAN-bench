import math
import backtrader as bt


class MACDATRStrategy(bt.Strategy):
    params = (
        ("macd_fast", 12),
        ("macd_slow", 26),
        ("macd_signal", 9),
        ("atr_period", 14),
        ("risk_pct", 0.01),
        ("sl_atr_mult", 1.0),
        ("tp_atr_mult", 3.0),
    )

    def __init__(self):
        self.macd = bt.ind.MACD(
            self.data.close,
            period_me1=self.p.macd_fast,
            period_me2=self.p.macd_slow,
            period_signal=self.p.macd_signal,
        )
        self.atr = bt.ind.ATR(self.data, period=self.p.atr_period)
        self.crossover = bt.ind.CrossOver(self.macd.macd, 0.0)

        self.order = None
        self.sl_order = None
        self.tp_order = None
        self.entry_atr = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status == order.Completed:
            if order == self.order:
                self.order = None
                entry_price = order.executed.price
                atr_val = self.entry_atr
                size = abs(order.executed.size)

                if order.isbuy():
                    sl_price = entry_price - self.p.sl_atr_mult * atr_val
                    tp_price = entry_price + self.p.tp_atr_mult * atr_val
                    self.sl_order = self.sell(
                        size=size, price=sl_price, exectype=bt.Order.Stop
                    )
                    self.tp_order = self.sell(
                        size=size,
                        price=tp_price,
                        exectype=bt.Order.Limit,
                        oco=self.sl_order,
                    )
                else:
                    sl_price = entry_price + self.p.sl_atr_mult * atr_val
                    tp_price = entry_price - self.p.tp_atr_mult * atr_val
                    self.sl_order = self.buy(
                        size=size, price=sl_price, exectype=bt.Order.Stop
                    )
                    self.tp_order = self.buy(
                        size=size,
                        price=tp_price,
                        exectype=bt.Order.Limit,
                        oco=self.sl_order,
                    )

            elif order == self.sl_order or order == self.tp_order:
                self.sl_order = None
                self.tp_order = None

        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            if order == self.order:
                self.order = None
            if order == self.sl_order or order == self.tp_order:
                self.sl_order = None
                self.tp_order = None

    def next(self):
        if self.order:
            return

        if not self.position:
            if self.crossover[0] > 0:
                atr_val = self.atr[0]
                if atr_val <= 0:
                    return
                risk_amount = self.broker.getvalue() * self.p.risk_pct
                size = int(risk_amount / (self.p.sl_atr_mult * atr_val))
                max_size = int(self.broker.getcash() / self.data.close[0])
                size = min(size, max_size)
                if size > 0:
                    self.entry_atr = atr_val
                    self.order = self.buy(size=size)

            elif self.crossover[0] < 0:
                atr_val = self.atr[0]
                if atr_val <= 0:
                    return
                risk_amount = self.broker.getvalue() * self.p.risk_pct
                size = int(risk_amount / (self.p.sl_atr_mult * atr_val))
                max_size = int(self.broker.getcash() / self.data.close[0])
                size = min(size, max_size)
                if size > 0:
                    self.entry_atr = atr_val
                    self.order = self.sell(size=size)
