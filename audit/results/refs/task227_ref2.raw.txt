import backtrader as bt


class UniversalPortfolio(bt.Strategy):
    def __init__(self):
        self.order = None

    def notify_order(self, order):
        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            self.order = None

    def next(self):
        # Need 252 daily returns, which require 253 close prices
        if len(self) < 253:
            return

        # Guard against pending orders
        if self.order is not None:
            return

        # Extract close prices for the 252-day window (253 bars)
        closes = [self.data.close[-i] for i in range(252, -1, -1)]
        returns = [
            closes[i] / closes[i - 1] - 1.0 for i in range(1, len(closes))
        ]

        # Candidate constant fractions w in {0.0, 0.1, ..., 1.0}
        weights = [i / 10.0 for i in range(11)]
        wealths = []

        for w in weights:
            wealth = 1.0
            for r in returns:
                wealth *= 1.0 + w * r
            wealths.append(wealth)

        total_wealth = sum(wealths)
        if total_wealth > 0:
            w_next = (
                sum(w * wealth for w, wealth in zip(weights, wealths))
                / total_wealth
            )
        else:
            w_next = 0.0

        # Rebalance target weight in asset
        self.order = self.order_target_percent(target=w_next)
