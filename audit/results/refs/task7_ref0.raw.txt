import backtrader as bt


class RSIStrategy(bt.Strategy):
    params = dict(
        rsi_period=14,
        rsi_oversold=30,
        rsi_overbought=70,
        stop_loss_pct=0.02,
        take_profit_pct=0.04,
    )

    def __init__(self):
        self.rsi = bt.indicators.RSI(self.data.close, period=self.p.rsi_period)
        self.order = None
        self.entering = False
        self.stop_price = None
        self.target_price = None

    def notify_order(self, order):
        if order.status == order.Completed and self.entering:
            entry_price = order.executed.price
            if order.isbuy():
                self.stop_price = entry_price * (1 - self.p.stop_loss_pct)
                self.target_price = entry_price * (1 + self.p.take_profit_pct)
            else:
                self.stop_price = entry_price * (1 + self.p.stop_loss_pct)
                self.target_price = entry_price * (1 - self.p.take_profit_pct)

        if order.status in (order.Completed, order.Canceled, order.Margin, order.Rejected):
            self.entering = False
            self.order = None

    def next(self):
        if self.order:
            return

        if not self.position:
            if self.rsi[-1] < self.p.rsi_oversold and self.rsi[0] >= self.p.rsi_oversold:
                self.entering = True
                self.order = self.buy()
            elif self.rsi[-1] > self.p.rsi_overbought and self.rsi[0] <= self.p.rsi_overbought:
                self.entering = True
                self.order = self.sell()
        else:
            if self.position.size > 0:
                if (self.rsi[0] > self.p.rsi_overbought
                        or self.data.low[0] <= self.stop_price
                        or self.data.high[0] >= self.target_price):
                    self.order = self.close()
            elif self.position.size < 0:
                if (self.rsi[0] < self.p.rsi_oversold
                        or self.data.high[0] >= self.stop_price
                        or self.data.low[0] <= self.target_price):
                    self.order = self.close()
