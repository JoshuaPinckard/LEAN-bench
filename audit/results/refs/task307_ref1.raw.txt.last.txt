import backtrader as bt


class AAPLIntradayADXMomentumStrategy(bt.Strategy):
    params = (
        ('AdxPeriod', 14),
        ('AdxThreshold', 25),
        ('MomentumPeriod', 5),
        ('ATRMultiplier', 2.0),
        ('AtrPeriod', 14),
    )

    def __init__(self):
        self.adx = bt.ind.ADX(self.data, period=self.p.AdxPeriod)
        self.momentum = bt.ind.Momentum(self.data, period=self.p.MomentumPeriod)
        self.momentum_cross = bt.ind.CrossOver(self.momentum, 0.0)
        self.atr = bt.ind.ATR(self.data, period=self.p.AtrPeriod)

        self.order = None
        self.entry_order = None
        self.stop_price = None

    def notify_order(self, order):
        if order.status in (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            if order.status == order.Completed and order is self.entry_order:
                if order.isbuy():
                    self.stop_price = (
                        order.executed.price
                        - self.p.ATRMultiplier * self.atr[0]
                    )
                else:
                    self.stop_price = (
                        order.executed.price
                        + self.p.ATRMultiplier * self.atr[0]
                    )

            if order is self.order:
                self.order = None
            if order is self.entry_order:
                self.entry_order = None

    def next(self):
        if self.order:
            return

        bullish_flip = self.momentum_cross[0] > 0
        bearish_flip = self.momentum_cross[0] < 0

        if not self.position:
            if self.adx[0] > self.p.AdxThreshold:
                if bullish_flip:
                    self.entry_order = self.buy()
                    self.order = self.entry_order
                elif bearish_flip:
                    self.entry_order = self.sell()
                    self.order = self.entry_order
            return

        if self.position.size > 0:
            if bearish_flip or (
                self.stop_price is not None and self.data.low[0] <= self.stop_price
            ):
                self.order = self.close()
        else:
            if bullish_flip or (
                self.stop_price is not None and self.data.high[0] >= self.stop_price
            ):
                self.order = self.close()
