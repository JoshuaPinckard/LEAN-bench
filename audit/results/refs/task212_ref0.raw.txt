```python
import backtrader as bt


class SingleAssetVolatilityRotation(bt.Strategy):
    params = dict(
        boll_period=20,
        boll_dev=2,
        atr_period=14,
        atr_sma_period=30,
        atr_filter_mult=0.5,
        rotation_pct=0.05,
        meanrev_pct=0.05,
        protect_pct=0.03,
        cooldown_days=30,
    )

    def __init__(self):
        self.boll = bt.indicators.BollingerBands(
            self.data, period=self.p.boll_period, devfactor=self.p.boll_dev)
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.atr_sma = bt.indicators.SMA(self.atr, period=self.p.atr_sma_period)

        self.order = None
        self.base_established = False
        self.in_tactical = False
        self.anchor_high = None
        self.sell_price = None
        self.tactical_size = 0.0
        self.cooldown_until = -1

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        if self.order:
            return

        if not self.base_established:
            self.order = self.buy()
            self.base_established = True
            return

        if self.in_tactical:
            price = self.data.close[0]
            if price <= self.anchor_high * (1 - self.p.meanrev_pct):
                self._buyback()
            elif price >= self.sell_price * (1 + self.p.protect_pct):
                self._buyback()
            return

        if len(self) < self.cooldown_until:
            return

        vol_ok = self.atr[0] > self.p.atr_filter_mult * self.atr_sma[0]
        if self.data.close[0] >= self.boll.top[0] and vol_ok:
            self._sell_tactical()

    def _sell_tactical(self):
        price = self.data.close[0]
        target_value = self.p.rotation_pct * self.broker.getvalue()
        size = target_value / price
        size = min(size, self.position.size)
        if size <= 0:
            return
        self.order = self.sell(size=size)
        self.tactical_size = size
        self.anchor_high = self.data.high[0]
        self.sell_price = price
        self.in_tactical = True

    def _buyback(self):
        self.order = self.buy(size=self.tactical_size)
        self.in_tactical = False
        self.anchor_high = None
        self.sell_price = None
        self.tactical_size = 0.0
        self.cooldown_until = len(self) + self.p.cooldown_days
```
