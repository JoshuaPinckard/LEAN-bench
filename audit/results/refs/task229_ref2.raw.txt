import backtrader as bt


class MeanReversionStrategy(bt.Strategy):
    params = (
        ("bb_period", 200),
        ("bb_dev2", 2.0),
        ("bb_dev3", 3.0),
        ("rsi_period", 14),
        ("rsi_long", 35),
        ("rsi_short", 65),
        ("atr_period", 14),
        ("target_kroner", 100000.0),
        ("risk_pct", 0.02),
    )

    def __init__(self):
        self.bb2 = bt.indicators.BollingerBands(
            self.data.close,
            period=self.p.bb_period,
            devfactor=self.p.bb_dev2,
        )
        self.bb3 = bt.indicators.BollingerBands(
            self.data.close,
            period=self.p.bb_period,
            devfactor=self.p.bb_dev3,
        )
        self.rsi = bt.indicators.RSI(self.data.close, period=self.p.rsi_period)
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)

        self.order = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return
        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            self.order = None

    def next(self):
        if self.order:
            return

        close = self.data.close[0]
        capital = self.broker.getvalue()

        if not self.position:
            # Long entry condition
            if close < self.bb2.bot[0] and self.rsi[0] < self.p.rsi_long:
                dist = max(close - self.bb3.bot[0], 1e-5)
                denom = max(dist, self.atr[0])
                target_size = self.p.target_kroner / close
                risk_size = (self.p.risk_pct * capital) / denom
                size = int(min(target_size, risk_size))
                if size > 0:
                    self.order = self.buy(size=size)

            # Short entry condition
            elif close > self.bb2.top[0] and self.rsi[0] > self.p.rsi_short:
                dist = max(self.bb3.top[0] - close, 1e-5)
                denom = max(dist, self.atr[0])
                target_size = self.p.target_kroner / close
                risk_size = (self.p.risk_pct * capital) / denom
                size = int(min(target_size, risk_size))
                if size > 0:
                    self.order = self.sell(size=size)

        else:
            if self.position.size > 0:
                # Long exit: crossing middle line (reversion) or touching lower 3σ band (anti-meltdown stop)
                if close >= self.bb2.mid[0] or self.data.low[0] <= self.bb3.bot[0]:
                    self.order = self.close()
            elif self.position.size < 0:
                # Short exit: crossing middle line (reversion) or touching upper 3σ band (anti-meltdown stop)
                if close <= self.bb2.mid[0] or self.data.high[0] >= self.bb3.top[0]:
                    self.order = self.close()
