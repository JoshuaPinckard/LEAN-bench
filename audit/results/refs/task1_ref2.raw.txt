import backtrader as bt


class InverseHeadAndShoulders(bt.Strategy):
    """
    Inverse Head and Shoulders strategy with 4h volume confirmation and daily 20-day SMA trend filter.
    Aggregates 1h bars into daily and 4-hour bars to evaluate pattern, volume, and breakout conditions.
    """

    def __init__(self):
        self.order = None
        self.sl = None
        self.tp = None
        self.current_neckline = None

        # Data aggregation attributes
        self.daily_bars = []
        self.current_daily = None
        self.last_date = None

        self.completed_4h_bars = []
        self.current_4h_bars = []
        self.last_red_4h_volume = None

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def find_ihs_pattern(self, daily_bars):
        """
        Detects an Inverse Head and Shoulders pattern on daily bars.
        Looks for Left Shoulder, Head, and Right Shoulder pivot lows.
        Returns: (pattern_found, neckline_price, head_low)
        """
        if len(daily_bars) < 20:
            return False, 0.0, 0.0

        bars = daily_bars[-60:]
        lows = [b['low'] for b in bars]
        highs = [b['high'] for b in bars]
        n = len(lows)

        pivots = []
        for i in range(1, n - 1):
            if lows[i] < lows[i - 1] and lows[i] < lows[i + 1]:
                pivots.append(i)

        if len(pivots) < 3:
            return False, 0.0, 0.0

        for r_idx in range(len(pivots) - 1, 1, -1):
            i_right = pivots[r_idx]
            for h_idx in range(r_idx - 1, 0, -1):
                i_head = pivots[h_idx]
                for l_idx in range(h_idx - 1, -1, -1):
                    i_left = pivots[l_idx]

                    if lows[i_head] < lows[i_left] and lows[i_head] < lows[i_right]:
                        if abs(lows[i_left] - lows[i_right]) / lows[i_left] <= 0.05:
                            sub_h1 = highs[i_left : i_head + 1]
                            p1_rel = sub_h1.index(max(sub_h1))
                            p1_idx = i_left + p1_rel
                            p1_high = highs[p1_idx]

                            sub_h2 = highs[i_head : i_right + 1]
                            p2_rel = sub_h2.index(max(sub_h2))
                            p2_idx = i_head + p2_rel
                            p2_high = highs[p2_idx]

                            if p2_idx != p1_idx:
                                slope = (p2_high - p1_high) / (p2_idx - p1_idx)
                                neckline = p1_high + slope * ((n - 1) - p1_idx)
                            else:
                                neckline = (p1_high + p2_high) / 2.0

                            head_low = lows[i_head]
                            if neckline > head_low:
                                return True, neckline, head_low
        return False, 0.0, 0.0

    def next(self):
        dt = self.data.datetime.date(0)
        o = self.data.open[0]
        h = self.data.high[0]
        l = self.data.low[0]
        c = self.data.close[0]
        v = self.data.volume[0]

        # 1. Update Daily Bar Aggregation
        if dt != self.last_date:
            if self.current_daily is not None:
                self.daily_bars.append(self.current_daily)
            self.current_daily = {'open': o, 'high': h, 'low': l, 'close': c, 'volume': v}
            self.last_date = dt
        else:
            self.current_daily['high'] = max(self.current_daily['high'], h)
            self.current_daily['low'] = min(self.current_daily['low'], l)
            self.current_daily['close'] = c
            self.current_daily['volume'] += v

        # 2. Manage Intraday Stop Loss / Take Profit
        if self.position:
            if self.sl is not None and l <= self.sl:
                self.close()
                return
            elif self.tp is not None and h >= self.tp:
                self.close()
                return

        # 3. Update 4h Bar Aggregation
        self.current_4h_bars.append({'open': o, 'high': h, 'low': l, 'close': c, 'volume': v})

        if len(self.current_4h_bars) == 4:
            b_open = self.current_4h_bars[0]['open']
            b_high = max(b['high'] for b in self.current_4h_bars)
            b_low = min(b['low'] for b in self.current_4h_bars)
            b_close = self.current_4h_bars[-1]['close']
            b_vol = sum(b['volume'] for b in self.current_4h_bars)

            bar_4h = {'open': b_open, 'high': b_high, 'low': b_low, 'close': b_close, 'volume': b_vol}
            self.completed_4h_bars.append(bar_4h)
            self.current_4h_bars = []

            # Alternative Exit: Close if 4h candle closes below neckline
            if self.position and self.current_neckline is not None:
                if bar_4h['close'] < self.current_neckline:
                    self.close()
                    return

            # Track last bearish (red) 4h candle volume
            if bar_4h['close'] < bar_4h['open']:
                self.last_red_4h_volume = bar_4h['volume']

            # Entry Evaluation on 4h candle completion
            if not self.position and self.order is None:
                # 20-day SMA trend filter on daily chart
                if len(self.daily_bars) >= 21:
                    daily_closes = [b['close'] for b in self.daily_bars]
                    sma20_curr = sum(daily_closes[-20:]) / 20.0
                    sma20_prev = sum(daily_closes[-21:-1]) / 20.0
                    trend_ok = sma20_curr >= sma20_prev
                else:
                    trend_ok = True

                if trend_ok:
                    pattern_found, neckline, head_low = self.find_ihs_pattern(self.daily_bars)
                    if pattern_found:
                        if bar_4h['close'] > neckline:
                            vol_ok = (
                                self.last_red_4h_volume is not None
                                and bar_4h['volume'] >= 1.5 * self.last_red_4h_volume
                            )
                            if vol_ok:
                                breakout_price = bar_4h['close']
                                stop_loss = head_low * 0.99
                                distance = neckline - head_low
                                take_profit = breakout_price + distance

                                risk_per_share = breakout_price - stop_loss
                                if risk_per_share > 0:
                                    capital = self.broker.getvalue()
                                    risk_amount = capital * 0.01
                                    size = int(risk_amount / risk_per_share)
                                    if size > 0:
                                        self.sl = stop_loss
                                        self.tp = take_profit
                                        self.current_neckline = neckline
                                        self.order = self.buy(size=size)
