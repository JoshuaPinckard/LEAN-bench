import backtrader as bt
from collections import deque


class VWAPBreakoutStrategy(bt.Strategy):
    params = dict(
        n=3,
        stop_pct=0.005,
    )

    def __init__(self):
        self.order = None
        self.entry_price = None
        self.cum_pv = 0.0
        self.cum_vol = 0.0
        self.current_date = None
        self.hist = deque(maxlen=self.p.n)

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return
        if order.status == order.Completed:
            if self.entry_price is None:
                self.entry_price = order.executed.price
            else:
                self.entry_price = None
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        bar_date = self.data.datetime.datetime(0).date()
        if self.current_date != bar_date:
            self.current_date = bar_date
            self.cum_pv = 0.0
            self.cum_vol = 0.0
            self.hist.clear()

        typical_price = (self.data.high[0] + self.data.low[0] + self.data.close[0]) / 3.0
        vol = self.data.volume[0]
        self.cum_pv += typical_price * vol
        self.cum_vol += vol
        vwap = self.cum_pv / self.cum_vol if self.cum_vol else typical_price

        close = self.data.close[0]

        if self.position:
            if not self.order:
                if self.position.size > 0:
                    stop_price = self.entry_price * (1.0 - self.p.stop_pct)
                    if self.data.low[0] <= stop_price:
                        self.order = self.close()
                    elif close < vwap:
                        self.order = self.close()
                elif self.position.size < 0:
                    stop_price = self.entry_price * (1.0 + self.p.stop_pct)
                    if self.data.high[0] >= stop_price:
                        self.order = self.close()
                    elif close > vwap:
                        self.order = self.close()
        elif not self.order:
            if len(self.hist) == self.p.n:
                if close > vwap and all(c < v for c, v in self.hist):
                    self.order = self.buy()
                elif close < vwap and all(c > v for c, v in self.hist):
                    self.order = self.sell()

        self.hist.append((close, vwap))
