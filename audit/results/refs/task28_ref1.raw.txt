import backtrader as bt
import math


class ShortTermEnergyETFStrategy(bt.Strategy):
    params = (
        ("sma_period", 50),
        ("rsi_period", 14),
    )

    def __init__(self):
        self.sma = bt.indicators.SimpleMovingAverage(
            self.data.close, period=self.p.sma_period
        )
        self.rsi = bt.indicators.RSI(self.data.close, period=self.p.rsi_period)
        self.order = None
        self.entry_price = None
        self.entry_date = None

    def next(self):
        if self.order:
            return

        current_close = self.data.close[0]

        if self.position:
            gain = (current_close - self.entry_price) / self.entry_price
            days_open = (self.data.datetime.date(0) - self.entry_date).days

            if gain >= 0.10 or gain <= -0.05 or days_open > 90:
                self.order = self.close(exectype=bt.Order.Close)
            return

        daily_increase = (current_close / self.data.close[-1]) - 1.0
        upward_trend = self.sma[0] > self.data.close[-50]
        valid_rsi = 40 <= self.rsi[0] <= 65

        if daily_increase >= 0.02 and upward_trend and valid_rsi:
            size = math.floor((self.broker.getvalue() * 0.05) / current_close)
            if size > 0:
                self.order = self.buy(size=size, exectype=bt.Order.Close)

    def notify_order(self, order):
        if order.status == order.Completed:
            if order.isbuy():
                self.entry_price = order.executed.price
                self.entry_date = self.data.datetime.date(0)
            else:
                self.entry_price = None
                self.entry_date = None
            self.order = None
        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            self.order = None
