import backtrader as bt


class MeanReversionTrend(bt.Strategy):
    params = (
        ("ema_period", 21),
        ("bb_period", 50),
        ("bb_dev", 1.5),
        ("pct_capital", 0.02),
        ("ema_threshold", 0.02),
    )

    def __init__(self):
        self.order = None

        self.ema = bt.indicators.EMA(self.data.close, period=self.p.ema_period)
        self.boll = bt.indicators.BollingerBands(
            self.data.close, period=self.p.bb_period, devfactor=self.p.bb_dev
        )

        self.cross_bot = bt.indicators.CrossOver(
            self.data.close, self.boll.lines.bot
        )
        self.cross_top = bt.indicators.CrossOver(
            self.data.close, self.boll.lines.top
        )
        self.cross_ema = bt.indicators.CrossOver(self.data.close, self.ema)

    def notify_order(self, order):
        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            self.order = None

    def next(self):
        if self.order:
            return

        close = self.data.close[0]
        high = self.data.high[0]
        low = self.data.low[0]
        ema = self.ema[0]
        bot = self.boll.lines.bot[0]
        top = self.boll.lines.top[0]

        if not self.position:
            dist = abs(close - ema) / ema

            # Long Entry
            if (
                self.cross_bot[0] > 0
                and close > ema
                and dist <= self.p.ema_threshold
            ):
                capital = self.broker.getvalue()
                size = int((capital * self.p.pct_capital) / close)
                if size > 0:
                    self.order = self.buy(size=size)

            # Short Entry
            elif (
                self.cross_top[0] < 0
                and close < ema
                and dist <= self.p.ema_threshold
            ):
                capital = self.broker.getvalue()
                size = int((capital * self.p.pct_capital) / close)
                if size > 0:
                    self.order = self.sell(size=size)

        else:
            if self.position.size > 0:
                # Long Exit:
                # - Touching upper BB band (High >= top) / Take-Profit
                # - Stop-loss at lower BB band (Low <= bot)
                # - Price crosses EMA(21) downward
                if high >= top or low <= bot or self.cross_ema[0] < 0:
                    self.order = self.close()

            elif self.position.size < 0:
                # Short Exit:
                # - Touching lower BB band (Low <= bot) / Take-Profit
                # - Stop-loss at upper BB band (High >= top)
                # - Price crosses EMA(21) upward
                if low <= bot or high >= top or self.cross_ema[0] > 0:
                    self.order = self.close()
