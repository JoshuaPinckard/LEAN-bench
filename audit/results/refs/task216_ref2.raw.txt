import backtrader as bt


class PeriodicRebalanceStrategy(bt.Strategy):
    params = (
        ("rebalance_period", "quarterly"),  # 'quarterly' or 'yearly'
        ("target_weights", None),  # dict, callable, scalar float, or None
        ("default_weight", 1.0),  # default target weight w_t if schedule silent
    )

    def __init__(self):
        self.order = None
        # Set slippage to 0.05% of price as specified
        self.broker.set_slippage_perc(0.0005)

    def notify_order(self, order):
        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            self.order = None

    def is_rebalance_date(self):
        if len(self) == 1:
            return True

        d0 = self.data.datetime.date(0)
        d1 = self.data.datetime.date(-1)

        period = str(self.p.rebalance_period).lower()
        if period == "yearly":
            return d0.year != d1.year
        elif period == "quarterly":
            q0 = (d0.year, (d0.month - 1) // 3)
            q1 = (d1.year, (d1.month - 1) // 3)
            return q0 != q1
        return False

    def get_target_weight(self):
        weights = self.p.target_weights
        if weights is None:
            return float(self.p.default_weight)

        if isinstance(weights, (int, float)):
            return float(weights)

        current_date = self.data.datetime.date(0)
        if callable(weights):
            return float(weights(current_date))

        if isinstance(weights, dict):
            if current_date in weights:
                return float(weights[current_date])
            date_str = current_date.strftime("%Y-%m-%d")
            if date_str in weights:
                return float(weights[date_str])

        return float(self.p.default_weight)

    def next(self):
        if self.order:
            return

        if not self.is_rebalance_date():
            return

        w_t = self.get_target_weight()

        if w_t == 0:
            if self.position.size != 0:
                self.order = self.close()
        else:
            portfolio_value = self.broker.getvalue()
            price = self.data.close[0]
            target_value = portfolio_value * w_t
            target_shares = int(target_value / price)

            delta = target_shares - self.position.size
            if delta > 0:
                self.order = self.buy(size=delta)
            elif delta < 0:
                self.order = self.sell(size=abs(delta))
