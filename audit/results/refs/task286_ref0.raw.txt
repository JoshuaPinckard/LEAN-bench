import backtrader as bt


class RollingCoveredCallsStrategy(bt.Strategy):
    def __init__(self):
        self.bb = bt.indicators.BollingerBands(self.data, period=20, devfactor=2)
        self.rsi = bt.indicators.RSI(self.data, period=14)
        self.atr = bt.indicators.ATR(self.data, period=14)
        self.vol_sma = bt.indicators.SMA(self.data.volume, period=20)
        self.cross_up = bt.indicators.CrossOver(self.data.close, self.bb.top)

        self.order = None
        self.core_bought = False
        self.core_qty = 0
        self.assigned_flat = False
        self.tranches = []  # each: {'qty', 'strike', 'expire_bar'}
        self.last_tranche_bar = None

    def notify_order(self, order):
        if order.status in (order.Completed, order.Canceled, order.Margin, order.Rejected):
            self.order = None

    def _core_size(self):
        cash = self.broker.getcash()
        price = self.data.close[0]
        return int(cash / price // 100) * 100

    def _active_sold_qty(self):
        return sum(t['qty'] for t in self.tranches)

    def next(self):
        if self.order:
            return

        bar = len(self)
        close = self.data.close[0]
        upper = self.bb.top[0]
        mid = self.bb.mid[0]
        rsi = self.rsi[0]
        atr = self.atr[0]
        vol = self.data.volume[0]
        vol_sma = self.vol_sma[0]

        if not self.core_bought:
            size = self._core_size()
            if size > 0:
                self.order = self.buy(size=size)
                self.core_qty = size
            self.core_bought = True
            return

        if self.assigned_flat:
            if close < mid and rsi < 50:
                size = self._core_size()
                if size > 0:
                    self.order = self.buy(size=size)
                    self.core_qty = size
                    self.tranches = []
                    self.last_tranche_bar = None
                    self.assigned_flat = False
            return

        if self.tranches and close > (upper + atr):
            self.order = self.close()
            self.tranches = []
            self.assigned_flat = True
            return

        if self.tranches and (close < mid or rsi < 45):
            qty = self._active_sold_qty()
            if qty > 0:
                self.order = self.buy(size=qty)
            self.tranches = []
            return

        order_placed = False
        for t in list(self.tranches):
            if bar >= t['expire_bar']:
                if close > upper:
                    t['strike'] = upper
                    t['expire_bar'] = bar + 7
                    if (rsi > 60 and self.last_tranche_bar != bar
                            and self.position.size > 0):
                        remaining_cap = self.core_qty - self._active_sold_qty()
                        if remaining_cap > 0:
                            add_qty = max(int(round(0.20 * self.position.size)), 100)
                            add_qty = min(add_qty, self.position.size, remaining_cap)
                            if add_qty > 0:
                                self.order = self.sell(size=add_qty)
                                self.tranches.append({
                                    'qty': add_qty,
                                    'strike': upper,
                                    'expire_bar': bar + 7,
                                })
                                self.last_tranche_bar = bar
                                order_placed = True
                else:
                    self.order = self.buy(size=t['qty'])
                    self.tranches.remove(t)
                    order_placed = True
                if order_placed:
                    break
        if order_placed:
            return

        if (self.cross_up[0] > 0 and rsi > 60 and vol > vol_sma
                and self.last_tranche_bar != bar
                and self.position.size > 0):
            remaining_cap = self.core_qty - self._active_sold_qty()
            if remaining_cap > 0:
                tranche_qty = max(int(round(0.20 * self.position.size)), 100)
                tranche_qty = min(tranche_qty, self.position.size, remaining_cap)
                if tranche_qty > 0:
                    self.order = self.sell(size=tranche_qty)
                    self.tranches.append({
                        'qty': tranche_qty,
                        'strike': upper,
                        'expire_bar': bar + 7,
                    })
                    self.last_tranche_bar = bar
