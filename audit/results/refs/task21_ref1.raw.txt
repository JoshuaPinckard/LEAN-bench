import backtrader as bt


class EMASochasticScalping(bt.Strategy):
    params = (
        ('ema_period', 20),
        ('stoch_period', 14),
        ('stoch_period_dfast', 3),
        ('stoch_period_dslow', 3),
        ('take_profit', 0.005),
        ('stop_loss', 0.003),
        ('capital_fraction', 0.01),
    )

    def __init__(self):
        self.order = None
        self.ema = bt.indicators.EMA(self.data.close, period=self.p.ema_period)
        self.stochastic = bt.indicators.Stochastic(
            self.data,
            period=self.p.stoch_period,
            period_dfast=self.p.stoch_period_dfast,
            period_dslow=self.p.stoch_period_dslow,
        )

    def notify_order(self, order):
        if order.status in (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            self.order = None

    def next(self):
        if self.order:
            return

        k = self.stochastic.percK[0]
        d = self.stochastic.percD[0]
        prev_k = self.stochastic.percK[-1]
        prev_d = self.stochastic.percD[-1]

        cross_up = prev_k < prev_d and k > d
        cross_down = prev_k > prev_d and k < d

        if self.position:
            entry_price = self.position.price

            if self.position.size > 0:
                if (
                    cross_down
                    or self.data.close[0] >= entry_price * (1 + self.p.take_profit)
                    or self.data.close[0] <= entry_price * (1 - self.p.stop_loss)
                ):
                    self.order = self.close()
            else:
                if (
                    cross_up
                    or self.data.close[0] <= entry_price * (1 - self.p.take_profit)
                    or self.data.close[0] >= entry_price * (1 + self.p.stop_loss)
                ):
                    self.order = self.close()
            return

        size = (self.broker.getvalue() * self.p.capital_fraction) / self.data.close[0]

        if (
            self.data.close[0] > self.ema[0]
            and cross_up
            and k < 20
            and d < 20
        ):
            self.order = self.buy(size=size)
        elif (
            self.data.close[0] < self.ema[0]
            and cross_down
            and k > 80
            and d > 80
        ):
            self.order = self.sell(size=size)
