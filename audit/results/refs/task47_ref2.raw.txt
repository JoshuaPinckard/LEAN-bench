import collections
import math
import statistics
import backtrader as bt


class FVGSwingFailureBreakout(bt.Strategy):
    params = (
        ("risk_pct", 0.01),  # 1% risk per trade
        ("rr_ratio", 3.33),  # 1:3.33 risk-to-reward ratio
        ("max_hold_bars", 48),  # 4 hours = 48 5m bars
    )

    def __init__(self):
        self.order = None

        # 4H Aggregation state
        self.current_4h_bucket = None
        self.current_4h_bar = None
        self.bars_4h = []

        # 4H Setup states
        self.fvg_active = False
        self.fvg_zone = None
        self.sfp_active = False
        self.mss_active = False
        self.setup_confirmed = False
        self.setup_bar_count = 0

        # Swing pivots on 4H
        self.swing_highs_4h = []
        self.swing_lows_4h = []

        # Trade management attributes
        self.entry_price = 0.0
        self.sl_price = 0.0
        self.tp1_price = 0.0
        self.tp2_price = 0.0
        self.risk_dist = 0.0
        self.tp1_hit = False
        self.bars_in_trade = 0
        self.initial_size = 0.0

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return
        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            self.order = None

    def _update_4h_pivots_and_signals(self):
        if len(self.bars_4h) < 3:
            return

        k = len(self.bars_4h) - 1
        prev = self.bars_4h[k - 1]
        p_prev = self.bars_4h[k - 2]
        curr = self.bars_4h[k]

        # Identify 3-bar Pivot Low on 4H
        if prev["low"] < p_prev["low"] and prev["low"] < curr["low"]:
            self.swing_lows_4h.append(prev["low"])
            if len(self.swing_lows_4h) > 10:
                self.swing_lows_4h.pop(0)

        # Identify 3-bar Pivot High on 4H
        if prev["high"] > p_prev["high"] and prev["high"] > curr["high"]:
            self.swing_highs_4h.append(prev["high"])
            if len(self.swing_highs_4h) > 10:
                self.swing_highs_4h.pop(0)

        # FVG Detection (3-candle gap on 4H: k-2 High < k Low)
        if p_prev["high"] < curr["low"]:
            self.fvg_zone = (p_prev["high"], curr["low"])

        # FVG Entry check
        if self.fvg_zone is not None:
            fvg_bot, fvg_top = self.fvg_zone
            if curr["low"] <= fvg_top and curr["high"] >= fvg_bot:
                self.fvg_active = True

        # SFP check: price drops below recent 4H swing low but closes above it
        if self.fvg_active and self.swing_lows_4h:
            recent_low = self.swing_lows_4h[-1]
            if curr["low"] < recent_low and curr["close"] > recent_low:
                self.sfp_active = True

        # MSS check: price closes above recent 4H swing high
        if self.sfp_active and self.swing_highs_4h:
            recent_high = self.swing_highs_4h[-1]
            if curr["close"] > recent_high:
                self.mss_active = True

        # Confirm 4H setup
        if self.fvg_active and self.sfp_active and self.mss_active:
            self.setup_confirmed = True
            self.setup_bar_count = 0

    def next(self):
        dt = self.data.datetime.datetime(0)

        # 4H bar aggregation from 5m data
        bucket = (dt.date(), dt.hour // 4)
        if self.current_4h_bucket is None:
            self.current_4h_bucket = bucket
            self.current_4h_bar = {
                "open": self.data.open[0],
                "high": self.data.high[0],
                "low": self.data.low[0],
                "close": self.data.close[0],
                "volume": self.data.volume[0],
                "datetime": dt,
            }
        elif bucket != self.current_4h_bucket:
            # Finalize previous 4H bar
            self.bars_4h.append(self.current_4h_bar)
            self._update_4h_pivots_and_signals()

            # Start new 4H bar
            self.current_4h_bucket = bucket
            self.current_4h_bar = {
                "open": self.data.open[0],
                "high": self.data.high[0],
                "low": self.data.low[0],
                "close": self.data.close[0],
                "volume": self.data.volume[0],
                "datetime": dt,
            }
        else:
            # Update current 4H bar
            self.current_4h_bar["high"] = max(
                self.current_4h_bar["high"], self.data.high[0]
            )
            self.current_4h_bar["low"] = min(
                self.current_4h_bar["low"], self.data.low[0]
            )
            self.current_4h_bar["close"] = self.data.close[0]
            self.current_4h_bar["volume"] += self.data.volume[0]

        # Manage existing open position
        if self.position.size > 0:
            self.bars_in_trade += 1

            # Stop Loss
            if self.data.low[0] <= self.sl_price:
                self.close()
                self.order = None
                return

            # Partial Take Profit 1 (50% position size)
            if not self.tp1_hit and self.data.high[0] >= self.tp1_price:
                exit_size = self.position.size / 2.0
                if exit_size > 0:
                    self.sell(size=exit_size)
                self.tp1_hit = True

            # Full Take Profit 2 (100% remaining position)
            if self.data.high[0] >= self.tp2_price:
                self.close()
                self.order = None
                return

            # Maximum holding time exit (4 hours = 48 5m bars)
            if self.bars_in_trade >= self.p.max_hold_bars:
                self.close()
                self.order = None
                return

            return

        # Do not enter if a pending order is active or on weekends
        if self.order is not None or dt.weekday() >= 5:
            return

        # Setup timeout after 24 hours (288 5m bars)
        if self.setup_confirmed:
            self.setup_bar_count += 1
            if self.setup_bar_count > 288:
                self.setup_confirmed = False
                self.fvg_active = False
                self.sfp_active = False
                self.mss_active = False

        # Entry Logic: 5m Order Block Breakout
        if self.setup_confirmed:
            if len(self.data) >= 2:
                prev_open = self.data.open[-1]
                prev_close = self.data.close[-1]
                prev_low = self.data.low[-1]

                # Identify 5m Bearish Order Block before upward move
                if prev_close < prev_open:
                    ob_high = prev_open
                    ob_low = prev_low

                    # Upward breakout on current 5m bar close
                    if self.data.close[0] > ob_high:
                        entry_p = self.data.close[0]
                        sl_p = ob_low
                        risk_d = entry_p - sl_p

                        if risk_d <= 0 or risk_d < entry_p * 0.001:
                            risk_d = entry_p * 0.005
                            sl_p = entry_p - risk_d

                        capital = self.broker.getvalue()
                        risk_amt = capital * self.p.risk_pct
                        size = risk_amt / risk_d

                        max_size = self.broker.getcash() / entry_p
                        size = min(size, max_size)

                        if size > 0:
                            self.entry_price = entry_p
                            self.sl_price = sl_p
                            self.risk_dist = risk_d
                            self.tp1_price = entry_p + (
                                0.5 * self.p.rr_ratio * risk_d
                            )
                            self.tp2_price = entry_p + (
                                self.p.rr_ratio * risk_d
                            )
                            self.tp1_hit = False
                            self.bars_in_trade = 0
                            self.initial_size = size

                            self.order = self.buy(size=size)

                            # Reset setup flags
                            self.setup_confirmed = False
                            self.fvg_active = False
                            self.sfp_active = False
                            self.mss_active = False
