import backtrader as bt


class SwingTrendStrategy(bt.Strategy):
    def __init__(self):
        self.sma = bt.ind.SMA(self.data.close, period=20)
        self.swing_high = bt.ind.Highest(self.data.high, period=5)(-1)
        self.swing_low = bt.ind.Lowest(self.data.low, period=5)(-1)

        self.crossover_high = bt.ind.CrossOver(self.data.close, self.swing_high)
        self.crossover_low = bt.ind.CrossOver(self.data.close, self.swing_low)

        self.order = None

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        if self.order:
            return

        if not self.position:
            if self.crossover_high[0] == 1 and self.data.close[0] > self.sma[0]:
                self.order = self.buy()
            elif self.crossover_low[0] == -1 and self.data.close[0] < self.sma[0]:
                self.order = self.sell()
        else:
            if self.position.size > 0:
                if self.crossover_low[0] == -1:
                    self.order = self.close()
            elif self.position.size < 0:
                if self.crossover_high[0] == 1:
                    self.order = self.close()
