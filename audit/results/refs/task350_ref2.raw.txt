import backtrader as bt


class MACloudWilliamsRStrategy(bt.Strategy):
    params = (
        ('stop_pct', 0.005),
    )

    def __init__(self):
        self.will = bt.indicators.WilliamsR(self.data, period=14)
        self.sma9 = bt.indicators.SMA(self.data.close, period=9)
        self.sma26 = bt.indicators.SMA(self.data.close, period=26)
        self.sma52 = bt.indicators.SMA(self.data.close, period=52)

        self.order = None
        self.stop_price = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        if self.order:
            return

        cloud_top = max(self.sma9[0], self.sma26[0], self.sma52[0])
        cloud_bot = min(self.sma9[0], self.sma26[0], self.sma52[0])

        if not self.position:
            self.stop_price = None

            # Long entry: Williams %R crosses below -80 AND close > cloud_top AND SMA9 crosses above SMA26
            will_cross_below_80 = self.will[-1] >= -80.0 and self.will[0] < -80.0
            sma9_cross_above_26 = self.sma9[-1] <= self.sma26[-1] and self.sma9[0] > self.sma26[0]
            close_above_cloud_top = self.data.close[0] > cloud_top

            if will_cross_below_80 and close_above_cloud_top and sma9_cross_above_26:
                self.order = self.buy()
                return

            # Short entry: Williams %R crosses above -20 AND close < cloud_bot AND SMA9 is below SMA26
            will_cross_above_20 = self.will[-1] <= -20.0 and self.will[0] > -20.0
            sma9_below_26 = self.sma9[0] < self.sma26[0]
            close_below_cloud_bot = self.data.close[0] < cloud_bot

            if will_cross_above_20 and close_below_cloud_bot and sma9_below_26:
                self.order = self.sell()
                return

            if self.stop_price is None:
                self.stop_price = candidate_stop
            else:
                self.stop_price = max(self.stop_price, candidate_stop)

            if self.data.close[0] < cloud_bot or self.data.low[0] <= self.stop_price:
                self.order = self.close()

        elif self.position.size < 0:
            candidate_stop = self.sma26[0] * (1.0 + self.p.stop_pct)
            if self.stop_price is None:
                self.stop_price = candidate_stop
            else:
                self.stop_price = min(self.stop_price, candidate_stop)

            if self.data.close[0] > cloud_top or self.data.high[0] >= self.stop_price:
