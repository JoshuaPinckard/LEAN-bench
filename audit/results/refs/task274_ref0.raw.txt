import backtrader as bt


class RegimeTradingStrategy(bt.Strategy):
    params = dict(
        bb_period=20,
        bb_devfactor=2,
        ema_fast_period=50,
        ema_slow_period=200,
        width_sma_period=200,
    )

    def __init__(self):
        self.bb = bt.indicators.BollingerBands(
            self.data.close, period=self.p.bb_period, devfactor=self.p.bb_devfactor
        )
        self.ema_fast = bt.indicators.EMA(self.data.close, period=self.p.ema_fast_period)
        self.ema_slow = bt.indicators.EMA(self.data.close, period=self.p.ema_slow_period)

        self.width = self.bb.lines.top - self.bb.lines.bot
        self.width_sma = bt.indicators.SMA(self.width, period=self.p.width_sma_period)

        self.cross_top = bt.indicators.CrossOver(self.data.close, self.bb.lines.top)
        self.cross_bot = bt.indicators.CrossOver(self.data.close, self.bb.lines.bot)
        self.cross_mid = bt.indicators.CrossOver(self.data.close, self.bb.lines.mid)

        self.order = None
        self.position_type = None

    def notify_order(self, order):
        if order.status in (order.Completed, order.Canceled, order.Margin, order.Rejected):
            self.order = None

    def notify_trade(self, trade):
        if trade.isclosed:
            self.position_type = None

    def next(self):
        if self.order:
            return

        trend_regime = self.width[0] > self.width_sma[0]
        mean_reversion_regime = not trend_regime

        if not self.position:
            size = int(self.broker.getcash() / self.data.close[0])
            if size <= 0:
                return

            if mean_reversion_regime:
                if self.data.close[0] <= self.bb.lines.bot[0]:
                    self.order = self.buy(size=size)
                    self.position_type = 'mr_long'
                elif self.data.close[0] >= self.bb.lines.top[0]:
                    self.order = self.sell(size=size)
                    self.position_type = 'mr_short'
            elif trend_regime:
                if self.ema_fast[0] > self.ema_slow[0] and self.cross_top[0] > 0:
                    self.order = self.buy(size=size)
                    self.position_type = 'trend_long'
                elif self.ema_fast[0] < self.ema_slow[0] and self.cross_bot[0] < 0:
                    self.order = self.sell(size=size)
                    self.position_type = 'trend_short'
        else:
            if self.position_type in ('mr_long', 'mr_short'):
                if self.cross_mid[0] != 0:
                    self.order = self.close()
            elif self.position_type == 'trend_long':
                if self.cross_mid[0] < 0:
                    self.order = self.close()
            elif self.position_type == 'trend_short':
                if self.cross_mid[0] > 0:
                    self.order = self.close()
