```python
import datetime
import backtrader as bt


class ORBFibonacciStrategy(bt.Strategy):
    params = (
        ("risk_pct", 0.01),
        ("rr_ratio", 1.5),
        ("sl_offset_pct", 0.001),
    )

    def __init__(self):
        self.order = None
        self.last_date = None
        self.orb_high = None
        self.orb_low = None
        self.orb_range = None
        self.state = "IDLE"
        self.touched_zone = False
        self.sl_price = None
        self.tp_price = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return
        self.order = None

    def next(self):
        dt = self.data.datetime.datetime(0)
        curr_date = dt.date()
        curr_time = dt.time()

        # Reset daily state on first bar of the trading day (09:30 bar)
        if self.last_date != curr_date:
            self.last_date = curr_date
            self.orb_high = self.data.high[0]
            self.orb_low = self.data.low[0]
            self.orb_range = max(self.orb_high - self.orb_low, 1e-5)
            self.state = "WAIT_BREAKOUT"
            self.touched_zone = False
            self.sl_price = None
            self.tp_price = None
            return

        # End-of-day exit for day trading rule (15:55 bar)
        if curr_time >= datetime.time(15, 55):
            if self.position:
                self.close()
            return

        # Position Risk Management (Intraday SL / TP evaluation)
        if self.position:
            if self.position.size > 0:
                if self.data.low[0] <= self.sl_price:
                    self.close()
                elif self.data.high[0] >= self.tp_price:
                    self.close()
            elif self.position.size < 0:
                if self.data.high[0] >= self.sl_price:
                    self.close()
                elif self.data.low[0] <= self.tp_price:
                    self.close()
            return

        # Pending order guard
        if self.order:
            return

        # Entry Signal Logic
        if self.state == "WAIT_BREAKOUT":
            if self.data.high[0] > self.orb_high:
                self.state = "WAIT_RETRACEMENT_LONG"
                self.touched_zone = False
            elif self.data.low[0] < self.orb_low:
                self.state = "WAIT_RETRACEMENT_SHORT"
                self.touched_zone = False

        if self.state == "WAIT_RETRACEMENT_LONG":
            fib_05 = self.orb_high - 0.50 * self.orb_range
            fib_0618 = self.orb_high - 0.618 * self.orb_range

            if self.data.low[0] <= fib_05 and self.data.high[0] >= fib_0618:
                self.touched_zone = True

            if self.touched_zone and self.data.close[0] > self.data.open[0]:
                entry_price = self.data.close[0]
                sl_price = fib_0618 - (self.p.sl_offset_pct * self.orb_range)
                risk = entry_price - sl_price
                if risk > 0:
                    tp_price = entry_price + (self.p.rr_ratio * risk)
                    cash_risk = self.broker.getvalue() * self.p.risk_pct
                    size = int(cash_risk / risk)
                    if size >= 1:
                        self.sl_price = sl_price
                        self.tp_price = tp_price
                        self.order = self.buy(size=size)
                        self.state = "TRADED"

        elif self.state == "WAIT_RETRACEMENT_SHORT":
            fib_05 = self.orb_low + 0.50 * self.orb_range
            fib_0618 = self.orb_low + 0.618 * self.orb_range

            if self.data.high[0] >= fib_05 and self.data.low[0] <= fib_0618:
                self.touched_zone = True

            if self.touched_zone and self.data.close[0] < self.data.open[0]:
                entry_price = self.data.close[0]
                sl_price = fib_0618 + (self.p.sl_offset_pct * self.orb_range)
                risk = sl_price - entry_price
                if risk > 0:
                    tp_price = entry_price - (self.p.rr_ratio * risk)
                    cash_risk = self.broker.getvalue() * self.p.risk_pct
                    size = int(cash_risk / risk)
                    if size >= 1:
                        self.sl_price = sl_price
                        self.tp_price = tp_price
                        self.order = self.sell(size=size)
                        self.state = "TRADED"
```
