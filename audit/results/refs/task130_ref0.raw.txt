import backtrader as bt


class AntiLossStreakSkipper(bt.Strategy):
    params = (
        ('fast_period', 10),
        ('slow_period', 30),
        ('stop_loss_pct', 0.02),
        ('take_profit_pct', 0.04),
        ('equity_pct', 0.01),
    )

    def __init__(self):
        self.fast_sma = bt.indicators.SMA(self.data.close, period=self.p.fast_period)
        self.slow_sma = bt.indicators.SMA(self.data.close, period=self.p.slow_period)
        self.crossover = bt.indicators.CrossOver(self.fast_sma, self.slow_sma)

        self.skip_mode = False

        self.entry_order = None
        self.stop_order = None
        self.limit_order = None
        self.entry_price = None

        self.hyp_pending = False
        self.hyp_active = False
        self.hyp_entry_price = None
        self.hyp_stop_price = None
        self.hyp_target_price = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if self.entry_order is not None and order.ref == self.entry_order.ref:
            if order.status == order.Completed:
                self.entry_price = order.executed.price
                size = order.executed.size
                stop_price = self.entry_price * (1.0 - self.p.stop_loss_pct)
                target_price = self.entry_price * (1.0 + self.p.take_profit_pct)
                stop_o = self.sell(exectype=bt.Order.Stop, price=stop_price, size=size)
                limit_o = self.sell(exectype=bt.Order.Limit, price=target_price, size=size, oco=stop_o)
                self.stop_order = stop_o
                self.limit_order = limit_o
            self.entry_order = None
            return

        if self.stop_order is not None and order.ref == self.stop_order.ref:
            if order.status == order.Completed:
                self.skip_mode = True
                self.stop_order = None
                self.limit_order = None
                self.entry_price = None
            elif order.status in [order.Canceled, order.Margin, order.Rejected]:
                self.stop_order = None
            return

        if self.limit_order is not None and order.ref == self.limit_order.ref:
            if order.status == order.Completed:
                self.skip_mode = False
                self.stop_order = None
                self.limit_order = None
                self.entry_price = None
            elif order.status in [order.Canceled, order.Margin, order.Rejected]:
                self.limit_order = None
            return

    def next(self):
        entry_signal = self.crossover[0] > 0

        if self.hyp_pending:
            self.hyp_entry_price = self.data.open[0]
            self.hyp_stop_price = self.hyp_entry_price * (1.0 - self.p.stop_loss_pct)
            self.hyp_target_price = self.hyp_entry_price * (1.0 + self.p.take_profit_pct)
            self.hyp_pending = False
            self.hyp_active = True

        if self.hyp_active:
            hit_stop = self.data.low[0] <= self.hyp_stop_price
            hit_target = self.data.high[0] >= self.hyp_target_price
            if hit_stop:
                # same-bar tie-break: stop-loss assumed to trigger first
                self.hyp_active = False
            elif hit_target:
                self.hyp_active = False
                self.skip_mode = False

        if self.entry_order is not None or self.position:
            return

        if entry_signal:
            if not self.skip_mode:
                size = int((self.p.equity_pct * self.broker.getvalue()) / self.data.close[0])
                if size > 0:
                    self.entry_order = self.buy(size=size)
            elif not self.hyp_active and not self.hyp_pending:
                self.hyp_pending = True
