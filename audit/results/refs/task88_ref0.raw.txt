import backtrader as bt


class DjiaSma50TransactionCostStrategy(bt.Strategy):
    params = dict(
        sma_period=50,
        trade_size=100,
        cost_pct=0.002,
    )

    def __init__(self):
        self.sma = bt.indicators.SMA(self.data.close, period=self.p.sma_period)
        self.order = None
        self.total_transaction_costs = 0.0
        self.gross_pnl = 0.0
        self.net_pnl = 0.0

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status == order.Completed:
            # Broker-level commission is fixed by the harness at 0.001;
            # the spec's 0.2% cost is tracked here as a separate internal ledger.
            trade_value = order.executed.price * abs(order.executed.size)
            self.total_transaction_costs += trade_value * self.p.cost_pct
            self.net_pnl = self.gross_pnl - self.total_transaction_costs

        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def notify_trade(self, trade):
        if trade.isclosed:
            self.gross_pnl += trade.pnl
            self.net_pnl = self.gross_pnl - self.total_transaction_costs

    def next(self):
        if self.order:
            return

        if not self.position:
            if self.data.close[0] > self.sma[0]:
                self.order = self.buy(size=self.p.trade_size)
        else:
            if self.data.close[0] < self.sma[0]:
                self.order = self.close()
