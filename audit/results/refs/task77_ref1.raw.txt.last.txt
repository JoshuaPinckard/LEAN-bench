```python
import backtrader as bt
import math
import statistics


class AvellanedaStoikovStrategy(bt.Strategy):
    params = (
        ("gamma", 0.1),
        ("mode", "Inventory"),
        ("volatility_period", 20),
        ("arrival_rate", 1.0),
        ("horizon", 1.0),
        ("max_inventory", 10),
    )

    def __init__(self):
        self.volatility = bt.indicators.StdDev(
            self.data.close, period=self.p.volatility_period
        )
        self.buy_order = None
        self.sell_order = None
        self.optimal_bid = None
        self.optimal_ask = None
        self.profits = []

    def _calculate_quotes(self):
        mid_price = float(self.data.close[0])
        sigma = float(self.volatility[0])

        if not math.isfinite(mid_price) or not math.isfinite(sigma):
            return None, None

        q = float(self.position.size)
        variance = sigma * sigma
        gamma = self.p.gamma
        k = self.p.arrival_rate

        if self.p.mode == "Symmetric":
            reservation_price = mid_price
        else:
            reservation_price = mid_price - q * gamma * variance * self.p.horizon

        half_spread = (
            (gamma * variance * self.p.horizon) / 2.0
            + math.log(1.0 + gamma / k) / gamma
        )

        self.optimal_bid = reservation_price - half_spread
        self.optimal_ask = reservation_price + half_spread
        return self.optimal_bid, self.optimal_ask

    def next(self):
        bid, ask = self._calculate_quotes()
        if bid is None:
            return

        if self.buy_order is None and self.sell_order is None:
            q = self.position.size

            if q < self.p.max_inventory:
                self.buy_order = self.buy(exectype=bt.Order.Limit, price=bid)

            if q > -self.p.max_inventory:
                self.sell_order = self.sell(exectype=bt.Order.Limit, price=ask)

    def notify_order(self, order):
        if order.status in (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            if order is self.buy_order:
                self.buy_order = None
            elif order is self.sell_order:
                self.sell_order = None

        if order.status == order.Completed:
            self._calculate_quotes()

            if order.isbuy() and self.sell_order is not None:
                self.cancel(self.sell_order)
            elif order.issell() and self.buy_order is not None:
                self.cancel(self.buy_order)

    def notify_trade(self, trade):
        if trade.isclosed:
            self.profits.append(trade.pnlcomm)

    def stop(self):
        self.total_profit = sum(self.profits)
        self.total_average_profit = (
            statistics.mean(self.profits) if self.profits else 0.0
        )
        self.standard_deviation_profit = (
            statistics.pstdev(self.profits) if len(self.profits) > 1 else 0.0
        )
        self.average_final_inventory = float(self.position.size)
        self.standard_deviation_final_inventory = 0.0
```
