import math
import backtrader as bt


class SlippageAssessmentStrategy(bt.Strategy):
    def __init__(self):
        self.order = None
        self.cum_volume = 0.0
        self.best_pcv = None

    def notify_order(self, order):
        if order.status in (order.Completed, order.Canceled, order.Margin, order.Rejected):
            self.order = None

    def next(self):
        volume = self.data.volume[0]
        high = self.data.high[0]
        low = self.data.low[0]

        price_change_per_volume = (math.log(high) - math.log(low)) / volume

        self.cum_volume += volume
        filter_passes = volume <= 0.05 * self.cum_volume

        if self.order:
            return

        if self.position:
            self.order = self.close(exectype=bt.Order.Close)
        elif filter_passes:
            if self.best_pcv is None or price_change_per_volume < self.best_pcv:
                self.best_pcv = price_change_per_volume

            max_affordable = int(self.broker.getcash() / self.data.close[0])
            size = int(max_affordable * (self.best_pcv / price_change_per_volume))

            if size > 0:
                self.order = self.buy(size=size, exectype=bt.Order.Close)
