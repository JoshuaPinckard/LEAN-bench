import backtrader as bt


class BreakoutRetestStrategy(bt.Strategy):

    def __init__(self):
        # 10-period EMA: support level and dynamic stop
        self.ema10 = bt.indicators.EMA(self.data.close, period=10)
        # Highest HIGH over the 50 bars strictly BEFORE the current bar
        # Using data.high(-1) delays the series by 1 so Highest(period=50)
        # computes max(high[t-1], ..., high[t-50]) at bar t; minperiod = 51
        self.breakout_level = bt.indicators.Highest(self.data.high(-1), period=50)
        # 50-bar average volume for demand confirmation
        self.avg_vol50 = bt.indicators.SMA(self.data.volume, period=50)

        self.order = None
        self.entry_price = None
        self.in_breakout_watch = False
        self.breakout_bar = None

    def notify_order(self, order):
        if order.status in [order.Completed]:
            if order.isbuy():
                self.entry_price = order.executed.price
            self.order = None
        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        if self.order:
            return

        if not self.position:
            if not self.in_breakout_watch:
                # Condition 1: today's close exceeds the highest HIGH of the prior 50 bars
                if self.data.close[0] > self.breakout_level[0]:
                    self.in_breakout_watch = True
                    self.breakout_bar = len(self)
            else:
                bars_after = len(self) - self.breakout_bar
                if bars_after > 5:
                    # 5-day retest window expired; reset and look for new breakout
                    self.in_breakout_watch = False
                    self.breakout_bar = None
                else:
                    ema_val = self.ema10[0]
                    # Retest: low touches or pierces EMA (pullback reached EMA)
                    # but close is at or above EMA (did not close below it)
                    # Demand: volume above 50-bar average
                    if (self.data.low[0] <= ema_val
                            and self.data.close[0] >= ema_val
                            and self.data.volume[0] > self.avg_vol50[0]):
                        size = int(0.02 * self.broker.getvalue() / self.data.close[0])
                        if size > 0:
                            self.order = self.buy(size=size)
                        self.in_breakout_watch = False
                        self.breakout_bar = None
        else:
            ema_val = self.ema10[0]
            # Stop-loss: price closes below the 10-day EMA
            if self.data.close[0] < ema_val:
                self.order = self.close()
            # Take-profit: 10% gain from entry price
            elif (self.entry_price is not None
                  and self.data.close[0] >= self.entry_price * 1.10):
                self.order = self.close()
