import backtrader as bt


class BreakoutRetest(bt.Strategy):
    """
    Breakout Retest Strategy Reference Implementation.

    Trades breakouts from a 50-day high followed by a retest of the 10-day EMA
    within 5 trading days with above-average volume confirmation.
    """

    params = (
        ("ema_period", 10),
        ("breakout_period", 50),
        ("vol_period", 50),
        ("retest_window", 5),
        ("capital_pct", 0.02),
        ("tp_pct", 0.10),
    )

    def __init__(self):
        self.ema = bt.indicators.EMA(self.data.close, period=self.p.ema_period)
        self.highest50 = bt.indicators.Highest(
            self.data.high, period=self.p.breakout_period
        )
        self.vol_sma = bt.indicators.SMA(
            self.data.volume, period=self.p.vol_period
        )

        self.order = None
        self.entry_price = None
        self.days_since_breakout = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status == order.Completed:
            if order.isbuy():
                self.entry_price = order.executed.price
            elif order.issell():
                self.entry_price = None

        self.order = None

    def next(self):
        # Increment breakout tracker
        if self.days_since_breakout is not None:
            self.days_since_breakout += 1

        # Breakout condition: close exceeds 50-day highest prior high
        if self.data.close[0] > self.highest50[-1]:
            self.days_since_breakout = 0

        if self.order:
            return

        if not self.position:
            # Retest window check (within 1 to 5 days after breakout)
            if (
                self.days_since_breakout is not None
                and 1 <= self.days_since_breakout <= self.p.retest_window
            ):
                # Retest: price pulls back to 10-day EMA but does not close below it
                retest = (self.data.low[0] <= self.ema[0]) and (
                    self.data.close[0] >= self.ema[0]
                )
                # Demand confirmation: volume > 50-day average volume
                demand = self.data.volume[0] > self.vol_sma[0]

                if retest and demand:
                    capital = self.broker.getvalue()
                    size = int(
                        (capital * self.p.capital_pct) / self.data.close[0]
                    )
                    if size > 0:
                        self.order = self.buy(size=size)
                        self.days_since_breakout = None
        else:
            # Exit condition 1: Stop-loss if price closes below 10-day EMA
            stop_loss = self.data.close[0] < self.ema[0]

            # Exit condition 2: Take-profit of 10% from entry price
            take_profit = (self.entry_price is not None) and (
                self.data.high[0] >= self.entry_price * (1.0 + self.p.tp_pct)
            )

            if stop_loss or take_profit:
                self.order = self.close()
