import backtrader as bt


class RegimeSwitchingStrategy(bt.Strategy):
    params = (
        ('sma_long_period', 200),
        ('sma_short_period', 50),
        ('rsi_period', 14),
        ('atr_period', 14),
        ('stddev_period', 20),
    )

    def __init__(self):
        self.sma200 = bt.indicators.SMA(self.data.close, period=self.p.sma_long_period)
        self.sma50 = bt.indicators.SMA(self.data.close, period=self.p.sma_short_period)
        self.rsi = bt.indicators.RSI(self.data.close, period=self.p.rsi_period)
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.price_stddev = bt.indicators.StandardDeviation(
            self.data.close, period=self.p.stddev_period
        )

        self.pending_order = None
        self.stop_order = None
        self.take_profit_order = None
        self.exit_order = None

    def _regime(self):
        price = self.data.close[0]
        sma200 = self.sma200[0]
        atr_higher_than_50_bars_ago = self.atr[0] > self.atr[-50]

        if (price > sma200 and self.rsi[0] > 50 and
                atr_higher_than_50_bars_ago):
            return 'uptrend'

        if (price < sma200 and self.rsi[0] < 50 and
                atr_higher_than_50_bars_ago):
            return 'downtrend'

        if (abs(price - sma200) / sma200 < 0.02 and
                atr_higher_than_50_bars_ago):
            return 'sideways_volatile'

        return 'stable'

    def _cancel_protection(self):
        if self.stop_order is not None:
            self.cancel(self.stop_order)
        if self.take_profit_order is not None:
            self.cancel(self.take_profit_order)

    def next(self):
        if self.pending_order is not None or self.exit_order is not None:
            return

        crossed_above_sma50 = (
            self.data.close[-1] < self.sma50[-1] and
            self.data.close[0] > self.sma50[0]
        )
        crossed_below_sma50 = (
            self.data.close[-1] > self.sma50[-1] and
            self.data.close[0] < self.sma50[0]
        )

        if self.position:
            if self.position.size > 0:
                if crossed_below_sma50 or self.rsi[0] < 30:
                    self._cancel_protection()
                    self.exit_order = self.close()
            else:
                if crossed_above_sma50 or self.rsi[0] > 70:
                    self._cancel_protection()
                    self.exit_order = self.close()
            return

        # The supplied feed is AAPL, which is neither a major Forex pair nor
        # a specified index; therefore this strategy may not open positions.
        return

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return

        if order.status == order.Completed:
            if order == self.pending_order:
                self.pending_order = None

                entry_price = order.executed.price
                stop_distance = 1.5 * self.atr[0]
                size = abs(order.executed.size)

                if order.isbuy():
                    self.stop_order = self.sell(
                        size=size,
                        exectype=bt.Order.Stop,
                        price=entry_price - stop_distance,
                    )
                    self.take_profit_order = self.sell(
                        size=size,
                        exectype=bt.Order.Limit,
                        price=entry_price + (3 * stop_distance),
                        oco=self.stop_order,
                    )
                else:
                    self.stop_order = self.buy(
                        size=size,
                        exectype=bt.Order.Stop,
                        price=entry_price + stop_distance,
                    )
                    self.take_profit_order = self.buy(
                        size=size,
                        exectype=bt.Order.Limit,
                        price=entry_price - (3 * stop_distance),
                        oco=self.stop_order,
                    )

            elif order == self.exit_order:
                self.exit_order = None
            elif order == self.stop_order:
                self.stop_order = None
                self.take_profit_order = None
            elif order == self.take_profit_order:
                self.take_profit_order = None
                self.stop_order = None

        elif order.status in (
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            if order == self.pending_order:
                self.pending_order = None
            elif order == self.exit_order:
                self.exit_order = None
            elif order == self.stop_order:
                self.stop_order = None
            elif order == self.take_profit_order:
                self.take_profit_order = None
