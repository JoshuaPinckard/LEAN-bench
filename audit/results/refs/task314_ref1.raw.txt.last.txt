import backtrader as bt


class ReferenceStrategy(bt.Strategy):
    params = (
        ("macd_me1", 12),
        ("macd_me2", 26),
        ("macd_signal", 9),
        ("atr_period", 14),
        ("risk_fraction", 0.01),
    )

    def __init__(self):
        self.macd = bt.indicators.MACD(
            self.data.close,
            period_me1=self.p.macd_me1,
            period_me2=self.p.macd_me2,
            period_signal=self.p.macd_signal,
        )
        self.crossover = bt.indicators.CrossOver(self.macd.macd, self.macd.signal)
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)

        self.entry_order = None
        self.exit_order = None
        self.stop_order = None
        self.entry_price = None
        self.entry_is_long = None

    def _size_from_atr(self):
        atr_value = float(self.atr[0])
        if atr_value <= 0:
            return 0
        return max(1, int((self.broker.getvalue() * self.p.risk_fraction) / atr_value))

    def next(self):
        if self.entry_order or self.exit_order:
            return

        if not self.position:
            size = self._size_from_atr()
            if not size:
                return

            if self.crossover[0] > 0:
                self.entry_order = self.buy(size=size)
            elif self.crossover[0] < 0:
                self.entry_order = self.sell(size=size)
            return

        opposite_signal = (
            (self.position.size > 0 and self.crossover[0] < 0)
            or (self.position.size < 0 and self.crossover[0] > 0)
        )

        if opposite_signal:
            if self.stop_order:
                self.cancel(self.stop_order)
            self.exit_order = self.close()

    def notify_order(self, order):
        if order.status not in (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            return

        if order is self.entry_order:
            self.entry_order = None

            if order.status == order.Completed:
                self.entry_price = order.executed.price
                self.entry_is_long = order.isbuy()

                if self.entry_is_long:
                    self.stop_order = self.sell(
                        exectype=bt.Order.Stop,
                        price=self.entry_price * 0.98,
                    )
                else:
                    self.stop_order = self.buy(
                        exectype=bt.Order.Stop,
                        price=self.entry_price * 1.02,
                    )

        elif order is self.exit_order:
            self.exit_order = None
            if order.status == order.Completed:
                self.entry_price = None
                self.entry_is_long = None

        elif order is self.stop_order:
            self.stop_order = None
            if order.status == order.Completed:
                self.entry_price = None
                self.entry_is_long = None
