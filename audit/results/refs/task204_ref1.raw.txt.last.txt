import backtrader as bt


class NXXTBaseBreakoutStrategy(bt.Strategy):
    def __init__(self):
        self.sma200 = bt.ind.SMA(self.data.close, period=200)
        self.sma20 = bt.ind.SMA(self.data.close, period=20)
        self.bb = bt.ind.BollingerBands(self.data.close, period=20, devfactor=2)
        self.bb_width = (self.bb.top - self.bb.bot) / self.bb.mid
        self.atr = bt.ind.ATR(self.data, period=14)
        self.macd = bt.ind.MACDHisto(
            self.data.close,
            period_me1=12,
            period_me2=26,
            period_signal=9,
        )
        self.rsi = bt.ind.RSI(self.data.close, period=14)
        self.volume_sma20 = bt.ind.SMA(self.data.volume, period=20)

        self.hh20 = bt.ind.Highest(self.data.high, period=20)
        self.ll20 = bt.ind.Lowest(self.data.low, period=20)
        self.low5 = bt.ind.Lowest(self.data.low, period=5)

        self.entry_order = None
        self.stop_order = None
        self.tp1_order = None
        self.tp2_order = None
        self.close_order = None

        self.pending_stop_price = None
        self.pending_base_high = None
        self.pending_base_low = None
        self.pending_base_height = None

        self.base_high = None
        self.base_height = None
        self.bars_held = 0

    def _is_order(self, order, tracked):
        return tracked is not None and order.ref == tracked.ref

    def _cancel_protective_orders(self):
        for order in (self.stop_order, self.tp1_order, self.tp2_order):
            if order is not None and order.alive():
                self.cancel(order)

    def next(self):
        if self.position:
            self.bars_held += 1

            if self.close_order is None:
                failsafe = (
                    self.data.close[0] < self.sma20[0]
                    and self.macd.histo[0] < 0
                )
                time_exit = self.bars_held >= 30

                if failsafe or time_exit:
                    self._cancel_protective_orders()
                    self.close_order = self.close()
            return

        if any((
            self.entry_order,
            self.stop_order,
            self.tp1_order,
            self.tp2_order,
            self.close_order,
        )):
            return

        base_high = self.hh20[-1]
        base_low = self.ll20[-1]
        base_height = base_high - base_low
        entry_price = base_high + 0.1 * self.atr[0]
        initial_stop = base_low - self.atr[0]

        trend = self.data.close[0] > self.sma200[0]
        consolidation = (
            self.bb_width[-1] < 0.08
            and (self.hh20[-1] - self.ll20[-1]) < 2.5 * self.atr[0]
        )
        higher_low = self.low5[0] > self.low5[-5]
        macd_cross = self.macd.histo[0] > 0 and self.macd.histo[-1] < 0
        macd_rising = (
            self.macd.histo[0] > self.macd.histo[-1]
            and self.macd.histo[-1] > self.macd.histo[-2]
            and self.macd.histo[-2] > self.macd.histo[-3]
        )
        momentum = (macd_cross or macd_rising) and self.rsi[0] > 50
        volume = self.data.volume[0] > 1.5 * self.volume_sma20[0]
        breakout = self.data.close[0] >= entry_price

        if trend and consolidation and higher_low and momentum and volume and breakout:
            risk_distance = entry_price - initial_stop
            if risk_distance <= 0:
                return

            capital = self.broker.getvalue()
            risk_size = (0.005 * capital) / risk_distance
            exposure_size = (0.02 * capital) / entry_price
            size = min(risk_size, exposure_size)

            if size <= 0:
                return

            self.pending_stop_price = initial_stop
            self.pending_base_high = base_high
            self.pending_base_low = base_low
            self.pending_base_height = base_height
            self.entry_order = self.buy(
                size=size,
                exectype=bt.Order.Stop,
                price=entry_price,
            )

    def notify_order(self, order):
        terminal = (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        )

        if order.status not in terminal:
            return

        if self._is_order(order, self.entry_order):
            self.entry_order = None

            if order.status == order.Completed:
                entry_price = order.executed.price
                total_size = abs(order.executed.size)
                tp1_size = total_size * 0.5
                tp2_size = total_size - tp1_size

                self.base_high = self.pending_base_high
                self.base_height = self.pending_base_height
                self.bars_held = 0

                self.stop_order = self.sell(
                    size=total_size,
                    exectype=bt.Order.Stop,
                    price=self.pending_stop_price,
                )
                self.tp1_order = self.sell(
                    size=tp1_size,
                    exectype=bt.Order.Limit,
                    price=entry_price + self.base_height,
                )
                self.tp2_order = self.sell(
                    size=tp2_size,
                    exectype=bt.Order.Limit,
                    price=entry_price + 2.0 * self.base_height,
                )

            self.pending_stop_price = None
            self.pending_base_high = None
            self.pending_base_low = None
            self.pending_base_height = None
            return

        if self._is_order(order, self.tp1_order):
            self.tp1_order = None

            if order.status == order.Completed and self.position:
                if self.stop_order is not None and self.stop_order.alive():
                    self.cancel(self.stop_order)

                self.stop_order = self.sell(
                    size=abs(self.position.size),
                    exectype=bt.Order.Stop,
                    price=self.base_high,
                )
            return

        if self._is_order(order, self.tp2_order):
            self.tp2_order = None
            if order.status == order.Completed:
                if self.stop_order is not None and self.stop_order.alive():
                    self.cancel(self.stop_order)
                if self.tp1_order is not None and self.tp1_order.alive():
                    self.cancel(self.tp1_order)
            return

        if self._is_order(order, self.stop_order):
            self.stop_order = None
            if order.status == order.Completed:
                if self.tp1_order is not None and self.tp1_order.alive():
                    self.cancel(self.tp1_order)
                if self.tp2_order is not None and self.tp2_order.alive():
                    self.cancel(self.tp2_order)
            return

        if self._is_order(order, self.close_order):
            self.close_order = None
            if order.status == order.Completed:
                self._cancel_protective_orders()
