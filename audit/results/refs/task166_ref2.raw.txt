import backtrader as bt


class IntraRangeMomentumTripleBarriers(bt.Strategy):
    params = (
        ("sma_period", 20),
        ("stoch_period", 14),
        ("stoch_dfast", 3),
        ("stoch_dslow", 3),
        ("bb_period", 20),
        ("bb_dev", 2.0),
        ("atr_period", 14),
        ("time_stop", 20),
        ("pos_size_pct", 0.01),
        ("min_daily_range_pct", 0.002),
        ("min_atr_pct", 0.0015),
    )

    def __init__(self):
        self.sma = bt.ind.SMA(self.data.close, period=self.params.sma_period)
        self.stoch = bt.ind.Stochastic(
            self.data,
            period=self.params.stoch_period,
            period_dfast=self.params.stoch_dfast,
            period_dslow=self.params.stoch_dslow,
        )
        self.bb = bt.ind.BollingerBands(
            self.data.close,
            period=self.params.bb_period,
            devfactor=self.params.bb_dev,
        )
        self.atr = bt.ind.ATR(self.data, period=self.params.atr_period)
        self.crossover = bt.ind.CrossOver(self.stoch.percK, self.stoch.percD)

        self.order = None
        self.entry_bar = 0

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            if self.position:
                self.entry_bar = len(self)
            self.order = None
        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        if self.order:
            return

        close = self.data.close[0]
        high = self.data.high[0]
        low = self.data.low[0]

        daily_range_pct = (high - low) / close
        atr_pct = self.atr[0] / close

        if not self.position:
            long_cond = (
                close > self.sma[0]
                and self.stoch.percK[0] > 66
                and self.crossover[0] > 0
                and daily_range_pct >= self.params.min_daily_range_pct
                and atr_pct > self.params.min_atr_pct
            )

            short_cond = (
                close < self.sma[0]
                and self.stoch.percK[0] < 33
                and self.crossover[0] < 0
                and daily_range_pct >= self.params.min_daily_range_pct
                and atr_pct > self.params.min_atr_pct
            )

            if long_cond:
                size = (self.broker.getvalue() * self.params.pos_size_pct) / close
                self.order = self.buy(size=size)
            elif short_cond:
                size = (self.broker.getvalue() * self.params.pos_size_pct) / close
                self.order = self.sell(size=size)

        else:
            bars_in_trade = len(self) - self.entry_bar

            if self.position.size > 0:
                tp_cond = high >= self.bb.lines.top[0]
                sl_cond = low <= self.bb.lines.bot[0]
                alt_cond = (close < self.sma[0]) or (bars_in_trade >= self.params.time_stop)

                if tp_cond or sl_cond or alt_cond:
                    self.order = self.close()

            elif self.position.size < 0:
                tp_cond = low <= self.bb.lines.bot[0]
                sl_cond = high >= self.bb.lines.top[0]
                alt_cond = (close > self.sma[0]) or (bars_in_trade >= self.params.time_stop)

                if tp_cond or sl_cond or alt_cond:
                    self.order = self.close()
