import backtrader as bt


class SMTCPatternStrategy(bt.Strategy):
    def __init__(self):
        self.ema = bt.indicators.EMA(self.data.close, period=9)
        self.order = None
        self.tp_active = False
        self.trail_stop = None

    def notify_order(self, order):
        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            self.order = None

    def next(self):
        if not self.position:
            self.tp_active = False
            self.trail_stop = None

            if self.order is not None:
                return

            c_prev = self.data.close[-1]
            o_prev = self.data.open[-1]
            l_prev = self.data.low[-1]

            c_curr = self.data.close[0]
            o_curr = self.data.open[0]

            is_bearish_prev = c_prev < o_prev
            is_bullish_curr = c_curr > o_curr

            if is_bearish_prev and is_bullish_curr:
                midpoint_prev = (o_prev + c_prev) / 2.0
                is_piercing = (o_curr < l_prev) and (c_curr >= midpoint_prev)
                is_engulfing = (o_curr <= c_prev) and (c_curr >= o_prev)

                pattern = is_piercing or is_engulfing

                if pattern and c_curr > 62.17:
                    self.order = self.buy()
        else:
            if self.order is not None:
                return

            c_curr = self.data.close[0]
            h_curr = self.data.high[0]

            if c_curr < 59.16:
                self.order = self.close()
            elif not self.tp_active:
                if h_curr >= 65.33:
                    self.tp_active = True
                    self.trail_stop = self.ema[0]
                    close_size = self.position.size * 0.5
                    self.order = self.close(size=close_size)
            else:
                self.trail_stop = max(self.trail_stop, self.ema[0])
                if c_curr < self.trail_stop:
                    self.order = self.close()
