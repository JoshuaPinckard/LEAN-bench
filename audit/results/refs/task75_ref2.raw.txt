import backtrader as bt


class TimeSeriesMomentum(bt.Strategy):
    params = (("period", 252),)

    def __init__(self):
        self.roc = bt.indicators.ROC(self.data.close, period=self.p.period)
        self.order = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            self.order = None

    def next(self):
        if self.order:
            return

        if not self.position:
            if self.roc[0] > 0:
                self.order = self.buy()
            elif self.roc[0] < 0:
                self.order = self.sell()
        else:
            if self.position.size > 0 and self.roc[0] < 0:
                self.order = self.close()
            elif self.position.size < 0 and self.roc[0] > 0:
                self.order = self.close()
