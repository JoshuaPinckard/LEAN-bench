import backtrader as bt


class ATHExitStrategy(bt.Strategy):
    params = (
        ('sma_fast_period', 20),
        ('sma_slow_period', 50),
        ('rsi_period', 14),
        ('vol_period', 20),
        ('rsi_ob_threshold', 70.0),
        ('vol_mult', 1.2),
        ('position_pct', 0.02),
    )

    def __init__(self):
        self.sma20 = bt.indicators.SMA(self.data.close, period=self.p.sma_fast_period)
        self.sma50 = bt.indicators.SMA(self.data.close, period=self.p.sma_slow_period)
        self.rsi = bt.indicators.RSI(self.data.close, period=self.p.rsi_period)
        self.sma_vol20 = bt.indicators.SMA(self.data.volume, period=self.p.vol_period)

        self.order = None
        self.max_high_so_far = None
        self.partial_tp_done = False

        self.in_ob = False
        self.curr_ob_max_price = 0.0
        self.curr_ob_max_rsi = 0.0
        self.prev_ob_max_price = 0.0
        self.prev_ob_max_rsi = 0.0

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        if self.max_high_so_far is None:
            self.max_high_so_far = max([self.data.high[-i] for i in range(1, len(self))])

        is_new_ath = self.data.high[0] > self.max_high_so_far
        if is_new_ath:
            self.max_high_so_far = self.data.high[0]

        bearish_div_signal = False
        if self.rsi[0] >= self.p.rsi_ob_threshold:
            if not self.in_ob:
                self.in_ob = True
                self.curr_ob_max_price = self.data.high[0]
                self.curr_ob_max_rsi = self.rsi[0]
            else:
                self.curr_ob_max_price = max(self.curr_ob_max_price, self.data.high[0])
                self.curr_ob_max_rsi = max(self.curr_ob_max_rsi, self.rsi[0])
        elif self.in_ob:
            self.in_ob = False
            if self.prev_ob_max_price > 0:
                if (self.curr_ob_max_price > self.prev_ob_max_price) and (self.curr_ob_max_rsi < self.prev_ob_max_rsi):
                    bearish_div_signal = True
            self.prev_ob_max_price = self.curr_ob_max_price
            self.prev_ob_max_rsi = self.curr_ob_max_rsi

        if self.order:
            return

        if not self.position:
            if is_new_ath:
                capital = self.broker.getvalue()
                target_val = capital * self.p.position_pct
                price = self.data.close[0]
                size = int(target_val / price)
                if size > 0:
                    self.order = self.buy(size=size)
                    self.partial_tp_done = False
        else:
            main_tp_signal = (self.data.close[0] < self.sma20[0]) and (self.data.volume[0] > self.p.vol_mult * self.sma_vol20[0])
            stop_loss_signal = self.data.close[0] < self.sma50[0]

            if main_tp_signal or stop_loss_signal or bearish_div_signal:
                self.order = self.close()
            elif is_new_ath and not self.partial_tp_done:
                tp_size = int(self.position.size * 0.5)
                if tp_size > 0:
                    self.order = self.sell(size=tp_size)
                    self.partial_tp_done = True
