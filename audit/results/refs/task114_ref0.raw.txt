import math

import backtrader as bt


class Ema200BreakoutRetestStrategy(bt.Strategy):
    params = dict(
        ema_period=200,
        sma_period=60,
        resistance_lookback=20,
        stop_loss_pct=0.30,
        tp_fallback_pct=0.05,
        risk_pct=0.02,
    )

    def __init__(self):
        self.ema = bt.indicators.ExponentialMovingAverage(
            self.data.close, period=self.p.ema_period
        )
        self.sma = bt.indicators.SimpleMovingAverage(
            self.data.close, period=self.p.sma_period
        )
        self.breakdown = bt.indicators.CrossDown(self.data.close, self.ema)
        self.resistance = bt.indicators.Highest(
            self.data.high, period=self.p.resistance_lookback
        )

        self.order = None
        self.awaiting_retest = False
        self.entry_price = None

    def notify_order(self, order):
        if order.status == order.Completed and order.isbuy():
            self.entry_price = order.executed.price
        if order.status in (order.Completed, order.Canceled, order.Margin, order.Rejected):
            self.order = None

    def next(self):
        if self.order:
            return

        if not self.position:
            if self.awaiting_retest:
                touched = self.data.low[0] <= self.ema[0] <= self.data.high[0]
                bullish_candle = self.data.close[0] > self.data.open[0]
                bounced = self.data.close[0] > self.ema[0]
                confirmed = touched and (bullish_candle or bounced)

                if confirmed:
                    resistance_now = self.resistance[0]
                    resistance_prior = self.resistance[-self.p.resistance_lookback]
                    descending = resistance_now < resistance_prior
                    broken = self.data.close[0] > resistance_now

                    if descending and broken:
                        cash_at_risk = self.broker.getvalue() * self.p.risk_pct
                        size = cash_at_risk / self.data.close[0]
                        self.order = self.buy(size=size)
                        self.awaiting_retest = False

            if self.breakdown[0]:
                self.awaiting_retest = True
        else:
            if math.isnan(self.sma[0]):
                tp_level = self.entry_price * (1 + self.p.tp_fallback_pct)
            else:
                tp_level = self.sma[0]
            sl_level = self.entry_price * (1 - self.p.stop_loss_pct)

            if self.data.close[0] >= tp_level or self.data.close[0] <= sl_level:
                self.order = self.close()
