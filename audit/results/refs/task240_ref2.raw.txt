import math
import backtrader as bt


class MeanReversionVolTargeting(bt.Strategy):
    params = (
        ("risk_factor", 100.0),
        ("rsi_long", 40.0),
        ("rsi_short", 60.0),
    )

    def __init__(self):
        self.order = None
        self.bb2 = bt.indicators.BollingerBands(self.data.close, period=100, devfactor=2.0)
        self.bb3 = bt.indicators.BollingerBands(self.data.close, period=100, devfactor=3.0)
        self.atr = bt.indicators.ATR(self.data, period=20)
        self.rsi = bt.indicators.RSI(self.data.close, period=14)
        self.crossover = bt.indicators.CrossOver(self.data.close, self.bb2.lines.mid)

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        if self.order:
            return

        close = self.data.close[0]
        bb2_top = self.bb2.lines.top[0]
        bb2_mid = self.bb2.lines.mid[0]
        bb2_bot = self.bb2.lines.bot[0]
        bb3_top = self.bb3.lines.top[0]
        bb3_bot = self.bb3.lines.bot[0]
        atr_val = self.atr[0]
        rsi_val = self.rsi[0]

        if not self.position:
            # Entry condition for long position
            if close < bb2_bot and rsi_val < self.p.rsi_long:
                cash = self.broker.getcash()
                capital = self.broker.getvalue()
                if atr_val > 0 and close > 0:
                    max_cash_shares = cash / close
                    target_vol_shares = math.floor(capital / (atr_val * self.p.risk_factor))
                    size = int(min(max_cash_shares, target_vol_shares))
                    if size > 0:
                        self.order = self.buy(size=size)

            # Entry condition for short position
            elif close > bb2_top and rsi_val > self.p.rsi_short:
                cash = self.broker.getcash()
                capital = self.broker.getvalue()
                if atr_val > 0 and close > 0:
                    max_cash_shares = cash / close
                    target_vol_shares = math.floor(capital / (atr_val * self.p.risk_factor))
                    size = int(min(max_cash_shares, target_vol_shares))
                    if size > 0:
                        self.order = self.sell(size=size)
        else:
            # Exit / Stop condition evaluation
            half_sigma_dist = 0.25 * (bb2_top - bb2_mid)
            mean_returned = (self.crossover[0] != 0) or (abs(close - bb2_mid) <= half_sigma_dist)

            if self.position.size > 0:
                stop_loss = close <= bb3_bot
                if stop_loss or mean_returned:
                    self.order = self.close()
            elif self.position.size < 0:
                stop_loss = close >= bb3_top
                if stop_loss or mean_returned:
                    self.order = self.close()
