import backtrader as bt


class KalmanTrendFlipStrategy(bt.Strategy):
    params = (
        ('kf_q', 1e-5),
        ('kf_r', 1e-2),
        ('kf_p0', 1.0),
        ('atr_period', 14),
        ('atr_mult', 2.0),
    )

    def __init__(self):
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.order = None
        self.pending_direction = None
        self.entry_price = None
        self.stop_price = None
        self.kf_initialized = False
        self.kf_estimate = None
        self.kf_covariance = None
        self.prev_above = None

    def next(self):
        close = self.data.close[0]

        if not self.kf_initialized:
            self.kf_estimate = close
            self.kf_covariance = self.p.kf_p0
            self.kf_initialized = True
        else:
            pred_estimate = self.kf_estimate
            pred_covariance = self.kf_covariance + self.p.kf_q
            kalman_gain = pred_covariance / (pred_covariance + self.p.kf_r)
            self.kf_estimate = pred_estimate + kalman_gain * (close - pred_estimate)
            self.kf_covariance = (1 - kalman_gain) * pred_covariance

        current_above = close > self.kf_estimate

        if self.order:
            self.prev_above = current_above
            return

        if self.position:
            if self.position.size > 0 and self.stop_price is not None and self.data.low[0] <= self.stop_price:
                self.order = self.close()
                self.prev_above = current_above
                return
            elif self.position.size < 0 and self.stop_price is not None and self.data.high[0] >= self.stop_price:
                self.order = self.close()
                self.prev_above = current_above
                return

        if self.prev_above is not None:
            if current_above and not self.prev_above:
                if self.position.size < 0:
                    self.pending_direction = 'long'
                    self.order = self.close()
                elif self.position.size == 0:
                    self.order = self.buy()
            elif not current_above and self.prev_above:
                if self.position.size > 0:
                    self.pending_direction = 'short'
                    self.order = self.close()
                elif self.position.size == 0:
                    self.order = self.sell()

        self.prev_above = current_above

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status == order.Completed:
            if self.position.size == 0:
                if self.pending_direction == 'long':
                    self.pending_direction = None
                    self.order = self.buy()
                    return
                elif self.pending_direction == 'short':
                    self.pending_direction = None
                    self.order = self.sell()
                    return
                self.entry_price = None
                self.stop_price = None
            else:
                self.entry_price = order.executed.price
                atr_val = self.atr[0]
                if self.position.size > 0:
                    self.stop_price = self.entry_price - self.p.atr_mult * atr_val
                else:
                    self.stop_price = self.entry_price + self.p.atr_mult * atr_val

        if order.status in [order.Canceled, order.Margin, order.Rejected]:
            self.pending_direction = None

        self.order = None
