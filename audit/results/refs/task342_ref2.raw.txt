import math
import backtrader as bt


class StochasticATRStrategy(bt.Strategy):
    params = (
        ("period", 14),
        ("period_dfast", 3),
        ("period_dslow", 3),
        ("atr_period", 14),
        ("risk_per_trade", 0.01),
        ("stop_pct", 0.02),
    )

    def __init__(self):
        self.stoch = bt.indicators.StochasticFull(
            self.data,
            period=self.p.period,
            period_dfast=self.p.period_dfast,
            period_dslow=self.p.period_dslow,
        )
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)

        self.order = None
        self.entry_price = None
        self.pending_opposite = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            if order.isbuy():
                if self.position.size > 0:
                    self.entry_price = order.executed.price
            elif order.issell():
                if self.position.size < 0:
                    self.entry_price = order.executed.price

            if self.position.size == 0:
                self.entry_price = None

            self.order = None

        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            self.order = None
            self.pending_opposite = None

    def _calc_size(self, is_long):
        equity = self.broker.getvalue()
        risk_amount = equity * self.p.risk_per_trade
        close_price = self.data.close[0]

        if close_price <= 0:
            return 0

        if is_long:
            stop_price = close_price * (1.0 - self.p.stop_pct)
        else:
            stop_price = close_price * (1.0 + self.p.stop_pct)

        stop_distance_dollars = abs(close_price - stop_price)
        if stop_distance_dollars <= 0:
            return 0

        size = math.floor(risk_amount / stop_distance_dollars)

        available_cash = self.broker.getcash()
        max_shares = math.floor(available_cash / close_price)
        size = min(size, max_shares)

        return max(0, int(size))

    def next(self):
        if self.order is not None:
            return

        percK = self.stoch.percK
        percD = self.stoch.percD

        bullish_cross = (percK[0] > percD[0]) and (percK[-1] <= percD[-1])
        bearish_cross = (percK[0] < percD[0]) and (percK[-1] >= percD[-1])

        long_signal = (percK[-1] < 20) and bullish_cross
        short_signal = (percK[-1] > 80) and bearish_cross

        if self.position.size > 0:
            entry_p = (
                self.entry_price
                if self.entry_price is not None
                else self.data.close[0]
            )
            long_stop = entry_p * (1.0 - self.p.stop_pct)

            if self.data.low[0] <= long_stop:
                self.order = self.close()
                return

            if (percK[-1] >= 50) and (percK[0] < 50):
                self.order = self.close()
                return

            if short_signal:
                self.pending_opposite = "short"
                self.order = self.close()
                return

        elif self.position.size < 0:
            entry_p = (
                self.entry_price
                if self.entry_price is not None
                else self.data.close[0]
            )
            short_stop = entry_p * (1.0 + self.p.stop_pct)

            if self.data.high[0] >= short_stop:
                self.order = self.close()
                return

            if (percK[-1] <= 50) and (percK[0] > 50):
                self.order = self.close()
                return

            if long_signal:
                self.pending_opposite = "long"
                self.order = self.close()
                return

        else:
            if self.pending_opposite == "short":
                size = self._calc_size(is_long=False)
                if size > 0:
                    self.order = self.sell(size=size)
                self.pending_opposite = None
                return
            elif self.pending_opposite == "long":
                size = self._calc_size(is_long=True)
                if size > 0:
                    self.order = self.buy(size=size)
                self.pending_opposite = None
                return

            if long_signal:
                size = self._calc_size(is_long=True)
                if size > 0:
                    self.order = self.buy(size=size)
            elif short_signal:
                size = self._calc_size(is_long=False)
                if size > 0:
                    self.order = self.sell(size=size)
