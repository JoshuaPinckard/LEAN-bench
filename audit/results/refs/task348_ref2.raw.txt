import backtrader as bt


class KalmanFilter(bt.Indicator):
    lines = ('kalman',)
    params = (
        ('Q', 0.0001),
        ('R', 0.01),
    )

    def __init__(self):
        self.addminperiod(1)
        self.x = None
        self.P = 1.0

    def next(self):
        z = self.data.close[0]
        if self.x is None:
            self.x = z
            self.P = 1.0
            self.lines.kalman[0] = self.x
            return

        # State prediction
        x_pred = self.x
        P_pred = self.P + self.p.Q

        # Measurement update
        K = P_pred / (P_pred + self.p.R)
        self.x = x_pred + K * (z - x_pred)
        self.P = (1.0 - K) * P_pred

        self.lines.kalman[0] = self.x


class KalmanATRStrategy(bt.Strategy):
    params = (
        ('q', 0.0001),
        ('r', 0.01),
        ('atr_period', 14),
        ('atr_mult', 2.0),
    )

    def __init__(self):
        self.kalman = KalmanFilter(self.data, Q=self.p.q, R=self.p.r)
        self.crossover = bt.ind.CrossOver(self.data.close, self.kalman)
        self.atr = bt.ind.ATR(self.data, period=self.p.atr_period)

        self.order = None
        self.stop_price = None
        self.pending_flip = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status == order.Completed:
            if self.position.size > 0:
                self.stop_price = (
                    order.executed.price - self.p.atr_mult * self.atr[0]
                )
            elif self.position.size < 0:
                self.stop_price = (
                    order.executed.price + self.p.atr_mult * self.atr[0]
                )
            else:
                self.stop_price = None

        self.order = None

    def next(self):
        if self.order:
            return

        if self.position.size > 0:
            if self.stop_price is not None and self.data.low[0] <= self.stop_price:
                self.pending_flip = None
                self.order = self.close()
            elif self.crossover[0] < 0:
                self.pending_flip = 'short'
                self.order = self.close()

        elif self.position.size < 0:
            if self.stop_price is not None and self.data.high[0] >= self.stop_price:
                self.pending_flip = None
                self.order = self.close()
            elif self.crossover[0] > 0:
                self.pending_flip = 'long'
                self.order = self.close()

        else:
            if self.pending_flip == 'short':
                self.order = self.sell()
                self.pending_flip = None
            elif self.pending_flip == 'long':
                self.order = self.buy()
                self.pending_flip = None
            else:
                if self.crossover[0] > 0:
                    self.order = self.buy()
                elif self.crossover[0] < 0:
                    self.order = self.sell()
