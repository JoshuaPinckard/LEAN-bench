import backtrader as bt


class SystematicLimitStopStrategy(bt.Strategy):
    """Reference implementation of Systematic Trading with Limit and Stop Orders.

    Generates signals on daily aggregated data using a 20-period SMA crossover,
    and executes entry limit orders and OCO exit brackets on hourly bars.
    """

    def __init__(self):
        self.daily_closes = []
        self.current_date = None
        self.day_close = None

        self.entry_order = None
        self.tp_order = None
        self.sl_order = None

    def notify_order(self, order):
        if order.status in [order.Completed]:
            if order == self.entry_order:
                self.entry_order = None
                entry_price = order.executed.price
                size = abs(order.executed.size)

                if order.isbuy():
                    tp_price = entry_price * 1.02
                    sl_price = entry_price * (1.0 - 0.015)
                    self.tp_order = self.sell(
                        exectype=bt.Order.Limit, price=tp_price, size=size
                    )
                    self.sl_order = self.sell(
                        exectype=bt.Order.Stop,
                        price=sl_price,
                        size=size,
                        oco=self.tp_order,
                    )
                else:
                    tp_price = entry_price * (1.0 - 0.02)
                    sl_price = entry_price * 1.015
                    self.tp_order = self.buy(
                        exectype=bt.Order.Limit, price=tp_price, size=size
                    )
                    self.sl_order = self.buy(
                        exectype=bt.Order.Stop,
                        price=sl_price,
                        size=size,
                        oco=self.tp_order,
                    )

            elif order in [self.tp_order, self.sl_order]:
                if order == self.tp_order and self.sl_order:
                    if self.sl_order.status in [order.Submitted, order.Accepted]:
                        self.cancel(self.sl_order)
                elif order == self.sl_order and self.tp_order:
                    if self.tp_order.status in [order.Submitted, order.Accepted]:
                        self.cancel(self.tp_order)
                self.tp_order = None
                self.sl_order = None

        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            if order == self.entry_order:
                self.entry_order = None
            if order == self.tp_order:
                self.tp_order = None
            if order == self.sl_order:
                self.sl_order = None

    def next(self):
        dt = self.data.datetime.date(0)

        if self.current_date is None:
            self.current_date = dt
            self.day_close = self.data.close[0]
            return

        if dt != self.current_date:
            # Day boundary transition: record previous day's close
            self.daily_closes.append(self.day_close)

            # Cancel any unfilled entry order from day t+1 upon entering day t+2
            if self.entry_order:
                self.cancel(self.entry_order)
                self.entry_order = None

            # Generate signal based on completed daily data
            if not self.position and self.entry_order is None:
                if len(self.daily_closes) >= 21:
                    c_t = self.daily_closes[-1]
                    c_t_prev = self.daily_closes[-2]
                    sma_t = sum(self.daily_closes[-20:]) / 20.0
                    sma_t_prev = sum(self.daily_closes[-21:-1]) / 20.0

                    buy_signal = (c_t_prev <= sma_t_prev) and (c_t > sma_t)
                    sell_signal = (c_t_prev >= sma_t_prev) and (c_t < sma_t)

                    if buy_signal:
                        limit_price = c_t
                        self.entry_order = self.buy(
                            exectype=bt.Order.Limit, price=limit_price
                        )
                    elif sell_signal:
                        limit_price = self.data.open[0]
                        self.entry_order = self.sell(
                            exectype=bt.Order.Limit, price=limit_price
                        )

            self.current_date = dt
            self.day_close = self.data.close[0]
        else:
            self.day_close = self.data.close[0]
