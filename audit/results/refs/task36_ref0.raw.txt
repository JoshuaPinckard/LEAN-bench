import backtrader as bt


class BreakoutAfterDoubleBottom(bt.Strategy):
    LOW_ZONE_CENTER = 135.0
    LOW_ZONE_TOL = 0.1
    RESISTANCE = 191.8
    STOP_LOSS = 132.3
    CONSOL_LOW = 134.2
    CONSOL_HIGH = 191.8
    CONSOL_BARS = 20
    VOL_SMA_PERIOD = 20
    VOL_BREAKOUT_MULT = 1.2
    RISK_PCT = 0.015
    PIVOT_WINDOW = 5

    def __init__(self):
        self.vol_sma = bt.indicators.SMA(self.data.volume, period=self.VOL_SMA_PERIOD)
        self.order = None
        self.pivots = []
        self.pattern_low = None
        self.watching_breakout = False
        self.stop_price = None
        self.take_profit_price = None

    def next(self):
        self._update_pivots()

        if self.order:
            return

        if self.position:
            self._check_exit()
            return

        if self.watching_breakout:
            self._check_entry()

    def _update_pivots(self):
        w = self.PIVOT_WINDOW
        if len(self) <= 2 * w:
            return
        low_c = self.data.low[-w]
        high_c = self.data.high[-w]
        lows_window = [self.data.low[-k] for k in range(0, 2 * w + 1)]
        highs_window = [self.data.high[-k] for k in range(0, 2 * w + 1)]
        if low_c == min(lows_window):
            self._register_pivot('low', low_c)
        if high_c == max(highs_window):
            self._register_pivot('high', high_c)

    def _register_pivot(self, ptype, price):
        if self.pivots and self.pivots[-1]['type'] == ptype:
            last = self.pivots[-1]
            if (ptype == 'low' and price < last['price']) or \
               (ptype == 'high' and price > last['price']):
                pivot = {'type': ptype, 'price': price}
                if ptype == 'low':
                    pivot['consolidated_before'] = self._check_consolidation()
                self.pivots[-1] = pivot
            return
        pivot = {'type': ptype, 'price': price}
        if ptype == 'low':
            pivot['consolidated_before'] = self._check_consolidation()
        self.pivots.append(pivot)
        self._check_double_bottom()

    def _check_consolidation(self):
        w = self.PIVOT_WINDOW
        if len(self) < w + self.CONSOL_BARS + 1:
            return False
        start = -(w + self.CONSOL_BARS)
        end = -(w + 1)
        for r in range(start, end + 1):
            c = self.data.close[r]
            if not (self.CONSOL_LOW <= c <= self.CONSOL_HIGH):
                return False
        return True

    def _check_double_bottom(self):
        if self.watching_breakout or self.pattern_low is not None:
            return
        if len(self.pivots) < 3:
            return
        p1, p2, p3 = self.pivots[-3], self.pivots[-2], self.pivots[-1]
        if p1['type'] != 'low' or p2['type'] != 'high' or p3['type'] != 'low':
            return
        lo, hi = self.LOW_ZONE_CENTER - self.LOW_ZONE_TOL, self.LOW_ZONE_CENTER + self.LOW_ZONE_TOL
        if not (lo <= p1['price'] <= hi):
            return
        if not (lo <= p3['price'] <= hi):
            return
        if p2['price'] <= hi or p2['price'] >= self.RESISTANCE:
            return
        if not p1.get('consolidated_before', False):
            return

        self.pattern_low = min(p1['price'], p3['price'])
        self.watching_breakout = True

    def _check_entry(self):
        if self.data.close[0] <= self.RESISTANCE:
            return
        if len(self.vol_sma) < 2 or self.vol_sma[-1] <= 0:
            return
        if self.data.volume[0] < self.VOL_BREAKOUT_MULT * self.vol_sma[-1]:
            return

        entry_price = self.data.close[0]
        risk_per_share = entry_price - self.STOP_LOSS
        if risk_per_share <= 0:
            self.watching_breakout = False
            self.pattern_low = None
            return

        height = self.RESISTANCE - self.pattern_low
        max_risk = self.RISK_PCT * self.broker.getvalue()
        size = int(max_risk / risk_per_share)
        if size <= 0:
            self.watching_breakout = False
            self.pattern_low = None
            return

        self.stop_price = self.STOP_LOSS
        self.take_profit_price = self.RESISTANCE + height
        self.order = self.buy(size=size)
        self.watching_breakout = False

    def _check_exit(self):
        price = self.data.close[0]
        if price <= self.stop_price:
            self.order = self.close()
        elif price >= self.take_profit_price:
            self.order = self.close()
        elif price < self.LOW_ZONE_CENTER:
            self.order = self.close()

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def notify_trade(self, trade):
        if trade.isclosed:
            self.pattern_low = None
            self.stop_price = None
            self.take_profit_price = None
