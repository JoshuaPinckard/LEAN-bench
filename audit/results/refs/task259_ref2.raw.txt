import backtrader as bt


class VolatilitySellingStrategy(bt.Strategy):
    """Reference implementation of Bullish Volatility Selling strategy."""

    def __init__(self):
        self.order = None
        self.peak_value = 0.0

        # Indicators
        self.bb = bt.indicators.BollingerBands(
            self.data, period=20, devfactor=2.0
        )
        self.ema50 = bt.indicators.EMA(self.data.close, period=50)
        self.sma200 = bt.indicators.SMA(self.data.close, period=200)
        self.rsi = bt.indicators.RSI(self.data.close, period=14)
        self.atr = bt.indicators.ATR(self.data, period=14)

        # Trade tracking variables
        self.entry_price = 0.0
        self.entry_bar = 0
        self.r_val = 0.0
        self.entry_bb_width = 0.0

    def notify_order(self, order):
        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            self.order = None

        if order.status == order.Completed and order.isbuy():
            self.entry_price = order.executed.price
            self.entry_bar = len(self)
            atr_val = self.atr[0]
            self.r_val = 1.5 * atr_val
            if self.bb.mid[0] != 0:
                self.entry_bb_width = (
                    self.bb.top[0] - self.bb.bot[0]
                ) / self.bb.mid[0]
            else:
                self.entry_bb_width = 0.0

    def next(self):
        # Update peak portfolio value for drawdown tracking
        curr_val = self.broker.getvalue()
        if self.peak_value == 0.0 or curr_val > self.peak_value:
            self.peak_value = curr_val

        # Guard pending orders
        if self.order:
            return

        if not self.position:
            # 1. Drawdown filter (< 10%)
            drawdown = (
                (self.peak_value - curr_val) / self.peak_value
                if self.peak_value > 0
                else 0.0
            )
            if drawdown >= 0.10:
                return

            # 2. Price filter (price >= SMA(200))
            if self.data.close[0] < self.sma200[0]:
                return

            # 3. Uptrend condition (EMA(50) > SMA(200))
            if not (self.ema50[0] > self.sma200[0]):
                return

            # 4. Oversold condition (RSI(14) < 35)
            if not (self.rsi[0] < 35):
                return

            # 5. Touch / break of lower Bollinger Band
            if not (self.data.low[0] <= self.bb.bot[0]):
                return

            # 6. Volatility filter (BB width in 2% - 8%)
            if self.bb.mid[0] == 0:
                return
            bb_width = (self.bb.top[0] - self.bb.bot[0]) / self.bb.mid[0]
            if not (0.02 <= bb_width <= 0.08):
                return

            # Position Sizing (1.5% risk of portfolio capital)
            risk_pct = 0.015
            risk_amount = curr_val * risk_pct
            r_per_share = 1.5 * self.atr[0]

            if r_per_share <= 0:
                return

            size = int(risk_amount / r_per_share)
            if size > 0:
                self.order = self.buy(size=size)

        else:
            bars_held = len(self) - self.entry_bar

            # 1. Take-Profit (+0.5R)
            tp_price = self.entry_price + 0.5 * self.r_val
            if self.data.high[0] >= tp_price:
                self.order = self.close()
                return

            # 2. Hard Stop-Loss (-1.0R = 1.5 * ATR below entry)
            hard_stop_price = self.entry_price - 1.0 * self.r_val
            if self.data.low[0] <= hard_stop_price:
                self.order = self.close()
                return

            # 3. Soft Stop-Loss (-0.8R evaluated on close)
            soft_stop_price = self.entry_price - 0.8 * self.r_val
            if self.data.close[0] <= soft_stop_price:
                self.order = self.close()
                return

            # 4. Volatility Exit (BB width increased by >= 20% from entry)
            if self.bb.mid[0] != 0:
                curr_bb_width = (
                    self.bb.top[0] - self.bb.bot[0]
                ) / self.bb.mid[0]
                if (
                    self.entry_bb_width > 0
                    and curr_bb_width >= 1.20 * self.entry_bb_width
                ):
                    self.order = self.close()
                    return

            # 5. Timebox Exit (Expiration at 60 bars / 0 DTE)
            if bars_held >= 60:
                self.order = self.close()
                return
