import backtrader as bt


class OrbStrategy(bt.Strategy):
    params = (
        ('opening_range_minutes', 30),
        ('entry_cutoff_minutes', 120),
        ('volume_lookback', 30),
    )

    def __init__(self):
        self.order = None
        self.current_date = None
        self.or_high = None
        self.or_low = None
        self.or_complete = False
        self.breakout_confirmed = False
        self.pullback_seen = False
        self.entry_price = None
        self.stop_price = None
        self.target_price = None
        self.volume = bt.indicators.SimpleMovingAverage(
            self.data.volume, period=self.p.volume_lookback
        )

    def _reset_day(self):
        self.or_high = None
        self.or_low = None
        self.or_complete = False
        self.breakout_confirmed = False
        self.pullback_seen = False

    def next(self):
        current_dt = self.data.datetime.datetime(0)
        current_date = current_dt.date()
        minutes_from_open = current_dt.hour * 60 + current_dt.minute - (9 * 60 + 30)

        if self.current_date != current_date:
            self.current_date = current_date
            self._reset_day()

        if minutes_from_open < 0:
            return

        if minutes_from_open < self.p.opening_range_minutes:
            high = float(self.data.high[0])
            low = float(self.data.low[0])
            self.or_high = high if self.or_high is None else max(self.or_high, high)
            self.or_low = low if self.or_low is None else min(self.or_low, low)
            return

        if self.or_high is None or self.or_low is None:
            return

        self.or_complete = True
        opening_range = self.or_high - self.or_low
        fib_236 = self.or_high - opening_range * 0.236
        fib_382 = self.or_high - opening_range * 0.382
        fib_500 = self.or_high - opening_range * 0.500
        fib_618 = self.or_high - opening_range * 0.618

        if self.position:
            if self.order is None:
                if float(self.data.low[0]) <= self.stop_price:
                    self.order = self.close()
                elif float(self.data.high[0]) >= self.target_price:
                    self.order = self.close()
            return

        if self.order is not None or minutes_from_open > self.p.entry_cutoff_minutes:
            return

        if not self.breakout_confirmed:
            if float(self.data.close[0]) > self.or_high:
                self.breakout_confirmed = True
            return

        upper_wick = float(self.data.high[0]) - max(
            float(self.data.open[0]), float(self.data.close[0])
        )
        body = abs(float(self.data.close[0]) - float(self.data.open[0]))
        selling_pressure = upper_wick > body

        in_pullback_zone = (
            float(self.data.low[0]) <= fib_382
            and float(self.data.high[0]) >= fib_618
        )

        if in_pullback_zone and selling_pressure:
            self.pullback_seen = True
            return

        if (
            self.pullback_seen
            and float(self.data.close[0]) > float(self.data.close[-1])
            and float(self.data.close[0]) > fib_382
        ):
            self.order = self.buy()
            self.stop_price = fib_618
            self.target_price = self.or_high + opening_range

    def notify_order(self, order):
        if order.status in (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            if order.status == order.Completed and order.isbuy():
                self.entry_price = order.executed.price
            self.order = None
