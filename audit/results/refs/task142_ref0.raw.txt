import backtrader as bt


class NXXTConsolidationTrading(bt.Strategy):
    params = dict(
        sma_fast_period=60,
        sma_slow_period=120,
        trend_lookback=10,
        support_level=170.0,
        support_zone_pct=0.015,
        take_profit_pct=0.03,
        risk_pct=0.015,
    )

    def __init__(self):
        self.sma_fast = bt.indicators.SMA(self.data.close, period=self.p.sma_fast_period)
        self.sma_slow = bt.indicators.SMA(self.data.close, period=self.p.sma_slow_period)
        self.sma_fast_prev = self.sma_fast(-self.p.trend_lookback)
        self.sma_slow_prev = self.sma_slow(-self.p.trend_lookback)
        self.order = None
        self.entry_price = None

    def notify_order(self, order):
        if order.status in (order.Completed, order.Canceled, order.Margin, order.Rejected):
            if order.status == order.Completed and order.isbuy():
                self.entry_price = order.executed.price
            self.order = None

    def next(self):
        if self.order:
            return

        close = self.data.close[0]
        support = self.p.support_level

        sma_fast_rising = self.sma_fast[0] > self.sma_fast_prev[0]
        sma_slow_rising = self.sma_slow[0] > self.sma_slow_prev[0]
        sma_fast_falling = self.sma_fast[0] < self.sma_fast_prev[0]
        sma_slow_falling = self.sma_slow[0] < self.sma_slow_prev[0]

        if not self.position:
            support_zone_high = support * (1.0 + self.p.support_zone_pct)
            in_support_zone = support <= close <= support_zone_high
            above_smas = close > self.sma_fast[0] and close > self.sma_slow[0]

            if in_support_zone and sma_fast_rising and sma_slow_rising and above_smas:
                stop_distance = close - support
                if stop_distance > 0:
                    risk_amount = self.broker.getvalue() * self.p.risk_pct
                    size = int(risk_amount / stop_distance)
                    if size > 0:
                        self.order = self.buy(size=size)
        else:
            if close < support:
                self.order = self.close()
            elif self.entry_price is not None and close >= self.entry_price * (1.0 + self.p.take_profit_pct):
                self.order = self.close()
            elif sma_fast_falling and sma_slow_falling:
                self.order = self.close()
