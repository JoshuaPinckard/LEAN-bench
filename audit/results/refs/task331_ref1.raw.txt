import backtrader as bt
import math


class SeasonalDonchianStrategy(bt.Strategy):
    params = (
        ("donchian_period", 20),
        ("atr_period", 14),
        ("atr_mult", 3.0),
        ("seasonal_long_thresh", 0.5),
        ("seasonal_short_thresh", -0.5),
    )

    class SeasonalScore(bt.Indicator):
        lines = ("seasonal",)

        def next(self):
            day_of_year = self.data.datetime.date(0).timetuple().tm_yday
            self.lines.seasonal[0] = math.sin(
                2.0 * math.pi * (day_of_year - 1) / 365.0
            )

    def __init__(self):
        self.donchian = bt.indicators.DonchianChannel(
            self.data,
            period=self.p.donchian_period,
        )
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)

        if "seasonal" in self.data.lines.getlinealiases():
            self.seasonal = self.data.seasonal
        else:
            self.seasonal = self.SeasonalScore(self.data)

        self.order = None
        self.trailing_stop = None

    def notify_order(self, order):
        if order.status in (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            if order is self.order:
                self.order = None

    def next(self):
        if self.order:
            return

        close = self.data.close[0]
        atr_distance = self.p.atr_mult * self.atr[0]

        long_signal = (
            close > self.donchian.dch[0]
            and self.seasonal[0] > self.p.seasonal_long_thresh
        )
        short_signal = (
            close < self.donchian.dcl[0]
            and self.seasonal[0] < self.p.seasonal_short_thresh
        )

        if self.position.size > 0:
            if self.trailing_stop is not None and close <= self.trailing_stop:
                self.order = self.close()
                self.trailing_stop = None
                return

            candidate_stop = close - atr_distance
            self.trailing_stop = (
                candidate_stop
                if self.trailing_stop is None
                else max(self.trailing_stop, candidate_stop)
            )

            if short_signal:
                self.order = self.close()
                self.trailing_stop = None
            return

        if self.position.size < 0:
            if self.trailing_stop is not None and close >= self.trailing_stop:
                self.order = self.close()
                self.trailing_stop = None
                return

            candidate_stop = close + atr_distance
            self.trailing_stop = (
                candidate_stop
                if self.trailing_stop is None
                else min(self.trailing_stop, candidate_stop)
            )

            if long_signal:
                self.order = self.close()
                self.trailing_stop = None
            return

        self.trailing_stop = None

        if long_signal:
            self.order = self.buy()
        elif short_signal:
            self.order = self.sell()
