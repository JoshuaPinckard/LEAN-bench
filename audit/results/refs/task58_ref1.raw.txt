import backtrader as bt


class SwingTradingStrategy(bt.Strategy):
    params = (
        ("risk_fraction", 0.01),
    )

    def __init__(self):
        self.ema50 = bt.indicators.EMA(self.data.close, period=50)
        self.ema21 = bt.indicators.EMA(self.data.close, period=21)
        self.rsi = bt.indicators.RSI(self.data.close, period=14)
        self.ema_cross = bt.indicators.CrossOver(self.ema21, self.ema50)

        self.entry_order = None
        self.stop_order = None
        self.target_order = None
        self.pending_stop_price = None
        self.latest_swing_low = None
        self.latest_swing_high = None

    def next(self):
        if len(self.data) >= 3:
            if self.data.low[-1] <= self.data.low[-2] and self.data.low[-1] < self.data.low[0]:
                self.latest_swing_low = self.data.low[-1]

            if self.data.high[-1] >= self.data.high[-2] and self.data.high[-1] > self.data.high[0]:
                self.latest_swing_high = self.data.high[-1]

        if self.entry_order or self.stop_order or self.target_order or self.position:
            return

        rsi_value = self.rsi[0]
        close = self.data.close[0]

        if (
            close > self.ema50[0]
            and self.ema_cross[0] > 0
            and rsi_value < 50
            and rsi_value >= 30
            and self.latest_swing_low is not None
        ):
            stop_price = min(self.latest_swing_low, self.ema50[0])
            risk_per_unit = close - stop_price
            if risk_per_unit > 0:
                size = (self.broker.getvalue() * self.p.risk_fraction) / risk_per_unit
                self.pending_stop_price = stop_price
                self.entry_order = self.buy(size=size)

        elif (
            close < self.ema50[0]
            and self.ema_cross[0] < 0
            and rsi_value > 50
            and rsi_value <= 70
            and self.latest_swing_high is not None
        ):
            stop_price = max(self.latest_swing_high, self.ema50[0])
            risk_per_unit = stop_price - close
            if risk_per_unit > 0:
                size = (self.broker.getvalue() * self.p.risk_fraction) / risk_per_unit
                self.pending_stop_price = stop_price
                self.entry_order = self.sell(size=size)

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return

        if order == self.entry_order:
            self.entry_order = None

            if order.status == order.Completed:
                entry_price = order.executed.price
                size = abs(order.executed.size)

                if order.isbuy():
                    target_price = entry_price + 2 * (entry_price - self.pending_stop_price)
                    self.stop_order = self.sell(
                        size=size,
                        exectype=bt.Order.Stop,
                        price=self.pending_stop_price,
                    )
                    self.target_order = self.sell(
                        size=size,
                        exectype=bt.Order.Limit,
                        price=target_price,
                        oco=self.stop_order,
                    )
                else:
                    target_price = entry_price - 2 * (self.pending_stop_price - entry_price)
                    self.stop_order = self.buy(
                        size=size,
                        exectype=bt.Order.Stop,
                        price=self.pending_stop_price,
                    )
                    self.target_order = self.buy(
                        size=size,
                        exectype=bt.Order.Limit,
                        price=target_price,
                        oco=self.stop_order,
                    )

            self.pending_stop_price = None

        if order == self.stop_order and order.status in (
            order.Completed, order.Canceled, order.Margin, order.Rejected
        ):
            self.stop_order = None

        if order == self.target_order and order.status in (
            order.Completed, order.Canceled, order.Margin, order.Rejected
        ):
            self.target_order = None
