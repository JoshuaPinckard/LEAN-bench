import backtrader as bt


class LeapCallProxyStrategy(bt.Strategy):
    params = (
        ("allocation", 0.25),
    )

    def __init__(self):
        self.sma50 = bt.indicators.SMA(self.data.close, period=50)
        self.sma200 = bt.indicators.SMA(self.data.close, period=200)
        self.atr14 = bt.indicators.ATR(self.data, period=14)
        self.price_sma14 = bt.indicators.SMA(self.data.close, period=14)
        self.bb = bt.indicators.BollingerBands(
            self.data.close, period=20, devfactor=2
        )
        self.volume_sma20 = bt.indicators.SMA(self.data.volume, period=20)
        self.stochastic = bt.indicators.Stochastic(self.data)

        self.order = None
        self.entry_price = None
        self.entry_bar = None
        self.last_exit_price = None
        self.exit_bar = None
        self.initial_size = None
        self.initial_entry_submitted = False

    def _maximum_size(self):
        return int(
            (self.broker.getvalue() * self.p.allocation) / self.data.close[0]
        )

    def _open_initial_position(self):
        size = self._maximum_size()
        if size > 0:
            self.order = self.buy(size=size)
            self.initial_entry_submitted = True

    def next(self):
        if self.order:
            return

        if not self.position:
            if not self.initial_entry_submitted:
                self._open_initial_position()
                return

            if (
                self.last_exit_price is not None
                and len(self.data) - self.exit_bar >= 10
                and abs(self.data.close[0] - self.last_exit_price)
                / self.last_exit_price <= 0.03
                and self.sma50[0] > self.sma200[0]
            ):
                size = min(self.initial_size, self._maximum_size())
                if size > 0:
                    self.order = self.buy(size=size)
            return

        bars_in_trade = len(self.data) - self.entry_bar
        bb_width_pct = (self.bb.top[0] - self.bb.bot[0]) / self.bb.mid[0]
        atr_pct = self.atr14[0] / self.price_sma14[0]
        price_increase = (
            (self.data.close[0] - self.entry_price) / self.entry_price
        )

        if (
            self.data.close[0] >= self.entry_price * 1.10
            or bars_in_trade >= 380
            or bb_width_pct - atr_pct > 0.10
            or (
                price_increase > 0.05
                and self.data.volume[0] > 1.5 * self.volume_sma20[0]
                and self.stochastic[0] > 80
            )
        ):
            self.order = self.close()

    def notify_order(self, order):
        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            if order.status == order.Completed:
                if order.isbuy():
                    self.entry_price = order.executed.price
                    self.entry_bar = len(self.data)
                    if self.initial_size is None:
                        self.initial_size = order.executed.size
                elif order.issell():
                    self.last_exit_price = order.executed.price
                    self.exit_bar = len(self.data)
                    self.entry_price = None
                    self.entry_bar = None

            self.order = None
