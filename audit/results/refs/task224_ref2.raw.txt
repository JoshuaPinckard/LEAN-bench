import math
import backtrader as bt


class SyntheticCoveredCallExit(bt.Strategy):
    params = (
        ("strike", 150.0),
        ("annual_limit_usd", 270000.0),
        ("lot", 100),
        ("initial_position", 2000),
    )

    def __init__(self):
        self.bb = bt.indicators.BollingerBands(
            self.data.close, period=20, devfactor=2.0
        )
        self.atr = bt.indicators.ATR(self.data, period=14)
        self.atr_pct = 100.0 * self.atr / self.data.close
        self.ema_atr_pct = bt.indicators.EMA(self.atr_pct, period=50)
        self.rsi = bt.indicators.RSI(self.data.close, period=14)
        self.macd = bt.indicators.MACDHisto(
            self.data.close, period_me1=12, period_me2=26, period_signal=9
        )
        self.sma_vol = bt.indicators.SMA(self.data.volume, period=20)

        self.order = None
        self.position_qty = self.p.initial_position
        self.sold_usd_this_year = 0.0
        self.last_year = None
        self.last_month = None
        self.lots_month = 0
        self.sold_this_month = False
        self.skipped_this_month = False

    def notify_order(self, order):
        if order.status in [order.Completed]:
            exec_shares = abs(order.executed.size)
            exec_usd = exec_shares * order.executed.price
            self.sold_usd_this_year += exec_usd
            self.position_qty = max(0, self.position_qty - exec_shares)
            self.order = None
        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def is_last_bar_of_month(self):
        try:
            return (
                self.data.datetime.date(0).month
                != self.data.datetime.date(1).month
            )
        except IndexError:
            return True

    def next(self):
        current_date = self.data.datetime.date(0)
        current_year = current_date.year
        current_month = current_date.month

        if self.last_year is None or current_year != self.last_year:
            self.sold_usd_this_year = 0.0
            self.last_year = current_year

        if self.last_month is None or current_month != self.last_month:
            self.last_month = current_month
            self.sold_this_month = False
            self.skipped_this_month = False

            remaining_limit_usd = max(
                0.0, self.p.annual_limit_usd - self.sold_usd_this_year
            )
            lots_limit = math.floor(
                remaining_limit_usd / (self.p.strike * self.p.lot)
            )
            lots_pos = math.floor(self.position_qty / self.p.lot)
            self.lots_month = max(0, min(lots_limit, lots_pos))

        if (
            self.lots_month <= 0
            or self.position_qty < self.p.lot
            or self.sold_this_month
            or self.skipped_this_month
            or self.order is not None
        ):
            return

        cond_2a = (self.data.high[0] >= self.p.strike) or (
            self.data.close[0] >= self.bb.top[0]
        )
        cond_2b = self.atr_pct[0] > self.ema_atr_pct[0]
        cond_2c = self.data.volume[0] >= self.sma_vol[0]
        sell_trigger_mid = cond_2a and cond_2b and cond_2c

        sell_trigger_eom = False
        if not sell_trigger_mid and self.is_last_bar_of_month():
            if self.data.close[0] >= self.p.strike:
                sell_trigger_eom = True

        if sell_trigger_mid or sell_trigger_eom:
            bullish_impulse = (
                (self.rsi[0] > 75)
                and (self.macd.histo[0] > 0)
                and (self.macd.histo[0] > self.macd.histo[-1])
            )

            if bullish_impulse:
                self.skipped_this_month = True
            else:
                shares_to_sell = self.lots_month * self.p.lot
                self.order = self.sell(size=shares_to_sell)
                self.sold_this_month = True
