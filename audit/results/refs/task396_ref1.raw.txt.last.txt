import backtrader as bt


class IntradayBreakoutStrategy(bt.Strategy):
    def __init__(self):
        self.bb = bt.indicators.BollingerBands(period=20, devfactor=2.0)
        self.atr = bt.indicators.ATR(period=14)

        self.band_width = self.bb.top - self.bb.bot
        self.squeeze_threshold = bt.indicators.SimpleMovingAverage(
            self.band_width, period=20
        )

        self.cross_top = bt.indicators.CrossOver(self.data.close, self.bb.top)
        self.cross_bot = bt.indicators.CrossOver(self.data.close, self.bb.bot)

        self.entry_order = None
        self.exit_order = None
        self.stop_order = None

    def next(self):
        if self.entry_order or self.exit_order:
            return

        if self.position.size > 0:
            if self.cross_top[0] < 0:
                if self.stop_order:
                    self.cancel(self.stop_order)
                self.exit_order = self.close()
            return

        if self.position.size < 0:
            if self.cross_bot[0] > 0:
                if self.stop_order:
                    self.cancel(self.stop_order)
                self.exit_order = self.close()
            return

        if self.stop_order:
            return

        if self.band_width[0] < self.squeeze_threshold[0]:
            if self.cross_top[0] > 0:
                self.entry_order = self.buy()
            elif self.cross_bot[0] < 0:
                self.entry_order = self.sell()

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return

        if order.status == order.Completed:
            if order is self.entry_order:
                self.entry_order = None
                if order.isbuy():
                    self.stop_order = self.sell(
                        exectype=bt.Order.Stop,
                        price=order.executed.price - 2 * self.atr[0],
                    )
                else:
                    self.stop_order = self.buy(
                        exectype=bt.Order.Stop,
                        price=order.executed.price + 2 * self.atr[0],
                    )
            elif order is self.exit_order:
                self.exit_order = None
                if self.stop_order:
                    self.cancel(self.stop_order)
            elif order is self.stop_order:
                self.stop_order = None
                if self.exit_order:
                    self.cancel(self.exit_order)

        elif order.status in (
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            if order is self.entry_order:
                self.entry_order = None
            elif order is self.exit_order:
                self.exit_order = None
            elif order is self.stop_order:
                self.stop_order = None
