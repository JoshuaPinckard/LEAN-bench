import backtrader as bt


class CorrelationMeanReversionStrategy(bt.Strategy):
    params = (
        ("LookbackPeriod", 20),
    )

    def __init__(self):
        self.corr = bt.indicators.Correlation(
            self.data.close,
            self.data.close(-1),
            period=self.p.LookbackPeriod,
        )
        self.corr_mean = bt.indicators.SimpleMovingAverage(
            self.corr,
            period=self.p.LookbackPeriod,
        )
        self.corr_std = bt.indicators.StandardDeviation(
            self.corr,
            period=self.p.LookbackPeriod,
        )

        self.order = None
        self.stop_order = None

    def notify_order(self, order):
        if order.status in (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            if order is self.stop_order:
                self.stop_order = None

            if order is self.order:
                self.order = None

                if order.status == order.Completed:
                    if self.position.size > 0 and order.isbuy():
                        self.stop_order = self.sell(
                            exectype=bt.Order.Stop,
                            price=order.executed.price * 0.98,
                        )
                    elif self.position.size < 0 and order.issell():
                        self.stop_order = self.buy(
                            exectype=bt.Order.Stop,
                            price=order.executed.price * 1.02,
                        )
                    elif self.stop_order is not None:
                        self.cancel(self.stop_order)
                        self.stop_order = None

    def next(self):
        if self.order is not None:
            return

        if self.corr_std[0] == 0 or self.corr_std[-1] == 0:
            return

        z = (self.corr[0] - self.corr_mean[0]) / self.corr_std[0]
        previous_z = (self.corr[-1] - self.corr_mean[-1]) / self.corr_std[-1]

        if self.position.size > 0:
            if previous_z < 0 and z >= 0:
                if self.stop_order is not None:
                    self.cancel(self.stop_order)
                    self.stop_order = None
                self.order = self.close()

        elif self.position.size < 0:
            if previous_z > 0 and z <= 0:
                if self.stop_order is not None:
                    self.cancel(self.stop_order)
                    self.stop_order = None
                self.order = self.close()

        elif z > 2.0 and z < previous_z:
            self.order = self.sell()

        elif z < -2.0 and z > previous_z:
            self.order = self.buy()
