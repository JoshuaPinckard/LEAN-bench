import backtrader as bt
from collections import deque


class DynamicSequenceStrategy(bt.Strategy):
    params = (
        ("record_window", 100),
        ("min_sequence_length", 2),
    )

    def __init__(self):
        self.sma10 = bt.indicators.SMA(self.data.close, period=10)
        self.order = None
        self.sequence_length = 0
        self._prior_sequence_lengths = deque(maxlen=self.p.record_window)

    def notify_order(self, order):
        if order.status in (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            self.order = None

    def next(self):
        sequence_condition = (
            self.data.close[0] >= self.data.close[-1]
            and self.sma10[0] >= self.sma10[-1]
        )

        if sequence_condition:
            self.sequence_length += 1
        else:
            self.sequence_length = 0

        is_record = (
            not self._prior_sequence_lengths
            or self.sequence_length > max(self._prior_sequence_lengths)
        )

        if not self.order:
            if self.position:
                if not sequence_condition:
                    self.order = self.close()
            elif (
                sequence_condition
                and self.sequence_length >= self.p.min_sequence_length
                and is_record
            ):
                self.order = self.buy()

        self._prior_sequence_lengths.append(self.sequence_length)
