```python
import backtrader as bt
import math


class ReferenceStrategy(bt.Strategy):
    def __init__(self):
        self.sma200 = bt.ind.SMA(self.data.close, period=200)
        self.sma200_20ago = self.sma200(-20)
        self.ema20 = bt.ind.EMA(self.data.close, period=20)
        self.ema50 = bt.ind.EMA(self.data.close, period=50)
        self.ema100 = bt.ind.EMA(self.data.close, period=100)
        self.ema200 = bt.ind.EMA(self.data.close, period=200)
        self.rsi = bt.ind.RSI(self.data.close, period=14)
        self.atr = bt.ind.ATR(self.data, period=14)

        self.bb = bt.ind.BollingerBands(self.data.close, period=20, devfactor=2.0)
        self.width = self.bb.top - self.bb.bot

        self.atr126 = bt.ind.SMA(self.atr, period=126)
        self.atr100 = bt.ind.SMA(self.atr, period=100)
        self.width100 = bt.ind.SMA(self.width, period=100)
        self.vol50 = bt.ind.SMA(self.data.volume, period=50)
        self.vol20 = bt.ind.SMA(self.data.volume, period=20)

        self.close_mid_cross = bt.ind.CrossOver(self.data.close, self.bb.mid)
        self.close_upper_cross = bt.ind.CrossOver(self.data.close, self.bb.top)
        self.close_lower_cross = bt.ind.CrossOver(self.data.close, self.bb.bot)
        self.ema_up_cross = bt.ind.CrossOver(self.ema20, self.ema50)
        self.ema_down_cross = bt.ind.CrossOver(self.ema50, self.ema20)

        self.stoch = bt.ind.Stochastic(
            self.data, period=14, period_dfast=3, period_dslow=3
        )
        self.stoch_cross = bt.ind.CrossOver(self.stoch.percK, self.stoch.percD)

        self.pending = {}
        self.primary = None
        self.auxiliary = {}
        self.next_trade_id = 1
        self.panic_until = -1
        self.gap_long_armed = False
        self.gap_short_armed = False

    def notify_order(self, order):
        record = self.pending.pop(order.ref, None)

        if order.status in [order.Canceled, order.Margin, order.Rejected]:
            if record and record["role"] == "entry":
                if record["bucket"] == "primary":
                    self.primary = None
                else:
                    self.auxiliary.pop(record["id"], None)
            return

        if order.status != order.Completed or not record:
            return

        if record["role"] == "entry":
            state = record["state"]
            state["entry"] = order.executed.price
            state["entry_bar"] = len(self)
            state["active"] = True
            if record["bucket"] == "primary":
                state["initial_stop"] = state["entry"] - 1.5 * state["atr"]
                state["stop"] = state["initial_stop"]
            else:
                direction = state["direction"]
                state["stop"] = state["entry"] - direction * state["atr"]
                state["target"] = state["entry"] + direction * state["target_atr"] * state["atr"]

        elif record["role"] == "exit":
            if record["bucket"] == "primary":
                self.primary = None
            else:
                self.auxiliary.pop(record["id"], None)

    def _submit_entry(self, bucket, state, direction, size):
        if size <= 0:
            return

        if bucket == "primary":
            order = self.buy(size=size)
            self.primary = state
        elif direction > 0:
            order = self.buy(size=size, tradeid=state["id"])
            self.auxiliary[state["id"]] = state
        else:
            order = self.sell(size=size, tradeid=state["id"])
            self.auxiliary[state["id"]] = state

        self.pending[order.ref] = {
            "role": "entry",
            "bucket": bucket,
            "id": state.get("id"),
            "state": state,
        }

    def _exit(self, bucket, state):
        if state.get("exit_pending"):
            return
        state["exit_pending"] = True
        order = self.close(size=state["size"], tradeid=state.get("id", 0))
        if order is not None:
            self.pending[order.ref] = {
                "role": "exit",
                "bucket": bucket,
                "id": state.get("id"),
            }

    def _aux_exposure(self):
        return sum(
            state["size"] * state.get("entry", self.data.close[0])
            for state in self.auxiliary.values()
            if not state.get("exit_pending")
        )

    def _aux_size(self):
        capital = self.broker.getvalue()
        remaining = max(0.0, 0.10 * capital - self._aux_exposure())
        risk_size = math.floor((0.01 * capital) / self.atr[0]) if self.atr[0] > 0 else 0
        exposure_size = math.floor(remaining / self.data.close[0]) if self.data.close[0] > 0 else 0
        return min(risk_size, exposure_size)

    def _primary_size(self):
        return math.floor((0.90 * self.broker.getvalue()) / self.data.close[0])

    def _panic_detected(self):
        for ago in range(5):
            hits = (
                int(self.atr[ago] > self.atr126[ago] * 1.5)
                + int(self.data.close[ago] < self.bb.bot[ago])
                + int(self.rsi[ago] < 30)
                + int(self.data.volume[ago] > self.vol50[ago] * 1.3)
            )
            if hits >= 3:
                return True
        return False

    def _new_aux_state(self, name, direction, target_atr, time_stop):
        trade_id = self.next_trade_id
        self.next_trade_id += 1
        return {
            "id": trade_id,
            "name": name,
            "direction": direction,
            "size": self._aux_size(),
            "atr": self.atr[0],
            "target_atr": target_atr,
            "time_stop": time_stop,
            "active": False,
            "exit_pending": False,
        }

    def _manage_primary(self):
        state = self.primary
        if not state or not state.get("active") or state.get("exit_pending"):
            return

        if self.data.close[0] > state["entry"]:
            state["stop"] = max(
                state["initial_stop"],
                self.data.close[0] - 2.0 * self.atr[0],
            )

        take_profit = self.data.close[0] >= state["entry"] + 3.0 * state["atr"]
        rsi_profit = self.rsi[0] > 70 and self.data.close[0] > self.bb.top[0]
        protective = self.ema_down_cross[0] < 0 or self.data.close[0] < self.ema50[0]
        stopped = self.data.close[0] <= state["stop"]

        if take_profit or rsi_profit or protective or stopped:
            self._exit("primary", state)

    def _manage_auxiliary(self):
        for state in list(self.auxiliary.values()):
            if not state.get("active") or state.get("exit_pending"):
                continue

            direction = state["direction"]
            age = len(self) - state["entry_bar"]
            stopped = (
                self.data.close[0] <= state["stop"]
                if direction > 0
                else self.data.close[0] >= state["stop"]
            )
            target = (
                self.data.close[0] >= state["target"]
                if direction > 0
                else self.data.close[0] <= state["target"]
            )
            timed = age >= state["time_stop"]

            if state["name"] == "squeeze":
                middle_return = (
                    self.data.close[0] <= self.bb.mid[0]
                    if direction > 0
                    else self.data.close[0] >= self.bb.mid[0]
                )
                exit_now = stopped or target or middle_return or timed
            elif state["name"] == "gap":
                ema_exit = (
                    self.data.close[0] < self.ema20[0]
                    if direction > 0
                    else self.data.close[0] > self.ema20[0]
                )
                exit_now = stopped or target or ema_exit or timed
            else:
                band_exit = (
                    self.data.close[0] >= self.bb.top[0]
                    if direction > 0
                    else self.data.close[0] <= self.bb.bot[0]
                )
                exit_now = stopped or target or band_exit or timed

            if exit_now:
                self._exit("auxiliary", state)

    def next(self):
        self._manage_primary()
        self._manage_auxiliary()

        if self._panic_detected():
            self.panic_until = len(self) + 10

        if (
            self.primary is None
            and len(self) <= self.panic_until
            and self.sma200[0] > self.sma200_20ago[0]
            and self.close_mid_cross[0] > 0
            and self.ema_up_cross[0] > 0
        ):
            self._submit_entry(
                "primary",
                {"size": self._primary_size(), "atr": self.atr[0], "active": False, "exit_pending": False},
                1,
                self._primary_size(),
            )

        squeeze = self.width[0] < self.width100[0] * 0.8 and self.atr[0] < self.atr100[0]
        if squeeze and self.close_upper_cross[0] > 0 and self.data.volume[0] > self.vol20[0] * 1.2:
            state = self._new_aux_state("squeeze", 1, 1.5, 3)
            self._submit_entry("auxiliary", state, 1, state["size"])
        elif squeeze and self.close_lower_cross[0] < 0 and self.data.volume[0] > self.vol20[0] * 1.2:
            state = self._new_aux_state("squeeze", -1, 1.5, 3)
            self._submit_entry("auxiliary", state, -1, state["size"])

        gap_up = (
            self.data.open[0] >= self.data.close[-1] + 1.5 * self.atr[-1]
            and self.data.close[-1] > self.bb.top[-1]
        )
        gap_down = (
            self.data.open[0] <= self.data.close[-1] - 1.5 * self.atr[-1]
            and self.data.close[-1] < self.bb.bot[-1]
        )
        if gap_up:
            self.gap_long_armed = True
        if gap_down:
            self.gap_short_armed = True

        stoch_up_20 = self.stoch_cross[0] > 0 and self.stoch.percK[-1] < 20
        stoch_down_20 = self.stoch_cross[0] < 0 and self.stoch.percK[-1] > 80

        if self.gap_long_armed and self.data.low[0] <= max(self.ema20[0], self.bb.mid[0]):
            self.gap_long_armed = False
            if stoch_up_20:
                state = self._new_aux_state("gap", 1, 2.0, 10)
                self._submit_entry("auxiliary", state, 1, state["size"])

        if self.gap_short_armed and self.data.high[0] >= min(self.ema20[0], self.bb.mid[0]):
            self.gap_short_armed = False
            if stoch_down_20:
                state = self._new_aux_state("gap", -1, 2.0, 10)
                self._submit_entry("auxiliary", state, -1, state["size"])

        long_trend = self.ema20[0] > self.ema50[0] > self.ema100[0] > self.ema200[0]
        short_trend = self.ema20[0] < self.ema50[0] < self.ema100[0] < self.ema200[0]

        if (
            long_trend
            and self.data.close[0] <= self.bb.mid[0]
            and self.stoch_cross[0] > 0
            and self.stoch.percK[-1] < 30
        ):
            state = self._new_aux_state("scalp", 1, 1.2, 8)
            self._submit_entry("auxiliary", state, 1, state["size"])
        elif (
            short_trend
            and self.data.close[0] >= self.bb.mid[0]
            and self.stoch_cross[0] < 0
            and self.stoch.percK[-1] > 70
        ):
            state = self._new_aux_state("scalp", -1, 1.2, 8)
            self._submit_entry("auxiliary", state, -1, state["size"])
```
