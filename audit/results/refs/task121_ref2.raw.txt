import backtrader as bt


class PullbackFibonacciEMA(bt.Strategy):
    params = (
        ("ema_period", 20),
        ("swing_period", 30),
        ("risk_pct", 0.01),
        ("rr_ratio", 9.0),
    )

    def __init__(self):
        self.ema = bt.indicators.EMA(self.data.close, period=self.p.ema_period)
        self.main_order = None

    def notify_order(self, order):
        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            if self.main_order and order.ref == self.main_order.ref:
                self.main_order = None

    def next(self):
        if len(self) < self.p.swing_period:
            return

        if self.position or self.main_order is not None:
            return

        highs = [
            self.data.high[-i] for i in range(self.p.swing_period - 1, -1, -1)
        ]
        lows = [
            self.data.low[-i] for i in range(self.p.swing_period - 1, -1, -1)
        ]

        high_max = max(highs)
        low_min = min(lows)
        idx_high = highs.index(high_max)
        idx_low = lows.index(low_min)

        if idx_high <= idx_low:
            return

        range_fib = high_max - low_min
        if range_fib <= 0:
            return

        fib_236 = high_max - 0.236 * range_fib

        if self.data.close[0] <= self.ema[0]:
            return

        cur_low = self.data.low[0]
        if cur_low > fib_236 or cur_low < low_min:
            return

        if self.data.close[0] <= self.data.open[0]:
            return

        entry_price = self.data.close[0]
        sl_price = low_min * 0.998
        risk_per_share = entry_price - sl_price

        if risk_per_share <= 0:
            return

        capital = self.broker.getvalue()
        risk_amount = capital * self.p.risk_pct
        size = int(risk_amount / risk_per_share)

        cash = self.broker.getcash()
        if size * entry_price > cash:
            size = int(cash / entry_price)

        if size <= 0:
            return

        tp_price = entry_price + self.p.rr_ratio * risk_per_share

        orders = self.buy_bracket(
            size=size, limitprice=tp_price, stopprice=sl_price
        )
        self.main_order = orders[0]
