import backtrader as bt


class TrailingStopShortStrategy(bt.Strategy):
    params = dict(
        entry_hour=13,
        entry_minute=10,
        size=10,
        trail_amount=0.19,
    )

    def __init__(self):
        self.order = None
        self.highest_since_entry = None

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return
        if order.status == order.Completed:
            side = 'SELL' if order.issell() else 'BUY'
            print(
                f"{bt.num2date(order.executed.dt)} {side} "
                f"volume={abs(order.executed.size)} price={order.executed.price}"
            )
        if order.status in (order.Completed, order.Canceled, order.Margin, order.Rejected):
            self.order = None

    def next(self):
        if self.order:
            return

        bar_time = self.data.datetime.time(0)

        if not self.position:
            self.highest_since_entry = None
            if bar_time.hour == self.p.entry_hour and bar_time.minute == self.p.entry_minute:
                self.order = self.sell(size=self.p.size)
            return

        current_high = self.data.high[0]
        if self.highest_since_entry is None or current_high > self.highest_since_entry:
            self.highest_since_entry = current_high

        stop_level = self.highest_since_entry - self.p.trail_amount
        if self.data.low[0] <= stop_level:
            self.order = self.close()
