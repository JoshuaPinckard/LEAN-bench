import math
import backtrader as bt


class DonchianChannel(bt.Indicator):
    lines = ('dch', 'dcl')
    params = (('period', 20),)

    def __init__(self):
        self.lines.dch = bt.indicators.Highest(self.data.high(-1), period=self.p.period)
        self.lines.dcl = bt.indicators.Lowest(self.data.low(-1), period=self.p.period)


class SeasonalIndicator(bt.Indicator):
    lines = ('seasonal',)

    def next(self):
        day_of_year = self.data.datetime.date(0).timetuple().tm_yday
        self.lines.seasonal[0] = math.sin(2 * math.pi * day_of_year / 365.0)


class DonchianATRSeasonalStrategy(bt.Strategy):
    params = (
        ('donchian_period', 20),
        ('atr_period', 14),
        ('atr_mult', 3.0),
        ('seasonal_long_thresh', 0.5),
        ('seasonal_short_thresh', -0.5),
    )

    def __init__(self):
        self.donchian = DonchianChannel(self.data, period=self.p.donchian_period)
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.seasonal = SeasonalIndicator(self.data)
        self.order = None
        self.stop_price = None

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        if self.order:
            return

        close = self.data.close[0]
        long_signal = close > self.donchian.dch[0] and self.seasonal.seasonal[0] > self.p.seasonal_long_thresh
        short_signal = close < self.donchian.dcl[0] and self.seasonal.seasonal[0] < self.p.seasonal_short_thresh

        if not self.position:
            if long_signal:
                self.order = self.buy()
            elif short_signal:
                self.order = self.sell()
            return

        stop_dist = self.p.atr_mult * self.atr[0]

        if self.position.size > 0:
            candidate = close - stop_dist
            self.stop_price = candidate if self.stop_price is None else max(self.stop_price, candidate)

            if short_signal:
                self.order = self.close()
                self.stop_price = None
                return
            if close <= self.stop_price:
                self.order = self.close()
                self.stop_price = None
                return
        else:
            candidate = close + stop_dist
            self.stop_price = candidate if self.stop_price is None else min(self.stop_price, candidate)

            if long_signal:
                self.order = self.close()
                self.stop_price = None
                return
            if close >= self.stop_price:
                self.order = self.close()
                self.stop_price = None
                return
