import backtrader as bt


class ZScoreBollingerMeanReversion(bt.Strategy):
    params = dict(
        bb_period=100,
        bb_devfactor=2,
        atr_period=14,
        entry_threshold=2.0,
        stop_threshold=3.0,
        half_life_frac=0.5,
        atr_stop_k=3.0,
        risk_pct=0.01,
    )

    def __init__(self):
        self.sma = bt.indicators.SMA(self.data.close, period=self.p.bb_period)
        self.bb = bt.indicators.BollingerBands(
            self.data.close, period=self.p.bb_period, devfactor=self.p.bb_devfactor
        )
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.crossover = bt.indicators.CrossOver(self.data.close, self.sma)

        self.order = None
        self.entry_z = None
        self.stop_price = None

    def _zscore(self):
        mid = self.sma[0]
        std = (self.bb.top[0] - mid) / 2.0
        if std == 0:
            return 0.0
        return (self.data.close[0] - mid) / std

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return
        self.order = None

    def next(self):
        if self.order:
            return

        z = self._zscore()

        if not self.position:
            if z <= -self.p.entry_threshold:
                size = (self.p.risk_pct * self.broker.getvalue()) / self.atr[0]
                self.entry_z = z
                self.stop_price = self.data.close[0] - self.p.atr_stop_k * self.atr[0]
                self.order = self.buy(size=size)
            elif z >= self.p.entry_threshold:
                size = (self.p.risk_pct * self.broker.getvalue()) / self.atr[0]
                self.entry_z = z
                self.stop_price = self.data.close[0] + self.p.atr_stop_k * self.atr[0]
                self.order = self.sell(size=size)
            return

        if self.position.size > 0:
            if self.crossover[0] > 0:
                self.order = self.close()
            elif abs(z) <= self.p.half_life_frac * abs(self.entry_z):
                self.order = self.close()
            elif z <= -self.p.stop_threshold or self.data.close[0] <= self.stop_price:
                self.order = self.close()
        else:
            if self.crossover[0] < 0:
                self.order = self.close()
            elif abs(z) <= self.p.half_life_frac * abs(self.entry_z):
                self.order = self.close()
            elif z >= self.p.stop_threshold or self.data.close[0] >= self.stop_price:
                self.order = self.close()
