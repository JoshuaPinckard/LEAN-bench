import backtrader as bt


class WilliamsRMomentumATRStrategy(bt.Strategy):
    params = (
        ("atr_mult", 2.0),
        ("update_stop", False),
    )

    def __init__(self):
        self.williams_r = bt.indicators.WilliamsR(self.data, period=14)
        self.momentum = bt.indicators.Momentum(self.data, period=14)
        self.atr = bt.indicators.ATR(self.data, period=14)

        self.order = None
        self.stop_order = None
        self.entry_price = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order == self.stop_order:
            if order.status == order.Completed:
                self.stop_order = None
                self.entry_price = None
            elif order.status in [order.Canceled, order.Margin, order.Rejected]:
                self.stop_order = None

        if order == self.order:
            if order.status == order.Completed:
                if order.isbuy() and self.position.size > 0:
                    self.entry_price = order.executed.price
                    self.stop_order = self.sell(
                        exectype=bt.Order.Stop,
                        price=self.entry_price - self.p.atr_mult * self.atr[0],
                    )
                elif order.issell() and self.position.size < 0:
                    self.entry_price = order.executed.price
                    self.stop_order = self.buy(
                        exectype=bt.Order.Stop,
                        price=self.entry_price + self.p.atr_mult * self.atr[0],
                    )
                elif self.position.size == 0:
                    self.entry_price = None

            if order.status in [
                order.Completed,
                order.Canceled,
                order.Margin,
                order.Rejected,
            ]:
                self.order = None

    def next(self):
        long_signal = (
            self.williams_r[-1] <= -80
            and self.williams_r[0] > -80
            and self.momentum[0] > 0
        )
        short_signal = (
            self.williams_r[-1] >= -20
            and self.williams_r[0] < -20
            and self.momentum[0] < 0
        )

        if self.position.size > 0 and self.p.update_stop and self.stop_order is None:
            self.stop_order = self.sell(
                exectype=bt.Order.Stop,
                price=self.entry_price - self.p.atr_mult * self.atr[0],
            )
        elif self.position.size < 0 and self.p.update_stop and self.stop_order is None:
            self.stop_order = self.buy(
                exectype=bt.Order.Stop,
                price=self.entry_price + self.p.atr_mult * self.atr[0],
            )

        if self.order:
            return

        if self.position.size > 0:
            if long_signal is False and short_signal:
                if self.stop_order:
                    self.cancel(self.stop_order)
                self.order = self.close()
            return

        if self.position.size < 0:
            if short_signal is False and long_signal:
                if self.stop_order:
                    self.cancel(self.stop_order)
                self.order = self.close()
            return

        if long_signal:
            self.order = self.buy()
        elif short_signal:
            self.order = self.sell()
