import backtrader as bt


class BuyAndHold(bt.Strategy):
    def __init__(self):
        self.order = None
        self.max_value = 0.0
        self.bought = False
        self.exited = False

    def start(self):
        self.broker.setcommission(
            commission=0.01,
            commtype=bt.CommInfoBase.COMM_PERSHARE,
            stocklike=True,
            mincomm=1.0,
        )

    def notify_order(self, order):
        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            self.order = None

    def next(self):
        val = self.broker.getvalue()
        if val > self.max_value:
            self.max_value = val

        if self.order:
            return

        if self.position:
            if self.max_value > 0:
                dd = (self.max_value - val) / self.max_value
                if dd >= 0.30:
                    self.order = self.close()
                    self.exited = True
        elif not self.bought and not self.exited:
            price = self.data.close[0]
            cash = self.broker.getcash()
            if cash > 1.0 and price > 0:
                size = int(min((cash - 1.0) / price, cash / (price + 0.01)))
                if size > 0:
                    self.order = self.buy(size=size)
                    self.bought = True
