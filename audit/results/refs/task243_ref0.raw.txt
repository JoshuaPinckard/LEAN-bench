import backtrader as bt


class BollingerBreakoutStrategy(bt.Strategy):
    params = dict(
        bb_period=20,
        bb_dev=2,
        sma_period=20,
        lookback=3,
        take_profit_pct=0.02,
        risk_pct=0.01,
    )

    def __init__(self):
        self.bb = bt.indicators.BollingerBands(
            self.data, period=self.p.bb_period, devfactor=self.p.bb_dev
        )
        self.sma = bt.indicators.SMA(self.data.close, period=self.p.sma_period)
        self.order = None
        self.entry_price = None

    def notify_order(self, order):
        if order.status in (order.Completed, order.Canceled, order.Margin, order.Rejected):
            self.order = None

    def next(self):
        if self.order:
            return

        if not self.position:
            breakout = self.data.close[0] > self.bb.top[0]
            prior_on_or_below = all(
                self.data.close[-i] <= self.bb.top[-i]
                for i in range(1, self.p.lookback + 1)
            )
            sma_rising = self.sma[0] > self.sma[-1]

            if breakout and prior_on_or_below and sma_rising:
                risk_per_share = self.data.close[0] - self.bb.top[0]
                if risk_per_share > 0:
                    capital = self.broker.getvalue()
                    risk_amount = self.p.risk_pct * capital
                    size = int(risk_amount / risk_per_share)
                    if size > 0:
                        self.entry_price = self.data.close[0]
                        self.order = self.buy(size=size)
        else:
            if self.data.close[0] < self.bb.top[0]:
                self.order = self.close()
            elif self.data.close[0] >= self.entry_price * (1 + self.p.take_profit_pct):
                self.order = self.close()
