import backtrader as bt


class IntradayVWAPBreakout(bt.Strategy):
    params = (
        ('N', 3),
        ('stop_loss_pct', 0.005),
    )

    def __init__(self):
        self.entry_order = None
        self.exit_order = None
        self.stop_order = None

        self._cum_pv = 0.0
        self._cum_volume = 0.0
        self._session_date = None
        self._previous_relations = []

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            if order is self.entry_order:
                self.entry_order = None

                if order.status == order.Completed:
                    if order.isbuy():
                        self.stop_order = self.sell(
                            exectype=bt.Order.Stop,
                            price=order.executed.price * (1.0 - self.p.stop_loss_pct),
                        )
                    else:
                        self.stop_order = self.buy(
                            exectype=bt.Order.Stop,
                            price=order.executed.price * (1.0 + self.p.stop_loss_pct),
                        )

            elif order is self.exit_order:
                self.exit_order = None

            elif order is self.stop_order:
                self.stop_order = None

    def next(self):
        dt = self.data.datetime.datetime(0)
        current_date = dt.date()

        if dt.hour == 9 and dt.minute == 30:
            self._cum_pv = 0.0
            self._cum_volume = 0.0
            self._previous_relations = []
            self._session_date = current_date
        elif self._session_date is None or current_date != self._session_date:
            self._cum_pv = 0.0
            self._cum_volume = 0.0
            self._previous_relations = []
            self._session_date = current_date

        typical_price = (self.data.high[0] + self.data.low[0] + self.data.close[0]) / 3.0
        self._cum_pv += typical_price * self.data.volume[0]
        self._cum_volume += self.data.volume[0]
        vwap = self._cum_pv / self._cum_volume if self._cum_volume else typical_price

        if self.position and self.exit_order is None:
            should_exit = (
                (self.position.size > 0 and self.data.close[0] < vwap) or
                (self.position.size < 0 and self.data.close[0] > vwap)
            )
            if should_exit:
                if self.stop_order is not None:
                    self.cancel(self.stop_order)
                self.exit_order = self.close()

        if not self.position and self.entry_order is None and self.exit_order is None:
            if len(self._previous_relations) == self.p.N:
                if all(relation < 0 for relation in self._previous_relations) and self.data.close[0] > vwap:
                    self.entry_order = self.buy()
                elif all(relation > 0 for relation in self._previous_relations) and self.data.close[0] < vwap:
                    self.entry_order = self.sell()

        relation = self.data.close[0] - vwap
        self._previous_relations.append(relation)
        if len(self._previous_relations) > self.p.N:
            self._previous_relations.pop(0)
