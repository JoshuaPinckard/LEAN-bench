import backtrader as bt


class StochKeltnerStrategy(bt.Strategy):
    params = (
        ('stoch_period', 14),
        ('stoch_dfast', 3),
        ('stoch_dslow', 3),
        ('ema_period', 20),
        ('atr_period', 14),
        ('keltner_mult', 2.0),
        ('stop_atr_mult', 2.0),
    )

    def __init__(self):
        # Stochastic %K/%D (14,3,3)
        self.stoch = bt.indicators.Stochastic(
            self.data,
            period=self.p.stoch_period,
            period_dfast=self.p.stoch_dfast,
            period_dslow=self.p.stoch_dslow
        )
        
        # Keltner Channel (20-period EMA, ATR(14) with 2x multiplier)
        self.ema = bt.indicators.EMA(self.data.close, period=self.p.ema_period)
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.upper_band = self.ema + self.p.keltner_mult * self.atr
        self.lower_band = self.ema - self.p.keltner_mult * self.atr

        # Order tracking attributes
        self.order = None
        self.stop_order = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            if order == self.order:
                self.order = None
                # Set protective stop order upon position entry completion
                if self.position.size > 0:
                    stop_price = order.executed.price - (self.p.stop_atr_mult * self.atr[0])
                    self.stop_order = self.sell(exectype=bt.Order.Stop, price=stop_price)
                elif self.position.size < 0:
                    stop_price = order.executed.price + (self.p.stop_atr_mult * self.atr[0])
                    self.stop_order = self.buy(exectype=bt.Order.Stop, price=stop_price)
            elif order == self.stop_order:
                self.stop_order = None

        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            if order == self.order:
                self.order = None
            elif order == self.stop_order:
                self.stop_order = None

    def next(self):
        # Guard against duplicate orders while an entry/exit order is pending
        if self.order:
            return

        if self.position.size > 0:
            # Long Exit: price reverts to middle band (EMA)
            if self.data.close[0] >= self.ema[0]:
                if self.stop_order:
                    self.cancel(self.stop_order)
                self.order = self.close()
        elif self.position.size < 0:
            # Short Exit: price reverts to middle band (EMA)
            if self.data.close[0] <= self.ema[0]:
                if self.stop_order:
                    self.cancel(self.stop_order)
                self.order = self.close()
        else:
            # Entry conditions
            long_cond = (self.stoch.percK[0] < 20) and (self.data.close[0] < self.lower_band[0])
            short_cond = (self.stoch.percK[0] > 80) and (self.data.close[0] > self.upper_band[0])

            if long_cond:
                self.order = self.buy()
            elif short_cond:
                self.order = self.sell()
