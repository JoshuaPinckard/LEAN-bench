import math
import backtrader as bt


def calculate_percentile(values, p):
    sorted_v = sorted(values)
    k = (len(sorted_v) - 1) * (p / 100.0)
    f = math.floor(k)
    c = math.ceil(k)
    if f == c:
        return sorted_v[int(k)]
    return sorted_v[int(f)] * (c - k) + sorted_v[int(c)] * (k - f)


class VolatilityCheapnessStrategy(bt.Strategy):
    def __init__(self):
        self.order = None
        self.entry_price = 0.0
        self.entry_atr = 0.0
        self.highest_price = 0.0
        self.lowest_price = 0.0

        # Bollinger Bands
        self.bb20 = bt.indicators.BollingerBands(
            self.data.close, period=20, devfactor=2.0
        )
        self.bb100 = bt.indicators.BollingerBands(
            self.data.close, period=100, devfactor=2.0
        )

        # Widths and Ratio
        self.bb20_width = (self.bb20.top - self.bb20.bot) / self.bb20.mid
        self.bb100_width = (self.bb100.top - self.bb100.bot) / self.bb100.mid
        self.ratio = self.bb20_width / self.bb100_width

        # Directional Indicators
        self.ema20 = bt.indicators.EMA(self.data.close, period=20)
        self.sma50 = bt.indicators.SMA(self.data.close, period=50)
        self.rsi14 = bt.indicators.RSI(self.data.close, period=14)
        self.macd = bt.indicators.MACD(
            self.data.close, period_me1=12, period_me2=26, period_signal=9
        )

        # Liquidity Filter & Volume SMA
        self.vol_sma20 = bt.indicators.SMA(self.data.volume, period=20)
        self.vol_sma100 = bt.indicators.SMA(self.data.volume, period=100)

        # ATR for Exits
        self.atr14 = bt.indicators.ATR(self.data, period=14)

    def notify_order(self, order):
        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            self.order = None

        if order.status == order.Completed:
            if self.position.size != 0:
                self.entry_price = order.executed.price
                self.entry_atr = self.atr14[0]
                self.highest_price = self.data.high[0]
                self.lowest_price = self.data.low[0]

    def next(self):
        if self.order:
            return

        if self.position.size > 0:
            self.highest_price = max(self.highest_price, self.data.high[0])
            stop_price = self.entry_price - 1.5 * self.entry_atr
            target_price = self.entry_price + 3.0 * self.entry_atr
            trailing_stop = self.highest_price - 1.0 * self.atr14[0]

            if (
                self.data.low[0] <= stop_price
                or self.data.high[0] >= target_price
                or self.data.low[0] <= trailing_stop
                or self.ema20[0] < self.sma50[0]
            ):
                self.order = self.close()
            return

        elif self.position.size < 0:
            self.lowest_price = min(self.lowest_price, self.data.low[0])
            stop_price = self.entry_price + 1.5 * self.entry_atr
            target_price = self.entry_price - 3.0 * self.entry_atr
            trailing_stop = self.lowest_price + 1.0 * self.atr14[0]

            if (
                self.data.high[0] >= stop_price
                or self.data.low[0] <= target_price
                or self.data.high[0] >= trailing_stop
                or self.ema20[0] > self.sma50[0]
            ):
                self.order = self.close()
            return

        # Liquidity filter check
        liquidity_ok = self.vol_sma20[0] > self.vol_sma100[0]
        if not liquidity_ok:
            return

        bullish_bias = (
            (self.ema20[0] > self.sma50[0])
            and (self.rsi14[0] > 55)
            and (self.macd.macd[0] > 0)
        )
        bearish_bias = (
            (self.ema20[0] < self.sma50[0])
            and (self.rsi14[0] < 45)
            and (self.macd.macd[0] < 0)
        )

        is_cheap_vol = False
        if len(self.ratio) >= 252:
            ratio_vals = self.ratio.get(size=252)
            p20 = calculate_percentile(ratio_vals, 20)
            is_cheap_vol = self.ratio[0] <= p20

        # Signal 1: Cheap Calls
        if is_cheap_vol and bullish_bias:
            self.order = self.buy()
            return

        # Signal 2: Cheap Puts
        if is_cheap_vol and bearish_bias:
            self.order = self.sell()
            return

        # Signal 3: Earnings analogs breakout
        if len(self.bb20_width) >= 21:
            bb21_vals = self.bb20_width.get(size=21)
            is_21_min = self.bb20_width[0] <= min(bb21_vals) + 1e-9
            vol_spike = self.data.volume[0] > 1.5 * self.vol_sma20[0]

            if is_21_min and vol_spike:
                if self.data.close[0] > self.bb20.top[0]:
                    self.order = self.buy()
                elif self.data.close[0] < self.bb20.bot[0]:
                    self.order = self.sell()
