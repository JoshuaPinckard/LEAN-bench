import backtrader as bt
import math


class MonthlyFirstDayStrategy(bt.Strategy):
    def __init__(self):
        self.order = None
        self.initial_portfolio_value = None
        self.final_portfolio_value = None

    def start(self):
        self.broker.setcash(1000.0)
        self.initial_portfolio_value = self.broker.getvalue()

    def next(self):
        if self.order:
            return

        current_date = self.data.datetime.date(0)
        previous_date = self.data.datetime.date(-1) if len(self) > 1 else None

        is_first_trading_day = (
            previous_date is None
            or current_date.month != previous_date.month
            or current_date.year != previous_date.year
        )

        is_last_trading_day = False
        try:
            next_date = self.data.datetime.date(1)
            is_last_trading_day = (
                current_date.month != next_date.month
                or current_date.year != next_date.year
            )
        except IndexError:
            is_last_trading_day = True

        if self.position and is_last_trading_day:
            self.order = self.close()
        elif not self.position and is_first_trading_day:
            commission = self.broker.getcommissioninfo(self.data).p.commission
            size = math.floor(
                self.broker.getcash() / (self.data.close[0] * (1.0 + commission))
            )
            if size > 0:
                self.order = self.buy(size=size)

    def notify_order(self, order):
        if order.status in (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            self.order = None

    def stop(self):
        self.final_portfolio_value = self.broker.getvalue()
