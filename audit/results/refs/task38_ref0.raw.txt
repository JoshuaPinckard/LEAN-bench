import backtrader as bt


class APlusReversalStrategy(bt.Strategy):
    def __init__(self):
        self.order = None
        self.stop_order = None
        self.spike_high = None
        self.tp1 = None
        self.tp2 = None
        self.tp3 = None

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            if order is self.order:
                self.order = None
            if order is self.stop_order:
                self.stop_order = None

    def next(self):
        if self.order:
            return

        if not self.position:
            if len(self) < 2:
                return

            open_ = self.data.open[0]
            prev_high = self.data.high[-1]
            prev_close = self.data.close[-1]
            ict_open = prev_close

            outside_day = open_ > prev_high
            gap_up = open_ > prev_close
            ict_signal = open_ > ict_open

            if outside_day and gap_up and ict_signal:
                self.tp1 = prev_close
                self.tp2 = prev_high
                self.tp3 = ict_open
                self.spike_high = self.data.high[0]
                self.order = self.sell()
        else:
            if self.stop_order is None and self.spike_high is not None:
                self.stop_order = self.buy(exectype=bt.Order.Stop, price=self.spike_high)

            low = self.data.low[0]
            if low <= self.tp1 or low <= self.tp2 or low <= self.tp3:
                if self.stop_order:
                    self.cancel(self.stop_order)
                self.order = self.close()
