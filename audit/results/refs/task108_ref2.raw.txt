import backtrader as bt


class CorrectionReversalStrategy(bt.Strategy):
    params = (
        ("sma_period", 50),
        ("consolidation_period", 20),
        ("rebound_pct", 0.06),
        ("key_level_short", 175.0),
        ("target_long", 250.0),
        ("target_short_1", 3350.0),
        ("target_short_2", 2700.0),
        ("risk_pct", 0.02),
    )

    def __init__(self):
        self.sma = bt.indicators.SimpleMovingAverage(
            self.data.close, period=self.params.sma_period
        )
        self.order = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return
        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            self.order = None

    def _get_anchor_level(self):
        lookback = min(len(self), 50)
        for i in range(2, lookback):
            bar_low = self.data.low[-i]
            sub_highs = [self.data.high[-j] for j in range(0, i)]
            if sub_highs and max(sub_highs) >= bar_low * (
                1.0 + self.params.rebound_pct
            ):
                return bar_low
        return self.data.low[-1]

    def next(self):
        if len(self) < self.params.sma_period:
            return

        if self.order:
            return

        close = self.data.close[0]
        open_p = self.data.open[0]
        high = self.data.high[0]
        low = self.data.low[0]
        volume = self.data.volume[0]

        prev_close = self.data.close[-1]
        prev_open = self.data.open[-1]
        prev_low = self.data.low[-1]
        prev_vol = self.data.volume[-1]

        cons_period = min(len(self), self.params.consolidation_period)
        range_high = max([self.data.high[-i] for i in range(cons_period)])
        range_low = min([self.data.low[-i] for i in range(cons_period)])

        anchor_level = self._get_anchor_level()

        if not self.position:
            # Long Entry Conditions
            in_consolidation = range_low <= close <= range_high
            false_breakdown = (
                (low < range_low or prev_low < range_low)
                and close > range_low
                and volume > prev_vol
            )
            sma_not_broken = close >= self.sma[0]
            anchor_bounce = low <= anchor_level * 1.01 and close > open_p
            buying_vol_rising = volume > prev_vol and close > prev_close
            weak_selling = (prev_close >= prev_open) or (
                len(self) >= 3 and prev_vol < self.data.volume[-2]
            )

            if (
                in_consolidation
                and false_breakdown
                and sma_not_broken
                and anchor_bounce
                and buying_vol_rising
                and weak_selling
            ):
                cash_alloc = self.broker.getvalue() * self.params.risk_pct
                size = max(1, int(cash_alloc / close))
                self.order = self.buy(size=size)
                return

            # Short Entry Conditions
            strong_selling = volume > prev_vol and close < open_p
            dynamic_supp_broken = close < prev_low
            below_key_level = (
                close < self.params.key_level_short
                and prev_close < self.params.key_level_short
            )

            if (
                in_consolidation
                and strong_selling
                and dynamic_supp_broken
                and below_key_level
            ):
                cash_alloc = self.broker.getvalue() * self.params.risk_pct
                size = max(1, int(cash_alloc / close))
                self.order = self.sell(size=size)
                return

        elif self.position.size > 0:
            # Exit Long Conditions
            recent_low = min([self.data.low[-i] for i in range(5)])
            stop_loss = min(prev_low, recent_low) * 0.99

            if (
                high >= self.params.target_long
                or close < self.sma[0]
                or close < prev_low
                or low <= stop_loss
            ):
                self.order = self.close()

        elif self.position.size < 0:
            # Exit Short Conditions
            stop_loss = max(self.params.key_level_short, range_high) * 1.01

            if (
                low <= self.params.target_short_1
                or low <= self.params.target_short_2
                or close > range_high
                or high >= stop_loss
            ):
                self.order = self.close()
