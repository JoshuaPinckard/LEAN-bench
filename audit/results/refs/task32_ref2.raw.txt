import backtrader as bt


class High52WeekStopLoss(bt.Strategy):
    params = (
        ("period", 252),
        ("stop_pct", 0.95),
        ("limit_pct", 0.90),
    )

    def __init__(self):
        self.hi252 = bt.indicators.Highest(
            self.data.close, period=self.p.period
        )
        self.crossover = bt.indicators.CrossOver(
            self.data.close, self.hi252(-1)
        )

        self.order = None
        self.stop_order = None
        self.stop_price = None

    def notify_order(self, order):
        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            if self.order == order:
                self.order = None
            if self.stop_order == order:
                self.stop_order = None

    def next(self):
        if self.order:
            return

        if not self.position:
            if self.crossover[0] > 0:
                cash = self.broker.getcash()
                price = self.data.close[0]
                size = int(cash / (price * 1.001))
                if size > 0:
                    self.order = self.buy(size=size)
        else:
            target_stop = self.p.stop_pct * self.hi252[0]
            target_limit = self.p.limit_pct * self.hi252[0]

            if self.stop_order is None:
                self.stop_price = target_stop
                self.stop_order = self.sell(
                    exectype=bt.Order.StopLimit,
                    price=target_stop,
                    plimit=target_limit,
                    size=self.position.size,
                )
            else:
                if target_stop > self.stop_price:
                    self.cancel(self.stop_order)
