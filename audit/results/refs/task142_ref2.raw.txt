import backtrader as bt


class NXXTConsolidationTrading(bt.Strategy):
    params = (
        ("support_level", 170.0),
        ("sma_fast_period", 60),
        ("sma_slow_period", 120),
        ("lookback_n", 5),
        ("take_profit_pct", 0.03),
        ("risk_pct", 0.01),
    )

    def __init__(self):
        self.order = None
        self.entry_price = None

        self.sma60 = bt.indicators.SMA(
            self.data.close, period=self.params.sma_fast_period
        )
        self.sma120 = bt.indicators.SMA(
            self.data.close, period=self.params.sma_slow_period
        )

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            if order.isbuy():
                self.entry_price = order.executed.price
            elif order.issell():
                self.entry_price = None

        self.order = None

    def next(self):
        if self.order:
            return

        if not self.position:
            # Condition 1: Price drops to support zone (~$170.00) and holds above it
            price_at_support = (
                self.data.low[0] <= self.params.support_level
                and self.data.close[0] >= self.params.support_level
            )

            # Condition 2: Both 60 SMA and 120 SMA are higher than N periods ago
            smas_rising = (
                self.sma60[0] > self.sma60[-self.params.lookback_n]
                and self.sma120[0] > self.sma120[-self.params.lookback_n]
            )

            # Condition 3: Price is above both moving averages
            price_above_smas = (
                self.data.close[0] > self.sma60[0]
                and self.data.close[0] > self.sma120[0]
            )

            if price_at_support and smas_rising and price_above_smas:
                risk_per_share = self.data.close[0] - self.params.support_level
                if risk_per_share > 0:
                    capital = self.broker.getvalue()
                    max_risk_amount = capital * self.params.risk_pct
                    size = int(max_risk_amount / risk_per_share)
                    cash = self.broker.getcash()
                    max_cash_size = int(cash / self.data.close[0])
                    size = min(size, max_cash_size)

                    if size > 0:
                        self.order = self.buy(size=size)
        else:
            entry_p = (
                self.entry_price
                if self.entry_price is not None
                else self.position.price
            )

            # Exit Condition 1: Stop-loss - close candle below support level
            stop_loss = self.data.close[0] < self.params.support_level

            # Exit Condition 2: Take-profit - reaching 2-5% profit (default 3%)
            take_profit = (
                self.data.close[0]
                >= entry_p * (1.0 + self.params.take_profit_pct)
            )

            # Exit Condition 3: Alternative exit - both SMAs turn down compared to N periods ago
            smas_falling = (
                self.sma60[0] < self.sma60[-self.params.lookback_n]
                and self.sma120[0] < self.sma120[-self.params.lookback_n]
            )

            if stop_loss or take_profit or smas_falling:
                self.order = self.close()
