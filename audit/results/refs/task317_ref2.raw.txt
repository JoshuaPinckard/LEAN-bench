import backtrader as bt


class MonthlySMAStrategy(bt.Strategy):
    params = (
        ('sma_period', 20),
        ('holding_days', 5),
        ('stop_loss_pct', 0.02),
    )

    def __init__(self):
        self.sma = bt.indicators.SimpleMovingAverage(
            self.data.close, period=self.p.sma_period
        )
        self.order = None
        self.stop_order = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            if order.isbuy():
                entry_price = order.executed.price
                stop_price = entry_price * (1.0 - self.p.stop_loss_pct)
                self.stop_order = self.sell(
                    size=order.executed.size,
                    exectype=bt.Order.Stop,
                    price=stop_price
                )
            elif order == self.stop_order:
                self.stop_order = None

        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            if order == self.order:
                self.order = None
            if order == self.stop_order and order.status != order.Completed:
                self.stop_order = None

    def next(self):
        # Determine the trading day index within the current month
        current_date = self.data.datetime.date(0)
        day_count = 1
        i = -1
        while True:
            try:
                past_date = self.data.datetime.date(i)
                if past_date.month == current_date.month and past_date.year == current_date.year:
                    day_count += 1
                    i -= 1
                else:
                    break
            except (IndexError, ValueError):
                break

        if self.order:
            return

        # On the 1st trading day of the month, evaluate entry signal
        if day_count == 1:
            if not self.position and self.data.close[0] > self.sma[0]:
                cash = self.broker.getcash()
                size = int(cash / (self.data.close[0] * 1.001))
                if size > 0:
                    self.order = self.buy(size=size)

        # On or after the 5th trading day of the month, close position
        elif day_count >= self.p.holding_days:
            if self.position:
                if self.stop_order:
                    self.cancel(self.stop_order)
                    self.stop_order = None
                self.order = self.close()
