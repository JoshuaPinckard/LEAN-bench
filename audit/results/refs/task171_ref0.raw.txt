import backtrader as bt


class SyntheticCoveredCallStrategy(bt.Strategy):
    params = dict(
        cycle_len=21,
        strike_mult=1.15,
        bb_period=20,
        bb_devfactor=2.0,
        atr_period=14,
        sma_period=20,
        sell_fraction=0.40,      # midpoint of stated 30-50% lock-in sell
        roll_fraction=0.70,      # upper bound of stated "up to 70%" roll sell
        min_core_fraction=0.60,  # midpoint of stated 50-70% core floor
        roll_bars_remaining=5,
        block_sell_bars_remaining=2,
    )

    def __init__(self):
        self.sma20 = bt.indicators.SMA(self.data.close, period=self.p.sma_period)
        self.bb = bt.indicators.BollingerBands(
            self.data.close, period=self.p.bb_period, devfactor=self.p.bb_devfactor
        )
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)

        self.order = None
        self.initial_size = None
        self.cycle_bar = 0
        self.K = None
        self.k_sold_this_cycle = False
        self.pending_buyback = 0

    def notify_order(self, order):
        if order.status == order.Completed and self.initial_size is None and order.isbuy():
            self.initial_size = order.executed.size
        if order.status in (order.Completed, order.Canceled, order.Margin, order.Rejected):
            self.order = None

    def _set_new_K(self):
        close = self.data.close[0]
        self.K = max(close * self.p.strike_mult, self.bb.top[0])

    def next(self):
        if self.order:
            return

        if self.initial_size is None:
            self.order = self.buy()
            return

        if self.K is None:
            self._set_new_K()

        close = self.data.close[0]
        current_size = self.position.size
        core_min_size = self.initial_size * self.p.min_core_fraction
        bars_remaining = (self.p.cycle_len - 1) - self.cycle_bar

        # Rule 3: deep ITM roll up and out
        if bars_remaining <= self.p.roll_bars_remaining and close > (self.K + self.atr[0]):
            max_sellable = current_size - core_min_size
            if max_sellable > 0:
                sell_amt = int(min(current_size * self.p.roll_fraction, max_sellable))
                if sell_amt > 0:
                    self.order = self.sell(size=sell_amt)
            self.cycle_bar = 0
            self._set_new_K()
            self.k_sold_this_cycle = False
            return

        # Rule 1: reaching K locks in gains (blocked near cycle end per rule 2)
        if (not self.k_sold_this_cycle) and bars_remaining > self.p.block_sell_bars_remaining and close >= self.K:
            max_sellable = current_size - core_min_size
            if max_sellable > 0:
                sell_amt = int(min(current_size * self.p.sell_fraction, max_sellable))
                if sell_amt > 0:
                    self.order = self.sell(size=sell_amt)
                    self.pending_buyback += sell_amt
            self.k_sold_this_cycle = True

        # Buyback of the locked-in portion on return to SMA(20) or a touch of the lower band
        if self.order is None and self.pending_buyback > 0:
            if close <= self.sma20[0] or self.data.low[0] <= self.bb.bot[0]:
                self.order = self.buy(size=self.pending_buyback)
                self.pending_buyback = 0

        # Cycle rollover
        if self.cycle_bar >= self.p.cycle_len - 1:
            self.cycle_bar = 0
            self._set_new_K()
            self.k_sold_this_cycle = False
        else:
            self.cycle_bar += 1
