import backtrader as bt


class MacdVolumeCrossStrategy(bt.Strategy):
    params = dict(
        macd_fast=12,
        macd_slow=26,
        macd_signal=9,
        vol_period=20,
        vol_mult=2.0,
        atr_period=14,
        atr_mult=2.0,
    )

    def __init__(self):
        self.macd = bt.indicators.MACD(
            self.data,
            period_me1=self.p.macd_fast,
            period_me2=self.p.macd_slow,
            period_signal=self.p.macd_signal,
        )
        self.crossover = bt.indicators.CrossOver(self.macd.macd, self.macd.signal)
        self.vol_avg = bt.indicators.SMA(self.data.volume, period=self.p.vol_period)
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)

        self.order = None
        self.stop_order = None

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted, order.Partial):
            return

        if order.status == order.Completed:
            if order == self.order:
                self.order = None
                entry_price = order.executed.price
                atr_val = self.atr[0]
                if order.isbuy():
                    size = order.executed.size
                    stop_price = entry_price - self.p.atr_mult * atr_val
                    self.stop_order = self.sell(
                        exectype=bt.Order.Stop, price=stop_price, size=size
                    )
                else:
                    size = abs(order.executed.size)
                    stop_price = entry_price + self.p.atr_mult * atr_val
                    self.stop_order = self.buy(
                        exectype=bt.Order.Stop, price=stop_price, size=size
                    )
            elif order == self.stop_order:
                self.stop_order = None
        elif order.status in (order.Canceled, order.Margin, order.Rejected):
            if order == self.order:
                self.order = None
            elif order == self.stop_order:
                self.stop_order = None

    def next(self):
        if self.order:
            return

        volume_spike = self.data.volume[0] >= self.p.vol_mult * self.vol_avg[0]
        bullish_signal = self.crossover[0] > 0 and volume_spike
        bearish_signal = self.crossover[0] < 0 and volume_spike

        if not bullish_signal and not bearish_signal:
            return

        pos_size = self.position.size

        if bullish_signal:
            if pos_size < 0:
                if self.stop_order:
                    self.cancel(self.stop_order)
                    self.stop_order = None
                self.close()
                self.order = self.buy()
            elif pos_size == 0:
                self.order = self.buy()
        elif bearish_signal:
            if pos_size > 0:
                if self.stop_order:
                    self.cancel(self.stop_order)
                    self.stop_order = None
                self.close()
                self.order = self.sell()
            elif pos_size == 0:
                self.order = self.sell()
