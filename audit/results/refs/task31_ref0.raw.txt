import backtrader as bt


class OpeningRangeBreakoutStrategy(bt.Strategy):
    params = dict(
        range_minutes=40,
        rsi_period=14,
        risk_pct=0.01,
        stop_buffer_pct=0.001,
        eod_close_time=(15, 55),
    )

    def __init__(self):
        self.rsi = bt.indicators.RSI(self.data.close, period=self.p.rsi_period)
        self.bars_per_range = max(1, int(round(self.p.range_minutes / 5)))

        self.current_day = None
        self.range_high = None
        self.range_low = None
        self.range_bar_count = 0
        self.range_set = False
        self.entered_today = False

        self.entry_order = None
        self.stop_order = None
        self.tp_order = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status not in [order.Completed, order.Canceled,
                                 order.Margin, order.Rejected, order.Expired]:
            return

        if order == self.entry_order:
            self.entry_order = None
        if order == self.stop_order:
            self.stop_order = None
        if order == self.tp_order:
            self.tp_order = None

    def _calc_size(self, entry_price, stop_price):
        risk_per_share = abs(entry_price - stop_price)
        if risk_per_share <= 0:
            return 0
        risk_amount = self.broker.getvalue() * self.p.risk_pct
        return int(risk_amount / risk_per_share)

    def next(self):
        dt = self.data.datetime.datetime(0)
        bar_date = dt.date()
        bar_time = dt.time()

        if self.current_day != bar_date:
            self.current_day = bar_date
            self.range_high = None
            self.range_low = None
            self.range_bar_count = 0
            self.range_set = False
            self.entered_today = False

        eod_hour, eod_minute = self.p.eod_close_time
        if bar_time.hour == eod_hour and bar_time.minute == eod_minute:
            if self.stop_order is not None:
                self.cancel(self.stop_order)
            if self.tp_order is not None:
                self.cancel(self.tp_order)
            if self.entry_order is not None:
                self.cancel(self.entry_order)
            if self.position:
                self.close()
            return

        if not self.range_set:
            high = self.data.high[0]
            low = self.data.low[0]
            self.range_high = high if self.range_high is None else max(self.range_high, high)
            self.range_low = low if self.range_low is None else min(self.range_low, low)
            self.range_bar_count += 1
            if self.range_bar_count >= self.bars_per_range:
                self.range_set = True
            return

        if self.position or self.entry_order is not None or self.entered_today:
            return

        close = self.data.close[0]
        volume = self.data.volume[0]
        prev_volume = self.data.volume[-1]
        volume_increasing = volume > prev_volume
        rsi = self.rsi[0]

        if close > self.range_high and rsi <= 70:
            if volume_increasing or rsi > 50:
                stop_price = self.range_low * (1 - self.p.stop_buffer_pct)
                tp_price = self.range_high
                size = self._calc_size(close, stop_price)
                if size > 0:
                    orders = self.buy_bracket(
                        size=size,
                        exectype=bt.Order.Market,
                        stopprice=stop_price,
                        stopexec=bt.Order.Stop,
                        limitprice=tp_price,
                        limitexec=bt.Order.Stop,
                    )
                    self.entry_order, self.stop_order, self.tp_order = orders
                    self.entered_today = True

        elif close < self.range_low and rsi >= 30:
            if volume_increasing or rsi < 50:
                stop_price = self.range_high * (1 + self.p.stop_buffer_pct)
                tp_price = self.range_low
                size = self._calc_size(close, stop_price)
                if size > 0:
                    orders = self.sell_bracket(
                        size=size,
                        exectype=bt.Order.Market,
                        stopprice=stop_price,
                        stopexec=bt.Order.Stop,
                        limitprice=tp_price,
                        limitexec=bt.Order.Stop,
                    )
                    self.entry_order, self.stop_order, self.tp_order = orders
                    self.entered_today = True
