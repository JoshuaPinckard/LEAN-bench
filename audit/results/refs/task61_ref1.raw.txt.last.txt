import backtrader as bt


class PositionalManagementBasedOnMarketOverheating(bt.Strategy):
    params = (
        ("overheating_states", ()),
        ("qld_proportion", 0.5),
        ("tqqq_proportion", 0.5),
    )

    STAY_HEAVY = "Stay heavy on positions"
    STAY_LIGHT = "Stay light on positions"
    INITIAL_OVERHEATING = "Initial phase of overheating"

    def __init__(self):
        self.order = None
        self.previous_state = None

    def notify_order(self, order):
        if order.status in (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            self.order = None

    def next(self):
        index = len(self) - 1
        if index >= len(self.p.overheating_states):
            return

        state = self.p.overheating_states[index]
        transitioned_to_heavy = (
            state == self.STAY_HEAVY and self.previous_state != self.STAY_HEAVY
        )
        transitioned_to_light = (
            state == self.STAY_LIGHT and self.previous_state != self.STAY_LIGHT
        )

        # QQQ/QLD/TQQQ and a Bitcoin 2x leveraged instrument are unavailable
        # in the single AAPL feed supplied by the harness, so no proxy trades
        # are submitted.
        if self.order is None:
            if transitioned_to_heavy:
                pass
            elif transitioned_to_light:
                pass

        self.previous_state = state
