import backtrader as bt


class IchimokuStyleSMAMomentum(bt.Strategy):
    params = (
        ('fast_period', 9),   # Tenkan analogue
        ('mid_period', 26),   # Kijun analogue
        ('slow_period', 52),  # long-term cloud support analogue
    )

    def __init__(self):
        self.sma9 = bt.indicators.SMA(self.data.close, period=self.p.fast_period)
        self.sma26 = bt.indicators.SMA(self.data.close, period=self.p.mid_period)
        self.sma52 = bt.indicators.SMA(self.data.close, period=self.p.slow_period)
        # +1 when SMA(9) crosses SMA(26) from below to above,
        # -1 when SMA(9) crosses SMA(26) from above to below.
        self.crossover = bt.indicators.CrossOver(self.sma9, self.sma26)
        self.order = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return
        if order.status in [order.Completed, order.Canceled,
                            order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        if self.order:
            return

        close = self.data.close[0]
        sma26 = self.sma26[0]
        sma52 = self.sma52[0]
        cross = self.crossover[0]

        if not self.position:
            # Enter long: bullish cross with price above both averages and bullish cloud.
            if cross > 0 and close > sma26 and close > sma52 and sma26 > sma52:
                self.order = self.order_target_percent(target=1.0)
            # Enter short: bearish cross with price below both averages and bearish cloud.
            elif cross < 0 and close < sma26 and close < sma52 and sma26 < sma52:
                self.order = self.order_target_percent(target=-1.0)

        elif self.position.size > 0:
            # Exit long on reverse cross or close below the lower of SMA(26)/SMA(52).
            if cross < 0 or close < min(sma26, sma52):
                self.order = self.close()

        elif self.position.size < 0:
            # Exit short on reverse cross or close above the higher of SMA(26)/SMA(52).
            if cross > 0 or close > max(sma26, sma52):
                self.order = self.close()
