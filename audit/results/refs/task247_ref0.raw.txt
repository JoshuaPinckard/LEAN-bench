import math
import backtrader as bt


class MeanReversionPseudoSpread(bt.Strategy):
    params = dict(
        sma_trend_period=200,
        sma_base_period=50,
        atr_period=14,
        atr_sma_period=30,
        long_entry_dev=0.95,
        short_entry_dev=1.05,
        vol_filter_max=0.05,
        entry_slippage_pct=0.005,
        stop_multiplier=1.1,
        liquidity_sma_period=20,
        liquidity_threshold=10_000_000,
        risk_pct=0.01,
        atr_risk_mult=2.0,
    )

    def __init__(self):
        self.sma200 = bt.indicators.SMA(self.data.close, period=self.p.sma_trend_period)
        self.sma50 = bt.indicators.SMA(self.data.close, period=self.p.sma_base_period)
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.atr_sma30 = bt.indicators.SMA(self.atr, period=self.p.atr_sma_period)
        self.dollar_vol_sma = bt.indicators.SMA(self.data.close * self.data.volume, period=self.p.liquidity_sma_period)
        self.crossover = bt.indicators.CrossOver(self.data.close, self.sma50)

        self.order = None
        self.pending_intent = None
        self.direction = None
        self.entry_price = None
        self.sma50_entry = None
        self.d0 = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status == order.Completed:
            if self.pending_intent == 'long_entry':
                self.direction = 'long'
                self.entry_price = order.executed.price
                self.sma50_entry = self.sma50[0]
                self.d0 = self.sma50_entry - self.entry_price
            elif self.pending_intent == 'short_entry':
                self.direction = 'short'
                self.entry_price = order.executed.price
                self.sma50_entry = self.sma50[0]
                self.d0 = self.entry_price - self.sma50_entry
            elif self.pending_intent == 'exit':
                self.direction = None
                self.entry_price = None
                self.sma50_entry = None
                self.d0 = None

        self.order = None
        self.pending_intent = None

    def next(self):
        if self.order:
            return

        price = self.data.close[0]

        if not self.position:
            vol_ok = self.atr_sma30[0] / price <= self.p.vol_filter_max
            liquidity_ok = self.dollar_vol_sma[0] > self.p.liquidity_threshold

            long_signal = (
                price > self.sma200[0]
                and price <= self.sma50[0] * self.p.long_entry_dev
                and vol_ok
                and liquidity_ok
            )
            short_signal = (
                price < self.sma200[0]
                and price >= self.sma50[0] * self.p.short_entry_dev
                and vol_ok
                and liquidity_ok
            )

            if not (long_signal or short_signal):
                return

            atr_val = self.atr[0]
            if atr_val <= 0:
                return
            equity = self.broker.getvalue()
            size = math.floor((self.p.risk_pct * equity) / (self.p.atr_risk_mult * atr_val))
            if size <= 0:
                return

            if long_signal:
                limit_price = price * (1 + self.p.entry_slippage_pct)
                self.pending_intent = 'long_entry'
                self.order = self.buy(exectype=bt.Order.Limit, price=limit_price, size=size)
            else:
                limit_price = price * (1 - self.p.entry_slippage_pct)
                self.pending_intent = 'short_entry'
                self.order = self.sell(exectype=bt.Order.Limit, price=limit_price, size=size)
            return

        if self.direction == 'long':
            d = self.sma50[0] - price
            stop_hit = d >= self.p.stop_multiplier * self.d0
            cross_exit = self.crossover[0] > 0
        else:
            d = price - self.sma50[0]
            stop_hit = d >= self.p.stop_multiplier * self.d0
            cross_exit = self.crossover[0] < 0

        if cross_exit or stop_hit:
            self.pending_intent = 'exit'
            self.order = self.close()
