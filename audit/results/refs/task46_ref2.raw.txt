import backtrader as bt
import math
import collections
import statistics


class ElliottWavePatternStrategy(bt.Strategy):
    params = (
        ('pivot_bars', 5),
        ('risk_pct', 0.02),
        ('fib_target', 1.618),
        ('pattern_tolerance', 0.03),
    )

    def __init__(self):
        self.order = None
        self.pivots = []  # List of tuples: (ptype, idx, price)
        self.sl_price = None
        self.tp_price = None

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def update_pivots(self):
        p = self.p.pivot_bars
        if len(self) < 2 * p + 1:
            return

        idx = len(self) - 1 - p
        high_val = self.data.high[-p]
        low_val = self.data.low[-p]

        is_pivot_high = True
        is_pivot_low = True

        for i in range(0, 2 * p + 1):
            offset = -i
            if offset == -p:
                continue
            if self.data.high[offset] >= high_val:
                is_pivot_high = False
            if self.data.low[offset] <= low_val:
                is_pivot_low = False

        if is_pivot_high:
            self._add_pivot('H', idx, high_val)
        elif is_pivot_low:
            self._add_pivot('L', idx, low_val)

    def _add_pivot(self, ptype, idx, price):
        if not self.pivots:
            self.pivots.append((ptype, idx, price))
            return

        last_type, last_idx, last_price = self.pivots[-1]
        if last_type == ptype:
            if ptype == 'H' and price > last_price:
                self.pivots[-1] = (ptype, idx, price)
            elif ptype == 'L' and price < last_price:
                self.pivots[-1] = (ptype, idx, price)
        else:
            self.pivots.append((ptype, idx, price))

    def check_5wave_down(self):
        if len(self.pivots) < 6:
            return False
        pts = self.pivots[-6:]
        if [p[0] for p in pts] != ['H', 'L', 'H', 'L', 'H', 'L']:
            return False
        h0, l1, h2, l3, h4, l5 = [p[2] for p in pts]
        if h2 < h0 and h4 < h2 and l3 < l1 and l5 < l3:
            w1 = h0 - l1
            w3 = h2 - l3
            if w1 > 0 and w3 >= 0.618 * w1:
                return True
        return False

    def check_5wave_up(self):
        if len(self.pivots) < 6:
            return False
        pts = self.pivots[-6:]
        if [p[0] for p in pts] != ['L', 'H', 'L', 'H', 'L', 'H']:
            return False
        l0, h1, l2, h3, l4, h5 = [p[2] for p in pts]
        if l2 > l0 and l4 > l2 and h3 > h1 and h5 > h3:
            w1 = h1 - l0
            w3 = h3 - l2
            if w1 > 0 and w3 >= 0.618 * w1:
                return True
        return False

    def get_bullish_pattern(self):
        if len(self.pivots) < 3:
            return False, None, None

        # Double Bottom: L, H, L
        if len(self.pivots) >= 3 and self.pivots[-1][0] == 'L' and self.pivots[-2][0] == 'H' and self.pivots[-3][0] == 'L':
            l2 = self.pivots[-1][2]
            h1 = self.pivots[-2][2]
            l1 = self.pivots[-3][2]
            if abs(l1 - l2) / l1 <= self.p.pattern_tolerance:
                return True, h1, min(l1, l2)

        # Inverse Head & Shoulders: L, H, L, H, L
        if len(self.pivots) >= 5 and [p[0] for p in self.pivots[-5:]] == ['L', 'H', 'L', 'H', 'L']:
            pts = self.pivots[-5:]
            l_ls, h1, l_h, h2, l_rs = pts[0][2], pts[1][2], pts[2][2], pts[3][2], pts[4][2]
            if l_h < l_ls and l_h < l_rs:
                return True, max(h1, h2), l_h

        return False, None, None

    def get_bearish_pattern(self):
        if len(self.pivots) < 3:
            return False, None, None

        # Double Top: H, L, H
        if len(self.pivots) >= 3 and self.pivots[-1][0] == 'H' and self.pivots[-2][0] == 'L' and self.pivots[-3][0] == 'H':
            h2 = self.pivots[-1][2]
            l1 = self.pivots[-2][2]
            h1 = self.pivots[-3][2]
            if abs(h1 - h2) / h1 <= self.p.pattern_tolerance:
                return True, l1, max(h1, h2)

        # Head & Shoulders: H, L, H, L, H
        if len(self.pivots) >= 5 and [p[0] for p in self.pivots[-5:]] == ['H', 'L', 'H', 'L', 'H']:
            pts = self.pivots[-5:]
            h_ls, l1, h_h, l2, h_rs = pts[0][2], pts[1][2], pts[2][2], pts[3][2], pts[4][2]
            if h_h > h_ls and h_h > h_rs:
                return True, min(l1, l2), h_h

        return False, None, None

    def next(self):
        self.update_pivots()

        if self.order:
            return

        is_5_down = self.check_5wave_down()
        is_5_up = self.check_5wave_up()

        has_bull_pattern, bull_neckline, bull_extreme = self.get_bullish_pattern()
        has_bear_pattern, bear_neckline, bear_extreme = self.get_bearish_pattern()

        close = self.data.close[0]
        high = self.data.high[0]
        low = self.data.low[0]

        if self.position:
            if self.position.size > 0:
                if self.tp_price and high >= self.tp_price:
                    self.order = self.close()
                elif self.sl_price and low <= self.sl_price:
                    self.order = self.close()
                elif is_5_up or has_bear_pattern:
                    self.order = self.close()
            elif self.position.size < 0:
                if self.tp_price and low <= self.tp_price:
                    self.order = self.close()
                elif self.sl_price and high >= self.sl_price:
                    self.order = self.close()
                elif is_5_down or has_bull_pattern:
                    self.order = self.close()
            return

        if is_5_down and has_bull_pattern and bull_neckline and bull_extreme:
            if close > bull_neckline or close >= bull_extreme * 1.01:
                risk_amt = self.broker.getvalue() * self.p.risk_pct
                risk_per_share = abs(close - bull_extreme)
                if risk_per_share > 0:
                    size = int(risk_amt / risk_per_share)
                    max_size = int(self.broker.getcash() / close)
                    size = min(size, max_size)
                    if size > 0:
                        self.sl_price = bull_extreme
                        pattern_height = abs(bull_neckline - bull_extreme)
                        self.tp_price = close + pattern_height * self.p.fib_target
                        self.order = self.buy(size=size)

        elif is_5_up and has_bear_pattern and bear_neckline and bear_extreme:
            if close < bear_neckline or close <= bear_extreme * 0.99:
                risk_amt = self.broker.getvalue() * self.p.risk_pct
                risk_per_share = abs(bear_extreme - close)
                if risk_per_share > 0:
                    size = int(risk_amt / risk_per_share)
                    max_size = int(self.broker.getcash() / close)
                    size = min(size, max_size)
                    if size > 0:
                        self.sl_price = bear_extreme
                        pattern_height = abs(bear_extreme - bear_neckline)
                        self.tp_price = close - pattern_height * self.p.fib_target
                        self.order = self.sell(size=size)
