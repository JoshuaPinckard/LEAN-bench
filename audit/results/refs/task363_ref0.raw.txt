import backtrader as bt


class StochZScoreMeanReversion(bt.Strategy):
    params = (
        ('stoch_period', 14),
        ('stoch_kperiod', 3),
        ('stoch_dperiod', 3),
        ('zscore_period', 20),
        ('slope_period', 20),
        ('z_entry', 2.0),
        ('stop_pct', 0.02),
    )

    def __init__(self):
        self.stoch = bt.indicators.Stochastic(
            self.data,
            period=self.p.stoch_period,
            period_dfast=self.p.stoch_kperiod,
            period_dslow=self.p.stoch_dperiod,
        )
        self.pctk = self.stoch.percK
        self.sma20 = bt.indicators.SMA(self.pctk, period=self.p.zscore_period)
        self.std20 = bt.indicators.StandardDeviation(self.pctk, period=self.p.zscore_period)

        self.order = None
        self.stop_order = None
        self.prev_slope = None
        self.pending_direction = None

    def _slope(self):
        n = self.p.slope_period
        if len(self.pctk) < n:
            return None
        ys = [self.pctk[-i] for i in range(n - 1, -1, -1)]
        xbar = (n - 1) / 2.0
        ybar = sum(ys) / n
        num = 0.0
        den = 0.0
        for x, y in enumerate(ys):
            dx = x - xbar
            num += dx * (y - ybar)
            den += dx * dx
        if den == 0:
            return 0.0
        return num / den

    def notify_order(self, order):
        if order.status == order.Completed:
            if order is self.order and self.pending_direction is not None:
                entry_price = order.executed.price
                if self.pending_direction == 'long':
                    stop_price = entry_price * (1.0 - self.p.stop_pct)
                    self.stop_order = self.sell(exectype=bt.Order.Stop, price=stop_price)
                else:
                    stop_price = entry_price * (1.0 + self.p.stop_pct)
                    self.stop_order = self.buy(exectype=bt.Order.Stop, price=stop_price)
                self.pending_direction = None
                self.order = None
                return

        if order.status in (order.Completed, order.Canceled, order.Margin, order.Rejected):
            if order is self.order:
                self.order = None
                self.pending_direction = None
            if order is self.stop_order:
                self.stop_order = None

    def next(self):
        cur_slope = self._slope()
        if cur_slope is None:
            return
        prev_slope = self.prev_slope
        self.prev_slope = cur_slope

        if self.std20[0] == 0:
            z = 0.0
        else:
            z = (self.pctk[0] - self.sma20[0]) / self.std20[0]

        if self.order:
            return

        if not self.position:
            if prev_slope is None:
                return
            if z < -self.p.z_entry and cur_slope > 0 and prev_slope <= 0:
                self.pending_direction = 'long'
                self.order = self.buy()
            elif z > self.p.z_entry and cur_slope < 0 and prev_slope >= 0:
                self.pending_direction = 'short'
                self.order = self.sell()
        else:
            crossed_above_50 = self.pctk[0] > 50.0 and self.pctk[-1] <= 50.0
            crossed_below_50 = self.pctk[0] < 50.0 and self.pctk[-1] >= 50.0

            if self.position.size > 0 and (crossed_above_50 or z >= 0):
                if self.stop_order:
                    self.cancel(self.stop_order)
                self.order = self.close()
            elif self.position.size < 0 and (crossed_below_50 or z <= 0):
                if self.stop_order:
                    self.cancel(self.stop_order)
                self.order = self.close()
