import backtrader as bt


class ROCSMACrossoverStrategy(bt.Strategy):
    def __init__(self):
        self.fast_sma = bt.indicators.SMA(self.data.close, period=20)
        self.slow_sma = bt.indicators.SMA(self.data.close, period=50)
        self.buy_signal = bt.indicators.CrossOver(self.fast_sma, self.slow_sma)

        self.order = None
        self.true_positive = 0
        self.false_positive = 0
        self.true_negative = 0
        self.false_negative = 0
        self.confusion_matrix = {}
        self.roc_metrics = {}
        self._update_metrics()

    def _update_metrics(self):
        self.confusion_matrix = {
            "true_positive": self.true_positive,
            "false_positive": self.false_positive,
            "true_negative": self.true_negative,
            "false_negative": self.false_negative,
        }

        positive_count = self.true_positive + self.false_negative
        negative_count = self.false_positive + self.true_negative

        self.roc_metrics = {
            "true_positive_rate": (
                self.true_positive / positive_count
                if positive_count else None
            ),
            "false_positive_rate": (
                self.false_positive / negative_count
                if negative_count else None
            ),
        }

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            self.order = None

    def next(self):
        signal = self.buy_signal[0] > 0
        is_positive_day = self.data.close[0] > self.data.open[0]
        is_negative_day = self.data.close[0] < self.data.open[0]

        if is_positive_day:
            if signal:
                self.true_positive += 1
            else:
                self.false_negative += 1
        elif is_negative_day:
            if signal:
                self.false_positive += 1
            else:
                self.true_negative += 1

        self._update_metrics()

        if self.order:
            return

        if self.position:
            self.order = self.close()
        elif signal:
            self.order = self.buy(size=1)
