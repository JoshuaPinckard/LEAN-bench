import backtrader as bt


class DeepDipLeaderPyramiding(bt.Strategy):
    def __init__(self):
        self.order = None
        self.last_exit_bar = None
        self.add_count = 0
        self.took_partial = False
        self.max_price_since_entry = None
        self.stop_price = None

        # Required indicators as specified
        self.sma200 = bt.ind.SMA(self.data.close, period=200)
        self.ema20 = bt.ind.EMA(self.data.close, period=20)
        self.ema10 = bt.ind.EMA(self.data.close, period=10)
        self.rsi = bt.ind.RSI(self.data.close, period=14)
        self.macd = bt.ind.MACD(
            self.data.close, periodme1=12, periodme2=26, signalperiod=9
        )
        self.bband = bt.ind.BollingerBands(
            self.data.close, period=20, devfactor=2
        )
        self.atr = bt.ind.ATR(self.data, period=14)

        # Helper indicators for entry/exit conditions
        self.h252 = bt.ind.Highest(self.data.close, period=252)
        self.co_close_ema20 = bt.ind.CrossOver(self.data.close, self.ema20)
        self.co_rsi_30 = bt.ind.CrossOver(self.rsi, 30.0)
        self.co_macd = bt.ind.CrossOver(self.macd.macd, self.macd.signal)

    def notify_order(self, order):
        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            self.order = None

    def notify_trade(self, trade):
        if trade.isclosed:
            self.last_exit_bar = len(self)

    def next(self):
        if self.order:
            return

        if not self.position:
            # 30-bar post-exit cooldown check
            if (
                self.last_exit_bar is not None
                and (len(self) - self.last_exit_bar) < 30
            ):
                return

            # Phase 1 Crash Filter: close <= 30% of maximum close over 252 bars
            crash_filter = self.data.close[0] <= 0.30 * self.h252[0]
            if not crash_filter:
                return

            # Entry Condition A: CrossOver(close, EMA(20)) upward AND CrossOver(RSI(14), 30) upward
            cond_a = (self.co_close_ema20[0] > 0) and (self.co_rsi_30[0] > 0)

            # Entry Condition B: Touch of lower BB and close above lower BB with RSI < 35
            cond_b = (
                (self.data.low[0] <= self.bband.lines.bot[0])
                and (self.data.close[0] > self.bband.lines.bot[0])
                and (self.rsi[0] < 35.0)
            )

            if cond_a or cond_b:
                risk_amount = self.broker.getvalue() * 0.01
                risk_per_share = 1.5 * self.atr[0]
                if risk_per_share > 0:
                    size = int(risk_amount / risk_per_share)
                    if size > 0:
                        self.order = self.buy(size=size)
                        self.add_count = 0
                        self.took_partial = False
                        self.max_price_since_entry = self.data.high[0]
                        self.stop_price = self.data.close[0] - risk_per_share
        else:
            # Track maximum price since entry
            if self.max_price_since_entry is None:
                self.max_price_since_entry = self.data.high[0]
            else:
                self.max_price_since_entry = max(
                    self.max_price_since_entry, self.data.high[0]
                )

            # Phase 2 (Growth Phase): activated when close > SMA(200) and MACD > 0
            in_growth_phase = (self.data.close[0] > self.sma200[0]) and (
                self.macd.macd[0] > 0
            )

            if in_growth_phase:
                trailing_stop = self.max_price_since_entry - 3.0 * self.atr[0]
                if self.stop_price is None:
                    self.stop_price = trailing_stop
                else:
                    self.stop_price = max(self.stop_price, trailing_stop)
            else:
                if self.stop_price is None:
                    self.stop_price = self.position.price - 1.5 * self.atr[0]

            # 1. Stop Evaluation
            if self.stop_price is not None and self.data.low[0] <= self.stop_price:
                self.order = self.close()
                return

            # 2. Profit Taking Exits
            if (
                not self.took_partial
                and self.data.high[0] >= self.position.price * 1.50
            ):
                sell_size = max(1, int(self.position.size * 0.5))
                self.order = self.sell(size=sell_size)
                self.took_partial = True
                return

            remainder_pt = self.data.high[0] >= self.position.price * 2.00
            bb_rsi_exit = (self.data.high[0] >= self.bband.lines.top[0]) and (
                self.rsi[0] > 75.0
            )

            if remainder_pt or bb_rsi_exit:
                self.order = self.close()
                return

            # 3. Trend Exits
            macd_cross_down = self.co_macd[0] < 0
            two_closes_below_ema20 = (
                self.data.close[0] < self.ema20[0]
                and self.data.close[-1] < self.ema20[-1]
            )

            if macd_cross_down or two_closes_below_ema20:
                self.order = self.close()
                return

            # 4. Pyramiding Adds
            if in_growth_phase and self.add_count < 3:
                if self.data.close[0] > self.position.price:
                    pullback = (self.data.low[0] <= self.ema20[0]) or (
                        self.data.low[0] <= self.bband.lines.bot[0]
                    )
                    rsi_range = 45.0 <= self.rsi[0] <= 60.0
                    macd_pos = self.macd.macd[0] > 0

                    if pullback and rsi_range and macd_pos:
                        risk_amount = self.broker.getvalue() * 0.01
                        risk_per_share = 1.5 * self.atr[0]
                        if risk_per_share > 0:
                            size = int(risk_amount / risk_per_share)
                            if size > 0:
                                self.order = self.buy(size=size)
                                self.add_count += 1
