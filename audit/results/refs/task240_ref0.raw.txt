import math
import backtrader as bt


class MeanReversionATRStrategy(bt.Strategy):
    params = dict(
        bb_period=100,
        bb_devfactor=2,
        bb_stop_devfactor=3,
        atr_period=20,
        risk_factor=100,
        exit_band_frac=0.25,
    )

    def __init__(self):
        self.bb2 = bt.indicators.BollingerBands(
            self.data.close, period=self.p.bb_period, devfactor=self.p.bb_devfactor)
        self.bb3 = bt.indicators.BollingerBands(
            self.data.close, period=self.p.bb_period, devfactor=self.p.bb_stop_devfactor)
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.crossover_mid = bt.indicators.CrossOver(self.data.close, self.bb2.lines.mid)
        self.order = None

    def notify_order(self, order):
        if order.status in (order.Completed, order.Canceled, order.Margin, order.Rejected):
            self.order = None

    def _target_size(self):
        atr_val = self.atr[0]
        if atr_val <= 0:
            return 0
        close = self.data.close[0]
        cash = self.broker.getcash()
        capital = self.broker.getvalue()
        vol_size = math.floor(capital / (atr_val * self.p.risk_factor))
        cash_size = cash / close
        return max(math.floor(min(cash_size, vol_size)), 0)

    def next(self):
        if self.order:
            return

        close = self.data.close[0]
        top2 = self.bb2.lines.top[0]
        bot2 = self.bb2.lines.bot[0]
        mid2 = self.bb2.lines.mid[0]
        bot3 = self.bb3.lines.bot[0]
        top3 = self.bb3.lines.top[0]

        if not self.position:
            size = self._target_size()
            if size <= 0:
                return
            if close < bot2:
                self.order = self.buy(size=size)
            elif close > top2:
                self.order = self.sell(size=size)
            return

        near_mean = abs(close - mid2) <= self.p.exit_band_frac * (top2 - mid2)

        if self.position.size > 0:
            if close <= bot3:
                self.order = self.close()
            elif self.crossover_mid[0] > 0 or near_mean:
                self.order = self.close()
        elif self.position.size < 0:
            if close >= top3:
                self.order = self.close()
            elif self.crossover_mid[0] < 0 or near_mean:
                self.order = self.close()
