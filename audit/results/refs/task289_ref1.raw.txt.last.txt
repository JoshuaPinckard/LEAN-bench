import backtrader as bt
import math


class MeanReversionStrategy(bt.Strategy):
    def __init__(self):
        self.ema = bt.indicators.EMA(self.data.close, period=20)
        self.bb = bt.indicators.BollingerBands(
            self.data.close, period=20, devfactor=2.0
        )
        self.bb_half = bt.indicators.BollingerBands(
            self.data.close, period=20, devfactor=0.5
        )
        self.atr = bt.indicators.ATR(self.data, period=14)
        self.atr_sma = bt.indicators.SMA(self.atr, period=60)
        self.volume_sma = bt.indicators.SMA(self.data.volume, period=20)

        self.sigma = (self.bb.top - self.bb.mid) / 2.0
        self.strike_minus_1sigma = self.ema - self.sigma
        self.strike_plus_1sigma = self.ema + self.sigma
        self.strike_minus_2sigma = self.ema - 2.0 * self.sigma
        self.strike_plus_2sigma = self.ema + 2.0 * self.sigma

        self.entry_order = None
        self.exit_order = None
        self.stop_order = None
        self.planned_stop = None
        self.entry_bar = None
        self.halted = False
        self.peak_value = None

    def start(self):
        self.peak_value = self.broker.getvalue()

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order is self.entry_order:
            if order.status == order.Completed:
                self.entry_order = None
                self.entry_bar = len(self)

                if self.position.size > 0:
                    self.stop_order = self.sell(
                        size=abs(self.position.size),
                        exectype=bt.Order.Stop,
                        price=self.planned_stop,
                    )
                elif self.position.size < 0:
                    self.stop_order = self.buy(
                        size=abs(self.position.size),
                        exectype=bt.Order.Stop,
                        price=self.planned_stop,
                    )
            elif order.status in [
                order.Canceled, order.Margin, order.Rejected
            ]:
                self.entry_order = None
                self.planned_stop = None

        if order is self.exit_order:
            if order.status in [
                order.Completed, order.Canceled, order.Margin, order.Rejected
            ]:
                if order.status == order.Completed and self.stop_order:
                    self.cancel(self.stop_order)
                    self.entry_bar = None
                self.exit_order = None

        if order is self.stop_order:
            if order.status in [
                order.Completed, order.Canceled, order.Margin, order.Rejected
            ]:
                if order.status == order.Completed:
                    self.entry_bar = None
                self.stop_order = None

    def _exit_position(self):
        if self.position and self.exit_order is None:
            if self.stop_order is not None:
                self.cancel(self.stop_order)
            self.exit_order = self.close()

    def next(self):
        value = self.broker.getvalue()
        self.peak_value = max(self.peak_value, value)

        if (self.peak_value - value) / self.peak_value >= 0.10:
            self.halted = True
            if self.entry_order is not None:
                self.cancel(self.entry_order)
            self._exit_position()
            return

        if self.halted:
            return

        if self.position:
            returned_to_fair = (
                self.bb_half.bot[0] < self.data.close[0] < self.bb_half.top[0]
            )
            time_expired = (
                self.entry_bar is not None and len(self) - self.entry_bar >= 20
            )

            if returned_to_fair or time_expired:
                self._exit_position()
            return

        if (
            self.entry_order is not None
            or self.exit_order is not None
            or self.stop_order is not None
        ):
            return

        if (
            self.atr[0] > 1.5 * self.atr_sma[0]
            or self.data.volume[0] < self.volume_sma[0]
            or abs(self.data.open[0] - self.data.close[-1]) > 1.5 * self.atr[0]
        ):
            return

        crossed_lower_up = (
            self.data.close[-1] < self.bb.bot[-1]
            and self.data.close[0] > self.bb.bot[0]
        )
        crossed_upper_down = (
            self.data.close[-1] > self.bb.top[-1]
            and self.data.close[0] < self.bb.top[0]
        )

        if crossed_lower_up:
            self.planned_stop = self.ema[0] - 2.0 * self.atr[0]
            risk_per_share = self.data.close[0] - self.planned_stop

            if risk_per_share > 0:
                size = math.floor(
                    (self.broker.getvalue() * 0.05) / risk_per_share
                )
                if size > 0:
                    self.entry_order = self.buy(size=size)

        elif crossed_upper_down:
            self.planned_stop = self.ema[0] + 2.0 * self.atr[0]
            risk_per_share = self.planned_stop - self.data.close[0]

            if risk_per_share > 0:
                size = math.floor(
                    (self.broker.getvalue() * 0.05) / risk_per_share
                )
                if size > 0:
                    self.entry_order = self.sell(size=size)
