import backtrader as bt


class NvdaResistanceBreakoutStrategy(bt.Strategy):
    params = dict(
        ema_period=8,
        entry_level=179.33,
        tp1_level=184.89,
        tp2_level=192.95,
        risk_per_trade=0.02,
    )

    def __init__(self):
        # Indicator: 8-period Exponential Moving Average.
        self.ema = bt.indicators.ExponentialMovingAverage(
            self.data.close, period=self.p.ema_period
        )
        self.order = None        # guards pending orders
        self.trail_stop = None   # current trailing-stop price level
        self.tp1_done = False     # first (50%) take-profit already taken

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return
        if order.status in (order.Completed, order.Canceled,
                             order.Margin, order.Rejected):
            self.order = None

    def next(self):
        # Only one working order at a time.
        if self.order:
            return

        # Flat: look for a breakout entry above the resistance level.
        if not self.position:
            self.trail_stop = None
            self.tp1_done = False

            # Entry: closing price of the current bar exceeds 179.33.
            if self.data.close[0] > self.p.entry_level:
                # The trailing-stop distance equals the EMA value, so the
                # potential loss per share (entry - initial stop) == EMA.
                risk_per_share = self.ema[0]
                if risk_per_share > 0:
                    cash_risk = self.p.risk_per_trade * self.broker.getvalue()
                    size = int(cash_risk / risk_per_share)
                    if size >= 1:
                        self.order = self.buy(size=size)
            return

        # In a position: maintain the trailing stop (ratchets upward only,
        # lagging the close by a distance equal to the 8-period EMA value).
        candidate = self.data.close[0] - self.ema[0]
        if self.trail_stop is None:
            self.trail_stop = candidate
        else:
            self.trail_stop = max(self.trail_stop, candidate)

        # Trailing stop hit -> exit the whole position (risk management first).
        if self.data.low[0] <= self.trail_stop:
            self.order = self.close()
            return

        # Final take-profit: close the remaining part upon reaching 192.95.
        if self.tp1_done:
            if self.data.high[0] >= self.p.tp2_level:
                self.order = self.close()
            return

        # First take-profit: close 50% of the position upon reaching 184.89.
        if self.data.high[0] >= self.p.tp1_level:
            half = abs(self.position.size) / 2.0
            if half > 0:
                self.order = self.sell(size=half)
                self.tp1_done = True
