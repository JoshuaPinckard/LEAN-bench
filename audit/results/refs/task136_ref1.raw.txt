import backtrader as bt
import math


class NVDAResistanceEMAStrategy(bt.Strategy):
    params = (("ema_period", 8),)

    def __init__(self):
        self.ema = bt.indicators.EMA(self.data.close, period=self.p.ema_period)
        self.order = None
        self.stop_order = None
        self.order_role = None
        self.stop_level = None
        self.first_take_profit_taken = False
        self.entry_stop_level = None

    def _place_stop(self):
        if not self.position or self.stop_level is None:
            return

        if self.stop_order is not None:
            self.cancel(self.stop_order)

        self.stop_order = self.sell(
            size=self.position.size,
            exectype=bt.Order.Stop,
            price=self.stop_level,
        )

    def _cancel_stop(self):
        if self.stop_order is not None:
            self.cancel(self.stop_order)

    def next(self):
        if self.order is not None:
            return

        if not self.position:
            if self.data.close[0] > 179.33:
                initial_stop = self.data.close[0] - self.ema[0]
                potential_loss_per_share = self.data.close[0] - initial_stop

                if potential_loss_per_share > 0:
                    size = (self.broker.getvalue() * 0.02) / potential_loss_per_share
                    if size > 0:
                        self.entry_stop_level = initial_stop
                        self.order_role = "entry"
                        self.order = self.buy(size=size)
            return

        if self.data.high[0] >= 192.95:
            self._cancel_stop()
            self.order_role = "final_take_profit"
            self.order = self.close()
            return

        if not self.first_take_profit_taken and self.data.high[0] >= 184.89:
            self._cancel_stop()
            self.order_role = "partial_take_profit"
            self.order = self.sell(size=self.position.size * 0.5)
            return

        candidate_stop = self.data.high[0] - self.ema[0]
        if self.stop_level is None or candidate_stop > self.stop_level:
            self.stop_level = candidate_stop
            self._place_stop()

    def notify_order(self, order):
        if order is self.stop_order:
            if order.status in [
                order.Completed,
                order.Canceled,
                order.Margin,
                order.Rejected,
            ]:
                self.stop_order = None
            return

        if order is not self.order:
            return

        if order.status in [order.Canceled, order.Margin, order.Rejected]:
            self.order = None
            self.order_role = None
            return

        if order.status != order.Completed:
            return

        role = self.order_role
        self.order = None
        self.order_role = None

        if role == "entry":
            self.stop_level = self.entry_stop_level
            self.entry_stop_level = None
            self.first_take_profit_taken = False
            self._place_stop()
        elif role == "partial_take_profit":
            self.first_take_profit_taken = True
            if self.position:
                self._place_stop()
        elif role == "final_take_profit":
            self.first_take_profit_taken = False
            self.stop_level = None
