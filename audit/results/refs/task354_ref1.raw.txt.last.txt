import backtrader as bt


class IntradayBreakoutStrategy(bt.Strategy):
    params = (
        ('period', 20),
        ('stop_loss', 0.02),
    )

    def __init__(self):
        self.sma = bt.indicators.SimpleMovingAverage(self.data.close, period=self.p.period)
        self.stddev = bt.indicators.StandardDeviation(self.data.close, period=self.p.period)
        self.upper_band = self.sma + self.stddev
        self.lower_band = self.sma - self.stddev

        self.order = None
        self.stop_order = None
        self.exit_requested = False

    def next(self):
        if self.order:
            return

        if not self.position:
            if self.data.close[-1] <= self.upper_band[-1] and self.data.close[0] > self.upper_band[0]:
                self.order = self.buy()
            elif self.data.close[-1] >= self.lower_band[-1] and self.data.close[0] < self.lower_band[0]:
                self.order = self.sell()
            return

        crosses_sma = (
            (self.position.size > 0 and self.data.close[-1] >= self.sma[-1] and self.data.close[0] < self.sma[0])
            or
            (self.position.size < 0 and self.data.close[-1] <= self.sma[-1] and self.data.close[0] > self.sma[0])
        )

        if crosses_sma:
            if self.stop_order:
                self.exit_requested = True
                self.cancel(self.stop_order)
            else:
                self.order = self.close()

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return

        if order == self.order:
            if order.status == order.Completed and not self.position and self.exit_requested:
                self.exit_requested = False
            elif order.status == order.Completed and self.position:
                if self.position.size > 0:
                    self.stop_order = self.sell(
                        exectype=bt.Order.Stop,
                        price=order.executed.price * (1.0 - self.p.stop_loss),
                    )
                else:
                    self.stop_order = self.buy(
                        exectype=bt.Order.Stop,
                        price=order.executed.price * (1.0 + self.p.stop_loss),
                    )
            if order.status in (order.Completed, order.Canceled, order.Margin, order.Rejected):
                self.order = None

        if order == self.stop_order:
            if order.status == order.Canceled and self.exit_requested:
                self.stop_order = None
                self.exit_requested = False
                self.order = self.close()
            elif order.status in (order.Completed, order.Canceled, order.Margin, order.Rejected):
                self.stop_order = None
