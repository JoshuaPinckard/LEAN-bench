import math
import backtrader as bt


class DailyRSIGatedIntradaySMACross(bt.Strategy):
    params = dict(
        rsi_period=14,
        rsi_threshold=30.0,
        sma_period=20,
        stop_loss_pct=0.02,
        take_profit_pct=0.04,
        position_pct=0.05,
    )

    def __init__(self):
        self.sma20 = bt.indicators.SMA(self.data.close, period=self.p.sma_period)
        self.crossover = bt.indicators.CrossOver(self.data.close, self.sma20)

        self.current_day = None
        self.prev_daily_close = None
        self._gains = []
        self._losses = []
        self.avg_gain = None
        self.avg_loss = None
        self.daily_rsi = None
        self.today_tradable = False

        self.order = None
        self.entry_price = None

    def _update_daily_rsi(self, new_close):
        if self.prev_daily_close is not None:
            change = new_close - self.prev_daily_close
            gain = change if change > 0 else 0.0
            loss = -change if change < 0 else 0.0
            if self.avg_gain is None:
                self._gains.append(gain)
                self._losses.append(loss)
                if len(self._gains) == self.p.rsi_period:
                    self.avg_gain = sum(self._gains) / self.p.rsi_period
                    self.avg_loss = sum(self._losses) / self.p.rsi_period
            else:
                period = self.p.rsi_period
                self.avg_gain = (self.avg_gain * (period - 1) + gain) / period
                self.avg_loss = (self.avg_loss * (period - 1) + loss) / period

            if self.avg_gain is not None:
                if self.avg_loss == 0:
                    self.daily_rsi = 100.0
                else:
                    rs = self.avg_gain / self.avg_loss
                    self.daily_rsi = 100.0 - 100.0 / (1.0 + rs)
        self.prev_daily_close = new_close

    def next(self):
        bar_dt = self.data.datetime.datetime(0)
        bar_date = bar_dt.date()

        if self.current_day is None:
            self.current_day = bar_date
        elif bar_date != self.current_day:
            self._update_daily_rsi(self.data.close[-1])
            self.today_tradable = (
                self.daily_rsi is not None and self.daily_rsi < self.p.rsi_threshold
            )
            self.current_day = bar_date

        if self.order:
            return

        if self.position:
            exit_signal = False
            if self.crossover[0] < 0:
                exit_signal = True
            if (bar_dt.hour, bar_dt.minute) >= (15, 45):
                exit_signal = True
            if self.entry_price is not None:
                price = self.data.close[0]
                if price <= self.entry_price * (1.0 - self.p.stop_loss_pct):
                    exit_signal = True
                if price >= self.entry_price * (1.0 + self.p.take_profit_pct):
                    exit_signal = True
            if exit_signal:
                self.order = self.close()
        else:
            if self.today_tradable and self.crossover[0] > 0:
                value = self.broker.getvalue()
                size = math.floor((value * self.p.position_pct) / self.data.close[0])
                if size > 0:
                    self.order = self.buy(size=size)

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return
        if order.status == order.Completed:
            if order.isbuy():
                self.entry_price = order.executed.price
            else:
                self.entry_price = None
        self.order = None
