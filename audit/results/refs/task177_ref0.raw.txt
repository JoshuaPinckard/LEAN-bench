import backtrader as bt


class LowIVGapEntryStrategy(bt.Strategy):
    params = dict(
        bb_period=20,
        bb_devfactor=2.0,
        ivr_period=100,
        ivr_threshold=20.0,
        gap_threshold=-0.004,
    )

    def __init__(self):
        self.bb = bt.indicators.BollingerBands(
            self.data, period=self.p.bb_period, devfactor=self.p.bb_devfactor
        )
        self.bandwidth = (self.bb.top - self.bb.bot) / self.bb.mid
        self.hi_bw = bt.indicators.Highest(self.bandwidth, period=self.p.ivr_period)
        self.lo_bw = bt.indicators.Lowest(self.bandwidth, period=self.p.ivr_period)
        self.order = None
        # Same-day open entry / close exit is otherwise unreachable: default
        # backtrader order matching defers fills to the following bar.
        self.broker.set_coo(True)
        self.broker.set_coc(True)

    def notify_order(self, order):
        if order.status in (order.Completed, order.Canceled, order.Margin, order.Rejected):
            self.order = None

    def next_open(self):
        if self.order is not None or self.position:
            return
        hi = self.hi_bw[0]
        lo = self.lo_bw[0]
        bw = self.bandwidth[0]
        iv_rank = 0.0 if hi == lo else (bw - lo) / (hi - lo) * 100.0
        if iv_rank > self.p.ivr_threshold:
            return
        gap = (self.data.open[0] - self.data.close[-1]) / self.data.close[-1]
        if gap < self.p.gap_threshold:
            return
        self.order = self.buy()

    def next(self):
        if self.position and self.order is None:
            self.order = self.close()
