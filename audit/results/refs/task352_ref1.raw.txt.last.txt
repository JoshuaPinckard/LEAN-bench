import backtrader as bt


class IntradayTrendStrategy(bt.Strategy):
    def __init__(self):
        self.sma = bt.indicators.SimpleMovingAverage(self.data.close, period=20)
        self.atr = bt.indicators.AverageTrueRange(self.data, period=14)
        self.upper_band = self.sma + (2 * self.atr)
        self.lower_band = self.sma - (2 * self.atr)
        self.order = None

    def notify_order(self, order):
        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            self.order = None

    def next(self):
        if self.order:
            return

        close = self.data.close[0]

        if not self.position:
            if close > self.upper_band[0]:
                self.order = self.buy()
            elif close < self.lower_band[0]:
                self.order = self.sell()
            return

        if self.position.size > 0:
            crossed_baseline = (
                self.data.close[-1] >= self.sma[-1] and close < self.sma[0]
            )
            stop_hit = self.data.low[0] <= self.lower_band[0]

            if crossed_baseline or stop_hit:
                self.order = self.close()

        elif self.position.size < 0:
            crossed_baseline = (
                self.data.close[-1] <= self.sma[-1] and close > self.sma[0]
            )
            stop_hit = self.data.high[0] >= self.upper_band[0]

            if crossed_baseline or stop_hit:
                self.order = self.close()
