import backtrader as bt


class DeepDipLeaderPyramidingStrategy(bt.Strategy):
    def __init__(self):
        self.sma200 = bt.indicators.SMA(self.data.close, period=200)
        self.ema20 = bt.indicators.EMA(self.data.close, period=20)
        self.ema10 = bt.indicators.EMA(self.data.close, period=10)
        self.rsi = bt.indicators.RSI(self.data.close, period=14)
        self.macd = bt.indicators.MACD(
            self.data.close, period_me1=12, period_me2=26, period_signal=9
        )
        self.bb = bt.indicators.BollingerBands(self.data.close, period=20, devfactor=2)
        self.atr = bt.indicators.ATR(self.data, period=14)
        self.h252 = bt.indicators.Highest(self.data.close, period=252)

        self.cross_close_ema20 = bt.indicators.CrossOver(self.data.close, self.ema20)
        self.cross_rsi_30 = bt.indicators.CrossOver(self.rsi, 30)
        self.cross_macd_signal = bt.indicators.CrossOver(self.macd.macd, self.macd.signal)

        self.order = None
        self.order_type = None

        self.entry_price = None
        self.initial_stop = None
        self.highest_since_entry = None
        self.growth_phase = False
        self.partial_taken = False
        self.below_ema20_count = 0
        self.add_count = 0
        self.exit_bar = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status == order.Completed:
            if order.isbuy():
                if self.order_type == 'entry':
                    self.entry_price = order.executed.price
                    self.initial_stop = self.entry_price - 1.5 * self.atr[0]
                    self.highest_since_entry = self.data.high[0]
                    self.add_count = 0
                    self.growth_phase = False
                    self.partial_taken = False
                    self.below_ema20_count = 0
                elif self.order_type == 'add':
                    self.add_count += 1
            elif order.issell():
                if self.order_type == 'partial_exit':
                    self.partial_taken = True
                elif self.order_type in ('full_exit', 'stop_exit', 'trend_exit'):
                    self.exit_bar = len(self)
                    self.entry_price = None
                    self.initial_stop = None
                    self.highest_since_entry = None
                    self.growth_phase = False
                    self.partial_taken = False
                    self.add_count = 0
                    self.below_ema20_count = 0

        self.order = None
        self.order_type = None

    def _check_entry(self):
        crash = self.data.close[0] <= 0.30 * self.h252[0]
        if not crash:
            return

        cond_a = self.cross_close_ema20[0] > 0 and self.cross_rsi_30[0] > 0

        touch_lower_bb = self.data.low[0] <= self.bb.bot[0]
        close_above_lower_bb = self.data.close[0] > self.bb.bot[0]
        cond_b = touch_lower_bb and close_above_lower_bb and self.rsi[0] < 35

        if cond_a or cond_b:
            stop_distance = 1.5 * self.atr[0]
            if stop_distance <= 0:
                return
            risk_amount = 0.01 * self.broker.getvalue()
            size = risk_amount / stop_distance
            self.order = self.buy(size=size)
            self.order_type = 'entry'

    def _manage_position(self):
        if not self.growth_phase and self.data.close[0] > self.sma200[0] and self.macd.macd[0] > 0:
            self.growth_phase = True

        if self.data.high[0] > self.highest_since_entry:
            self.highest_since_entry = self.data.high[0]

        if self.data.close[0] < self.ema20[0]:
            self.below_ema20_count += 1
        else:
            self.below_ema20_count = 0

        if self.growth_phase:
            stop_price = self.highest_since_entry - 3 * self.atr[0]
        else:
            stop_price = self.initial_stop

        if self.data.low[0] <= stop_price:
            self.order = self.close()
            self.order_type = 'stop_exit'
            return

        trend_exit = (self.cross_macd_signal[0] < 0) or (self.below_ema20_count >= 2)
        if trend_exit:
            self.order = self.close()
            self.order_type = 'trend_exit'
            return

        avg_price = self.position.price

        if not self.partial_taken and self.data.close[0] >= avg_price * 1.5:
            size_to_sell = abs(self.position.size) / 2
            self.order = self.sell(size=size_to_sell)
            self.order_type = 'partial_exit'
            return

        touch_upper_bb = self.data.high[0] >= self.bb.top[0]
        full_profit = self.data.close[0] >= avg_price * 2.0 or (touch_upper_bb and self.rsi[0] > 75)
        if full_profit:
            self.order = self.close()
            self.order_type = 'full_exit'
            return

        if self.add_count < 3:
            phase2_active = self.data.close[0] > self.sma200[0] and self.macd.macd[0] > 0
            pullback = (self.data.low[0] <= self.ema20[0]) or (self.data.low[0] <= self.bb.bot[0])
            rsi_range = 45 <= self.rsi[0] <= 60
            unrealized_profit = self.data.close[0] > avg_price
            if phase2_active and pullback and rsi_range and unrealized_profit:
                self.order = self.buy()
                self.order_type = 'add'

    def next(self):
        if self.order:
            return

        if self.position:
            self._manage_position()
        else:
            if self.exit_bar is not None and (len(self) - self.exit_bar) < 30:
                return
            self._check_entry()
