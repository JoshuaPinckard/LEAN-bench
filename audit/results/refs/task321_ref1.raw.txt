import backtrader as bt


class RSIFailureSwingStrategy(bt.Strategy):
    params = (("rsi_period", 14),)

    def __init__(self):
        self.rsi = bt.indicators.RSI(self.data.close, period=self.p.rsi_period)

        self.order = None
        self.stop_order = None
        self.entry_price = None

        self.long_state = "seek_first"
        self.rsi_low1 = None
        self.long_candidate = None

        self.short_state = "seek_first"
        self.rsi_high1 = None
        self.short_candidate = None

    def _reset_setups(self):
        self.long_state = "seek_first"
        self.rsi_low1 = None
        self.long_candidate = None
        self.short_state = "seek_first"
        self.rsi_high1 = None
        self.short_candidate = None

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return

        role = getattr(order.info, "role", None)

        if order.status == order.Completed:
            if role == "entry":
                self.order = None
                self.entry_price = order.executed.price
                self._reset_setups()

                if order.isbuy():
                    self.stop_order = self.sell(
                        exectype=bt.Order.Stop,
                        price=self.entry_price * 0.98,
                    )
                else:
                    self.stop_order = self.buy(
                        exectype=bt.Order.Stop,
                        price=self.entry_price * 1.02,
                    )
                self.stop_order.info.role = "stop"

            elif role == "exit":
                self.order = None
                if self.stop_order is not None:
                    self.cancel(self.stop_order)
                self.stop_order = None
                self.entry_price = None
                self._reset_setups()

            elif role == "stop":
                self.stop_order = None
                self.entry_price = None
                self._reset_setups()

        elif order.status in (
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            if order is self.order:
                self.order = None
            if order is self.stop_order:
                self.stop_order = None

    def next(self):
        if len(self) < 2:
            return

        rsi_now = self.rsi[0]
        rsi_prev = self.rsi[-1]

        if rsi_now != rsi_now or rsi_prev != rsi_prev:
            return

        if self.order is not None:
            return

        if self.position:
            if self.position.size > 0 and rsi_prev <= 70 and rsi_now > 70:
                if self.stop_order is not None:
                    self.cancel(self.stop_order)
                self.order = self.close()
                self.order.info.role = "exit"

            elif self.position.size < 0 and rsi_prev >= 30 and rsi_now < 30:
                if self.stop_order is not None:
                    self.cancel(self.stop_order)
                self.order = self.close()
                self.order.info.role = "exit"
            return

        if self.long_state == "seek_first":
            if rsi_now < 30:
                self.long_candidate = rsi_now
                self.long_state = "first_oversold"

        elif self.long_state == "first_oversold":
            if rsi_now < 30:
                self.long_candidate = min(self.long_candidate, rsi_now)
            elif rsi_prev <= 30 and rsi_now > 30:
                self.rsi_low1 = self.long_candidate
                self.long_candidate = None
                self.long_state = "seek_second"

        elif self.long_state == "seek_second":
            if rsi_now < 30:
                self.long_candidate = rsi_now
                self.long_state = "second_oversold"

        elif self.long_state == "second_oversold":
            if rsi_now < 30:
                self.long_candidate = min(self.long_candidate, rsi_now)
            elif rsi_prev <= 30 and rsi_now > 30:
                rsi_low2 = self.long_candidate
                self.long_candidate = None
                if rsi_low2 > self.rsi_low1:
                    self.order = self.buy()
                    self.order.info.role = "entry"
                else:
                    self.rsi_low1 = rsi_low2
                    self.long_state = "seek_second"

        if self.short_state == "seek_first":
            if rsi_now > 70:
                self.short_candidate = rsi_now
                self.short_state = "first_overbought"

        elif self.short_state == "first_overbought":
            if rsi_now > 70:
                self.short_candidate = max(self.short_candidate, rsi_now)
            elif rsi_prev >= 70 and rsi_now < 70:
                self.rsi_high1 = self.short_candidate
                self.short_candidate = None
                self.short_state = "seek_second"

        elif self.short_state == "seek_second":
            if rsi_now > 70:
                self.short_candidate = rsi_now
                self.short_state = "second_overbought"

        elif self.short_state == "second_overbought":
            if rsi_now > 70:
                self.short_candidate = max(self.short_candidate, rsi_now)
            elif rsi_prev >= 70 and rsi_now < 70:
                rsi_high2 = self.short_candidate
                self.short_candidate = None
                if rsi_high2 < self.rsi_high1:
                    self.order = self.sell()
                    self.order.info.role = "entry"
                else:
                    self.rsi_high1 = rsi_high2
                    self.short_state = "seek_second"
