import backtrader as bt


class HmaAtrStopStrategy(bt.Strategy):
    params = (
        ('hma_period', 9),
        ('atr_period', 14),
    )

    def __init__(self):
        self.hma = bt.indicators.HullMovingAverage(
            self.data.close, period=self.p.hma_period
        )
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)

        self.entry_order = None
        self.stop_order = None
        self.pending_stop_price = None
        self.pending_direction = None

    def next(self):
        if self.position or self.entry_order or self.stop_order:
            return

        was_falling = self.hma[-1] < self.hma[-2]
        is_rising = self.hma[0] > self.hma[-1]
        was_rising = self.hma[-1] > self.hma[-2]
        is_falling = self.hma[0] < self.hma[-1]

        if was_falling and is_rising:
            self.pending_direction = 'long'
            self.pending_stop_price = self.data.low[0] - (2 * self.atr[0])
            self.entry_order = self.buy()

        elif was_rising and is_falling:
            self.pending_direction = 'short'
            self.pending_stop_price = self.data.high[0] + (2 * self.atr[0])
            self.entry_order = self.sell()

    def notify_order(self, order):
        if order.status in (
            order.Submitted,
            order.Accepted,
        ):
            return

        if order == self.entry_order:
            if order.status == order.Completed:
                if self.pending_direction == 'long':
                    self.stop_order = self.close(
                        exectype=bt.Order.Stop,
                        price=self.pending_stop_price,
                    )
                else:
                    self.stop_order = self.close(
                        exectype=bt.Order.Stop,
                        price=self.pending_stop_price,
                    )

            self.entry_order = None
            self.pending_stop_price = None
            self.pending_direction = None

        elif order == self.stop_order:
            if order.status in (
                order.Completed,
                order.Canceled,
                order.Margin,
                order.Rejected,
            ):
                self.stop_order = None
