import backtrader as bt


class BreakoutStrategy(bt.Strategy):
    params = (("stochastic_period", 260),)

    def __init__(self):
        self.stochastic = bt.indicators.StochasticFast(
            self.data,
            period=self.p.stochastic_period,
            period_d=1,
        )
        self.order = None

    def notify_order(self, order):
        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            self.order = None

    def next(self):
        if self.order:
            return

        percent_k = self.stochastic.percK[0]
        previous_percent_k = self.stochastic.percK[-1]

        if not self.position:
            if percent_k > 85 and previous_percent_k <= 85:
                self.order = self.order_target_percent(target=1.0)
        elif percent_k < 1:
            self.order = self.close()
