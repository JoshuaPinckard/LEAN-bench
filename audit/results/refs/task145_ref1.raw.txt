import backtrader as bt


class RecentStrongPriceMoveDetectionStrategy(bt.Strategy):
    params = (
        ("price_change_threshold", 20.0),
        ("period", 10),
        ("holding_period", 5),
        ("volume_dollar_threshold", 1000000.0),
        ("stop_loss_pct", 0.05),
        ("take_profit_pct", 0.10),
    )

    def __init__(self):
        self.avg_volume = bt.indicators.SMA(self.data.volume, period=self.p.period)
        self.order = None
        self.stop_order = None
        self.take_profit_order = None
        self.entry_bar = None

    def next(self):
        if self.order:
            return

        if self.position:
            if self.entry_bar is not None and len(self) >= self.entry_bar + self.p.holding_period:
                if self.stop_order:
                    self.cancel(self.stop_order)
                if self.take_profit_order:
                    self.cancel(self.take_profit_order)
                self.order = self.close()
            return

        period_open_price = self.data.open[-(self.p.period - 1)]
        percentage_change = (
            (self.data.close[0] / period_open_price - 1.0) * 100.0
        )
        average_daily_dollar_volume = self.avg_volume[0] * self.data.close[0]

        if (
            abs(percentage_change) > self.p.price_change_threshold
            and average_daily_dollar_volume > self.p.volume_dollar_threshold
        ):
            available_cash = self.broker.getcash()
            size = int(available_cash / (self.data.close[0] * 1.001))
            if size > 0:
                self.order = self.buy(size=size)

    def notify_order(self, order):
        if order.status in (
            order.Submitted,
            order.Accepted,
        ):
            return

        if order.status == order.Completed:
            if order.isbuy():
                self.entry_bar = len(self)
                entry_price = order.executed.price

                self.stop_order = self.sell(
                    size=order.executed.size,
                    exectype=bt.Order.Stop,
                    price=entry_price * (1.0 - self.p.stop_loss_pct),
                )
                self.take_profit_order = self.sell(
                    size=order.executed.size,
                    exectype=bt.Order.Limit,
                    price=entry_price * (1.0 + self.p.take_profit_pct),
                    oco=self.stop_order,
                )
            else:
                self.entry_bar = None
                self.stop_order = None
                self.take_profit_order = None

        if order.status in (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            if order == self.order:
                self.order = None
            if order == self.stop_order:
                self.stop_order = None
            if order == self.take_profit_order:
                self.take_profit_order = None
