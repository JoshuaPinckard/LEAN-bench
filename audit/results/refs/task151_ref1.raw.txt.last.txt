import backtrader as bt


class StreamingTickAnomalyStrategy(bt.Strategy):
    params = (
        ("sma_period", 10),
        ("rsi_period", 14),
        ("volume_period", 50),
    )

    def __init__(self):
        self.sma = bt.indicators.SimpleMovingAverage(
            self.data.close, period=self.p.sma_period
        )
        self.rsi = bt.indicators.RSI(self.data.close, period=self.p.rsi_period)
        self.average_volume = bt.indicators.SimpleMovingAverage(
            self.data.volume, period=self.p.volume_period
        )
        self.cross_down = bt.indicators.CrossDown(self.data.close, self.sma)
        self.cross_up = bt.indicators.CrossUp(self.data.close, self.sma)

        self.order = None
        self.active_positions = 0
        self.last_entry_tick = None
        self.closing_position = False

    def notify_order(self, order):
        if order.status == order.Completed:
            if order.isbuy() or order.issell():
                if self.closing_position:
                    self.active_positions = 0
                    self.closing_position = False
                elif not order.info.get("is_close", False):
                    self.active_positions += 1
                    self.last_entry_tick = len(self)

            self.order = None

        elif order.status in (order.Canceled, order.Margin, order.Rejected):
            self.order = None
            if self.closing_position:
                self.closing_position = False

    def _close_position(self):
        self.closing_position = True
        self.order = self.close()
        if self.order is not None:
            self.order.addinfo(is_close=True)

    def next(self):
        if self.order:
            return

        price = self.data.close[0]
        sma = self.sma[0]
        rsi = self.rsi[0]
        volume = self.data.volume[0]
        average_volume = self.average_volume[0]

        if self.position.size:
            entry_price = self.position.price

            if self.position.size > 0:
                if (
                    price <= entry_price * (1.0 - 0.005)
                    or price >= entry_price * (1.0 + 0.008)
                    or self.cross_down[0] < 0
                    or rsi < 30
                    or rsi > 70
                ):
                    self._close_position()

            else:
                if (
                    price >= entry_price * (1.0 + 0.005)
                    or price <= entry_price * (1.0 - 0.008)
                    or self.cross_up[0] > 0
                    or rsi < 30
                    or rsi > 70
                ):
                    self._close_position()

            return

        dt = self.data.datetime.datetime(0)
        if dt.hour == 9 and 30 <= dt.minute < 35:
            return

        if self.active_positions >= 3:
            return

        if self.last_entry_tick is not None and len(self) - self.last_entry_tick < 3:
            return

        volume_confirmed = volume > average_volume * 1.20
        rsi_neutral = 40 <= rsi <= 60
        size = (self.broker.getvalue() * 0.01) / price

        if price > sma * 1.001 and rsi_neutral and volume_confirmed:
            self.order = self.buy(size=size)

        elif price < sma * 0.999 and rsi_neutral and volume_confirmed:
            self.order = self.sell(size=size)
