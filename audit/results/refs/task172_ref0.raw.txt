import backtrader as bt


class RsiDivergenceBreakoutStrategy(bt.Strategy):
    params = dict(
        rsi_period=14,
        vol_sma_period=20,
        breakout_lookback=20,
        exit_lookback=10,
        pivot_width=2,
        stop_loss_pct=0.02,
        take_profit_pct=0.06,
        risk_pct=0.01,
        agg_bars=4,
    )

    def __init__(self):
        # --- synthetic 4H bar aggregation state (built from the 1h feed) ---
        self._agg_count = 0
        self._agg_high = None
        self._agg_low = None
        self._agg_close = None
        self._agg_volume = 0.0

        # --- confirmed 4H series ---
        self.h4_close = []
        self.h4_high = []
        self.h4_low = []
        self.h4_volume = []
        self.h4_rsi = []

        # --- Wilder RSI(14) state ---
        self._prev_close = None
        self._gains = []
        self._losses = []
        self._avg_gain = None
        self._avg_loss = None

        # --- pivot / divergence state ---
        self._confirmed_lows = []
        self._confirmed_highs = []
        self.bullish_div_active = False
        self.bearish_div_active = False

        # --- order tracking (canonical guard) ---
        self.entry_order = None
        self.stop_order = None
        self.limit_order = None
        self.exit_order = None
        self.entry_price = None

    def _update_rsi(self, close):
        if self._prev_close is None:
            self._prev_close = close
            self.h4_rsi.append(None)
            return
        change = close - self._prev_close
        self._prev_close = close
        gain = change if change > 0 else 0.0
        loss = -change if change < 0 else 0.0
        n = self.p.rsi_period

        if self._avg_gain is None:
            self._gains.append(gain)
            self._losses.append(loss)
            if len(self._gains) < n:
                self.h4_rsi.append(None)
                return
            self._avg_gain = sum(self._gains) / n
            self._avg_loss = sum(self._losses) / n
        else:
            self._avg_gain = (self._avg_gain * (n - 1) + gain) / n
            self._avg_loss = (self._avg_loss * (n - 1) + loss) / n

        if self._avg_loss == 0:
            rsi = 100.0
        else:
            rs = self._avg_gain / self._avg_loss
            rsi = 100.0 - (100.0 / (1.0 + rs))
        self.h4_rsi.append(rsi)

    def _finalize_h4_bar(self):
        self.h4_close.append(self._agg_close)
        self.h4_high.append(self._agg_high)
        self.h4_low.append(self._agg_low)
        self.h4_volume.append(self._agg_volume)
        self._update_rsi(self._agg_close)
        self._agg_volume = 0.0

    def _process_new_h4_bar(self):
        n = len(self.h4_close)
        w = self.p.pivot_width
        candidate = n - 1 - w

        if candidate >= w:
            lows = self.h4_low
            highs = self.h4_high

            is_low = all(lows[candidate] < lows[candidate - k] for k in range(1, w + 1)) and \
                     all(lows[candidate] < lows[candidate + k] for k in range(1, w + 1))
            if is_low:
                rsi_val = self.h4_rsi[candidate]
                self._confirmed_lows.append((candidate, lows[candidate], rsi_val))
                if len(self._confirmed_lows) >= 2:
                    prev_pt = self._confirmed_lows[-2]
                    curr_pt = self._confirmed_lows[-1]
                    if prev_pt[2] is not None and curr_pt[2] is not None and \
                       curr_pt[1] < prev_pt[1] and curr_pt[2] > prev_pt[2]:
                        self.bullish_div_active = True
                    else:
                        self.bullish_div_active = False

            is_high = all(highs[candidate] > highs[candidate - k] for k in range(1, w + 1)) and \
                      all(highs[candidate] > highs[candidate + k] for k in range(1, w + 1))
            if is_high:
                rsi_val = self.h4_rsi[candidate]
                self._confirmed_highs.append((candidate, highs[candidate], rsi_val))
                if len(self._confirmed_highs) >= 2:
                    prev_pt = self._confirmed_highs[-2]
                    curr_pt = self._confirmed_highs[-1]
                    if prev_pt[2] is not None and curr_pt[2] is not None and \
                       curr_pt[1] > prev_pt[1] and curr_pt[2] < prev_pt[2]:
                        self.bearish_div_active = True
                    else:
                        self.bearish_div_active = False

        # --- alternative exits (checked on 4H closes) ---
        if self.position and self.exit_order is None:
            breakdown = False
            if n > self.p.exit_lookback:
                support = min(self.h4_low[-(self.p.exit_lookback + 1):-1])
                breakdown = self.h4_close[-1] < support
            if self.bearish_div_active or breakdown:
                if self.stop_order is not None:
                    self.cancel(self.stop_order)
                if self.limit_order is not None:
                    self.cancel(self.limit_order)
                self.exit_order = self.close()
                self.bearish_div_active = False

        # --- entry: divergence + breakout + volume confirmation ---
        if not self.position and self.entry_order is None and self.exit_order is None:
            if self.bullish_div_active and n > self.p.breakout_lookback and \
               len(self.h4_volume) >= self.p.vol_sma_period:
                current_close = self.h4_close[-1]
                resistance = max(self.h4_high[-(self.p.breakout_lookback + 1):-1])
                vol_avg = sum(self.h4_volume[-self.p.vol_sma_period:]) / self.p.vol_sma_period
                current_vol = self.h4_volume[-1]
                if current_close > resistance and current_vol > vol_avg:
                    dt = self.data.datetime.date(0)
                    if dt.year == 2025 and dt.month in (11, 12):
                        entry_price_est = current_close
                        risk_amount = self.broker.getvalue() * self.p.risk_pct
                        risk_per_share = self.p.stop_loss_pct * entry_price_est
                        size = risk_amount / risk_per_share
                        if size > 0:
                            self.entry_order = self.buy(size=size)
                            self.bullish_div_active = False

    def next(self):
        o = self.data.open[0]
        h = self.data.high[0]
        l = self.data.low[0]
        c = self.data.close[0]
        v = self.data.volume[0]

        if self._agg_count == 0:
            self._agg_high = h
            self._agg_low = l
        else:
            self._agg_high = max(self._agg_high, h)
            self._agg_low = min(self._agg_low, l)
        self._agg_close = c
        self._agg_volume += v
        self._agg_count += 1

        if self._agg_count == self.p.agg_bars:
            self._finalize_h4_bar()
            self._agg_count = 0
            self._process_new_h4_bar()

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return

        if order.status == order.Completed:
            if order == self.entry_order:
                self.entry_price = order.executed.price
                self.entry_order = None
                stop_price = self.entry_price * (1 - self.p.stop_loss_pct)
                limit_price = self.entry_price * (1 + self.p.take_profit_pct)
                self.stop_order = self.sell(
                    exectype=bt.Order.Stop, price=stop_price, size=self.position.size
                )
                self.limit_order = self.sell(
                    exectype=bt.Order.Limit, price=limit_price, size=self.position.size,
                    oco=self.stop_order
                )
            elif order in (self.stop_order, self.limit_order):
                self.stop_order = None
                self.limit_order = None
                self.exit_order = None
                self.entry_price = None
            elif order == self.exit_order:
                self.exit_order = None
                self.stop_order = None
                self.limit_order = None
                self.entry_price = None

        elif order.status in (order.Canceled, order.Margin, order.Rejected):
            if order == self.entry_order:
                self.entry_order = None
            if order == self.stop_order:
                self.stop_order = None
            if order == self.limit_order:
                self.limit_order = None
            if order == self.exit_order:
                self.exit_order = None
