import backtrader as bt


class IntradayDeviationFromOpen(bt.Strategy):
    params = (
        ('deviation_bps', 50),
    )

    def __init__(self):
        self.current_day = None
        self.opening_price = None
        self.traded_today = False
        self.order = None

    def notify_order(self, order):
        if order.status in (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            self.order = None

    def next(self):
        bar_day = self.data.datetime.date(0)

        if self.current_day is not None and bar_day != self.current_day:
            if self.position and self.order is None:
                self.order = self.close()
            self.opening_price = self.data.open[0]
            self.current_day = bar_day
            self.traded_today = False
        elif self.current_day is None:
            self.current_day = bar_day
            self.opening_price = self.data.open[0]

        if self.position or self.order is not None or self.traded_today:
            return

        deviation = self.p.deviation_bps / 10000.0
        close_price = self.data.close[0]

        if close_price < self.opening_price * (1 - deviation):
            self.order = self.buy()
            self.traded_today = True
        elif close_price > self.opening_price * (1 + deviation):
            self.order = self.sell()
            self.traded_today = True

    def stop(self):
        if self.position and self.order is None:
            self.order = self.close()
