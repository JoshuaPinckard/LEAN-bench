import backtrader as bt


class ATHExitStrategy(bt.Strategy):
    params = (
        ('sma_fast_period', 20),
        ('sma_slow_period', 50),
        ('rsi_period', 14),
        ('volume_period', 20),
    )

    def __init__(self):
        self.sma20 = bt.indicators.SimpleMovingAverage(
            self.data.close, period=self.p.sma_fast_period
        )
        self.sma50 = bt.indicators.SimpleMovingAverage(
            self.data.close, period=self.p.sma_slow_period
        )
        self.rsi = bt.indicators.RSI(self.data.close, period=self.p.rsi_period)
        self.volume_sma20 = bt.indicators.SimpleMovingAverage(
            self.data.volume, period=self.p.volume_period
        )

        self.order = None
        self.ath = None
        self.partial_taken = False

        self.in_overbought = False
        self.current_rsi_peak = None
        self.current_peak_price = None
        self.previous_rsi_peak = None
        self.previous_peak_price = None

    def notify_order(self, order):
        if order.status in (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            self.order = None

    def next(self):
        close = self.data.close[0]
        rsi = self.rsi[0]

        if self.ath is None or close > self.ath:
            is_new_ath = self.ath is not None
            self.ath = close
        else:
            is_new_ath = False

        if rsi > 70:
            if not self.in_overbought:
                self.in_overbought = True
                self.current_rsi_peak = rsi
                self.current_peak_price = close
            elif rsi > self.current_rsi_peak:
                self.current_rsi_peak = rsi
                self.current_peak_price = close

        bearish_divergence = False
        if self.in_overbought and rsi < 70:
            if (
                self.previous_rsi_peak is not None
                and self.current_peak_price > self.previous_peak_price
                and self.current_rsi_peak < self.previous_rsi_peak
            ):
                bearish_divergence = True

            self.previous_rsi_peak = self.current_rsi_peak
            self.previous_peak_price = self.current_peak_price
            self.in_overbought = False
            self.current_rsi_peak = None
            self.current_peak_price = None

        if not self.position or self.order:
            return

        volume_confirmed = self.data.volume[0] > (
            1.20 * self.volume_sma20[0]
        )

        if (
            volume_confirmed
            and (
                close < self.sma20[0]
                or close < self.sma50[0]
                or bearish_divergence
            )
        ):
            self.order = self.close()
            return

        if is_new_ath and not self.partial_taken:
            self.order = self.sell(size=abs(self.position.size) * 0.50)
            self.partial_taken = True
