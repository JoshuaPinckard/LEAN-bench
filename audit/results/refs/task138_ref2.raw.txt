import backtrader as bt


class RSIStrategy(bt.Strategy):
    params = (
        ('rsi_period', 14),
        ('oversold', 30),
        ('overbought', 70),
    )

    def __init__(self):
        self.rsi = bt.indicators.RSI(self.data.close, period=self.p.rsi_period)
        self.order = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        if self.order:
            return

        if not self.position:
            # Long Entry: RSI drops below 30 and then rises back above 30
            if self.rsi[-1] < self.p.oversold and self.rsi[0] > self.p.oversold:
                price = self.data.close[0]
                cash = self.broker.getcash()
                size = int(cash / (price * 1.001))
                if size > 0:
                    self.order = self.buy(size=size)

            # Short Entry: RSI rises above 70 and then falls back below 70
            elif self.rsi[-1] > self.p.overbought and self.rsi[0] < self.p.overbought:
                price = self.data.close[0]
                value = self.broker.getvalue()
                size = int(value / (price * 1.001))
                if size > 0:
                    self.order = self.sell(size=size)
        else:
            # Close Long Position when RSI rises above 70
            if self.position.size > 0:
                if self.rsi[0] > self.p.overbought:
                    self.order = self.close()

            # Close Short Position when RSI drops below 30
            elif self.position.size < 0:
                if self.rsi[0] < self.p.oversold:
                    self.order = self.close()
