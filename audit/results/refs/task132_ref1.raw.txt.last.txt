import backtrader as bt
import math
from collections import deque


class SystematicTradingWithLimitAndStopOrders(bt.Strategy):
    def __init__(self):
        self._daily_closes = deque(maxlen=21)
        self._current_day = None
        self._entry_ref = None

    def next(self):
        current_day = self.data.datetime.date(0)

        if self._current_day is None:
            self._current_day = current_day
            return

        if current_day == self._current_day:
            return

        # At the first hourly bar of day t+1, day t is complete.
        previous_close = self.data.close[-1]
        self._daily_closes.append(previous_close)
        self._current_day = current_day

        if self.position or self._entry_ref is not None:
            return

        if len(self._daily_closes) < 21:
            return

        closes = list(self._daily_closes)
        previous_sma = sum(closes[:-1]) / 20.0
        current_sma = sum(closes[-20:]) / 20.0
        prior_close = closes[-2]
        signal_close = closes[-1]

        valid_until = math.floor(self.data.datetime[0]) + 0.999999

        if prior_close < previous_sma and signal_close > current_sma:
            order = self.buy(
                exectype=bt.Order.Limit,
                price=signal_close,
                valid=valid_until,
            )
            self._entry_ref = order.ref

        elif prior_close > previous_sma and signal_close < current_sma:
            order = self.sell(
                exectype=bt.Order.Limit,
                price=self.data.open[0],
                valid=valid_until,
            )
            self._entry_ref = order.ref

    def notify_order(self, order):
        if order.status in (
            bt.Order.Completed,
            bt.Order.Canceled,
            bt.Order.Margin,
            bt.Order.Rejected,
        ):
            if order.ref == self._entry_ref:
                self._entry_ref = None

        if order.status != bt.Order.Completed:
            return

        if order.ref != self._entry_ref and order.isbuy() and self.position.size > 0:
            return

        if order.ref != self._entry_ref and order.issell() and self.position.size < 0:
            return

        if order.ref == self._entry_ref:
            return

        entry_price = order.executed.price
        size = abs(order.executed.size)

        if order.isbuy():
            take_profit = self.sell(
                size=size,
                exectype=bt.Order.Limit,
                price=entry_price * 1.02,
            )
            self.sell(
                size=size,
                exectype=bt.Order.Stop,
                price=entry_price * 0.985,
                oco=take_profit,
            )
        else:
            take_profit = self.buy(
                size=size,
                exectype=bt.Order.Limit,
                price=entry_price * 0.98,
            )
            self.buy(
                size=size,
                exectype=bt.Order.Stop,
                price=entry_price * 1.015,
                oco=take_profit,
            )
