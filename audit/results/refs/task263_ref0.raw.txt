import backtrader as bt


class QuantValueStrategy(bt.Strategy):
    def __init__(self):
        self.sma50 = bt.indicators.SMA(self.data.close, period=50)
        self.sma200 = bt.indicators.SMA(self.data.close, period=200)
        self.crossover = bt.indicators.CrossOver(self.sma50, self.sma200)
        self.bb = bt.indicators.BollingerBands(self.data.close, period=20, devfactor=2)
        self.rsi = bt.indicators.RSI(self.data.close, period=14)
        self.vol_sma20 = bt.indicators.SMA(self.data.volume, period=20)
        self.vol_sma50 = bt.indicators.SMA(self.data.volume, period=50)
        self.atr = bt.indicators.ATR(self.data, period=14)

        self.order = None
        self.entry_price = None

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return
        if order.status == order.Completed:
            if order.isbuy():
                self.entry_price = order.executed.price
        if order.status in (order.Completed, order.Canceled, order.Margin, order.Rejected):
            self.order = None

    def next(self):
        if self.order:
            return

        if not self.position:
            trend_ok = self.data.close[0] > self.sma200[0]

            golden_cross_recent = any(self.crossover[-i] == 1 for i in range(5))

            bb_top = self.bb.top[0]
            bb_bot = self.bb.bot[0]
            bb_mid = self.bb.mid[0]
            band_width = bb_top - bb_bot
            percent_b = (self.data.close[0] - bb_bot) / band_width if band_width != 0 else 0.0
            value_ok = (self.data.close[0] < bb_mid) and (percent_b <= 0.2)

            rsi_ok = 40 <= self.rsi[0] <= 60

            liquidity_ok = (self.vol_sma20[0] > 100000) and (self.vol_sma20[0] > self.vol_sma50[0])

            stability_ok = (self.atr[0] / self.data.close[0]) <= 0.04

            if trend_ok and golden_cross_recent and value_ok and rsi_ok and liquidity_ok and stability_ok:
                price = self.data.close[0]
                atr_val = self.atr[0]
                stop_distance = max(0.10 * price, 2 * atr_val)
                capital = self.broker.getvalue()
                risk_amount = 0.01 * capital
                risk_based_shares = risk_amount / stop_distance
                risk_based_value = risk_based_shares * price
                cash_allocation = min(0.25 * capital, risk_based_value)
                size = int(cash_allocation / price)

                if size > 0:
                    self.order = self.buy(size=size)

        else:
            take_profit = self.data.close[0] >= self.entry_price * 1.20
            stop_loss = self.data.close[0] <= self.entry_price * 0.90
            death_cross = self.crossover[0] == -1
            overheat = (self.rsi[0] >= 70) or (self.data.close[0] >= self.bb.top[0])
            risk_spike = (self.atr[0] / self.data.close[0]) > 0.06

            if take_profit or stop_loss or death_cross or overheat or risk_spike:
                self.order = self.close()
