import backtrader as bt


class RegimeTradingStrategy(bt.Strategy):
    def __init__(self):
        self.order = None
        self.entry_type = None

        self.bband = bt.indicators.BollingerBands(
            self.data.close, period=20, devfactor=2.0
        )
        self.width = self.bband.top - self.bband.bot
        self.width_sma = bt.indicators.SMA(self.width, period=200)

        self.ema50 = bt.indicators.EMA(self.data.close, period=50)
        self.ema200 = bt.indicators.EMA(self.data.close, period=200)

        self.cross_top = bt.indicators.CrossOver(self.data.close, self.bband.top)
        self.cross_bot = bt.indicators.CrossOver(self.data.close, self.bband.bot)
        self.cross_mid = bt.indicators.CrossOver(self.data.close, self.bband.mid)

    def notify_order(self, order):
        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            self.order = None
            if not self.position:
                self.entry_type = None

    def next(self):
        if self.order:
            return

        if not self.position:
            is_mr = self.width[0] <= self.width_sma[0]
            is_trend = self.width[0] > self.width_sma[0]

            if is_mr:
                if self.data.close[0] <= self.bband.bot[0]:
                    size = int((self.broker.getcash() * 0.99) / self.data.close[0])
                    if size > 0:
                        self.entry_type = "mr_long"
                        self.order = self.buy(size=size)
                elif self.data.close[0] >= self.bband.top[0]:
                    size = int((self.broker.getcash() * 0.99) / self.data.close[0])
                    if size > 0:
                        self.entry_type = "mr_short"
                        self.order = self.sell(size=size)
            elif is_trend:
                if self.ema50[0] > self.ema200[0] and self.cross_top[0] > 0:
                    size = int((self.broker.getcash() * 0.99) / self.data.close[0])
                    if size > 0:
                        self.entry_type = "trend_long"
                        self.order = self.buy(size=size)
                elif self.ema50[0] < self.ema200[0] and self.cross_bot[0] < 0:
                    size = int((self.broker.getcash() * 0.99) / self.data.close[0])
                    if size > 0:
                        self.entry_type = "trend_short"
                        self.order = self.sell(size=size)
        else:
            if self.position.size > 0:
                if self.entry_type == "mr_long":
                    if self.cross_mid[0] != 0:
                        self.order = self.close()
                elif self.entry_type == "trend_long":
                    if self.cross_mid[0] < 0:
                        self.order = self.close()
            elif self.position.size < 0:
                if self.entry_type == "mr_short":
                    if self.cross_mid[0] != 0:
                        self.order = self.close()
                elif self.entry_type == "trend_short":
                    if self.cross_mid[0] > 0:
                        self.order = self.close()
