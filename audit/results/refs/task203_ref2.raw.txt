import backtrader as bt


class MeanReversionZScore(bt.Strategy):
    params = (
        ("lookback", 20),
        ("half_life", 10),
        ("entry_threshold", 2.0),
        ("stoploss_factor", 1.5),
    )

    def __init__(self):
        self.order = None
        self.bb = bt.indicators.BollingerBands(
            self.data.close, period=self.p.lookback, devfactor=2
        )
        band_width = self.bb.top - self.bb.mid
        self.z = 2.0 * (self.data.close - self.bb.mid) / band_width
        self.z_ema = bt.indicators.EMA(self.z, period=self.p.half_life)

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return
        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            self.order = None

    def next(self):
        if self.order:
            return

        stoploss_threshold = self.p.entry_threshold * self.p.stoploss_factor
        z_val = self.z[0]
        z_ema_now = self.z_ema[0]
        z_ema_prev = self.z_ema[-1]
        band_width_val = self.bb.top[0] - self.bb.bot[0]

        if self.position:
            if self.position.size > 0:
                cross_mean = z_ema_prev <= 0 and z_ema_now > 0
                stop_loss = z_val >= stoploss_threshold
                if cross_mean or stop_loss:
                    self.order = self.close()
            elif self.position.size < 0:
                cross_mean = z_ema_prev >= 0 and z_ema_now < 0
                stop_loss = z_val <= -stoploss_threshold
                if cross_mean or stop_loss:
                    self.order = self.close()
        else:
            if band_width_val < 1e-5:
                return

            if z_val < -self.p.entry_threshold:
                self.order = self.buy()
            elif z_val > self.p.entry_threshold:
                self.order = self.sell()
