import datetime
import backtrader as bt


class DoubleTopStrategy(bt.Strategy):
    def __init__(self):
        self.order = None
        self.stop_price = None
        self.peaks = []  # List of tuples: (bar_index, high_value)

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return
        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            self.order = None
            if not self.position:
                self.stop_price = None

    def next(self):
        if len(self) < 5:
            return

        is_last_bar = (
            self.data.datetime.time(0) >= datetime.time(15, 45)
        ) or (len(self) == self.data.buflen())

        # Position management
        if self.position:
            # Stop-loss check
            if (
                self.stop_price is not None
                and self.data.high[0] >= self.stop_price
            ):
                if not self.order:
                    self.order = self.close()
                return

            # Intraday end-of-day exit check
            if is_last_bar:
                if not self.order:
                    self.order = self.close()
                return

        # Peak detection for bar at index -2 (N=2 window: -4, -3 vs -2 vs -1, 0)
        h_i = self.data.high[-2]
        if (
            h_i > self.data.high[-3]
            and h_i > self.data.high[-4]
            and h_i >= self.data.high[-1]
            and h_i >= self.data.high[0]
        ):

            p_idx = len(self) - 3
            p_high = h_i
            self.peaks.append((p_idx, p_high))

            # Double Top pattern evaluation
            if len(self.peaks) >= 2:
                p1_idx, p1_high = self.peaks[-2]
                p2_idx, p2_high = self.peaks[-1]

                bars_between = p2_idx - p1_idx
                diff_pct = abs(p2_high - p1_high) / max(p1_high, p2_high)
                is_bearish = self.data.close[-1] < self.data.open[-1]

                if (
                    bars_between >= 5
                    and diff_pct <= 0.02
                    and is_bearish
                    and not self.position
                    and not self.order
                    and not is_last_bar
                ):
                    higher_high = max(p1_high, p2_high)
                    self.stop_price = higher_high * 1.01
                    self.order = self.sell()
