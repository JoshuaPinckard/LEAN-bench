import backtrader as bt


class BreakRetest(bt.Strategy):
    def __init__(self):
        self.order = None
        self.entry_price = None

        self.current_date = None
        self.current_day_high = None
        self.current_day_low = None
        self.previous_day_high = None
        self.previous_day_low = None

        self.premarket_high = None
        self.premarket_low = None
        self.first_five_high = None
        self.first_five_low = None

        self.breakout_active = False
        self.breakout_level = None
        self.bars_since_breakout = 0
        self.retested = False

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            if order.status == order.Completed:
                if order.isbuy():
                    self.entry_price = order.executed.price
                elif order.issell():
                    self.entry_price = None

            if order == self.order:
                self.order = None

    def _reset_breakout(self):
        self.breakout_active = False
        self.breakout_level = None
        self.bars_since_breakout = 0
        self.retested = False

    def next(self):
        dt = self.data.datetime.datetime(0)
        bar_date = dt.date()
        close = self.data.close[0]
        high = self.data.high[0]
        low = self.data.low[0]

        if self.current_date != bar_date:
            if self.current_date is not None:
                self.previous_day_high = self.current_day_high
                self.previous_day_low = self.current_day_low

            self.current_date = bar_date
            self.current_day_high = high
            self.current_day_low = low
            self.premarket_high = None
            self.premarket_low = None
            self.first_five_high = None
            self.first_five_low = None
        else:
            self.current_day_high = max(self.current_day_high, high)
            self.current_day_low = min(self.current_day_low, low)

        is_premarket = dt.hour < 9 or (dt.hour == 9 and dt.minute < 30)
        is_first_five_minutes = dt.hour == 9 and dt.minute == 30

        if is_premarket:
            self.premarket_high = (
                high if self.premarket_high is None else max(self.premarket_high, high)
            )
            self.premarket_low = (
                low if self.premarket_low is None else min(self.premarket_low, low)
            )

        if is_first_five_minutes:
            self.first_five_high = high
            self.first_five_low = low

        if self.order:
            return

        if self.position:
            if close >= self.entry_price * 1.015 or close <= self.entry_price * 0.995:
                self.order = self.close()
            return

        levels = [
            level
            for level in (
                self.previous_day_high,
                self.premarket_high,
                self.first_five_high,
            )
            if level is not None
        ]

        if not levels:
            return

        resistance = max(levels)

        if not self.breakout_active:
            if close > resistance:
                self.breakout_active = True
                self.breakout_level = resistance
                self.bars_since_breakout = 0
                self.retested = False
            return

        self.bars_since_breakout += 1

        if self.bars_since_breakout > 5:
            self._reset_breakout()
            return

        if not self.retested:
            if close <= self.breakout_level:
                self.retested = True
            return

        if close > self.breakout_level:
            stop_loss_price = close * 0.995
            risk_per_share = close - stop_loss_price
            position_size = (self.broker.getvalue() * 0.01) / risk_per_share
            self.order = self.buy(size=position_size)
            self._reset_breakout()
