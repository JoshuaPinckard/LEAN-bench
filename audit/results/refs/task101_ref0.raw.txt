import backtrader as bt


class TrailingRunnerMACDStrategy(bt.Strategy):
    params = dict(
        atr_period=14,
        sma_period=9,
        macd_fast=12,
        macd_slow=9,
        macd_signal=26,
        atr_mult=2.0,
    )

    def __init__(self):
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.sma = bt.indicators.SMA(self.data, period=self.p.sma_period)
        self.macd = bt.indicators.MACD(
            self.data,
            period_me1=self.p.macd_fast,
            period_me2=self.p.macd_slow,
            period_signal=self.p.macd_signal,
        )
        self.macd_cross = bt.indicators.CrossOver(self.macd.macd, self.macd.signal)

        self.order = None
        self.entry_price = None
        self.stop_price = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status == order.Completed:
            if order.isbuy():
                self.entry_price = order.executed.price
                self.stop_price = self.entry_price - self.p.atr_mult * self.atr[0]
            else:
                self.entry_price = None
                self.stop_price = None
            self.order = None
        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        if self.order:
            return

        if not self.position:
            if self.data.close[0] > self.sma[0] and self.macd_cross[0] > 0:
                self.order = self.buy()
            return

        candidate_stop = self.data.high[0] - self.p.atr_mult * self.atr[0]
        if self.stop_price is None or candidate_stop > self.stop_price:
            self.stop_price = candidate_stop

        if self.data.low[0] <= self.stop_price:
            self.order = self.close()
            return

        if self.data.close[0] > self.entry_price and self.macd_cross[0] < 0:
            self.order = self.close()
