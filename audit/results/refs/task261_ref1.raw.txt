import backtrader as bt


class Strategy(bt.Strategy):
    params = (
        ('risk_fraction', 0.01),
    )

    def __init__(self):
        self.macd = bt.indicators.MACD(
            self.data.close,
            period_me1=12,
            period_me2=26,
            period_signal=9,
        ).macd
        self.atr = bt.indicators.ATR(self.data, period=14)

        self.entry_order = None
        self.stop_order = None
        self.target_order = None
        self.entry_atr = None

    def next(self):
        if self.entry_order or self.stop_order or self.target_order or self.position:
            return

        atr_value = self.atr[0]
        if atr_value <= 0:
            return

        risk_amount = self.broker.getvalue() * self.p.risk_fraction
        size = risk_amount / atr_value
        if size <= 0:
            return

        if self.macd[-1] < 0 and self.macd[0] > 0:
            self.entry_atr = atr_value
            self.entry_order = self.buy(size=size)

        elif self.macd[-1] > 0 and self.macd[0] < 0:
            self.entry_atr = atr_value
            self.entry_order = self.sell(size=size)

    def notify_order(self, order):
        if order.status in [order.Canceled, order.Margin, order.Rejected]:
            if order is self.entry_order:
                self.entry_order = None
                self.entry_atr = None
            elif order is self.stop_order:
                self.stop_order = None
            elif order is self.target_order:
                self.target_order = None
            return

        if order.status != order.Completed:
            return

        if order is self.entry_order:
            self.entry_order = None
            entry_price = order.executed.price
            size = abs(order.executed.size)

            if order.isbuy():
                stop_price = entry_price - self.entry_atr
                target_price = entry_price + (3 * self.entry_atr)
                self.stop_order = self.sell(
                    size=size,
                    exectype=bt.Order.Stop,
                    price=stop_price,
                )
                self.target_order = self.sell(
                    size=size,
                    exectype=bt.Order.Limit,
                    price=target_price,
                    oco=self.stop_order,
                )
            else:
                stop_price = entry_price + self.entry_atr
                target_price = entry_price - (3 * self.entry_atr)
                self.stop_order = self.buy(
                    size=size,
                    exectype=bt.Order.Stop,
                    price=stop_price,
                )
                self.target_order = self.buy(
                    size=size,
                    exectype=bt.Order.Limit,
                    price=target_price,
                    oco=self.stop_order,
                )

            self.entry_atr = None

        elif order is self.stop_order:
            self.stop_order = None
            if self.target_order:
                self.cancel(self.target_order)

        elif order is self.target_order:
            self.target_order = None
            if self.stop_order:
                self.cancel(self.stop_order)
