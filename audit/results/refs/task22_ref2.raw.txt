import backtrader as bt


class TrendRSIHardStopStrategy(bt.Strategy):
    """
    Reference implementation of Trend Strategy with RSI Filter and Hard Stop-Loss.
    - Daily RSI (14-period) filter on a 15-minute EUR/USD feed.
    - Long entry when Daily RSI < 30.
    - Short entry when Daily RSI > 70.
    - Position sizing: Loss at Stop-Loss equals 1% of total account value.
    - Stop-Loss: 0.1% of entry price.
    - Take-Profit: 1.0% of entry price + 1 pip spread requirement.
    """
    params = (
        ('rsi_period', 14),
        ('rsi_low', 30.0),
        ('rsi_high', 70.0),
        ('stop_loss', 0.001),     # 0.1%
        ('take_profit', 0.01),    # 1.0%
        ('pip_value', 0.0001),    # 1 pip for EUR/USD
        ('risk_pct', 0.01),       # 1.0% of capital risked per trade
    )

    def __init__(self):
        self.rsi = bt.indicators.RSI(
            self.data.close,
            period=self.p.rsi_period,
            timeframe=bt.TimeFrame.Days
        )
        self.order = None
        self.sl_order = None
        self.tp_order = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            if order == self.order:
                self.order = None
                exec_price = order.executed.price
                exec_size = abs(order.executed.size)

                if order.isbuy():
                    sl_price = exec_price * (1.0 - self.p.stop_loss)
                    tp_price = exec_price * (1.0 + self.p.take_profit) + self.p.pip_value
                    self.sl_order = self.sell(
                        size=exec_size,
                        price=sl_price,
                        exectype=bt.Order.Stop
                    )
                    self.tp_order = self.sell(
                        size=exec_size,
                        price=tp_price,
                        exectype=bt.Order.Limit,
                        oco=self.sl_order
                    )
                else:
                    sl_price = exec_price * (1.0 + self.p.stop_loss)
                    tp_price = exec_price * (1.0 - self.p.take_profit) - self.p.pip_value
                    self.sl_order = self.buy(
                        size=exec_size,
                        price=sl_price,
                        exectype=bt.Order.Stop
                    )
                    self.tp_order = self.buy(
                        size=exec_size,
                        price=tp_price,
                        exectype=bt.Order.Limit,
                        oco=self.sl_order
                    )
            elif order in [self.sl_order, self.tp_order]:
                self.sl_order = None
                self.tp_order = None

        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            if order == self.order:
                self.order = None
            if order in [self.sl_order, self.tp_order]:
                self.sl_order = None
                self.tp_order = None

    def next(self):
        if self.order or self.position:
            return

        current_price = self.data.close[0]
        if current_price <= 0:
            return

        # Position sizing calculation:
        # Potential loss = size * (current_price * stop_loss) = risk_pct * account_value
        # size = (risk_pct * account_value) / (stop_loss * current_price)
        account_value = self.broker.getvalue()
        target_risk = account_value * self.p.risk_pct
        sl_distance = current_price * self.p.stop_loss

        if sl_distance <= 0:
            return

        size = target_risk / sl_distance

        if self.rsi[0] < self.p.rsi_low:
            self.order = self.buy(size=size)
        elif self.rsi[0] > self.p.rsi_high:
            self.order = self.sell(size=size)
