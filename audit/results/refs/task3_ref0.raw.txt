import backtrader as bt


class SwingTradingRSIFilter(bt.Strategy):
    params = dict(
        sma_period=50,
        rsi_period=14,
        lookback=20,
        risk_pct=0.01,
    )

    def __init__(self):
        self.sma = bt.indicators.SMA(self.data.close, period=self.p.sma_period)
        self.rsi = bt.indicators.RSI(self.data.close, period=self.p.rsi_period)
        self.highest_high = bt.indicators.Highest(self.data.high, period=self.p.lookback)
        self.order = None
        self.target_price = None

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return
        if order.status in (order.Completed, order.Canceled, order.Margin, order.Rejected):
            self.order = None
            if order.status == order.Completed and order.issell():
                self.target_price = None

    def next(self):
        if self.order:
            return

        if self.position:
            if self.data.high[0] >= self.target_price:
                self.order = self.close()
            return

        if self.data.close[0] > self.sma[0] and self.rsi[0] < 50:
            # -1 excludes the current (entry) bar: "20 bars prior to entering"
            self.target_price = self.highest_high[-1]
            size = (self.p.risk_pct * self.broker.getvalue()) / self.data.close[0]
            self.order = self.buy(size=size)
