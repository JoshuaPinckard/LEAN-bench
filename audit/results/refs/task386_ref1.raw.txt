import backtrader as bt


class BreakoutStrategy(bt.Strategy):
    params = (
        ("period", 20),
        ("stop_loss", 0.02),
    )

    def __init__(self):
        self.highest_high = bt.indicators.Highest(self.data.high(-1), period=self.p.period)
        self.lowest_low = bt.indicators.Lowest(self.data.low(-1), period=self.p.period)
        self.sma = bt.indicators.SimpleMovingAverage(self.data.close, period=self.p.period)

        self.order = None
        self.stop_order = None
        self.entry_price = None

    def notify_order(self, order):
        if order.status in (order.Completed, order.Canceled, order.Margin, order.Rejected):
            if order == self.order:
                if order.status == order.Completed and order.isbuy() and self.position.size > 0:
                    self.entry_price = order.executed.price
                    self.stop_order = self.sell(
                        exectype=bt.Order.Stop,
                        price=self.entry_price * (1.0 - self.p.stop_loss),
                    )
                elif order.status == order.Completed and order.issell() and self.position.size < 0:
                    self.entry_price = order.executed.price
                    self.stop_order = self.buy(
                        exectype=bt.Order.Stop,
                        price=self.entry_price * (1.0 + self.p.stop_loss),
                    )
                elif self.position.size == 0:
                    self.entry_price = None
                self.order = None

            if order == self.stop_order:
                self.stop_order = None
                if self.position.size == 0:
                    self.entry_price = None

    def next(self):
        if self.order:
            return

        long_signal = (
            self.data.close[0] > self.highest_high[0]
            and self.data.close[0] > self.sma[0]
        )
        short_signal = (
            self.data.close[0] < self.lowest_low[0]
            and self.data.close[0] < self.sma[0]
        )

        if not self.position:
            if long_signal:
                self.order = self.buy()
            elif short_signal:
                self.order = self.sell()
        elif self.position.size > 0 and short_signal:
            if self.stop_order:
                self.cancel(self.stop_order)
                self.stop_order = None
            self.order = self.close()
        elif self.position.size < 0 and long_signal:
            if self.stop_order:
                self.cancel(self.stop_order)
                self.stop_order = None
            self.order = self.close()
