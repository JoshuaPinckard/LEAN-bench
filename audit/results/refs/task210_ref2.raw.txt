import backtrader as bt
import math
import collections
import statistics


class LowVolatilityMonthlyStrategy(bt.Strategy):
    def __init__(self):
        self.order = None

        # Indicators
        self.bb = bt.indicators.BollingerBands(
            self.data.close, period=20, devfactor=2.0
        )
        self.width = (self.bb.top - self.bb.bot) / self.bb.mid

        self.atr = bt.indicators.ATR(self.data, period=14)
        self.sma_atr = bt.indicators.SMA(self.atr, period=252)

        self.sma_vol = bt.indicators.SMA(self.data.volume, period=252)

    def notify_order(self, order):
        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            self.order = None

    def _get_percentile(self, data, p):
        sdata = sorted(data)
        n = len(sdata)
        if n == 0:
            return 0.0
        if n == 1:
            return sdata[0]
        k = (n - 1) * (p / 100.0)
        f = math.floor(k)
        c = math.ceil(k)
        if f == c:
            return sdata[int(k)]
        return sdata[int(f)] * (c - k) + sdata[int(c)] * (k - f)

    def is_last_trading_day_of_month(self):
        try:
            curr_month = self.data.datetime.date(0).month
            next_month = self.data.datetime.date(1).month
            return curr_month != next_month
        except IndexError:
            return True

    def next(self):
        if len(self) < 126:
            return

        if not self.is_last_trading_day_of_month():
            return

        # 1. Relative BB Width <= 20th percentile over trailing 252 days
        w_size = min(len(self.width), 252)
        w_window = self.width.get(size=w_size)
        curr_width = self.width[0]
        p20_width = self._get_percentile(w_window, 20)
        cond_width = curr_width <= p20_width

        # 2. ATR(14) < SMA(ATR(14), 252)
        cond_atr = self.atr[0] < self.sma_atr[0]

        # 3. SMA(volume, 252) >= 1,000,000 shares
        cond_vol = self.sma_vol[0] >= 1000000

        conditions_met = cond_width and cond_atr and cond_vol

        if conditions_met:
            if not self.position and self.order is None:
                price = self.data.close[0]
                cash = self.broker.getcash()
                size = int(cash / (price * 1.001))
                if size > 0:
                    self.order = self.buy(size=size)
        else:
            if self.position and self.order is None:
                self.order = self.close()
