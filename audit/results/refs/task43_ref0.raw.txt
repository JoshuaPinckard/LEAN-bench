import backtrader as bt


class BollingerBandsMeanReversion(bt.Strategy):
    params = dict(period=20, devfactor=2.0)

    def __init__(self):
        self.bb = bt.indicators.BollingerBands(
            self.data.close, period=self.p.period, devfactor=self.p.devfactor
        )
        self.middle_bb = self.bb.lines.mid
        self.upper_bb = self.bb.lines.top
        self.lower_bb = self.bb.lines.bot

        self.cross_lower = bt.indicators.CrossOver(self.data.close, self.lower_bb)
        self.cross_upper = bt.indicators.CrossOver(self.data.close, self.upper_bb)
        self.cross_middle = bt.indicators.CrossOver(self.data.close, self.middle_bb)

        self.order = None

    def notify_order(self, order):
        if order.status in (order.Completed, order.Canceled, order.Margin, order.Rejected):
            self.order = None

    def next(self):
        if self.order:
            return

        position_size = self.position.size

        if position_size == 0:
            if self.cross_lower[0] > 0:
                size = int(self.broker.getcash() / self.data.close[0])
                if size > 0:
                    self.order = self.buy(size=size)
            elif self.cross_upper[0] < 0:
                size = int(self.broker.getcash() / self.data.close[0])
                if size > 0:
                    self.order = self.sell(size=size)
        elif position_size > 0:
            if self.cross_middle[0] < 0:
                self.order = self.close()
        elif position_size < 0:
            if self.cross_middle[0] > 0:
                self.order = self.close()
