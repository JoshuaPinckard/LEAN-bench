import backtrader as bt


class CSILiteMomentumStrategy(bt.Strategy):
    def __init__(self):
        self.atr = bt.indicators.ATR(self.data, period=14)
        self.macd = bt.indicators.MACDHisto(
            self.data, period_me1=12, period_me2=26, period_signal=9
        )
        self.sma_price = bt.indicators.SMA(self.data.close, period=50)
        self.sma_atr = bt.indicators.SMA(self.atr, period=50)
        self.sma_vol = bt.indicators.SMA(self.data.volume, period=50)
        self.rsi = bt.indicators.RSI(self.data.close, period=14)
        self.macd_cross = bt.indicators.CrossOver(self.macd.macd, self.macd.signal)

        self.order = None
        self.tranche_count = 0
        self.peak_s = 0.0

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        if self.order:
            return

        s = (
            max(0.0, self.macd.histo[0])
            * (self.atr[0] / self.sma_atr[0])
            * (self.data.volume[0] / self.sma_vol[0])
            * 100.0
        )

        if self.position:
            if s > self.peak_s:
                self.peak_s = s

            macd_cross_down = self.macd_cross[0] < 0
            if s < 0.7 * self.peak_s or macd_cross_down or self.rsi[0] < 45:
                self.order = self.close()
                return

            if self.tranche_count < 3 and s > 1.5:
                exposure = (self.position.size * self.data.close[0]) / self.broker.getvalue()
                if exposure < 0.5:
                    cash_for_tranche = 0.05 * self.broker.getvalue()
                    size = cash_for_tranche / self.data.close[0]
                    self.order = self.buy(size=size)
                    self.tranche_count += 1
        else:
            if (
                s > 1.5
                and self.data.close[0] > self.sma_price[0]
                and self.macd.macd[0] > self.macd.signal[0]
            ):
                cash_for_tranche = 0.05 * self.broker.getvalue()
                size = cash_for_tranche / self.data.close[0]
                self.order = self.buy(size=size)
                self.tranche_count = 1
                self.peak_s = s
