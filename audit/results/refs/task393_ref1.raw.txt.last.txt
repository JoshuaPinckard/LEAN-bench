import backtrader as bt


class DonchianWidthMeanReversionStrategy(bt.Strategy):
    params = (
        ("DeviationMultiplier", 2.0),
    )

    def __init__(self):
        donchian = bt.ind.DonchianChannel(self.data, period=20)
        self.width = donchian.hi - donchian.lo
        self.width_sma = bt.ind.SMA(self.width, period=20)

        self.upper_extreme = self.width_sma * self.p.DeviationMultiplier
        self.lower_extreme = self.width_sma / self.p.DeviationMultiplier

        self.cross_up_upper = bt.ind.CrossUp(self.width, self.upper_extreme)
        self.cross_down_upper = bt.ind.CrossDown(self.width, self.upper_extreme)
        self.cross_up_lower = bt.ind.CrossUp(self.width, self.lower_extreme)
        self.cross_down_lower = bt.ind.CrossDown(self.width, self.lower_extreme)
        self.cross_down_mean = bt.ind.CrossDown(self.width, self.width_sma)
        self.cross_up_mean = bt.ind.CrossUp(self.width, self.width_sma)

        self.upper_regime = False
        self.lower_regime = False
        self.order = None
        self.stop_order = None

    def next(self):
        if self.order:
            return

        if self.cross_up_upper[0]:
            self.upper_regime = True
            self.lower_regime = False
        elif self.cross_down_lower[0]:
            self.lower_regime = True
            self.upper_regime = False

        if self.position:
            if self.position.size < 0 and self.cross_down_mean[0]:
                if self.stop_order:
                    self.cancel(self.stop_order)
                self.order = self.close()
            elif self.position.size > 0 and self.cross_up_mean[0]:
                if self.stop_order:
                    self.cancel(self.stop_order)
                self.order = self.close()
            return

        if self.upper_regime and self.cross_down_upper[0]:
            self.order = self.sell()
        elif self.lower_regime and self.cross_up_lower[0]:
            self.order = self.buy()

    def notify_order(self, order):
        if order.status == order.Completed:
            if order == self.order:
                if order.isbuy():
                    self.stop_order = self.sell(
                        exectype=bt.Order.Stop,
                        price=order.executed.price * (1.0 - 0.02),
                    )
                else:
                    self.stop_order = self.buy(
                        exectype=bt.Order.Stop,
                        price=order.executed.price * (1.0 + 0.02),
                    )
                self.order = None
            elif order == self.stop_order:
                self.stop_order = None

        elif order.status in (order.Canceled, order.Margin, order.Rejected):
            if order == self.order:
                self.order = None
            elif order == self.stop_order:
                self.stop_order = None
