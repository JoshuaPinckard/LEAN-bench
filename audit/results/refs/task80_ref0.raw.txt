Implementing a manual 1h-aggregation + 1m Order Block detection strategy since the harness provides only the 1-minute feed.

```python
import backtrader as bt


class FVGOrderBlockStrategy(bt.Strategy):
    params = dict(
        risk_pct=0.01,
        h1_sma_period=20,
        adx_period=14,
        adx_threshold=20,
        ob_body_ratio=0.7,
    )

    def __init__(self):
        self.adx = bt.indicators.ADX(self.data, period=self.p.adx_period)

        self.cur_h1 = None
        self.h1_bars = []
        self.fvg_zones = []
        self.trend_up = False

        self.active_zone = None
        self.ob_high = None
        self.ob_broken = False
        self.pending_breakout = None

        self.order = None

    def _update_h1(self):
        dt = self.data.datetime.datetime(0)
        hour_ts = dt.replace(minute=0, second=0, microsecond=0)

        if self.cur_h1 is None:
            self.cur_h1 = dict(hour=hour_ts, open=self.data.open[0],
                                high=self.data.high[0], low=self.data.low[0],
                                close=self.data.close[0])
        elif hour_ts == self.cur_h1['hour']:
            self.cur_h1['high'] = max(self.cur_h1['high'], self.data.high[0])
            self.cur_h1['low'] = min(self.cur_h1['low'], self.data.low[0])
            self.cur_h1['close'] = self.data.close[0]
        else:
            self.h1_bars.append(self.cur_h1)
            self._on_new_h1_bar()
            self.cur_h1 = dict(hour=hour_ts, open=self.data.open[0],
                                high=self.data.high[0], low=self.data.low[0],
                                close=self.data.close[0])

    def _on_new_h1_bar(self):
        bars = self.h1_bars
        if len(bars) >= 3:
            c1, c2, c3 = bars[-3], bars[-2], bars[-1]
            if c1['high'] < c3['low']:
                self.fvg_zones.append(dict(low=c1['high'], high=c3['low'], mitigated=False))

        if len(bars) >= self.p.h1_sma_period:
            recent = [b['close'] for b in bars[-self.p.h1_sma_period:]]
            avg = sum(recent) / len(recent)
            self.trend_up = bars[-1]['close'] > avg
        else:
            self.trend_up = False

    def _update_ob(self):
        rng = self.data.high[0] - self.data.low[0]
        if rng <= 0:
            return
        is_bullish = self.data.close[0] > self.data.open[0]
        body = abs(self.data.close[0] - self.data.open[0])
        if is_bullish and (body / rng) >= self.p.ob_body_ratio:
            self.ob_high = self.data.high[-1]
            self.ob_broken = False

    def _update_zone_touch(self):
        if self.active_zone is None:
            for z in reversed(self.fvg_zones):
                if z['mitigated']:
                    continue
                if self.data.low[0] <= z['high'] and self.data.high[0] >= z['low']:
                    self.active_zone = z
                    break
        else:
            z = self.active_zone
            if self.data.close[0] < z['low']:
                z['mitigated'] = True
                self.active_zone = None

    def _try_enter(self, confirm):
        entry_price = self.data.close[0]
        sl_price = confirm['breakout_low']
        risk_per_unit = entry_price - sl_price
        if risk_per_unit <= 0:
            return

        candidates = [z for z in self.fvg_zones if not z['mitigated'] and z['low'] > entry_price]
        if not candidates:
            return
        target_zone = min(candidates, key=lambda z: z['low'])
        tp_price = target_zone['low']

        equity = self.broker.getvalue()
        size = (equity * self.p.risk_pct) / risk_per_unit
        if size <= 0:
            return

        self.order = self.buy_bracket(size=size, exectype=bt.Order.Market,
                                       stopprice=sl_price, limitprice=tp_price)[0]

        if self.active_zone is not None:
            self.active_zone['mitigated'] = True
            self.active_zone = None

    def next(self):
        self._update_h1()
        if len(self) < 2:
            return

        self._update_ob()
        self._update_zone_touch()

        if self.pending_breakout is not None:
            confirm = self.pending_breakout
            self.pending_breakout = None
            if self.data.close[0] > confirm['breakout_close'] and not self.position and not self.order:
                self._try_enter(confirm)
            return

        if self.position or self.order:
            return

        if (self.ob_high is not None and not self.ob_broken and self.active_zone is not None
                and self.trend_up and self.adx[0] > self.p.adx_threshold):
            if self.data.close[-1] <= self.ob_high < self.data.close[0]:
                self.ob_broken = True
                self.pending_breakout = dict(
                    breakout_close=self.data.close[0],
                    breakout_low=self.data.low[0],
                )

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return
        if order == self.order and order.status in [order.Completed, order.Canceled,
                                                      order.Margin, order.Rejected]:
            self.order = None
```
