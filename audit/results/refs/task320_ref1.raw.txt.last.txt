import backtrader as bt


class DoubleTopStrategy(bt.Strategy):
    def __init__(self):
        self.broker.set_coc(True)
        self.addminperiod(5)

        self.last_peak_index = None
        self.last_peak_high = None

        self.entry_order = None
        self.stop_order = None
        self.exit_order = None
        self.stop_price = None

    def notify_order(self, order):
        if order.status not in (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            return

        if order is self.entry_order:
            self.entry_order = None
        elif order is self.stop_order:
            self.stop_order = None
        elif order is self.exit_order:
            self.exit_order = None

    def next(self):
        is_last_session_bar = (
            self.data.datetime.time(0).hour == 15
            and self.data.datetime.time(0).minute == 45
        )

        if is_last_session_bar:
            if self.position and self.exit_order is None:
                if self.stop_order is not None:
                    self.cancel(self.stop_order)
                self.exit_order = self.close()
            return

        p2_is_peak = (
            self.data.high[-2] > self.data.high[-3]
            and self.data.high[-2] > self.data.high[-4]
            and self.data.high[-2] >= self.data.high[-1]
            and self.data.high[-2] >= self.data.high[0]
        )

        if not p2_is_peak:
            return

        p2_index = len(self) - 3
        p2_high = self.data.high[-2]

        if self.last_peak_index is not None:
            sufficiently_separated = p2_index - self.last_peak_index >= 5
            similar_height = (
                abs(p2_high - self.last_peak_high)
                / max(self.last_peak_high, p2_high)
                <= 0.02
            )
            bearish_confirmation = self.data.close[-1] < self.data.open[-1]

            if (
                sufficiently_separated
                and similar_height
                and bearish_confirmation
                and not self.position
                and self.entry_order is None
                and self.exit_order is None
            ):
                self.stop_price = max(self.last_peak_high, p2_high) * 1.01
                self.entry_order = self.sell()
                self.stop_order = self.buy(
                    exectype=bt.Order.Stop,
                    price=self.stop_price,
                )

        self.last_peak_index = p2_index
        self.last_peak_high = p2_high
