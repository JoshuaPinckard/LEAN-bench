import backtrader as bt


class ROCTechnicalTradingRuleStrategy(bt.Strategy):
    params = (
        ('fast_period', 20),
        ('slow_period', 50),
    )

    def __init__(self):
        self.sma_fast = bt.indicators.SMA(self.data.close, period=self.p.fast_period)
        self.sma_slow = bt.indicators.SMA(self.data.close, period=self.p.slow_period)

        self.order = None

        self.true_positives = 0
        self.false_positives = 0
        self.true_negatives = 0
        self.false_negatives = 0

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        buy_signal = self.sma_fast[0] > self.sma_slow[0]
        positive_return = self.data.close[0] > self.data.open[0]

        if buy_signal:
            if positive_return:
                self.true_positives += 1
            else:
                self.false_positives += 1
        else:
            if positive_return:
                self.false_negatives += 1
            else:
                self.true_negatives += 1

        if self.order:
            return

        if not self.position:
            if buy_signal:
                self.order = self.buy(size=1)
        else:
            self.order = self.close()

    def stop(self):
        tp, fp = self.true_positives, self.false_positives
        tn, fn = self.true_negatives, self.false_negatives
        tpr = tp / (tp + fn) if (tp + fn) > 0 else 0.0
        fpr = fp / (fp + tn) if (fp + tn) > 0 else 0.0
        print('Confusion Matrix -> TP: %d, FP: %d, TN: %d, FN: %d' % (tp, fp, tn, fn))
        print('True Positive Rate: %.4f' % tpr)
        print('False Positive Rate: %.4f' % fpr)
