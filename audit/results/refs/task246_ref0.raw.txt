import math

import backtrader as bt


class PostDisasterCapitulationBounce(bt.Strategy):
    params = (
        ('sma_fast', 20),
        ('sma_slow', 50),
        ('rsi_period', 14),
        ('atr_period', 14),
        ('vol_period', 20),
        ('cap_window', 5),
        ('active_window', 10),
        ('drop_pct', 0.15),
        ('vol_spike_mult', 7.0),
        ('vol_ease_mult', 0.8),
        ('volatility_pct', 0.03),
        ('tp2_pct', 0.5),
        ('risk_pct', 0.02),
    )

    def __init__(self):
        self.sma20 = bt.indicators.SMA(self.data.close, period=self.p.sma_fast)
        self.sma50 = bt.indicators.SMA(self.data.close, period=self.p.sma_slow)
        self.rsi = bt.indicators.RSI(self.data.close, period=self.p.rsi_period)
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.volsma = bt.indicators.SMA(self.data.volume, period=self.p.vol_period)
        self.rsi_cross = bt.indicators.CrossOver(self.rsi, 30)

        self.order = None
        self.cap_bar = None
        self.cap_low = None
        self.cap_high_vol = None
        self.range_L = None
        self.range_H = None

        self.stop_price = None
        self.tp2_price = None

    def notify_order(self, order):
        if order.status in (order.Completed, order.Canceled, order.Margin, order.Rejected):
            self.order = None

    def next(self):
        self._update_capitulation_state()

        if self.order is not None:
            return

        if self.position:
            self._check_exits()
        else:
            self._check_entry()

    def _update_capitulation_state(self):
        drop_confirmed = self.data.close[0] <= self.data.close[-self.p.cap_window] * (1 - self.p.drop_pct)
        if not drop_confirmed:
            return

        best_offset = None
        best_vol = None
        for offset in range(self.p.cap_window):
            if len(self) - offset < self.p.vol_period:
                continue
            is_down = self.data.close[-offset] < self.data.open[-offset]
            vol_spike = self.data.volume[-offset] > self.p.vol_spike_mult * self.volsma[-offset]
            if is_down and vol_spike:
                v = self.data.volume[-offset]
                if best_vol is None or v > best_vol:
                    best_vol = v
                    best_offset = offset

        if best_offset is None:
            return

        candidate_bar = len(self) - best_offset
        window_expired = self.cap_bar is not None and len(self) > self.cap_bar + self.p.active_window

        if self.cap_bar is None or window_expired or best_vol > self.cap_high_vol:
            self.cap_bar = candidate_bar
            self.cap_low = self.data.low[-best_offset]
            self.cap_high_vol = best_vol
            highs = [self.data.high[-(best_offset + k)] for k in range(self.p.vol_period)]
            self.range_H = max(highs)
            self.range_L = self.cap_low

    def _in_active_window(self):
        return self.cap_bar is not None and len(self) <= self.cap_bar + self.p.active_window

    def _check_entry(self):
        if not self._in_active_window() or self.range_L is None or self.range_H is None:
            return

        positive_day = self.data.close[0] > self.data.open[0]
        rsi_cross_up = self.rsi_cross[0] > 0
        vol_normal = self.data.volume[0] >= self.p.vol_ease_mult * self.volsma[0]
        vol_range = (self.data.high[0] - self.data.low[0]) / self.data.open[0] >= self.p.volatility_pct
        trend_ok = self.data.close[0] > self.sma20[0]

        if not (positive_day and rsi_cross_up and vol_normal and vol_range and trend_ok):
            return

        L = self.range_L
        H = self.range_H
        stop = L - 1.0 * self.atr[0]
        entry_price = self.data.close[0]
        risk_per_share = entry_price - stop
        if risk_per_share <= 0:
            return

        equity = self.broker.getvalue()
        size = math.floor((self.p.risk_pct * equity) / risk_per_share)
        if size < 1:
            return

        self.stop_price = stop
        self.tp2_price = L + self.p.tp2_pct * (H - L)
        self.order = self.buy(size=size)

    def _check_exits(self):
        if self.data.low[0] <= self.stop_price:
            self.order = self.close()
            return

        touched_tp2 = self.data.high[0] >= self.tp2_price
        touched_sma50 = self.data.low[0] <= self.sma50[0] <= self.data.high[0]
        if touched_tp2 or touched_sma50:
            self.order = self.close()
