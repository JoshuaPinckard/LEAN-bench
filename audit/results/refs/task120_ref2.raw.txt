import backtrader as bt


class MartingaleRSIBB(bt.Strategy):
    params = (
        ("rsi_period", 14),
        ("rsi_oversold", 30),
        ("rsi_overbought", 70),
        ("rsi_exit_long", 50),
        ("rsi_exit_short", 50),
        ("bb_period", 20),
        ("bb_dev", 2.0),
        ("initial_lot", 0.01),
        ("lot_exp", 1.2345),
        ("max_levels", 6),
        ("max_dd", 350.0),
    )

    def __init__(self):
        self.rsi = bt.indicators.RSI(self.data.close, period=self.p.rsi_period)
        self.bb = bt.indicators.BollingerBands(
            self.data.close, period=self.p.bb_period, devfactor=self.p.bb_dev
        )

        self.order = None
        self.trades_count = 0
        self.peak_value = None

    def notify_order(self, order):
        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            if order.status in [order.Canceled, order.Margin, order.Rejected]:
                if self.position.size == 0:
                    self.trades_count = 0
                elif self.trades_count > 0:
                    self.trades_count -= 1
            self.order = None

        if self.position.size == 0:
            self.trades_count = 0

    def next(self):
        current_value = self.broker.getvalue()
        if self.peak_value is None or current_value > self.peak_value:
            self.peak_value = current_value

        drawdown = self.peak_value - current_value

        # Maximum Drawdown Rule ($350 total account drawdown limit)
        if drawdown >= self.p.max_dd:
            if self.position.size != 0:
                if not self.order:
                    self.order = self.close()
                    self.peak_value = current_value
            return

        # Guard pending orders
        if self.order:
            return

        close = self.data.close[0]
        rsi = self.rsi[0]
        bb_top = self.bb.lines.top[0]
        bb_bot = self.bb.lines.bot[0]

        pos_size = self.position.size

        if pos_size > 0:
            # Exit Long: Close > Upper BB or RSI > 50
            if close > bb_top or rsi > self.p.rsi_exit_long:
                self.order = self.close()
                return

            # Long Averaging (Martingale) Entry
            if self.trades_count < self.p.max_levels:
                if close < bb_bot and rsi < self.p.rsi_oversold:
                    size = self.p.initial_lot * (self.p.lot_exp**self.trades_count)
                    self.order = self.buy(size=size)
                    self.trades_count += 1

        elif pos_size < 0:
            # Exit Short: Close < Lower BB or RSI < 50
            if close < bb_bot or rsi < self.p.rsi_exit_short:
                self.order = self.close()
                return

            # Short Averaging (Martingale) Entry
            if self.trades_count < self.p.max_levels:
                if close > bb_top and rsi > self.p.rsi_overbought:
                    size = self.p.initial_lot * (self.p.lot_exp**self.trades_count)
                    self.order = self.sell(size=size)
                    self.trades_count += 1

        else:
            # Initial Entry
            if close < bb_bot and rsi < self.p.rsi_oversold:
                size = self.p.initial_lot
                self.order = self.buy(size=size)
                self.trades_count = 1
            elif close > bb_top and rsi > self.p.rsi_overbought:
                size = self.p.initial_lot
                self.order = self.sell(size=size)
                self.trades_count = 1
