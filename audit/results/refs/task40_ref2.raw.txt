import backtrader as bt


class PowerHourReversal(bt.Strategy):
    params = (
        ('rsi_period', 14),
        ('ema_period', 20),
        ('rsi_oversold', 30.0),
        ('rsi_overbought', 70.0),
        ('tp_pct', 0.015),
        ('sl_pct', 0.008),
        ('position_pct', 0.02),
        ('volatility_threshold', 0.015),
    )

    def __init__(self):
        self.rsi = bt.indicators.RSI(self.data.close, period=self.p.rsi_period)
        self.ema = bt.indicators.EMA(self.data.close, period=self.p.ema_period)
        self.order = None
        self.daily_date = None
        self.daily_high = 0.0
        self.daily_low = 0.0

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def _is_end_of_session(self):
        t = self.data.datetime.time(0)
        if t.hour > 15 or (t.hour == 15 and t.minute >= 30):
            return True
        try:
            if self.data.datetime.date(1) != self.data.datetime.date(0):
                return True
        except IndexError:
            return True
        return False

    def next(self):
        # Update daily high / low tracking for volatility filter
        current_date = self.data.datetime.date(0)
        if self.daily_date != current_date:
            self.daily_date = current_date
            self.daily_high = self.data.high[0]
            self.daily_low = self.data.low[0]
        else:
            self.daily_high = max(self.daily_high, self.data.high[0])
            self.daily_low = min(self.daily_low, self.data.low[0])

        if self.order:
            return

        # Position exit management
        if self.position:
            entry_price = self.position.price
            if self.position.size > 0:
                tp_price = entry_price * (1.0 + self.p.tp_pct)
                sl_price = entry_price * (1.0 - self.p.sl_pct)
                if self.data.high[0] >= tp_price or self.data.low[0] <= sl_price or self._is_end_of_session():
                    self.order = self.close()
            elif self.position.size < 0:
                tp_price = entry_price * (1.0 - self.p.tp_pct)
                sl_price = entry_price * (1.0 + self.p.sl_pct)
                if self.data.low[0] <= tp_price or self.data.high[0] >= sl_price or self._is_end_of_session():
                    self.order = self.close()
            return

        # Entry rules check: last hour of session
        t = self.data.datetime.time(0)
        in_last_hour = (t.hour >= 15)
        if not in_last_hour:
            return

        # Volatility check: daily range exceeds 1.5%
        if self.daily_low <= 0:
            return
        volatility = (self.daily_high - self.daily_low) / self.daily_low
        if volatility <= self.p.volatility_threshold:
            return

        # Position sizing: 2% of total capital
        capital = self.broker.getvalue()
        size = int((capital * self.p.position_pct) / self.data.close[0])
        if size <= 0:
            size = 1

        # Long entry logic: Price < EMA(20) and RSI crosses above 30 from below
        long_cond = (self.data.close[0] < self.ema[0]) and (self.rsi[-1] < self.p.rsi_oversold) and (self.rsi[0] > self.p.rsi_oversold)

        # Short entry logic: Price > EMA(20) and RSI crosses below 70 from above
        short_cond = (self.data.close[0] > self.ema[0]) and (self.rsi[-1] > self.p.rsi_overbought) and (self.rsi[0] < self.p.rsi_overbought)

        if long_cond:
            self.order = self.buy(size=size)
        elif short_cond:
            self.order = self.sell(size=size)
