import backtrader as bt


class SwingTradingStrategy(bt.Strategy):
    params = (
        ('ema_fast', 21),
        ('ema_slow', 50),
        ('rsi_period', 14),
        ('swing_period', 10),
        ('risk_pct', 0.01),
        ('rsi_overbought', 70.0),
        ('rsi_oversold', 30.0),
    )

    def __init__(self):
        self.ema50 = bt.indicators.EMA(self.data.close, period=self.p.ema_slow)
        self.ema21 = bt.indicators.EMA(self.data.close, period=self.p.ema_fast)
        self.rsi = bt.indicators.RSI(self.data.close, period=self.p.rsi_period)
        self.crossover = bt.indicators.CrossOver(self.ema21, self.ema50)
        self.order = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return
        if self.order and order.ref == self.order.ref:
            if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
                self.order = None

    def next(self):
        if self.position.size != 0 or self.order is not None:
            return

        close = self.data.close[0]
        ema50 = self.ema50[0]
        rsi = self.rsi[0]

        # Long Entry Conditions
        if close > ema50 and self.crossover[0] > 0 and rsi < 50.0:
            if rsi >= self.p.rsi_oversold:
                swing_low = min([self.data.low[-i] for i in range(self.p.swing_period)])
                sl_level = min(swing_low, ema50)
                risk_dist = close - sl_level
                if risk_dist > 0:
                    capital = self.broker.getvalue()
                    risk_amount = capital * self.p.risk_pct
                    size = int(risk_amount / risk_dist)
                    if size > 0:
                        tp_level = close + 2.0 * risk_dist
                        orders = self.buy_bracket(
                            size=size,
                            stopprice=sl_level,
                            limitprice=tp_level
                        )
                        self.order = orders[0]

        # Short Entry Conditions
        elif close < ema50 and self.crossover[0] < 0 and rsi > 50.0:
            if rsi <= self.p.rsi_overbought:
                swing_high = max([self.data.high[-i] for i in range(self.p.swing_period)])
                sl_level = max(swing_high, ema50)
                risk_dist = sl_level - close
                if risk_dist > 0:
                    capital = self.broker.getvalue()
                    risk_amount = capital * self.p.risk_pct
                    size = int(risk_amount / risk_dist)
                    if size > 0:
                        tp_level = close - 2.0 * risk_dist
                        orders = self.sell_bracket(
                            size=size,
                            stopprice=sl_level,
                            limitprice=tp_level
                        )
                        self.order = orders[0]
