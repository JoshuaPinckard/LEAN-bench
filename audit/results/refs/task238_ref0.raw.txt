import backtrader as bt


class ScalpingStrategy(bt.Strategy):
    params = dict(
        use_module_a=True,
        use_module_b=True,
        use_rsi_filter=True,
        use_macd_filter=True,
        use_atr_vol_filter=False,
        use_bbands_squeeze=False,
        risk_perc=0.01,
        initial_stop_atr_mult=1.5,
        trailing_atr_mult=1.0,
        tp_r_mult=2.0,
        time_stop_bars=10,
        atr_vol_sma_period=50,
        bb_width_sma_period=20,
    )

    def __init__(self):
        self.ema5 = bt.indicators.EMA(self.data, period=5)
        self.ema8 = bt.indicators.EMA(self.data, period=8)
        self.ema13 = bt.indicators.EMA(self.data, period=13)
        self.ema9 = bt.indicators.EMA(self.data, period=9)
        self.ema21 = bt.indicators.EMA(self.data, period=21)

        self.rsi = bt.indicators.RSI(self.data, period=14)

        self.macd = bt.indicators.MACD(
            self.data, period_me1=12, period_me2=26, period_signal=9
        )
        self.macd_hist = self.macd.macd - self.macd.signal

        self.bb = bt.indicators.BollingerBands(self.data, period=20, devfactor=2)
        self.bb_width = self.bb.top - self.bb.bot
        self.bb_width_sma = bt.indicators.SMA(
            self.bb_width, period=self.p.bb_width_sma_period
        )

        self.atr = bt.indicators.ATR(self.data, period=14)
        self.atr_sma = bt.indicators.SMA(self.atr, period=self.p.atr_vol_sma_period)

        self.order = None

        self.a_long_pb_high = None
        self.a_short_pb_low = None
        self.b_long_touch_high = None
        self.b_short_touch_low = None

        self.direction = None
        self.entry_price = None
        self.initial_stop_price = None
        self.tp_price = None
        self.extreme_price = None
        self.bars_in_trade = 0

    def _module_a_signals(self):
        long_sig = False
        short_sig = False

        bullish = (
            self.ema5[0] > self.ema8[0] > self.ema13[0]
            and (self.ema5[0] - self.ema8[0]) > (self.ema5[-1] - self.ema8[-1])
            and (self.ema8[0] - self.ema13[0]) > (self.ema8[-1] - self.ema13[-1])
        )
        if bullish:
            if self.data.low[0] <= self.ema5[0]:
                self.a_long_pb_high = self.data.high[0]
            elif self.a_long_pb_high is not None and self.data.high[0] > self.a_long_pb_high:
                rsi_ok = (not self.p.use_rsi_filter) or self.rsi[0] < 70
                macd_ok = (not self.p.use_macd_filter) or self.macd_hist[0] > 0
                atr_ok = (not self.p.use_atr_vol_filter) or self.atr[0] > self.atr_sma[0]
                bb_ok = (not self.p.use_bbands_squeeze) or self.bb_width[0] < self.bb_width_sma[0]
                if rsi_ok and macd_ok and atr_ok and bb_ok:
                    long_sig = True
                self.a_long_pb_high = None
        else:
            self.a_long_pb_high = None

        bearish = (
            self.ema5[0] < self.ema8[0] < self.ema13[0]
            and (self.ema8[0] - self.ema5[0]) > (self.ema8[-1] - self.ema5[-1])
            and (self.ema13[0] - self.ema8[0]) > (self.ema13[-1] - self.ema8[-1])
        )
        if bearish:
            if self.data.high[0] >= self.ema5[0]:
                self.a_short_pb_low = self.data.low[0]
            elif self.a_short_pb_low is not None and self.data.low[0] < self.a_short_pb_low:
                rsi_ok = (not self.p.use_rsi_filter) or self.rsi[0] > 30
                macd_ok = (not self.p.use_macd_filter) or self.macd_hist[0] < 0
                atr_ok = (not self.p.use_atr_vol_filter) or self.atr[0] > self.atr_sma[0]
                bb_ok = (not self.p.use_bbands_squeeze) or self.bb_width[0] < self.bb_width_sma[0]
                if rsi_ok and macd_ok and atr_ok and bb_ok:
                    short_sig = True
                self.a_short_pb_low = None
        else:
            self.a_short_pb_low = None

        return long_sig, short_sig

    def _module_b_signals(self):
        long_sig = False
        short_sig = False

        flag = self.bb_width[0] < self.bb_width_sma[0]
        touch = (
            self.data.low[0] <= self.ema9[0] <= self.data.high[0]
            or self.data.low[0] <= self.ema21[0] <= self.data.high[0]
        )

        long_trend = (
            self.ema9[0] > self.ema21[0]
            and (self.ema9[0] - self.ema21[0]) > (self.ema9[-1] - self.ema21[-1])
            and (self.ema9[-1] - self.ema21[-1]) > (self.ema9[-2] - self.ema21[-2])
        )
        if long_trend:
            if flag and touch:
                self.b_long_touch_high = self.data.high[0]
            elif self.b_long_touch_high is not None and self.data.high[0] > self.b_long_touch_high:
                rsi_ok = (not self.p.use_rsi_filter) or (40 <= self.rsi[0] <= 65)
                macd_ok = (not self.p.use_macd_filter) or self.macd_hist[0] > 0
                atr_ok = (not self.p.use_atr_vol_filter) or self.atr[0] > self.atr_sma[0]
                bb_ok = (not self.p.use_bbands_squeeze) or self.bb_width[0] < self.bb_width_sma[0]
                if rsi_ok and macd_ok and atr_ok and bb_ok:
                    long_sig = True
                self.b_long_touch_high = None
        else:
            self.b_long_touch_high = None

        short_trend = (
            self.ema9[0] < self.ema21[0]
            and (self.ema21[0] - self.ema9[0]) > (self.ema21[-1] - self.ema9[-1])
            and (self.ema21[-1] - self.ema9[-1]) > (self.ema21[-2] - self.ema9[-2])
        )
        if short_trend:
            if flag and touch:
                self.b_short_touch_low = self.data.low[0]
            elif self.b_short_touch_low is not None and self.data.low[0] < self.b_short_touch_low:
                rsi_ok = (not self.p.use_rsi_filter) or (35 <= self.rsi[0] <= 60)
                macd_ok = (not self.p.use_macd_filter) or self.macd_hist[0] < 0
                atr_ok = (not self.p.use_atr_vol_filter) or self.atr[0] > self.atr_sma[0]
                bb_ok = (not self.p.use_bbands_squeeze) or self.bb_width[0] < self.bb_width_sma[0]
                if rsi_ok and macd_ok and atr_ok and bb_ok:
                    short_sig = True
                self.b_short_touch_low = None
        else:
            self.b_short_touch_low = None

        return long_sig, short_sig

    def _reset_position_state(self):
        self.direction = None
        self.entry_price = None
        self.initial_stop_price = None
        self.tp_price = None
        self.extreme_price = None
        self.bars_in_trade = 0

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return

        if order.status == order.Completed:
            if self.position.size == 0:
                self._reset_position_state()
            elif order.isbuy():
                self.entry_price = order.executed.price
                dist = self.p.initial_stop_atr_mult * self.atr[0]
                self.initial_stop_price = self.entry_price - dist
                self.tp_price = self.entry_price + self.p.tp_r_mult * dist
                self.extreme_price = self.entry_price
                self.bars_in_trade = 0
                self.direction = 'long'
            else:
                self.entry_price = order.executed.price
                dist = self.p.initial_stop_atr_mult * self.atr[0]
                self.initial_stop_price = self.entry_price + dist
                self.tp_price = self.entry_price - self.p.tp_r_mult * dist
                self.extreme_price = self.entry_price
                self.bars_in_trade = 0
                self.direction = 'short'

        self.order = None

    def _enter(self, direction):
        dist = self.p.initial_stop_atr_mult * self.atr[0]
        if dist <= 0:
            return
        risk_amount = self.p.risk_perc * self.broker.getvalue()
        size = int(risk_amount / dist)
        if size < 1:
            return
        if direction == 'long':
            self.order = self.buy(size=size)
        else:
            self.order = self.sell(size=size)

    def _manage_position(self, opp_long_sig, opp_short_sig):
        self.bars_in_trade += 1
        if self.direction == 'long':
            self.extreme_price = max(self.extreme_price, self.data.high[0])
            trailing = max(
                self.ema21[0], self.extreme_price - self.p.trailing_atr_mult * self.atr[0]
            )
            stop_price = max(self.initial_stop_price, trailing)
            exit_now = (
                self.data.low[0] <= stop_price
                or self.data.high[0] >= self.tp_price
                or opp_short_sig
                or self.data.close[0] < self.ema13[0]
                or self.bars_in_trade >= self.p.time_stop_bars
            )
            if exit_now:
                self.order = self.close()
        else:
            self.extreme_price = min(self.extreme_price, self.data.low[0])
            trailing = min(
                self.ema21[0], self.extreme_price + self.p.trailing_atr_mult * self.atr[0]
            )
            stop_price = min(self.initial_stop_price, trailing)
            exit_now = (
                self.data.high[0] >= stop_price
                or self.data.low[0] <= self.tp_price
                or opp_long_sig
                or self.data.close[0] > self.ema13[0]
                or self.bars_in_trade >= self.p.time_stop_bars
            )
            if exit_now:
                self.order = self.close()

    def next(self):
        if self.order:
            return

        long_a = short_a = long_b = short_b = False
        if self.p.use_module_a:
            long_a, short_a = self._module_a_signals()
        if self.p.use_module_b:
            long_b, short_b = self._module_b_signals()

        long_sig = long_a or long_b
        short_sig = short_a or short_b

        if not self.position:
            if long_sig:
                self._enter('long')
            elif short_sig:
                self._enter('short')
        else:
            self._manage_position(long_sig, short_sig)
