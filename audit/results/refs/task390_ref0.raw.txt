import backtrader as bt


class TrendFilteredCrossoverStrategy(bt.Strategy):
    params = (
        ('size', 100),
    )

    def __init__(self):
        self.sma9 = bt.indicators.SMA(self.data.close, period=9)
        self.sma26 = bt.indicators.SMA(self.data.close, period=26)
        self.sma52 = bt.indicators.SMA(self.data.close, period=52)
        self.adx = bt.indicators.ADX(self.data, period=14)
        self.atr = bt.indicators.ATR(self.data, period=14)
        self.crossover = bt.indicators.CrossOver(self.sma9, self.sma26)

        self.order = None
        self.stop_order = None
        self.stop_price = None

    def _cancel_stop(self):
        if self.stop_order is not None:
            self.cancel(self.stop_order)

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status == order.Completed:
            if order == self.order:
                self.order = None
                if self.position:
                    if self.position.size > 0:
                        self.stop_price = self.position.price - 2.0 * self.atr[0]
                        self.stop_order = self.sell(
                            exectype=bt.Order.Stop,
                            price=self.stop_price,
                            size=self.position.size,
                        )
                    elif self.position.size < 0:
                        self.stop_price = self.position.price + 2.0 * self.atr[0]
                        self.stop_order = self.buy(
                            exectype=bt.Order.Stop,
                            price=self.stop_price,
                            size=abs(self.position.size),
                        )
            elif order == self.stop_order:
                self.stop_order = None
                self.stop_price = None
        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            if order == self.order:
                self.order = None
            elif order == self.stop_order:
                self.stop_order = None

    def next(self):
        if self.order:
            return

        long_signal = (
            self.crossover[0] > 0
            and self.data.close[0] > self.sma52[0]
            and self.adx[0] > 20
        )
        short_signal = (
            self.crossover[0] < 0
            and self.data.close[0] < self.sma52[0]
            and self.adx[0] > 20
        )

        if not self.position:
            if long_signal:
                self.order = self.buy(size=self.p.size)
            elif short_signal:
                self.order = self.sell(size=self.p.size)
            return

        if self.position.size > 0:
            if short_signal:
                self._cancel_stop()
                self.order = self.sell(size=self.position.size + self.p.size)
                return
            candidate = self.data.close[0] - 2.0 * self.atr[0]
            if self.stop_price is None or candidate > self.stop_price:
                self.stop_price = candidate
                self._cancel_stop()
                self.stop_order = self.sell(
                    exectype=bt.Order.Stop,
                    price=self.stop_price,
                    size=self.position.size,
                )
        else:
            if long_signal:
                self._cancel_stop()
                self.order = self.buy(size=abs(self.position.size) + self.p.size)
                return
            candidate = self.data.close[0] + 2.0 * self.atr[0]
            if self.stop_price is None or candidate < self.stop_price:
                self.stop_price = candidate
                self._cancel_stop()
                self.stop_order = self.buy(
                    exectype=bt.Order.Stop,
                    price=self.stop_price,
                    size=abs(self.position.size),
                )
