import backtrader as bt


class TrendDayMACrossover(bt.Strategy):
    """
    Reference implementation of the 'Trend Day MA Crossover' Strategy.
    """

    def __init__(self):
        self.order = None
        self.seq_state = 0

        # Moving Averages
        self.ma20 = bt.ind.SMA(self.data.close, period=20)
        self.ma30 = bt.ind.SMA(self.data.close, period=30)
        self.ma60 = bt.ind.SMA(self.data.close, period=60)

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        if self.order:
            return

        if not self.position:
            cross_ma20 = self.data.close[0] > self.ma20[0] and self.data.close[-1] <= self.ma20[-1]
            cross_ma30 = self.data.close[0] > self.ma30[0] and self.data.close[-1] <= self.ma30[-1]

            # Sequence Step 1: Price crosses and closes above MA20
            if self.seq_state == 0:
                if cross_ma20:
                    self.seq_state = 1

            # Sequence Step 2: After that, price crosses and closes above MA30
            if self.seq_state == 1:
                if cross_ma30:
                    self.seq_state = 2
                elif self.data.close[0] < self.ma20[0]:
                    self.seq_state = 0

            # Sequence Step 3 & 4: Price above MA60 with increasing volume and growing bar size
            if self.seq_state == 2:
                above_ma60 = self.data.close[0] > self.ma60[0]
                vol_increasing = self.data.volume[0] > self.data.volume[-1]
                bar_growing = (self.data.high[0] - self.data.low[0]) > (self.data.high[-1] - self.data.low[-1])

                if above_ma60 and vol_increasing and bar_growing:
                    self.order = self.buy()
                    self.seq_state = 0
                elif self.data.close[0] < self.ma30[0]:
                    self.seq_state = 0
        else:
            self.seq_state = 0

            # Take-profit: Price reaches level 0.278 or 0.28
            tp_hit = self.data.high[0] >= 0.278

            # Stop-loss: Price closes below level 0.265 on increasing volume (red bar)
            is_red_bar = self.data.close[0] < self.data.open[0]
            vol_inc = self.data.volume[0] > self.data.volume[-1]
            sl_hit = (self.data.close[0] < 0.265) and is_red_bar and vol_inc

            if tp_hit or sl_hit:
                self.order = self.close()
