import backtrader as bt


class RSIFailureSwingStrategy(bt.Strategy):
    def __init__(self):
        self.rsi = bt.indicators.RSI(self.data, period=14)
        self.addminperiod(2)

        self.entry_order = None
        self.exit_order = None
        self.stop_order = None

        self.long_state = 0
        self.rsi_low1 = None
        self.rsi_low2 = None

        self.short_state = 0
        self.rsi_high1 = None
        self.rsi_high2 = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order == self.entry_order:
            if order.status == order.Completed:
                if order.isbuy():
                    stop_price = order.executed.price * 0.98
                    self.stop_order = self.sell(exectype=bt.Order.Stop, price=stop_price)
                else:
                    stop_price = order.executed.price * 1.02
                    self.stop_order = self.buy(exectype=bt.Order.Stop, price=stop_price)
            self.entry_order = None
        elif order == self.exit_order:
            self.exit_order = None
        elif order == self.stop_order:
            self.stop_order = None

    def next(self):
        if self.entry_order or self.exit_order:
            return

        rsi0 = self.rsi[0]
        rsi1 = self.rsi[-1]
        rsi2 = self.rsi[-2]

        if self.position:
            if self.position.size > 0 and rsi1 < 70 <= rsi0:
                if self.stop_order:
                    self.cancel(self.stop_order)
                self.exit_order = self.close()
            elif self.position.size < 0 and rsi0 <= 30 < rsi1:
                if self.stop_order:
                    self.cancel(self.stop_order)
                self.exit_order = self.close()
            return

        is_trough = rsi2 > rsi1 and rsi1 < rsi0
        is_peak = rsi2 < rsi1 and rsi1 > rsi0

        if is_trough and rsi1 < 30:
            if self.long_state == 0:
                self.rsi_low1 = rsi1
                self.long_state = 1
            elif self.long_state == 2:
                if rsi1 > self.rsi_low1:
                    self.rsi_low2 = rsi1
                    self.long_state = 3

        if self.long_state == 1 and rsi1 < 30 <= rsi0:
            self.long_state = 2

        if self.long_state == 3 and rsi1 < 30 <= rsi0:
            self.entry_order = self.buy()
            self.long_state = 0

        if is_peak and rsi1 > 70:
            if self.short_state == 0:
                self.rsi_high1 = rsi1
                self.short_state = 1
            elif self.short_state == 2:
                if rsi1 < self.rsi_high1:
                    self.rsi_high2 = rsi1
                    self.short_state = 3

        if self.short_state == 1 and rsi0 <= 70 < rsi1:
            self.short_state = 2

        if self.short_state == 3 and rsi0 <= 70 < rsi1:
            self.entry_order = self.sell()
            self.short_state = 0
