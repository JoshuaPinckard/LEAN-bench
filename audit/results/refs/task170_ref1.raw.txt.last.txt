import backtrader as bt


class SwingBreakoutStrategy(bt.Strategy):
    def __init__(self):
        self.sma200 = bt.indicators.SMA(self.data.close, period=200)
        self.bbands = bt.indicators.BollingerBands(
            self.data.close, period=20, devfactor=2.0
        )
        self.atr = bt.indicators.ATR(self.data, period=14)
        self.volume_sma = bt.indicators.SMA(self.data.volume, period=20)
        self.resistance = bt.indicators.Highest(self.data.high, period=20)
        self.support = bt.indicators.Lowest(self.data.low, period=20)

        self.entry_order = None
        self.entry_price = None
        self.initial_stop = None

        self.exit_roles = {}
        self.slot_remaining = {"t1": 0.0, "t2": 0.0, "t3": 0.0}

        self.highest_close_since_entry = None
        self.trail_level = None
        self.t3_stop_order = None
        self.t3_stop_price = None
        self.t3_replacing = False

    def _bb_width(self, ago=0):
        return (self.bbands.top[ago] - self.bbands.bot[ago]) / self.bbands.mid[ago]

    def _low_volatility_consolidation(self):
        qualifying_bars = 0
        for ago in range(10):
            atr_condition = (self.atr[ago] / self.data.close[ago]) < 0.02
            bb_condition = self._bb_width(ago) < 0.04
            if atr_condition and bb_condition:
                qualifying_bars += 1
        return qualifying_bars >= 8

    def _place_t3_stop(self):
        if self.slot_remaining["t3"] <= 0 or self.position.size <= 0:
            return

        self.t3_stop_order = self.sell(
            size=self.slot_remaining["t3"],
            exectype=bt.Order.Stop,
            price=self.t3_stop_price,
        )
        self.exit_roles[self.t3_stop_order.ref] = "t3_stop"

    def _create_exit_orders(self, filled_size):
        third = filled_size / 3.0

        self.slot_remaining["t1"] = third
        self.slot_remaining["t2"] = third
        self.slot_remaining["t3"] = third

        t1_stop = self.sell(
            size=third,
            exectype=bt.Order.Stop,
            price=self.initial_stop,
        )
        self.exit_roles[t1_stop.ref] = "t1_stop"

        t1_target = self.sell(
            size=third,
            exectype=bt.Order.Limit,
            price=self.entry_price * 1.02,
            oco=t1_stop,
        )
        self.exit_roles[t1_target.ref] = "t1_target"

        t2_stop = self.sell(
            size=third,
            exectype=bt.Order.Stop,
            price=self.initial_stop,
        )
        self.exit_roles[t2_stop.ref] = "t2_stop"

        t2_target = self.sell(
            size=third,
            exectype=bt.Order.Limit,
            price=self.entry_price * 1.07,
            oco=t2_stop,
        )
        self.exit_roles[t2_target.ref] = "t2_target"

        self.highest_close_since_entry = self.data.close[0]
        self.trail_level = self.highest_close_since_entry - 3.0 * self.atr[0]
        self.t3_stop_price = max(self.initial_stop, self.trail_level)
        self._place_t3_stop()

    def notify_order(self, order):
        if order.status in [
            order.Submitted,
            order.Accepted,
        ]:
            return

        if order is self.entry_order:
            if order.status == order.Completed:
                self.entry_order = None
                self._create_exit_orders(order.executed.size)
                return

            if order.status in [
                order.Canceled,
                order.Margin,
                order.Rejected,
            ]:
                self.entry_order = None
                return

        role = self.exit_roles.get(order.ref)

        if order.status == order.Completed:
            if role == "t1_target" or role == "t1_stop":
                self.slot_remaining["t1"] = 0.0
            elif role == "t2_target" or role == "t2_stop":
                self.slot_remaining["t2"] = 0.0
            elif role == "t3_stop":
                self.slot_remaining["t3"] = 0.0
                self.t3_stop_order = None
                self.t3_replacing = False

        elif order.status in [
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            if role == "t3_stop":
                self.t3_stop_order = None
                if (
                    self.t3_replacing
                    and self.slot_remaining["t3"] > 0
                    and self.position.size > 0
                ):
                    self.t3_replacing = False
                    self._place_t3_stop()

    def next(self):
        if self.entry_order is not None:
            volatility_jump = (self.atr[0] / self.data.close[0]) >= 0.025
            bb_expansion = self._bb_width() >= 0.06
            gap_above_entry = self.data.open[0] > self.entry_price
            stop_reached = self.data.low[0] <= self.initial_stop

            if gap_above_entry or volatility_jump or bb_expansion or stop_reached:
                self.cancel(self.entry_order)
            return

        if self.position.size > 0:
            if self.data.close[0] > self.highest_close_since_entry:
                self.highest_close_since_entry = self.data.close[0]

            candidate_trail = self.highest_close_since_entry - 3.0 * self.atr[0]
            if candidate_trail > self.trail_level:
                self.trail_level = candidate_trail

            desired_stop = max(self.initial_stop, self.trail_level)
            if (
                self.slot_remaining["t3"] > 0
                and self.t3_stop_order is not None
                and desired_stop > self.t3_stop_price
                and not self.t3_replacing
            ):
                self.t3_stop_price = desired_stop
                self.t3_replacing = True
                self.cancel(self.t3_stop_order)
            return

        if (
            self.data.close[0] > self.sma200[0]
            and self._low_volatility_consolidation()
            and self.data.volume[0] > self.volume_sma[0] * 1.3
        ):
            self.entry_price = self.resistance[0] + max(
                0.001 * self.data.close[0],
                0.25 * self.atr[0],
            )
            self.initial_stop = self.support[0] - 0.25 * self.atr[0]
            self.entry_order = self.buy(
                exectype=bt.Order.Stop,
                price=self.entry_price,
            )
