import backtrader as bt


class RegimeSwitchingStrategy(bt.Strategy):
    params = dict(
        sma_trend_period=200,
        sma_signal_period=50,
        rsi_period=14,
        atr_period=14,
        atr_baseline_period=50,
        stddev_period=20,
        sideways_proximity_pct=0.02,
        entry_max_dev_pct=0.05,
        rsi_exit_long=30,
        rsi_exit_short=70,
        stop_atr_mult=1.5,
        target_stop_mult=3.0,
        position_pct=0.01,
    )

    def __init__(self):
        self.sma200 = bt.indicators.SimpleMovingAverage(
            self.data.close, period=self.p.sma_trend_period)
        self.sma50 = bt.indicators.SimpleMovingAverage(
            self.data.close, period=self.p.sma_signal_period)
        self.rsi = bt.indicators.RSI(self.data.close, period=self.p.rsi_period)
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.atr_baseline = bt.indicators.SimpleMovingAverage(
            self.atr, period=self.p.atr_baseline_period)
        self.stddev20 = bt.indicators.StandardDeviation(
            self.data.close, period=self.p.stddev_period)
        self.cross50 = bt.indicators.CrossOver(self.data.close, self.sma50)

        self.order = None
        self.exit_order = None
        self.stop_order = None
        self.limit_order = None

    def _get_regime(self):
        price = self.data.close[0]
        sma200 = self.sma200[0]
        atr_high = self.atr[0] > self.atr_baseline[0]
        proximity = abs(price - sma200) / sma200

        if price > sma200 and self.rsi[0] > 50 and atr_high:
            return 1
        if price < sma200 and self.rsi[0] < 50 and atr_high:
            return 2
        if proximity < self.p.sideways_proximity_pct and atr_high:
            return 3
        return 4

    def _place_protective_orders(self, order):
        entry_price = order.executed.price
        entry_atr = self.atr[0]
        stop_distance = self.p.stop_atr_mult * entry_atr
        target_distance = self.p.target_stop_mult * stop_distance
        size = abs(order.executed.size)

        if order.isbuy():
            stop_price = entry_price - stop_distance
            target_price = entry_price + target_distance
            self.stop_order = self.sell(
                exectype=bt.Order.Stop, price=stop_price, size=size)
            self.limit_order = self.sell(
                exectype=bt.Order.Limit, price=target_price, size=size)
        else:
            stop_price = entry_price + stop_distance
            target_price = entry_price - target_distance
            self.stop_order = self.buy(
                exectype=bt.Order.Stop, price=stop_price, size=size)
            self.limit_order = self.buy(
                exectype=bt.Order.Limit, price=target_price, size=size)

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status == order.Completed:
            if order == self.order:
                self._place_protective_orders(order)
            elif order == self.stop_order and self.limit_order is not None:
                self.cancel(self.limit_order)
            elif order == self.limit_order and self.stop_order is not None:
                self.cancel(self.stop_order)

        if order == self.order:
            self.order = None
        elif order == self.exit_order:
            self.exit_order = None
        elif order == self.stop_order:
            self.stop_order = None
        elif order == self.limit_order:
            self.limit_order = None

    def next(self):
        if self.order or self.exit_order:
            return

        price = self.data.close[0]
        sma200 = self.sma200[0]
        dev_from_sma200 = abs(price - sma200) / sma200

        if self.position.size == 0:
            regime = self._get_regime()

            if regime == 1 and self.cross50[0] > 0:
                if dev_from_sma200 <= self.p.entry_max_dev_pct:
                    size = (self.p.position_pct * self.broker.getvalue()) / price
                    if size > 0:
                        self.order = self.buy(size=size)

            elif regime == 2 and self.cross50[0] < 0:
                if dev_from_sma200 <= self.p.entry_max_dev_pct:
                    size = (self.p.position_pct * self.broker.getvalue()) / price
                    if size > 0:
                        self.order = self.sell(size=size)

        elif self.position.size > 0:
            if self.cross50[0] < 0 or self.rsi[0] < self.p.rsi_exit_long:
                if self.stop_order is not None:
                    self.cancel(self.stop_order)
                if self.limit_order is not None:
                    self.cancel(self.limit_order)
                self.exit_order = self.close()

        elif self.position.size < 0:
            if self.cross50[0] > 0 or self.rsi[0] > self.p.rsi_exit_short:
                if self.stop_order is not None:
                    self.cancel(self.stop_order)
                if self.limit_order is not None:
                    self.cancel(self.limit_order)
                self.exit_order = self.close()
