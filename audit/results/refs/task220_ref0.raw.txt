import backtrader as bt


class AggressiveCombinedStrategy(bt.Strategy):
    params = (
        ('bb_period', 20),
        ('bb_dev', 2),
        ('rsi_period', 14),
        ('macd_fast', 12),
        ('macd_slow', 26),
        ('macd_signal', 9),
        ('ema_period', 50),
        ('atr_period', 14),
        ('rsi_low', 40),
        ('rsi_high', 60),
        ('rsi_oversold', 30),
        ('rsi_overbought', 70),
        ('scalp_tp_atr_mult', 1.25),
        ('scalp_stop_atr_mult', 1.0),
        ('scalp_time_stop', 4),
        ('scalp_extreme_lookback', 5),
        ('swing_tp_atr_mult', 2.5),
        ('swing_stop_atr_mult', 1.0),
        ('swing_time_stop', 7),
        ('pullback_atr_mult', 0.5),
        ('vol_filter_threshold', 0.03),
        ('risk_pct', 0.015),
    )

    def __init__(self):
        self.bb = bt.indicators.BollingerBands(
            self.data.close, period=self.p.bb_period, devfactor=self.p.bb_dev)
        self.rsi = bt.indicators.RSI(self.data.close, period=self.p.rsi_period)
        self.macd = bt.indicators.MACD(
            self.data.close,
            period_me1=self.p.macd_fast,
            period_me2=self.p.macd_slow,
            period_signal=self.p.macd_signal)
        self.ema50 = bt.indicators.EMA(self.data.close, period=self.p.ema_period)
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)

        self.cross_lower = bt.indicators.CrossOver(self.data.close, self.bb.bot)
        self.cross_upper = bt.indicators.CrossOver(self.bb.top, self.data.close)
        self.macd_cross = bt.indicators.CrossOver(self.macd.macd, self.macd.signal)

        self.addminperiod(2)

        self.order = None
        self.position_type = None
        self.entry_bar = None
        self.stop_price = None
        self.tp_atr_price = None

    def notify_order(self, order):
        if order.status in (order.Completed, order.Canceled, order.Margin, order.Rejected):
            self.order = None

    def _hist(self, ago=0):
        return self.macd.macd[ago] - self.macd.signal[ago]

    def _size_for_stop(self, stop_distance):
        if stop_distance <= 0:
            return 0
        value = self.broker.getvalue()
        risk_amount = self.p.risk_pct * value
        return int(risk_amount / stop_distance)

    def _enter_scalp(self, is_long):
        lb = self.p.scalp_extreme_lookback
        atr0 = self.atr[0]
        close = self.data.close[0]
        if is_long:
            local_extreme = min(self.data.low.get(size=lb))
            stop_price = local_extreme - self.p.scalp_stop_atr_mult * atr0
            stop_distance = close - stop_price
            size = self._size_for_stop(stop_distance)
            if size <= 0:
                return
            self.order = self.buy(size=size)
            self.tp_atr_price = close + self.p.scalp_tp_atr_mult * atr0
        else:
            local_extreme = max(self.data.high.get(size=lb))
            stop_price = local_extreme + self.p.scalp_stop_atr_mult * atr0
            stop_distance = stop_price - close
            size = self._size_for_stop(stop_distance)
            if size <= 0:
                return
            self.order = self.sell(size=size)
            self.tp_atr_price = close - self.p.scalp_tp_atr_mult * atr0

        self.position_type = 'scalp_long' if is_long else 'scalp_short'
        self.entry_bar = len(self)
        self.stop_price = stop_price

    def _enter_swing(self, is_long):
        atr0 = self.atr[0]
        close = self.data.close[0]
        stop_distance = self.p.swing_stop_atr_mult * atr0
        size = self._size_for_stop(stop_distance)
        if size <= 0:
            return
        if is_long:
            self.order = self.buy(size=size)
            self.tp_atr_price = close + self.p.swing_tp_atr_mult * atr0
        else:
            self.order = self.sell(size=size)
            self.tp_atr_price = close - self.p.swing_tp_atr_mult * atr0

        self.position_type = 'swing_long' if is_long else 'swing_short'
        self.entry_bar = len(self)
        self.stop_price = None

    def _manage_position(self):
        close = self.data.close[0]
        bars_held = len(self) - self.entry_bar

        if self.position_type == 'scalp_long':
            if close <= self.stop_price or close >= self.bb.mid[0] or close >= self.tp_atr_price:
                self.order = self.close()
            elif bars_held >= self.p.scalp_time_stop:
                self.order = self.close()

        elif self.position_type == 'scalp_short':
            if close >= self.stop_price or close <= self.bb.mid[0] or close <= self.tp_atr_price:
                self.order = self.close()
            elif bars_held >= self.p.scalp_time_stop:
                self.order = self.close()

        elif self.position_type == 'swing_long':
            if close >= self.tp_atr_price or close >= self.bb.top[0]:
                self.order = self.close()
            elif self.macd_cross[0] == -1:
                self.order = self.close()
            elif bars_held >= self.p.swing_time_stop:
                self.order = self.close()

        elif self.position_type == 'swing_short':
            if close <= self.tp_atr_price or close <= self.bb.bot[0]:
                self.order = self.close()
            elif self.macd_cross[0] == 1:
                self.order = self.close()
            elif bars_held >= self.p.swing_time_stop:
                self.order = self.close()

    def next(self):
        if self.order:
            return

        if self.position:
            self._manage_position()
            return

        atr = self.atr[0]
        close = self.data.close[0]
        if (atr / close) > self.p.vol_filter_threshold:
            return

        rsi = self.rsi[0]
        hist0 = self._hist(0)
        hist1 = self._hist(-1)
        hist2 = self._hist(-2)

        if self.cross_lower[0] == 1 and self.p.rsi_low <= rsi <= self.p.rsi_high:
            momentum_long = (self.macd_cross[0] == 1) or (hist0 > hist1 > hist2)
            if momentum_long:
                self._enter_scalp(is_long=True)
                return

        if self.cross_upper[0] == 1 and self.p.rsi_low <= rsi <= self.p.rsi_high:
            momentum_short = (self.macd_cross[0] == -1) or (hist0 < hist1 < hist2)
            if momentum_short:
                self._enter_scalp(is_long=False)
                return

        trend_up = close > self.ema50[0] and self.ema50[0] > self.ema50[-1]
        if trend_up:
            macd_bull = self.macd.macd[0] > self.macd.signal[0] and self.macd.macd[0] > 0
            pullback = abs(close - self.bb.mid[0]) <= self.p.pullback_atr_mult * atr
            rsi_ok = (self.p.rsi_low <= rsi <= self.p.rsi_high) or \
                     (self.rsi[-1] < self.p.rsi_oversold and rsi > self.p.rsi_low)
            if macd_bull and pullback and rsi_ok:
                self._enter_swing(is_long=True)
                return

        trend_down = close < self.ema50[0] and self.ema50[0] < self.ema50[-1]
        if trend_down:
            macd_bear = self.macd.macd[0] < self.macd.signal[0] and self.macd.macd[0] < 0
            pullback = abs(close - self.bb.mid[0]) <= self.p.pullback_atr_mult * atr
            rsi_ok = (self.p.rsi_low <= rsi <= self.p.rsi_high) or \
                     (self.rsi[-1] > self.p.rsi_overbought and rsi < self.p.rsi_high)
            if macd_bear and pullback and rsi_ok:
                self._enter_swing(is_long=False)
                return
