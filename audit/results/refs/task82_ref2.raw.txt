import backtrader as bt


class TrailingStopShortStrategy(bt.Strategy):
    """
    Trailing Stop-Loss Strategy for Short Positions.

    Opens a short position of 10 units at 13:10 UTC and manages it using a
    trailing stop-loss set at the maximum price reached since activation minus
    the trail amount of 0.19 price units.
    """

    params = (
        ("size", 10),
        ("trail_amount", 0.19),
        ("entry_hour", 13),
        ("entry_minute", 10),
    )

    def __init__(self):
        self.order = None
        self.highest_price = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            self.order = None

        if order.status == order.Completed:
            if self.position:
                self.highest_price = self.data.high[0]
            else:
                self.highest_price = None

    def next(self):
        if self.order:
            return

        if not self.position:
            dt_time = self.data.datetime.time(0)
            if (
                dt_time.hour == self.p.entry_hour
                and dt_time.minute == self.p.entry_minute
            ):
                self.order = self.sell(size=self.p.size)
        else:
            if self.highest_price is None:
                self.highest_price = self.data.high[0]
            else:
                self.highest_price = max(self.highest_price, self.data.high[0])

            stop_level = self.highest_price - self.p.trail_amount

            if self.data.low[0] <= stop_level:
                self.order = self.close()
