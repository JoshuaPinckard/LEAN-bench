import backtrader as bt


class RSIStrategy(bt.Strategy):
    params = (
        ("rsi_period", 14),
        ("overbought", 70.0),
        ("oversold", 30.0),
        ("stop_loss", 0.02),
        ("take_profit", 0.04),
    )

    def __init__(self):
        self.rsi = bt.indicators.RSI(
            self.data.close,
            period=self.p.rsi_period
        )
        self.order = None
        self.sl_order = None
        self.tp_order = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            if order == self.order:
                self.order = None
                entry_price = order.executed.price
                if self.position.size > 0:
                    sl_price = entry_price * (1.0 - self.p.stop_loss)
                    tp_price = entry_price * (1.0 + self.p.take_profit)
                    self.tp_order = self.sell(
                        exectype=bt.Order.Limit,
                        price=tp_price
                    )
                    self.sl_order = self.sell(
                        exectype=bt.Order.Stop,
                        price=sl_price,
                        oco=self.tp_order
                    )
                elif self.position.size < 0:
                    sl_price = entry_price * (1.0 + self.p.stop_loss)
                    tp_price = entry_price * (1.0 - self.p.take_profit)
                    self.tp_order = self.buy(
                        exectype=bt.Order.Limit,
                        price=tp_price
                    )
                    self.sl_order = self.buy(
                        exectype=bt.Order.Stop,
                        price=sl_price,
                        oco=self.tp_order
                    )
            elif order in (self.sl_order, self.tp_order):
                if order == self.sl_order:
                    self.sl_order = None
                if order == self.tp_order:
                    self.tp_order = None

        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            if order == self.order:
                self.order = None
            if order == self.sl_order:
                self.sl_order = None
            if order == self.tp_order:
                self.tp_order = None

    def next(self):
        if self.order:
            return

        if not self.position:
            # Open long: RSI falls below 30 and then rises back above 30
            if self.rsi[-1] < self.p.oversold and self.rsi[0] > self.p.oversold:
                self.order = self.buy()
            # Open short: RSI rises above 70 and then falls back below 70
            elif self.rsi[-1] > self.p.overbought and self.rsi[0] < self.p.overbought:
                self.order = self.sell()
        else:
            if self.position.size > 0:
                # Close long position when RSI exceeds 70
                if self.rsi[0] > self.p.overbought:
                    self._cancel_sl_tp()
                    self.order = self.close()
            elif self.position.size < 0:
                # Close short position when RSI falls below 30
                if self.rsi[0] < self.p.oversold:
                    self._cancel_sl_tp()
                    self.order = self.close()

    def _cancel_sl_tp(self):
        if self.sl_order:
            self.cancel(self.sl_order)
            self.sl_order = None
        if self.tp_order:
            self.cancel(self.tp_order)
            self.tp_order = None
