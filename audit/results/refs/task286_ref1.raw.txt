import backtrader as bt


class RollingCoveredCalls(bt.Strategy):
    def __init__(self):
        self.bb = bt.indicators.BollingerBands(
            self.data.close, period=20, devfactor=2
        )
        self.rsi = bt.indicators.RSI(self.data.close, period=14)
        self.atr = bt.indicators.ATR(self.data, period=14)
        self.volume_sma = bt.indicators.SimpleMovingAverage(
            self.data.volume, period=20
        )
        self.cross_upper = bt.indicators.CrossOver(
            self.data.close, self.bb.lines.top
        )

        self.order = None
        self.core_started = False
        self.core_size = 0
        self.active_tranches = []
        self.last_tranche_bar = None
        self.assigned = False

    def notify_order(self, order):
        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            if order.status == order.Completed:
                action = getattr(order, "_rolling_action", None)

                if action == "core_entry":
                    self.core_size = abs(order.executed.size)
                    self.assigned = False

                elif action == "tranche_sale":
                    self.active_tranches.append({
                        "size": abs(order.executed.size),
                        "strike": order._rolling_strike,
                        "expires": order._rolling_expires,
                    })

                elif action == "buyback":
                    self.active_tranches = []

                elif action == "assignment":
                    self.active_tranches = []
                    self.assigned = True

            if order == self.order:
                self.order = None

    def _active_sold(self):
        return sum(tranche["size"] for tranche in self.active_tranches)

    def _can_sell_tranche(self):
        if self.order is not None:
            return False
        if self.position.size <= 0:
            return False
        if self.last_tranche_bar == len(self):
            return False
        return self._active_sold() < self.core_size

    def _sell_tranche(self):
        if not self._can_sell_tranche():
            return

        size = int(self.position.size * 0.20)
        if self.position.size >= 100:
            size = max(size, 100)

        remaining_capacity = self.core_size - self._active_sold()
        size = min(size, self.position.size, remaining_capacity)

        if size <= 0:
            return

        self.order = self.sell(size=size)
        self.order._rolling_action = "tranche_sale"
        self.order._rolling_strike = self.bb.lines.top[0]
        self.order._rolling_expires = len(self) + 7
        self.last_tranche_bar = len(self)

    def next(self):
        if not self.core_started:
            self.order = self.buy()
            self.order._rolling_action = "core_entry"
            self.core_started = True
            return

        if self.order is not None:
            return

        if self.assigned:
            if (
                self.position.size == 0
                and self.data.close[0] < self.bb.lines.mid[0]
                and self.rsi[0] < 50
            ):
                self.order = self.buy()
                self.order._rolling_action = "core_entry"
            return

        if self.active_tranches:
            if (
                self.data.close[0] < self.bb.lines.mid[0]
                or self.rsi[0] < 45
            ):
                size = self._active_sold()
                if size > 0:
                    self.order = self.buy(size=size)
                    self.order._rolling_action = "buyback"
                return

            if self.data.close[0] > self.bb.lines.top[0] + self.atr[0]:
                self.order = self.close()
                self.order._rolling_action = "assignment"
                return

            rolled = False
            for tranche in self.active_tranches:
                if (
                    len(self) == tranche["expires"]
                    and self.data.close[0] > self.bb.lines.top[0]
                ):
                    tranche["strike"] = self.bb.lines.top[0]
                    tranche["expires"] += 7
                    rolled = True

            if rolled and self.rsi[0] > 60:
                self._sell_tranche()
                return

        if (
            self.cross_upper[0] > 0
            and self.rsi[0] > 60
            and self.data.volume[0] > self.volume_sma[0]
        ):
            self._sell_tranche()
