import backtrader as bt
import math


class BollingerBandsMATrendFollowingSystem(bt.Strategy):
    params = (
        ('risk_per_trade', 0.02),
    )

    def __init__(self):
        self.bbands = bt.indicators.BollingerBands(
            self.data.close, period=20, devfactor=2.0
        )
        self.ma44 = bt.indicators.SimpleMovingAverage(
            self.data.close, period=44
        )

        self.band_width = self.bbands.top - self.bbands.bot
        self.avg_band_width = bt.indicators.SimpleMovingAverage(
            self.band_width, period=20
        )

        self.order = None

    def notify_order(self, order):
        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            if order is self.order:
                self.order = None

    def next(self):
        if self.order:
            return

        close = self.data.close[0]
        upper = self.bbands.top[0]
        lower = self.bbands.bot[0]
        ma = self.ma44[0]

        in_consolidation = self.band_width[0] < (0.20 * self.avg_band_width[0])

        if self.position:
            if self.position.size > 0:
                if close < upper or (
                    self.data.close[-1] >= self.ma44[-1] and close < ma
                ):
                    self.order = self.close()
            else:
                if close > lower or (
                    self.data.close[-1] <= self.ma44[-1] and close > ma
                ):
                    self.order = self.close()
            return

        if in_consolidation:
            return

        long_breakout = (
            close > ma
            and self.data.close[-1] <= self.bbands.top[-1]
            and close > upper
        )
        short_breakout = (
            close < ma
            and self.data.close[-1] >= self.bbands.bot[-1]
            and close < lower
        )

        if long_breakout:
            risk_per_share = close - lower
            if risk_per_share > 0:
                size = math.floor(
                    (self.broker.getvalue() * self.p.risk_per_trade) / risk_per_share
                )
                if size > 0:
                    self.order = self.buy_bracket(
                        size=size,
                        exectype=bt.Order.Market,
                        stopprice=lower,
                        stopexec=bt.Order.Stop,
                    )[0]

        elif short_breakout:
            risk_per_share = upper - close
            if risk_per_share > 0:
                size = math.floor(
                    (self.broker.getvalue() * self.p.risk_per_trade) / risk_per_share
                )
                if size > 0:
                    self.order = self.sell_bracket(
                        size=size,
                        exectype=bt.Order.Market,
                        stopprice=upper,
                        stopexec=bt.Order.Stop,
                    )[0]
