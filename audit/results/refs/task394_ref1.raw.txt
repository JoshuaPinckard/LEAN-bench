class ParabolicSARVolumeBreakout(bt.Strategy):
    params = (
        ('acceleration', 0.02),
        ('max_acceleration', 0.2),
        ('volume_period', 20),
    )

    def __init__(self):
        self.sar = bt.indicators.ParabolicSAR(
            self.data,
            af=self.p.acceleration,
            afmax=self.p.max_acceleration,
        )
        self.volume_sma = bt.indicators.SimpleMovingAverage(
            self.data.volume,
            period=self.p.volume_period,
        )
        self.order = None

    def notify_order(self, order):
        if order.status in (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            self.order = None

    def next(self):
        if self.order:
            return

        close = self.data.close[0]
        sar = self.sar[0]
        high_volume = self.data.volume[0] > self.volume_sma[0]

        if self.position.size > 0:
            if close < sar:
                self.order = self.close()
            return

        if self.position.size < 0:
            if close > sar:
                self.order = self.close()
            return

        if high_volume and close > sar and self.data.close[-1] <= self.sar[-1]:
            self.order = self.buy()
        elif high_volume and close < sar and self.data.close[-1] >= self.sar[-1]:
            self.order = self.sell()
