import backtrader as bt
import math


class QuantValueStrategy(bt.Strategy):
    def __init__(self):
        self.sma50 = bt.indicators.SMA(self.data.close, period=50)
        self.sma200 = bt.indicators.SMA(self.data.close, period=200)
        self.cross = bt.indicators.CrossOver(self.sma50, self.sma200)

        self.bbands = bt.indicators.BollingerBands(self.data.close, period=20, devfactor=2)
        self.rsi = bt.indicators.RSI(self.data.close, period=14)
        self.vol_sma20 = bt.indicators.SMA(self.data.volume, period=20)
        self.vol_sma50 = bt.indicators.SMA(self.data.volume, period=50)
        self.atr = bt.indicators.ATR(self.data, period=14)

        self.order = None
        self.entry_price = None

    def notify_order(self, order):
        if order.status == order.Completed:
            if order.isbuy():
                self.entry_price = order.executed.price
            elif not self.position:
                self.entry_price = None

        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        if self.order:
            return

        close = self.data.close[0]
        atr_ratio = self.atr[0] / close
        upper = self.bbands.top[0]
        middle = self.bbands.mid[0]
        lower = self.bbands.bot[0]
        band_width = upper - lower
        percent_b = (close - lower) / band_width if band_width != 0 else float("inf")

        if self.position:
            if (
                close >= self.entry_price * 1.20
                or close <= self.entry_price * 0.90
                or self.cross[0] < 0
                or self.rsi[0] >= 70
                or close >= upper
                or atr_ratio > 0.06
            ):
                self.order = self.close()
            return

        golden_cross_recent = any(self.cross[-i] > 0 for i in range(5))

        if (
            close > self.sma200[0]
            and golden_cross_recent
            and close < middle
            and percent_b <= 0.2
            and 40 <= self.rsi[0] <= 60
            and self.vol_sma20[0] > 100000
            and self.vol_sma20[0] > self.vol_sma50[0]
            and atr_ratio <= 0.04
        ):
            capital = self.broker.getvalue()
            stop_distance = max(0.10 * close, 2 * self.atr[0])
            risk_size = (0.01 * capital) / stop_distance
            allocation_size = (0.25 * capital) / close
            size = math.floor(min(risk_size, allocation_size))

            if size > 0:
                self.order = self.buy(size=size)
