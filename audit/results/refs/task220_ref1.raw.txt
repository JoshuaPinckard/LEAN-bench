import backtrader as bt


class AggressiveCombinedStrategy(bt.Strategy):
    def __init__(self):
        self.bb = bt.indicators.BollingerBands(
            self.data.close, period=20, devfactor=2
        )
        self.rsi = bt.indicators.RSI(self.data.close, period=14)
        self.macd = bt.indicators.MACD(
            self.data.close, period_me1=12, period_me2=26, period_signal=9
        )
        self.ema = bt.indicators.EMA(self.data.close, period=50)
        self.atr = bt.indicators.ATR(self.data, period=14)

        self.close_lower_cross = bt.indicators.CrossOver(
            self.data.close, self.bb.bot
        )
        self.upper_close_cross = bt.indicators.CrossOver(
            self.bb.top, self.data.close
        )
        self.macd_cross = bt.indicators.CrossOver(
            self.macd.macd, self.macd.signal
        )
        self.histogram = self.macd.macd - self.macd.signal

        self.pending_order = None
        self.entry_order = None
        self.exit_order = None

        self.lower_outside = False
        self.upper_outside = False
        self.lower_extreme = None
        self.upper_extreme = None

        self.pending_style = None
        self.pending_stop = None
        self.pending_atr = None
        self.pending_mid = None

        self.position_style = None
        self.entry_price = None
        self.stop_price = None
        self.entry_atr = None
        self.entry_mid = None
        self.entry_bar = None

        self.risk_pct = 0.01
        self.scalp_atr_target = 1.0
        self.scalp_time_stop = 3
        self.swing_atr_target = 2.0
        self.swing_time_stop = 5

    def notify_order(self, order):
        terminal = (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        )

        if order.status not in terminal:
            return

        if self.pending_order is order:
            self.pending_order = None

        if order.status == order.Completed:
            if self.entry_order is order:
                self.position_style = self.pending_style
                self.entry_price = order.executed.price
                self.stop_price = self.pending_stop
                self.entry_atr = self.pending_atr
                self.entry_mid = self.pending_mid
                self.entry_bar = len(self)
                self.entry_order = None

            elif self.exit_order is order:
                self._clear_trade()
                self.exit_order = None
        else:
            if self.entry_order is order:
                self.entry_order = None
                self.pending_style = None
                self.pending_stop = None
                self.pending_atr = None
                self.pending_mid = None
            elif self.exit_order is order:
                self.exit_order = None

    def _clear_trade(self):
        self.position_style = None
        self.entry_price = None
        self.stop_price = None
        self.entry_atr = None
        self.entry_mid = None
        self.entry_bar = None

    def _size_for_risk(self, price, stop_price):
        stop_distance = abs(price - stop_price)
        if stop_distance <= 0:
            return 0

        commission_info = self.broker.getcommissioninfo(self.data)
        entry_commission = abs(commission_info.getcommission(1, price))
        exit_commission = abs(commission_info.getcommission(1, stop_price))
        risk_per_unit = stop_distance + entry_commission + exit_commission

        if risk_per_unit <= 0:
            return 0

        return (self.broker.getvalue() * self.risk_pct) / risk_per_unit

    def _enter(self, is_long, style, stop_price):
        size = self._size_for_risk(self.data.close[0], stop_price)
        if size <= 0:
            return

        self.pending_style = style
        self.pending_stop = stop_price
        self.pending_atr = self.atr[0]
        self.pending_mid = self.bb.mid[0]

        if is_long:
            self.entry_order = self.buy(size=size)
        else:
            self.entry_order = self.sell(size=size)

        self.pending_order = self.entry_order

    def _exit(self):
        self.exit_order = self.close()
        self.pending_order = self.exit_order

    def _manage_position(self):
        held_bars = len(self) - self.entry_bar
        is_long = self.position.size > 0

        if self.position_style == "scalp":
            if is_long:
                stop_hit = self.data.low[0] <= self.stop_price
                target_hit = (
                    self.data.high[0] >= self.bb.mid[0]
                    or self.data.high[0]
                    >= self.entry_price + self.scalp_atr_target * self.entry_atr
                )
            else:
                stop_hit = self.data.high[0] >= self.stop_price
                target_hit = (
                    self.data.low[0] <= self.bb.mid[0]
                    or self.data.low[0]
                    <= self.entry_price - self.scalp_atr_target * self.entry_atr
                )

            if stop_hit or target_hit or held_bars >= self.scalp_time_stop:
                self._exit()

        elif self.position_style == "swing":
            if is_long:
                stop_hit = self.data.low[0] <= self.stop_price
                target_hit = (
                    self.data.high[0]
                    >= self.entry_price + self.swing_atr_target * self.entry_atr
                    or self.data.high[0] >= self.bb.top[0]
                )
                reverse_cross = self.macd_cross[0] < 0
            else:
                stop_hit = self.data.high[0] >= self.stop_price
                target_hit = (
                    self.data.low[0]
                    <= self.entry_price - self.swing_atr_target * self.entry_atr
                    or self.data.low[0] <= self.bb.bot[0]
                )
                reverse_cross = self.macd_cross[0] > 0

            if (
                stop_hit
                or target_hit
                or reverse_cross
                or held_bars >= self.swing_time_stop
            ):
                self._exit()

    def next(self):
        if self.pending_order:
            return

        if self.position:
            self._manage_position()
            return

        was_lower_outside = self.lower_outside
        was_upper_outside = self.upper_outside
        lower_extreme = self.lower_extreme
        upper_extreme = self.upper_extreme

        if self.data.close[0] < self.bb.bot[0]:
            self.lower_outside = True
            self.lower_extreme = (
                self.data.low[0]
                if self.lower_extreme is None
                else min(self.lower_extreme, self.data.low[0])
            )
        else:
            self.lower_outside = False
            self.lower_extreme = None

        if self.data.close[0] > self.bb.top[0]:
            self.upper_outside = True
            self.upper_extreme = (
                self.data.high[0]
                if self.upper_extreme is None
                else max(self.upper_extreme, self.data.high[0])
            )
        else:
            self.upper_outside = False
            self.upper_extreme = None

        rsi_neutral = 40 <= self.rsi[0] <= 60
        long_hist_toward_zero = (
            self.histogram[-2] < self.histogram[-1] < self.histogram[0] <= 0
        )
        short_hist_toward_zero = (
            self.histogram[-2] > self.histogram[-1] > self.histogram[0] >= 0
        )

        scalp_long = (
            was_lower_outside
            and self.data.close[0] >= self.bb.bot[0]
            and self.close_lower_cross[0] > 0
            and rsi_neutral
            and (self.macd_cross[0] > 0 or long_hist_toward_zero)
        )
        scalp_short = (
            was_upper_outside
            and self.data.close[0] <= self.bb.top[0]
            and self.upper_close_cross[0] > 0
            and rsi_neutral
            and (self.macd_cross[0] < 0 or short_hist_toward_zero)
        )

        if scalp_long and lower_extreme is not None:
            self._enter(
                True, "scalp", lower_extreme - self.atr[0]
            )
            return

        if scalp_short and upper_extreme is not None:
            self._enter(
                False, "scalp", upper_extreme + self.atr[0]
            )
            return

        if self.data.close[0] == 0 or self.atr[0] / self.data.close[0] > 0.03:
            return

        long_rsi_ok = rsi_neutral or (
            self.rsi[-1] <= 40 and self.rsi[0] > 40
        )
        short_rsi_ok = rsi_neutral or (
            self.rsi[-1] >= 60 and self.rsi[0] < 60
        )

        swing_long = (
            self.data.close[0] > self.ema[0]
            and self.ema[0] > self.ema[-1]
            and self.macd.macd[0] > self.macd.signal[0]
            and self.macd.macd[0] > 0
            and abs(self.data.close[0] - self.bb.mid[0]) <= 0.5 * self.atr[0]
            and long_rsi_ok
        )
        swing_short = (
            self.data.close[0] < self.ema[0]
            and self.ema[0] < self.ema[-1]
            and self.macd.macd[0] < self.macd.signal[0]
            and self.macd.macd[0] < 0
            and abs(self.data.close[0] - self.bb.mid[0]) <= 0.5 * self.atr[0]
            and short_rsi_ok
        )

        if swing_long:
            self._enter(True, "swing", self.data.close[0] - self.atr[0])
        elif swing_short:
            self._enter(False, "swing", self.data.close[0] + self.atr[0])
