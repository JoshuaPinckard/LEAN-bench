import math

import backtrader as bt


class IntradaySmaCrossoverStrategy(bt.Strategy):
    params = (
        ('sma_fast_period', 10),
        ('sma_slow_period', 20),
        ('atr_period', 14),
        ('proximity_atr_mult', 0.5),
        ('stop_loss_pct', 0.01),
        ('risk_pct', 0.01),
        ('session_open_minutes', 9 * 60 + 30),
        ('session_close_minutes', 16 * 60),
        ('open_buffer_minutes', 5),
        ('close_buffer_minutes', 30),
        ('forced_close_buffer_minutes', 5),
    )

    def __init__(self):
        self.sma_fast = bt.indicators.SMA(self.data.close, period=self.p.sma_fast_period)
        self.sma_slow = bt.indicators.SMA(self.data.close, period=self.p.sma_slow_period)
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.crossover = bt.indicators.CrossOver(self.sma_fast, self.sma_slow)
        self.order = None
        self.entry_price = None

    def _minutes_of_day(self):
        t = self.data.datetime.time(0)
        return t.hour * 60 + t.minute

    def _calc_size(self, price):
        risk_per_share = price * self.p.stop_loss_pct
        if risk_per_share <= 0:
            return 0
        risk_amount = self.broker.getvalue() * self.p.risk_pct
        return int(math.floor(risk_amount / risk_per_share))

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return
        if order.status == order.Completed:
            self.entry_price = order.executed.price if self.position.size != 0 else None
        self.order = None

    def next(self):
        if self.order:
            return

        minutes = self._minutes_of_day()
        entry_cutoff_open = self.p.session_open_minutes + self.p.open_buffer_minutes
        entry_cutoff_close = self.p.session_close_minutes - self.p.close_buffer_minutes
        forced_close_minute = self.p.session_close_minutes - self.p.forced_close_buffer_minutes

        if self.position:
            if minutes >= forced_close_minute:
                self.order = self.close()
                return
            if self.position.size > 0:
                stop_price = self.entry_price * (1 - self.p.stop_loss_pct)
                if self.data.low[0] <= stop_price or self.crossover[0] < 0:
                    self.order = self.close()
            else:
                stop_price = self.entry_price * (1 + self.p.stop_loss_pct)
                if self.data.high[0] >= stop_price or self.crossover[0] > 0:
                    self.order = self.close()
            return

        if minutes < self.p.session_open_minutes or minutes >= self.p.session_close_minutes:
            return
        if minutes < entry_cutoff_open or minutes >= entry_cutoff_close:
            return

        close_price = self.data.close[0]
        proximity = self.p.proximity_atr_mult * self.atr[0]

        if self.crossover[0] > 0 and abs(close_price - self.sma_slow[0]) <= proximity:
            size = self._calc_size(close_price)
            if size > 0:
                self.order = self.buy(size=size)
        elif self.crossover[0] < 0 and abs(close_price - self.sma_slow[0]) <= proximity:
            size = self._calc_size(close_price)
            if size > 0:
                self.order = self.sell(size=size)
