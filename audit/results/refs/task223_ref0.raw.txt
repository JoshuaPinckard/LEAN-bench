import backtrader as bt


class TrendEnsembleStrategy(bt.Strategy):
    params = dict(
        macd1=12,
        macd2=26,
        macdsig=9,
        sma_fast=20,
        sma_slow=50,
        ema_fast=10,
        rsi_period=14,
        rsi_upper=55,
        rsi_lower=45,
        boll_period=20,
        boll_dev=2,
        stoch_period=14,
        stoch_dfast=3,
        stoch_dslow=3,
        stoch_upper=80,
        stoch_lower=20,
        score_entry=2,
    )

    def __init__(self):
        self.macd = bt.indicators.MACDHisto(
            self.data,
            period_me1=self.p.macd1,
            period_me2=self.p.macd2,
            period_signal=self.p.macdsig,
        )
        self.sma20 = bt.indicators.SMA(self.data.close, period=self.p.sma_fast)
        self.sma50 = bt.indicators.SMA(self.data.close, period=self.p.sma_slow)
        self.ema10 = bt.indicators.EMA(self.data.close, period=self.p.ema_fast)
        self.rsi = bt.indicators.RSI(self.data.close, period=self.p.rsi_period)
        self.boll = bt.indicators.BollingerBands(
            self.data.close, period=self.p.boll_period, devfactor=self.p.boll_dev
        )
        self.stoch = bt.indicators.Stochastic(
            self.data,
            period=self.p.stoch_period,
            period_dfast=self.p.stoch_dfast,
            period_dslow=self.p.stoch_dslow,
        )
        self.stoch_cross = bt.indicators.CrossOver(self.stoch.percK, self.stoch.percD)

        self.order = None

    def notify_order(self, order):
        if order.status in (order.Completed, order.Canceled, order.Margin, order.Rejected):
            self.order = None

    def next(self):
        if self.order:
            return

        score = 0

        if self.macd.histo[0] > 0:
            score += 1
        elif self.macd.histo[0] < 0:
            score -= 1

        if (self.sma20[0] > self.sma50[0]
                and self.ema10[0] > self.ema10[-1]
                and self.sma20[0] > self.sma20[-1]):
            score += 1
        elif (self.sma20[0] < self.sma50[0]
                and self.ema10[0] < self.ema10[-1]
                and self.sma20[0] < self.sma20[-1]):
            score -= 1

        if self.rsi[0] > self.p.rsi_upper:
            score += 1
        elif self.rsi[0] < self.p.rsi_lower:
            score -= 1

        width_today = self.boll.top[0] - self.boll.bot[0]
        width_yesterday = self.boll.top[-1] - self.boll.bot[-1]
        if self.data.close[0] > self.boll.mid[0] and width_today > width_yesterday:
            score += 1
        elif self.data.close[0] < self.boll.mid[0] and width_today > width_yesterday:
            score -= 1

        if self.stoch_cross[0] > 0 and self.stoch.percK[0] > self.p.stoch_lower:
            score += 1
        elif self.stoch_cross[0] < 0 and self.stoch.percK[0] < self.p.stoch_upper:
            score -= 1

        if score >= self.p.score_entry and self.position.size <= 0:
            self.order = self.order_target_percent(target=1.0)
        elif score <= -self.p.score_entry and self.position.size >= 0:
            self.order = self.order_target_percent(target=-1.0)
