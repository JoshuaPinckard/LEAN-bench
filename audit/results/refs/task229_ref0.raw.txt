import backtrader as bt


class MeanReversionBBRSIStrategy(bt.Strategy):
    def __init__(self):
        self.bb_signal = bt.indicators.BollingerBands(
            self.data.close, period=200, devfactor=2
        )
        self.bb_stop = bt.indicators.BollingerBands(
            self.data.close, period=200, devfactor=3
        )
        self.rsi = bt.indicators.RSI(self.data.close, period=14)
        self.atr = bt.indicators.ATR(self.data, period=14)
        self.cross_mid = bt.indicators.CrossOver(
            self.data.close, self.bb_signal.lines.mid
        )
        self.order = None

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        if self.order:
            return

        close = self.data.close[0]

        if not self.position:
            if close < self.bb_signal.lines.bot[0] and self.rsi[0] < 35:
                distance = abs(close - self.bb_stop.lines.bot[0])
                risk_distance = max(distance, self.atr[0])
                if risk_distance > 0:
                    capital = self.broker.getvalue()
                    target_size = 100000.0 / close
                    risk_size = (0.02 * capital) / risk_distance
                    size = min(target_size, risk_size)
                    if size > 0:
                        self.order = self.buy(size=size)
            elif close > self.bb_signal.lines.top[0] and self.rsi[0] > 65:
                distance = abs(self.bb_stop.lines.top[0] - close)
                risk_distance = max(distance, self.atr[0])
                if risk_distance > 0:
                    capital = self.broker.getvalue()
                    target_size = 100000.0 / close
                    risk_size = (0.02 * capital) / risk_distance
                    size = min(target_size, risk_size)
                    if size > 0:
                        self.order = self.sell(size=size)
        else:
            if self.position.size > 0:
                if self.cross_mid[0] > 0:
                    self.order = self.close()
                elif close <= self.bb_stop.lines.bot[0]:
                    self.order = self.close()
            elif self.position.size < 0:
                if self.cross_mid[0] < 0:
                    self.order = self.close()
                elif close >= self.bb_stop.lines.top[0]:
                    self.order = self.close()
