import math
import statistics

import backtrader as bt


class VolatilityRankingStrategy(bt.Strategy):
    params = (
        ('start_hour', 8),
        ('start_minute', 0),
        ('end_hour', 9),
        ('end_minute', 0),
    )

    def __init__(self):
        self._current_day = None
        self._log_returns = []
        self.daily_volatility = {}
        self.ranked_days = []

        self._start_minutes = self.p.start_hour * 60 + self.p.start_minute
        self._end_minutes = self.p.end_hour * 60 + self.p.end_minute

    def _finalize_day(self, day):
        if day is not None and len(self._log_returns) > 0:
            self.daily_volatility[day] = statistics.pstdev(self._log_returns)

    def next(self):
        dt = self.data.datetime.datetime(0)
        current_date = dt.date()

        if self._current_day is None:
            self._current_day = current_date
        elif current_date != self._current_day:
            self._finalize_day(self._current_day)
            self._current_day = current_date
            self._log_returns = []

        current_minutes = dt.hour * 60 + dt.minute
        if self._start_minutes <= current_minutes < self._end_minutes:
            open_price = self.data.open[0]
            close_price = self.data.close[0]
            self._log_returns.append(math.log(close_price / open_price))

    def stop(self):
        self._finalize_day(self._current_day)
        self.ranked_days = sorted(
            self.daily_volatility.items(),
            key=lambda item: item[1],
            reverse=True,
        )
