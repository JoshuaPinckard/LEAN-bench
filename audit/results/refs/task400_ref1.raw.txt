import backtrader as bt


class AdaptiveChannelEMAStrategy(bt.Strategy):
    params = (
        ("lookback", 10),
        ("stop_multiplier", 2),
    )

    def __init__(self):
        self.fast_ema = bt.indicators.EMA(self.data.close, period=2)
        self.slow_ema = bt.indicators.EMA(self.data.close, period=30)
        self.atr = bt.indicators.ATR(self.data, period=14)

        self.upper_channel = bt.indicators.Highest(
            self.data.high(-1), period=self.p.lookback
        )
        self.lower_channel = bt.indicators.Lowest(
            self.data.low(-1), period=self.p.lookback
        )

        self.order = None
        self.stop_order = None

    def next(self):
        if self.order:
            return

        long_signal = (
            self.data.close[0] > self.upper_channel[0]
            and self.fast_ema[0] > self.slow_ema[0]
        )
        short_signal = (
            self.data.close[0] < self.lower_channel[0]
            and self.fast_ema[0] < self.slow_ema[0]
        )

        if not self.position:
            if long_signal:
                self.order = self.buy()
            elif short_signal:
                self.order = self.sell()
        elif self.position.size > 0 and short_signal:
            if self.stop_order:
                self.cancel(self.stop_order)
            self.order = self.close()
        elif self.position.size < 0 and long_signal:
            if self.stop_order:
                self.cancel(self.stop_order)
            self.order = self.close()

    def notify_order(self, order):
        if order.status in (
            order.Submitted,
            order.Accepted,
        ):
            return

        if order.status == order.Completed:
            if order is self.stop_order:
                self.stop_order = None
            elif order is self.order:
                self.order = None

                if self.position.size > 0:
                    self.stop_order = self.sell(
                        exectype=bt.Order.Stop,
                        price=order.executed.price
                        - self.p.stop_multiplier * self.atr[0],
                    )
                elif self.position.size < 0:
                    self.stop_order = self.buy(
                        exectype=bt.Order.Stop,
                        price=order.executed.price
                        + self.p.stop_multiplier * self.atr[0],
                    )

        elif order.status in (
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            if order is self.order:
                self.order = None
            if order is self.stop_order:
                self.stop_order = None
