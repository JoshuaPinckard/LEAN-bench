import backtrader as bt


class InstantTradingDeltaTriggers(bt.Strategy):
    """
    Reference implementation of 'Instant Trading with Delta Triggers' strategy.
    """

    params = (
        ('buy_position_delta', 0.02),
        ('sell_position_delta', 0.02),
    )

    def __init__(self):
        self.order = None

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        if self.order:
            return

        if len(self.data) < 2:
            return

        close_curr = self.data.close[0]
        close_prev = self.data.close[-1]

        if not self.position:
            # Check long entry
            pct_change_up = (close_curr - close_prev) / close_prev
            if pct_change_up >= self.p.buy_position_delta:
                cash = self.broker.getcash()
                size = int((cash * 0.999) / close_curr)
                if size > 0:
                    self.order = self.buy(size=size)
                return

            # Check short entry
            pct_change_down = (close_prev - close_curr) / close_prev
            if pct_change_down >= self.p.sell_position_delta:
                cash = self.broker.getcash()
                size = int((cash * 0.999) / close_curr)
                if size > 0:
                    self.order = self.sell(size=size)
                return
        else:
            entry_price = self.position.price
            if self.position.size > 0:
                # Long position exit: price drops by sell_position_delta from entry AND trade is profitable
                price_dropped = (entry_price - close_curr) / entry_price >= self.p.sell_position_delta
                is_profitable = close_curr > entry_price
                if price_dropped and is_profitable:
                    self.order = self.close()
            elif self.position.size < 0:
                # Short position exit: price rises by buy_position_delta from entry AND trade is profitable
                price_risen = (close_curr - entry_price) / entry_price >= self.p.buy_position_delta
                is_profitable = close_curr < entry_price
                if price_risen and is_profitable:
                    self.order = self.close()
