import backtrader as bt


class StochasticKeltnerStrategy(bt.Strategy):
    params = (
        ('stoch_period', 14),
        ('stoch_period_dfast', 3),
        ('stoch_period_dslow', 3),
        ('ema_period', 20),
        ('atr_period', 14),
        ('atr_multiplier', 2.0),
    )

    def __init__(self):
        self.stoch = bt.indicators.Stochastic(
            self.data,
            period=self.p.stoch_period,
            period_dfast=self.p.stoch_period_dfast,
            period_dslow=self.p.stoch_period_dslow,
        )
        self.middle_band = bt.indicators.EMA(
            self.data.close, period=self.p.ema_period
        )
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.upper_band = self.middle_band + (
            self.atr * self.p.atr_multiplier
        )
        self.lower_band = self.middle_band - (
            self.atr * self.p.atr_multiplier
        )

        self.order = None
        self.stop_order = None

    def notify_order(self, order):
        if order.status in [
            order.Completed, order.Canceled, order.Margin, order.Rejected
        ]:
            if order == self.order:
                self.order = None

            if order == self.stop_order:
                self.stop_order = None

            if order.status == order.Completed and order.isbuy() and self.position.size > 0:
                self.stop_order = self.sell(
                    exectype=bt.Order.Stop,
                    price=order.executed.price - (
                        self.atr[0] * self.p.atr_multiplier
                    ),
                )
            elif order.status == order.Completed and order.issell() and self.position.size < 0:
                self.stop_order = self.buy(
                    exectype=bt.Order.Stop,
                    price=order.executed.price + (
                        self.atr[0] * self.p.atr_multiplier
                    ),
                )

    def next(self):
        if self.order:
            return

        if not self.position:
            if self.stoch.percK[0] < 20 and self.data.low[0] < self.lower_band[0]:
                self.order = self.buy()
            elif self.stoch.percK[0] > 80 and self.data.high[0] > self.upper_band[0]:
                self.order = self.sell()
        elif self.position.size > 0 and self.data.close[0] >= self.middle_band[0]:
            if self.stop_order:
                self.cancel(self.stop_order)
            self.order = self.close()
        elif self.position.size < 0 and self.data.close[0] <= self.middle_band[0]:
            if self.stop_order:
                self.cancel(self.stop_order)
            self.order = self.close()
