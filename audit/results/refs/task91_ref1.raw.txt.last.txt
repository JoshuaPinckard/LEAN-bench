import backtrader as bt


class SignificantMoveDetection(bt.Strategy):
    params = (
        ('atr_period', 14),
        ('ema_period', 50),
    )

    def __init__(self):
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.ema = bt.indicators.EMA(self.data.close, period=self.p.ema_period)

        self.order = None
        self.order_kind = None
        self.pending_entry_atr = None
        self.entry_price = None
        self.entry_atr = None
        self.last_entry_direction = None
        self.long_rearmed = True
        self.short_rearmed = True

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            if order.status == order.Completed:
                if self.order_kind == 'entry':
                    self.entry_price = order.executed.price
                    self.entry_atr = self.pending_entry_atr
                    self.last_entry_direction = 1 if order.isbuy() else -1
                elif self.order_kind == 'exit':
                    self.entry_price = None
                    self.entry_atr = None

            if order == self.order:
                self.order = None
                self.order_kind = None
                self.pending_entry_atr = None

    def next(self):
        close = self.data.close[0]
        previous_close = self.data.close[-1]
        ema = self.ema[0]
        atr = self.atr[0]

        if close < ema:
            self.long_rearmed = True
        elif close > ema:
            self.short_rearmed = True

        if self.order:
            return

        if self.position:
            if self.position.size > 0:
                take_profit = self.entry_price + (2.0 * self.entry_atr)
                stop_loss = self.entry_price - (1.0 * self.entry_atr)
                trend_reversal = (
                    self.data.close[-1] >= self.ema[-1] and close < ema
                )

                if close >= take_profit or close <= stop_loss or trend_reversal:
                    self.order = self.close()
                    self.order_kind = 'exit'

            else:
                take_profit = self.entry_price - (2.0 * self.entry_atr)
                stop_loss = self.entry_price + (1.0 * self.entry_atr)
                trend_reversal = (
                    self.data.close[-1] <= self.ema[-1] and close > ema
                )

                if close <= take_profit or close >= stop_loss or trend_reversal:
                    self.order = self.close()
                    self.order_kind = 'exit'

            return

        risk_amount = self.broker.getvalue() * 0.01
        size = risk_amount / atr

        long_signal = (
            close > ema
            and (close - previous_close) >= (1.5 * atr)
            and self.long_rearmed
        )
        short_signal = (
            close < ema
            and (previous_close - close) >= (1.5 * atr)
            and self.short_rearmed
        )

        if long_signal:
            self.order = self.buy(size=size)
            self.order_kind = 'entry'
            self.pending_entry_atr = atr
            self.long_rearmed = False

        elif short_signal:
            self.order = self.sell(size=size)
            self.order_kind = 'entry'
            self.pending_entry_atr = atr
            self.short_rearmed = False
