import backtrader as bt
import math


class IntradayMeanReversionStrategy(bt.Strategy):
    params = (
        ("bb_period", 20),
        ("bb_devfactor", 2),
        ("sma_period", 20),
        ("rsi_period", 14),
        ("atr_period", 14),
    )

    def __init__(self):
        self.bb = bt.indicators.BollingerBands(
            self.data.close, period=self.p.bb_period, devfactor=self.p.bb_devfactor
        )
        self.sma = bt.indicators.SMA(self.data.close, period=self.p.sma_period)
        self.rsi = bt.indicators.RSI(self.data.close, period=self.p.rsi_period)
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.cross = bt.indicators.CrossOver(self.data.close, self.bb.bot)

        self._pending_order = None
        self._order_purpose = None
        self._session_date = None
        self._last_date = None
        self._last_atr = None
        self._last_close = None
        self._no_trade_day = False
        self._day_high = None
        self._base_size = 0
        self._take_profit_taken = False
        self._reduction_done = False

    def notify_order(self, order):
        if order.status not in (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            return

        purpose = self._order_purpose if order is self._pending_order else None

        if order is self._pending_order:
            self._pending_order = None
            self._order_purpose = None

        if order.status == order.Completed:
            if purpose == "entry":
                self._take_profit_taken = False
                self._reduction_done = False
                self._day_high = self.data.high[0]
            elif purpose == "take_profit":
                self._take_profit_taken = True

            if not self.position:
                self._base_size = 0
                self._day_high = None
                self._take_profit_taken = False
                self._reduction_done = False
        elif purpose == "entry":
            self._base_size = 0

    def _submit(self, order, purpose):
        self._pending_order = order
        self._order_purpose = purpose

    def _maximum_affordable_size(self, price):
        cash = self.broker.getcash()
        comminfo = self.broker.getcommissioninfo(self.data)
        leverage = comminfo.get_leverage()
        unit_commission = comminfo.getcommission(1, price)
        unit_cash_requirement = (price / leverage) + unit_commission

        if unit_cash_requirement <= 0:
            return 0
        return int(cash / unit_cash_requirement)

    def next(self):
        dt = self.data.datetime.datetime(0)
        current_date = dt.date()
        price = self.data.close[0]

        if self._last_date is None:
            self._session_date = current_date
            self._no_trade_day = (
                self.bb.mid[0] != 0
                and (self.bb.top[0] - self.bb.bot[0]) / self.bb.mid[0] > 0.015
            )
        elif current_date != self._last_date:
            self._session_date = current_date
            prior_atr_percent = (
                self._last_atr / self._last_close
                if self._last_atr is not None and self._last_close
                else 0.0
            )
            self._no_trade_day = (
                (
                    self.bb.mid[0] != 0
                    and (self.bb.top[0] - self.bb.bot[0]) / self.bb.mid[0] > 0.015
                )
                or prior_atr_percent > 0.015
            )
            self._day_high = self.data.high[0] if self.position else None
            self._reduction_done = False

        if self.position and self._day_high is not None:
            self._day_high = max(self._day_high, self.data.high[0])

        if self._pending_order is None and self.position:
            if self._day_high is not None and price <= self._day_high * 0.98:
                self._submit(self.close(), "stop")
            elif (
                not self._take_profit_taken
                and price >= self.position.price * 1.003
            ):
                size = int(self.position.size * 0.5)
                if size > 0:
                    self._submit(self.sell(size=size), "take_profit")
            elif (
                not self._reduction_done
                and (dt.hour == 15 and 0 <= dt.minute < 30)
            ):
                target_size = int(self._base_size * 0.5)
                reduction_size = int(self.position.size) - target_size
                self._reduction_done = True
                if reduction_size > 0:
                    self._submit(self.sell(size=reduction_size), "reduction")

        if (
            self._pending_order is None
            and not self.position
            and not self._no_trade_day
            and dt.hour == 9
            and 30 <= dt.minute < 60
        ):
            band_width = (
                (self.bb.top[0] - self.bb.bot[0]) / self.bb.mid[0]
                if self.bb.mid[0] != 0
                else math.inf
            )
            atr_percent = self.atr[0] / price if price else math.inf

            if (
                self.cross[0] > 0
                and self.rsi[0] < 45
                and (band_width <= 0.005 or atr_percent <= 0.008)
            ):
                capital = self.broker.getvalue()
                base_size = int(capital / price) if price > 0 else 0
                target_multiplier = 1.25 if atr_percent > 0.01 else 1.5
                target_size = int((capital * target_multiplier) / price)
                affordable_size = self._maximum_affordable_size(price)

                self._base_size = min(base_size, affordable_size)
                entry_size = min(target_size, affordable_size)

                if entry_size > 0:
                    self._submit(self.buy(size=entry_size), "entry")

        self._last_date = current_date
        self._last_atr = self.atr[0]
        self._last_close = price
