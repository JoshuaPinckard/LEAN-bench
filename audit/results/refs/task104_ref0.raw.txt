```python
import backtrader as bt


class ConsolidationBreakout(bt.Strategy):
    params = dict(
        sma_period=120,
        vol_period=20,
        lower_bound=180.00,
        upper_bound=193.00,
        take_profit=200.00,
        vol_mult=1.5,
        trend_lookback=5,
        risk_pct=0.015,
    )

    def __init__(self):
        self.sma = bt.indicators.SMA(self.data.close, period=self.p.sma_period)
        self.vol_sma = bt.indicators.SMA(self.data.volume, period=self.p.vol_period)
        self.addminperiod(self.p.sma_period + self.p.trend_lookback)
        self.order = None

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        if self.order:
            return

        close = self.data.close[0]

        if not self.position:
            trend_up = self.sma[0] > self.sma[-self.p.trend_lookback]
            above_sma = close > self.sma[0]
            was_consolidating = self.p.lower_bound <= self.data.close[-1] <= self.p.upper_bound
            volume_ok = self.data.volume[0] <= self.p.vol_mult * self.vol_sma[0]
            breakout = close > self.p.upper_bound

            if trend_up and above_sma and was_consolidating and volume_ok and breakout:
                stop_price = self.sma[0]
                risk_per_share = close - stop_price
                portfolio_value = self.broker.getvalue()
                risk_amount = self.p.risk_pct * portfolio_value
                size = int(risk_amount / risk_per_share)
                if size > 0:
                    self.order = self.buy(size=size)
        else:
            if close >= self.p.take_profit:
                self.order = self.close()
            elif close < self.sma[0]:
                self.order = self.close()
            elif close < self.p.lower_bound:
                self.order = self.close()
```
