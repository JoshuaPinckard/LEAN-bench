import backtrader as bt


class BearishWeeklyShortATRStrategy(bt.Strategy):
    def __init__(self):
        self.atr = bt.indicators.ATR(self.data, period=14)

        self.entry_order = None
        self.close_order = None
        self.tp_order = None
        self.sl_order = None
        self.entry_bar = None
        self.entry_atr = None

    def next(self):
        if self.entry_order is not None or self.close_order is not None:
            return

        if self.position:
            if self.entry_bar is not None and len(self) - self.entry_bar >= 20:
                if self.tp_order is not None:
                    self.cancel(self.tp_order)
                if self.sl_order is not None:
                    self.cancel(self.sl_order)
                self.close_order = self.close()
            return

        if len(self) % 7 == 0:
            k = 2
            atr_val = self.atr[0]
            risk_per_trade = 0.01 * self.broker.getvalue()
            size = risk_per_trade / (k * atr_val)
            self.entry_atr = atr_val
            self.entry_order = self.sell(size=size)

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status == order.Completed:
            if order == self.entry_order:
                entry_price = order.executed.price
                size = abs(order.executed.size)
                k = 2
                atr_val = self.entry_atr
                tp_price = entry_price - 0.5 * k * atr_val
                sl_price = entry_price + k * atr_val

                self.entry_order = None
                self.entry_bar = len(self)
                self.tp_order = self.buy(exectype=bt.Order.Limit, price=tp_price, size=size)
                self.sl_order = self.buy(exectype=bt.Order.Stop, price=sl_price, size=size,
                                          oco=self.tp_order)
            elif order == self.tp_order:
                self.tp_order = None
                self.sl_order = None
                self.entry_bar = None
            elif order == self.sl_order:
                self.sl_order = None
                self.tp_order = None
                self.entry_bar = None
            elif order == self.close_order:
                self.close_order = None
                self.tp_order = None
                self.sl_order = None
                self.entry_bar = None
        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            if order == self.entry_order:
                self.entry_order = None
            elif order == self.tp_order:
                self.tp_order = None
            elif order == self.sl_order:
                self.sl_order = None
            elif order == self.close_order:
                self.close_order = None
