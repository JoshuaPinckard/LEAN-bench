import backtrader as bt


class Sma200ExitStrategy(bt.Strategy):
    params = (
        ("sma_period", 200),
        ("capital_pct", 0.01),
    )

    def __init__(self):
        self.sma = bt.indicators.SimpleMovingAverage(
            self.data.close, period=self.params.sma_period
        )
        self.crossover = bt.indicators.CrossOver(self.data.close, self.sma)
        self.order = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return
        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            self.order = None

    def next(self):
        if self.order:
            return

        if not self.position:
            if self.crossover > 0:
                capital = self.broker.getvalue() * self.params.capital_pct
                size = capital / self.data.close[0]
                self.order = self.buy(size=size)
        else:
            if self.crossover < 0:
                self.order = self.close()
