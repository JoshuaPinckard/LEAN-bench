class WeeklyORBStrategy(bt.Strategy):
    params = dict(
        ema_period=21,
        bb_period=20,
        bb_devfactor=2.0,
        stoch_period=14,
        stoch_pfast=3,
        stoch_pslow=3,
        atr_period=14,
        vol_sma_period=20,
        fib_low_mult=0.618,
        fib_high_mult=0.382,
        tp_mult=1.5,
        sl_atr_mult=0.25,
        risk_pct=0.01,
    )

    def __init__(self):
        self.ema21 = bt.indicators.EMA(self.data.close, period=self.p.ema_period)
        self.bb = bt.indicators.BollingerBands(
            self.data.close, period=self.p.bb_period, devfactor=self.p.bb_devfactor
        )
        self.stoch = bt.indicators.Stochastic(
            self.data,
            period=self.p.stoch_period,
            period_dfast=self.p.stoch_pfast,
            period_dslow=self.p.stoch_pslow,
        )
        self.crossover = bt.indicators.CrossOver(self.stoch.percK, self.stoch.percD)
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.vol_sma = bt.indicators.SMA(self.data.volume, period=self.p.vol_sma_period)

        self.order = None

        self.current_week = None
        self.wor_high = None
        self.wor_low = None
        self.wor_range = None
        self.breakout_confirmed = False

        self.entry_price = None
        self.tp_level = None
        self.sl_level = None
        self.entry_wor_low = None

    def notify_order(self, order):
        if order.status in (order.Completed, order.Canceled, order.Margin, order.Rejected):
            if order.status == order.Completed:
                if order.isbuy():
                    self.entry_price = order.executed.price
                    self.tp_level = self.entry_price + self.p.tp_mult * self.wor_range
                    self.sl_level = self.wor_low - self.p.sl_atr_mult * self.atr[0]
                    self.entry_wor_low = self.wor_low
                else:
                    self.entry_price = None
                    self.tp_level = None
                    self.sl_level = None
                    self.entry_wor_low = None
            self.order = None

    def next(self):
        dt = self.data.datetime.date(0)
        iso_year, iso_week, _ = dt.isocalendar()
        week_id = (iso_year, iso_week)

        if week_id != self.current_week:
            self.current_week = week_id
            self.wor_high = self.data.high[0]
            self.wor_low = self.data.low[0]
            self.wor_range = self.wor_high - self.wor_low
            self.breakout_confirmed = False

        if self.order:
            return

        close = self.data.close[0]

        if self.position:
            high = self.data.high[0]
            low = self.data.low[0]
            sl_hit = self.sl_level is not None and (
                low <= self.sl_level or close < self.entry_wor_low
            )
            tp_hit = self.tp_level is not None and high >= self.tp_level
            if sl_hit:
                self.order = self.close()
            elif tp_hit:
                self.order = self.close()
        else:
            if self.breakout_confirmed and self.wor_range is not None and self.wor_range > 0:
                low = self.data.low[0]
                fib_lower = self.wor_high - self.p.fib_low_mult * self.wor_range
                fib_upper = self.wor_high - self.p.fib_high_mult * self.wor_range
                pullback = (fib_lower <= low <= fib_upper) or (fib_lower <= close <= fib_upper)
                trend = close > self.ema21[0] and close > self.bb.mid[0]
                reversal = self.crossover[0] > 0

                if pullback and trend and reversal:
                    equity = self.broker.getvalue()
                    size = (self.p.risk_pct * equity) / close
                    if size > 0:
                        self.order = self.buy(size=size)

        if close > self.wor_high:
            self.breakout_confirmed = True
