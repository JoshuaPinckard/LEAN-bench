import math
import backtrader as bt


class MACDIntradayATRStrategy(bt.Strategy):
    params = (
        ('macd_fast', 12),
        ('macd_slow', 26),
        ('macd_signal', 9),
        ('atr_period', 14),
        ('stop_loss_pct', 0.02),
        ('risk_pct', 0.01),
    )

    def __init__(self):
        self.macd = bt.indicators.MACD(
            self.data.close,
            period_me1=self.params.macd_fast,
            period_me2=self.params.macd_slow,
            period_signal=self.params.macd_signal,
        )
        self.crossover = bt.indicators.CrossOver(
            self.macd.macd, self.macd.signal
        )
        self.atr = bt.indicators.ATR(self.data, period=self.params.atr_period)
        self.order = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return
        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            self.order = None

    def get_size(self):
        atr_val = self.atr[0]
        if atr_val <= 0 or math.isnan(atr_val):
            return 0
        risk_amount = self.broker.getvalue() * self.params.risk_pct
        size = int(risk_amount / atr_val)
        price = self.data.close[0]
        if price > 0:
            max_size = int(self.broker.getcash() / price)
            size = min(size, max_size)
        return max(size, 0)

    def next(self):
        if self.order:
            return

        if self.position.size > 0:
            stop_price = self.position.price * (1.0 - self.params.stop_loss_pct)
            if self.data.low[0] <= stop_price:
                self.order = self.close()
            elif self.crossover[0] < 0:
                new_size = self.get_size()
                if new_size > 0:
                    self.order = self.sell(size=self.position.size + new_size)
                else:
                    self.order = self.close()

        elif self.position.size < 0:
            stop_price = self.position.price * (1.0 + self.params.stop_loss_pct)
            if self.data.high[0] >= stop_price:
                self.order = self.close()
            elif self.crossover[0] > 0:
                new_size = self.get_size()
                if new_size > 0:
                    self.order = self.buy(
                        size=abs(self.position.size) + new_size
                    )
                else:
                    self.order = self.close()

        else:
            if self.crossover[0] > 0:
                size = self.get_size()
                if size > 0:
                    self.order = self.buy(size=size)
            elif self.crossover[0] < 0:
                size = self.get_size()
                if size > 0:
                    self.order = self.sell(size=size)
