import math
import backtrader as bt


class DonchianChannel(bt.Indicator):
    lines = ('dch', 'dcl', 'dcm')
    params = (('period', 20),)

    def __init__(self):
        self.lines.dch = bt.ind.Highest(self.data.high(-1), period=self.p.period)
        self.lines.dcl = bt.ind.Lowest(self.data.low(-1), period=self.p.period)
        self.lines.dcm = (self.lines.dch + self.lines.dcl) / 2.0


class SeasonalFilter(bt.Indicator):
    lines = ('seasonal',)

    def next(self):
        dt = self.data.datetime.date(0)
        doy = dt.timetuple().tm_yday
        self.lines.seasonal[0] = math.sin(2 * math.pi * doy / 365.25)


class DonchianSeasonalStrategy(bt.Strategy):
    params = (
        ('donchian_period', 20),
        ('atr_period', 14),
        ('atr_mult', 3.0),
        ('seasonal_long_thresh', 0.5),
        ('seasonal_short_thresh', -0.5),
    )

    def __init__(self):
        self.donchian = DonchianChannel(self.data, period=self.p.donchian_period)
        self.atr = bt.ind.ATR(self.data, period=self.p.atr_period)

        if hasattr(self.data.lines, 'seasonal'):
            self.seasonal = self.data.seasonal
        else:
            self.seasonal = SeasonalFilter(self.data).seasonal

        self.order = None
        self.trailing_stop = None

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        if self.order:
            return

        close = self.data.close[0]
        long_signal = (close > self.donchian.dch[0]) and (self.seasonal[0] > self.p.seasonal_long_thresh)
        short_signal = (close < self.donchian.dcl[0]) and (self.seasonal[0] < self.p.seasonal_short_thresh)

        if self.position.size > 0:
            current_stop = close - self.p.atr_mult * self.atr[0]
            if self.trailing_stop is None:
                self.trailing_stop = current_stop
            else:
                self.trailing_stop = max(self.trailing_stop, current_stop)

            if close <= self.trailing_stop:
                self.order = self.close()
                self.trailing_stop = None
            elif short_signal:
                self.order = self.close()
                self.trailing_stop = None

        elif self.position.size < 0:
            current_stop = close + self.p.atr_mult * self.atr[0]
            if self.trailing_stop is None:
                self.trailing_stop = current_stop
            else:
                self.trailing_stop = min(self.trailing_stop, current_stop)

            if close >= self.trailing_stop:
                self.order = self.close()
                self.trailing_stop = None
            elif long_signal:
                self.order = self.close()
                self.trailing_stop = None

        else:
            self.trailing_stop = None
            if long_signal:
                self.order = self.buy()
            elif short_signal:
                self.order = self.sell()
