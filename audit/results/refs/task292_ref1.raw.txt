import backtrader as bt
import math


class ReferenceStrategy(bt.Strategy):
    def __init__(self):
        self.sma10 = bt.indicators.SMA(self.data.close, period=10)
        self.sma20 = bt.indicators.SMA(self.data.close, period=20)
        self.atr = bt.indicators.ATR(self.data, period=14)
        self.crossover = bt.indicators.CrossOver(self.sma10, self.sma20)

        self.entry_order = None
        self.exit_order = None
        self.stop_order = None
        self.entry_price = None

    def _in_regular_session(self):
        current = self.data.datetime.datetime(0)
        minutes = current.hour * 60 + current.minute
        return 9 * 60 + 30 <= minutes < 16 * 60

    def _entry_window(self):
        current = self.data.datetime.datetime(0)
        minutes = current.hour * 60 + current.minute
        return 9 * 60 + 35 <= minutes < 15 * 60 + 30

    def _cancel_stop(self):
        if self.stop_order is not None:
            self.cancel(self.stop_order)

    def _exit_position(self):
        if self.exit_order is None:
            self._cancel_stop()
            self.exit_order = self.close()

    def next(self):
        current = self.data.datetime.datetime(0)
        minutes = current.hour * 60 + current.minute

        if self.position:
            if minutes == 15 * 60 + 55:
                self._exit_position()
                return

            if self.position.size > 0:
                if self.crossover[0] < 0:
                    self._exit_position()
            else:
                if self.crossover[0] > 0:
                    self._exit_position()
            return

        if (not self._in_regular_session() or
                not self._entry_window() or
                self.entry_order is not None or
                self.exit_order is not None or
                self.stop_order is not None):
            return

        near_sma20 = abs(self.data.close[0] - self.sma20[0]) <= 0.5 * self.atr[0]
        quantity = math.floor(
            self.broker.getvalue() * 0.01 / (self.data.close[0] * 0.01)
        )

        if quantity <= 0:
            return

        if self.crossover[0] > 0 and near_sma20:
            self.entry_order = self.buy(size=quantity)
        elif self.crossover[0] < 0 and near_sma20:
            self.entry_order = self.sell(size=quantity)

    def notify_order(self, order):
        if order.status in [order.Completed]:
            if order is self.entry_order:
                self.entry_order = None
                self.entry_price = order.executed.price

                if self.position.size > 0:
                    self.stop_order = self.sell(
                        size=self.position.size,
                        exectype=bt.Order.Stop,
                        price=self.entry_price * 0.99,
                    )
                elif self.position.size < 0:
                    self.stop_order = self.buy(
                        size=abs(self.position.size),
                        exectype=bt.Order.Stop,
                        price=self.entry_price * 1.01,
                    )

            elif order is self.exit_order:
                self.exit_order = None
                self.entry_price = None

            elif order is self.stop_order:
                self.stop_order = None
                self.entry_price = None

        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            if order is self.entry_order:
                self.entry_order = None
            elif order is self.exit_order:
                self.exit_order = None
            elif order is self.stop_order:
                self.stop_order = None
