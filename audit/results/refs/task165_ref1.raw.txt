import backtrader as bt


class Fibo114XBBATRProxy(bt.Strategy):
    params = (
        ("risk_fraction", 0.01),
    )

    def __init__(self):
        self.sma200 = bt.indicators.SMA(self.data.close, period=200)
        self.bb = bt.indicators.BollingerBands(
            self.data.close, period=20, devfactor=2
        )
        self.ema9 = bt.indicators.EMA(self.data.close, period=9)
        self.atr14 = bt.indicators.ATR(self.data, period=14)

        self.long_cross = bt.indicators.CrossOver(self.data.close, self.ema9)
        self.short_cross = bt.indicators.CrossOver(self.ema9, self.data.close)

        self.order = None
        self.long_armed = False
        self.short_armed = False
        self.long_stop = None
        self.short_stop = None
        self.long_peak = None
        self.short_trough = None

    def notify_order(self, order):
        if order.status in (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            self.order = None

        if order.status == order.Completed:
            if order.isbuy() and self.position.size > 0:
                self.long_peak = order.executed.price
            elif order.issell() and self.position.size < 0:
                self.short_trough = order.executed.price

    def next(self):
        if self.order:
            return

        width = self.bb.top[0] - self.bb.bot[0]
        atr = self.atr14[0]

        long_signal = self.long_armed and self.long_cross[0] > 0
        short_signal = self.short_armed and self.short_cross[0] > 0

        if not self.position:
            if long_signal and self.data.close[0] > self.sma200[0]:
                stop = self.long_stop
                distance = self.data.close[0] - stop
                if distance > 0:
                    size = (self.broker.getvalue() * self.p.risk_fraction) / distance
                    self.order = self.buy(size=size)
                    self.long_armed = False
                    self.short_armed = False

            elif short_signal and self.data.close[0] < self.sma200[0]:
                stop = self.short_stop
                distance = stop - self.data.close[0]
                if distance > 0:
                    size = (self.broker.getvalue() * self.p.risk_fraction) / distance
                    self.order = self.sell(size=size)
                    self.long_armed = False
                    self.short_armed = False

        elif self.position.size > 0:
            self.long_peak = max(self.long_peak, self.data.high[0])
            trailing_stop = self.long_peak - (atr * 1.5)
            effective_stop = max(self.long_stop, trailing_stop)

            if (
                self.data.low[0] <= effective_stop
                or self.data.high[0] >= self.bb.mid[0]
                or short_signal
            ):
                self.order = self.close()

        else:
            self.short_trough = min(self.short_trough, self.data.low[0])
            trailing_stop = self.short_trough + (atr * 1.5)
            effective_stop = min(self.short_stop, trailing_stop)

            if (
                self.data.high[0] >= effective_stop
                or self.data.low[0] <= self.bb.mid[0]
                or long_signal
            ):
                self.order = self.close()

        if (
            self.data.close[0] < self.bb.bot[0]
            and (self.bb.bot[0] - self.data.close[0]) >= 0.114 * width
        ):
            self.long_armed = True
            self.long_stop = (self.bb.bot[0] - 0.114 * width) - (0.5 * atr)

        if (
            self.data.close[0] > self.bb.top[0]
            and (self.data.close[0] - self.bb.top[0]) >= 0.114 * width
        ):
            self.short_armed = True
            self.short_stop = (self.bb.top[0] + 0.114 * width) + (0.5 * atr)
