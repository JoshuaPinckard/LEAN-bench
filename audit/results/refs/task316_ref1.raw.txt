import backtrader as bt


class VolatilityTrendBandStrategy(bt.Strategy):
    params = (
        ("atr_period", 10),
        ("atr_multiplier", 3),
    )

    def __init__(self):
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.ma = bt.indicators.EMA(self.data.close, period=20)

        self.signal = self.data.close - self.ma
        self.signal_mean = bt.indicators.SMA(self.signal, period=20)
        self.signal_std = bt.indicators.StdDev(self.signal, period=20)

        self.upper_band = self.signal_mean + 2 * self.signal_std
        self.lower_band = self.signal_mean - 2 * self.signal_std

        self.cross_upper = bt.indicators.CrossOver(self.signal, self.upper_band)
        self.cross_lower = bt.indicators.CrossOver(self.signal, self.lower_band)
        self.cross_mean = bt.indicators.CrossOver(self.signal, self.signal_mean)

        self.order = None
        self.stop_order = None

    def next(self):
        if self.order:
            return

        if not self.position:
            expansion = abs(self.signal[0] - self.signal[-1]) > 1.0 * self.signal_std[0]

            if self.cross_upper[0] > 0 and expansion:
                self.order = self.buy()
            elif self.cross_lower[0] < 0 and expansion:
                self.order = self.sell()
            return

        if self.position.size > 0 and self.cross_mean[0] < 0:
            if self.stop_order:
                self.cancel(self.stop_order)
            self.order = self.close()
        elif self.position.size < 0 and self.cross_mean[0] > 0:
            if self.stop_order:
                self.cancel(self.stop_order)
            self.order = self.close()

    def notify_order(self, order):
        if order.status in (order.Completed, order.Canceled, order.Margin, order.Rejected):
            if order is self.order:
                self.order = None

            if order is self.stop_order:
                self.stop_order = None

        if order.status == order.Completed:
            if order.isbuy() and self.position.size > 0 and order is not self.stop_order:
                self.stop_order = self.sell(
                    exectype=bt.Order.Stop,
                    price=order.executed.price - self.p.atr_multiplier * self.atr[0],
                )
            elif order.issell() and self.position.size < 0 and order is not self.stop_order:
                self.stop_order = self.buy(
                    exectype=bt.Order.Stop,
                    price=order.executed.price + self.p.atr_multiplier * self.atr[0],
                )
