import backtrader as bt


class DynamicExitSMAStrategy(bt.Strategy):
    """
    Reference implementation of a trading strategy with dynamic exit rules
    based on current trade profit percentage and SMA crossover entry signals.
    """

    params = (
        ("fast_period", 20),
        ("slow_period", 50),
        ("profit_threshold", 10.0),
    )

    def __init__(self):
        self.fast_sma = bt.indicators.SMA(
            self.data.close, period=self.params.fast_period
        )
        self.slow_sma = bt.indicators.SMA(
            self.data.close, period=self.params.slow_period
        )
        self.crossover = bt.indicators.CrossOver(self.fast_sma, self.slow_sma)

        self.order = None
        self.entry_price = None

    def notify_order(self, order):
        if order.status in [order.Completed]:
            if order.isbuy():
                self.entry_price = order.executed.price
            elif order.issell():
                self.entry_price = None
            self.order = None
        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def sell_below10(self):
        """Close position when trade profit is below 10%."""
        self.order = self.close()

    def sell_abv10(self):
        """Close position when trade profit is above 10%."""
        self.order = self.close()

    def next(self):
        if self.order:
            return

        if not self.position:
            # Bullish crossover: Fast SMA crosses Slow SMA from below upward
            if self.crossover[0] > 0:
                cash = self.broker.getcash()
                price = self.data.close[0]
                # Sizing: 100% of available capital adjusted for 0.1% commission
                size = int(cash / (price * 1.001))
                if size > 0:
                    self.order = self.buy(size=size)
        else:
            # Bearish crossover trigger to evaluate dynamic exit conditions
            if self.crossover[0] < 0:
                entry_price = (
                    self.entry_price
                    if self.entry_price is not None
                    else self.position.price
                )
                profit_pct = (
                    (self.data.close[0] - entry_price) / entry_price
                ) * 100.0

                if profit_pct < self.params.profit_threshold:
                    self.sell_below10()
                else:
                    self.sell_abv10()
