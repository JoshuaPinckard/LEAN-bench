import backtrader as bt


class LunchBreakFade(bt.Strategy):
    params = ()

    def __init__(self):
        self.rsi = bt.indicators.RSI(self.data.close, period=14)
        self.rsi_prev = self.rsi(-1)
        self.volume_sma = bt.indicators.SMA(self.data.volume, period=20)

        self.order = None
        self.stop_order = None
        self.entry_price = None

        self.current_day = None
        self.morning_open = None
        self.bias = None

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            if order == self.stop_order:
                self.stop_order = None
                if order.status == order.Completed:
                    self.entry_price = None
                return

            if order == self.order:
                if order.status == order.Completed:
                    if order.isbuy() and self.position.size > 0:
                        self.entry_price = order.executed.price
                        self.stop_order = self.sell(
                            size=self.position.size,
                            exectype=bt.Order.Stop,
                            price=self.entry_price * 0.98,
                        )
                    elif order.issell() and self.position.size < 0:
                        self.entry_price = order.executed.price
                        self.stop_order = self.buy(
                            size=abs(self.position.size),
                            exectype=bt.Order.Stop,
                            price=self.entry_price * 1.02,
                        )
                    elif not self.position:
                        self.entry_price = None

                self.order = None

    def next(self):
        dt = self.data.datetime.datetime(0)
        day = dt.date()
        bar_time = (dt.hour, dt.minute)

        if day != self.current_day:
            self.current_day = day
            self.morning_open = None
            self.bias = None

        if bar_time == (9, 30):
            self.morning_open = self.data.open[0]

        if bar_time == (11, 30) and self.morning_open is not None:
            change = ((self.data.close[0] - self.morning_open) / self.morning_open) * 100.0
            if change >= 0.5:
                self.bias = "rally"
            elif change <= -0.5:
                self.bias = "drop"
            else:
                self.bias = None

        if bar_time == (14, 30):
            if self.position and self.order is None:
                if self.stop_order is not None:
                    self.cancel(self.stop_order)
                self.order = self.close()
            return

        if self.order is not None:
            return

        if self.position:
            if self.position.size < 0 and self.rsi_prev <= 60 and self.rsi[0] > 60:
                if self.stop_order is not None:
                    self.cancel(self.stop_order)
                self.order = self.close()
            elif self.position.size > 0 and self.rsi_prev >= 40 and self.rsi[0] < 40:
                if self.stop_order is not None:
                    self.cancel(self.stop_order)
                self.order = self.close()
            return

        if not ((11, 30) <= bar_time < (14, 30)):
            return

        if self.data.volume[0] >= self.volume_sma[0]:
            return

        if self.bias == "rally" and self.rsi_prev >= 60 and self.rsi[0] < 60:
            self.order = self.sell()
        elif self.bias == "drop" and self.rsi_prev <= 40 and self.rsi[0] > 40:
            self.order = self.buy()
