import backtrader as bt


class VolumeSmaCrossoverStrategy(bt.Strategy):
    params = dict(
        vol_sma_fast=10,
        vol_sma_slow=50,
        price_sma_period=200,
        stop_loss_pct=0.01,
        take_profit_pct=0.02,
        use_take_profit=True,
        risk_pct=0.01,
    )

    def __init__(self):
        self.vol_sma_fast = bt.indicators.SMA(self.data.volume, period=self.p.vol_sma_fast)
        self.vol_sma_slow = bt.indicators.SMA(self.data.volume, period=self.p.vol_sma_slow)
        self.price_sma = bt.indicators.SMA(self.data.close, period=self.p.price_sma_period)

        self.vol_cross = bt.indicators.CrossOver(self.vol_sma_fast, self.vol_sma_slow)
        self.price_cross = bt.indicators.CrossOver(self.data.close, self.price_sma)

        self.order = None
        self.stop_order = None
        self.tp_order = None

    def _cancel_protective_orders(self):
        if self.stop_order:
            self.cancel(self.stop_order)
            self.stop_order = None
        if self.tp_order:
            self.cancel(self.tp_order)
            self.tp_order = None

    def _place_protective_orders(self, entry_price):
        size = abs(self.position.size)
        if self.position.size > 0:
            stop_price = entry_price * (1 - self.p.stop_loss_pct)
            self.stop_order = self.sell(size=size, exectype=bt.Order.Stop, price=stop_price)
            if self.p.use_take_profit:
                tp_price = entry_price * (1 + self.p.take_profit_pct)
                self.tp_order = self.sell(size=size, exectype=bt.Order.Limit, price=tp_price)
        else:
            stop_price = entry_price * (1 + self.p.stop_loss_pct)
            self.stop_order = self.buy(size=size, exectype=bt.Order.Stop, price=stop_price)
            if self.p.use_take_profit:
                tp_price = entry_price * (1 - self.p.take_profit_pct)
                self.tp_order = self.buy(size=size, exectype=bt.Order.Limit, price=tp_price)

    def _enter_long(self):
        price = self.data.close[0]
        stop_price = price * (1 - self.p.stop_loss_pct)
        stop_distance = price - stop_price
        if stop_distance <= 0:
            return
        risk_amount = self.broker.getvalue() * self.p.risk_pct
        size = int(risk_amount / stop_distance)
        if size < 1:
            return
        self.order = self.buy(size=size)

    def _enter_short(self):
        price = self.data.close[0]
        stop_price = price * (1 + self.p.stop_loss_pct)
        stop_distance = stop_price - price
        if stop_distance <= 0:
            return
        risk_amount = self.broker.getvalue() * self.p.risk_pct
        size = int(risk_amount / stop_distance)
        if size < 1:
            return
        self.order = self.sell(size=size)

    def next(self):
        if self.order:
            return

        if not self.position:
            if self.vol_cross[0] > 0 and self.data.close[0] > self.price_sma[0]:
                self._enter_long()
            elif self.vol_cross[0] < 0 and self.data.close[0] < self.price_sma[0]:
                self._enter_short()
        else:
            if self.position.size > 0:
                if self.vol_cross[0] < 0 or self.price_cross[0] < 0:
                    self._cancel_protective_orders()
                    self.order = self.close()
            elif self.position.size < 0:
                if self.vol_cross[0] > 0 or self.price_cross[0] > 0:
                    self._cancel_protective_orders()
                    self.order = self.close()

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted, order.Partial):
            return

        if order.status == order.Completed:
            if order is self.stop_order:
                self.stop_order = None
                if self.tp_order:
                    self.cancel(self.tp_order)
                    self.tp_order = None
                self.order = None
            elif order is self.tp_order:
                self.tp_order = None
                if self.stop_order:
                    self.cancel(self.stop_order)
                    self.stop_order = None
                self.order = None
            elif order is self.order:
                if self.position:
                    self._place_protective_orders(order.executed.price)
                self.order = None
        elif order.status in (order.Canceled, order.Margin, order.Rejected):
            if order is self.order:
                self.order = None
            elif order is self.stop_order:
                self.stop_order = None
            elif order is self.tp_order:
                self.tp_order = None
