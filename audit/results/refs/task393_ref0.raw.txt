import backtrader as bt


class DonchianWidthMeanReversion(bt.Strategy):
    params = (
        ('donchian_period', 20),
        ('width_sma_period', 20),
        ('deviation_multiplier', 2.0),
        ('stop_loss_pct', 0.02),
    )

    def __init__(self):
        self.donchian_hi = bt.ind.Highest(self.data.high, period=self.p.donchian_period)
        self.donchian_lo = bt.ind.Lowest(self.data.low, period=self.p.donchian_period)
        self.width = self.donchian_hi - self.donchian_lo
        self.width_sma = bt.ind.SMA(self.width, period=self.p.width_sma_period)
        self.upper_extreme = self.width_sma * self.p.deviation_multiplier
        self.lower_extreme = self.width_sma / self.p.deviation_multiplier

        self.cross_up_upper = bt.ind.CrossUp(self.width, self.upper_extreme)
        self.cross_down_lower = bt.ind.CrossDown(self.width, self.lower_extreme)
        self.cross_down_upper = bt.ind.CrossDown(self.width, self.upper_extreme)
        self.cross_up_lower = bt.ind.CrossUp(self.width, self.lower_extreme)
        self.cross_down_mean = bt.ind.CrossDown(self.width, self.width_sma)
        self.cross_up_mean = bt.ind.CrossUp(self.width, self.width_sma)

        self.upper_regime = False
        self.lower_regime = False
        self.order = None
        self.stop_price = None

    def notify_order(self, order):
        if order.status == order.Completed:
            if self.position:
                entry_price = order.executed.price
                if self.position.size > 0:
                    self.stop_price = entry_price * (1 - self.p.stop_loss_pct)
                else:
                    self.stop_price = entry_price * (1 + self.p.stop_loss_pct)
            else:
                self.stop_price = None
            self.order = None
        elif order.status in (order.Canceled, order.Margin, order.Rejected):
            self.order = None

    def next(self):
        if self.cross_up_upper[0]:
            self.upper_regime = True
            self.lower_regime = False
        if self.cross_down_lower[0]:
            self.lower_regime = True
            self.upper_regime = False

        if self.order:
            return

        if self.position:
            if self.position.size > 0:
                stop_hit = self.data.low[0] <= self.stop_price
                if self.cross_up_mean[0] or stop_hit:
                    self.order = self.close()
            else:
                stop_hit = self.data.high[0] >= self.stop_price
                if self.cross_down_mean[0] or stop_hit:
                    self.order = self.close()
        else:
            if self.upper_regime and self.cross_down_upper[0]:
                self.order = self.sell()
            elif self.lower_regime and self.cross_up_lower[0]:
                self.order = self.buy()
