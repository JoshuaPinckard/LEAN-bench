import math
import backtrader as bt


class ProfitBackedExpansion(bt.Strategy):
    def __init__(self):
        self.order = None

        # Base position tracking
        self.has_base = False
        self.base_entry_price = None
        self.base_shares = 0

        # Additional tranche tracking
        self.tranche_shares = 0
        self.tranche_entry_price = None
        self.tp_price = None
        self.tranche_entry_bar = None

        # Indicators
        self.highest_252 = bt.indicators.Highest(self.data.high, period=252)
        self.rsi = bt.indicators.RSI(self.data.close, period=14)
        self.sma_v20 = bt.indicators.SMA(self.data.volume, period=20)
        self.sma_v100 = bt.indicators.SMA(self.data.volume, period=100)

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            if order.isbuy():
                if not self.has_base:
                    self.has_base = True
                    self.base_entry_price = order.executed.price
                    self.base_shares = abs(order.executed.size)
                else:
                    self.tranche_shares = abs(order.executed.size)
                    self.tranche_entry_price = order.executed.price
                    self.tp_price = self.tranche_entry_price * 1.12
                    self.tranche_entry_bar = len(self)
            elif order.issell():
                self.tranche_shares = 0
                self.tranche_entry_price = None
                self.tp_price = None
                self.tranche_entry_bar = None

        self.order = None

    def next(self):
        if self.order is not None:
            return

        # 1. Establish base position if not already present
        if not self.has_base:
            self.order = self.buy()
            return

        # 2. Additional tranche logic
        if self.tranche_shares == 0:
            cond1 = self.data.close[0] >= 0.9 * self.highest_252[0]
            cond2 = self.rsi[0] > 50
            cond3 = (
                self.data.close[0] / self.base_entry_price - 1.0
            ) >= 0.4
            cond4 = self.sma_v20[0] > self.sma_v100[0]

            if cond1 and cond2 and cond3 and cond4:
                unrealized_profit_base = (
                    self.data.close[0] - self.base_entry_price
                ) * self.base_shares
                buy_shares = math.floor(
                    0.5 * unrealized_profit_base / self.data.close[0]
                )
                if buy_shares > 0:
                    self.order = self.buy(size=buy_shares)
        else:
            tp_hit = self.data.close[0] >= self.tp_price
            drawdown_exit = self.data.close[0] <= 0.8 * self.base_entry_price
            time_stop = (len(self) - self.tranche_entry_bar) >= 126

            if tp_hit or drawdown_exit or time_stop:
                self.order = self.sell(size=self.tranche_shares)
