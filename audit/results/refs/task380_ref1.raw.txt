import backtrader as bt


class MACDATRTendFollowingStrategy(bt.Strategy):
    params = (
        ("stake", 100),
    )

    def __init__(self):
        self.macd = bt.indicators.MACDHisto(
            self.data.close,
            period_me1=12,
            period_me2=26,
            period_signal=9,
        )
        self.atr = bt.indicators.ATR(self.data, period=14)

        self.entry_order = None
        self.exit_order = None
        self.stop_order = None

    def next(self):
        histogram_crosses_above = (
            self.macd.histo[-1] <= 0.5 and self.macd.histo[0] > 0.5
        )
        histogram_crosses_below = (
            self.macd.histo[-1] >= -0.5 and self.macd.histo[0] < -0.5
        )

        if self.position:
            opposing_signal = (
                (self.position.size > 0 and histogram_crosses_below)
                or (self.position.size < 0 and histogram_crosses_above)
            )

            if opposing_signal and self.exit_order is None:
                if self.stop_order is not None:
                    self.cancel(self.stop_order)
                self.exit_order = self.close()

            return

        if self.entry_order is not None or self.exit_order is not None:
            return

        if histogram_crosses_above:
            self.entry_order = self.buy(size=self.p.stake)
        elif histogram_crosses_below:
            self.entry_order = self.sell(size=self.p.stake)

    def notify_order(self, order):
        if order.status in (
            order.Submitted,
            order.Accepted,
        ):
            return

        if order.ref == getattr(self.entry_order, "ref", None):
            if order.status == order.Completed:
                if order.isbuy():
                    stop_price = order.executed.price - (2 * self.atr[0])
                    self.stop_order = self.sell(
                        size=order.executed.size,
                        exectype=bt.Order.Stop,
                        price=stop_price,
                    )
                else:
                    stop_price = order.executed.price + (2 * self.atr[0])
                    self.stop_order = self.buy(
                        size=abs(order.executed.size),
                        exectype=bt.Order.Stop,
                        price=stop_price,
                    )
            self.entry_order = None

        if order.ref == getattr(self.exit_order, "ref", None):
            self.exit_order = None

        if order.ref == getattr(self.stop_order, "ref", None):
            self.stop_order = None
