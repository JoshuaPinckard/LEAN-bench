```python
import backtrader as bt


class SupertrendStrategy(bt.Strategy):
    params = (
        ("atr_period", 10),
        ("multiplier", 3.0),
    )

    def __init__(self):
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.order = None
        self.prev_final_upper = None
        self.prev_final_lower = None
        self.prev_trend = None
        self.prev_close = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        close_val = self.data.close[0]
        high_val = self.data.high[0]
        low_val = self.data.low[0]
        atr_val = self.atr[0]

        hl2 = (high_val + low_val) / 2.0
        basic_upper = hl2 + self.p.multiplier * atr_val
        basic_lower = hl2 - self.p.multiplier * atr_val

        if self.prev_final_upper is None:
            final_upper = basic_upper
            final_lower = basic_lower
            if close_val <= final_upper:
                trend = -1
            else:
                trend = 1
        else:
            if basic_upper < self.prev_final_upper or self.prev_close > self.prev_final_upper:
                final_upper = basic_upper
            else:
                final_upper = self.prev_final_upper

            if basic_lower > self.prev_final_lower or self.prev_close < self.prev_final_lower:
                final_lower = basic_lower
            else:
                final_lower = self.prev_final_lower

            if self.prev_trend == 1:
                if close_val < final_lower:
                    trend = -1
                else:
                    trend = 1
            else:
                if close_val > final_upper:
                    trend = 1
                else:
                    trend = -1

        if not self.order and self.prev_trend is not None:
            if self.prev_trend == -1 and trend == 1:
                if self.position.size < 0:
                    self.close()
                    self.order = self.buy()
                elif self.position.size == 0:
                    self.order = self.buy()
            elif self.prev_trend == 1 and trend == -1:
                if self.position.size > 0:
                    self.close()
                    self.order = self.sell()
                elif self.position.size == 0:
                    self.order = self.sell()

        self.prev_final_upper = final_upper
        self.prev_final_lower = final_lower
        self.prev_trend = trend
        self.prev_close = close_val
```
