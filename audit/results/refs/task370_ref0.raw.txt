import backtrader as bt


class SessionVWAP(bt.Indicator):
    lines = ('vwap',)
    plotinfo = dict(subplot=False)

    def __init__(self):
        self.cum_pv = 0.0
        self.cum_vol = 0.0
        self.last_date = None
        self.addminperiod(1)

    def next(self):
        dt = self.data.datetime.date(0)
        if dt != self.last_date:
            self.cum_pv = 0.0
            self.cum_vol = 0.0
            self.last_date = dt
        typical_price = (self.data.high[0] + self.data.low[0] + self.data.close[0]) / 3.0
        self.cum_pv += typical_price * self.data.volume[0]
        self.cum_vol += self.data.volume[0]
        if self.cum_vol != 0:
            self.lines.vwap[0] = self.cum_pv / self.cum_vol
        else:
            self.lines.vwap[0] = typical_price


class VwapAtrTrendStrategy(bt.Strategy):
    params = (
        ('atr_period', 14),
        ('atr_mult', 2.0),
        ('stop_pct', 0.01),
        ('use_atr_stop', True),
        ('use_pct_stop', True),
        ('stake', 1),
    )

    def __init__(self):
        self.vwap = SessionVWAP(self.data)
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.crossover = bt.indicators.CrossOver(self.data.close, self.vwap.vwap)
        self.order = None
        self.entry_price = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return
        if order.status == order.Completed:
            if self.position:
                self.entry_price = order.executed.price
            else:
                self.entry_price = None
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def _stop_level(self, is_long):
        if is_long:
            atr_stop = self.entry_price - self.p.atr_mult * self.atr[0]
            pct_stop = self.entry_price * (1 - self.p.stop_pct)
            if self.p.use_atr_stop and self.p.use_pct_stop:
                return max(atr_stop, pct_stop)
            if self.p.use_atr_stop:
                return atr_stop
            if self.p.use_pct_stop:
                return pct_stop
            return None
        else:
            atr_stop = self.entry_price + self.p.atr_mult * self.atr[0]
            pct_stop = self.entry_price * (1 + self.p.stop_pct)
            if self.p.use_atr_stop and self.p.use_pct_stop:
                return min(atr_stop, pct_stop)
            if self.p.use_atr_stop:
                return atr_stop
            if self.p.use_pct_stop:
                return pct_stop
            return None

    def next(self):
        if self.order:
            return

        if not self.position:
            if self.crossover > 0:
                self.order = self.buy(size=self.p.stake)
            elif self.crossover < 0:
                self.order = self.sell(size=self.p.stake)
            return

        if self.position.size > 0:
            stop_level = self._stop_level(True)
            if stop_level is not None and self.data.low[0] <= stop_level:
                self.order = self.close()
                return
            if self.crossover < 0:
                self.order = self.close()
                return
        elif self.position.size < 0:
            stop_level = self._stop_level(False)
            if stop_level is not None and self.data.high[0] >= stop_level:
                self.order = self.close()
                return
            if self.crossover > 0:
                self.order = self.close()
                return
