class WilliamsPercentRMeanReversion(bt.Strategy):
    params = dict(
        wr_period=14,
        sma_period=20,
        stddev_period=20,
        oversold=-80.0,
        overbought=-20.0,
        stop_pct=0.02,
    )

    def __init__(self):
        self.wr = bt.indicators.WilliamsR(self.data, period=self.p.wr_period)
        self.sma_wr = bt.indicators.SMA(self.wr, period=self.p.sma_period)
        self.dev = self.wr - self.sma_wr
        self.stddev_dev = bt.indicators.StandardDeviation(self.dev, period=self.p.stddev_period)

        self.order = None
        self.stop_price = None

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return

        if order.status == order.Completed:
            if self.position:
                if order.isbuy():
                    self.stop_price = order.executed.price * (1 - self.p.stop_pct)
                else:
                    self.stop_price = order.executed.price * (1 + self.p.stop_pct)
            else:
                self.stop_price = None

        self.order = None

    def next(self):
        if self.order:
            return

        wr = self.wr[0]
        wr_prev = self.wr[-1]
        sma_wr = self.sma_wr[0]
        sma_wr_prev = self.sma_wr[-1]

        if self.position:
            if self.position.size > 0:
                stop_hit = self.data.close[0] <= self.stop_price
                exit_cross = wr_prev >= sma_wr_prev and wr < sma_wr
            else:
                stop_hit = self.data.close[0] >= self.stop_price
                exit_cross = wr_prev <= sma_wr_prev and wr > sma_wr

            if stop_hit or exit_cross:
                self.order = self.close()
            return

        entry_filter = abs(self.dev[0]) > 2 * self.stddev_dev[0]
        if not entry_filter:
            return

        long_cross = wr_prev <= self.p.oversold and wr > self.p.oversold
        short_cross = wr_prev >= self.p.overbought and wr < self.p.overbought

        if long_cross:
            self.order = self.buy()
        elif short_cross:
            self.order = self.sell()
