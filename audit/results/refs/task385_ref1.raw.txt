import backtrader as bt


class AdaptiveMACDBandStrategy(bt.Strategy):
    params = (
        ('macd_fast', 12),
        ('macd_slow', 26),
        ('macd_signal', 9),
        ('band_period', 20),
        ('band_stddev', 2),
        ('atr_period', 14),
    )

    def __init__(self):
        self.macd = bt.indicators.MACD(
            self.data,
            period_me1=self.p.macd_fast,
            period_me2=self.p.macd_slow,
            period_signal=self.p.macd_signal,
        )
        self.histogram = self.macd.macd - self.macd.signal
        self.hist_sma = bt.indicators.SMA(
            self.histogram, period=self.p.band_period
        )
        self.hist_stddev = bt.indicators.StandardDeviation(
            self.histogram, period=self.p.band_period
        )
        self.upper_band = self.hist_sma + self.p.band_stddev * self.hist_stddev
        self.lower_band = self.hist_sma - self.p.band_stddev * self.hist_stddev

        self.macd_cross = bt.indicators.CrossOver(self.macd.macd, self.macd.signal)
        self.hist_upper_cross = bt.indicators.CrossOver(
            self.histogram, self.upper_band
        )
        self.hist_lower_cross = bt.indicators.CrossOver(
            self.histogram, self.lower_band
        )
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)

        self.order = None
        self.order_kind = None
        self.stop_order = None

    def next(self):
        if self.order:
            return

        long_setup = self.macd_cross[0] > 0 and self.hist_upper_cross[0] > 0
        short_setup = self.macd_cross[0] < 0 and self.hist_lower_cross[0] < 0

        if not self.position:
            if long_setup:
                self.order_kind = 'long_entry'
                self.order = self.buy()
            elif short_setup:
                self.order_kind = 'short_entry'
                self.order = self.sell()
        elif self.position.size > 0 and short_setup:
            if self.stop_order:
                self.cancel(self.stop_order)
            self.order_kind = 'exit'
            self.order = self.close()
        elif self.position.size < 0 and long_setup:
            if self.stop_order:
                self.cancel(self.stop_order)
            self.order_kind = 'exit'
            self.order = self.close()

    def notify_order(self, order):
        if order is self.stop_order:
            if order.status in (
                order.Completed,
                order.Canceled,
                order.Margin,
                order.Rejected,
            ):
                self.stop_order = None
            return

        if order is not self.order:
            return

        if order.status not in (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            return

        order_kind = self.order_kind
        self.order = None
        self.order_kind = None

        if order.status != order.Completed:
            return

        if order_kind == 'long_entry':
            entry_price = order.executed.price
            stop_price = max(
                entry_price * 0.98,
                entry_price - 2 * self.atr[0],
            )
            self.stop_order = self.sell(
                exectype=bt.Order.Stop,
                price=stop_price,
            )
        elif order_kind == 'short_entry':
            entry_price = order.executed.price
            stop_price = min(
                entry_price * 1.02,
                entry_price + 2 * self.atr[0],
            )
            self.stop_order = self.buy(
                exectype=bt.Order.Stop,
                price=stop_price,
            )
