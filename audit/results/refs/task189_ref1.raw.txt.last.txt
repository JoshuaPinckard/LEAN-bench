import backtrader as bt
import math


class MomentumVolumeVolatilityStrategy(bt.Strategy):
    params = (
        ("sma_fast_period", 20),
        ("sma_slow_period", 50),
        ("rsi_period", 14),
        ("volume_sma_period", 20),
        ("atr_period", 20),
        ("atr_sma_period", 1950),
    )

    def __init__(self):
        self.sma20 = bt.indicators.SMA(self.data.close, period=self.p.sma_fast_period)
        self.sma50 = bt.indicators.SMA(self.data.close, period=self.p.sma_slow_period)
        self.rsi = bt.indicators.RSI(self.data.close, period=self.p.rsi_period)
        self.volume_sma20 = bt.indicators.SMA(
            self.data.volume, period=self.p.volume_sma_period
        )
        self.atr20 = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.atr_sma1950 = bt.indicators.SMA(
            self.atr20, period=self.p.atr_sma_period
        )
        self.sma_cross = bt.indicators.CrossOver(self.sma20, self.sma50)

        self.order = None
        self.entry_price = None
        self.initial_stop = None
        self.crossover_bars_remaining = 0

    def _in_entry_session(self):
        current = self.data.datetime.datetime(0)
        return (
            (current.hour > 9 or (current.hour == 9 and current.minute >= 30))
            and (current.hour < 16 or (current.hour == 16 and current.minute == 0))
        )

    def _last_10_completed_bars_low(self):
        return min(self.data.low[-offset] for offset in range(1, 11))

    def next(self):
        if self.order:
            return

        if not self.position:
            if self.sma_cross[0] > 0:
                self.crossover_bars_remaining = 5

            if self.crossover_bars_remaining >= 0:
                price = self.data.close[0]
                stop = self._last_10_completed_bars_low()

                entry_conditions = (
                    self._in_entry_session()
                    and 10 <= price <= 200
                    and self.data.volume[0] > self.volume_sma20[0] * 1.2
                    and 40 <= self.rsi[0] <= 60
                    and self.atr20[0] > self.atr_sma1950[0]
                )

                if entry_conditions:
                    risk_per_share = price - stop
                    if (
                        risk_per_share > 0
                        and risk_per_share <= price * 0.02
                    ):
                        size = math.floor(
                            self.broker.getvalue() * 0.01 / risk_per_share
                        )
                        if size > 0:
                            self.initial_stop = stop
                            self.order = self.buy(size=size)
                            self.crossover_bars_remaining = -1
                            return

                self.crossover_bars_remaining -= 1

            return

        trailing_stop = self._last_10_completed_bars_low()
        exit_conditions = (
            self.data.close[0] < trailing_stop
            or (
                self.entry_price is not None
                and self.data.close[0] >= self.entry_price * 1.02
            )
            or self.sma_cross[0] < 0
        )

        if exit_conditions:
            self.order = self.close()

    def notify_order(self, order):
        if order.status in (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            if order.status == order.Completed and order.isbuy():
                self.entry_price = order.executed.price

            if order.status != order.Completed or (
                order.issell() and not self.position
            ):
                if not self.position:
                    self.entry_price = None
                    self.initial_stop = None

            self.order = None
