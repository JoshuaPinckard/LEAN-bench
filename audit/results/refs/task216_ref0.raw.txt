import backtrader as bt


class TargetWeightRebalanceStrategy(bt.Strategy):
    params = (
        ('weights', [1.0]),
        ('rebalance_period', 'quarterly'),
        ('slippage_perc', 0.0005),
    )

    def __init__(self):
        self.order = None
        self._last_period = None
        self._weight_idx = 0
        self.broker.set_slippage_perc(perc=self.p.slippage_perc)

    def _current_period(self, dt):
        if self.p.rebalance_period == 'yearly':
            return dt.year
        quarter = (dt.month - 1) // 3 + 1
        return (dt.year, quarter)

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        if self.order:
            return

        dt = self.data.datetime.date(0)
        period = self._current_period(dt)

        # None -> first bar always counts as the first rebalance date
        if period != self._last_period:
            self._last_period = period
            if self._weight_idx < len(self.p.weights):
                w = self.p.weights[self._weight_idx]
            else:
                w = self.p.weights[-1]
            self._weight_idx += 1
            self.order = self.order_target_percent(target=w)
