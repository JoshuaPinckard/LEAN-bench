import backtrader as bt


class ConsolidationBreakoutStrategy(bt.Strategy):
    def __init__(self):
        self.ema8 = bt.indicators.EMA(self.data.close, period=8)
        self.order = None
        self.first_exit_locked = False
        self.first_exit_order = False

    def next(self):
        if self.order:
            return

        close = self.data.close[0]

        if not self.position:
            self.first_exit_locked = False
            self.first_exit_order = False
            if close > 8.47:
                self.order = self.buy()
            return

        if close < 8.02:
            self.order = self.close()
        elif not self.first_exit_locked and close >= 9.08:
            self.first_exit_order = True
            self.order = self.sell(size=self.position.size / 2.0)
        elif self.first_exit_locked and close < self.ema8[0]:
            self.order = self.close()

    def notify_order(self, order):
        if order.status == order.Completed:
            if order.issell() and self.first_exit_order:
                self.first_exit_locked = True
                self.first_exit_order = False

        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            if order.status != order.Completed and self.first_exit_order:
                self.first_exit_order = False
            self.order = None
