import backtrader as bt


class VolumePriceSMAStrategy(bt.Strategy):
    params = (
        ("vol_fast_period", 10),
        ("vol_slow_period", 50),
        ("price_sma_period", 200),
        ("stop_pct", 0.01),
        ("tp_pct", 0.02),
        ("risk_pct", 0.01),
    )

    def __init__(self):
        self.vol_sma_fast = bt.indicators.SMA(
            self.data.volume, period=self.p.vol_fast_period
        )
        self.vol_sma_slow = bt.indicators.SMA(
            self.data.volume, period=self.p.vol_slow_period
        )
        self.vol_cross = bt.indicators.CrossOver(
            self.vol_sma_fast, self.vol_sma_slow
        )

        self.price_sma = bt.indicators.SMA(
            self.data.close, period=self.p.price_sma_period
        )
        self.price_cross = bt.indicators.CrossOver(
            self.data.close, self.price_sma
        )

        self.entry_order = None
        self.stop_order = None
        self.tp_order = None

    def _cancel_pending_exit_orders(self):
        if self.stop_order:
            self.cancel(self.stop_order)
            self.stop_order = None
        if self.tp_order:
            self.cancel(self.tp_order)
            self.tp_order = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            if order == self.entry_order:
                self.entry_order = None
                exec_price = order.executed.price
                size = abs(order.executed.size)

                if order.isbuy():
                    stop_price = exec_price * (1.0 - self.p.stop_pct)
                    self.stop_order = self.sell(
                        size=size, exectype=bt.Order.Stop, price=stop_price
                    )
                    if self.p.tp_pct and self.p.tp_pct > 0:
                        tp_price = exec_price * (1.0 + self.p.tp_pct)
                        self.tp_order = self.sell(
                            size=size,
                            exectype=bt.Order.Limit,
                            price=tp_price,
                            oco=self.stop_order,
                        )
                else:
                    stop_price = exec_price * (1.0 + self.p.stop_pct)
                    self.stop_order = self.buy(
                        size=size, exectype=bt.Order.Stop, price=stop_price
                    )
                    if self.p.tp_pct and self.p.tp_pct > 0:
                        tp_price = exec_price * (1.0 - self.p.tp_pct)
                        self.tp_order = self.buy(
                            size=size,
                            exectype=bt.Order.Limit,
                            price=tp_price,
                            oco=self.stop_order,
                        )

            elif order in [self.stop_order, self.tp_order]:
                if order == self.stop_order:
                    self.stop_order = None
                if order == self.tp_order:
                    self.tp_order = None
                if not self.position:
                    self._cancel_pending_exit_orders()

        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            if order == self.entry_order:
                self.entry_order = None
            if order == self.stop_order:
                self.stop_order = None
            if order == self.tp_order:
                self.tp_order = None

    def next(self):
        if self.position.size > 0:
            if self.vol_cross[0] < 0 or self.price_cross[0] < 0:
                self._cancel_pending_exit_orders()
                self.close()
            return

        if self.position.size < 0:
            if self.vol_cross[0] > 0 or self.price_cross[0] > 0:
                self._cancel_pending_exit_orders()
                self.close()
            return

        if self.entry_order:
            return

        risk_amount = self.broker.getvalue() * self.p.risk_pct
        stop_dist = self.data.close[0] * self.p.stop_pct

        if stop_dist <= 0:
            return

        size = int(risk_amount / stop_dist)
        if size <= 0:
            return

        if self.vol_cross[0] > 0 and self.data.close[0] > self.price_sma[0]:
            self.entry_order = self.buy(size=size)
        elif self.vol_cross[0] < 0 and self.data.close[0] < self.price_sma[0]:
            self.entry_order = self.sell(size=size)
