import backtrader as bt


class MomentumTrendStrategy(bt.Strategy):
    params = (
        ("stop_loss_method", "percent"),  # "percent" or "atr"
        ("reverse_on_opposite_signal", False),
    )

    def __init__(self):
        self.momentum = (self.data.close / self.data.close(-10) - 1) * 100
        self.sma = bt.indicators.SimpleMovingAverage(self.data.close, period=50)
        self.atr = bt.indicators.ATR(self.data, period=14)

        self.pending_orders = set()
        self.entry_order_refs = set()
        self.entry_price = None
        self.entry_atr = None

    def _submit_entry(self, is_long):
        order = self.buy() if is_long else self.sell()
        self.pending_orders.add(order.ref)
        self.entry_order_refs.add(order.ref)

    def _submit_close(self):
        order = self.close()
        self.pending_orders.add(order.ref)

    def next(self):
        if self.pending_orders:
            return

        momentum = self.momentum[0]

        if not self.position:
            if momentum > 5 and self.data.close[0] > self.sma[0]:
                self._submit_entry(is_long=True)
            elif momentum < -5 and self.data.close[0] < self.sma[0]:
                self._submit_entry(is_long=False)
            return

        if self.position.size > 0:
            if self.p.stop_loss_method == "atr":
                stop_price = self.entry_price - self.entry_atr
            else:
                stop_price = self.entry_price * 0.98

            opposite_signal = momentum < -5
            stop_hit = self.data.low[0] <= stop_price

            if opposite_signal or stop_hit:
                self._submit_close()
                if opposite_signal and self.p.reverse_on_opposite_signal:
                    self._submit_entry(is_long=False)

        else:
            if self.p.stop_loss_method == "atr":
                stop_price = self.entry_price + self.entry_atr
            else:
                stop_price = self.entry_price * 1.02

            opposite_signal = momentum > 5
            stop_hit = self.data.high[0] >= stop_price

            if opposite_signal or stop_hit:
                self._submit_close()
                if opposite_signal and self.p.reverse_on_opposite_signal:
                    self._submit_entry(is_long=True)

    def notify_order(self, order):
        if order.status in (
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            self.pending_orders.discard(order.ref)

            if order.status == order.Completed and order.ref in self.entry_order_refs:
                self.entry_price = order.executed.price
                self.entry_atr = self.atr[0]

            self.entry_order_refs.discard(order.ref)
