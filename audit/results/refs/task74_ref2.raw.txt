import backtrader as bt


class DynamicStopStrategy(bt.Strategy):
    params = (
        ("size", 0.1),
        ("sl_points", 10.0),
        ("tp_points", 50.0),
        ("step_points", 10.0),
    )

    def __init__(self):
        self.order = None
        self.entry_price = None
        self.sl_price = None
        self.tp_price = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            if not self.position:
                self.entry_price = None
                self.sl_price = None
                self.tp_price = None
            else:
                self.entry_price = order.executed.price
                if self.position.size > 0:
                    self.sl_price = self.entry_price - self.p.sl_points
                    self.tp_price = self.entry_price + self.p.tp_points
                elif self.position.size < 0:
                    self.sl_price = self.entry_price + self.p.sl_points
                    self.tp_price = self.entry_price - self.p.tp_points

        self.order = None

    def next(self):
        if self.order:
            return

        if not self.position:
            if self.data.close[0] > self.data.open[0]:
                self.order = self.buy(size=self.p.size)
            elif self.data.close[0] < self.data.open[0]:
                self.order = self.sell(size=self.p.size)
        else:
            if self.position.size > 0:
                high_p = self.data.high[0]
                low_p = self.data.low[0]

                profit = round(high_p - self.entry_price, 6)
                if profit >= self.p.step_points:
                    step = int(profit // self.p.step_points)
                    new_sl = self.entry_price + (step - 1) * self.p.step_points
                    self.sl_price = max(self.sl_price, new_sl)

                if high_p >= self.tp_price:
                    self.order = self.close()
                elif low_p <= self.sl_price:
                    self.order = self.close()

            elif self.position.size < 0:
                high_p = self.data.high[0]
                low_p = self.data.low[0]

                profit = round(self.entry_price - low_p, 6)
                if profit >= self.p.step_points:
                    step = int(profit // self.p.step_points)
                    new_sl = self.entry_price - (step - 1) * self.p.step_points
                    self.sl_price = min(self.sl_price, new_sl)

                if low_p <= self.tp_price:
                    self.order = self.close()
                elif high_p >= self.sl_price:
                    self.order = self.close()
