import backtrader as bt


class MicroMomentumStrategy(bt.Strategy):
    params = dict(
        ema_fast_period=5,
        ema_slow_period=10,
        rsi_period=7,
        vol_sma_period=20,
        atr_period=14,
        tp_atr_mult=0.15,
        sl_atr_mult=0.10,
        time_exit_bars=3,
        atr_pct_threshold=0.001,
        rsi_long_threshold=55,
        rsi_short_threshold=45,
        position_pct=0.001,
        slippage_pct=0.0005,
        close_cutoff_minutes=15 * 60 + 55,
    )

    def __init__(self):
        self.broker.set_slippage_perc(
            perc=self.p.slippage_pct,
            slip_open=True,
            slip_limit=True,
            slip_match=True,
            slip_out=False,
        )

        self.ema_fast = bt.indicators.EMA(self.data.close, period=self.p.ema_fast_period)
        self.ema_slow = bt.indicators.EMA(self.data.close, period=self.p.ema_slow_period)
        self.rsi = bt.indicators.RSI(self.data.close, period=self.p.rsi_period)
        self.vol_sma = bt.indicators.SMA(self.data.volume, period=self.p.vol_sma_period)
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)
        self.crossover = bt.indicators.CrossOver(self.ema_fast, self.ema_slow)

        self.order = None
        self.entry_price = None
        self.entry_bar = None
        self.tp_price = None
        self.sl_price = None

    def notify_order(self, order):
        if order.status in (order.Submitted, order.Accepted):
            return

        if order.status == order.Completed:
            if self.position.size != 0 and self.entry_price is None:
                self.entry_price = order.executed.price
                self.entry_bar = len(self)
                atr_val = self.atr[0]
                if self.position.size > 0:
                    self.tp_price = self.entry_price + self.p.tp_atr_mult * atr_val
                    self.sl_price = self.entry_price - self.p.sl_atr_mult * atr_val
                else:
                    self.tp_price = self.entry_price - self.p.tp_atr_mult * atr_val
                    self.sl_price = self.entry_price + self.p.sl_atr_mult * atr_val
            elif self.position.size == 0:
                self.entry_price = None
                self.entry_bar = None
                self.tp_price = None
                self.sl_price = None

        self.order = None

    def _attempt_close(self):
        if not self.order:
            self.order = self.close()

    def next(self):
        t = self.data.datetime.time(0)
        current_minutes = t.hour * 60 + t.minute
        near_close = current_minutes >= self.p.close_cutoff_minutes

        if self.position:
            if near_close:
                self._attempt_close()
                return

            close_val = self.data.close[0]
            if self.position.size > 0:
                tp_sl_hit = close_val >= self.tp_price or close_val <= self.sl_price
            else:
                tp_sl_hit = close_val <= self.tp_price or close_val >= self.sl_price

            if tp_sl_hit:
                self._attempt_close()
                return

            if len(self) - self.entry_bar >= self.p.time_exit_bars:
                self._attempt_close()
            return

        if self.order:
            return

        atr_val = self.atr[0]
        close_val = self.data.close[0]

        volume_filter = self.data.volume[0] > self.vol_sma[0]
        volatility_filter = (atr_val / close_val) > self.p.atr_pct_threshold

        if not (volatility_filter and volume_filter):
            return

        size = (self.p.position_pct * self.broker.getvalue()) / close_val
        if size <= 0:
            return

        if self.crossover[0] > 0 and self.rsi[0] > self.p.rsi_long_threshold:
            self.order = self.buy(size=size)
        elif self.crossover[0] < 0 and self.rsi[0] < self.p.rsi_short_threshold:
            self.order = self.sell(size=size)
