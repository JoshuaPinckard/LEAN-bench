import backtrader as bt


class ConsolidationBreakoutStrategy(bt.Strategy):
    params = dict(
        bb_period=20,
        bb_dev=2,
        rsi_period=14,
        consolidation_bars=10,
        volume_period=20,
        volume_mult=1.5,
        take_profit_pct=0.02,
        stop_loss_pct=0.015,
        quick_return_bars=3,
        position_pct=0.01,
    )

    def __init__(self):
        self.bb = bt.indicators.BollingerBands(
            self.data.close, period=self.p.bb_period, devfactor=self.p.bb_dev
        )
        self.rsi = bt.indicators.RSI(self.data.close, period=self.p.rsi_period)
        self.avg_volume = bt.indicators.SimpleMovingAverage(
            self.data.volume, period=self.p.volume_period
        )

        self.order = None
        self.entry_pending = False
        self.entry_price = None
        self.consolidation_count = 0
        self.bars_since_entry = 0

    def notify_order(self, order):
        if order.status == order.Completed and self.entry_pending:
            self.entry_price = order.executed.price
            self.bars_since_entry = 0

        if order.status in (order.Completed, order.Canceled, order.Margin, order.Rejected):
            self.order = None
            self.entry_pending = False

    def next(self):
        if self.order:
            return

        close = self.data.close[0]
        upper = self.bb.top[0]
        lower = self.bb.bot[0]
        mid = self.bb.mid[0]
        in_bands = lower <= close <= upper

        if self.position:
            self.bars_since_entry += 1
            is_long = self.position.size > 0
            exit_signal = False

            if self.bars_since_entry <= self.p.quick_return_bars and in_bands:
                exit_signal = True
            elif is_long:
                if close >= self.entry_price * (1 + self.p.take_profit_pct):
                    exit_signal = True
                elif close <= self.entry_price * (1 - self.p.stop_loss_pct):
                    exit_signal = True
                elif close < mid:
                    exit_signal = True
            else:
                if close <= self.entry_price * (1 - self.p.take_profit_pct):
                    exit_signal = True
                elif close >= self.entry_price * (1 + self.p.stop_loss_pct):
                    exit_signal = True
                elif close > mid:
                    exit_signal = True

            if exit_signal:
                self.order = self.close()
            return

        volume_ok = self.data.volume[0] > self.p.volume_mult * self.avg_volume[0]

        if self.consolidation_count >= self.p.consolidation_bars and volume_ok:
            capital = self.broker.getvalue()
            size = (self.p.position_pct * capital) / close

            if close > upper and self.rsi[0] > 50:
                self.order = self.buy(size=size)
                self.entry_pending = True
            elif close < lower and self.rsi[0] < 50:
                self.order = self.sell(size=size)
                self.entry_pending = True

        if in_bands:
            self.consolidation_count += 1
        else:
            self.consolidation_count = 0
