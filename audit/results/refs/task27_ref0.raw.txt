import backtrader as bt


class LongConditionExitPriceStrategy(bt.Strategy):
    params = dict(
        fast_period=10,
        slow_period=30,
    )

    def __init__(self):
        self.sma_fast = bt.indicators.SMA(self.data.close, period=self.p.fast_period)
        self.sma_slow = bt.indicators.SMA(self.data.close, period=self.p.slow_period)
        self.longcondition = self.sma_fast > self.sma_slow

        self.order = None
        self.previous_exit_price = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status == order.Completed:
            if order.issell():
                self.previous_exit_price = order.executed.price
            self.order = None
        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        if self.order:
            return

        if not self.position:
            signal_ok = bool(self.longcondition[0])
            price_ok = (
                self.previous_exit_price is None
                or self.data.close[-1] < self.previous_exit_price
            )
            if signal_ok and price_ok:
                cash = self.broker.getcash()
                price = self.data.close[0]
                size = int(cash / price)
                if size > 0:
                    self.order = self.buy(size=size)
        else:
            if not self.longcondition[0]:
                self.order = self.close()
