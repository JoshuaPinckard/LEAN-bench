import backtrader as bt


class RegimeSwitchingStrategy(bt.Strategy):
    params = (
        ("sma_long_period", 200),
        ("sma_short_period", 50),
        ("rsi_period", 14),
        ("atr_period", 14),
        ("stddev_period", 20),
        ("atr_ma_period", 50),
        ("close_sma_threshold", 0.02),
        ("trend_filter_threshold", 0.05),
        ("sl_atr_mult", 1.5),
        ("tp_sl_mult", 3.0),
        ("capital_risk_pct", 0.01),
    )

    def __init__(self):
        self.sma200 = bt.ind.SMA(self.data.close, period=self.p.sma_long_period)
        self.sma50 = bt.ind.SMA(self.data.close, period=self.p.sma_short_period)
        self.rsi = bt.ind.RSI(self.data.close, period=self.p.rsi_period)
        self.atr = bt.ind.ATR(self.data, period=self.p.atr_period)
        self.stddev = bt.ind.StdDev(self.data.close, period=self.p.stddev_period)
        self.atr_sma50 = bt.ind.SMA(self.atr, period=self.p.atr_ma_period)
        self.crossover = bt.ind.CrossOver(self.data.close, self.sma50)

        self.order = None
        self.is_entry = False
        self.pending_atr = None
        self.entry_price = None
        self.sl_price = None
        self.tp_price = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            if self.is_entry:
                self.entry_price = order.executed.price
                self.sl_dist = self.p.sl_atr_mult * self.pending_atr
                self.tp_dist = self.p.tp_sl_mult * self.sl_dist

                if order.isbuy():
                    self.sl_price = self.entry_price - self.sl_dist
                    self.tp_price = self.entry_price + self.tp_dist
                else:
                    self.sl_price = self.entry_price + self.sl_dist
                    self.tp_price = self.entry_price - self.tp_dist
                self.is_entry = False
            else:
                self.entry_price = None
                self.sl_price = None
                self.tp_price = None
        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            if self.is_entry:
                self.is_entry = False

        self.order = None

    def get_regime(self):
        close = self.data.close[0]
        sma200 = self.sma200[0]
        rsi = self.rsi[0]
        atr = self.atr[0]
        atr_sma = self.atr_sma50[0]

        diff_pct = abs(close - sma200) / sma200
        high_vol = atr > atr_sma

        if diff_pct < self.p.close_sma_threshold and high_vol:
            return 3  # Sideways / Volatile
        elif close > sma200 and rsi > 50 and high_vol:
            return 1  # Uptrend / Bull
        elif close < sma200 and rsi < 50 and high_vol:
            return 2  # Downtrend / Bear
        else:
            return 4  # Stable / Low Volatility

    def next(self):
        if self.order:
            return

        regime = self.get_regime()

        if not self.position:
            diff_pct = abs(self.data.close[0] - self.sma200[0]) / self.sma200[0]
            if diff_pct > self.p.trend_filter_threshold:
                return

            if regime == 1:
                if self.crossover[0] > 0:
                    size = (self.broker.getvalue() * self.p.capital_risk_pct) / self.data.close[0]
                    if size > 0:
                        self.pending_atr = self.atr[0]
                        self.is_entry = True
                        self.order = self.buy(size=size)
            elif regime == 2:
                if self.crossover[0] < 0:
                    size = (self.broker.getvalue() * self.p.capital_risk_pct) / self.data.close[0]
                    if size > 0:
                        self.pending_atr = self.atr[0]
                        self.is_entry = True
                        self.order = self.sell(size=size)
        else:
            if self.position.size > 0:
                sl_hit = self.sl_price is not None and self.data.low[0] <= self.sl_price
                tp_hit = self.tp_price is not None and self.data.high[0] >= self.tp_price
                cross_exit = self.crossover[0] < 0
                rsi_exit = self.rsi[0] < 30
                regime_exit = regime == 2

                if sl_hit or tp_hit or cross_exit or rsi_exit or regime_exit:
                    self.order = self.close()

            elif self.position.size < 0:
                sl_hit = self.sl_price is not None and self.data.high[0] >= self.sl_price
                tp_hit = self.tp_price is not None and self.data.low[0] <= self.tp_price
                cross_exit = self.crossover[0] > 0
                rsi_exit = self.rsi[0] > 70
                regime_exit = regime == 1

                if sl_hit or tp_hit or cross_exit or rsi_exit or regime_exit:
                    self.order = self.close()
