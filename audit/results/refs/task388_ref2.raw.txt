import backtrader as bt


class WilliamsRMomentumATR(bt.Strategy):
    params = (
        ("williams_period", 14),
        ("momentum_period", 14),
        ("atr_period", 14),
        ("atr_mult", 2.0),
        ("trailing_stop", False),
    )

    def __init__(self):
        self.willr = bt.indicators.WilliamsR(
            self.data, period=self.p.williams_period
        )
        self.mom = bt.indicators.Momentum(
            self.data.close, period=self.p.momentum_period
        )
        self.atr = bt.indicators.ATR(self.data, period=self.p.atr_period)

        self.order = None
        self.entry_price = None
        self.entry_atr = None
        self.stop_price = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            if order.isbuy():
                if self.position.size > 0:
                    self.entry_price = order.executed.price
                    self.entry_atr = self.atr[0]
            elif order.issell():
                if self.position.size < 0:
                    self.entry_price = order.executed.price
                    self.entry_atr = self.atr[0]

            if not self.position:
                self.entry_price = None
                self.entry_atr = None
                self.stop_price = None

        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            pass

        self.order = None

    def next(self):
        if self.order:
            return

        long_signal = (
            self.willr[-1] <= -80.0 and self.willr[0] > -80.0
        ) and (self.mom[0] > 0)
        short_signal = (
            self.willr[-1] >= -20.0 and self.willr[0] < -20.0
        ) and (self.mom[0] < 0)

        if self.position.size > 0:
            if self.p.trailing_stop:
                current_stop = (
                    self.data.close[0] - self.p.atr_mult * self.atr[0]
                )
                if self.stop_price is None:
                    self.stop_price = current_stop
                else:
                    self.stop_price = max(self.stop_price, current_stop)
            else:
                if self.entry_price is not None and self.entry_atr is not None:
                    self.stop_price = (
                        self.entry_price - self.p.atr_mult * self.entry_atr
                    )
                else:
                    self.stop_price = (
                        self.data.close[0] - self.p.atr_mult * self.atr[0]
                    )

            if self.data.low[0] <= self.stop_price:
                self.order = self.close()
                self.stop_price = None
            elif short_signal:
                self.order = self.close()
                self.stop_price = None

        elif self.position.size < 0:
            if self.p.trailing_stop:
                current_stop = (
                    self.data.close[0] + self.p.atr_mult * self.atr[0]
                )
                if self.stop_price is None:
                    self.stop_price = current_stop
                else:
                    self.stop_price = min(self.stop_price, current_stop)
            else:
                if self.entry_price is not None and self.entry_atr is not None:
                    self.stop_price = (
                        self.entry_price + self.p.atr_mult * self.entry_atr
                    )
                else:
                    self.stop_price = (
                        self.data.close[0] + self.p.atr_mult * self.atr[0]
                    )

            if self.data.high[0] >= self.stop_price:
                self.order = self.close()
                self.stop_price = None
            elif long_signal:
                self.order = self.close()
                self.stop_price = None

        else:
            self.stop_price = None
            if long_signal:
                self.order = self.buy()
            elif short_signal:
                self.order = self.sell()
