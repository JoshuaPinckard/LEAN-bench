import backtrader as bt


class WilliamsRMomentumStrategy(bt.Strategy):
    params = (
        ('wr_period', 14),
        ('mom_period', 14),
        ('atr_period', 14),
        ('atr_mult', 2.0),
        ('trailing_stop', False),
    )

    def __init__(self):
        self.wr = bt.indicators.WilliamsR(period=self.p.wr_period)
        self.mom = bt.indicators.Momentum(period=self.p.mom_period)
        self.atr = bt.indicators.ATR(period=self.p.atr_period)
        self.order = None
        self.stop_price = None

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None

    def next(self):
        if self.order:
            return

        wr_prev = self.wr[-1]
        wr_curr = self.wr[0]
        mom_curr = self.mom[0]

        long_signal = (wr_prev <= -80.0) and (wr_curr > -80.0) and (mom_curr > 0.0)
        short_signal = (wr_prev >= -20.0) and (wr_curr < -20.0) and (mom_curr < 0.0)

        if self.position.size > 0:
            if self.stop_price is None:
                self.stop_price = self.position.price - self.p.atr_mult * self.atr[0]
            if self.p.trailing_stop:
                candidate = self.data.close[0] - self.p.atr_mult * self.atr[0]
                self.stop_price = max(self.stop_price, candidate)
            if self.data.low[0] <= self.stop_price or short_signal:
                self.order = self.close()
                self.stop_price = None
                return

        elif self.position.size < 0:
            if self.stop_price is None:
                self.stop_price = self.position.price + self.p.atr_mult * self.atr[0]
            if self.p.trailing_stop:
                candidate = self.data.close[0] + self.p.atr_mult * self.atr[0]
                self.stop_price = min(self.stop_price, candidate)
            if self.data.high[0] >= self.stop_price or long_signal:
                self.order = self.close()
                self.stop_price = None
                return

        else:
            self.stop_price = None
            if long_signal:
                self.order = self.buy()
            elif short_signal:
                self.order = self.sell()
