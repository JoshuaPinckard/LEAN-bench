import backtrader as bt


class WeeklyORBShort(bt.Strategy):
    def __init__(self):
        self.ema20 = bt.indicators.EMA(self.data.close, period=20)
        self.sma50 = bt.indicators.SMA(self.data.close, period=50)

        self.week_key = None
        self.week_high = None
        self.week_low = None
        self.week_mid = None
        self.breakout_active = False
        self.days_since_breakout = 0

        self.order = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return
        if order is self.order and order.status in [
            order.Completed, order.Canceled, order.Margin, order.Rejected
        ]:
            self.order = None

    def next(self):
        dt = self.data.datetime.date(0)
        year, week, _ = dt.isocalendar()
        week_key = (year, week)

        if week_key != self.week_key:
            self.week_key = week_key
            self.week_high = self.data.high[0]
            self.week_low = self.data.low[0]
            self.week_mid = (self.week_high + self.week_low) / 2.0
            self.breakout_active = False
            self.days_since_breakout = 0
            return

        if self.position or self.order is not None:
            return

        if not self.breakout_active:
            if self.data.close[0] < self.week_low:
                self.breakout_active = True
                self.days_since_breakout = 0
            return

        self.days_since_breakout += 1

        if self.days_since_breakout <= 2:
            touched_and_crossed = (
                self.data.high[0] >= self.week_mid
                and self.data.close[0] <= self.week_mid
            )
            trend_ok = (
                self.data.close[0] < self.ema20[0]
                and self.data.close[0] < self.sma50[0]
                and self.ema20[0] < self.sma50[0]
                and self.ema20[0] < self.ema20[-1]
            )
            if touched_and_crossed and trend_ok:
                stop_price = self.week_high
                target_price = self.week_low
                entry_price_est = self.data.close[0]
                stop_distance = stop_price - entry_price_est
                if stop_distance > 0:
                    risk_pct = 0.015
                    risk_amount = self.broker.getvalue() * risk_pct
                    size = int(risk_amount / stop_distance)
                    if size >= 1:
                        orders = self.sell_bracket(
                            size=size,
                            exectype=bt.Order.Market,
                            stopprice=stop_price,
                            limitprice=target_price,
                        )
                        self.order = orders[0]

        if self.days_since_breakout >= 2:
            self.breakout_active = False
