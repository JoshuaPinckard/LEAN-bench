import backtrader as bt
import math


class PseudoSpreadMeanReversion(bt.Strategy):
    params = (
        ("risk_pct", 0.01),
        ("atr_multiple", 1.0),
        ("liquidity_threshold", 1_000_000.0),
    )

    def __init__(self):
        self.sma200 = bt.indicators.SMA(self.data.close, period=200)
        self.sma50 = bt.indicators.SMA(self.data.close, period=50)
        self.atr14 = bt.indicators.ATR(self.data, period=14)
        self.atr_sma30 = bt.indicators.SMA(self.atr14, period=30)
        self.volume_sma20 = bt.indicators.SMA(self.data.volume, period=20)
        self.cross = bt.indicators.CrossOver(self.data.close, self.sma50)

        self.order = None
        self.entry_order = None
        self.d0 = None

    def notify_order(self, order):
        if order.status == order.Completed:
            if order is self.entry_order:
                if order.isbuy():
                    self.d0 = self.sma50[0] - order.executed.price
                else:
                    self.d0 = order.executed.price - self.sma50[0]

            if order is self.order:
                self.order = None

            if order is self.entry_order:
                self.entry_order = None

        elif order.status in (
            order.Canceled,
            order.Margin,
            order.Rejected,
        ):
            if order is self.order:
                self.order = None

            if order is self.entry_order:
                self.entry_order = None

    def next(self):
        if self.order:
            return

        close = self.data.close[0]
        volatility_ok = (self.atr_sma30[0] / close) <= 0.05
        liquidity_ok = (
            self.volume_sma20[0] * close
        ) > self.p.liquidity_threshold

        if self.position:
            if self.position.size > 0:
                deviation = self.sma50[0] - close
                if self.cross[0] > 0 or deviation >= 1.1 * self.d0:
                    self.order = self.close()
            else:
                deviation = close - self.sma50[0]
                if self.cross[0] < 0 or deviation >= 1.1 * self.d0:
                    self.order = self.close()
            return

        qty = math.floor(
            (self.p.risk_pct * self.broker.getvalue())
            / (self.p.atr_multiple * self.atr14[0])
        )

        if qty <= 0:
            return

        if (
            close > self.sma200[0]
            and close <= self.sma50[0] * 0.95
            and volatility_ok
            and liquidity_ok
        ):
            self.order = self.buy(
                size=qty,
                exectype=bt.Order.Limit,
                price=close * 1.005,
            )
            self.entry_order = self.order

        elif (
            close < self.sma200[0]
            and close >= self.sma50[0] * 1.05
            and volatility_ok
            and liquidity_ok
        ):
            self.order = self.sell(
                size=qty,
                exectype=bt.Order.Limit,
                price=close * 0.995,
            )
            self.entry_order = self.order
