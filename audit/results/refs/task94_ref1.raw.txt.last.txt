import backtrader as bt


class NXXTMultiMovingAverageTrendFollowingStrategy(bt.Strategy):
    params = (
        ("risk_per_trade", 0.02),
        ("take_profit", 0.10),
    )

    def __init__(self):
        self.sma30 = bt.indicators.SMA(self.data.close, period=30)
        self.sma60 = bt.indicators.SMA(self.data.close, period=60)
        self.sma120 = bt.indicators.SMA(self.data.close, period=120)

        self.atr14 = bt.indicators.ATR(self.data, period=14)
        self.atr30_average = bt.indicators.SMA(self.atr14, period=30)

        self.order = None
        self.entry_price = None
        self.consolidation_closes = []

    def notify_order(self, order):
        if order.status == order.Completed:
            if order.isbuy():
                self.entry_price = order.executed.price
            elif order.issell():
                self.entry_price = None
            self.order = None
        elif order.status in (order.Canceled, order.Margin, order.Rejected):
            self.order = None

    def next(self):
        if self.order:
            return

        close = self.data.close[0]

        if self.position:
            if (self.entry_price is not None and
                    self.data.high[0] >= self.entry_price * (1.0 + self.p.take_profit)):
                self.order = self.close()
            elif close < self.sma30[0]:
                self.order = self.close()
            return

        consolidation_high = max(self.consolidation_closes) if self.consolidation_closes else None
        consolidation_ready = len(self.consolidation_closes) >= 5

        above_moving_averages = (
            close > self.sma30[0] and
            close > self.sma60[0] and
            close > self.sma120[0]
        )
        moving_averages_rising = (
            self.sma30[0] > self.sma30[-1] and
            self.sma60[0] > self.sma60[-1] and
            self.sma120[0] > self.sma120[-1]
        )
        high_volatility = self.atr14[0] > self.atr30_average[0] * 1.5

        if (consolidation_ready and
                close > consolidation_high and
                above_moving_averages and
                moving_averages_rising and
                not high_volatility):
            stop_loss_price = self.sma30[0]
            risk_per_share = close - stop_loss_price

            if risk_per_share > 0:
                size = (self.broker.getvalue() * self.p.risk_per_trade) / risk_per_share
                self.order = self.buy(size=size, exectype=bt.Order.Close)

        self.consolidation_closes.append(close)
        while (len(self.consolidation_closes) > 1 and
               (max(self.consolidation_closes) - min(self.consolidation_closes)) /
               min(self.consolidation_closes) > 0.05):
            self.consolidation_closes.pop(0)
