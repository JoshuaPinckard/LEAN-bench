import backtrader as bt


class WeeklyORBStrategy(bt.Strategy):
    def __init__(self):
        self.ema21 = bt.indicators.EMA(self.data.close, period=21)
        self.bb = bt.indicators.BollingerBands(
            self.data.close, period=20, devfactor=2.0
        )
        self.stoch = bt.indicators.Stochastic(
            self.data, period=14, period_dfast=3, period_dslow=3
        )
        self.atr = bt.indicators.ATR(self.data, period=14)
        self.stoch_cross = bt.indicators.CrossOver(
            self.stoch.percK, self.stoch.percD
        )
        self.vol_sma = bt.indicators.SMA(self.data.volume, period=20)

        self.order = None
        self.current_week = None
        self.wor_high = None
        self.wor_low = None
        self.wor_range = None
        self.breakout_seen = False

        self.entry_price = None
        self.tp_price = None
        self.sl_price = None
        self.entry_wor_low = None

        self.pending_wor_range = None
        self.pending_wor_low = None
        self.pending_atr = None

    def notify_order(self, order):
        if order.status in [
            order.Completed,
            order.Canceled,
            order.Margin,
            order.Rejected,
        ]:
            if order.status == order.Completed and order.isbuy():
                self.entry_price = order.executed.price
                self.tp_price = self.entry_price + 1.5 * self.pending_wor_range
                self.sl_price = (
                    self.pending_wor_low - 0.25 * self.pending_atr
                )
                self.entry_wor_low = self.pending_wor_low
            self.order = None

    def next(self):
        dt = self.data.datetime.date(0)
        iso_year, iso_week, _ = dt.isocalendar()
        current_week_key = (iso_year, iso_week)

        if current_week_key != self.current_week:
            self.current_week = current_week_key
            self.wor_high = self.data.high[0]
            self.wor_low = self.data.low[0]
            self.wor_range = self.wor_high - self.wor_low
            self.breakout_seen = False
        else:
            if self.data.close[0] > self.wor_high:
                self.breakout_seen = True

        if self.order is not None:
            return

        if self.position:
            if self.tp_price is not None and self.data.high[0] >= self.tp_price:
                self.order = self.close()
            elif (
                self.sl_price is not None and self.data.low[0] <= self.sl_price
            ):
                self.order = self.close()
            elif (
                self.entry_wor_low is not None
                and self.data.close[0] < self.entry_wor_low
            ):
                self.order = self.close()
        else:
            if not self.breakout_seen or self.wor_range <= 0:
                return

            gz_low = self.wor_high - 0.618 * self.wor_range
            gz_high = self.wor_high - 0.382 * self.wor_range

            pullback_ok = (gz_low <= self.data.low[0] <= gz_high) or (
                gz_low <= self.data.close[0] <= gz_high
            )
            trend_ok = (self.data.close[0] > self.ema21[0]) and (
                self.data.close[0] > self.bb.mid[0]
            )
            reversal_ok = self.stoch_cross[0] > 0

            if pullback_ok and trend_ok and reversal_ok:
                equity = self.broker.getvalue()
                size = (0.01 * equity) / self.data.close[0]
                if size > 0:
                    self.pending_wor_range = self.wor_range
                    self.pending_wor_low = self.wor_low
                    self.pending_atr = self.atr[0]
                    self.order = self.buy(size=size)
