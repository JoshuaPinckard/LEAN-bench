```python
import backtrader as bt

class TradingStrategy(bt.Strategy):
    params = (
        ('rsi_period', 14),
        ('atr_period', 14),
        ('atr_mult', 2.0),
        ('trailing_stop', False),
    )
    
    def __init__(self):
        self.rsi = bt.indicators.RSI(self.data.close, period=self.params.rsi_period)
        self.atr = bt.indicators.ATR(self.data, period=self.params.atr_period)
        self.order = None
        self.entry_price = None
        self.stop_price = None
        
    def next(self):
        if self.order:
            return
        
        if not self.position:
            if (self.rsi[-1] <= 30 and self.rsi[0] > 30 
                and self.data.close[0] > self.data.close[-1]):
                self.entry_price = self.data.close[0]
                self.stop_price = self.entry_price - self.params.atr_mult * self.atr[0]
                self.order = self.buy()
                
            elif (self.rsi[-1] >= 70 and self.rsi[0] < 70 
                  and self.data.close[0] < self.data.close[-1]):
                self.entry_price = self.data.close[0]
                self.stop_price = self.entry_price + self.params.atr_mult * self.atr[0]
                self.order = self.sell()
        else:
            if self.position.size > 0:
                if self.data.low[0] <= self.stop_price:
                    self.order = self.sell()
                elif (self.rsi[-1] >= 70 and self.rsi[0] < 70 
                      and self.data.close[0] < self.data.close[-1]):
                    self.order = self.sell()
                
                if self.params.trailing_stop:
                    new_stop = self.entry_price - self.params.atr_mult * self.atr[0]
                    if new_stop > self.stop_price:
                        self.stop_price = new_stop
                        
            elif self.position.size < 0:
                if self.data.high[0] >= self.stop_price:
                    self.order = self.buy()
                elif (self.rsi[-1] <= 30 and self.rsi[0] > 30 
                      and self.data.close[0] > self.data.close[-1]):
                    self.order = self.buy()
                
                if self.params.trailing_stop:
                    new_stop = self.entry_price + self.params.atr_mult * self.atr[0]
                    if new_stop < self.stop_price:
                        self.stop_price = new_stop
    
    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None
```
