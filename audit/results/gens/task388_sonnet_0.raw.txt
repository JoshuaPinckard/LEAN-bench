import backtrader as bt

class TradingStrategy(bt.Strategy):
    params = (
        ('willr_period', 14),
        ('mom_period', 14),
        ('atr_period', 14),
        ('atr_mult', 2.0),
        ('trailing_stop', False),
    )

    def __init__(self):
        self.stoch = bt.indicators.Stochastic(
            self.data,
            period=self.params.willr_period,
            period_dfast=1,
        )
        self.atr = bt.indicators.ATR(self.data, period=self.params.atr_period)
        self.order = None
        self.stop_price = None
        self.trade_direction = 0

    def next(self):
        if self.order:
            return

        willr_curr = self.stoch.percK[0] - 100.0
        willr_prev = self.stoch.percK[-1] - 100.0
        momentum = self.data.close[0] - self.data.close[-self.params.mom_period]

        long_signal = willr_prev <= -80.0 and willr_curr > -80.0 and momentum > 0.0
        short_signal = willr_prev >= -20.0 and willr_curr < -20.0 and momentum < 0.0

        if not self.position:
            if long_signal:
                self.order = self.buy()
                self.trade_direction = 1
            elif short_signal:
                self.order = self.sell()
                self.trade_direction = -1
        else:
            if self.trade_direction == 1:
                if self.params.trailing_stop and self.stop_price is not None:
                    new_stop = self.data.close[0] - self.params.atr_mult * self.atr[0]
                    if new_stop > self.stop_price:
                        self.stop_price = new_stop
                if self.stop_price is not None:
                    if self.data.close[0] <= self.stop_price or short_signal:
                        self.order = self.close()
                        self.trade_direction = 0
            elif self.trade_direction == -1:
                if self.params.trailing_stop and self.stop_price is not None:
                    new_stop = self.data.close[0] + self.params.atr_mult * self.atr[0]
                    if new_stop < self.stop_price:
                        self.stop_price = new_stop
                if self.stop_price is not None:
                    if self.data.close[0] >= self.stop_price or long_signal:
                        self.order = self.close()
                        self.trade_direction = 0

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            if order.status == order.Completed:
                ep = order.executed.price
                if order.isbuy() and self.trade_direction == 1:
                    self.stop_price = ep - self.params.atr_mult * self.atr[0]
                elif order.issell() and self.trade_direction == -1:
                    self.stop_price = ep + self.params.atr_mult * self.atr[0]
                else:
                    self.stop_price = None
            self.order = None
