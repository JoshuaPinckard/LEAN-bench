import backtrader as bt

class TradingStrategy(bt.Strategy):
    params = (
        ('fast_period', 13),
        ('slow_period', 40),
        ('rsi_period', 14),
        ('bars_per_period', 6),
        ('hold_periods', 6),
        ('stop_loss', 0.015),
        ('take_profit', 0.030),
        ('rsi_threshold', 70.0),
    )

    def __init__(self):
        self.fast_sma = bt.indicators.SMA(self.data.close, period=self.params.fast_period)
        self.slow_sma = bt.indicators.SMA(self.data.close, period=self.params.slow_period)
        self.crossover = bt.indicators.CrossOver(self.fast_sma, self.slow_sma)
        self.rsi = bt.indicators.RSI(self.data.close, period=self.params.rsi_period)
        self.macd = bt.indicators.MACD(self.data.close)
        self.order = None
        self.entry_price = None
        self.entry_bar = None

    def next(self):
        if self.order:
            return

        bar_num = len(self)
        bpp = self.params.bars_per_period
        period_bar = bar_num % bpp

        if not self.position:
            if period_bar == 0 and bar_num > bpp:
                ma_crosses = 0
                rsi_sum = 0.0

                for i in range(1, bpp + 1):
                    if self.crossover[-i] > 0:
                        ma_crosses += 1
                    rsi_sum += self.rsi[-i]

                avg_rsi = rsi_sum / bpp

                if ma_crosses == 1 and avg_rsi > self.params.rsi_threshold:
                    self.order = self.buy()
        else:
            if self.entry_price is not None:
                current_price = self.data.close[0]

                if current_price <= self.entry_price * (1.0 - self.params.stop_loss):
                    self.order = self.sell()
                    return

                if current_price >= self.entry_price * (1.0 + self.params.take_profit):
                    self.order = self.sell()
                    return

                hold_bars = self.params.hold_periods * bpp
                if self.entry_bar is not None and bar_num >= self.entry_bar + hold_bars:
                    self.order = self.sell()

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            if order.status == order.Completed:
                if order.isbuy():
                    self.entry_price = order.executed.price
                    self.entry_bar = len(self)
                else:
                    self.entry_price = None
                    self.entry_bar = None
            self.order = None
