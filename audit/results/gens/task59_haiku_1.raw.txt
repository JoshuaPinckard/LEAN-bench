import backtrader as bt

class TradingStrategy(bt.Strategy):
    params = (
        ('fast_period', 13),
        ('slow_period', 40),
        ('rsi_period', 14),
        ('stop_loss_pct', 0.015),
        ('take_profit_pct', 0.03),
    )
    
    def __init__(self):
        self.sma_fast = bt.indicators.SMA(self.data.close, period=self.params.fast_period)
        self.sma_slow = bt.indicators.SMA(self.data.close, period=self.params.slow_period)
        self.rsi = bt.indicators.RSI(self.data.close, period=self.params.rsi_period)
        self.crossover = bt.indicators.CrossOver(self.sma_fast, self.sma_slow)
        
        self.order = None
        self.entry_price = 0
        self.bars_held = 0
        self.bar_count = 0
        self.crossovers_in_window = 0
        self.rsi_sum = 0
        self.rsi_count = 0
    
    def next(self):
        if self.order:
            return
        
        self.bar_count += 1
        
        if self.crossover > 0:
            self.crossovers_in_window += 1
        
        self.rsi_sum += self.rsi
        self.rsi_count += 1
        
        if self.bar_count % 6 == 0:
            avg_rsi = self.rsi_sum / self.rsi_count
            
            if not self.position:
                if self.crossovers_in_window == 1 and avg_rsi > 70:
                    self.order = self.buy()
                    self.entry_price = self.data.close
                    self.bars_held = 0
            
            self.crossovers_in_window = 0
            self.rsi_sum = 0
            self.rsi_count = 0
        
        if self.position:
            self.bars_held += 1
            current_price = self.data.close
            
            if current_price < self.entry_price * (1 - self.params.stop_loss_pct):
                self.order = self.sell()
            elif current_price > self.entry_price * (1 + self.params.take_profit_pct):
                self.order = self.sell()
            elif self.bars_held >= 36:
                self.order = self.sell()
    
    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None
