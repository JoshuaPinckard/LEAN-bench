import backtrader as bt

class TradingStrategy(bt.Strategy):
    params = (
        ('fast_period', 13),
        ('slow_period', 40),
        ('rsi_period', 14),
        ('hold_periods', 6),
        ('stop_loss_pct', 0.015),
        ('take_profit_pct', 0.030),
    )
    
    def __init__(self):
        self.fast_sma = bt.indicators.SMA(self.data.close, period=self.params.fast_period)
        self.slow_sma = bt.indicators.SMA(self.data.close, period=self.params.slow_period)
        self.crossover = bt.indicators.CrossOver(self.fast_sma, self.slow_sma)
        self.rsi = bt.indicators.RSI(self.data.close, period=self.params.rsi_period)
        self.order = None
        self.entry_price = None
        self.entry_bar = None
        
    def next(self):
        if self.order:
            return
            
        if not self.position:
            if self.crossover > 0 and self.rsi > 70:
                self.entry_price = self.data.close[0]
                self.entry_bar = len(self)
                self.order = self.buy()
        else:
            bars_held = len(self) - self.entry_bar
            
            if bars_held >= self.params.hold_periods:
                self.order = self.sell()
            elif self.data.close[0] <= self.entry_price * (1 - self.params.stop_loss_pct):
                self.order = self.sell()
            elif self.data.close[0] >= self.entry_price * (1 + self.params.take_profit_pct):
                self.order = self.sell()
    
    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None
