import backtrader as bt

class TradingStrategy(bt.Strategy):
    params = (
        ('fast_period', 9),
        ('mid_period', 26),
        ('slow_period', 52),
    )
    
    def __init__(self):
        self.sma_fast = bt.indicators.SMA(self.data.close, period=self.params.fast_period)
        self.sma_mid = bt.indicators.SMA(self.data.close, period=self.params.mid_period)
        self.sma_slow = bt.indicators.SMA(self.data.close, period=self.params.slow_period)
        self.crossover = bt.indicators.CrossOver(self.sma_fast, self.sma_mid)
        self.order = None
        
    def next(self):
        if self.order:
            return
            
        if not self.position:
            if self.crossover > 0:
                if (self.data.close[0] > self.sma_mid[0] and 
                    self.data.close[0] > self.sma_slow[0] and 
                    self.sma_mid[0] > self.sma_slow[0]):
                    self.order = self.buy()
            
            elif self.crossover < 0:
                if (self.data.close[0] < self.sma_mid[0] and 
                    self.data.close[0] < self.sma_slow[0] and 
                    self.sma_mid[0] < self.sma_slow[0]):
                    self.order = self.sell()
        else:
            if self.position.size > 0:
                lower_sma = min(self.sma_mid[0], self.sma_slow[0])
                if self.crossover < 0 or self.data.close[0] < lower_sma:
                    self.order = self.sell()
            
            elif self.position.size < 0:
                higher_sma = max(self.sma_mid[0], self.sma_slow[0])
                if self.crossover > 0 or self.data.close[0] > higher_sma:
                    self.order = self.buy()
                
    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None
