import backtrader as bt

class TradingStrategy(bt.Strategy):
    params = (
        ('fast_period', 9),
        ('mid_period', 26),
        ('slow_period', 52),
    )

    def __init__(self):
        self.sma9 = bt.indicators.SMA(self.data.close, period=self.params.fast_period)
        self.sma26 = bt.indicators.SMA(self.data.close, period=self.params.mid_period)
        self.sma52 = bt.indicators.SMA(self.data.close, period=self.params.slow_period)
        self.crossover = bt.indicators.CrossOver(self.sma9, self.sma26)
        self.order = None

    def next(self):
        if self.order:
            return

        close = self.data.close[0]
        sma26 = self.sma26[0]
        sma52 = self.sma52[0]

        if not self.position:
            if (self.crossover > 0 and
                    close > sma26 and
                    close > sma52 and
                    sma26 > sma52):
                size = int(self.broker.getcash() / close)
                if size > 0:
                    self.order = self.buy(size=size)

            elif (self.crossover < 0 and
                    close < sma26 and
                    close < sma52 and
                    sma26 < sma52):
                size = int(self.broker.getcash() / close)
                if size > 0:
                    self.order = self.sell(size=size)

        else:
            if self.position.size > 0:
                lower_cloud = min(sma26, sma52)
                if self.crossover < 0 or close < lower_cloud:
                    self.order = self.sell()

            elif self.position.size < 0:
                upper_cloud = max(sma26, sma52)
                if self.crossover > 0 or close > upper_cloud:
                    self.order = self.buy()

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None
