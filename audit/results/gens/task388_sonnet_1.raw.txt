import backtrader as bt


class TradingStrategy(bt.Strategy):
    params = (
        ('willr_period', 14),
        ('rsi_period', 14),
        ('atr_period', 14),
        ('atr_mult', 2.0),
        ('trailing_stop', False),
    )

    def __init__(self):
        stoch = bt.indicators.Stochastic(
            self.data,
            period=self.params.willr_period,
            period_dfast=1,
            period_dslow=1,
            safediv=True,
        )
        self.willr = stoch.percK - 100.0
        self.rsi = bt.indicators.RSI(self.data.close, period=self.params.rsi_period)
        self.atr = bt.indicators.ATR(self.data, period=self.params.atr_period)
        self.order = None
        self.stop_price = None
        self.trade_dir = 0

    def next(self):
        if self.order:
            return

        willr_curr = self.willr[0]
        willr_prev = self.willr[-1]
        long_signal = willr_prev <= -80.0 and willr_curr > -80.0 and self.rsi[0] > 50.0
        short_signal = willr_prev >= -20.0 and willr_curr < -20.0 and self.rsi[0] < 50.0

        if self.position:
            if self.params.trailing_stop and self.stop_price is not None:
                if self.trade_dir == 1:
                    candidate = self.data.close[0] - self.params.atr_mult * self.atr[0]
                    if candidate > self.stop_price:
                        self.stop_price = candidate
                elif self.trade_dir == -1:
                    candidate = self.data.close[0] + self.params.atr_mult * self.atr[0]
                    if candidate < self.stop_price:
                        self.stop_price = candidate

            if self.stop_price is not None:
                if self.trade_dir == 1 and self.data.low[0] <= self.stop_price:
                    self.order = self.sell()
                    self.trade_dir = 0
                    return
                if self.trade_dir == -1 and self.data.high[0] >= self.stop_price:
                    self.order = self.buy()
                    self.trade_dir = 0
                    return

            if self.trade_dir == 1 and short_signal:
                self.order = self.sell()
                self.trade_dir = 0
            elif self.trade_dir == -1 and long_signal:
                self.order = self.buy()
                self.trade_dir = 0
        else:
            if long_signal:
                self.trade_dir = 1
                self.stop_price = self.data.close[0] - self.params.atr_mult * self.atr[0]
                self.order = self.buy()
            elif short_signal:
                self.trade_dir = -1
                self.stop_price = self.data.close[0] + self.params.atr_mult * self.atr[0]
                self.order = self.sell()

    def notify_order(self, order):
        if order.status == order.Completed:
            fill = order.executed.price
            if order.isbuy() and self.trade_dir == 1:
                self.stop_price = fill - self.params.atr_mult * self.atr[0]
            elif order.issell() and self.trade_dir == -1:
                self.stop_price = fill + self.params.atr_mult * self.atr[0]
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None
