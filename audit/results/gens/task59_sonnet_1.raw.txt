import backtrader as bt

class TradingStrategy(bt.Strategy):
    params = (
        ('fast_period', 13),
        ('slow_period', 40),
        ('rsi_period', 14),
        ('agg_bars', 6),
        ('hold_bars', 36),
        ('stop_loss_pct', 0.015),
        ('take_profit_pct', 0.030),
        ('rsi_threshold', 70.0),
    )

    def __init__(self):
        self.fast_sma = bt.indicators.SMA(self.data.close, period=self.params.fast_period)
        self.slow_sma = bt.indicators.SMA(self.data.close, period=self.params.slow_period)
        self.crossover = bt.indicators.CrossOver(self.fast_sma, self.slow_sma)
        self.rsi = bt.indicators.RSI(self.data.close, period=self.params.rsi_period)
        self.macd = bt.indicators.MACD(self.data.close)
        self.order = None
        self.entry_price = 0.0
        self.bars_held = 0

    def next(self):
        if self.order:
            return

        if self.position:
            self.bars_held += 1
            price = self.data.close[0]

            if price <= self.entry_price * (1.0 - self.params.stop_loss_pct):
                self.order = self.sell()
                return

            if price >= self.entry_price * (1.0 + self.params.take_profit_pct):
                self.order = self.sell()
                return

            if self.bars_held >= self.params.hold_bars:
                self.order = self.sell()

        else:
            if len(self) % self.params.agg_bars != 0:
                return

            agg = self.params.agg_bars

            ma_crosses = 0
            for i in range(agg):
                if self.crossover[-i] > 0:
                    ma_crosses += 1

            rsi_sum = 0.0
            for i in range(agg):
                rsi_sum += self.rsi[-i]
            avg_rsi = rsi_sum / agg

            if ma_crosses == 1 and avg_rsi > self.params.rsi_threshold:
                cash = self.broker.getcash()
                price = self.data.close[0]
                size = int(cash / price)
                if size > 0:
                    self.order = self.buy(size=size)

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            if order.status == order.Completed and order.isbuy():
                self.entry_price = order.executed.price
                self.bars_held = 0
            self.order = None
