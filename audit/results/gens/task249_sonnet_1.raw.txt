import backtrader as bt

class TradingStrategy(bt.Strategy):
    params = (
        ('tenkan_period', 9),
        ('kijun_period', 26),
        ('senkou_period', 52),
    )

    def __init__(self):
        self.sma_tenkan = bt.indicators.SMA(self.data.close, period=self.params.tenkan_period)
        self.sma_kijun = bt.indicators.SMA(self.data.close, period=self.params.kijun_period)
        self.sma_senkou = bt.indicators.SMA(self.data.close, period=self.params.senkou_period)
        self.crossover = bt.indicators.CrossOver(self.sma_tenkan, self.sma_kijun)
        self.order = None

    def next(self):
        if self.order:
            return

        close = self.data.close[0]
        kijun = self.sma_kijun[0]
        senkou = self.sma_senkou[0]

        if not self.position:
            if (self.crossover[0] > 0 and
                    close > kijun and
                    close > senkou and
                    kijun > senkou):
                size = int(self.broker.getcash() / close)
                if size > 0:
                    self.order = self.buy(size=size)

            elif (self.crossover[0] < 0 and
                    close < kijun and
                    close < senkou and
                    kijun < senkou):
                size = int(self.broker.getcash() / close)
                if size > 0:
                    self.order = self.sell(size=size)

        elif self.position.size > 0:
            cloud_bottom = min(kijun, senkou)
            if self.crossover[0] < 0 or close < cloud_bottom:
                self.order = self.close()

        elif self.position.size < 0:
            cloud_top = max(kijun, senkou)
            if self.crossover[0] > 0 or close > cloud_top:
                self.order = self.close()

    def notify_order(self, order):
        if order.status in [order.Completed, order.Canceled, order.Margin, order.Rejected]:
            self.order = None
